"""
Propose a low‑score step function.

The algorithm consists of three stages:
1. **Coarse search over several problem sizes** – we try n = 4000,
   5000, 6000 and 8000 (adjusted to the remaining time) and run a
   quick L‑BFGS polish from a diverse pool of initial families
   (spectral, triangular, Hann, Gaussian, Arcsine, inverse‑sqrt,

2. **Intensive local optimisation** on the chosen length:
   - Long L‑BFGS polishing with a fine λ‑schedule.
   - Exponentiated‑gradient descent on the exact max‑convolution.

   - A final, more aggressive L‑BFGS refinement (λ down to 1e‑12).
   - A greedy hill‑climb for polishing.

3. **Evaluation** – the resulting vector is passed to `evaluate_sequence`
   (the official scorer).  All operations respect the non‑negativity
   and sum‑normalisation constraints required by the objective.

The code is written with only top‑level functions (no closures) and
uses NumPy / SciPy for FFTs and optimisation.  It adapts its time
budget so that it always finishes before the 1000 s deadline.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay (heavier on the left)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int, power: float = 0.5) -> np.ndarray:
    """
    Discrete analogue of the Beta(1‑p, 1‑p) density:
        a_i ∝ 1 / (x_i (1 - x_i))^p,  x_i = (i+0.5)/n
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))  # p=0.5 is classic arcsine
    if power != 0.5:
        a = 1.0 / (x * (1.0 - x)) ** power
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    L‑BFGS polishing across a decreasing λ schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent using the top‑fraction of the convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # threshold mask
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient via reverse‑a & mask
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry of the
    self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # build gradient for the max entry
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Candidate generation for a given length
# ----------------------------------------------------------------------


def build_candidate_pool(n: int, rng: np.random.Generator):
    """Create a diverse list of initial probability vectors."""
    cand = []

    # deterministic families
    cand.append(spectral_init(n))
    cand.append(triangular_init(n))
    cand.append(hann_init(n))
    cand.append(blackman_init(n))
    cand.append(gaussian_init(n, sigma_factor=rng.uniform(0.15, 0.35)))
    cand.append(inv_sqrt_init(n))
    cand.append(arcsine_init(n))

    # randomised families
    for _ in range(3):
        cand.append(random_initial(n, rng))
    for _ in range(2):
        cand.append(pow_decay_init(n, rng))
    for _ in range(2):
        beta = rng.uniform(4.0, 12.0)
        cand.append(kaiser_init(n, beta))

    # parameterised arcsine (vary exponent)
    for _ in range(2):
        p = rng.uniform(0.3, 0.7)  # power in 0.3..0.7
        cand.append(arcsine_init(n, power=p))

    return cand

# Coarse search over several lengths
# ----------------------------------------------------------------------


def quick_optimize_for_n(n: int, rng: np.random.Generator,
                         deadline: float):
    """
    Perform a fast L‑BFGS polish for a given n.
    Returns the best vector and its fast_score.
    """
    fft_len = next_pow2(2 * n - 1)

    # Build pool and take best raw candidate
    candidates = build_candidate_pool(n, rng)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # Light L‑BFGS polish with a short λ‑schedule
    lam_schedule = [0.5, 0.2, 0.07, 0.02, 0.008, 0.003, 1e-3, 5e-4, 1e-4]
    maxiter_per_lam = 800

    # A few random blends to avoid getting stuck
    while time.time() < deadline:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           deadline,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[quick‑opt] n={n}, new best fast_score={best_score:.8f}")
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the 1000 s time limit and returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 0 – coarse scan over a few problem sizes
    # ------------------------------------------------------------------
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5  # safety margin

    # Time allocated to the coarse scan (≈10 % of total)
    scan_deadline = start_time + 0.10 * budget_s
    length_options = [4000, 5000, 6000, 8000]
    best_overall_a = None
    best_overall_score = math.inf
    best_n = None

    for n in length_options:
        if time.time() >= scan_deadline:
            break
        print(f"[scan] starting quick optimisation for n={n}")
        a_opt, sc_opt = quick_optimize_for_n(n, rng, scan_deadline)
        if sc_opt < best_overall_score:
            best_overall_score = sc_opt
            best_overall_a = a_opt.copy()
            best_n = n
            print(f"[scan] new global best: n={n}, fast_score={sc_opt:.8f}")

    # If we could not finish any scan (e.g. tiny budget), fall back to n=5000
    if best_overall_a is None:
        best_n = 5000
        best_overall_a = spectral_init(best_n)

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing on the chosen length
    # ------------------------------------------------------------------
    n = best_n
    fft_len = next_pow2(2 * n - 1)
    print(f"[main] chosen n={n}, starting full optimisation")

    # Build a richer initial pool (including previous best)
    pool = build_candidate_pool(n, rng)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, prev)
            interp = np.clip(interp, 0.0, None)
            s = interp.sum()
            if s > 0:
                interp /= s
            else:
                interp = np.full(n, 1.0 / n)
            pool.append(interp)

    # Include the best from scan
    pool.append(best_overall_a)

    # Select the best raw candidate
    best_a = pool[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in pool[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score={best_score:.8f}")

    # ------------------- L‑BFGS phase -------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11,
        5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4  # blend factor
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------- Exponentiated gradient -------------------
    phase2_end = start_time + 0.65 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.02)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------- Exact sub‑gradient -------------------
    phase3_end = start_time + 0.80 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------- Final L‑BFGS refinement -------------------
    phase4_end = start_time + 0.90 * budget_s
    if time.time() < phase4_end:
        # Very fine λ‑schedule, more iterations
        fine_lam = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8,
                    2e-8, 1e-8, 5e-9, 2e-9, 1e-9,
                    5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11,
                    5e-12, 2e-12, 1e-12]
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                           fine_lam,
                                           maxiter_per_lam=2500,
                                           deadline=phase4_end,
                                           fft_len=fft_len,
                                           n=n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------- Greedy hill‑climb -------------------
    final_deadline = overall_deadline
    if time.time() < final_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        final_deadline,
                                        rng,
                                        max_iter=120000,
                                        delta_frac=0.06)
        print(f"[hill‑climb] final fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Evaluate with the official scorer and return the sequence.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
