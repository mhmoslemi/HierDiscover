"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1. Build a diverse pool of initial distributions (uniform, arcsine,
   random, power‑law, triangular, previous best) and keep the one with
   the lowest cheap score `2·n·max(conv)`.
2. Run a multi‑phase *smooth exponentiated‑gradient* optimisation:
   - The objective is a smoothed maximum of the self‑convolution,
     obtained by a soft‑max with a temperature `α` that is increased
     from 1 to ~50 during the run.
   - The gradient of the soft‑max is computed by a single FFT‑based
     cross‑correlation.  An exponentiated‑gradient update
     `a_i ← a_i·exp(−η·g_i)` is performed and the vector is projected
     back onto the probability simplex.
   - The learning‑rate `η` is decreased along the same schedule.
3. Refine the result with a projected sub‑gradient descent (as in the
   reference implementation) to fine‑tune the true max‑convolution.
4. Perform a final greedy hill‑climb (exact convolution updates) for
   additional local improvement.
5. Return the best sequence found; the harness will evaluate it with
   `evaluate_sequence`.

All steps respect the 1000 s budget; the algorithm works on the
simplex, so the final score equals `2·n·max(conv)`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (used after the smooth phase)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

    if remaining > 2.0:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
