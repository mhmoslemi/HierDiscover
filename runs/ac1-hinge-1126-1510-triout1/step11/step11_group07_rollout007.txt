def adam_optimize(a_start, deadline, n, fft_len, rng, beta_min=1.0, beta_max=2000.0, step0=0.1, beta_increase_iters=20000):
   b = np.log(a_start + 1e-12)  # initial logits
   m = np.zeros_like(b)
   v = np.zeros_like(b)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   it = 0
   best_a = a_start.copy()
   best_score = fast_score(best_a, fft_len)
   while time.time() < deadline:
        # compute a
        b_max = b.max()
        a = np.exp(b - b_max)  # subtract max for numeric stability
        a /= a.sum()
        # FFTs
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        # compute beta
        # use linear schedule based on fraction progress
        fraction = (it+1) / max(1, (deadline - start_time) * 200) ??? Actually we need bound total it.
        # But we don't know total iterations; we can compute based on elapsed proportion of deadline:
        elapsed = time.time() - start_time_global
        progress = elapsed / (deadline - start_time_global)
        beta = beta_min * ((beta_max / beta_min) ** progress)   # exponential increase
        # compute weights
        max_conv = conv.max()
        # subtract max to avoid overflow
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        w /= w_sum
        # gradient using FFT
        fm = np.fft.fft(w, fft_len)
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_vec = np.real(np.fft.ifft(fa_rev * fm))[:n]
        grad_a = 4.0 * n * grad_vec  # scaling
        # gradient wrt b
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = step0 / (np.sqrt(v_hat) + eps)
        b = b - step * m_hat
        # maybe apply clipping to b to avoid overflow large values: maybe set max/min to ±50
        b = np.clip(b, -50, 50)
        # evaluate fast_score
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
   return best_a, best_score

"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1. Build a few diverse initial probability vectors (uniform, arcsine‑shaped,
   random draws, and the previous best if available).  Keep the best by the
   cheap fast_score (2·n·max‐convolution).

2. Refine the chosen vector with a *softmax‑Adam* optimizer.
   * a is represented as a softmax of unconstrained logits b
   * The objective is a smoothed version of the max‑convolution:
         f_β(a) = 2·n/β·log Σ_k exp(β·conv_k(a))
     where β is increased slowly from a small value (≈10) to a large value
     (≈2000).  This yields a smooth gradient that gradually focuses on the
     true maximum.
   * Gradient of the smoothed objective is obtained analytically in the
     Fourier domain (single FFT per iteration).  Adam’s adaptive moments
     accelerate convergence.
   * All updates stay on the simplex automatically via the softmax.

3. Perform a final *greedy hill‑climb* that moves a tiny fraction of mass
   from high‑gradient indices to low‑gradient ones, accepting only improving
   moves.  The convolution is updated incrementally, so the step is cheap.

4. Return the best sequence found before the deadline.  The harness will
   evaluate it with the supplied `evaluate_sequence` routine.

The whole search respects the 1000 s time budget and works with up to
2 CPU cores (NumPy’s FFT uses multithreading automatically).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) using FFT (n = len(a))."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Softmax‑Adam optimiser (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  start_time: float,
                  deadline: float,
                  fft_len: int,
                  beta_min: float = 10.0,
                  beta_max: float = 2000.0,
                  step0: float = 0.05,
                  rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Adam on the softmax‑smoothed max‑convolution.
    a_start must sum to 1.
    Returns (best_a, best_score) where best_score = 2·n·max(conv).
    """
    n = a_start.shape[0]
    # initialise logits b
    eps = 1e-12
    b = np.log(a_start + eps)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    # pre‑allocate buffers to avoid re‑allocation inside loop
    conv_len = 2 * n - 1

    while time.time() < deadline:
        # ---- softmax ----
        b_max = b.max()
        a = np.exp(b - b_max)
        a_sum = a.sum()
        a /= a_sum

        # ---- convolution (FFT) ----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:conv_len]

        # ---- beta schedule (linear in elapsed fraction) ----
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = min(1.0, max(0.0, elapsed / total))
        beta = beta_min + (beta_max - beta_min) * progress

        # ---- softmax weights over convolution entries ----
        max_conv = conv.max()
        # subtraction of max_conv avoids overflow
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        w /= w_sum

        # ---- gradient of smoothed objective ----
        # grad_a = 4·n·ifft(FFT(a_rev) * FFT(w))
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        fm = np.fft.fft(w, fft_len)
        grad_part = np.real(np.fft.ifft(fa_rev * fm))[:n]          # shape (n,)
        grad_a = 4.0 * n * grad_part                               # d f / d a

        # ---- gradient w.r.t. logits b (softmax Jacobian) ----
        dot = np.dot(a, grad_a)                                    # scalar
        grad_b = a * (grad_a - dot)                                # shape (n,)

        # ---- Adam update on b ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + 1e-8)
        b -= step * m_hat

        # keep logits bounded (prevents overflow in exp)
        b = np.clip(b, -50.0, 50.0)

        # ---- keep track of the best solution ----
        cur_score = float(2.0 * n * max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a power of two for efficient FFTs.
    # 4096 provides a good trade‑off between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1. Build a small collection of diverse starting points.
    # ------------------------------------------------------------------
    candidates = []

    # Uniform distribution.
    candidates.append(np.full(n, 1.0 / n))

    # Deterministic arcsine‑shaped distribution.
    candidates.append(arcsine_initial(n))

    # Several random draws.
    for _ in range(4):
        candidates.append(random_initial(n, rng))

    # Previous best (if supplied by the harness).
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            candidates.append(a_old)

    # ------------------------------------------------------------------
    # Keep the best primitive according to the cheap fast_score.
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a tiny safety margin

    # ------------------------------------------------------------------
    # 2. Softmax‑Adam optimisation (≈ 70 % of remaining budget).
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.70 * (deadline - time.time())
    best_a, best_score = adam_optimize(
        best_a, start_time, phase1_deadline, fft_len,
        beta_min=10.0,
        beta_max=2000.0,
        step0=0.05,
        rng=rng
    )
    print(f"[adam] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final greedy hill‑climb (use the remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Official evaluation (the harness will also call it).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
