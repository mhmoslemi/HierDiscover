"""
Propose a low-evaluation step‑function sequence.
Algorithm
---------
1. Build a few simple initial distributions (uniform, arcsine,
   symmetric beta‑distribution with random exponent).
2. Run a lightweight random‑parameter search over a one‑parameter
   symmetric beta family (a_i ∝ (i/n)^{α‑1}(1‑i/n)^{α‑1}) and mix with
   the uniform distribution; keep the best candidate.
3. Refine the best candidate with a *smoothed* maximum optimisation:
   replace max(conv) by a log‑sum‑exp with a temperature that is
   increased slowly.  The gradient is obtained via FFT and a
   multiplicative (exponentiated) update with projection onto the simplex.
4. A short projected sub‑gradient phase and a greedy hill‑climb clean up
   any remaining slack.
5. The resulting vector is returned; the harness evaluates it with
   ``evaluate_sequence``.
The routine respects the supplied time budget (default 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a

def beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric beta(α,α) density on a discrete grid of size n."""
    # positions in (0,1)
    i = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-14
    i = np.clip(i, eps, 1.0 - eps)
    shape = (i ** (alpha - 1.0)) * ((1.0 - i) ** (alpha - 1.0))
    return shape

def mix_uniform_beta(alpha: float, mix: float, n: int) -> np.ndarray:
    """Mixture of uniform (weight mix) and symmetric beta(α,α) (weight 1‑mix)."""
    uniform = np.ones(n, dtype=np.float64)
    beta = beta_shape(alpha, n)
    a = mix * uniform + (1.0 - mix) * beta
    a /= a.sum()
    return a

def random_parameter_search(n: int, fft_len: int, rng: np.random.Generator,
                            deadline: float, start_a: np.ndarray, start_score: float):
    """Search a few hundred random β‑family candidates and keep the best."""
    best_a = start_a.copy()
    best_score = start_score
    # allocate up to 10 % of the remaining budget (or at most 30 s)
    max_time = min(30.0, deadline - time.time())
    while time.time() < deadline and (time.time() - (deadline - max_time)) < max_time:
        alpha = rng.uniform(0.05, 3.0)
        mix = rng.uniform(0.0, 1.0)
        cand = mix_uniform_beta(alpha, mix, n)
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[param‑search] best_score = {best_score:.6f}")
    return best_a, best_score

def smoothed_exponential_gradient_optimize(a_start: np.ndarray,
                                           deadline: float,
                                           n: int,
                                           fft_len: int,
                                           rng: np.random.Generator,
                                           eta0: float = 30.0,
                                           lambda_start: float = 5.0,
                                           lambda_factor: float = 1.02,
                                           lambda_max: float = 4000.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a soft‑max approximation of max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    lam = lambda_start
    while time.time() < deadline:
        # convolution of a with itself
        conv = full_conv(a, fft_len)

        # soft‑max weights
        # subtract max for numerical stability
        max_conv = conv.max()
        # exponent with current λ
        exps = np.exp(lam * (conv - max_conv))
        w = exps / exps.sum()   # shape (2n‑1,)

        # gradient of the smoothed objective
        a_rev = a[::-1]
        fm = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # step size (simple diminishing schedule)
        eta = eta0 / math.sqrt(it + 1.0)

        # multiplicative update and re‑normalisation
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
            s = a.sum()
        a /= s

        # track best
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # anneal λ
        lam = min(lambda_max, lam * lambda_factor)

        it += 1
    return best_a, best_score

def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        conv = full_conv(a, fft_len)

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        step = step0 / math.sqrt(it + 1.0)

        # simple back‑tracking: try decreasing step until improvement
        improved = False
        cur_score = fast_score(a, fft_len)
        for _ in range(4):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < cur_score:
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random jitter to escape stagnation
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score

def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1. Assemble a few simple starting vectors (uniform, arcsine, random,
   possible previous best).  Keep the best of these according to the
   cheap `fast_score`.

2. Perform a lightweight random search over two low‑dimensional families
   of symmetric distributions:
   * beta(α,α) on a discrete grid, optionally mixed with a uniform component;
   * a generalized arcsine family a_i ∝ ((i+δ)*(n+½−i+δ))^{‑p}.
   The best candidate of this stage becomes the new starting point.

3. Refine the candidate with a *smoothed* exponential‑gradient descent.
   The non‑smooth `max(conv)` is replaced by a log‑sum‑exp
   (soft‑max) with a temperature λ that is slowly increased.
   The gradient of the smoothed objective is obtained via FFT;
   a multiplicative update followed by simplex normalisation is used.

4. A short projected sub‑gradient phase cleans up any remaining slack.

5. A greedy hill‑climb with exact convolution updates performs a final
   local optimisation.

All phases respect the global time budget (default 1000 s).  The final
vector is returned; the harness evaluates it with the supplied
`evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (edges are heavier)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Families of parameterised shapes
# ----------------------------------------------------------------------


def beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric beta(α,α) density on a discrete grid of size n."""
    i = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-14
    i = np.clip(i, eps, 1.0 - eps)
    shape = (i ** (alpha - 1.0)) * ((1.0 - i) ** (alpha - 1.0))
    return shape


def mix_uniform_beta(alpha: float, mix: float, n: int) -> np.ndarray:
    """Mixture of uniform (weight mix) and symmetric beta(α,α)."""
    uniform = np.ones(n, dtype=np.float64)
    beta = beta_shape(alpha, n)
    a = mix * uniform + (1.0 - mix) * beta
    a /= a.sum()
    return a


def arcsine_general(p: float, delta: float, n: int) -> np.ndarray:
    """Generalised arcsine family:
       a_i ∝ ((i+δ)*(n+½−i+δ))^{‑p} ,  i = 0,…,n‑1.
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    numer = (idx + delta) * (n + 0.5 - idx + delta)
    a = numer ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Random parameter search over the two families
# ----------------------------------------------------------------------


def random_parameter_search(n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            deadline: float,
                            start_a: np.ndarray,
                            start_score: float) -> tuple[np.ndarray, float]:
    """Explore a few hundred low‑dimensional families and keep the best."""
    best_a = start_a.copy()
    best_score = start_score

    start_time = time.time()
    max_time = min(30.0, deadline - start_time)   # at most 30 s

    while time.time() < deadline and (time.time() - start_time) < max_time:
        if rng.random() < 0.5:
            # beta family
            alpha = rng.uniform(0.05, 3.0)
            mix = rng.uniform(0.0, 1.0)
            cand = mix_uniform_beta(alpha, mix, n)
        else:
            # generalized arcsine family
            p = rng.uniform(0.1, 1.2)
            delta = rng.uniform(0.0, 2.0)
            cand = arcsine_general(p, delta, n)

        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[param‑search] best_score = {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (soft‑max of convolution)
# ----------------------------------------------------------------------


def smoothed_exponential_gradient_optimize(a_start: np.ndarray,
                                           deadline: float,
                                           n: int,
                                           fft_len: int,
                                           rng: np.random.Generator,
                                           eta0: float = 30.0,
                                           lambda_start: float = 5.0,
                                           lambda_factor: float = 1.02,
                                           lambda_max: float = 4000.0) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on a soft‑max
    approximation of max(conv).  λ (temperature) is increased slowly.
    """
    a = a_start.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len)
    best_score = float(2.0 * n * conv.max())

    it = 0
    lam = lambda_start

    while time.time() < deadline:
        # current convolution
        conv = full_conv(a, fft_len)
        max_conv = conv.max()

        # soft‑max weights (log‑sum‑exp)
        exps = np.exp(lam * (conv - max_conv))
        w = exps / exps.sum()                # shape (2n‑1,)

        # gradient of the smoothed objective
        a_rev = a[::-1]
        fm = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # diminishing step size
        eta = eta0 / math.sqrt(it + 1.0)

        # multiplicative update
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total <= 0.0:
            a = best_a.copy()
            total = a.sum()
        a /= total

        # update best record
        cur_score = float(2.0 * n * max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # anneal λ
        lam = min(lambda_max, lam * lambda_factor)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (active‑set version)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    Active set = all offsets whose convolution is within `eps_frac` of the max.
    """
    a = a_start.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len)
    best_score = float(2.0 * n * conv.max())

    it = 0
    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        step = step0 / math.sqrt(it + 1.0)

        # back‑tracking line search
        cur_score = float(2.0 * n * max_val)
        improved = False
        for _ in range(4):
            cand = a - step * g
            cand = simplex_projection(cand)
            cand_conv = full_conv(cand, fft_len)
            cand_score = float(2.0 * n * cand_conv.max())
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
            if cand_score < cur_score:
                a = cand
                improved = True
                break
            step *= 0.5

        if not improved:
            # tiny random jab to escape plateaus
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (exact convolution updates)
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that transfers a small fraction of mass between
    entries, accepting only improving moves."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact linear convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_score = float(2.0 * n * conv_new.max())

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    init_candidates = [
        np.full(n, 1.0 / n),             # uniform

        if old.ndim == 1 and old.size > 0:

                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()

    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # 3. Random parameter search over two low‑dimensional families
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_parameter_search(
            n, fft_len, rng, deadline, best_a, best_score

    # 4. Smoothed exponential‑gradient descent (≈ 55 % of remaining time)

        best_a, best_score = smoothed_exponential_gradient_optimize(
            best_a, phase1_deadline, n, fft_len, rng,
            eta0=30.0, lambda_start=5.0, lambda_factor=1.02, lambda_max=4000.0
        )
        print(f"[smoothed‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Projected sub‑gradient refinement (≈ 25 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.25 * (deadline - time.time())
        best_a, best_score = projected_subgradient_optimize(
            best_a, phase2_deadline, n, fft_len, rng,
            eps_frac=0.001, step0=0.2
        )
        print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final greedy hill‑climb (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
