def adam_smooth_optimize(a_start, deadline, n, fft_len, rng,
                         beta1=0.9, beta2=0.999, lr=0.1,
                         temp_factor=0.05, temp_decay=0.5):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    initial_temp = None

    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

        max_val = conv.max()
        if initial_temp is None:
            initial_temp = max_val * temp_factor + 1e-12

        # temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        # exponential decay to 1% of init temp
        T = initial_temp * (0.01 ** (elapsed / total))

        # compute softmax weights
        # shift to avoid overflow
        shifted = conv - max_val
        w = np.exp(shifted / T)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(w) / w.shape[0]
            w_sum = 1.0
        w = w / w_sum

        # gradient
        a_rev = a[::-1]
        fm = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n-1:n-1+n]

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1.  Try several problem sizes (n = 2000, 4000, 6000, 8000).  
    For each size build a small family of candidate shapes:
        * arcsine (β(½,½))  
        * power‑law families a_i ∝ (i·(n‑i))^{-p}  with p∈[0.35,1.0]  
        * mixtures of the above with a uniform background.
    All candidates are normalised to sum 1 and evaluated with a cheap FFT‑based
    score = 2·n·max(conv(a,a)).  The best candidate among all n is kept.

2.  Refine the best candidate with a **smoothed Adam optimisation**.
    The hard max in the objective is replaced by a temperature‑controlled
    soft‑max:
        L_T(a) = T·log Σ_i exp(conv_i/T) .
    The gradient is computed analytically via one more FFT.  Adam’s
    adaptive step‑sizes together with a slowly decaying temperature give a
    fast descent even on the non‑convex landscape.

3.  A projected sub‑gradient phase (active‑set mask) cleans up the remaining
    tiny wiggles.

4.  A final greedy hill‑climb with exact incremental convolution updates
    fine‑tunes the solution.

All steps respect the overall 1000 s budget.  The routine returns the
best sequence found; the harness will evaluate it with the supplied
`evaluate_sequence` function.
"""

import time

# ----------------------------------------------------------------------
# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max‑convolution
# ----------------------------------------------------------------------


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         lr: float = 0.15,
                         temp_factor: float = 0.05,
                         min_temp_factor: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a temperature‑smoothed max
    convolution L_T(a) = T·log Σ exp(conv_i/T).

    The temperature T gradually decays from
        T0 = temp_factor * max(conv)
    to
        T_end = min_temp_factor * T0
    over the allocated time.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    init_temp = None

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        if init_temp is None:
            init_temp = max_val * temp_factor + 1e-14

        # temperature schedule: exponential decay to min_temp_factor * init_temp
        elapsed = time.time() - start
        total = deadline - start
        # fraction of time elapsed in [0,1]
        frac = elapsed / total if total > 0 else 1.0
        T = init_temp * (min_temp_factor ** frac)

        # soft‑max weights
        # shift by max for numerical stability
        shifted = conv - max_val
        # clip exponent to avoid overflow
        exp_arg = np.clip(shifted / max(T, 1e-16), -50.0, 50.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.shape[0])
            w_sum = 1.0
        w = w / w_sum

        # gradient of Σ w_i * conv_i w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        # the slice n‑1 … n‑1+n‑1 selects the gradient for the original variables
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()

        # accept if an improvement, otherwise keep current a
        cur_score = fast_score(a, fft_len)
        if cand_score < cur_score:
            a = a_candidate
        else:
            # occasional jitter to escape plateaus
            if rng.random() < 0.01:
                jitter = rng.normal(scale=1e-4, size=n)
                a = a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()

        # safeguard against NaNs
        if np.any(np.isnan(a)):
            a = best_a.copy()

    return best_a, best_score

def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    Active set = all convolution entries within eps_frac of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)

            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()

            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1

    return best_a, best_score

# Greedy hill‑climb with exact convolution updates

def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem sizes to explore
    # ------------------------------------------------------------------
    n_candidates = [2000, 4000, 6000, 8000]
    best_a = None
    best_score = float('inf')
    best_n = None
    best_fft_len = None

    # ------------------------------------------------------------------
    # 1. Coarse grid over shape parameters (p and uniform mix)
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.35, 1.0, 14)          # power‑law exponent
    alpha_vals = np.linspace(0.0, 0.15, 11)      # mixing weight with uniform

    for n in n_candidates:
        fft_len = next_pow2(2 * n - 1)
        idx = np.arange(n, dtype=np.float64) + 0.5
        prod = idx * (n + 0.5 - idx)               # positive for all i

        # baseline candidates
        candidates = []

        # pure arcsine (p = 0.5)
        a_arc = 1.0 / np.sqrt(prod)
        a_arc /= a_arc.sum()
        candidates.append(a_arc)

        # random initial (useful for diversity)
        candidates.append(random_initial(n, rng))

        # grid over (p, α)
        for p in p_vals:
            base = prod ** (-p)                     # shape with exponent p
            base_sum = base.sum()
            if base_sum == 0:
                continue
            base /= base_sum

            for alpha in alpha_vals:
                # mixture with uniform component
                a = (1.0 - alpha) * base + alpha * (1.0 / n)
                a_sum = a.sum()
                if a_sum == 0:
                    continue
                a /= a_sum
                candidates.append(a)

        # evaluate all candidates for this n
        for a in candidates:
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_n = n
                best_fft_len = fft_len

    print(f"[grid] best fast_score = {best_score:.6f} (n = {best_n})")

    # ------------------------------------------------------------------
    # 2. Incorporate previous best if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != best_n:
                # interpolate to current size
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, best_n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            if a_old.sum() == 0:
                a_old = np.full_like(a_old, 1.0 / a_old.size)
            a_old /= a_old.sum()
            sc = fast_score(a_old, best_fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a_old.copy()
                print(f"[prev] integrated previous best, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # 3. Adam optimisation with smoothed max‑convolution
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.5 * (deadline - time.time())
    best_a, best_score = adam_smooth_optimize(
        best_a, phase1_deadline, best_n, best_fft_len, rng,
        beta1=0.9, beta2=0.999, lr=0.12,
        temp_factor=0.05, min_temp_factor=0.001
    )
    print(f"[adam] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (fine‑tuning)
    # ------------------------------------------------------------------
    phase2_deadline = time.time() + 0.3 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, best_n, best_fft_len, rng,
        eps_frac=0.001, step0=0.25
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final greedy hill‑climb (exact convolution updates)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
