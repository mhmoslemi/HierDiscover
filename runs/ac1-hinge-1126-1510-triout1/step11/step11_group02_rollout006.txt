evaluation with the official `evaluate_sequence` function.

The whole routine respects a global time budget (default 1000 s) and
always returns the best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (∑a = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence (list of floats) found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between speed and quality
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with the uniform distribution (score = 2)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (uses the cheap FFT surrogate)
    # ------------------------------------------------------------------
    def try_candidate(a: np.ndarray, tag: str = ""):
        nonlocal best_a, best_sc
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[update] {tag} → fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – dense sweep of exponential families.
    # The theoretical optimum is at rate ≈ ln(n) / n.
    # We explore a fine linear window around that point and a wide log‑span.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # theoretical optimal rate
    rate_opt = np.log(n) / n
    # fine linear sweep (±80 % around optimum)
    linear_rates = np.linspace(rate_opt * 0.2, rate_opt * 2.0, 250)
    # broad log‑spaced sweep (covers very small / larger rates)
    log_rates = np.logspace(np.log10(5e-4), np.log10(5e-3), 200)
    all_rates = np.unique(np.concatenate([linear_rates, log_rates]))

    i = np.arange(n, dtype=float)

    for rate in all_rates:
        if time.time() >= deadline:
            break
        a = np.exp(-rate * i)
        a /= a.sum()
        try_candidate(a, f"exp rate={rate:.6e}")

    # ------------------------------------------------------------------
    # Phase 2 – local multiplicative jitter around the current best.
    # This explores the neighbourhood on the simplex.
    # ------------------------------------------------------------------
    jitter_iters = 250
    while time.time() < deadline and jitter_iters > 0:
        jitter_iters -= 1
        # multiplicative log‑normal perturbation
        pert = np.exp(rng.normal(loc=0.0, scale=0.03, size=n))
        a = best_a * pert
        a = np.maximum(a, 0.0)
        a /= a.sum()
        try_candidate(a, "jitter")

    # ------------------------------------------------------------------
    # Phase 3 – final refinement with smooth‑max L‑BFGS.
    # The objective is g(a)=maxₖ (a*a)ₖ; we smooth it with a log‑sum‑exp.
    # ------------------------------------------------------------------
    def softmax(y):
        m = np.max(y)
        e = np.exp(y - m)
        return e / e.sum()

    def objective_and_grad(y, lam):
        """Smoothed max of a*a where a = softmax(y). Returns (obj, grad_y)."""
        a = softmax(y)

        # convolution a*a via FFT (full length 2n‑1)
        fa = np.fft.fft(a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv_full.max()
        # smoothing parameter
        lam_eff = max_c * lam + 1e-12
        shifted = conv_full - max_c
        exp_shift = np.exp(shifted / lam_eff)
        Z = exp_shift.sum()
        smooth_max = max_c + lam_eff * np.log(Z)
        f = 2.0 * n * smooth_max

        # gradient w.r.t. convolution entries
        w = exp_shift / Z                     # d smooth_max / d conv_i

        # back‑propagation through convolution (a is on the simplex)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        # indices corresponding to the central n values
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]

        # chain rule through softmax
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        return f, grad_y

    # start from the best point found so far (logits)
    y = np.log(best_a + 1e-12)

    # λ schedule from coarse to fine smoothing
    lam_schedule = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(lam,),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 1200, "gtol": 1e-12, "disp": False},
            )
            y = res.x
            a = softmax(y)
            try_candidate(a, f"lbfgs λ={lam:.3e}")
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

    # ------------------------------------------------------------------
    # Final official evaluation (uses the exact convolution implementation)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
