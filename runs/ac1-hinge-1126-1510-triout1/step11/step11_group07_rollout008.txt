    s = exp_u / np.sum(exp_u)
    dot = np.dot(s, half_grad)
    grad_u = s * (half_grad - dot)
    return grad_u, conv, a

"""
Propose an improved step‑function sequence.

Algorithm
---------
1. Build a symmetric initial distribution on the probability simplex.
   – deterministic arcsine shape (known to be a good starting point)
   – optionally reuse the previous best if available.

2. Optimise the *smoothed* maximum of the autocorrelation:
   • Parameterise the distribution by half‑logits `u` (size n//2) and recover the
     full symmetric probability vector via a softmax.
   • For a temperature τ compute a soft‑max weighting of the convolution values.
     The gradient of τ·log‑sum‑exp(conv/τ) w.r.t. the logits is obtained analytically
     using FFT‑based convolutions (O(n log n)).
   • Perform an Adam‑type update while exponentially cooling τ from 1e‑3 to 1e‑7.
     The whole annealing phase consumes ≈ 60 % of the time budget and keeps the
     best vector (exact max‑convolution) encountered.

3. Refine the best candidate with a projected sub‑gradient descent on the *true*
   max‑convolution (the code from the previous solution, but started from the
   improved point).  This takes ≈ 30 % of the remaining time.

4. A final greedy hill‑climb with exact incremental convolution updates
   (again the earlier implementation) is performed for the last few seconds.

The algorithm respects the simplex constraints at every step, enforces symmetry,
and uses only FFT‑based convolutions – fast enough to run many thousands of
iterations inside the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic utilities (same as in the reference code)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetric parametrisation (logits → full probability vector)
# ----------------------------------------------------------------------


def logits_to_full_a(u: np.ndarray) -> np.ndarray:
    """
    Convert half‑logits `u` (size m) to a symmetric probability vector `a`
    of length n = 2*m.
    """
    m = u.shape[0]
    # softmax of u
    u_max = np.max(u)
    exp_u = np.exp(u - u_max)
    s = exp_u / np.sum(exp_u)          # size m, sums to 1
    half = 0.5 * s                     # each half‑mass is 0.5
    a = np.concatenate([half, half[::-1]])
    return a


def compute_grad_and_score(u: np.ndarray, tau: float,
                           n: int, fft_len: int):
    """
    For given half‑logits `u` and temperature `tau` compute:
        * gradient w.r.t. `u` (for the smoothed objective τ·log‑sum‑exp)
        * the full probability vector `a`
        * the exact fast_score = 2·n·max(conv)
    """
    a = logits_to_full_a(u)

    # ----- convolution and smoothed max -----
    conv = full_conv(a, fft_len)                 # length 2n‑1
    max_c = conv.max()

    # softmax weights over conv (stable)
    shifted = conv - max_c                       # ≤ 0
    w_exp = np.exp(shifted / tau)               # may underflow -> 0, fine
    w = w_exp / w_exp.sum()                     # ∑ w = 1

    # ----- gradient of τ·log‑sum‑exp w.r.t. a -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    grad_full = np.real(np.fft.ifft(fa_rev * fw))   # length fft_len
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]       # slice of length n

    # ----- map gradient to half‑logits -----
    m = n // 2
    rev_idx = n - 1 - np.arange(m)                # counterpart indices
    half_grad = 0.5 * (grad_a[:m] + grad_a[rev_idx])

    # softmax of u (again) – needed for chain rule
    u_max = np.max(u)
    exp_u = np.exp(u - u_max)
    s = exp_u / np.sum(exp_u)                     # size m, ∑ s = 1

    dot = np.dot(s, half_grad)
    grad_u = s * (half_grad - dot)                # shape m

    # ----- fast score (exact max of conv) -----
    score = float(2.0 * n * max_c)

    return grad_u, a, score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max) – reuse previous version
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates (same as before)
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence using
    temperature‑annealed smoothed‑max gradient descent (Adam) followed by
    projected sub‑gradient and a final hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # ---------- Problem size ----------
    # 4000 works well, but 4096 (power of two) simplifies FFT handling.
    n = 4096
    m = n // 2
    fft_len = next_pow2(2 * n - 1)

    # ---------- Initial symmetric distribution ----------
    # deterministic arcsine shape, expressed via half‑logits.
    arcsine_full = arcsine_initial(n)
    half = arcsine_full[:m] * 2.0                     # sum to 1
    u_best = np.log(half + 1e-15)                    # logits for the half

    # If a previous best is available, try to use it (interpolated).
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            old /= old.sum()
            half_old = old[:m] * 2.0
            u_old = np.log(half_old + 1e-15)
            # keep the better of arcsine / old (quick cheap score)
            a_arc = logits_to_full_a(u_best)
            a_old = logits_to_full_a(u_old)
            if fast_score(a_old, fft_len) < fast_score(a_arc, fft_len):
                u_best = u_old

    # ---------- Timing ----------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    # ---------- Phase 1: annealed smoothed‑max optimisation ----------
    # allocate ≈ 60 % of the remaining budget
    phase1_deadline = time.time() + 0.60 * (deadline - time.time())

    # Adam state
    m_adam = np.zeros(m)
    v_adam = np.zeros(m)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # temperature schedule
    tau_start = 1e-3
    tau_end = 1e-7
    phase1_start = time.time()
    phase1_total = phase1_deadline - phase1_start

    best_score = float('inf')
    best_u = u_best.copy()
    best_a = logits_to_full_a(best_u)

    while time.time() < phase1_deadline:
        now = time.time()
        frac = (now - phase1_start) / phase1_total
        # exponential cooling of τ
        tau = tau_start * (tau_end / tau_start) ** frac

        # gradient, current vector and fast score
        grad_u, a_curr, score_curr = compute_grad_and_score(u_best, tau, n, fft_len)

        # keep best (exact max‑convolution) seen so far
        if score_curr < best_score:
            best_score = score_curr
            best_u = u_best.copy()
            best_a = a_curr.copy()

        # Adam update
        t_iter += 1
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_u
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m_adam / (1.0 - beta1 ** t_iter)
        v_hat = v_adam / (1.0 - beta2 ** t_iter)

        # base learning rate – scaled down as τ gets tiny
        base_lr = 0.2
        lr = base_lr * (1.0 - frac) + 0.01 * frac   # slowly decay

        u_best = u_best - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    # after phase‑1, rebuild the best full vector
    a_best = best_a
    best_score = fast_score(a_best, fft_len)
    print(f"[phase‑1] best fast_score = {best_score:.6f}")

    # ---------- Phase 2: projected sub‑gradient on the true max ----------
    # allocate ≈ 30 % of remaining time
    phase2_deadline = time.time() + 0.30 * (deadline - time.time())
    a_best, best_score = projected_subgradient_optimize(
        a_best, phase2_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.2
    )
    print(f"[phase‑2] fast_score = {best_score:.6f}")

    # ---------- Phase 3: final greedy hill‑climb ----------
    if time.time() < deadline:
        a_best, best_score = hill_climb_incremental(
            a_best, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ---------- Official evaluation ----------
    final_seq = a_best.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
