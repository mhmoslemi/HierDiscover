```python
"""
Search for a non‑negative step sequence that minimizes the
evaluation function `evaluate_sequence`.  The algorithm combines
a systematic parametric sweep, a large random Dirichlet‑style
exploration, aggressive L‑BFGS optimisation of a smoothed
log‑sum‑exp surrogate, projected Adam on the true max‑convolution,
rich simulated‑annealing moves and a final local pairwise
adjustment.  Time is split among phases so that the best
candidate seen at any moment is returned within the 1000 s
budget.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of the official score for a vector with sum 1:
    2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the autocorrelation a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: ∂conv/∂a = 2 * (a convolved with w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation building blocks
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting every stage
    from the result of the previous stage.  Returns the best vector
    (on the simplex) and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # logits (stay on simplex internally)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of the max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                     # any convex combo is permitted

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
        a = a_candidate

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with three move types (transfer, swap, triple‑redistribute).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_rev_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(alpha * np.ones(n))
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "exp_rev":
        rate = rng.uniform(0.0002, 0.0015)
        d = np.minimum(i, n - 1 - i)
        a = np.exp(-rate * d)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "dirichlet":
        alpha = rng.uniform(0.01, 0.5)
        a = rng.dirichlet(alpha * np.ones(n))
    elif family == "random":
        a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Systematic sweep over a modest set of parametric shapes."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential family
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law family
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic family (coarse grid)
    for mu in np.linspace(0.2 * n, 0.8 * n, 7):
        for sigma in np.linspace(200.0, 1200.0, 7):
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Gaussian family (few sigmas)
    for sigma in np.linspace(300.0, 1500.0, 9):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gauss(n, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gauss sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Fast random exploration over many families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.20   # spend ~20% of total budget

    families = [
        "power",
        "exp",
        "exp_rev",
        "gauss",
        "double_exp",
        "bimodal",
        "logistic",
        "binary_sparse",
        "dirichlet",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[rand] new best fast_score={best_sc:.6f} (family={fam})")
    return best_a, best_sc

# Local pairwise fine‑tuning
# ----------------------------------------------------------------------


def pairwise_local_opt(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that repeatedly transfers a tiny fraction
    from a relatively large entry to a relatively small one.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)
    cur_sc = best_sc

    k = max(10, n // 200)  # size of the “large / small” pools (≈0.5% of n)

    while time.time() < deadline:
        # identify pools
        large_idx = np.argpartition(-a, k)[:k]
        small_idx = np.argpartition(a, k)[:k]

        i = rng.choice(large_idx)
        j = rng.choice(small_idx)
        if i == j:
            continue

        delta = a[i] * rng.random() * 0.03   # move at most 3 % of the donor mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non‑negativity (should hold because delta ≤ a[i])
        sc_new = fast_score(a_new, fft_len, n)

        if sc_new < best_sc:
            best_sc, best_a = sc_new, a_new.copy()
            # print only occasionally to avoid clutter
        if sc_new < cur_sc:
            a = a_new
            cur_sc = sc_new

    return best_a, best_sc

"""
Search for a non‑negative step sequence that minimizes the evaluation
function ``evaluate_sequence``.  The algorithm mixes a systematic grid
search, a broad random Dirichlet‑style sweep, aggressive L‑BFGS on a
log‑sum‑exp surrogate, projected Adam on the true max‑convolution,
rich simulated‑annealing moves and a final cheap pairwise hill‑climb.
Time is split among phases so that the best candidate seen at any
moment is returned within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of the official score for a vector with sum = 1:
    2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the autocorrelation a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. each convolution entry
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting every stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)            # logits (stay on simplex internally)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of the max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                     # any convex combination is admissible

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
        a = a_candidate

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with three move types (transfer, swap, triple‑redistribute).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:                                   # simple transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:                                 # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:                                                # triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_rev_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "exp_rev":
        rate = rng.uniform(0.0002, 0.0015)
        d = np.minimum(i, n - 1 - i)
        a = np.exp(-rate * d)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "dirichlet":
        alpha = rng.uniform(0.01, 0.5)
        a = rng.dirichlet(alpha * np.ones(n))
    elif family == "random":
        a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

# Parametric sweeps
# ----------------------------------------------------------------------


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Systematic sweep over a modest set of parametric shapes."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential family
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law family
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic family (coarse grid)
    for mu in np.linspace(0.2 * n, 0.8 * n, 7):
        for sigma in np.linspace(200.0, 1200.0, 7):
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Gaussian family (few sigmas)
    for sigma in np.linspace(300.0, 1500.0, 9):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gauss(n, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gauss sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Fast random exploration over many families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10   # spend ~10 % of total budget

    families = [
        "power",
        "exp",
        "exp_rev",
        "gauss",
        "double_exp",
        "bimodal",
        "logistic",
        "binary_sparse",
        "dirichlet",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[rand] new best fast_score={best_sc:.6f} (family={fam})")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap pairwise fine‑tuning
# ----------------------------------------------------------------------


def pairwise_local_opt(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny fraction from a relatively large entry
    to a relatively small one, accepting only improvements.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)
    cur_sc = best_sc

    # size of the “large / small” pools (≈0.5 % of n, at least 10 entries)
    k = max(10, n // 200)

    while time.time() < deadline:
        large_idx = np.argpartition(-a, k)[:k]
        small_idx = np.argpartition(a, k)[:k]

        i = rng.choice(large_idx)
        j = rng.choice(small_idx)
        if i == j:
            continue

        delta = a[i] * rng.random() * 0.03   # at most 3 % of donor mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        sc_new = fast_score(a_new, fft_len, n)

        if sc_new < best_sc:
            best_sc, best_a = sc_new, a_new.copy()
        if sc_new < cur_sc:
            a = a_new
            cur_sc = sc_new

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 offers a good trade‑off between quality and speed
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation in the smoothed surrogate

    # initialise with the uniform vector
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    # optionally start from a historic construction
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, hs)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            sc_old = fast_score(a_old, fft_len_conv, n)
            if sc_old < best_sc:
                best_a, best_sc = a_old, sc_old
                print(f"[init] previous best fast_score={best_sc:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 0 – systematic grid search
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.20 * budget_s
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, phase0_end)
    if sc_grid < best_sc:
        best_a, best_sc = a_grid, sc_grid
        print(f"[phase0] grid best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random sweep
    # ------------------------------------------------------------------
    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_sc:
        best_a, best_sc = a_rand, sc_rand
        print(f"[phase1] random sweep best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 5e-07, 2e-07,
    ]
    maxiter_per_lam = 1800

    while time.time() < phase2_end:
        # jitter current best or start from a fresh exponential profile
        if rng.random() < 0.30:
            a0 = best_a * rng.normal(1.0, 0.03, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            rate = rng.uniform(0.0003, 0.0009)
            a0 = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a0, lam_schedule, maxiter_per_lam, phase2_end,
            fft_len_conv, fft_len_g, n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[phase2] improved fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        best_a, best_sc = subgradient_adam_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase3] Adam sub‑gradient best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (rich move set)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.90 * budget_s
    if time.time() < phase4_end:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            phase4_end,
            rng,
        )
        print(f"[phase4] annealing best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – very cheap pairwise hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = pairwise_local_opt(
            best_a, fft_len_conv, n, deadline, rng
        )
        print(f"[phase5] pairwise local best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
