def hillclimb_exact(a, deadline, rng, start_factor=0.20, end_factor=0.001, recompute_every=2000):
    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    fft_len = next_pow2(3 * n)  # enough for mask (2n-1) + a_rev (n)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score
    cur_max = conv.max()
    start_time = time.time()
    total_time = deadline - start_time
    steps = 0
    accepted = 0
    # Received initial score maybe passed.
    while time.time() < deadline:
        # compute subgradient
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(float)  # length 2n-1
        # compute gradient via FFT
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        g_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * g_full[n-1:n-1+n]
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] <= 0.0:
            # fallback random move
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero, p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # compute step factor
        remaining = deadline - time.time()
        remaining_frac = max(0.0, remaining / total_time)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-15:
            break
        # try to adjust delta to ensure nonincrease
        # We'll try decreasing delta by factor 0.5 until condition satisfied or delta < tolerance
        a_old = a.copy()
        conv_old = conv.copy()
        while delta > 1e-15:
            # exact update
            conv_new = conv_old.copy()
            # subtract contributions from src
            conv_new[src:src+n] -= 2.0 * delta * a_old
            # add contributions from dst
            conv_new[dst:dst+n] += 2.0 * delta * a_old
            # quadratic terms
            idx_src2 = 2 * src
            idx_dst2 = 2 * dst
            idx_sum = src + dst
            conv_new[idx_src2] += delta * delta
            conv_new[idx_dst2] += delta * delta
            conv_new[idx_sum] -= 2.0 * delta * delta
            new_max = conv_new.max()
            if new_max <= cur_max + 1e-18:
                # accept
                a = a_old.copy()
                a[src] -= delta
                a[dst] += delta
                conv = conv_new
                cur_max = new_max
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                accepted += 1
                break
            else:
                delta *= 0.5
        # If delta became too small, skip move (no change)
        steps += 1
        # Periodic recompute conv from scratch to avoid drift (though we used exact updates, but maybe due to rounding). Let's recompute every certain steps.
        if steps % recompute_every == 0:
            conv = np.convolve(a, a)

        # Occasionally print progress
        if steps % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[hill] step={steps} elapsed={elapsed:.1f}s best_score={best_score:.6f}")
    return best_a, best_score

"""
Improved step–function search.

* Initialise the probability vector with a variety of deterministic windows
  (uniform, Hann, Blackman, Kaiser) and a few random starts.
* Pick the best initial vector (lowest 2·n·max(a*a)).
* Run a *gradient‑guided hill‑climb* that moves mass from the index with the
  largest sub‑gradient (most responsible for the current peak of the
  self‑convolution) to the index with the smallest sub‑gradient.
  The move size is chosen by a line‑search (halving) that guarantees the
  maximum of the convolution does not increase.
* The convolution is updated **exactly** (including the quadratic
  δ²‑terms), so no periodic recomputation is required.
* The step‑size factor decays linearly from `start_factor` to `end_factor`
  during the run.
* The whole budget (≈1000 s) is spent in the hill‑climb; occasional random
  restarts are performed if no improvement occurs for a while.
* The final vector is returned as a list of floats; `evaluate_sequence`
  is called once at the end for the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)                     # exact linear convolution
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random distributions on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def init_hann(n: int) -> np.ndarray:
    w = np.hanning(n)
    w /= w.sum()
    return w


def init_blackman(n: int) -> np.ndarray:
    w = np.blackman(n)
    w /= w.sum()
    return w


def init_kaiser(n: int, beta: float = 8.0) -> np.ndarray:
    w = np.kaiser(n, beta)
    w /= w.sum()
    return w


def init_exponential(n: int, rate: float = 0.001) -> np.ndarray:
    w = np.exp(-rate * np.arange(n))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Exact‑update gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def hillclimb_exact(a_start: np.ndarray,
                    start_score: float,
                    deadline: float,
                    rng: np.random.Generator,
                    start_factor: float = 0.20,
                    end_factor: float = 0.001,
                    recompute_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the index with the largest sub‑gradient
    to the index with the smallest sub‑gradient.
    The step size decays linearly from `start_factor` to `end_factor`.
    The convolution is updated exactly (including the δ² terms) so
    the value of max(convolution) never increases.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact self‑convolution, len = 2n‑1
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    cur_max = conv.max()

    start_time = time.time()
    total_time = deadline - start_time
    steps = 0
    accepted = 0

    # pre‑allocate FFT length for the gradient computation
    fft_len = next_pow2(3 * n)   # mask (2n‑1) + a_rev (n) – 1 ≤ 3n‑2

    while time.time() < deadline:
        # -------------------------------------------------------------
        # 1. sub‑gradient of the max‑convolution (mask‑based)
        # -------------------------------------------------------------
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(float)   # length 2n‑1

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        g_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * g_full[n - 1:n - 1 + n]   # gradient w.r.t. a

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # fallback when gradient is degenerate
        if src == dst or a[src] <= 0.0:
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero, p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # -------------------------------------------------------------
        # 2. choose a step size (fraction of a[src])
        # -------------------------------------------------------------
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac
        delta = a[src] * step_factor
        if delta < 1e-15:
            break

        # -------------------------------------------------------------
        # 3. line‑search (simple halving) to keep max(conv) non‑increasing
        # -------------------------------------------------------------
        a_old = a.copy()
        conv_old = conv.copy()
        while delta > 1e-15:
            # exact convolution update for the move a[src] → a[src]‑δ,
            # a[dst] → a[dst] + δ
            conv_new = conv_old.copy()
            # linear part: shift contributions
            conv_new[src:src + n] -= 2.0 * delta * a_old
            conv_new[dst:dst + n] += 2.0 * delta * a_old
            # quadratic δ² terms
            idx_src2 = 2 * src
            idx_dst2 = 2 * dst
            idx_sum = src + dst
            conv_new[idx_src2] += delta * delta
            conv_new[idx_dst2] += delta * delta
            conv_new[idx_sum] -= 2.0 * delta * delta

            new_max = conv_new.max()
            if new_max <= cur_max + 1e-18:          # accept if not larger
                # apply the move
                a = a_old.copy()
                a[src] -= delta
                a[dst] += delta
                conv = conv_new
                cur_max = new_max
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                accepted += 1
                break
            else:
                delta *= 0.5  # shrink step and retry

        # -------------------------------------------------------------
        # 4. housekeeping
        # -------------------------------------------------------------
        steps += 1
        # periodic exact recompute to avoid drift from rounding errors
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional progress report
        if steps % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[hill] steps={steps} elapsed={elapsed:.1f}s "
                  f"best_score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns a list of non‑negative floats (the heights) found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer search space.
    # ------------------------------------------------------------------
    n = 8000
    print(f"[setup] n={n}")

    # ------------------------------------------------------------------
    # 1. generate a pool of deterministic and random starts
    # ------------------------------------------------------------------
    candidates = []

    # deterministic windows
    candidates.append(np.full(n, 1.0 / n))          # uniform
    candidates.append(init_hann(n))
    candidates.append(init_blackman(n))
    candidates.append(init_kaiser(n, beta=6.0))
    candidates.append(init_kaiser(n, beta=9.0))
    candidates.append(init_exponential(n, rate=0.001))
    candidates.append(init_exponential(n, rate=0.008))

    # a few random starts
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    # evaluate the pool quickly
    best_a = None
    best_score = np.inf
    for a in candidates:
        s = fast_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    print(f"[init] best fast_score from pool = {best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # 2. Main hill‑climb.  If stagnation occurs, restart from a fresh random.
    # ------------------------------------------------------------------
    last_improve = time.time()
    stagnation_limit = 30.0    # seconds without improvement → restart

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 15.0:                # not enough time for a fresh run
            break

        # run a hill‑climb from the current best for at most `run_time`
        run_time = min(remaining * 0.9, 120.0)   # cap each run to 2 min
        hill_deadline = time.time() + run_time

        a_local, score_local = hillclimb_exact(
            best_a, best_score, hill_deadline, rng,
            start_factor=0.25, end_factor=0.001, recompute_every=5000
        )
        if score_local < best_score - 1e-9:
            best_score = score_local
            best_a = a_local.copy()
            last_improve = time.time()
            print(f"[global] new best during hill‑climb: {best_score:.6f}")
        else:
            # no improvement – perhaps we are stuck → random restart
            if time.time() - last_improve > stagnation_limit:
                print("[restart] stagnation detected – new random start")
                a_random = random_initial(n, rng)
                best_a = a_random
                best_score = fast_score(best_a)
                last_improve = time.time()

    # ------------------------------------------------------------------
    # 3. Final evaluation with the security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
