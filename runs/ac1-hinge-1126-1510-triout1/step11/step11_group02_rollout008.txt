def subgradient_descent(a0, best_score, fft_len, fft_len_g, n, deadline, rng):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    # momentum
    m = np.zeros_like(a)
    beta_mom = 0.9
    lr0 = 0.5  # initial learning rate
    # We'll adapt step based on time fraction
    start = time.time()
    phase_duration = deadline - start
    while time.time() < deadline:
        # compute conv and gradient
        Fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(Fa * Fa))[:2*n-1]
        max_val = conv.max()
        # identify active shifts
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        mask_sum = mask.sum()
        mask /= mask_sum
        # gradient
        a_rev = a[::-1]
        F_rev = np.fft.fft(a_rev, fft_len_g)
        F_mask = np.fft.fft(mask, fft_len_g)
        conv_rev = np.real(np.fft.ifft(F_rev * F_mask))
        g = 2.0 * conv_rev[n-1:n-1+n]

        m = beta_mom * m + (1 - beta_mom) * g

        elapsed = time.time() - start

        a_new = simplex_projection(a_new)

            best_sc = sc

"""
Search for a non‑negative step sequence with a low evaluation score.

Algorithm
---------
0️⃣ Initialise from the historic best (if available) or the uniform vector.
1️⃣ Systematic grid + random sweep over a rich collection of parametric
   families (including a Beta‐distribution family that can produce U‑,
   bell‑ and flat‑shaped profiles).
2️⃣ Long projected sub‑gradient descent on the exact max‑convolution
   (Nesterov‑type momentum, decaying step‑size).  This directly minimises
   the true objective without any smoothing.
3️⃣ A short simulated‑annealing phase with transfer / swap / triple‑redistribution
   moves to escape possible plateaus.
4️⃣ Final polish with a few L‑BFGS runs on the smooth log‑sum‑exp
   surrogate (very small smoothing λ) to clean up the remaining
   sub‑optimalities.
All phases respect a global deadline; the best vector seen at any moment
is returned and finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (uses Σa = 1)."""
    Fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Beta(α,β) on [0,1] discretised onto n points."""
    x = np.linspace(0.0, 1.0, n)
    # avoid 0**neg or (1‑x)**neg by clipping
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta_param - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "beta":
        alpha = rng.uniform(0.2, 5.0)
        beta_param = rng.uniform(0.2, 5.0)
        a = generate_beta(n, alpha, beta_param)
    else:  # pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def systematic_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
    best_a: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Light systematic sweep over exponential, power‑law and beta families."""
    # exponential rates
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_score
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_score:.6f}")

    # power‑law (increasing)
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_score
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_score:.6f}")

    # beta family (coarse grid)
    alphas = np.linspace(0.3, 4.0, 14)
    betas = np.linspace(0.3, 4.0, 14)
    for alpha in alphas:
        for beta_param in betas:
            if time.time() >= deadline:
                return best_a, best_score
            a = generate_beta(n, alpha, beta_param)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(
                    f"[grid] beta α={alpha:.2f}, β={beta_param:.2f} → score={best_score:.6f}"
                )
    return best_a, best_score


def cheap_random_sweep(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    best_a: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Quick random exploration over many families (incl. beta)."""
    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "beta",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[rand] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score


def subgradient_descent(
    a_start: np.ndarray,
    best_score: float,
    fft_len: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Nesterov‑style momentum.
    Works directly on the true max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = best_score

    # momentum term
    m = np.zeros_like(a)
    beta = 0.9
    init_lr = 0.6

    start = time.time()
    phase_len = deadline - start

    while time.time() < deadline:
        # convolution a*a
        Fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active shifts (all within tiny eps of the max)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[np.argmax(conv)] = 1.0
            cnt = 1.0
        mask /= cnt

        # gradient of the max (sub‑gradient)
        a_rev = a[::-1]
        Fa_rev = np.fft.fft(a_rev, fft_len_g)
        Fmask = np.fft.fft(mask, fft_len_g)
        conv_rev = np.real(np.fft.ifft(Fa_rev * Fmask))
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # decaying learning rate
        elapsed = time.time() - start
        lr = init_lr * (1.0 - elapsed / phase_len)
        if lr < 5e-4:
            lr = 5e-4

        a_candidate = a - lr * m
        a_candidate = simplex_projection(a_candidate)

        cand_sc = fast_score(a_candidate, fft_len, n)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()
            print(f"[sgd] new best fast_score={best_sc:.6f}")

        a = a_candidate

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(1.0, 0.02, size=n)
            a = np.maximum(a * jitter, 0.0)
            a /= a.sum()
    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb with a square‑root cooling schedule.
    Moves:
      – transfer mass between two positions,
      – swap two entries,
      – redistribute mass among three indices.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature
    T0 = max(0.08 * start_score, 1e-6)
    t0 = time.time()
    total = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # transfer a fraction from i → j
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution (preserves total mass)
            idx = rng.choice(n, size=3, replace=False)
            total_triple = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total_triple * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def final_lbfgs_refine(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very short L‑BFGS polish on the smoothed log‑sum‑exp surrogate.
    Uses an ultra‑small λ schedule.
    """
    y = np.log(a_start + 1e-12)  # logits – stays on the simplex
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    # aggressive λ schedule heading to zero
    lam_schedule = [0.05, 0.02, 0.008, 0.003, 0.001, 5e-04, 2e-04, 1e-04, 5e-05]
    maxiter = 1500

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] polished fast_score={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size (use historic length if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        n = hs.size if hs.ndim == 1 and hs.size > 0 else 4000
    else:
        n = 4000

    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform) possibly replaced by historic best
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a1 = np.interp(x_new, x_old, hs)
                a1 = np.maximum(a1, 0.0)
                a1 /= a1.sum()
            else:
                a1 = hs.copy()
                a1 = np.maximum(a1, 0.0)
                a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] historic fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – systematic sweep + cheap random sweep
    # --------------------------------------------------------------
    phase0_end = start_time + 0.12 * budget_s
    best_a, best_score = systematic_grid_search(
        n, fft_len_conv, rng, phase0_end, best_a, best_score
    )
    best_a, best_score = cheap_random_sweep(
        n, fft_len_conv, phase0_end, rng, best_a, best_score
    )
    print(f"[phase0] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – long projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    phase1_end = start_time + 0.70 * budget_s
    best_a, best_score = subgradient_descent(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        phase1_end,
        rng,
    )
    print(f"[phase1] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing refinement
    # --------------------------------------------------------------
    phase2_end = start_time + 0.90 * budget_s
    best_a, best_score = simulated_annealing(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase2_end,
        rng,
    )
    print(f"[phase2] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing – short L‑BFGS run with ultra‑small λ
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = final_lbfgs_refine(
            best_a, fft_len_conv, fft_len_g, n, deadline
        )
        print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
