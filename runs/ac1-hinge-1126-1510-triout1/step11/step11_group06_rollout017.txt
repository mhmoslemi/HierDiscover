def subgrad_max(a, fft_len):
    conv = np.convolve(a,a)
    max_val = conv.max()
    eps = 1e-12 * max_val
    mask = (conv >= max_val - eps).astype(float)   # weight each active index equally (or normalized)
    # Compute gradient of each active entry wrt a: For each k, grad_k = 2 * a_rev shifted. Efficiency via FFT.

def hill_climb(a_start, init_score, deadline, rng, step_start=0.5, step_end=1e-4, recompute_every=1000):

   a = a_start.copy()
   conv = np.convolve(a, a)
   n = len(a)
   fft_len = next_pow2(2*n-1)
   best_a = a.copy()
   best_score = init_score

   # Record start time for step schedule
   start_time = time.time()
   total = deadline - start_time

   while time.time() < deadline:
       cur_max = conv.max()
       cur_score = 2 * n * cur_max
       # Subgradient
       eps = cur_max * 1e-12 + 1e-12
       mask = (conv >= cur_max - eps).astype(np.float64)

       # compute gradient using FFT
       a_rev = a[::-1]
       grad_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                      np.fft.fft(a_rev, fft_len)))
       g = 2 * grad_full[n-1:n-1+n]  # shape n

       src = int(np.argmax(g))
       dst = int(np.argmin(g))

       if src == dst or a[src] < 1e-18:
           # pick random
           nonzero = np.where(a > 1e-12)[0]
           if nonzero.size == 0:
               break
           src = int(rng.choice(nonzero, p=a[nonzero]/a[nonzero].sum()))
           dst = int(rng.integers(0, n))
           if src == dst:
               continue

       # compute step factor based on remaining time globally
       remaining_frac = max(0.0, (deadline - time.time())/total)
       step_factor = step_end + (step_start - step_end) * remaining_frac

       # Move proportion of source mass or fixed epsilon
       delta = a[src] * step_factor
       if delta < 1e-12:
           continue
       delta = min(delta, a[src])  # cannot exceed source mass

       a_old = a.copy()
       conv_old = conv.copy()

       # Update convolution (linear term only) similar to guided_hc
       # conv = conv - 2*delta*a shifted from src + 2*delta*a shifted to dst
       # The shift operation: convolution of a with Dirac at src is a shifted.
       # Equivalent to conv[src:src+n] -= 2*delta*a (because conv_k includes sum_i a_i*a_{k-i}; subtract contributions where i=src)
       conv[src:src + n] -= 2.0 * delta * a_old
       conv[dst:dst + n] += 2.0 * delta * a_old

       # Quadratic terms: we ignore small delta^2 but for accuracy we could recompute conv exactly after some steps
       a[src] -= delta
       a[dst] += delta

       new_max = conv.max()
       new_score = 2.0 * n * new_max

       if new_score <= cur_score + 1e-12:   # accept non-worsening
           if new_score < best_score:
               best_score = new_score
               best_a = a.copy()
           # accepted
       else:
           # reject
           a = a_old
           conv = conv_old

       # periodic recompute
       if np.random.rand() < 0.001:  # small probability to recompute.
           conv = np.convolve(a, a)

   return best_a, best_score

We need to test if our increments are correct. The update conv[src:src+n] -= 2δ a_old is because original conv includes contributions from a[src] * a[i] and a[i] * a[src] for each i. When we reduce a[src] by δ, each term a[src] * a[i] changes to (a[src]-δ) * a[i] = a[src]*a[i] - δ a[i]; similarly a[i] * a[src] reduces by δ a[i]; total reduction = 2δ a[i]. So vector to subtract is 2δ*a_old[i] for each i, which appears in conv at index src+i (since sum indexes). Works.

When we increase a[dst] by δ, we add 2δ*a[i] contributions at conv[dst + i]. So conv[dst:dst+n] += 2δ*a_old[i].

But note a_old[i] is the old distribution (before moving). However for contributions involving dst and src simultaneously, there is also cross term -2δ^2 at index src+dst (since the product a_src * a_dst originally contributed a_src a_dst + a_dst a_src = 2 a_src a_dst, after change it's (a_src - δ) (a_dst + δ) + (a_dst + δ) (a_src - δ) = 2 (a_src a_dst + δ a_dst - δ a_src - δ^2). The linear terms accounted earlier (by adjusting conv with -2δ a_src or +2δ a_dst). The δ^2 term leads to -2δ^2, which we ignore. For small δ relative to a values, error small.

"""
Optimise a non‑negative step‑function (probability vector) to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant, so we keep the vector normalised (sum=1) and
optimise the constant C = 2·n·max(conv(a,a)).

The search consists of three stages:

1.  Several short but intense L‑BFGS runs on a smooth log‑sum‑exp
    approximation of the max‑convolution.  This quickly drives the
    solution into a good basin.

2.  A short Adam optimisation of the same smoothed objective.

3.  A large allocation of time to *gradient‑guided hill‑climbing*.
    The hill‑climb repeatedly moves a fraction of mass from the index
    with the largest sub‑gradient (most responsible for the current peak)
    to the index with the smallest sub‑gradient, accepting the move only
    if the peak does not increase.  The step size decays linearly with
    the remaining time; the full convolution is recomputed periodically
    to eliminate drift from the linear update.

To avoid getting stuck, the hill‑climb is launched many times from
perturbed copies of the current best solution as well as from fresh
random initialisations.  The best sequence found before the deadline is
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

# Smoothed objective for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.20,
                      end_factor: float = 0.001,
                      recompute_every: int = 2000):
    """
    Move mass from the index with the largest sub‑gradient to the index with
    the smallest sub‑gradient.  Step size decays linearly with remaining
    time.  Convolution is updated incrementally; the exact convolution is
    recomputed every *recompute_every* accepted moves.
    Returns the best a and its fast_score found before *deadline*.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution (len 2n‑1)
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        delta = min(delta, a[src])

        # save state for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental convolution update (linear part only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move the mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score + 1e-14:
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject the move
            a = a_old
            conv = conv_old

        steps += 1

        # periodic exact recompute
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional diagnostics
        if steps % 10000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 entries give a good trade‑off (larger sizes
    # increase flexibility but also cost more time per iteration)
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the previously found construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # Linear interpolation to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS bursts from random starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s   # ~150 s
    lam_schedule = np.geomspace(0.5, 1e-6, num=20)
    maxiter_per_lam = 500
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective (short warm‑up)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02,
                     0.01, 0.005, 0.002, 0.001, 5e-4]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=3000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.15
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – many restarts of gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        while time.time() < deadline:
            remaining = deadline - time.time()
            # allocate a modest chunk (max 30 s) for each restart
            chunk = min(30.0, remaining * 0.5)
            local_deadline = time.time() + chunk

            # 80 % of the time start from a perturbed copy of the best,
            # 20 % from a fresh random start.
            if rng.random() < 0.8:
                a_start = best_a.copy()
                noise = rng.normal(scale=0.01, size=n)
                a_start += noise
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()
            else:
                a_start = random_initial(n, rng)

            start_score = fast_score(a_start, fft_len)
            a_new, score_new = guided_hill_climb(
                a_start, start_score, local_deadline, rng,
                start_factor=0.4, end_factor=0.001, recompute_every=1000
            )
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                print(f"[restart_hc] new best fast_score={best_score:.6f}")

            # occasional short refinement directly on the global best
            if time.time() + 5.0 < deadline:
                refine_deadline = time.time() + min(5.0, deadline - time.time())
                start_score_best = fast_score(best_a, fft_len)
                a_ref, score_ref = guided_hill_climb(
                    best_a, start_score_best, refine_deadline, rng,
                    start_factor=0.2, end_factor=0.001, recompute_every=2000
                )
                if score_ref < best_score:
                    best_score = score_ref
                    best_a = a_ref.copy()
                    print(f"[refine_hc] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
