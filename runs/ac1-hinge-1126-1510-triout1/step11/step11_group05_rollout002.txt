def compute_subgradient(a, conv, max_val, eps=1e-12):
    mask = (conv >= max_val - eps).astype(float)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, L)  # L = length for conv (maybe L2)
    fa_rev = np.fft.fft(a_rev, L)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    # gradient for objective = 2 * grad_conv central part

def compute_subgrad(a, n, L):
    # compute conv
    fa = np.fft.fft(a, L)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_val = conv.max()
    # mask for entries near max_val
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    rev_a = a[::-1]
    fm = np.fft.fft(mask, L)
    fa_rev = np.fft.fft(rev_a, L)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n-1:n-1+n]  # length n
    return g, conv, max_val

def adam_logit_subgrad_optimize(a_start, deadline, n, L, rng, max_iters=200000):
    # same as original Adam but using subgradient.
    # This will likely converge faster.
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.04 # step for logit
    y = np.log(a_start + 1e-12)

        g, _, _ = compute_subgrad(a, n, L)
        # compute gradient w.r.t logits (softmax chain)
        mean_g = np.dot(a, g)

def targeted_local_search(a_start, deadline, n, L, rng, max_iters=500000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, L, n)
    # compute gradient once maybe
    g, _, _ = compute_subgrad(a, n, L)
    it = 0
    while time.time() < deadline and it < max_iters:
        # Choose donor as index with largest g (most contribution)
        donor = int(np.argmax(g))
        receiver = int(np.argmin(g))
        if donor == receiver:
            break
        if a[donor] <= 0:
            # recompute gradient if donor has zero mass
            g, _, _ = compute_subgrad(a, n, L)
            continue
        # Transfer fraction
        eps = a[donor] * rng.uniform(0.01, 0.2)  # move up to 20% of donor mass
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        # ensure positivity
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        new_score = fast_score(a_new, L, n)
        if new_score < best_score:
            a = a_new
            best_score = new_score
            best_a = a_new.copy()
            # recompute gradient after acceptance
            g, _, _ = compute_subgrad(a, n, L)
        else:
            # maybe still accept with small probability for escape
            # use simple temperature schedule
            temp = 0.001 * (0.99 ** it)
            if temp > 0 and rng.random() < np.exp(-(new_score - best_score) / temp):
                a = a_new
                g, _, _ = compute_subgrad(a, n, L)
        it += 1
    return best_a, best_score

"""
Propose a low‑score step‑function sequence.

Algorithm outline (≈ 1000 s budget)

1.  Initialise a diverse pool (uniform, random, exponential, edge‑heavy,
    previous best) and keep the best vector on the probability simplex.
2.  Adam optimisation on a *smooth* log‑sum‑exp surrogate of the
    max‑convolution (many more steps than before).
3.  Adam optimisation on the *true* sub‑gradient of the max‑convolution
    (logits → soft‑max, Adam on the resulting gradient).
4.  Exponentiated‑gradient descent on the exact max‑convolution, with a
    larger initial step‑size and a sqrt‑decay schedule.
5.  Targeted local search: repeatedly move a fraction of the mass from the
    index with the largest sub‑gradient (most responsible for the peak) to
    the index with the smallest sub‑gradient; accept improving moves and
    occasionally uphill moves according to a simple temperature schedule.
6.  Convex‑combination (mixture) line‑search over the top‑K vectors found.
7.  Return the best list; also store it globally as ``height_sequence_1``
    for reuse by future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective = 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return float(2.0 * n * conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_and_conv(a: np.ndarray, n: int, L: int):
    """
    Return (g, conv, max_val) where
    - conv is the linear self‑convolution a*a (length 2n‑1)
    - max_val = conv.max()
    - g is a sub‑gradient of max(conv) w.r.t. a (size n)
    """
    fa = np.fft.fft(a, L)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, L)
    fa_rev = np.fft.fft(a_rev, L)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    # central block corresponds to indices n‑1 … n‑1+n‑1
    g = 2.0 * grad_conv[n - 1 : n - 1 + n]
    return g, conv_full, max_val


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_objective_and_grad(y: np.ndarray, n: int, Lc: int, Lg: int,
                              lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, Lc)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, Lg)
    fw = np.fft.fft(w, Lg)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize_smooth(a_start: np.ndarray,
                         lam_schedule,
                         steps_per_lam: int,
                         deadline: float,
                         Lc: int,
                         Lg: int,
                         n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    """
    beta1, beta2, eps, lr = 0.9, 0.999, 1e-8, 0.05
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, Lc, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = smooth_objective_and_grad(
                y, n, Lc, Lg, lam=lam
            )
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            sc = fast_score(a, Lc, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

def adam_optimize_subgrad(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          L: int,
                          rng: np.random.Generator,

    working on soft‑max logits.
    """

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        sc = fast_score(a, L, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          L: int,
                          rng: np.random.Generator,
                          init_scale: float = 20.0,
                          min_scale: float = 0.1):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, L, n)

    it = 0
    while time.time() < deadline:
        g, _, _ = subgrad_and_conv(a, n, L)
        scale = max(min_scale, init_scale / np.sqrt(it + 1))
        exp_term = np.exp(np.clip(-scale * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = best_a.copy()
        sc = fast_score(a, L, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fallback, cheap)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              L: int,
                              init_eta: float = 0.7):
    """Simple Euclidean projected sub‑gradient descent."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, L, n)

    it = 0
    eta = init_eta
    while time.time() < deadline:
        g, _, _ = subgrad_and_conv(a, n, L)
        a_step = a - eta * g
        a = proj_simplex(a_step)

        sc = fast_score(a, L, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-9:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (mass transfer guided by the sub‑gradient)
# ----------------------------------------------------------------------


def targeted_local_search(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          L: int,
                          rng: np.random.Generator,
                          max_iters: int = 500_000):
    """
    Move a fraction of mass from the index with the largest gradient
    (most responsible for the current peak) to the index with the smallest.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, L, n)

    g, _, _ = subgrad_and_conv(a, n, L)

    it = 0
    while time.time() < deadline and it < max_iters:
        donor = int(np.argmax(g))
        receiver = int(np.argmin(g))
        if a[donor] <= 0:
            g, _, _ = subgrad_and_conv(a, n, L)
            continue
        # transfer a modest fraction of the donor's mass
        eps = a[donor] * rng.uniform(0.01, 0.20)
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_score = fast_score(a_new, L, n)
        if new_score < best_score:
            a = a_new
            best_a = a_new.copy()
            best_score = new_score
            g, _, _ = subgrad_and_conv(a, n, L)
        else:
            # occasional uphill move (simple temperature schedule)
            temp = 0.001 * (0.995 ** it)
            if temp > 0 and rng.random() < np.exp(-(new_score - best_score) / temp):
                a = a_new
                g, _, _ = subgrad_and_conv(a, n, L)
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------


def mixture_search(candidates, scores, n, L, deadline):
    """Golden‑section line search over all unordered pairs."""
    best_a = candidates[0].copy()
    best_score = scores[0]

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            lo, hi = 0.0, 1.0
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, L, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, L, n)

            while time.time() < deadline and (hi - lo) > 1e-4:
                if fc < fd:
                    hi, fd = d, fc
                    d = c
                    c = hi - invphi * (hi - lo)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, L, n)
                else:
                    lo, fc = c, fd
                    c = d
                    d = lo + invphi * (hi - lo)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, L, n)

            tm = 0.5 * (lo + hi)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            ftm = fast_score(mix_tm, L, n)

            if ftm < best_score:
                best_score = ftm
                best_a = mix_tm.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a modestly large n gives enough flexibility.
    # ------------------------------------------------------------------
    n = 4000
    Lc = next_pow2(2 * n - 1)          # length for convolution FFT
    Lg = next_pow2(3 * n - 2)          # length for gradient‑related FFTs

    # ------------------------------------------------------------------
    # Initialise from the historic best if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            if a_start.sum() > 0:
                a_start = a_start / a_start.sum()
            else:
                a_start = np.full(n, 1.0 / n)
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build diverse initial pool.
    # ------------------------------------------------------------------
    pool = [
        a_start.copy(),
        random_initial(n, rng),
        random_initial(n, rng),
        add_noise(a_start, rng.uniform(0.001, 0.01), rng),
        add_noise(a_start, rng.uniform(0.002, 0.02), rng),
    ]

    best_a = pool[0].copy()
    best_score = fast_score(best_a, Lc, n)
    for cand in pool[1:]:
        sc = fast_score(cand, Lc, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed surrogate (more steps than before).
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 400   # increased granularity

    for r in range(6):
        if time.time() >= deadline:
            break
        a0 = best_a.copy() if r == 0 else add_noise(best_a,
                                                    rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize_smooth(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            Lc,
            Lg,
            n,
        )
        print(f"[Adam‑smooth restart {r + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best (smooth Adam): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the true sub‑gradient (logits).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = adam_optimize_subgrad(
            best_a,
            deadline,
            n,
            Lc,
            rng,
        )
        print(f"[Adam‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best (Adam sub‑grad): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent on true max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            Lc,
            rng,
            init_scale=20.0,
            min_scale=0.05,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted local search guided by the sub‑gradient.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = targeted_local_search(
            best_a,
            deadline,
            n,
            Lc,
            rng,
            max_iters=300_000,
        )
        print(f"[Targeted‑local] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after local search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Mixture line‑search among top candidates.
    # ------------------------------------------------------------------
    # Gather a handful of promising vectors.
    top_vecs = [best_a]
    top_scores = [best_score]
    for _ in range(5):
        cand = random_initial(n, rng)
        sc = fast_score(cand, Lc, n)
        top_vecs.append(cand)
        top_scores.append(sc)

    # keep the best K = 5 vectors
    sorted_pairs = sorted(zip(top_scores, top_vecs), key=lambda x: x[0])
    K = 5
    sel_scores, sel_vectors = zip(*sorted_pairs[:K])

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(sel_vectors),
            list(sel_scores),
            n,
            Lc,
            deadline,
        )
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
