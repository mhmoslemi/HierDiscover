Implementation of coordinate gradient descent:

```
def gradient_descent_mass_transfer(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute gradient g about objective (max conv)
        # compute conv and mask
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # grad (positive)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]  # length n

        # identify donors and receivers based on g
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        if g[i_max] <= g[i_min] + 1e-12:
            break # no improvement possible
        # step size: proportion of a[i_max]
        step = min(a[i_max] * 0.5, 0.001) # limited to 0.1%? tune.

        # transfer step mass
        a[i_max] -= step
        a[i_min] += step

        # minor numeric adjustments: ensure nonnegative
        if a[i_max] < 0: a[i_max] = 0.0
        # renormalize sum to 1 (should stay sum=1)
        # but due to rounding errors may deviate; we can renorm
        a /= a.sum()

        # compute score
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score
```

def alternating_projection_init(n, rng, iterations=50):
    # start with random a on simplex
    a = np.random.dirichlet(np.ones(n))
    L = next_pow2(2*n-1)  # for convolution later but here we need FFT length for a maybe same L
    for _ in range(iterations):
        # zero pad to L
        a_padded = np.zeros(L)
        a_padded[:n] = a
        # FFT
        F = np.fft.fft(a_padded)
        # get phases
        phi = np.angle(F)
        # set magnitude to 1 (or constant)
        F_new = np.exp(1j * phi)  # magnitude 1
        # Inverse FFT to get real values (per possibly complex)
        a_new_full = np.real(np.fft.ifft(F_new))
        # Take first n entries as new a
        a = np.maximum(a_new_full[:n], 0.0)
        # project onto simplex
        a = proj_simplex(a)
    return a
```

"""
Propose a low‑score step function.

The algorithm follows a multi‑phase strategy:

1. Build a diverse pool of starting vectors (uniform, random, exponential,
   spectral, and a custom alternating‑projection initializer).  The best of
   these is kept as the current incumbent.

2. Run many Adam restarts on a smooth log‑sum‑exp surrogate of the
   max‑convolution objective.  The smoothing parameter (lam) is gradually
   reduced and the number of optimisation steps per lam is increased
   compared with the previous baseline.  This phase dominates the budget.

3. Refine the incumbent with three deterministic descent methods:
   * Exponentiated‑gradient descent on the true max‑convolution.
   * Euclidean projected sub‑gradient descent.
   * A simple coordinate‑descent that transfers mass from the index with the
     largest sub‑gradient to the index with the smallest sub‑gradient.

4. Perform a cheap simulated‑annealing local search to escape possible
   shallow minima.

5. Optionally mix the best few candidates with a golden‑section line search.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0).  The best
vector is stored in the global name ``height_sequence_1`` for reuse in later
runs.

The search works with a large problem size (n = 8000) which provides more
degrees of freedom and typically yields a lower bound than the previous
n = 4000 experiments, while still fitting comfortably in the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective = 2·n·max(a*a) assuming ∑a = 1.
    This matches `evaluate_sequence` when the sequence is normalised.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return float(2.0 * n * conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def alternating_proj_init(n: int, rng: np.random.Generator,
                         iters: int = 70) -> np.ndarray:
    """
    Alternating projection between the simplex (non‑negative, sum‑1) and the
    set of vectors whose FFT has constant magnitude (i.e. unit magnitude).
    This often yields a diverse, low‑convolution candidate.
    """
    L = next_pow2(2 * n - 1)
    # start from a Dirichlet draw (uniform on simplex)
    a = rng.random(n)
    a = proj_simplex(a)
    for _ in range(iters):
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        # keep the phase, force magnitude = 1 (except DC component)
        phi = np.angle(F)
        F_new = np.exp(1j * phi)   # magnitude 1
        a_full = np.real(np.fft.ifft(F_new))
        a = np.maximum(a_full[:n], 0.0)
        a = proj_simplex(a)
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          init_scale: float = 12.0,
                          scale_decay: float = 0.99):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    scale = init_scale
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        # guard against overflow
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        scale *= scale_decay
        if scale < 0.1:
            scale = 0.1
    return best_a, best_score

def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              init_eta: float = 0.7):
    """
    Euclidean projected sub‑gradient descent on max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple coordinate‐descent mass‑transfer optimiser
# ----------------------------------------------------------------------


def coordinate_descent_optimize(a_start: np.ndarray,
                                deadline: float,
                                n: int,
                                fft_len: int,
                                rng: np.random.Generator,
                                max_iter: int = 500000,
                                eps_factor: float = 0.06):
    """
    Repeatedly move a small fraction of mass from the index with the
    largest sub‑gradient to the index with the smallest sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        donor = np.argmax(g)
        receiver = np.argmin(g)

        if donor == receiver:
            break

        # amount to transfer – a small fraction of the donor's mass
        step = min(a[donor] * eps_factor, a[donor] * 0.5)
        if step <= 0.0:
            break

        a[donor] -= step
        a[receiver] += step

        # re‑normalise to guard against numerical drift
        a /= a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 2e-3,
                               final_temp: float = 1e-8,
                               max_iter: int = 10_000_000):
    """
    Random mass‑transfer moves with a geometric temperature decay.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        temp = max(final_temp, init_temp * (0.9995 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.07)  # slightly larger moves
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    Golden‑section line search over convex mixtures of each unordered pair.
    Returns (best_a, best_score).
    """
    best_a = candidates[0].copy()
    best_score = scores[0]

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            lo, hi = 0.0, 1.0
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

            while time.time() < deadline and (hi - lo) > 1e-4:
                if fc < fd:
                    hi, fd = d, fc
                    d = c
                    c = hi - invphi * (hi - lo)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, fft_len, n)
                else:
                    lo, fc = c, fd
                    c = d
                    d = lo + invphi * (hi - lo)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, fft_len, n)

            tm = 0.5 * (lo + hi)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            ftm = fast_score(mix_tm, fft_len, n)

            if ftm < best_score:
                best_score = ftm
                best_a = mix_tm.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------------
    # Problem size – a larger n gives more freedom and a lower theoretical bound.
    # ---------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ---------------------------------------------------------------
    # Initialise candidates (interpolating any previous result if present)
    # ---------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.maximum(a_prev, 0.0)
            a_prev /= a_prev.sum()
            candidates.append(a_prev)

    candidates.extend([
        spectral_init(n),
        random_initial(n, rng),
        add_noise(np.full(n, 1.0 / n), rng.uniform(0.001, 0.01), rng),
        alternating_proj_init(n, rng)
    ])

    # ---------------------------------------------------------------
    # Pick the best of the initial pool
    # ---------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for cand in candidates:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ---------------------------------------------------------------
    # Phase 1 – Adam restarts on smoothed surrogate
    # ---------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 300   # more steps per lam than the baseline

    for r in range(12):
        if time.time() >= deadline:
            break
        a0 = best_a.copy() if r == 0 else add_noise(best_a,
                                                    rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {r + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (true max)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_scale=20.0,
            scale_decay=0.995,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgrad_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            init_eta=1.0,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 4 – Simple coordinate‑descent mass‑transfer optimiser
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = coordinate_descent_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            max_iter=600000,
            eps_factor=0.07,
        )
        print(f"[Coord‑descent] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after coordinate descent: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 5 – Simulated annealing
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_search(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_temp=2e-3,
            final_temp=1e-8,
            max_iter=8_000_000,
        )
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after anneal: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 6 – Convex mixture search among a few promising vectors
    # ---------------------------------------------------------------
    # prepare a small set of promising vectors
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(5):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the best K of them
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    sel_scores, sel_vectors = zip(*ordered[:K])

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(sel_vectors),
            list(sel_scores),
            n,
            fft_len_conv,
            deadline,
        )
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ---------------------------------------------------------------
    # Final evaluation using the provided security‑checked routine
    # ---------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
