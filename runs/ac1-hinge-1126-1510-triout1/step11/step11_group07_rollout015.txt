def smooth_gradient_optimize(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_conv = conv.max()
        # Temperature T schedule
        T = max_conv * max(0.02, 0.4 / np.sqrt(it + 1.0))
        # Compute softmax weights
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.zeros_like(conv)
            w[n-1] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # compute gradient
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * grad_conv[n-1:n-1+n]

        # learning rate
        eta = 30.0 / np.sqrt(it + 1.0)  # similar to exp gradient
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

propose_candidate
-----------------
A multi‑phase optimizer for the scale‑invariant step‑function problem
---------------------------------------------------------------

The objective   2·n·max(conv(a,a)) / (sum(a)**2)  is unchanged under scaling,
so we work with a probability simplex (sum = 1).  The algorithm consists
of several fast stages that respect the 1000 s budget:

1. **Diverse starts** – uniform, arcsine‑shaped, random, and the previous best
   (if supplied).  The cheapest FFT‑based `fast_score` picks the best seed.

2. **Smoothed‑max gradient descent** – we minimise a temperature‑smoothed
   version of the max‑convolution:
       F_T(a) = T·log∑_k exp(conv_k / T)
   The gradient is obtained via a single FFT‑convolution with the soft‑max
   weights.  A multiplicative (exponentiated) update together with a
   decreasing learning‑rate and a temperature schedule yields very fine
   improvements near the optimum.

3. **Plain exponentiated‑gradient** – identical to the original implementation

4. **Projected sub‑gradient descent** – a classic Euclidean‑projection onto
   the simplex refines the solution once the active set of near‑maximal

5. **Greedy hill‑climb** – exact convolution updates move a tiny fraction of

   steps.  This final polish often yields the last few hundredths of a point.

All stages respect the time limit; the best sequence found is returned as a

    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # decreasing learning rate
        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (new component)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(a_start: np.ndarray,
                             deadline: float,
                             n: int,
                             fft_len: int,
                             rng: np.random.Generator,
                             eta0: float = 30.0,
                             temp_factor: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a temperature‑smoothed maximum:
        F_T(a) = T * log(∑_k exp(conv_k / T))

    The gradient is 2 * (a_rev convolved with the soft‑max weights).
    A multiplicative update with a decreasing step size is used, followed
    by a projection onto the simplex (normalisation).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # temperature schedule – starts relatively warm and cools down
        T = max_val * max(temp_factor / np.sqrt(it + 1.0), 1e-8)

        # soft‑max weights (stable formulation)
        w = np.exp((conv - max_val) / T)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to a hard mask if numerical issues appear
            eps = max_val * 1e-12 + 1e-12
            w = (conv >= max_val - eps).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient = 2 * (a_rev * w)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 is a good compromise between quality and speed
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1. Build a small collection of diverse starting points
    # --------------------------------------------------------------
    init_candidates = [
        np.full(n, 1.0 / n),          # uniform
        arcsine_initial(n),           # deterministic arcsine
        random_initial(n, rng)        # random
    ]

    # include previous best if present
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            if a_old.sum() == 0.0:
                a_old = np.full(n, 1.0 / n)
            else:
                a_old = a_old / a_old.sum()
            init_candidates.append(a_old)

    # keep the best primitive (cheap evaluation)
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # 2. Smoothed‑max gradient descent (≈ 40 % of remaining time)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.4 * (deadline - time.time())
    best_a, best_score = smooth_gradient_optimize(
        best_a, phase1_deadline, n, fft_len, rng, eta0=30.0, temp_factor=0.2
    )
    print(f"[smooth‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient descent (≈ 20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.2 * (deadline - time.time())
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_deadline, n, fft_len, rng, eta0=30.0
        )
        print(f"[exp‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent (≈ 20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + 0.2 * (deadline - time.time())
        best_a, best_score = projected_subgradient_optimize(
            best_a, phase3_deadline, n, fft_len, rng,
            eps_frac=0.001, step0=0.2
        )
        print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final greedy hill‑climb (use any time that remains)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
