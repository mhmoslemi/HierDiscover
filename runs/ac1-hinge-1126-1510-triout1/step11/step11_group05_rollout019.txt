def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    fa = np.fft.fft(a, fft_len)

"""
Propose a low‑score step‑function sequence.

The optimizer uses three complementary strategies:

1.  A large pool of diverse initial vectors (uniform, random, power‑law,
    several exponential shapes, and noisy copies of the current best).

2.  A long Adam optimisation on a smooth log‑sum‑exp surrogate of the
    max‑convolution objective.  The smoothing parameter (``lam``) is
    reduced gradually and many optimisation steps are taken per value.

3.  A final exponentiated‑gradient (mirror‑descent) phase on the *exact*
    max‑convolution objective, with many thousands of updates and a
    decreasing learning‑rate schedule.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0).  The best
vector found is stored globally as ``height_sequence_1`` for reuse.
"""


import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective = 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return float(2.0 * n * conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Random / parametric initialisers
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


def power_init(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def exponential_init(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i), normalised to sum 1."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = uniform_init(n)
    else:
        a /= a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          init_scale: float = 12.0,
                          max_iter: int = 30000):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # forward convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient w.r.t. a (sub‑gradient of the max)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # decreasing step‑size
        eta = init_scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              init_eta: float = 0.7,
                              max_iter: int = 20000):
    """
    Euclidean projected sub‑gradient descent on max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local search (light version)
# ----------------------------------------------------------------------


def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 5e-3,
                               final_temp: float = 1e-7,
                               max_iter: int = 200_000):
    """
    Simple mass‑transfer annealing; cheap but can still escape shallow
    local minima.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        temp = max(final_temp, init_temp * (0.9999 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score or (temp > 0.0 and
                                     rng.random() < np.exp(-(new_score - cur_score) / temp)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand coefficients give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # length for a*a
    fft_len_g = next_pow2(3 * n - 2)          # length needed for the gradient step

    # ------------------------------------------------------------------
    # Initialise from previous global result if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size == n:
            a_start = prev.copy()
        else:
            # interpolate previous solution to current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
    else:
        a_start = uniform_init(n)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # ------------------------------------------------------------------
    candidates = [
        a_start,
        uniform_init(n),
        random_init(n, rng),
        power_init(n, rng),
    ]

    # add a few exponentials with different rates (including very small ones)
    for lam in [0.0002, 0.0003, 0.0004, 0.0005, 0.0006, 0.0010]:
        candidates.append(exponential_init(n, lam))

    # add a noisy copy of the current best
    candidates.append(add_noise(a_start, rng.uniform(0.001, 0.01), rng))

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate (many restarts, many steps)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 500          # a bit more than the original code

    # a few more restarts than before
    for r in range(12):
        if time.time() >= deadline:
            break
        # choose a starting point: either the overall best or a noisy variant
        a0 = best_a.copy() if r == 0 else add_noise(best_a,
                                                    rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {r + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (many iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_scale=10.0,
            max_iter=30000,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgrad_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            init_eta=0.8,
            max_iter=20000,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Light simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_search(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_temp=5e-3,
            final_temp=1e-7,
            max_iter=200_000,
        )
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after anneal: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Optional mixture search among a few top vectors
    # ------------------------------------------------------------------
    # gather a handful of promising vectors (current best + a few random)
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_init(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the K best
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    sel_scores, sel_vectors = zip(*selected)

    if time.time() < deadline:
        # simple convex combination of the two best vectors
        a1, a2 = sel_vectors[0], sel_vectors[1]
        lo, hi = 0.0, 1.0
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        invphi = (np.sqrt(5.0) - 1.0) / 2.0
        while time.time() < deadline and (hi - lo) > 1e-4:
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a1 + c * a2
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len_conv, n)

            mix_d = (1.0 - d) * a1 + d * a2
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len_conv, n)

            if fc < fd:
                hi, fd = d, fc
                d = c
                c = hi - invphi * (hi - lo)
            else:
                lo, fc = c, fd
                c = d
                d = lo + invphi * (hi - lo)

        t_opt = 0.5 * (lo + hi)
        a_mix = (1.0 - t_opt) * a1 + t_opt * a2
        a_mix /= a_mix.sum()
        fm = fast_score(a_mix, fft_len_conv, n)
        print(f"[Mixture] score={fm:.6f}")
        if fm < best_score:
            best_score, best_a = fm, a_mix.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
