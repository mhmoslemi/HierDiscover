"""
Propose a low‑evaluation step‑function.

The algorithm explores a wide range of initial shapes (uniform,
arcsine‑shaped, power‑law, exponential) and keeps the best by the cheap
`fast_score = 2·n·max(conv)`.  Then a long run of exponentiated‑gradient
descent (multiplicative mirror‑descent on the simplex) followed by
projected sub‑gradient descent refines the candidate.  Finally a small
simulated‑annealing walk and a greedy hill‑climb polish the result.

A larger problem size (n = 8000) reduces the discretisation error of the
continuous optimum, allowing the final score to approach the theoretical
limit 2·sup(f∗f) ≈ 1.5.  The whole search respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def power_initial(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p/2}  (symmetric power‑law, p≥0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = (i * (n + 1 - i)) ** (-p / 2.0)
    a = w / w.sum()
    return a


def exponential_initial(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    a = w / w.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = power_initial(n, p)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          eps_frac: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the max‑convolution.
    The active mask contains all indices that are within eps_frac·max of the
    current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * eps_frac + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        grad_conv = np.real(np.fft.ifft(np.fft.fft(a_rev, fft_len) *
                                        np.fft.fft(mask, fft_len)))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)  # multiplicative update

        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = best_a.copy()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set contains all indices within `eps_frac` of the current
    maximum of the convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        grad_conv = np.real(np.fft.ifft(np.fft.fft(a_rev, fft_len) *
                                        np.fft.fft(mask, fft_len)))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        # simple back‑tracking line‑search
        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)

            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()

            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # small random jitter to escape plateaus
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Moves a random fraction of mass from a source index to a destination.
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12

    while time.time() < deadline:
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small fraction of mass from a heavier
    index to a lighter one, accepting only moves that strictly improve the score.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution (only the linear contribution)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """

    # Problem size – 8000 gives a finer discretisation of the continuous problem.
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1. Build a diverse collection of starting points
    # --------------------------------------------------------------
    init_candidates = []

    # a) deterministic baselines
    init_candidates.append(uniform_initial(n))
    init_candidates.append(arcsine_initial(n))

    # b) a sweep over power‑law exponents
    for p in np.linspace(0.6, 1.8, 7):          # 0.6, 0.8, … ,1.8
        init_candidates.append(power_initial(n, p))

    # c) exponential decays
    for lam in np.linspace(0.002, 0.018, 5):
        init_candidates.append(exponential_initial(n, lam))

    init_candidates.append(random_initial(n, rng))

    # e) warm‑start from a previous run, if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # interpolate to the new size
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            init_candidates.append(a_old)

    # Keep the best primitive by cheap evaluation
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient descent (≈ 55 % of remaining time)

    phase1_deadline = time.time() + 0.55 * (deadline - time.time())
    best_a, best_score = exp_gradient_optimize(
        best_a, phase1_deadline, n, fft_len, rng,

    )
    print(f"[exp‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent (≈ 30 % of remaining time)
    # --------------------------------------------------------------
    phase2_deadline = time.time() + 0.30 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.2
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Global simulated annealing (remaining time – a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline - 5.0, rng
        )
        print(f"[SA] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )

    # --------------------------------------------------------------
    # 7. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
