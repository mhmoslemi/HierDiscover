"""
Low‑evaluation step‑function search.

The algorithm proceeds in three stages:

1. **Diverse initialisation** – several families of distributions are
   generated (uniform, arcsine, Beta(p,p) with p<1, edge‑heavy, random)
   and the cheapest one (according to the cheap `fast_score`) is kept.

2. **Smooth exponential‑gradient descent** – we minimise the
   smooth approximation of the max‑convolution

        S_β(a) = (1/β)·log Σ_i exp(β·(a*a)[i])

   using mirror descent on the probability simplex.
   The temperature β is increased linearly from 1 to 200,
   the learning rate η decays as 1/√t, and after every update we
   enforce symmetry a_i = a_{n‑1‑i}.  The gradient of S_β is
   computed exactly with two FFTs.

3. **Refinement** – a short projected sub‑gradient phase, a
   simulated‑annealing walk and a greedy hill‑climb are performed,
   all using the cheap `fast_score`.  Finally the harness evaluates the
   result with `evaluate_sequence`.

All phases respect a total budget of 1000 s and the code prints
intermediate scores so the progress can be monitored.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diverse random point on the simplex."""
    a = rng.random(n)
    a /= a.sum()
    return a


def beta_initial(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta(p,p) density on the discrete grid.
    p < 1 → heavier edges, p = 0.5 is the arcsine law.
    """
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – xs is never 0 or 1 because of the 0.5 offset
    log_a = (p - 1.0) * (np.log(xs) + np.log(1.0 - xs))
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def edge_initial(n: int, edge_frac: float, ratio: float) -> np.ndarray:
    """
    Heavy mass on the two edge blocks.
    edge_frac : fraction of indices belonging to each edge block.
    ratio     : weight of an edge index relative to a centre index.
    """
    e = max(1, int(edge_frac * n))
    # interior weight L, edge weight H = ratio·L
    L = 1.0 / (2 * e * ratio + (n - 2 * e))
    H = ratio * L
    a = np.full(n, L, dtype=np.float64)
    a[:e] = H
    a[-e:] = H
    return a


# ----------------------------------------------------------------------
# Smooth exponential‑gradient descent on the soft‑max of convolution
# ----------------------------------------------------------------------


def smooth_exp_gradient_optimize(a_start: np.ndarray,
                                 start_time: float,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 eta0: float = 5.0,
                                 beta_start: float = 1.0,
                                 beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the smoothed max‑convolution.
    The temperature β grows linearly from beta_start to beta_end.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    total_time = deadline - start_time

    while time.time() < deadline:
        # ----- convolution -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()

        # ----- temperature schedule -----
        elapsed = time.time() - start_time
        fraction = elapsed / total_time
        beta = beta_start + (beta_end - beta_start) * fraction

        # ----- soft‑max weights -----
        shifted = conv - max_val                     # ≤ 0, max at 0
        w = np.exp(beta * shifted)                  # w_i = exp(β·(c_i‑max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max -----
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # ----- mirror‑descent update (exponential) -----
        eta = eta0 / math.sqrt(it + 1.0)
        # clip to avoid overflow inside exp
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term

        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
            s = a.sum()
        a /= s

        # ----- enforce symmetry (helps stability) -----
        a = (a + a[::-1]) * 0.5

        # ----- bookkeeping -----
        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Projected sub‑gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # define active mask (slightly relaxed)
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        # sub‑gradient of the (non‑smoothed) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step size schedule (diminishing)
        step = step0 / math.sqrt(it + 1.0)

        # back‑tracking line search (max 5 trials)
        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # tiny random jitter to escape plateaus
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk on the simplex."""
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12  # initial temperature

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points give a good trade‑off between quality
    # and speed.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of candidate initialisations
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(uniform_initial(n))

    # arcsine
    init_candidates.append(arcsine_initial(n))

    # random
    init_candidates.append(random_initial(n, rng))

    # Beta‑family (p in (0.2, 0.9))
    for p in np.linspace(0.2, 0.9, 8):
        init_candidates.append(beta_initial(n, p))

    # Edge‑heavy families
    for edge_frac in [0.006, 0.012, 0.018, 0.024, 0.030]:
        for ratio in [5, 10, 20, 50, 100]:
            init_candidates.append(edge_initial(n, edge_frac, ratio))

    # previous best (if supplied by the harness)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            init_candidates.append(a_old)

    # ------------------------------------------------------------------
    # 2. Keep the cheapest candidate according to the cheap score
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # 4. Smooth exponential‑gradient descent (≈ 45 % of the time)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.45 * (deadline - time.time())
    best_a, best_score = smooth_exp_gradient_optimize(
        best_a, start_time, phase1_deadline, n, fft_len, rng,
        eta0=5.0, beta_start=1.0, beta_end=200.0
    )
    print(f"[smooth‑exp‑grad] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (≈ 30 % of the remaining time)
    # ------------------------------------------------------------------
    phase2_deadline = time.time() + 0.30 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.2
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Global simulated annealing (use whatever time is left – a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline - 5.0, rng
        )
        print(f"[SA] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation (the harness will also call this)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
