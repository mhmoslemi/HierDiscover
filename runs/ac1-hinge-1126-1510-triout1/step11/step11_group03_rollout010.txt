    n = len(a)

    # Use scipy.optimize.linprog where A_ub is A (dense? It expects dense or sparse)

"""
Propose a non‑negative step‑function with a low evaluation score.

Algorithm
---------
1.  Build a diverse pool of initial candidates (random families,
    classical windows, Gaussian / cosine‑power / edge‑heavy parametrisations).
2.  Aggressive L‑BFGS optimisation of a log‑sum‑exp smooth surrogate
    for the L∞‐convolution objective (λ‑schedule decreasing to 1e‑5).
3.  Long exponentiated‑gradient descent on the exact max‑convolution
    (sub‑gradient, 1/√t step‑size).
4.  Fine‑tuned L‑BFGS with a very low λ.
5.  Linear‑programming refinement: for the current vector *f* solve

       max  Σ b_j
       s.t. f * b  ≤  M    (pointwise),   b ≥ 0

    where M = max (f * f).  The optimiser returns a vector *g* that can be
    mixed with *f* (small convex combination) to strictly lower the true
    score.  Only a few LP solves are needed, so the cost is modest.
6.  Short simulated‑annealing polish.
7.  Final evaluation with the official ``evaluate_sequence`` function.

The whole pipeline respects the 1000 s budget and stores the best
sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)).
    Assumes ``a`` sums to 1 (probability simplex).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    ``y`` lives in ℝⁿ; ``a = softmax(y)`` belongs to the probability simplex.
    ``lam`` is the smoothing factor (relative to the current max value).
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)     # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f_val, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
      * mass transfer between two arbitrary positions
      * swapping two entries
      * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:
                j = rng.integers(0, n)
            if i == j:
                continue

            # move mass from larger to smaller entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def _conv_matrix_template(n: int):
    """Pre‑compute the row/col indices for the convolution matrix (size (2n‑1)×n)."""
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=np.int32) for j in range(n)])
    shape = (2 * n - 1, n)
    return rows, cols, shape


def solve_lp_for_f(f: np.ndarray, M: float, rows, cols, shape):
    """
    Linear program:
        maximise    Σ b_j
        subject to  (f * b)_k ≤ M   for all k,
                    b_j ≥ 0.
    Returns the optimal vector ``b`` (may be all‑zero if LP fails).
    """
    n = f.shape[0]
    data = np.tile(f, n)
    A = sparse.csr_matrix((data, (rows, cols)), shape=shape)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise –Σb ⇒ maximise Σb
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None


def lp_refinement_stage(a: np.ndarray,
                        cur_score: float,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """
    Repeatedly solve the LP described above and mix the solution with the
    current vector.  Mixing is performed with a few preset coefficients;
    the best improvement (if any) is kept.
    """
    rows, cols, shape = _conv_matrix_template(n)
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline - 2.0:   # keep a tiny buffer for final steps
        M = fast_score(best_a, fft_len) / (2.0 * n)

        b = solve_lp_for_f(best_a, M, rows, cols, shape)
        if b is None or b.sum() == 0.0:
            break

        b_norm = b / b.sum()

        improved = False
        # try a few mixing coefficients – larger ones first
        for t in (0.5, 0.3, 0.1):
            cand = (1.0 - t) * best_a + t * b_norm
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            score = fast_score(cand, fft_len)
            if score < best_score:
                best_score = score
                best_a = cand.copy()
                improved = True
                print(f"[LP‑mix] new best fast_score = {best_score:.9f}  t={t:.2f}")
                break
        if not improved:
            # No improvement – stop the LP refinement loop
            break

    return best_a, best_score
