"""
step‑function optimiser – 1000 s budget (enhanced version)

Main ideas
-----------
* Work on a *probability vector* (∑a_i = 1).  For such vectors the official
  score equals ``2·n·max(conv(a,a))`` (the ‘fast_score’), so we optimise
  the max‑convolution directly.
* Use a *larger problem size* (n≈8000).  The theoretical lower bound
  2·n·(1/(2n‑1)) ≈ 1 is approached more closely when more degrees of
  freedom are available.
* Three‑phase strategy
  1. **Diverse warm‑starts + a cascade of L‑BFGS** on a smoothed
     log‑sum‑exp surrogate.  The λ‑schedule goes down to 1e‑6, far
     smaller than in the reference code.
  2. **Long exponentiated‑gradient (mirror‑SGD)** on the true L∞‑objective.
     The step‑size decays as η₀/√t but we restart it once with a
     smaller η₀ to escape plateaus.
  3. **Fine L‑BFGS polish + a short simulated‑annealing hill‑climb**.
* All phases respect the 1000 s deadline.  The best vector found is stored
  in the global variable ``height_sequence_1`` and returned as a plain
  list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the official score when ``sum(a)=1``:
        2·n·max(conv(a,a))
    Convolution is performed by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(
        ['uniform',
         'random',
         'exp',
         'power',
         'edge',
         'bimodal',
         'sparse',
         'decreasing',
         'increasing']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'decreasing':
        # sort a random vector in decreasing order – strong left‑heavy shape
        a = np.sort(rng.random(n))[::-1]
    else:  # increasing
        a = np.sort(rng.random(n))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smooth max‑convolution value and its gradient w.r.t.
    the log‑variables y (a = softmax(y)).
    """
    # --- softmax to ensure a lives on the simplex ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution via FFT ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # Effective temperature – scaling by max_c keeps the smoothing
    # relative to the magnitude of the function.
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c                # ≤0
    exp_shift = np.exp(shifted / lam_eff)      # stabilised soft‑max weights
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z               # d smooth_max / d conv_i
    # back‑prop through the convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that corresponds to gradient w.r.t. a
    g_vec = conv_rev[n - 1: n - 1 + n]   # length n

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        # recover probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0,
                          max_iters: int = 10_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution.
    The update stays on the simplex via multiplicative weights.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # --- exact convolution and mask of (almost) maximal entries ---
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # periodic cheap score update
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                # occasional log
                if (it % 1024) == 0:
                    print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20        # maximal fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    # Use a size that comfortably fits into the time budget.
    n = 8000                     # feel free to adjust (≈2× the previous size)
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the current length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Randomly generated families (including monotone shapes)
    for _ in range(16):
        pool.append(random_initial(n, rng))

    # 3) Classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.15 * budget_s       # L‑BFGS warm‑up
    phase2_end = start_time + 0.80 * budget_s       # long EG phase
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – cascade of L‑BFGS on a smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.15, 0.10, 0.07,
        0.04, 0.025, 0.015, 0.008, 0.0045,
        0.0025, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001, 5e-6, 1e-6
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter to increase diversity
        if rng.random() < 0.30:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    # We run a first EG pass, then restart once with a smaller η₀.
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=20.0,
            max_iters=20_000_000
        )
        print(f"[Phase‑2‑EG1] fast_score={best_score:.9f}")

        # second pass with a reduced learning‑rate to refine
        if time.time() < phase2_end:
            best_a, best_score = exp_gradient_optimize(
                best_a,
                phase2_end,
                n,
                fft_len,
                rng,
                eta0=6.0,
                max_iters=12_000_000
            )
            print(f"[Phase‑2‑EG2] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS polish (very low λ) + simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-6, 3e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase‑3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Phase‑3‑Anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final normalisation & official evaluation
    # ------------------------------------------------------------------
    # Protect against any accidental numerical drift.
    s = best_a.sum()
    if s > 0:
        best_a = best_a / s

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # the official scorer
    print(f"[final] evaluated score = {final_eval:.9f}")

    # store for possible reuse in a later run
    globals()['height_sequence_1'] = final_seq
    return final_seq
