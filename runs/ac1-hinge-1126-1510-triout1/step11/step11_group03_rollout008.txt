def differential_evolution_optimize(pop, deadline, n, fft_len, rng):
    best_idx = np.argmin([fast_score(p, fft_len) for p in pop])
    best_score = fast_score(pop[best_idx], fft_len)
    best = pop[best_idx].copy()
    CR = 0.7
    F = 0.8
    while time.time() < deadline:
        for i in range(len(pop)):
            indices = rng.choice([j for j in range(len(pop)) if j != i], 3, replace=False)
            a,b,c = pop[indices[0]], pop[indices[1]], pop[indices[2]]
            mutant = a + F * (b - c)
            mutant = np.maximum(mutant, 0.0)
            mutant_sum = mutant.sum()
            if mutant_sum == 0:
                mutant = pop[i].copy()
            else:
                mutant = mutant / mutant_sum
            # crossover
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(n)] = True
            trial[mask] = mutant[mask]
            # renormalize
            trial = np.maximum(trial,0.0)
            s= trial.sum()

                trial = pop[i].copy()

                trial = trial / s

                pop[i] = trial
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        # optional early exit if improvement not found after many loops?
    return best, best_score

"""
Propose a step‑function that lowers the 2·n·max(conv)/sum² score.
Algorithm (≈1000 s budget)

1. Initialise a diverse population (random windows, Gaussian, cosine‑power,
   edge‑heavy, uniform, plus the previous best if any).  All vectors lie on
   the probability simplex (sum = 1).

2. Aggressive L‑BFGS on a log‑sum‑exp smooth approximation of the L∞‑objective.
   The λ‑schedule goes down to 1e‑7, producing a strong local optimum.

3. Long mirrored exponentiated‑gradient (multiplicative) descent on the
   exact max‑convolution (sub‑gradient).  Step‑size η₀ decays as 1/√t.

4. Differential‑Evolution (DE) population search.  DE explores the space
   globally while keeping each candidate on the simplex.  After each successful
   trial we apply a few EG steps to fine‑tune locally.

5. Fine L‑BFGS with an ultra‑low λ‑schedule (1e‑7 → 1e‑9) followed by a brief
   Metropolis anneal that swaps/ transfers mass.

6. Return the best sequence found and store it in the global variable
   `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Score 2·n·max(conv(a,a)) assuming `a` already normalised to sum = 1.
    Uses FFT for O(n log n) speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add isotropic Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution using log‑sum‑exp.
    y lives in ℝⁿ; a = softmax(y) (∑a=1, a≥0).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)      # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                          # d smooth_max / d conv_i

    # back‑prop through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]         # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best vector (∑=1) and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑space representation
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Exponentiated‑gradient (multiplicative) descent using the exact sub‑gradient
    of max(conv).  Stops when `deadline` is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Differential Evolution (global population search)
# ----------------------------------------------------------------------


def differential_evolution_optimize(pop: list[np.ndarray],
                                    deadline: float,
                                    n: int,
                                    fft_len: int,
                                    rng: np.random.Generator,
                                    F: float = 0.8,
                                    CR: float = 0.7,
                                    inner_eg_iters: int = 5):
    """
    Classic DE with simplex projection.  After a successful trial we perform
    a few cheap EG steps to polish locally.
    Returns the best individual and its fast_score.
    """
    # store current scores to avoid recomputation
    scores = np.array([fast_score(ind, fft_len) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(pop)):
            # mutation
            idxs = rng.choice([j for j in range(len(pop)) if j != i], 3, replace=False)
            a1, a2, a3 = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a1 + F * (a2 - a3)
            mutant = np.maximum(mutant, 0.0)
            s = mutant.sum()
            if s == 0.0:
                mutant = pop[i].copy()
            else:
                mutant = mutant / s

            # crossover
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial[mask] = mutant[mask]
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s == 0.0:
                trial = pop[i].copy()
            else:
                trial = trial / s

            # optional tiny EG polish
            for _ in range(inner_eg_iters):
                # one EG step (very cheap)
                fa = np.fft.fft(trial, fft_len)
                conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
                max_val = conv.max()
                eps = max_val * 1e-12 + 1e-12
                mask2 = (conv >= max_val - eps).astype(np.float64)
                a_rev = trial[::-1]
                fm = np.fft.fft(mask2, fft_len)
                fa_rev = np.fft.fft(a_rev, fft_len)
                grad_conv = np.real(np.fft.ifft(fm * fa_rev))
                g = 2.0 * grad_conv[n - 1: n - 1 + n]
                eta = 10.0 / np.sqrt(_ + 1.0)
                exp_term = np.exp(np.clip(-eta * g, -40, 40))
                trial = trial * exp_term
                s = trial.sum()
                if s == 0.0:
                    break
                trial = trial / s

            trial_score = fast_score(trial, fft_len)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
        # optional early‑stop when improvement stalls
        # (omitted to keep the loop simple)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple Metropolis annealer (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """
    Very short Metropolis annealer that moves a fraction of mass between
    two positions or swaps entries.  Temperature linearly cools.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])
        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:
                j = rng.integers(0, n)
            if i == j:
                continue
            # ensure we move from the larger entry
            src, dst = (i, j) if a[i] > a[j] else (j, i)
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True
        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------
    n = 8000                           # feel free to adjust if time permits
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial diverse pool (size ≈ 30)
    # --------------------------------------------------------------
    pool = []

    # 0) previously discovered best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) classic windows (if SciPy signal is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # 2) random families (multiple draws)
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # 3) parametric families
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # ensure we have at least a minimal size
    if not pool:
        pool.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05           # safety margin
    phase1_end = start_time + 0.20 * budget_s        # aggressive L‑BFGS
    phase2_end = start_time + 0.55 * budget_s        # long EG descent
    phase3_end = start_time + 0.85 * budget_s        # DE global search

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 800
    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.3:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=30.0
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Differential Evolution (global exploration)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        # initialise a small population around the current best
        pop = [best_a.copy()]
        while len(pop) < 20:
            cand = add_noise(best_a, sigma=rng.uniform(0.001, 0.01), rng=rng)
            pop.append(cand)
        # fill up to 30 individuals with random pool members
        while len(pop) < 30:
            pop.append(rng.choice(pool))

        best_a, best_score = differential_evolution_optimize(
            pop,
            phase3_end,
            n,
            fft_len,
            rng,
            F=0.7,
            CR=0.75,
            inner_eg_iters=3
        )
        print(f"[DE] best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – final fine‑L‑BFGS + short annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        fine_lam_sched = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1500,
                                        deadline=deadline - 20.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final official evaluation and store result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)      # provided by the platform
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
