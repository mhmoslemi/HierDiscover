"""
optimise_step_function.py

Search for a non‑negative sequence a_0,…,a_{n‑1} that minimises

    score = 2·n·max_k (a * a)_k               (sum a = 1)

The algorithm works in several stages:

1. Choose a moderately large problem size (n = 8000).  A larger n gives a
   lower theoretical lower‑bound (≈1 + 1/(2n‑1)) while keeping the FFT
   cost modest.

2. Initialise a from a previously stored solution (if any) by linear
   interpolation, otherwise from a diverse set of random/simple shapes.

3. Run many short L‑BFGS optimisations on a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing parameter is
   gradually reduced, giving increasingly accurate descent directions.

4. Refine the best point with Adam (again on the smooth objective).

5. Perform a large number of exponentiated‑gradient steps on the *true*
   max‑convolution (sub‑gradient given by the mask of maximal entries).

6. A final “guided hill‑climb” repeatedly moves a small fraction of mass
   from the index with the largest sub‑gradient to the index with the
   smallest, accepting the move only when the max‑convolution does not
   increase.  The step‑size decays linearly with the remaining time and
   the exact convolution is recomputed periodically.

All operations use FFT‑based convolutions (real‑valued rFFT) for speed.
The total wall‑clock budget is enforced (default 1000 s).  The returned
list of floats is directly fed to the provided `evaluate_sequence`
function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns the first 2*len(a)-1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the objective for a probability vector a
    (sum a == 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random probability distribution on n points.
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'cosine', 'triangular']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'cosine':
        x = np.arange(n) / (n - 1)
        a = 1.0 + np.cos(2 * np.pi * x)  # positive values
    else:  # triangular
        x = np.arange(n) / (n - 1)
        a = 1.0 - np.abs(x - 0.5) * 2.0

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def interpolate_to_length(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation (or down‑sampling) of a 1‑D array to length new_n.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total == 0:
        new = np.full(new_n, 1.0 / new_n)
    else:
        new /= total
    return new


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of convolution) for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Returns (objective, gradient) where a = softmax(y) and the objective
    is a smoothed version of 2·n·max(conv(a,a)).
    """
    # softmax -> probability vector
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via rFFT
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
    g_vec = conv_rev[n - 1: n - 1 + n]      # size n, gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimizer on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    s = best_a.sum()
    if s == 0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= s
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum

            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Sub‑gradient descent on the true max‑convolution using an
    exponentiated‑gradient update.  Returns the best a found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # compute full convolution (FFT)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        total = a.sum()
        if total <= 0.0:
            a = best_a.copy()
            total = a.sum()
        a /= total

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.20,
                      end_factor: float = 0.001,
                      recompute_every: int = 2000):
    """
    Moves a fraction of mass from the index with the largest sub‑gradient
    (most responsible for the current peak) to the index with the smallest.
    The step size decays linearly with the remaining time.
    Returns the best a and its fast_score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score

    phase_start = time.time()
    total_rem = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero, p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total_rem)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        if delta > a[src]:
            delta = a[src]

        # keep copies for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental update of convolution (linear term only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move the mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            # accept
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject
            a = a_old
            conv = conv_old

        steps += 1

        # Periodic exact recompute to avoid drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional progress report
        if steps % 20000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns a list of floats that will be fed to `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # 8000 fits comfortably in the 1000 s budget.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            a0 = interpolate_to_length(old, n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS bursts from random starts (≈10 % of time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.10 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-5, num=12)
    maxiter_per_lam = 200
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective (≈15 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02,
                     0.01, 0.005, 0.002, 0.001, 0.0005]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=3000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.13
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient on the true max‑convolution
    #          (≈40 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=15.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – gradient‑guided hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_hill_climb(
            best_a, best_score, deadline, rng,
            start_factor=0.25, end_factor=0.001, recompute_every=2500
        )
        print(f"[guided_hc] final best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
