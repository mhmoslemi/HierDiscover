def random_local_search(a_start, deadline, n, fft_len, rng, max_iter=1e9):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    while time.time() < deadline and it < max_iter:
        # Choose src and dst
        src = rng.integers(n)
        if a[src] == 0:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # Choose epsilon
        step_frac = rng.random() * 0.1  # up to 10%
        epsilon = a[src] * step_frac
        a_new = a.copy()
        a_new[src] -= epsilon
        a_new[dst] += epsilon
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        # else keep a as is (or maybe revert)
        it += 1
    return best_a, best_score
```

```
def guided_hill_climb_exact(a_start, start_score, deadline, rng,
                            start_factor=0.4, end_factor=0.001,
                            recompute_every=500):
    n = a_start.shape[0]
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    fft_len = next_pow2(2*n-1)

    start_time = time.time()
    total = deadline - start_time
    steps = 0
    accepted = 0

    # Precompute a_rev_fft? We'll compute each iteration as needed.

    while time.time() < deadline:
        # compute conv using FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # compute mask of near-max
        eps = max_b * 1e-12 + 1e-12
        mask = (conv >= max_b - eps).astype(np.float64)

        # gradient
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n-1:n-1+n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # fallback degenerate case
        if src == dst or a[src] <= 0.0:
            nonzero_idx = np.where(a > 1e-12)[0]
            if nonzero_idx.size == 0:
                break
            src = int(rng.choice(nonzero_idx, p=a[nonzero_idx]/a[nonzero_idx].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # compute step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac
        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        if delta > a[src]:
            delta = a[src]

        # create new candidate
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # evaluate new candidate exact
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[:2*n-1]
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # accept if not worse
        if new_score <= cur_score + 1e-15:  # allow tiny tolerance
            a = a_new
            cur_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # else reject; nothing changes

        steps += 1
        # optional periodic recompute but now each step recomputes anyway

        if steps % 5000 == 0:
            print(f"[guided_exact] step={steps}, best={best_score}")

    return best_a, best_score
```

def upscale(a, new_n):
    old_n = a.shape[0]
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)

"""
Propose a step‐function sequence with a low evaluation score.

The search proceeds in stages:

1. **Small‑scale global optimisation** (n≈4000)  
   – Random initialisations → L‑BFGS on a smooth log‑sum‑exp surrogate.  
   – Adam (stochastic) refinement of the same surrogate.

2. **Direct max‑convolution optimisation** (still n≈4000)  
   – Exponentiated‑gradient descent on the true max‑convolution.

3. **Upscaling** to a larger support (n≈8000) to reduce discretisation error.

4. **Exact gradient‑guided hill‑climb** on the large problem.  
   Each step computes the exact self‑convolution (FFT), the sub‑gradient
   of its ∞‑norm, moves a fraction of mass from the most “responsible”
   index to the least, and accepts the move only if the peak does not grow.
   The step‑size decays linearly with the remaining time.

5. **Fine‑grained random mixing** – occasional convex combinations with
   fresh random distributions and tiny random transfers, kept only if they
   improve the score.

All stages respect the 1000 s budget and the sequence is always kept
non‑negative and normalised (sum = 1), which makes the evaluation
independent of an overall scale.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a probability vector a (sum≈1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'triangular']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(x - 0.5) * 2.0   # peak in the centre, linear drop to edges
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Returns (smoothed objective, gradient) where a = softmax(y).
    The smoothed objective approximates 2·n·max(a*a) using a log‑sum‑exp.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Cascade of L‑BFGS optimisations over a decreasing λ schedule.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns best a and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly minimising max(a*a).
    Returns best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact gradient‑guided hill‑climb (full FFT evaluation each step)
# ----------------------------------------------------------------------


def guided_hill_climb_exact(a_start: np.ndarray,
                            start_score: float,
                            deadline: float,
                            rng: np.random.Generator,
                            start_factor: float = 0.40,
                            end_factor: float = 0.001,
                            recompute_every: int = 0):
    """
    Gradient‑guided hill‑climb where every candidate move is evaluated
    with the exact self‑convolution (FFT).  Mass is shifted from the
    index with the largest sub‑gradient to the one with the smallest.
    The step size decays linearly with the remaining time.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    fft_len = next_pow2(2 * n - 1)
    start_time = time.time()
    total = deadline - start_time
    steps = 0
    accepted = 0

    while time.time() < deadline:
        # exact convolution of current a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # mask of (near) maximal entries
        eps = max_b * 1e-12 + 1e-12
        mask = (conv >= max_b - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]   # length n

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # if gradient is degenerate, fall back to a random move
        if src == dst or a[src] <= 0.0:
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac
        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        delta = min(delta, a[src])

        # propose new distribution
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # evaluate new candidate exactly
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max()

        if new_score <= cur_score + 1e-15:   # accept non‑worsening moves
            a = a_new
            cur_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # else reject

        steps += 1
        if steps % 5000 == 0:
            print(f"[guided_exact] steps={steps}, best={best_score:.6f}")

    return best_a, best_score

        dst = int(rng.integers(0, best_a.shape[0]))
        if src != dst:
            eps = best_a[src] * rng.random() * 0.05   # ≤5 % of src mass
            a_new = best_a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
            # normalisation is preserved automatically
            new_score = fast_score(a_new, next_pow2(2 * a_new.shape[0] - 1))
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[tiny‑move] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
