"""
Search for a low‑score non‑negative step sequence.

The algorithm combines:
 - systematic sweep over several parametric families,
 - a broad random Dirichlet/Family exploration,
 - a multi‑stage continuation of a smoothed‑max L‑BFGS optimizer
   (log‑sum‑exp smoothing with a decreasing λ schedule),
 - a long projected Adam run on the true max‑convolution,
 - a simulated‑annealing hill‑climber with rich moves,
 - a final polishing L‑BFGS with a tiny λ.

All phases respect a global deadline (default 1000 s).  The best
candidate seen at any moment is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the linear part (first 2*n‑1 entries)
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Fast surrogate for the official evaluation (assumes sum(a)=1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (conv = a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a: only the central n entries of conv_rev are needed
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    Returns the best vector (as flat ndarray) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits – stays on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:          # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(a0: np.ndarray,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution.
    Returns the best vector found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)   # 1st moment
    v = np.zeros_like(a)   # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.5

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # identify all entries that are within relative tolerance of the max
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                    # any convex combination is a sub‑gradient

        # sub‑gradient of max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        fm = np.fft.fft(mask, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update (projected)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[adam] iter {it}, best fast_score={best_score:.6f}")

        a = a_candidate

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len_conv: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a square‑root cooling schedule.
    Moves:
      – transfer a random fraction from i → j,
      – swap two positions,
      – redistribute mass among three random indices,
      – scale a random subset (increase / decrease) while re‑normalising.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(4)
        a_new = a.copy()

        if move_type == 0:
            # transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        elif move_type == 2:
            # triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        else:
            # random scaling of a small random subset
            idx = rng.choice(n, size=rng.integers(1, max(2, n // 200)), replace=False)
            factor = 1.0 + rng.uniform(-0.3, 0.3)
            a_new[idx] = np.maximum(a_new[idx] * factor, 0.0)
            a_new = simplex_projection(a_new)

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a

def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def generate_gauss(n: int, sigma: float, mu: float = None) -> np.ndarray:
    i = np.arange(n, dtype=float)
    if mu is None:
        mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def generate_cosine_window(n: int, width: float) -> np.ndarray:
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.cos(np.pi * i / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def generate_triangular(n: int) -> np.ndarray:
    # symmetric triangular distribution (peak in middle)
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a

def generate_random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    draws = rng.gamma(alpha, 1.0, size=n)
    draws = np.maximum(draws, 0.0)
    total = draws.sum()
    if total == 0.0:
        draws = np.full(n, 1.0 / n)
    else:
        draws /= total
    return draws

def cheap_random_sweep(n: int,
                       fft_len_conv: int,
                       budget_s: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random exploration over many families for a short budget."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)

    start = time.time()
    deadline = start + budget_s * 0.15   # 15 % of total budget

    families = [
        "exp", "power", "gauss", "cos", "bimodal", "logistic",
        "binary_sparse", "double_exp", "random_dirichlet"
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        if fam == "exp":
            rate = rng.uniform(0.00002, 0.0018)
            a = generate_exp(n, rate)
        elif fam == "power":
            p = rng.uniform(-12.0, -0.1)
            inc = rng.choice([True, False])
            a = generate_power(n, p, inc)
        elif fam == "gauss":
            sigma = rng.uniform(100.0, 5000.0)
            a = generate_gauss(n, sigma)
        elif fam == "cos":
            width = rng.uniform(100.0, n / 2.0)
            a = generate_cosine_window(n, width)
        elif fam == "bimodal":
            mu1 = rng.uniform(0.0, n / 2.0)
            mu2 = rng.uniform(n / 2.0, n - 1)
            sigma1 = rng.uniform(50.0, 800.0)
            sigma2 = rng.uniform(50.0, 800.0)
            i = np.arange(n, dtype=float)
            a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
            a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
            a = a1 + a2
            a = np.maximum(a, 0.0)
            a /= a.sum()
        elif fam == "logistic":
            mu = rng.uniform(0.0, n - 1)
            sigma = rng.uniform(100.0, 2500.0)
            a = generate_logistic(n, mu, sigma)
        elif fam == "binary_sparse":
            prob = rng.uniform(0.01, 0.3)
            mask = rng.random(n) < prob
            if not np.any(mask):
                mask[0] = True
            a = mask.astype(float)
        elif fam == "double_exp":
            rate = rng.uniform(0.00002, 0.0018)
            a = generate_double_exp(n, rate)
        elif fam == "random_dirichlet":
            alpha = rng.uniform(0.05, 5.0)
            a = generate_random_dirichlet(n, alpha, rng)
        else:
            a = rng.random(n)

        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

"""
Search for a low‑score non‑negative step sequence.

The algorithm mixes:
 – systematic sweeps over several parametric families,
 – a cheap random Dirichlet/‑family exploration,
 – a multi‑stage continuation of a smoothed‑max L‑BFGS optimiser
   (log‑sum‑exp smoothing with a decreasing λ schedule),
 – a long projected Adam run on the true max‑convolution,
 – a simulated‑annealing hill‑climber with a rich move set,
 – a final polishing L‑BFGS pass with an ultra‑small λ.

All phases respect a global deadline (default 1000 s).  The best candidate
seen at any moment is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Fast surrogate for the official evaluation (assumes sum(a)=1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (conv = a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    Returns the best vector (as flat ndarray) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits – stays on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:          # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(a0: np.ndarray,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution.
    Returns the best vector found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)   # 1st moment
    v = np.zeros_like(a)   # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.5

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                    # any convex combination is a sub‑gradient

        # sub‑gradient of max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        fm = np.fft.fft(mask, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update (projected)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
            if it % 500 == 0:
                print(f"[adam] iter {it}, best fast_score={best_score:.6f}")

        a = a_candidate

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len_conv: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a square‑root cooling schedule.
    Moves:
      – transfer a random fraction from i → j,
      – swap two positions,
      – redistribute mass among three random indices,
      – scale a random subset (increase/decrease) and re‑normalise.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(4)
        a_new = a.copy()

        if move_type == 0:
            # Transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # Swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        elif move_type == 2:
            # Triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        else:
            # Random scaling of a small random subset
            subset_size = rng.integers(1, max(2, n // 200))
            idx = rng.choice(n, size=subset_size, replace=False)
            factor = 1.0 + rng.uniform(-0.3, 0.3)
            a_new[idx] = np.maximum(a_new[idx] * factor, 0.0)
            a_new = simplex_projection(a_new)

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float, mu: float = None) -> np.ndarray:
    i = np.arange(n, dtype=float)
    if mu is None:
        mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cosine_window(n: int, width: float) -> np.ndarray:
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.cos(np.pi * i / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def generate_random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    draws = rng.gamma(alpha, 1.0, size=n)
    draws = np.maximum(draws, 0.0)
    total = draws.sum()
    if total == 0.0:
        draws = np.full(n, 1.0 / n)
    else:
        draws /= total
    return draws


def parametric_grid_search(n: int,
                           fft_len_conv: int,
                           rng: np.random.Generator,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Systematic sweep over several families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)

    # exponential rates
    for rate in np.linspace(0.00005, 0.0015, 31):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power exponents
    for p in np.linspace(-10.0, -0.2, 25):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic families
    mus = np.linspace(0.2 * n, 0.8 * n, 9)
    sigmas = np.linspace(100.0, 2000.0, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Gaussian families
    for sigma in np.linspace(200.0, 4000.0, 15):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gauss(n, sigma)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gauss sigma={sigma:.0f} → score={best_sc:.6f}")

    # cosine window
    for width in np.linspace(200.0, n / 2.0, 12):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_cosine_window(n, width)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] cosine width={width:.0f} → score={best_sc:.6f}")

    # triangular
    a = generate_triangular(n)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
        print(f"[grid] triangular → score={best_sc:.6f}")

    # double exponential
    for rate in np.linspace(0.00005, 0.0015, 21):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_double_exp(n, rate)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] double_exp rate={rate:.5f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(n: int,
                       fft_len_conv: int,
                       budget_s: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random exploration over many families for a short budget."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)

    start = time.time()
    deadline = start + budget_s * 0.15   # 15 % of total budget

    families = [
        "exp", "power", "gauss", "cos", "bimodal", "logistic",
        "binary_sparse", "double_exp", "random_dirichlet"
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        if fam == "exp":
            rate = rng.uniform(0.00002, 0.0018)
            a = generate_exp(n, rate)
        elif fam == "power":
            p = rng.uniform(-12.0, -0.1)
            inc = rng.choice([True, False])
            a = generate_power(n, p, inc)
        elif fam == "gauss":
            sigma = rng.uniform(100.0, 5000.0)
            a = generate_gauss(n, sigma)
        elif fam == "cos":
            width = rng.uniform(100.0, n / 2.0)
            a = generate_cosine_window(n, width)
        elif fam == "bimodal":
            mu1 = rng.uniform(0.0, n / 2.0)
            mu2 = rng.uniform(n / 2.0, n - 1)
            sigma1 = rng.uniform(50.0, 800.0)
            sigma2 = rng.uniform(50.0, 800.0)
            i = np.arange(n, dtype=float)
            a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
            a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
            a = a1 + a2
            a = np.maximum(a, 0.0)
            a /= a.sum()
        elif fam == "logistic":
            mu = rng.uniform(0.0, n - 1)
            sigma = rng.uniform(100.0, 2500.0)
            a = generate_logistic(n, mu, sigma)
        elif fam == "binary_sparse":
            prob = rng.uniform(0.01, 0.3)
            mask = rng.random(n) < prob
            if not np.any(mask):
                mask[0] = True
            a = mask.astype(float)
        elif fam == "double_exp":
            rate = rng.uniform(0.00002, 0.0018)
            a = generate_double_exp(n, rate)
        elif fam == "random_dirichlet":
            alpha = rng.uniform(0.05, 5.0)
            a = generate_random_dirichlet(n, alpha, rng)
        else:
            a = rng.random(n)

        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem parameters
    # ------------------------------------------------------------------
    # The choice of n trades off solution quality against runtime.
    # 6000 gives a reasonable bound while keeping FFTs cheap.
    n = 6000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Initial candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Incorporate any historic construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1_sum = a1.sum()
            if a1_sum > 0.0:
                a1 /= a1_sum
                sc1 = fast_score(a1, fft_len_conv, n)
                if sc1 < best_score:
                    best_a, best_score = a1, sc1
                    print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – systematic grid search + cheap random sweep
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing allocation for subsequent phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS intensive stage
    phase2_end = start_time + 0.65 * budget_s   # Adam sub‑gradient stage
    phase3_end = start_time + 0.90 * budget_s   # Simulated annealing stage

    # ------------------------------------------------------------------
    # Phase 1 – aggressive multi‑restart L‑BFGS with λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 5e-07, 2e-07,
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        if rng.random() < 0.35:
            a_start = best_a * rng.normal(1.0, 0.03, size=n)
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            a_start = a_start / s if s > 0.0 else np.full(n, 1.0 / n)
        else:
            alpha = rng.uniform(0.5, 3.0)
            a_start = generate_random_dirichlet(n, alpha, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing with richer move set
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – L‑BFGS with ultra‑small λ
    # ------------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [1e-07, 5e-08, 2e-08, 1e-08, 5e-09]
        a_final, sc_final = lbfgs_continuation(
            best_a.copy(),
            tiny_sched,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    best_a = simplex_projection(best_a)   # final safety projection
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
