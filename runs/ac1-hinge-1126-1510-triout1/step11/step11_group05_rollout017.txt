"""
Hybrid optimizer for the step‑function bound.
-------------------------------------------------
The algorithm proceeds in several stages within a 1000‑s budget:

1.  Initialise a diverse set of candidates (previous best, random,
    spectral, and noisy copies) on the probability simplex.
2.  Adam optimisation on a log‑sum‑exp smooth surrogate of the
    max‑convolution, with a long continuation schedule.
3.  Exponentiated‑gradient descent on the exact max‑convolution.
4.  Projected sub‑gradient descent (Euclidean projection onto the simplex).
5.  A linear‑programming “mass‑redistribution’’ step: maximise the total
    mass of a new vector g under the constraints (f * g)_k ≤ M,
    where M is the current maximal convolution value of f.
    An extreme‑point solution of this LP moves mass from positions where
    f is large to positions where its reversed counterpart is small,
    which tends to lower the peak autocorrelation.
6.  Simulated‑annealing local search with a geometric temperature decay.
7.  Convex‑combination (golden‑section line search) among the best few
    vectors discovered.

All vectors are kept normalised (∑a = 1) so that the fast score
`2·n·max(conv)` equals the true objective.  The best vector is stored
in the historic global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective = 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random / spectral initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          scale: float = 10.0):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              init_eta: float = 0.5):
    """
    Euclidean projected sub‑gradient descent on max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local search
# ----------------------------------------------------------------------


def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 1e-3,
                               final_temp: float = 1e-8,
                               max_iter: int = 500_000):
    """
    Random mass‑transfer moves with a geometric temperature schedule.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        # geometric temperature decay
        temp = max(final_temp, init_temp * (0.9995 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each unordered pair of vectors perform a cheap golden‑section line
    search over t∈[0,1] to minimise the bound.
    Returns (best_a, best_score).
    """
    best_a = candidates[0].copy()
    best_score = scores[0]

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            lo, hi = 0.0, 1.0
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

            while time.time() < deadline and (hi - lo) > 1e-4:
                if fc < fd:
                    hi, fd = d, fc
                    d = c
                    c = hi - invphi * (hi - lo)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, fft_len, n)
                else:
                    lo, fc = c, fd
                    c = d
                    d = lo + invphi * (hi - lo)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, fft_len, n)

            tm = 0.5 * (lo + hi)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            ftm = fast_score(mix_tm, fft_len, n)

            if ftm < best_score:
                best_score = ftm
                best_a = mix_tm.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mass redistribution step
# ----------------------------------------------------------------------


def lp_redistribute(a: np.ndarray, max_k: int, n: int, fft_len: int,
                    deadline: float, alpha: float = 0.5):
    """
    Solve  max ∑ g_i   s.t. (a * g)_k ≤ max_k  ∀k in a selected set,
    g_i ≥ 0.  Return the blended vector (1‑α)·a + α·g_norm.
    """
    # compute full convolution just to pick a few active constraints
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    # pick the K largest entries (they dominate the max)
    K = min(15, conv_full.size)
    idx = np.argsort(-conv_full)[:K]

    # build the (small) constraint matrix T g ≤ max_k
    a_rev = a[::-1]
    rows = []
    for k in idx:
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        row[js] = a_rev[k - js]
        rows.append(row)
    A = np.vstack(rows)
    b = np.full(A.shape[0], max_k)

    # linear program: maximise ∑g  ⇔ minimise -∑g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return a, fast_score(a, fft_len, n)   # fall back to unchanged vector

    g = res.x
    if g.sum() == 0:
        return a, fast_score(a, fft_len, n)

    g_norm = g / g.sum()
    blended = (1.0 - alpha) * a + alpha * g_norm
    blended /= blended.sum()
    return blended, fast_score(blended, fft_len, n)

    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 300   # increased compared with the reference code

    for r in range(12):   # a few more restarts
        if time.time() >= deadline:
            break
        a0 = best_a.copy() if r == 0 else add_noise(best_a,
                                                    rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len,
            n,
        )
        print(f"[Adam restart {r + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            scale=15.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgrad_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            init_eta=0.8,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – linear‑programming mass redistribution
    # --------------------------------------------------------------
    if time.time() < deadline:
        # perform a handful of LP‑guided updates
        for it_lp in range(8):
            if time.time() >= deadline:
                break
            # current maximal convolution value
            fa = np.fft.fft(best_a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            M = conv.max()
            blended, new_score = lp_redistribute(best_a, M, n, fft_len,
                                                 deadline, alpha=0.5)
            print(f"[LP iter {it_lp + 1}] score={new_score:.6f}")
            if new_score < best_score:
                best_score, best_a = new_score, blended.copy()
                print(f"   ==> new global best after LP: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – simulated‑annealing local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_search(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-3,
            final_temp=1e-9,
            max_iter=500_000,
        )
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – convex‑combination (mixture) search among top vectors
    # --------------------------------------------------------------
    if time.time() < deadline:
        # collect a few random top‑k candidates for mixture search
        extra_vecs = [random_initial(n, rng) for _ in range(5)]
        candidates = [best_a] + extra_vecs
        scores = [fast_score(v, fft_len, n) for v in candidates]

        # keep the K best
        K = 5
        ordered = sorted(zip(scores, candidates), key=lambda x: x[0])
        sel_scores, sel_vecs = zip(*ordered[:K])

        a_opt, sc_opt = mixture_search(
            list(sel_vecs),
            list(sel_scores),
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
