        if contributions.sum() == 0:

            src_idx = rng.choice(idx, p=probs)
        # decide destination index distinct from src_idx

        # compute fast_score

"""
step‑function optimiser – 1000 s budget (enhanced)

The optimiser consists of three phases:

1. **Diverse starts + aggressive L‑BFGS on a smoothed log‑sum‑exp
   surrogate**.  A cascade of λ‑values down to 1e‑6 drives many random
   starts deep into good basins.

2. **Long mirror‑SGD (exponentiated‑gradient) on the exact L∞ objective**.
   The sub‑gradient is computed by FFT; step‑size decays as η₀/√t.

3. **Targeted simulated‑annealing** that always attacks the shift with
   the current maximal convolution value by moving mass away from the
   most harmful pair of indices.  This sharpens the result beyond the
   generic anneal used previously.

All phases respect the 1000 s overall deadline, keep the vector on the
probability simplex, and finally evaluate the raw metric with
`evaluate_sequence`.  The best sequence is stored in the global variable
`height_sequence_1` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT (real part only).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'quadratic']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:               # quadratic (convex hill)
        x = np.linspace(-1, 1, n)
        a = 1.0 - x * x
        a = np.clip(a, 0, None)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_projection(noisy)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns the smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # ---------- soft‑max ----------
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---------- convolution ----------
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # effective smoothing (avoid division by zero)
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient ----------
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # convolution of w with reversed a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the n‑dimensional gradient block
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_iters: int = 15_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        # cheap best‑track
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score

def targeted_annealing(a_start: np.ndarray,
                       start_score: float,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator):
    """
    Move mass away from the most harmful pair of indices for the shift
    that currently attains the maximal convolution value.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    # temperature aligned to current best score
    temp0 = max(best_score * 0.04, 1e-12)
    max_frac = 0.25     # up to 25 % of src mass may be moved in one step

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-14)

        # compute full convolution once per outer loop
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        s_max = int(np.argmax(conv))
        max_val = conv[s_max]

        # indices that participate in this shift
        i_lo = max(0, s_max - (n - 1))
        i_hi = min(n - 1, s_max)
        idx = np.arange(i_lo, i_hi + 1)
        jdx = s_max - idx

        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total == 0.0:
            # fall back to random move
            src = rng.integers(n)
        else:
            # sample source index proportional to its contribution
            probs = contrib / total
            src = rng.choice(idx, p=probs)

        # choose a destination that is *not* the source
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer a modest amount of mass away from src
        delta = a[src] * rng.uniform(0.0, max_frac)
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, fft_len)
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n provides more flexibility.
    # ------------------------------------------------------------------
    n = 6000                       # increase dimension for better bound
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families (more than before)
    for _ in range(20):
        pool.append(random_initial(n, rng))

    # 3) Classical windows – useful baselines
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # allocate roughly 15 % to the L‑BFGS warm‑up, 70 % to EG,
    # remaining 15 % to the targeted anneal
    phase1_end = start_time + 0.15 * budget_s
    phase2_end = start_time + 0.85 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS cascade on a smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.8, 0.5, 0.3, 0.18, 0.11, 0.07,
        0.045, 0.028, 0.018, 0.011, 0.007,
        0.0045, 0.0028, 0.0018, 0.0012, 0.0008,
        0.0005, 0.0003, 0.00018, 0.00011, 0.00007,
        0.000045, 0.00003, 0.000018, 0.000011, 0.000007,
        0.0000045, 0.000003, 0.0000018, 0.0000011, 0.0000007,
        0.0000004, 0.0000002, 0.0000001
    ]
    maxiter_per_lam = 2000

    best_a = None
    best_score = float('inf')
    # run as many restarts as time permits
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter to increase diversity
        if rng.random() < 0.4:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.015), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:                                 # fallback
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (exact L∞)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=22.0,
            max_iters=20_000_000
        )
        print(f"[exp‑grad] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted annealing fine‑tune
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_annealing(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[targeted‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
