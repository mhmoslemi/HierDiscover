"""
Propose a low‑evaluation step‑function sequence.

Algorithm overview
------------------
1. Choose a relatively large problem size (n = 8000) – finer discretisation
   gives a better approximation of the continuous optimum.
2. Build an initial pool of candidate distributions:
   * uniform, arcsine, random,
   * a parametrised family based on symmetric Beta(α,α) and
     power‑law shapes, optionally mixed with uniform,
   * the previous best sequence (if supplied).
   The parametrised family is explored with a short random search
   (≈ 20 s) and the best result is taken as the starting point.
3. Refine the starter with a long run of exponentiated‑gradient descent
   on the true ‑max‑convolution (≈ 50 % of the remaining budget).
4. Continue with a projected sub‑gradient method (≈ 30 % of the remaining
   budget) – an active set of near‑maximal convolution entries yields a
   useful sub‑gradient.
5. Finish with a greedy hill‑climb that maintains the convolution
   incrementally (the rest of the budget).
6. Return the final normalized list; the harness will call
   `evaluate_sequence` to obtain the official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a)).  Lower is better."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generators

def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalized to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = uniform_initial(n)
    else:
        a /= a_sum
    return a


def power_initial(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+0.5)^{‑p}+ (n‑i‑0.5)^{‑p}."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = (idx ** (-p)) + ((n + 0.5 - idx) ** (-p))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def mixed_initial(base: np.ndarray, n: int, mix: float) -> np.ndarray:
    """Convex mix of a base shape with the uniform distribution."""
    uniform = np.full(n, 1.0 / n)
    a = mix * base + (1.0 - mix) * uniform
    a /= a.sum()
    return a

# Parametric search (fast, uses only fast_score)
# ----------------------------------------------------------------------


def parametric_search_best(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    """Explore a low‑dimensional family of shapes and return the best."""
    best_a = uniform_initial(n)
    best_score = fast_score(best_a, fft_len)
    start = time.time()

    while time.time() - start < time_budget:
        shape_type = rng.choice(['beta', 'power'])
        if shape_type == 'beta':
            alpha = rng.uniform(0.05, 1.0)          # α < 1 ⇒ heavier edges
            base = beta_initial(n, alpha)
        else:  # power
            p = rng.uniform(0.2, 3.0)
            base = power_initial(n, p)

        # optionally mix with uniform
        mix = rng.uniform(0.0, 1.0)
        a = mixed_initial(base, n, mix)

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 50.0,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = best_a.copy()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eps_frac: float = 0.001,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        # back‑tracking line search (up to 5 reductions)
        improved = False
        cur_score = fast_score(a, fft_len)
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < cur_score:
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # small stochastic jitter to escape stagnation
            jitter = rng.normal(scale=1e-5, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental updates
# ----------------------------------------------------------------------


def hill_climb_incremental(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            # occasional full recompute to avoid drift
            conv = np.convolve(best, best)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 8000 gives a finer discretisation without hurting speed.
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1. Build a small collection of diverse starting points
    # --------------------------------------------------------------
    init_candidates = []

    # deterministic primitives
    init_candidates.append(uniform_initial(n))
    init_candidates.append(arcsine_initial(n))
    init_candidates.append(random_initial(n, rng))

    # previous best (if available)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # interpolate to current size
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            init_candidates.append(a_old)

    # --------------------------------------------------------------
    # 2. Quick parametric sweep for a better starter
    # --------------------------------------------------------------
    param_time = min(20.0, 0.02 * budget_s)  # ≤20 s or 2 % of the total budget
    param_best_a, param_best_score = parametric_search_best(
        n, fft_len, rng, param_time
    )
    init_candidates.append(param_best_a)

    # keep the best primitive / parametric candidate (cheap evaluation)
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient descent (≈ 50 % of remaining time)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.5 * (deadline - time.time())
    best_a, best_score = exp_gradient_optimize(
        best_a, phase1_deadline, n, fft_len, rng, eta0=50.0
    )
    print(f"[exp‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent (≈ 30 % of remaining time)
    # --------------------------------------------------------------
    phase2_deadline = time.time() + 0.3 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.3
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final greedy hill‑climb (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
