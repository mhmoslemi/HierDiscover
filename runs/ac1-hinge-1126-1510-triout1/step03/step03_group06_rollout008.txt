def smoothed_gradient_descent(a0, max_iters, step0, p, deadline):
    a = a0.copy()
    n = a.shape[0]
    a = project_simplex(a) # ensure sum=1
    best_seq = a.copy()
    best_score = compute_score(a)  # compute true objective
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv.
        conv, max_b = compute_conv_and_max(a)  # conv length 2n-1
        # compute Lp norm:
        S = np.sum(conv ** p)
        if S == 0:
            break
        L = S ** (1.0 / p)
        # compute w = conv ** (p-1)
        w = conv ** (p - 1)
        # compute correlation
        grad = correlation_slice(w, a)  # length n
        # scale
        grad = (2.0 * grad) / (L ** (p - 1))
        # step schedule: step = step0 / it**0.5 or step0 * (0.99**it) maybe.
        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)
        # evaluate true objective occasionally:
        if it % 100 == 0 or it==max_iters:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[smooth p={p} it={it}] new best={score:.6f}")
    return best_seq, best_score

"""
Gradient‑smoothed simplex optimisation for the max‑self‑convolution objective.

The evaluation score is   2·n·max(conv(a,a)) / (sum(a)²) .
Because the expression is homogeneous we fix Σa = 1 and optimise a on the
probability simplex.  Instead of a sub‑gradient on the active peak we
 minimise a smooth approximation  ‖a*a‖ₚ = (∑ (a*a)ᵖ)¹/ᵖ for a sequence of
increasing p (2, 4, 8, 16, 32, 64).  The gradient of this p‑norm is
available in closed form via a convolution, allowing cheap O(n log n)
updates with FFTs.  After the smoothed phases a cheap simulated‑annealing
refinement moves a tiny amount of mass between random coordinates.
Multiple restarts, different sequence lengths and several families of
initialisations (uniform, power‑law, power‑law·exponential) are performed
until the time budget expires.  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_len_for_conv(n: int) -> int:
    """Length of the zero‑padded FFT that can hold a full linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximum entry.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation score 2·n·max(conv)/sum(a)²."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    _, max_b = compute_conv_and_max(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Correlation needed for the smoothed gradient
# ----------------------------------------------------------------------
def correlation_slice(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    For vectors a (length n) and w (length 2n‑1) return the slice
    g_i = Σ_j a_j · w_{i+j} for i = 0…n‑1.
    Implemented via FFT (correlation of w with a).
    """
    n = a.shape[0]
    L = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a[::-1], n=L)
    conv_rev = np.fft.irfft(W * A_rev, n=L)
    start = n - 1
    return conv_rev[start : start + n]


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex (p‑norm of the convolution)
# ----------------------------------------------------------------------
def smoothed_gradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    p: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise the p‑norm ‖a*a‖ₚ while staying on the simplex.
    The update uses the exact gradient
        ∇L = 2·correlation(conv^{p‑1}, a) / L^{p‑1},
    where L = (∑ convᵖ)¹/ᵖ.
    """
    a = project_simplex(a0.copy())
    n = a.shape[0]

    best_seq = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full self‑convolution
        conv, _ = compute_conv_and_max(a)

        # p‑norm of the convolution
        S = np.sum(conv ** p)
        if S <= 0.0:
            break
        Lp = S ** (1.0 / p)                # ‖conv‖ₚ

        # gradient of the p‑norm (before projection)
        w = conv ** (p - 1)                # conv^{p‑1}
        grad = correlation_slice(w, a)     # Σ_j a_j·conv_{i+j}^{p‑1}
        grad = (2.0 * grad) / (Lp ** (p - 1))

        # step size schedule (standard diminishing step)
        step = step0 / np.sqrt(it)

        # gradient step + projection onto the simplex
        a = project_simplex(a - step * grad)

        # occasional true‑score check (every 200 iters)
        if it % 200 == 0 or it == max_iters:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[smooth p={p:2d} it={it:5d}] new best={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny amounts of probability mass between random coordinates."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = compute_score(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of the move
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # move up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = compute_score(new)

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score
