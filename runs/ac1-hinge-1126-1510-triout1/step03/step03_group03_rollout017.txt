Propose a non‑negative height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two stages.

1️⃣ **Broad random search** – thousands of candidates are sampled from
   a variety of parametric families (power‑law, exponential,
   two‑sided power‑law, arcsine, tapered‑arcsine, cosine, Gaussian).
   The best candidate (lowest fast evaluation) is kept.

2️⃣ **Smooth‑max Adam refinement** – starting from the best candidate we
   minimise the *smooth* maximum of the self‑convolution.
   For the vector `a` we keep `x = log(a)` and use the Adam optimiser
   on the smooth‑max objective

        Φβ(a) = (1/β)·log Σk exp(β·(a∗a)k)

   The gradient of Φβ can be written as a convolution and is obtained
   efficiently with two FFTs each iteration.  The temperature β is
   increased from 5 to 30 during the run, focusing the optimisation
   more and more on the true maximum.  After each update we project
   back onto the probability simplex (normalise to sum 1).  Occasional
   jitter helps to escape plateaus.

The whole routine respects the 1000 s budget, prints progress and
returns a plain Python list ready for the provided `evaluate_sequence`.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective using FFT (seq must be non‑negative)."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}.  They satisfy (a∗a)_k = 1 for all k≥0
    (infinite series) and behave like 1/√i.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_arcsine_tapered(n: int, exponent: float = 1.0) -> np.ndarray:
    """
    Tapered arcsine: a_i = arcsine_i * sin(π·(i+0.5)/n)^{exponent}.
    The sine window forces the ends toward zero while preserving the
    favourable flat‑convolution property of the pure arcsine.
    """
    base = generate_arcsine_like(n)
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** exponent
    return base * w
