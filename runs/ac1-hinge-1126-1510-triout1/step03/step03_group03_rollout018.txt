"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in several stages:

1.  Multi‑scale optimisation – start with a modest length (4000), optimise,
    then interpolate the best vector to a larger length (8000, 12000, …).
    Larger n gives a finer approximation of the continuous problem.

2.  At each scale a smooth approximation of the “max‑convolution’’ is built.
    The current self‑convolution `c = a*a` is computed with FFT.
    A small fraction of the largest convolution entries (top‑k) is turned into
    a weight vector `w` (probability simplex).  The gradient of the
    “soft‑max’’ surrogate is
        ∇ = 2·(a ⊛ w)                     (⊛ = linear convolution)
    which is also computed by FFT.

3.  An additive gradient step with a decaying learning rate is performed,
    followed by Euclidean projection onto the probability simplex
    (guarantees non‑negativity and Σa = 1).

4.  Occasional jitter (tiny random perturbation) helps to escape plateaus.
    The best sequence seen so far (according to a cheap FFT‑based score)
    is stored and finally evaluated with the official `evaluate_sequence`.

The whole routine respects the 1000 s budget and returns a plain Python list.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Cheap estimator of 2·n·max(conv(seq,seq))/(sum(seq)**2) using FFT.
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Un‑normalised coefficients a_i = (2i−1)/(2i)·a_{i−1};  a_0 = 1.
    Behaves roughly like 1/√i and gives a good starting point.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_n = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Multi‑scale optimisation
    # ------------------------------------------------------------------
    # Lengths we will visit (increasing).  Final length ≈ 20000 (still fast).
    length_schedule = [4000, 8000, 12000, 18000, 20000]

    best_overall_seq = None
    best_overall_score = math.inf

    # keep an intermediate vector that will be up‑scaled to the next length
    current_seq = None

    for n in length_schedule:
        # Stop early if time is gone
        if time.time() >= deadline - 20.0:
            break

        print(f"\n=== Stage with length={n} (time left {deadline - time.time():.1f}s) ===")

        # --------------------------------------------------------------
        # Initialise the candidate for this scale
        # --------------------------------------------------------------
        if current_seq is None:
            # Try to use the previously discovered construction if it matches the length
            try:
                init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
                if init.size == n:
                    current_seq = init.copy()
                else:
                    # interpolate the global candidate to the required size
                    current_seq = upscale_sequence(init, n)
            except Exception:
                # fallback: arcsine‑like shape
                current_seq = generate_arcsine_like(n)
        else:
            # upscale the vector obtained from the previous stage
            current_seq = upscale_sequence(current_seq, n)

        # non‑negativity and normalisation
        current_seq = np.maximum(current_seq, 0.0)
        current_seq = project_simplex(current_seq)

        # keep the best sequence seen in this scale
        best_local_seq = current_seq.copy()
        best_local_score = fast_score(best_local_seq)
        print(f"[init] fast_score = {best_local_score:.6f}")

        # --------------------------------------------------------------
        # Pre‑compute lengths for FFTs (conv a*a and a*w)
        # --------------------------------------------------------------
        fft_len_conv = next_pow2(2 * n - 1)          # for a*a
        fft_len_grad = next_pow2(3 * n - 2)          # for a*w (linear conv)

        # hyper‑parameters (tuned a bit experimentally)
        base_lr = 0.8          # initial additive step size
        lr_decay = 1e-4        # controls decay of the step size
        top_fraction = 0.001   # fraction of convolution entries used as active set
        jitter_interval = 6000
        jitter_scale = 0.005
        eval_interval = 2000

        iteration = 0
        while time.time() < deadline - 1.0:
            iteration += 1

            # ----------------------------------------------------------
            # 1) self‑convolution a*a  (FFT, linear conv)
            # ----------------------------------------------------------
            Fa = np.fft.rfft(current_seq, n=fft_len_conv)
            conv_full = np.fft.irfft(Fa * Fa, n=fft_len_conv)[: 2 * n - 1]

            # ----------------------------------------------------------
            # 2) build a sparse weight vector w that concentrates on the
            #    largest convolution entries (soft active set)
            # ----------------------------------------------------------
            k = max(1, int(top_fraction * conv_full.size))
            # argpartition gives the indices of the top‑k entries (unsorted)
            top_idx = np.argpartition(conv_full, -k)[-k:]
            w = np.zeros_like(conv_full, dtype=np.float64)
            w[top_idx] = 1.0 / k   # uniform weight over the active set
            # tiny epsilon everywhere to keep the kernel strictly positive
            eps = 1e-12
            w += eps
            w /= w.sum()

            # ----------------------------------------------------------
            # 3) gradient = 2·(a ⊛ w)   (linear convolution)
            # ----------------------------------------------------------
            Fa_g = np.fft.rfft(current_seq, n=fft_len_grad)
            Fw_g = np.fft.rfft(w, n=fft_len_grad)
            grad_full = np.fft.irfft(Fa_g * Fw_g, n=fft_len_grad)
            # central slice corresponds to indices n-1 … n-1+n-1
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # ----------------------------------------------------------
            # 4) additive update + projection onto the simplex
            # ----------------------------------------------------------
            lr = base_lr / (1.0 + iteration * lr_decay)
            current_seq = current_seq - lr * grad
            current_seq = project_simplex(current_seq)

            # ----------------------------------------------------------
            # 5) occasional jitter to avoid stagnation
            # ----------------------------------------------------------
            if iteration % jitter_interval == 0:
                noise = np.random.randn(n) * jitter_scale * current_seq
                current_seq = np.clip(current_seq + noise, 0.0, None)
                current_seq = project_simplex(current_seq)

            # ----------------------------------------------------------
            # 6) periodic evaluation (cheap) and bookkeeping
            # ----------------------------------------------------------
            if iteration % eval_interval == 0:
                sc = fast_score(current_seq)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_seq = current_seq.copy()
                    print(
                        f"[{time.time() - start_time:.1f}s] n={n} iter={iteration} "
                        f"new local best = {sc:.6f}"
                    )

        # --------------------------------------------------------------
        # End of this scale – keep the best found so far
        # --------------------------------------------------------------
        print(f"[stage end] length={n} best fast_score={best_local_score:.6f}")
        # store for the next up‑scale
        current_seq = best_local_seq.copy()

        # compare against the global best
        if best_local_score < best_overall_score:
            best_overall_score = best_local_score
            best_overall_seq = best_local_seq.copy()
            print(
                f"[global] new overall best (length={n}) fast_score={best_overall_score:.6f}"
            )

    # Final verification with the official evaluator
    # ------------------------------------------------------------------

    print(
        f"\nFinished. Best fast_score = {best_overall_score:.6f}, "
        f"official score = {final_score:.6f}"
    )
    return final_seq
