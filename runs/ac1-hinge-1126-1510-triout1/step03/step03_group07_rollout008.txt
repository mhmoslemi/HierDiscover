"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code follows three main ideas:

1. **Rich parametric warm‑start** – try a dense grid of simple families
   (power‑law, exponential, U‑shaped, Beta and the classic arcsine‐type
   shape).  The best candidate (smallest max‑convolution) is kept.

2. **Two‑stage first‑order optimisation** – a long Adam phase on a
   smooth approximation of the max (log‑sum‑exp) followed by a
   sub‑gradient phase on the true max.  Both phases work on the
   probability simplex (the vector is always projected onto the
   simplex after each update).

3. **Gradient‑guided greedy mass transfers** – after the continuous
   optimisation we repeatedly move a small amount of mass from the
   coordinates with the largest (worst) sub‑gradient to the ones with
   the smallest sub‑gradient, accepting the move only if the max‑convolution
   improves.  This fine‑tunes the solution beyond what the pure
   sub‑gradient descent can achieve.

All heavy work (convolution, gradient) is carried out with FFTs
(O(n log n) per iteration).  The routine respects the supplied time
budget and returns the best sequence found, scaled to the normalisation
used by `evaluate_sequence` (sum = √(2 n)).
"""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum(v)=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of <weights, a*a> w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    n = a.shape[0]
    # The correlation result is centred; we extract the part that aligns with a.
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    # tolerance generous enough to capture numerically equal peak entries
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin so that we can finish cleanly
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1) Choose the length of the vector.
    # If the global variable `height_sequence_1` exists we keep its length,
    # otherwise we fall back to a modest size (8000) that still fits comfortably
    # into memory.
    try:
        init_vec = np.array(height_sequence_1, dtype=np.float64)
        n = init_vec.shape[0]
        # normalise to L1‑norm = 1 (the scaling does not affect the objective)
        a = init_vec / init_vec.sum()
        print(f"Using length n={n} from existing construction.")
    except Exception:
        n = 8000
        a = None
        print(f"No global construction found – using n={n}.")

    # FFT helpers (same mask used throughout the run)
    L = 1 << ((2 * n - 1).bit_length())      # smallest power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ------------------------------------------------------------------
    # 2) Parametric sweep – collect a handful of simple families and keep the best.
    best_a = None
    best_max = np.inf

    xs = np.arange(n, dtype=np.float64)

    # helper to evaluate a candidate (assumes sum(a)=1)
    def _eval_max(candidate: np.ndarray) -> float:
        conv = _compute_convolution(candidate, L)
        return conv.max()

    print("--- Phase 0: parametric sweep ---")
    # power‑law family a_i ∝ (i+1)^p + (n-i)^p  (symmetric)
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1) ** p
        cand = base + base[::-1]          # enforce symmetry – helps the max stay central
        cand = cand / cand.sum()
        cur_max = _eval_max(cand)
        if cur_max < best_max:
            best_max, best_a = cur_max, cand.copy()
            print(f"  new best (power p={p:.2f}) max={cur_max:.8e}")

    # exponential family a_i ∝ exp(-λ i)  (and mirror)
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        cand = exp_shape + exp_shape[::-1]
        cand = cand / cand.sum()
        cur_max = _eval_max(cand)
        if cur_max < best_max:
            best_max, best_a = cur_max, cand.copy()
            print(f"  new best (exp λ={lam:.2f}) max={cur_max:.8e}")

    # double‑exponential (U‑shaped) family
    for lam in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        cand = left + left[::-1]
        cand = cand / cand.sum()
        cur_max = _eval_max(cand)
        if cur_max < best_max:
            best_max, best_a = cur_max, cand.copy()
            print(f"  new best (U‑shape λ={lam:.2f}) max={cur_max:.8e}")

    # Beta‑density family (discrete Beta(a,b) shape)
    alphas = np.linspace(0.2, 2.0, 10)
    betas  = np.linspace(0.2, 2.0, 10)
    x_mid = (xs + 0.5) / n
    for a_par in alphas:
        for b_par in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a_par - 1)) * ((1.0 - x_mid) ** (b_par - 1))
            if np.all(beta_shape > 0):
                cand = beta_shape / beta_shape.sum()
                cur_max = _eval_max(cand)
                if cur_max < best_max:
                    best_max, best_a = cur_max, cand.copy()
                    print(f"  new best (Beta a={a_par:.2f} b={b_par:.2f}) max={cur_max:.8e}")

    # arcsine‑like family  a_i ∝ (x (1‑x))^{-p}
    for p in np.linspace(0.3, 0.9, 13):
        if time.time() > deadline:
            break
        arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
        arcsine = arcsine ** p          # shape parameter
        cand = arcsine / arcsine.sum()
        cur_max = _eval_max(cand)
        if cur_max < best_max:
            best_max, best_a = cur_max, cand.copy()
            print(f"  new best (arcsine p={p:.2f}) max={cur_max:.8e}")

    # If no sweep succeeded (unlikely), fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _eval_max(best_a)

    # If we already had a vector from `height_sequence_1`, compare it.
    if a is not None:
        cur_max = _eval_max(a)
        if cur_max < best_max:
            best_max, best_a = cur_max, a.copy()
            print(f"  initial global candidate better: max={cur_max:.8e}")
    a = best_a.copy()
    print(f"Initial best max={best_max:.8e}   score≈{2 * n * best_max:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam descent on a smoothed max (log‑sum‑exp).
    print("--- Phase 1: Adam on smoothed max ---")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                     # slightly larger base learning rate
    max_adam_iters = 120000       # more iterations than the previous run
    alpha_start, alpha_end = 2.0, 500.0

    it = 0
    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        # smooth‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate the true max (not the smooth one)
        new_max = _compute_convolution(a_new, L).max()
        if new_max < best_max:
            best_max, best_a = new_max, a_new.copy()
            print(f"  Adam it {it:5d}  improved max={new_max:.8e}  score≈{2 * n * new_max:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 4) Sub‑gradient descent on the true max.
    print("--- Phase 2: sub‑gradient descent on true max ---")
    sub_max_iters = 120000
    sub_it = 0
    step0 = 0.8
    while time.time() < deadline and sub_it < sub_max_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        w = _submax_weights(conv)             # uniform on active set
        grad = _gradient_from_weights(a, w, phase, L)

        # Polyak‑style diminishing step (heuristic)
        step = step0 / math.sqrt(sub_it)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_max = _compute_convolution(a_new, L).max()
        if new_max < best_max:
            best_max, best_a = new_max, a_new.copy()
            print(f"  Sub‑grad it {sub_it:5d}  improved max={new_max:.8e}  score≈{2 * n * new_max:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 5) Gradient‑guided greedy local moves (mass transfers).
    print("--- Phase 3: gradient‑guided greedy moves ---")
    a_curr = best_a.copy()
    move_it = 0
    while time.time() < deadline:
        move_it += 1
        conv = _compute_convolution(a_curr, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_a = cur_max, a_curr.copy()
            print(f"  Greedy move it {move_it:5d}  improved max={cur_max:.8e}  score≈{2 * n * cur_max:.6f}")

        # sub‑gradient (active‑set) gives a direction of change
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a_curr, w, phase, L)

        # pick source (largest positive gradient) and sink (most negative)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # very unlikely, pick a random pair instead
            src = np.random.choice(n, p=a_curr)
            dst = np.random.randint(0, n)

        # move a modest fraction of the source mass
        delta = min(0.02 * a_curr[src], 0.02)   # cap at 2 % of source
        if delta <= 0:
            continue

        a_new = a_curr.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # projection not needed – we kept the simplex exactly

        new_max = _compute_convolution(a_new, L).max()
        if new_max < cur_max - 1e-12:
            # accept strictly improving moves
            a_curr = a_new
        else:
            # try a smaller delta (down‑scale geometrically)
            delta *= 0.5
            if delta > 1e-8:
                a_new = a_curr.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                new_max = _compute_convolution(a_new, L).max()
                if new_max < cur_max - 1e-12:
                    a_curr = a_new

    # ------------------------------------------------------------------
    # 6) Final random hill‑climbing (tiny perturbations) – safety net.
    print("--- Phase 4: final random hill‑climbing ---")
    a_curr = best_a.copy()
    while time.time() < deadline:
        i = np.random.choice(n, p=a_curr)   # favour larger entries
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.01 * a_curr[i]   # up to 1 % transfer
        if delta <= 0:
            continue
        a_new = a_curr.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        new_max = _compute_convolution(a_new, L).max()
        if new_max < best_max:
            best_max, best_a = new_max, a_new.copy()
            a_curr = a_new
            print(f"  Random move improved max={new_max:.8e}  score≈{2 * n * new_max:.6f}")

    # ------------------------------------------------------------------
    # 7) Scale to the normalisation used by `evaluate_sequence`.
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_a * target_sum).tolist()

    # Log the final score (using the provided evaluator – does not affect the answer)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
              f"after {it} Adam iters, {sub_it} sub‑grad iters ---")
    except Exception:
        # In case `evaluate_sequence` is not available during testing
        print(f"--- Finished (local score not computed) ---")

    return final_seq
