"""
Propose a low‑autocorrelation step function.

Algorithm
---------
1. **Initialisation** – take the known construction (``height_sequence_1``)
   if it exists, otherwise fall back to a few simple analytic shapes
   (uniform, arcsine, exponential, power‑law).  All candidates are
   normalised to unit L1‑norm (the required √(2 n) scaling is applied only
   at the very end).

2. **Adam on the smoothed max** – replace `max(conv)` by a soft‑max
   `log‑sum‑exp(α·conv)/α`.  The gradient of this smooth objective is
   `2·correlation(weights, a)`.  Adam updates are performed on the simplex
   (projection after each step).  The smoothing parameter `α` grows from a
   small to a large value, steering the iterate towards a basin of a true
   optimum.

3. **Polyak sub‑gradient descent** – once the smooth phase has converged,
   switch to genuine sub‑gradients of the hard max (uniform weights on
   the active set).  The step size is chosen by the Polyak rule
   `step = (cur_max - best_max)/(‖grad‖²)`, which yields fast monotone
   decrease.

4. **Simulated‑annealing hill climbing** – finally a stochastic local
   search makes occasional uphill moves, allowing the procedure to escape
   small plateaus.  The temperature is linearly cooled and the move is a
   tiny mass transfer between two randomly chosen coordinates.

All heavy operations (convolution, gradient) use FFTs (`O(L log L)`) and the
whole routine respects the supplied time budget (≈ 1000 s).  The best
sequence found is returned after scaling its L1‑norm to √(2 n) – the exact
format expected by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    # reversed a in the frequency domain:
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # centre part of the circular correlation = linear correlation
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # fallback (should never happen)
        w[conv.argmax()] = 1.0
    return w
