def project_simplex(v):
    # v may be any real numbers
    # ensure non-negative
    v = np.maximum(v, 0)
    if v.sum() == 0:
        # return uniform
        return np.full_like(v, 1.0 / v.size)
    # project onto sum=1
    # use algorithm: sort descending, find rho, theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    n=5000
    fft_len = next_pow2(2*n - 1)

    # initialize sequence as uniform
    a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    print("[init] fast_score=", best_score)

    # parameters
    eta0 = 0.5   # initial step size
    eta_decay = 0.9995   # decay per iteration
    min_eta = 1e-5

    iter=0
    while time.time() < deadline:
        # compute convolution
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        # identify indices close to max within tolerance
        eps = max_val * 1e-12 + 1e-15
        max_idx = np.where(conv >= max_val - eps)[0]
        # compute gradient
        g = np.zeros_like(a)
        for k in max_idx:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            i = np.arange(low, high+1)
            j = k - i
            g[i] += a[j]
        # optional average
        # g = g / len(max_idx)

        # step size
        eta = max(min_eta, eta0 * (eta_decay ** iter))

        # Exponential update
        a = a * np.exp(-eta * g)
        # renormalize
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= sum_a

        # evaluate score
        cur_score = 2.0 * n * max_val  # but max_val from old a; we need recompute after update
        # compute new max after update
        conv_new = conv_fft(a, fft_len, n)
        cur_score = 2.0 * n * conv_new.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iter}] improved score {best_score:.6f}")

        # occasional random perturbation to avoid local minima
        if iter % 500 == 0 and iter > 0:
            # restart a little
            perturb = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + perturb, 0)
            a /= a.sum()
        iter += 1

    # final hill climbing local search (similar to earlier)
    hc_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score, fft_len, n, hc_deadline, max_iter=2000)

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print("[final] evaluated score =", final_val)
    return final_seq
```

"""
subgradient‑based optimisation of the max‑auto‑convolution.

The objective to minimise (for a probability vector `a` with sum 1) is  

    F(a) = 2·n·max_k (a*a)[k] .  

`max_k (a*a)[k]` is a convex, piece‑wise linear function; its sub‑gradient at
the active shift indices `S = {k | (a*a)[k] = max}` is  

    g_i = Σ_{k∈S} a_{k-i}      (i.e. the sum of the coefficients that
                                 participate in the maximal convolutions).

A simple projected (exponential) sub‑gradient step  

    a ← a · exp(-η·g) ,   a ← a/∑a

keeps the iterates non‑negative and on the probability simplex.
The algorithm starts from several sensible initialisations
(uniform, exponential decay, linear ramp, previous best, random Dirichlet)
and runs a long sub‑gradient descent with a geometrically decreasing step‑size.
After the descent a greedy hill‑climb moves a small fraction of mass between
random coordinates and keeps any strict improvement.  The best sequence found
is returned; it can be fed to ``evaluate_sequence`` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    a is assumed to have length n and sum(a)=1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep the real part and truncate to the needed length
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective ``2·n·max(a*a)`` for a probability vector.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10  # up to 10 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        max_iter -= 1

    return best, best_score


def subgradient_descent(
    a0: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 200000,
    eta0: float = 0.3,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Projected (exponential) sub‑gradient descent on the max‑convolution.
    Returns the best sequence seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution of the current iterate
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # indices where the maximum is attained (within a tiny tolerance)
        eps = max_val * 1e-12 + 1e-15
        max_idx = np.where(conv >= max_val - eps)[0]

        # sub‑gradient g_i = Σ_{k∈S} a_{k-i}
        g = np.zeros_like(a)
        for k in max_idx:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            i = np.arange(low, high + 1)
            j = k - i
            g[i] += a[j]

        # step‑size (geometric decay)
        eta = eta0 * (decay ** it)
        if eta < 1e-8:
            eta = 1e-8

        # exponential update keeps non‑negativity automatically
        a = a * np.exp(-eta * g)

        # renormalise to stay on the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional tiny random perturbation to escape flat regions
        if it % 5000 == 0 and it > 0:
            a = a * np.exp(rng.normal(scale=0.01, size=n))
            a /= a.sum()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats (its sum is 1, which is optimal for the
    internal objective).  The function respects the time budget provided.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0    # small safety margin

    # ------------------------------------------------------------------
    # problem size – reuse a previous long construction if available
    # ------------------------------------------------------------------
    n = 5000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = max(n, int(arr.size))

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # build a pool of diverse starting points (all are normalised to sum=1)
    # ------------------------------------------------------------------
    starts = []

    # 1) uniform
    starts.append(np.full(n, 1.0 / n))

    # 2) exponential decay from the left
    rate = rng.uniform(0.001, 0.02)
    a_exp = np.exp(-rate * np.arange(n))
    a_exp /= a_exp.sum()
    starts.append(a_exp)

    # 3) linear ramp (large on the left, small on the right)
    a_lin = np.linspace(1.0, 0.0, n)
    a_lin = np.maximum(a_lin, 0.0)
    a_lin /= a_lin.sum()
    starts.append(a_lin)

    # 4) previous construction if available and of the right size
    if prev is not None and arr.size == n:
        a_prev = arr.astype(float)
        a_prev /= a_prev.sum()
        starts.append(a_prev)

    # 5) a few random Dirichlet samples with different concentrations
    for conc in [0.3, 0.8, 2.0]:
        a_dir = rng.dirichlet(np.full(n, conc))
        starts.append(a_dir)

    # keep the current best (initially the uniform one)
    best_a = starts[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    # ------------------------------------------------------------------
    # run sub‑gradient descent from each start until time runs out
    # ------------------------------------------------------------------
    for idx, a_start in enumerate(starts):
        if time.time() >= deadline - 5.0:
            break
        print(f"[init] start {idx} score = {fast_score(a_start, fft_len, n):.6f}")

        a_opt, score_opt = subgradient_descent(
            a_start,
            fft_len,
            n,
            deadline,
            max_iters=150000,
            eta0=0.3,
            decay=0.9995,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[update] improved after start {idx}: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final aggressive hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=12000,
        )
        print(f"[final‑hill] score after hill‑climb = {best_score:.6f}")

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
