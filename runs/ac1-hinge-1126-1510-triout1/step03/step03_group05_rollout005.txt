def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # parameters
    n_choices = [8000, 10000, 12000, 15000]
    num_restarts = 5
    max_iter_per_restart = 8000  # may be limited by time
    eps_small = 1e-12

    best_seq = None
    best_score = float('inf')

    for restart in range(num_restarts):
        # check time
        if time.time() > deadline:
            break

        # generate random shape
        n = random.choice(n_choices)

        family = random.choice(['power','exp','beta','mixture'])
        if family == 'power':
            p = random.uniform(0.05, 1.5)
            a = generate_power_law(n, p)
        elif family == 'exp':
            lam = random.uniform(1e-4, 5e-2)
            a = generate_exponential(n, lam)
        elif family == 'beta':
            alpha = random.uniform(0.05, 2.0)
            beta = random.uniform(0.05, 2.0)
            a = generate_beta_shape(n, alpha, beta)
        else:  # mixture of two power laws
            p1 = random.uniform(0.05, 1.5)
            p2 = random.uniform(0.05, 1.5)
            w = random.uniform(0.2, 5.0)
            a = generate_two_sided_power_law(n, p1, p2, w, w)

        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < eps_small:
            a = np.full(n, 1.0/n)
        else:
            a = a / total

        # start optimization using Adam on logits
        # convert to logits y = log a (add epsilon)
        y = np.log(a + 1e-16)

        # Adam state
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        lr0 = 0.5   # initial learning rate
        iteration = 0
        best_local_seq = a.copy()
        best_local_score = fast_score(a)

        while True:
            now = time.time()
            if now > deadline:
                break
            if iteration >= max_iter_per_restart:
                break
            iteration += 1

            # get a from y via softmax
            y_shifted = y - np.max(y)  # for stability
            exp_y = np.exp(y_shifted)
            a = exp_y / exp_y.sum()

            # compute convolution
            n = a.shape[0]
            fft_len = next_pow2(2*n)
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
            M = conv.max()
            cur_score = 2.0 * n * M  # sum=1

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()

            # compute active set
            eps_act = 1e-12
            active_idx = np.where(conv >= M - eps_act)[0]
            # compute subgradient g = sum_k in active 2 * shifted a
            # We'll implement using loops if active set small:
            if active_idx.size <= 10:
                g = np.zeros_like(a)
                for k in active_idx:
                    i_start = max(0, k - (n - 1))
                    i_end = min(n - 1, k)
                    i_range = np.arange(i_start, i_end + 1)
                    j_range = k - i_range
                    g[i_range] += 2.0 * a[j_range]
            else:
                # convolution of a with mask
                mask = np.zeros(2*n-1, dtype=np.float64)
                mask[active_idx] = 1.0
                # convolve via FFT
                fm = np.fft.rfft(mask, n=fft_len)
                g = np.fft.irfft(fa * fm, n=fft_len)[:n] * 2.0

            # gradient w.r.t y via softmax Jacobian
            ag = np.dot(a, g)  # scalar
            grad_y = a * (g - ag)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** iteration)
            v_hat = v / (1 - beta2 ** iteration)
            lr = lr0 / (np.sqrt(iteration) + 1.0)
            y = y - lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # occasional random jitter
            if iteration % 2000 == 0:
                sigma = 0.01 / np.sqrt(n)
                y += sigma * np.random.randn(n)

        # after loop, evaluate best_local_seq using fast_score
        local_score = fast_score(best_local_seq)
        print(f"[restart {restart+1}] best local score {local_score:.6f} length {best_local_seq.size}")

        if local_score < best_score:
            best_score = local_score
            best_seq = best_local_seq.copy()
            print(f"   >>> new global best {best_score:.6f}")

    # final verification with given evaluate_sequence
    if best_seq is None:
        # fallback: uniform
        n = 10000
        best_seq = np.full(n, 1.0/n)
        best_score = fast_score(best_seq)

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final evaluated score: {final_score:.6f}")
    return final_seq

* **Random initialisation** – generate a few candidate shapes (power‑law,
  exponential, two‑sided power‑law, Beta‑pdf) of moderate length
  (≈10k–15k).  One candidate is the previously discovered sequence
  `height_sequence_1` if it exists.

* **Soft‑max re‑parameterisation** – the coefficients are represented as
  `a_i = softmax(y_i)`.  This guarantees non‑negativity and the simplex
  constraint without an explicit projection.

* **Projected sub‑gradient with Adam** – for the current `a` we compute the
  linear convolution `c = a * a` via an FFT, find the maximal value `M`
  and the (typically tiny) active set of indices where `c_k` is within
  `1e‑12` of `M`.  The sub‑gradient of `M` w.r.t. `a` is

      g_i = 2 * sum_{k∈active} a_{k-i}   (indices outside the support are ignored)

  The gradient w.r.t. the log‑variables is
      ∇_y = a * (g - ⟨a,g⟩),

  and an Adam optimiser (β₁=0.9, β₂=0.999) with a slowly decaying step
  size `lr / (sqrt(t)+1)` is used.  Every few thousand iterations a
  small Gaussian jitter is added to `y` to escape plateaus.

* **Time‑budget handling** – the total budget (≈1000 s) is split between
  several independent restarts; each restart runs as many Adam steps as
  time permits.  The best sequence seen overall is finally verified with
  the provided `evaluate_sequence` function.

The code is fully self‑contained, uses only NumPy and the supplied
`evaluate_sequence`, and respects the 2‑CPU, no‑I/O constraints.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(seq,seq)) / (sum(seq)**2) using an FFT.
    The sequence must be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Core optimiser (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on `a` (initialised from ``a0``) to minimise
    the quantity 2·n·max(conv(a,a)).  Returns the best sequence encountered
    and its fast_score.
    """
    n = a0.shape[0]
    # initialise logits (log‑softmax variables)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ---------- soft‑max ----------
        y_shift = y - np.max(y)          # for numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # always lies on the simplex

        # ---------- convolution ----------
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- sub‑gradient ----------
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        g = np.zeros_like(a)

        # active set is usually tiny; use explicit loops
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ---------- gradient w.r.t. logits ----------
        ag = np.dot(a, g)                     # scalar  ⟨a,g⟩
        grad_y = a * (g - ag)                 # Jᵀ·g for soft‑max Jacobian

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low
    ``evaluate_sequence`` value.  The routine respects the 1000 s time
    budget, prints progress, and returns the best sequence found as a
    plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 0) build a pool of random initial candidates (including the
    #    previously discovered one if it exists)
    # ------------------------------------------------------------------
    init_candidates = []

    # previous best, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s < 1e-12:
                prev = np.full(prev.shape, 1.0 / prev.shape[0])
            else:
                prev = prev / s
            init_candidates.append(prev.copy())
            print("[info] using previously discovered sequence as a start")
    except Exception:
        pass

    # a modest default (two‑sided power‑law) to guarantee at least one start
    default_len = 10000
    default_seq = generate_two_sided_power_law(default_len, 0.55, 0.55, 1.0, 1.0)
    default_seq = np.maximum(default_seq, 0.0)
    default_seq = default_seq / default_seq.sum()
    init_candidates.append(default_seq.copy())

    # ------------------------------------------------------------------
    # 1) Randomised family search – fill the pool up to ~8 seeds
    # ------------------------------------------------------------------
    length_choices = [8000, 10000, 12000, 15000]
    families = ["power", "exp", "twosided", "beta"]
    desired_pool = 8

    while len(init_candidates) < desired_pool and time.time() < deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 1.5)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.05, 1.5)
            pr = random.uniform(0.05, 1.5)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        else:  # beta
            a = random.uniform(0.05, 2.0)
            b = random.uniform(0.05, 2.0)
            seq = generate_beta_shape(n, a, b)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        init_candidates.append(seq)

    # ------------------------------------------------------------------
    # 2) Optimise each candidate with Adam (time‑budget split)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    per_candidate = remaining / len(init_candidates)
    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(f"[optim] start candidate #{idx+1}/{len(init_candidates)} "
              f"(len={a0.size}) – time budget {per_candidate:.1f}s")
        seq, val = adam_optimize(a0, per_candidate, time.time())
        print(f"[optim] candidate #{idx+1} best score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback to a simple uniform distribution
        n = 10000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
