"""
Improved search for a low‑score non‑negative step sequence.

The algorithm consists of four phases:

1.  Random restarts (various shapes) + a “anti‑correlated” permutation
    (large values are placed opposite to other large values).  Each restart
    runs a continuation L‑BFGS on the smooth log‑sum‑exp surrogate of the
    max‑convolution, with a long schedule of decreasing smoothing factors.
    After every L‑BFGS stage a short hill‑climb is performed.

2.  Projected sub‑gradient descent with back‑tracking line‑search (the
    most effective stage in our experiments).

3.  Simulated annealing that transfers a random fraction of mass between
    two coordinates; temperature decays slowly and larger moves are
    allowed.

4.  A final long hill‑climb (pure greedy mass‑transfer) to polish the
    solution until the time budget ends.

All phases respect the simplex (non‑negative, sum = 1) and use FFT‑based
convolutions for O(n log n) speed.  The best sequence found before the
deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. the logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) where the gradient is taken w.r.t. y.
    The objective equals 2·n·smooth_max(a*a) with a = softmax(y).
    """
    a = softmax(y)

    # a * a convolution (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. Conv entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution w with reversed a (needed for chain rule)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that aligns with the original a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimisation
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_factors,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each smoothing factor in lam_factors, starting each
    stage from the result of the previous one.  Return the best a and its
    score observed during the run.
    """
    # initialise from the start distribution
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # any numerical trouble – just skip this lam
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # a quick hill‑climb after each lam to escape tiny local bumps
        hc_deadline = min(deadline, time.time() + 0.2)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=800)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (core of Phase 2)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    The iterate always stays in the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # current full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # active set

        # sub‑gradient of max(convolution)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(7):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)  # still move a little

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (Phase 4)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two indices i, j and move a small random fraction from i to j.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the source mass can be moved
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation (Phase 3)
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a temperature that decays slowly.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.06                     # start temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving up to 15 % of the source mass
        delta = cur_a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)

        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.93
            T = max(T, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: anti‑correlated permutation of a sorted vector
# ----------------------------------------------------------------------
def anti_permute_desc(v_desc: np.ndarray) -> np.ndarray:
    """
    Given a descendingly sorted vector, return a permutation where the
    largest values occupy the left part and the smallest values the right
    part (alternating high‑low).  This tends to minimise the overlap of
    large values when the vector is convolved with itself.
    """
    n = v_desc.size
    out = np.empty_like(v_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(v_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n]
            a0[a0 < 0] = 0.0
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            n = 2000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    phase1_end = start_time + 0.60 * budget_s   # 60 % for random restarts + L‑BFGS
    phase2_end = start_time + 0.85 * budget_s   # next 25 % for sub‑gradient
    phase3_end = deadline                       # remaining 15 % for annealing

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts + L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_factors = [0.5, 0.4, 0.3, 0.22, 0.15, 0.10,
                  0.07, 0.05, 0.03, 0.015, 0.008, 0.004]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        # -------- generate a base shape ---------------------------------
        mode = rng.choice(['random', 'exp', 'power', 'edge'])
        if mode == 'random':
            base = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0008, 0.015)
            base = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.0)
            base = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:   # edge‑heavy
            edge_frac = rng.uniform(0.01, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            base = np.full(n, a_center)
            base[:m_edge] = a_edge
            base[-m_edge:] = a_edge

        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base = np.full(n, 1.0 / n)
        base /= base.sum()

        # -------- produce a few permutations of the base -----------------
        candidates = [base]
        # anti‑correlated order (large opposite small)
        v_desc = np.sort(base)[::-1]
        candidates.append(anti_permute_desc(v_desc))
        # reverse order (just in case)
        candidates.append(base[::-1])
        # alternating high‑low without sorting
        alt = base.copy()
        idx = np.arange(n)
        mask = (idx % 2 == 0)
        alt[mask] = base[mask][::-1]   # simply shuffle even positions
        candidates.append(alt)

        # -------- run optimisation on each start vector -----------------
        for start_vec in candidates:
            if time.time() >= phase1_end:
                break
            a_opt, score_opt = lbfgs_continuation(start_vec,
                                                  lam_factors,
                                                  maxiter_per_lam,
                                                  phase1_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
            if score_opt < best_score:
                best_score, best_a = score_opt, a_opt.copy()
                print(f"[restart] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (the most potent phase)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.3)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as the remaining time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the sequence
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
