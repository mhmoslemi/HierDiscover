```
def generate_mixture(n, shape_type, shape_param, mix_alpha):
    if shape_type == 'exp':
        w = np.exp(-shape_param * np.arange(n))
    elif shape_type == 'power':
        idx = np.arange(1, n+1)
        w = idx ** (-shape_param)
    elif shape_type == 'cos2':
        cos2 = np.cos(np.pi * np.arange(n) / (n-1)) ** 2
        w = 1 + shape_param * cos2
    elif shape_type == 'exp_sym':
        w = np.exp(-shape_param * np.arange(n)) + np.exp(-shape_param * np.arange(n-1, -1, -1))
    # other shapes
    w_norm = w / w.sum()
    uniform = np.full(n, 1.0 / n)
    a = (1 - mix_alpha) * uniform + mix_alpha * w_norm
    # ensure sum 1
    a = a / a.sum()
    return a
```

"""
Improved search for a low‑value step sequence.

The evaluation metric  E(a)=2·n·max(a*a)/(sum a)^2  is scale‑invariant,
so we restrict the search to the probability simplex (∑a=1, a_i≥0).

Idea
----
1.  Very cheap parametric search: generate families of one‑parameter
    distributions (exponential, power‑law, cosine‑squared) and mix them
    with the uniform distribution.  The mixture is evaluated with a fast
    FFT based convolution – this gives thousands of candidates in a few
    seconds and often already beats the previous best bound.

2.  The best candidate is refined by a projected sub‑gradient method
    (directly minimises the true max‑convolution) followed by a short
    L‑BFGS optimisation of a smooth log‑sum‑exp surrogate.

3.  Finally a greedy hill‑climb moves a small amount of mass between
    two coordinates while keeping the sum fixed; it polishes the result.

All stages respect the supplied time budget (default 1000 s) and print
progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """2·n·max(a*a) – a is already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exact max)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active mask (indices attaining the max, with tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to length n
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best iterate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps ∑a = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Transfer a tiny fraction of mass between two random positions."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective (log‑sum‑exp) and its gradient w.r.t. y.
    Returns f = 2·n·smooth_max(conv) and ∇_y f.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # compute g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y
