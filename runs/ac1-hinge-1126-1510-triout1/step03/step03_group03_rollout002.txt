"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear convolution.  The algorithm works in two stages.

1. **Broad random sampling** – a few thousand candidates are drawn from a
   diverse collection of analytic families (power‑law, exponential,
   two‑sided power‑law, arcsine, Hann, Gaussian, Beta‑like) with several
   lengths.  The best candidate measured with a fast FFT based score is kept.

2. **Soft‑max gradient descent (mirror descent)** – starting from the best
   candidate we repeatedly
      * compute the convolution,
      * build a soft‑max weight `w` over the lag indices (temperature `β`
        starting small and slowly increased, so the weight concentrates on
        the current peak of the convolution),
      * compute the gradient of the soft‑max objective
        `J_β(a)=log‑sum‑exp(β·conv)/β` w.r.t. `a`.  Using the convolution theorem
        this gradient is a cross‑correlation and is obtained with two FFTs.
      * perform a multiplicative (mirror‑descent) update
        `a_i ← a_i·exp(-η·grad_i)` and project back onto the probability simplex.
      * occasionally add a tiny random jitter to escape plateaus.

The update uses a decaying step size `η` and a slowly increasing temperature
`β`.  Periodically the fast score is recomputed and the best sequence so far
is stored.  The whole routine respects the 1000 s budget (a small safety
margin is kept).  Finally the best sequence is returned as a plain Python
list; it can be handed to the provided `evaluate_sequence` function.

The method is different from the previous multiplicative sub‑gradient on the
single peak – it uses a smooth approximation of the max and a full gradient,
which usually leads to a lower final bound.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2) via FFT."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{-1/2}: a_i ~ 1/√i (flattening conv)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_hann(n: int) -> np.ndarray:
    """Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the window. sigma_frac ∈ (0,0.5) is σ / n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_like(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on the index interval."""
    t = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n  # avoid zeros
    return (t ** (a - 1)) * ((1 - t) ** (b - 1))

def refine_sequence(
    a: np.ndarray,
    deadline: float,
    start_time: float,
    eval_interval: int = 2000,
    jitter_interval: int = 5000,
) -> np.ndarray:
    """
    Perform mirror‑descent on the soft‑max approximation of max(conv).
    Returns the best sequence found before `deadline`.
    """
    n = a.shape[0]
    L = next_pow2(3 * n)                # length for convolution / correlation FFTs

    # hyper‑parameters
    eta0 = 2000.0                        # initial step‑size (multiplicative)
    eta_decay = 5000.0                   # controls decay of η
    beta = 0.5                           # start temperature
    beta_max = 200.0
    beta_mul = 1.002                     # geometric increase each iteration
    jitter_scale = 0.015                 # relative jitter magnitude

    best = a.copy()
    best_score = fast_score(best)

    iteration = 0
    while time.time() < deadline - 0.5:
        iteration += 1

        # ----- convolution of a with itself -----
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

        # ----- soft‑max weights over lags -----
        scaled = beta * conv
        # numeric stability
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ----- gradient of the soft‑max objective -----
        Wa = np.fft.rfft(w, n=L)
        cross = np.fft.irfft(np.conj(Fa) * Wa, n=L)[:n]   # linear cross‑correlation
        grad = 2.0 * cross                               # ∂J/∂a_i

        # ----- multiplicative mirror‑descent update -----
        eta = eta0 / (1.0 + iteration / eta_decay)
        a = a * np.exp(-eta * grad)

        # ----- simplex projection (ensures non‑negativity & sum‑1) -----
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

            a = np.clip(a + noise, 0.0, None)
            a_sum = a.sum()
            if a_sum <= 0:

            else:
                a = a / a_sum

                best = a.copy()
                print(

                    f"new best score={cur_score:.6f}"
                )
    return best

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_hann(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)   # σ / n
            cand = generate_gaussian(n, sigma_frac)

        elif family == "beta":
            a_param = random.uniform(0.5, 5.0)
            b_param = random.uniform(0.5, 5.0)
            cand = generate_beta_like(n, a_param, b_param)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new random best ({family}) n={n} "
                f"score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Soft‑max mirror‑descent refinement (remaining time)
    # ------------------------------------------------------------------
    refined = refine_sequence(
        best_seq.copy(),
        deadline,
        start,
        eval_interval=2000,
        jitter_interval=5000,
    )
    # keep the best seen during refinement
    cand_score = fast_score(refined)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = refined.copy()
        print(f"[{time.time() - start:.1f}s] post‑refine best score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
