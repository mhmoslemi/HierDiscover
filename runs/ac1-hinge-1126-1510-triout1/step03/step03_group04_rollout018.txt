from scipy.optimize import linprog

def lp_refine(a, M, t):
    n = len(a)
    A = sp.diags(a, np.arange(n), shape=(2*n-1, n), format='csr')
    b = np.full(2*n-1, M, dtype=float)
    bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return a  # fall back
    b_opt = res.x
    # scale to sum 1
    b_opt_sum = b_opt.sum()
    if b_opt_sum == 0:
        return a
    g = b_opt / b_opt_sum
    a_new = (1 - t) * a + t * g
    return a_new

"""
Propose a candidate step‑sequence with a low evaluation score.

Algorithm
---------
1. **Initialisation** – Load the previous best sequence (if any) and
   normalise it to a probability simplex, otherwise start from a uniform
   distribution of length n≈2000.

2. **Linear‑program (LP) refinement** – For the current sequence a,
   compute M = max(a*a).  Solve the linear programme

        maximise   Σ_j b_j
        subject to  (a * b)[k] ≤ M   for every k = 0,…,2n−2
                    b_j ≥ 0

   (convolution with a is linear in b, so the constraints are linear).
   The optimiser returns a vector b with the largest possible total mass
   under the same convolution bound.  After normalising b we move a a
   fraction t towards b (projected back onto the simplex).  Repeating
   this step strictly decreases the max‑convolution and yields a very
   efficient reduction of the score.

3. **Projected sub‑gradient descent** – A few iterations of the classic
   sub‑gradient method on the true max‑convolution fine‑tune the sequence.

4. **Simulated annealing** – Non‑local stochastic moves (mass transfer
   between two positions) help escape any remaining local minima.

5. **Final greedy hill‑climb** – A fast deterministic polish that repeatedly
   transfers a tiny amount of mass between two coordinates while keeping the
   sum fixed.

All convolutions are performed via FFT (O(n log n)).  The whole routine
obeys the 1000‑second budget (default) and prints progress information.
The returned list contains non‑negative heights ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Accepts improving moves always, uphill moves with a
    temperature‑dependent probability.  Temperature decays geometrically.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, t: float) -> np.ndarray:
    """
    One LP refinement step.
    maximise Σ b_j    subject to (a * b)[k] ≤ M   for all k,
    where M = max(a*a).  The result b is normalised to sum 1 and a is moved
    a fraction t towards b (projected on the simplex).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)          # full linear convolution, size 2n‑1
    M = conv.max()

    # Build Toeplitz matrix A where A_{k,j} = a_{k-j} (zero when out of range)
    A = sp.diags(a, offsets=np.arange(n), shape=(2 * n - 1, n), format='csr')
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise sum(b)  ⇔ minimise -sum(b)
    c_obj = -np.ones(n)

    # Solve LP (highs is fast and accepts sparse matrices)
    res = opt.linprog(c=c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 8})

    if not res.success:
        # fallback: keep the current a
        return a

    b_opt = res.x
    s = b_opt.sum()
    if s == 0:
        return a
    g = b_opt / s                      # normalise to a probability vector

    # convex combination and projection onto simplex
    a_new = (1 - t) * a + t * g
    return simplex_projection(a_new)

# Main entry point
# ----------------------------------------------------------------------
