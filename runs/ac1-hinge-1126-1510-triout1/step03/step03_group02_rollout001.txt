"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1.  Choose a problem size `n` (≥ 4000).  If a previous construction
    (`height_sequence_1`) is available we reuse its length.
2.  Represent the coefficients `a_i ≥ 0` via a soft‑max of unconstrained
    variables `y` (so the simplex constraint Σ a_i = 1 is satisfied).
3.  Minimise a *smooth* approximation of the peak of the auto‑convolution
    using many L‑BFGS runs with a geometric continuation schedule for the
    smoothing parameter λ.  Several random initialisations (uniform,
    exponential, power‑law, edge‑biased, perturbation of the current best)
    are tried; the best result is kept.
4.  A **targeted hill‑climb** is then performed.  The shift `k_max`
    where the current convolution attains its maximum is located; mass is
    repeatedly moved away from the indices that contribute most to this
    peak (chosen proportionally to a_i·a_{k_max‑i}) to a random destination.
    The move is accepted only if the (exact) convolution peak strictly
    decreases.
5.  A final high‑precision L‑BFGS pass with very small λ and a last
    aggressive hill‑climb are performed until the time budget expires.
6.  The resulting non‑negative list is returned.  It is already normalised
    to sum = 1; `evaluate_sequence` will rescale it automatically.

The method differs from the previous baseline by (i) using a finer
λ‑schedule, (ii) increasing the number of L‑BFGS iterations per stage,
(iii) adding a peak‑focused hill‑climb, and (iv) employing a larger
population of restarts.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns the first (2·n‑1) real samples.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def smooth_max_objective(y: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         fft_len_g: int,
                         lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the peak of a*a and its gradient w.r.t. y.

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for the forward convolution and the
        auxiliary convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width λ = max_c·lam_factor.

    Returns
    -------
    f : float
        Smoothed objective value (≈ 2·n·max(a*a)).
    grad_y : ndarray
        Gradient w.r.t. y (same shape as y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-15  # avoid exact zero

    # smooth max = max_c + lam * log( Σ exp((c_i - max_c)/lam) )
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z  # d smooth_max / d conv_i

    # gradient w.r.t. a : g_i = Σ_j a_j * w_{i+j}
    # This is a convolution of a (reversed) with w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max:
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_factors: np.ndarray,
                      maxiter_stage: int = 800) -> np.ndarray:
    """
    Continuation L‑BFGS: run L‑BFGS for each λ in `lam_factors`,
    feeding the final point of a stage as the start of the next.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=smooth_max_objective,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_stage,
                                "gtol": 1e-14,
                                "disp": False})
        y = res.x
    return y


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 12000,
                        max_move_frac: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that concentrates on the shift `k_max` where the
    convolution attains its maximum.  Mass is moved away from the
    indices that contribute most to this peak.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (exact, not FFT‑based – size still modest)
        conv = np.convolve(best, best)
        k_max = int(np.argmax(conv))
        # overlapping index range for this shift
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx

        # contribution of each pair (i, j)
        contrib = best[idx] * best[jdx]
        total = contrib.sum()
        if total == 0.0:
            # fallback to uniform random move
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(idx, p=probs)

        # source index i must have some mass
        if best[i] <= 1e-12:
            continue

        # move a fraction of its mass to a random destination not in the
        # current overlapping block (to avoid immediate re‑increase)
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        # choose destination far from the current block
        dest_candidates = np.setdiff1d(np.arange(n), np.arange(i_start, i_end + 1))
        if dest_candidates.size == 0:
            # all indices belong to block – pick any different index
            dest = rng.integers(0, n)
            while dest == i:
                dest = rng.integers(0, n)
        else:
            dest = rng.choice(dest_candidates)

        best[i] -= delta
        best[dest] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep new configuration
        else:
            # revert move
            best[i] += delta
            best[dest] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The search stops shortly before this limit.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (its sum will be 1 after
        normalisation; `evaluate_sequence` rescales it as needed).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 5000
    # reuse previous construction if it exists and is long enough
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = max(n, int(arr.size))

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise with uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)  # log‑softmax parameters
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # A geometric λ‑schedule that goes far into the non‑smooth regime
    lam_factors = np.geomspace(0.5, 0.001, num=30)
    maxiter_stage = 800  # more iterations per stage than the baseline
    iteration = 0

    # ------------------------------------------------------------------
    # multi‑restart L‑BFGS loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick an initialisation strategy
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # a few larger entries at the left edge, the rest tiny
            m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(y_start=y0,
                                  n=n,
                                  fft_len_conv=fft_len_conv,
                                  fft_len_g=fft_len_g,
                                  lam_factors=lam_factors,
                                  maxiter_stage=maxiter_stage)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 hc_deadline,
                                                 max_iter=8000,
                                                 max_move_frac=0.12)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final deep refinement with tighter λ
    # ------------------------------------------------------------------
    # continue the same continuation schedule but with many more L‑BFGS steps
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(fun=smooth_max_objective,
                       x0=best_y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 1500,
                                "gtol": 1e-14,
                                "disp": False})
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.5g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 deadline,
                                                 max_iter=15000,
                                                 max_move_frac=0.15)
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        # If the harness didn't import evaluate_sequence, we still return
        print(f"[final] evaluate_sequence failed: {exc}")
    return final_seq
