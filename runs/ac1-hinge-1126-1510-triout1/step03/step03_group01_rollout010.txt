"""
Propose a low‑evaluation step sequence.

Algorithm
---------

* Work on the probability simplex (∑a = 1) via the soft‑max
  parameterisation a_i = softmax(y_i).
* Approximate the L∞‑norm of the auto‑convolution with a
  smooth log‑sum‑exp (controlled by a “lam’’ factor) and minimise
  the resulting smooth objective with L‑BFGS‑B.
* After every successful L‑BFGS run a **targeted hill‑climb** is
  performed: the current maximal convolution entry is identified,
  we locate the pair (i,j) that contributes most to this entry and
  move a small fraction of mass from the larger partner to a random
  index that does not belong to the current peak.  The move is kept
  only if the (fast) convolution max decreases.
* The whole procedure is repeated with many random restarts until the
  allocated time budget (≈1000 s) expires.  The best solution found
  is finally polished with a finer continuation schedule and a final
  targeted hill‑climb before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    The result is identical to evaluate_sequence() up to the sum‑square
    normalisation.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and a sum of exactly 1.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. the
    unconstrained vector y (soft‑max space).  The smoothing parameter
    is lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth maximum of the convolution -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z   # d smooth_max / d conv_i  (shape 2n‑1)

    # compute g_i = Σ_j a_j * w_{i+j}
    # this is a convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part corresponding to indices 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # a few heavy masses at the two ends, lighter interior
        m_edge = max(1, int(n * rng.uniform(0.008, 0.025)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set (pairwise sums distinct) – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # avoid double of cand
            if cand * 2 in sums:
                continue
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.3):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:   # "best" – handled outside
        raise ValueError(f"unknown mode {mode}")
    return y0


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that focuses on the currently most offending
    convolution entry.  At each step we locate the pair (i,j) that
    contributes most to the max entry and move a small fraction of the
    larger partner's mass to a random index outside the peak's support.
    The move is kept only if the fast score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        conv = conv_fft(best, fft_len, n)
        k = int(conv.argmax())                     # index of the current peak

        # feasible i values are those with 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)

        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        prod = best[i_vals] * best[j_vals]

        # pick the pair with the largest product contribution
        idx = int(prod.argmax())
        i = int(i_vals[idx])
        j = int(j_vals[idx])

        # source is the heavier of the two
        src = i if best[i] >= best[j] else j

        # pick a target index **outside** the interval [i_min, i_max]
        # (these indices do not directly affect the current peak)
        while True:
            tgt = rng.integers(0, n)
            if tgt < i_min or tgt > i_max:
                break

        # move a modest fraction of the source mass
        if best[src] <= 0.0:
            continue
        delta = best[src] * rng.random() * 0.07   # up to 7 % of src
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[tgt] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta

    return best, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple generic hill‑climb (random mass transfers) used for occasional
    polishing when time is abundant.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS‑B optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The routine stops a couple of seconds
        before the deadline to guarantee a return value.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a reasonably large n gives a rich search space.
    # ------------------------------------------------------------------
    n = 5000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 1000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise with a uniform distribution (the known baseline).
    # If a global construction is available we also use it as a seed.
    # ------------------------------------------------------------------
    uniform_a = np.full(n, 1.0 / n)
    best_a = uniform_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    # If a previous construction exists, seed it.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=float)
            if cand.ndim == 1 and cand.size == n:
                cand = np.clip(cand, 0, None)
                total = cand.sum()
                if total > 0.0:
                    cand /= total
                    best_a = cand.copy()
                    best_y = np.log(best_a + 1e-12)
                    best_score = fast_score(best_a, fft_len_conv, n)
                    print("[init] using external seed, fast score =", best_score)
        except Exception:
            pass

    print("[init] n =", n, "uniform fast score =", fast_score(uniform_a, fft_len_conv, n))

    # ------------------------------------------------------------------
    # Optimisation schedule (coarse → fine smoothing)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 0.001, num=14))

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            # small Gaussian perturbation of the current best
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=350,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- focused hill‑climb -------------------------------------------------
        # We give the climb a short time slice (≈0.6 s) to keep the overall loop fast.
        hc_deadline = min(deadline, time.time() + 0.55)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iters=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement: finer smoothing schedule and a last targeted climb.
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-5, num=20))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # One more targeted polishing with whatever time remains.
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, final_deadline, max_iters=5000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation (the harness will clip/scale if necessary)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print("[final] evaluated score =", final_val)
    return final_seq
