"""
Propose a low‑score step sequence.

Algorithm
---------
1. **Parameterisation** – Work on the log‑softmax parameters `y`
   (so the simplex constraint is always satisfied).

2. **Continuation optimisation** – For a decreasing smoothing
   parameter `lam` (log‑sum‑exp approximation of the max) run a
   limited‑memory BFGS optimiser (`scipy.optimize.minimize`) from many
   random starts.  Each stage finishes with a short hill‑climb.

3. **Projected sub‑gradient descent** – Refine the best candidate
   using the exact sub‑gradient of the max‑convolution (mask of the
   active indices) and a simple back‑tracking line‑search.

4. **Fine‑grained hill‑climb** – Randomly move a small fraction of
   mass between two coordinates, accepting only improving moves.
   The remaining time is spent here.

All convolutions are performed with FFTs (real → O(n log n) speed).
The total wall‑clock budget is split automatically; the best feasible
sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max computation
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimisation (multiple restarts)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing parameter `lam` in `lam_schedule`,
    starting each stage from the result of the previous stage.
    Returns the best distribution found and its fast_score.
    """
    # Initialise log‑softmax parameters
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] lam={lam:.5f} new best score={best_score:.6f}")

        # a short hill‑climb after each lam improvement
        sub_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        sub_deadline,
                                        max_iter=1200)

    return best_a, best_score

# ----------------------------------------------------------------------
# Projected sub‑gradient descent with simple back‑tracking line search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent on the
    exact max‑convolution.  The simplex constraint is enforced by a
    projection after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current full convolution and its active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (see paper)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            # still move a little to avoid stagnation
            a = simplex_projection(a - step * g)

        it += 1
    return best_a, best_score

# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    repeatedly move a small random fraction of mass from one coordinate
    to another and keep the move only if it strictly improves the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # Phase 1 – continuation L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
    maxiter_per_lam = 400

    while time.time() < phase1_end:
        # Choose a random start distribution
        mode = rng.choice(['random', 'exp', 'power', 'sin', 'edge'])
        if mode == 'random':
            a_tmp = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0004, 0.015)
            a_tmp = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.2)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == 'sin':
            # a half‑sine window, strictly positive
            a_tmp = np.sin(np.pi * (np.arange(n) + 0.5) / n) + 0.01
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.006, 0.035)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge

        a_tmp /= a_tmp.sum()
        a_opt, score_opt = lbfgs_continuation(a_tmp,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[phase1] new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.2)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb / fine tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=8000)
        print(f"[final hill‑climb] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to python list and evaluate with the external checker
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
