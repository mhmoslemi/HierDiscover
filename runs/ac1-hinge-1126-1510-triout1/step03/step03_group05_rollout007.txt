"""
Propose the best non‑negative step‑function (a list of heights) within a
given time budget.

Algorithm
---------

1. **Broad stochastic search** – Randomly generate a large pool of
   candidate shapes (power‑law, exponential, two‑sided power‑law,
   Beta‑distribution and a sine‑shape).  All candidates are normalised
   to unit L¹‑norm and scored with a fast FFT based “score”:
        score = 2·n·max(conv(a,a))          (n = len(a))

   The ten best candidates (lowest scores) are kept as an elite set.

2. **Smoothed exponentiated‑gradient refinement** – Each elite candidate
   is refined independently on the simplex by minimising the maximal
   entry of the auto‑convolution.
   * The exact objective (max‑convolution) is approximated by a
     smoothened version using a soft‑max (log‑sum‑exp) with a
     temperature β that is slowly increased.
   * The gradient of the smooth objective can be written as a
     convolution of the current weight vector with the reversed
     candidate; this is computed efficiently by FFT.
   * An *exponentiated‑gradient* (multiplicative) update
     ``a_i ← a_i·exp(−η·g_i)`` is performed, followed by a projection
     onto the probability simplex (simple renormalisation).
   * Step‑size η decays like 1/√t, momentum is added by an Adam‑style
     moving‑average of the gradients, and occasional Gaussian noise
     is injected to escape plateaus.

3. **Final verification** – The globally best refined sequence is
   converted to a plain Python list and returned.  The supplied
   ``evaluate_sequence`` routine is called once at the end to obtain
   the official score.

The method exploits the fact that the objective is scale‑invariant
(only the shape of the sequence matters) and works well for lengths
of a few × 10⁴ within the 1000 s time limit.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of   2·n·max(conv(seq,seq)) / (sum(seq)²).

    The sequence is assumed non‑negative and sum(seq) > 0.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(‑lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(‑p_left) + w_right·(n‑i)^(‑p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‐pdf shape:
        a_i ∝ (i+1)^(α‑1)·(n‑i)^(β‑1).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sine_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1)), a strictly positive sine shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.sin(np.pi * i / (n + 1))


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient refinement
# ----------------------------------------------------------------------


def refine_exponential(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.8,
    beta_start: float = 5.0,
    beta_max: float = 60.0,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate on the simplex using a smoothed gradient of the
    max‑convolution.  Returns the best sequence found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]

    # Normalise to the simplex
    a = project_simplex(a)

    # FFT lengths for the two convolutions we need
    fft_len2 = next_pow2(2 * n)          # for a*a
    fft_len3 = next_pow2(3 * n)          # for w * a_rev

    # Adam‑style moving averages
    m = np.zeros_like(a)                # 1st moment
    v = np.zeros_like(a)                # 2nd moment
    beta1, beta2 = 0.9, 0.999

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    it = 0
    while time.time() < start_time + time_limit:
        it += 1

        # ---- current convolution and objective ----
        fa = np.fft.rfft(a, n=fft_len2)
        conv = np.fft.irfft(fa * fa, n=fft_len2)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M          # sum(a)=1 ⇒ fast_score = 2·n·M

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # ---- temperature schedule (soft‑max) ----
        beta = beta_start + (beta_max - beta_start) * min(1.0, it / 20000.0)
        # weights w_k = exp(beta·(conv_k ‑ M))  ∈ (0, 1]
        w = np.exp(beta * (conv - M))

        # ---- gradient of the smoothed max (vectorized via FFT) ----
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len3)
        fa_rev = np.fft.rfft(a_rev, n=fft_len3)
        g_full = np.fft.irfft(fw * fa_rev, n=fft_len3)[: 3 * n - 2]
        # central n entries give ∂/∂a_i
        start = n - 1
        g = 2.0 * g_full[start : start + n]

        # ---- Adam update (multiplicative) ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate η_t = η₀ / √t  (combined with Adam’s scaling)
        eta = eta0 / (np.sqrt(it) + 1.0)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + 1e-8))

        # projection onto the simplex (just renormalise, non‑negativity preserved)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- occasional random perturbation to escape plateaus ----
        if it % 5000 == 0:
            a = a + 0.001 * np.random.randn(n)
            a = np.clip(a, 0, None)
            a = a / a.sum()

    final_score = fast_score(best_seq)
    return best_seq, final_score
