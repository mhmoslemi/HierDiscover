"""
A fast, deterministic search for a low‑evaluation step‑function.

The algorithm combines three ideas that complement each other:

1. **Smooth‑max L‑BFGS optimisation** – the classic continuation
   approach (soft‑max re‑parameterisation, geometric decay of the
   smoothing factor).  A fairly long schedule and a generous per‑stage
   iteration budget are used so that the optimiser can approach a good
   local minimum.

2. **Linear‑programming refinement** – for a given candidate f we solve
   a LP
        maximise    Σ b_j
        subject to  (f * b)[s] ≤ M   ∀ s
                    b_j ≥ 0,
   where M = max (f*f).  The solution b can carry *more* total mass than
   f while respecting the same convolution bound.  Mixing f with the
   normalised b (convex combination) provably cannot increase M and
   often lowers it.

3. **Greedy hill‑climb** – a cheap, random mass‑transfer operator that
   accepts only strict improvements.  It finishes each optimisation
   stage and also serves as a last polishing step.

The search runs many random restarts (the bulk of the time budget is
spent in the L‑BFGS phases).  The best sequence found so far is kept
and periodically perturbed; the LP refinement is applied every few
iterations.  All numbers are kept non‑negative and are finally returned
as a plain list ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
#  Utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a (real part) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of the smooth max ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}   (convolution of a reversed with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = scipy.optimize.minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # heavy mass at the two ends (≈3 % of the indices)
        m_edge = max(1, int(n * rng.uniform(0.02, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set – about √n elements
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if conflict:
                continue
            if candidate * 2 in sums:
                continue
            chosen.append(candidate)
            for c in chosen:
                sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode!r}")
    return y0


def lp_refine(a: np.ndarray, n: int, time_limit: float) -> np.ndarray:
    """
    Linear‑programming refinement:
    maximise Σ b_j  subject to (a*b)[s] ≤ M   and b ≥ 0,
    where M = max(a*a).  Returns the mixed vector (1‑t)·a + t·g,
    with t=0.5 and g = b / Σ b.
    """
    # current bound
    conv = conv_fft(a, next_pow2(2 * n - 1), n)
    M = conv.max()

    # Build the Toeplitz‑like constraint matrix efficiently.
    # Row indices are i + j, column indices are j, values are a_i.
    i_vec = np.arange(n, dtype=np.int64)
    j_vec = np.arange(n, dtype=np.int64)

    rows = (i_vec[:, None] + j_vec[None, :]).ravel()            # shape (n*n,)
    cols = np.tile(j_vec, n)                                   # shape (n*n,)
    data = np.repeat(a, n)                                     # shape (n*n,)

    A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                shape=(2 * n - 1, n))

    # Linear program: maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=float)
    ub = M * np.ones(2 * n - 1, dtype=float)

    # Solve with HiGHS (fast and robust)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"time_limit": time_limit},
    )
    if not res.success:
        return a

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return a

    g = b / sum_b                # normalise to sum = 1
    t = 0.5                      # mixing parameter
    new_a = (1.0 - t) * a + t * g
    new_a /= new_a.sum()         # insure sum exactly 1 (numerical safety)
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a step‑function sequence with a low evaluation score.

    The function respects the supplied time budget (default 1000 s) and
    returns the best sequence found.  It combines L‑BFGS continuation,
    occasional LP refinement and a final greedy polishing hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # Problem size – a moderate n keeps each iteration cheap while still
    # offering a sufficient search space.
    # ------------------------------------------------------------------
    n = 3000

    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1000:

            n = max(n, best_a.size)
        else:

    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)    # representation in soft‑max space
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # A deep continuation schedule (more stages than the reference code)
    lam_schedule = list(np.geomspace(1.0, 5e-5, num=28))

                best_y = np.log(best_a + 1e-12)

                )

        # ----------- aggressive greedy hill‑climb -------------
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_iter=3500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final refinement: a very fine continuation schedule
    # ------------------------------------------------------------------
    fine_lam = list(np.geomspace(0.3, 1e-5, num=22))
    for lam in fine_lam:
        if time.time() >= deadline:
            break
        res = scipy.optimize.minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # last greedy polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)        # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
