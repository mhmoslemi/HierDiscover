def subgradient_adam(a0, max_iters, step0, deadline):
    a = a0.copy()
    a = np.maximum(a,0)
    a /= a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_score = np.inf
    best_seq = a.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print("[adam it {}] new best={:.6f}".format(it, score))
        # Determine active set
        # take top k indices where conv close to max_b
        tolerance = 1e-6 * max_b  # absolute tolerance
        active = np.where(conv >= max_b - tolerance)[0]
        # limit to max_k
        max_k = min(active.size, 5)
        if active.size > max_k:
            # select k largest conv indices
            top_idx = np.argpartition(conv, -max_k)[-max_k:]
            active = top_idx
        # compute gradient as average of each active gradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi+1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection
        a = project_simplex(a_new)
    return best_seq, best_score

def adam_optimize(a0, max_iters, base_lr, deadline):
    a = a0.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a_sum
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_score = np.inf

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[adam it {it}] new best={score:.6f}")
        # compute active indices
        if max_b == 0:
            # all zero? break
            break
        # find top indices
        # Determine threshold: maybe 1e-6 relative
        rel_tol = 1e-8
        threshold = max_b * (1 - rel_tol)
        # we want at least one index; compute indices where conv >= max_b*(1-rel_tol)
        # Might include all if max is plateau.
        active_candidates = np.where(conv >= threshold)[0]
        if active_candidates.size > 5:
            # Choose 5 highest conv values
            # Using argpartition: get indices of top 5 conv values
            # Actually conv may have many indices with similar close to max; we pick top 5 based on conv value
            top_idx = np.argpartition(conv, -5)[-5:]
            active = top_idx
        else:
            active = active_candidates
        # compute gradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi+1, dtype=np.int64)
            # a[k-idx] shape same as idx
            grad[idx] += 2.0 * a[k - idx]
        grad = grad / max(1, len(active))

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a_new = a - base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

    # after loop, compute final score
    final_score = 2.0 * n * compute_max_conv(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()

"""
Propose a low‑score height sequence.

The algorithm mixes three ideas:

1. **Diverse initial shapes** – uniform, power‑law, exponential decay,
   Gaussian, or a noisy copy of the current best.  This helps explore
   different regions of the feasible simplex.

2. **Projected Adam optimisation** – a fast adaptive‑gradient method on the
   probability simplex.  At each iteration the gradient of the currently
   active convolution peak(s) is computed (averaged over the top‑few peaks)
   and an Adam step is taken, followed by projection onto the simplex.
   A small fraction of restarts uses an exponential‑gradient (EG) update
   which is naturally simplex‑preserving.

3. **Light stochastic refinement** – after the main descent a simulated‑annealing
   local search moves a small amount of mass between random coordinates.
   This often escapes shallow minima.

The search runs many independent restarts for several candidate
lengths (2000, 4000 and 8000) until the time budget (default 1000 s)
expires or a score below 1.503 is found.  The best sequence observed
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximum entry.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    base_lr: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adaptive‑gradient (Adam) descent on the probability simplex.
    The sub‑gradient is taken as the average of gradients of the
    top few active convolution peaks.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_score = float("inf")
    top_k = 5                     # number of peaks used for the sub‑gradient

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[adam it {it}] new best={score:.6f}")

        # --- active set -------------------------------------------------
        # pick up to `top_k` indices with largest convolution values
        if top_k >= conv.size:
            active = np.arange(conv.size, dtype=np.int64)
        else:
            active = np.argpartition(conv, -top_k)[-top_k:]

        # --- gradient (average over the active set) --------------------
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(active))

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a - step)

    # final evaluation (in case the last iterate is better)
    final_score = 2.0 * n * compute_max_conv(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (multiplicative updates)
# ----------------------------------------------------------------------
def eg_optimize(
    a0: np.ndarray,
    max_iters: int,
    eta: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exponential map.
    The update is a_i ← a_i·exp(−η·∇_i) followed by renormalisation.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s

    n = a.shape[0]
    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[eg it {it}] new best={score:.6f}")

        # gradient from the single highest peak
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative update; clamp to avoid underflow/overflow
        factor = np.exp(-eta * grad)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # optional decay of η (slow)
        eta = eta / (1.0 + 1e-4 * it)

    final_score = 2.0 * n * compute_max_conv(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random walk that moves a small amount of mass between two coordinates.
    Accepts improvements and occasional uphill moves according to a temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # decide direction of the mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # move up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # Metropolis acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature *= 0.9999
        if temperature < 1e-6:
            temperature = 1e-6

    return best_seq, best_score

            a0 += 0.02 * np.random.randn(n)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()
        else:
            # discrete Gaussian (centered)
            sigma = np.random.uniform(0.12, 0.30) * n
            centre = (n - 1) / 2.0
            idx = np.arange(n, dtype=float)
            a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
            a0 += 0.0001 * np.random.rand(n)          # tiny jitter
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Choose optimiser: Adam (≈70 %) or EG (≈30 %).
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        # Rough cost per iteration grows linearly with n
        time_per_iter = 0.0016 * (n / 4000.0)
        max_iters = int(min(30000, max(1, time_left / time_per_iter)))
        if max_iters < 5:
            break

        if np.random.rand() < 0.7:
            # Adam
            base_lr = 0.05
            seq, score = adam_optimize(a0, max_iters, base_lr, deadline)
        else:
            # Exponential‑gradient
            eta = 0.5
            seq, score = eg_optimize(a0, max_iters, eta, deadline)

        # --------------------------------------------------------------
        # Light stochastic refinement (if we still have time).
        # --------------------------------------------------------------
        time_now = time.time()
        time_rem = deadline - time_now
        if time_rem > 0.5:
            # allocate a few thousand cheap refinement steps
            steps = int(min(8000, time_rem / 0.0009))
            seq_ref, score_ref = local_refine(seq, steps, deadline)
            if score_ref < score:
                seq, score = seq_ref, score_ref

        # --------------------------------------------------------------
        # Update global best.
        # --------------------------------------------------------------
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"[global] new best score={score:.6f} (n={n})")
            # Stop early if we cross the target.
            if global_best_score < 1.503:
                print(f"Target reached: {global_best_score:.6f}")
                break

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=float)
        global_best_seq = a

    # Convert to a plain Python list as required by the harness.
    return [float(x) for x in global_best_seq.tolist()]
