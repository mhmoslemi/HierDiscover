"""
Propose a step‑function that (hopefully) beats the current best bound.

Algorithm
---------
1. **Randomised shape generation** – many candidate sequences are drawn from
   a variety of parametric families (power‑law, exponential, two‑sided,
   Beta‑pdf, sinusoid, arcsine and random mixtures).  All candidates are
   normalised to sum = 1 and quickly scored with an FFT‑based “fast_score”.
   The top‑few (≤ 8) are kept as an elite pool.

2. **Smoothed‑max gradient refinement** – each elite candidate is refined
   by projected gradient descent on the probability simplex.  The objective
   2·n·max(conv(a,a)) is approximated by a soft‑max (temperature α) which
   yields a smooth gradient:
        ∇F ≈ 2·(w * a_rev)   with  w_k ∝ exp(α·conv_k)
   (convolution performed by FFT).  The temperature α is increased
   continuously, the step‑size decays like 1/√t and Nesterov‑style momentum
   (β≈0.9) is used.  After each update the vector is symmetrised
   (a←(a+a[::-1])/2) and projected onto the simplex.

3. **Final verification** – the best sequence obtained from all refinements
   is evaluated with the provided `evaluate_sequence` function and returned
   as a plain Python list.

The whole search respects the 1000 s time budget; about 40 % of the time
is spent on the stochastic exploration and the remaining time is shared
equally among the elite candidates for refinement.
"""

import time
import random
import math

import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a non‑negative sequence with sum = 1.
    Uses an FFT‑based convolution; much faster than the full evaluator.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # convolution via FFT, zero‑padded to a power of two
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Shape generators

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sine_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n).  Positive on the whole interval."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def generate_arcsine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt( (i+1)*(n-i) ).  Peaks at the ends, small in the centre."""
    i = np.arange(1, n + 1, dtype=np.float64)
    eps = 1e-12
    return 1.0 / np.sqrt(i * (n - i + 1) + eps)


def generate_mixture(
    n: int,
    seq1: np.ndarray,
    seq2: np.ndarray,
    w: float,
) -> np.ndarray:
    """Convex combination w·seq1 + (1‑w)·seq2."""
    return w * seq1 + (1.0 - w) * seq2

# Smoothed‑max gradient refinement
# ----------------------------------------------------------------------


def refine_sequence_smooth(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    momentum: float = 0.9,
    eta0: float = 1.0,
    alpha_start: float = 5.0,
    alpha_end: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex using a soft‑max approximation
    of the max‑convolution.  Returns the best sequence found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)                     # ensure feasibility
    n = a.shape[0]

    # make the sequence symmetric – this never hurts the max‑convolution
    a = (a + a[::-1]) / 2.0

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    v = np.zeros_like(a)                       # momentum accumulator
    it = 0
    no_improve = 0

    # FFT sizes
    fft_len_conv = next_pow2(2 * n)            # for a*a
    fft_len_grad = next_pow2(3 * n)            # for w * a_rev (length ≤ 3n‑2)

    while time.time() < start_time + time_limit:
        it += 1

        # ---- 1) current convolution and score --------------------------------
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- 2) soft‑max weights ------------------------------------------------
        # temperature schedule – linearly increase α
        elapsed = time.time() - start_time
        total = time_limit
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, elapsed / total)

        # exp(α·(conv‑M)) stays in [0,1] → safe numerics
        exp_vals = np.exp(alpha * (conv - M))
        w = exp_vals / exp_vals.sum()                         # shape (2n‑1,)

        # ---- 3) gradient of the smoothed objective ----------------------------
        # gradient = 2 * (w * a_rev) sliced appropriately
        rev_a = a[::-1]                                       # reversed copy
        fw = np.fft.rfft(w, n=fft_len_grad)
        fra = np.fft.rfft(rev_a, n=fft_len_grad)
        grad_full = np.fft.irfft(fw * fra, n=fft_len_grad)
        # slice the part that corresponds to indices 0 … n‑1
        start_idx = n - 1
        grad = 2.0 * grad_full[start_idx : start_idx + n]

        # ---- 4) momentum / step size -----------------------------------------
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * grad
        cand = a - eta * v

        # ---- 5) enforce symmetry & simplex ------------------------------------
        cand = (cand + cand[::-1]) / 2.0
        a = project_simplex(cand)

        # ---- 6) occasional random perturbation to escape plateaus ------------
        if no_improve > 5000:
            sigma = 0.001 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score for the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score

        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = np.full(init_seq.shape, 1.0 / init_seq.shape[0])
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Randomised family search (≈40 % of budget)
    # ------------------------------------------------------------------
    length_choices = [6000, 8000, 10000, 12000, 14000]
    families = ["power", "exp", "twosided", "beta", "sine", "arc", "mix"]
    top_k = [(best_score, best_seq.copy())]

    rand_deadline = start + 0.4 * budget_s
    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.1, 1.2)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.1, 1.2)
            p_r = random.uniform(0.1, 1.2)
            w_l = random.uniform(0.2, 5.0)
            w_r = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            alpha = random.uniform(0.05, 1.5)
            beta = random.uniform(0.05, 1.5)
            seq = generate_beta_shape(n, alpha, beta)

        elif family == "sine":
            seq = generate_sine_shape(n)

        elif family == "arc":
            seq = generate_arcsine_shape(n)

        else:  # "mix" – convex combination of two random families
            # pick two families at random (excluding "mix" itself)
            fam1, fam2 = random.sample(
                ["power", "exp", "twosided", "beta", "sine", "arc"], 2
            )
            # generate the two components
            if fam1 == "power":
                seq1 = generate_power_law(n, random.uniform(0.1, 1.2))
            elif fam1 == "exp":
                seq1 = generate_exponential(n, random.uniform(1e-4, 5e-2))
            elif fam1 == "twosided":
                seq1 = generate_two_sided_power_law(
                    n,
                    random.uniform(0.1, 1.2),
                    random.uniform(0.1, 1.2),
                    random.uniform(0.2, 5.0),
                    random.uniform(0.2, 5.0),
                )
            elif fam1 == "beta":
                seq1 = generate_beta_shape(
                    n, random.uniform(0.05, 1.5), random.uniform(0.05, 1.5)
                )
            elif fam1 == "sine":
                seq1 = generate_sine_shape(n)
            else:
                seq1 = generate_arcsine_shape(n)

            if fam2 == "power":
                seq2 = generate_power_law(n, random.uniform(0.1, 1.2))
            elif fam2 == "exp":
                seq2 = generate_exponential(n, random.uniform(1e-4, 5e-2))
            elif fam2 == "twosided":
                seq2 = generate_two_sided_power_law(
                    n,
                    random.uniform(0.1, 1.2),
                    random.uniform(0.1, 1.2),
                    random.uniform(0.2, 5.0),
                    random.uniform(0.2, 5.0),
                )
            elif fam2 == "beta":
                seq2 = generate_beta_shape(
                    n, random.uniform(0.05, 1.5), random.uniform(0.05, 1.5)
                )
            elif fam2 == "sine":
                seq2 = generate_sine_shape(n)
            else:
                seq2 = generate_arcsine_shape(n)

            w = random.uniform(0.2, 0.8)
            seq = generate_mixture(n, seq1, seq2, w)

        # normalise and defensive checks
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time() - start:.1f}s] new global best ({family}) n={n} "
                f"score={sc:.6f}"
            )

        # maintain an elite set of up to 8 candidates
        if len(top_k) < 8:
            top_k.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(top_k)), key=lambda i: top_k[i][0])
            if sc < top_k[worst_idx][0]:
                top_k[worst_idx] = (sc, seq.copy())

    # sort elite pool (ascending by score)
    top_k.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2) Smoothed‑max gradient refinement on elite candidates
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted. Final score = {final_score:.6f}")
        return final_seq

    per_candidate = time_remaining / len(top_k)

    for idx, (sc, seq) in enumerate(top_k):
        if time.time() > deadline:
            break
        print(
            f"[refine] elite #{idx+1}/{len(top_k)} – start score={sc:.6f}, n={seq.size}"
        )
        refined_seq, refined_score = refine_sequence_smooth(
            seq, per_candidate, time.time()
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
