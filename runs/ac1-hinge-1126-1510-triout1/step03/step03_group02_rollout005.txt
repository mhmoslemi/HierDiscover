"""
search for a low‑score step‑function.

The algorithm combines three complementary strategies:

1. **Multi‑restart L‑BFGS** on the probability simplex using a smooth‑max
   approximation of the auto‑convolution peak.  The smoothing parameter λ is
   decreased geometrically (continuation) so that the optimiser gradually
   focuses on the true max‑operator.

2. **Projected sub‑gradient descent** on the exact max‑convolution objective.
   For the current peak index p, the sub‑gradient of max_i (a*a)_i is
   g_i = 2·a_{p−i} (when the index is valid, otherwise 0).  After each
   step the vector is projected onto the simplex (non‑negative, unit sum).

3. **Aggressive hill‑climbing** that swaps small fractions of mass between
   coordinates whenever the fast FFT‑based score improves.

All convolutions are performed with FFTs (O(n log n)).  The search is
time‑bounded (default 1000 s) and works for up to ~10 k coefficients.
The best sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector that sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# L‑BFGS with smooth‑max
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (the log‑soft‑max coordinates).

    Returns
    -------
    f : float
        Smooth approximation of 2·n·max(a*a).
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y`` (same shape as ``y``).
    """
    a = softmax(y)

    # ---- convolution a*a (full, length 2n‑1) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# hill climbing on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12  # up to 12 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    init_step: float,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   max_i (a*a)_i .
    Returns the best vector encountered and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for t in range(max_iter):
        if time.time() >= deadline:
            break

        # current convolution and the peak index
        conv = conv_fft(a, fft_len, n)
        p = int(np.argmax(conv))

        # sub‑gradient of the max‑operator
        idx = p - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # decreasing step size
        step = init_step / np.sqrt(t + 1.0)

        # gradient step + projection onto the simplex
        a = proj_simplex(a - step * g)

        # evaluate (fast) score
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with low evaluation score.

    The routine runs multiple random restarts, each consisting of:
      * continuation L‑BFGS on a smooth‑max surrogate,
      * a short projected sub‑gradient descent on the exact max‑convolution,
      * an aggressive hill‑climb.
    The best sequence discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # margin for safety

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    # try to reuse a previously found construction if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)
        if prev_arr.ndim == 1 and prev_arr.size >= 2000:
            n = int(prev_arr.size)
            a_prev = np.maximum(prev_arr, 0.0)
            if a_prev.sum() > 0:
                a_prev = a_prev / a_prev.sum()
            else:
                a_prev = np.full(n, 1.0 / n)
        else:
            n = 8000
            a_prev = None
    else:
        n = 8000
        a_prev = None

    # ensure a reasonable minimum size
    n = max(n, 5000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initial uniform solution (baseline)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # if we have a previous construction, use it as a starting point
    if a_prev is not None and a_prev.size == n:
        best_a = a_prev
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)
        print(f"[init] loaded previous sequence, score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 0.001, num=25)   # smooth‑max continuation
    maxiter_stage = 200                               # L‑BFGS iterations per stage
    subgrad_step = 0.6                                 # initial step size for sub‑grad
    subgrad_iters = 3000                               # cap per sub‑gradient phase

    iteration = 0

    # --------------------------------------------------------------
    # multi‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        mode = rng.choice(["random", "exp", "power", "edge", "triangular", "best"])

        # ----- generate a fresh starting point on the simplex -----
        if mode == "random":
            a0 = rng.random(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
        elif mode == "triangular":
            # symmetric tent shape: a_i ∝ (i+1)*(n-i)
            idx = np.arange(n, dtype=float)
            a0 = (idx + 1) * (n - idx)
        else:  # perturb the current best
            a0 = best_a + rng.normal(scale=0.03, size=n)

        # normalise to the simplex
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

        # ----- L‑BFGS continuation -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- sub‑gradient phase (exact max) -----
        sub_deadline = min(deadline, time.time() + 1.2)  # short slice of remaining budget
        a_sg, sg_score = subgradient_optimize(
            a_start=best_a,
            n=n,
            fft_len=fft_len_conv,
            init_step=subgrad_step,
            max_iter=subgrad_iters,
            deadline=sub_deadline,
        )
        if sg_score < best_score:
            best_score = sg_score
            best_a = a_sg.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] sub‑grad improved fast score = {best_score:.6f}")

        # ----- aggressive hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final deep refinement on the current best
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 400, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # final hill‑climb with any remaining time
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation (for logging) and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception:
        print("[final] evaluation unavailable (will be done by the harness)")

    return final_seq
