"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

using a two‑phase strategy:

1️⃣  Random exploration of many parametric families (power‑law, exponential,
    two‑sided power‑law, arcsine‑like, cosine, Gaussian, beta‑like).  The
    best normalised sequence found by the cheap FFT based “fast_score”
    is kept.

2️⃣  Multi‑gradient mirror‑descent refinement.
    At each iteration we compute the self‑convolution,
    select the top % of indices with the largest values (active set)
    and take the gradient of the sum of those entries.
    The gradient is efficiently obtained as a correlation via FFT.
    A multiplicative update (mirrored on the probability simplex) keeps
    the sequence non‑negative and preserves the normalisation.
    Decaying step‑size, occasional jitter and periodic evaluation
    drive the solution toward a flatter convolution, i.e. a lower
    max/average ratio.

The routine respects a 1000 s budget, prints progress and finally
returns the best sequence as a plain Python list of floats.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT based convolution.  ``seq`` is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  The (unnormalised) shape behaves like 1/√i.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian, not normalised."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{alpha}·(n‑i)^{beta}.
    Allows a wide variety of unimodal / U‑shaped / heavy‑tailed shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    left = i ** alpha
    right = (n - i + 1) ** beta
    return left * right

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # keep a small safety margin

    # --------------------------------------------------------------
    # 0️⃣  Initialise with a known good sequence or an arcsine shape
    # --------------------------------------------------------------
    try:

    best_seq = init.copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]

            cand = generate_power_law(n, p)

        elif family == "twosided":

            pr = random.uniform(0.30, 0.90)
            wl = random.uniform(0.5, 2.0)

        # ----- FFT of current a (single transform for this iteration) -----
        A = np.fft.rfft(a, n=fft_len)

        # ----- self‑convolution (full length) -----
        conv_full = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]

        # ----- identify active set (top % of values) -----
        top_frac = 0.01                     # 1 % of entries
        k = max(1, int(top_frac * conv_full.size))
        # indices of the k largest values

            a = np.clip(a, 0.0, None)
            sum_a = a.sum()
