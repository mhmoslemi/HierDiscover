"""
Propose a step‑function with a low evaluation score.

Algorithm
---------

* **Length selection** – try a few lengths (the length of the supplied
  `height_sequence_1` and a few nearby values).  The evaluation score is
  invariant under rescaling, therefore the most promising length is kept.

* **Rich parametric sweep** – generate many simple families (power‑law,
  exponential, double‑exponential, Beta, sinusoid, triangular, arcsine)
  on a fine grid of parameters.  Each candidate is projected onto the
  probability simplex and its maximal self‑convolution is recorded.
  The best candidate of the sweep becomes the start point for the next
  phase.

* **Adam optimisation on a smoothed max** – replace the hard max by a
  soft‑max (`log‑sum‑exp`) with a temperature that grows from 4 to 200.
  The gradient of the smoothed objective is `2·correlation(weights, a)`,
  computed by FFTs.  A projected Adam optimiser (β₁=0.9, β₂=0.999) is run
  for as many iterations as the remaining time allows.

* **Sub‑gradient descent on the true max** – once the smoothing is strong,
  switch to the true max using a uniform weight on the active set.
  A diminishing step size `η/√t` is used, again with simplex projection.

* **Random hill‑climbing / simulated‑annealing** – finish with cheap
  mass‑transfer moves between random coordinates.  Moves that improve
  the maximal convolution are kept; worsening moves are accepted with a
  tiny probability that decays with time (a light annealing schedule).

* **Scaling** – the best L1‑normalised vector is finally multiplied by
  `sqrt(2·n)` (the required normalisation) and returned as a plain
  Python list.

All heavy work (convolution, correlation, gradient) is performed with
FFT‑based O(L log L) kernels, where `L` is the next power of two ≥ 2 n − 1.
The routine respects the supplied time budget (default 1000 s) and
prints progress information.
"""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(
    a: np.ndarray, weights: np.ndarray, phase: np.ndarray, L: int
) -> np.ndarray:
    """
    Gradient of <weights, a*a> w.r.t. a.
    Implements 2·correlation(weights, a) using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase  # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights over the active set of the hard max."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:  # fallback – should never happen
        w[conv.argmax()] = 1.0
    return w

def _parametric_sweep(n: int, L: int, best_seq: np.ndarray, best_max: float):
    """
    Extensive sweep over a large family of analytic shapes.
    Returns (best_seq, best_max) after the sweep.
    """
    xs = np.arange(n, dtype=np.float64)

    # power‑law (including decreasing shapes)
    for p in np.linspace(-1.5, 5.0, 66):
        base = (xs + 1.0) ** p
        for cand in (base, base[::-1]):
            cand = cand / cand.sum()
            conv = _compute_convolution(cand, L)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"  sweep power p={p:.3f}  max={cur_max:.8e}")

    for lam in np.linspace(0.0, 15.0, 61):
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):

            conv = _compute_convolution(cand, L)
            cur_max = conv.max()
            if cur_max < best_max:

                print(f"  sweep exp λ={lam:.3f}  max={cur_max:.8e}")

    # double‑exponential (U‑shape)

        left = np.exp(-lam * xs)

        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  sweep U‑shape λ={lam:.3f}  max={cur_max:.8e}")

                    print(f"  sweep Beta a={a:.3f} b={b:.3f}  max={cur_max:.8e}")

    # sinusoidal power family

    conv = _compute_convolution(cand, L)

        print(f"  sweep triangular max={cur_max:.8e}")

    for cand in (arcsine, arcsine[::-1]):

        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:

def _adam_descent(
    a0: np.ndarray,
    n: int,
    L: int,
    phase: np.ndarray,
    deadline: float,
    max_iters: int = 50000,
    alpha_start: float = 4.0,
    alpha_end: float = 200.0,
    base_lr: float = 0.12,
):
    """Projected Adam minimising the smoothed max."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_a = cur_max, a.copy()
            print(f"  Adam iter {it:5d}  new best max={cur_max:.8e}")

        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_max


def _subgrad_descent(
    a0: np.ndarray,
    n: int,
    L: int,
    phase: np.ndarray,
    deadline: float,
    max_iters: int = 60000,
):
    """Projected sub‑gradient descent on the true max."""
    a = a0.copy()
    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()

    for t in range(1, max_iters + 1):

def _random_hill_climb(

    max_delta_frac: float = 0.05,

        if i == j:
            continue
        # transfer up to max_delta_frac of the source mass
        delta = np.random.rand() * max_delta_frac * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < -1e-12:
            continue
        conv = _compute_convolution(a_new, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_a = cur_max, a_new.copy()
            a = a_new  # continue climbing from the improved point
            print(f"  hill‑climb {it:6d} improved max={cur_max:.8e}")
        else:
            # light simulated‑annealing: accept w.p. exp(-Δ/τ) with τ slowly decaying
            # (here τ is very small, we keep it simple and reject)
            pass
    return best_a, best_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine respects a time budget (seconds) and returns a plain
    Python list of floats (the final sequence is already rescaled to
    satisfy sum(a) = sqrt(2*n) as required by `evaluate_sequence`).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a set of lengths to explore
    # ------------------------------------------------------------------
    try:
        # a previously discovered construction may be available globally
        base_vec = np.array(height_sequence_1, dtype=np.float64)
        base_n = len(base_vec)
    except Exception:
        base_vec = None
        base_n = 6000

    # we probe a handful of nearby lengths; the best after scaling is kept
    lengths = sorted(
        {base_n, base_n - 1000, base_n + 1000, 5000, 8000, 10000}
    )
    lengths = [n for n in lengths if n > 1000]  # avoid too tiny problems

    global_best_seq = None
    global_best_score = float("inf")
    global_best_n = None

    # ------------------------------------------------------------------
    # 2.  Main loop over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Trying length n={n} ===")
        L = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
        phase = np.exp(2j * np.pi * np.arange(L) / L)

        # ------------------------------------------------------------------
        #   Initialise with a known construction or uniform vector
        # ------------------------------------------------------------------
        if base_vec is not None and len(base_vec) == n:
            a0 = base_vec / base_vec.sum()
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)

        best_seq = a0.copy()
        best_max = _compute_convolution(best_seq, L).max()
        print(f"  initial max={best_max:.8e}")

        # ------------------------------------------------------------------
        #   Phase 1 – rich parametric sweep
        # ------------------------------------------------------------------
        best_seq, best_max = _parametric_sweep(n, L, best_seq, best_max)
        print(f"  after sweep best max={best_max:.8e}")

        # ------------------------------------------------------------------
        #   Phase 2 – Adam on smoothed max
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(remaining * 0.45, 120.0)  # cap to ~2 min
        best_seq, best_max = _adam_descent(
            best_seq, n, L, phase, adam_deadline,
            max_iters=50000,
            alpha_start=4.0,
            alpha_end=200.0,
            base_lr=0.12,
        )
        print(f"  after Adam best max={best_max:.8e}")

        # ------------------------------------------------------------------
        #   Phase 3 – sub‑gradient descent on true max
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.30, 180.0)  # cap to ~3 min
        best_seq, best_max = _subgrad_descent(
            best_seq, n, L, phase, sub_deadline, max_iters=60000
        )
        print(f"  after sub‑grad best max={best_max:.8e}")

        # ------------------------------------------------------------------
        #   Phase 4 – random hill‑climbing / simulated annealing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            best_seq, best_max = _random_hill_climb(
                best_seq, n, L, deadline - 0.5, max_delta_frac=0.06
            )
            print(f"  after hill‑climb best max={best_max:.8e}")

        # ------------------------------------------------------------------
        #   Compute the final score (after scaling to sqrt(2 n))
        # ------------------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)
        final_seq = (best_seq * target_sum).tolist()
        # use evaluate_sequence only for logging – it is already imported
        try:
            final_score = evaluate_sequence(final_seq)
        except Exception:
            final_score = float("inf")
        print(f"--- Length {n} finished: score={final_score:.6f} "
              f"(max_conv={best_max:.8e}) ---")

        if final_score < global_best_score:
            global_best_score = final_score
            global_best_seq = final_seq
            global_best_n = n

    # ------------------------------------------------------------------
    # 3.  Return the overall best sequence found
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – uniform vector of a reasonable size
        n = 6000
        seq = np.full(n, 1.0 / n, dtype=np.float64)
        seq = (seq * math.sqrt(2.0 * n)).tolist()
        return seq

    print(f"\n=== Overall best: n={global_best_n}  score={global_best_score:.6f} ===")
    return global_best_seq
