    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)

def objective_logsumexp(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    # compute conv
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    # compute smooth max
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z
    # gradient
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n-1:n-1+n]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def objective_pnorm(y, n, fft_len_conv, fft_len_g, p):
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    conv = np.maximum(conv, 0)  # non-negative
    conv_p = conv ** p
    S = conv_p.sum()
    f_val = S ** (1.0/p)
    f = 2.0 * n * f_val
    # gradient w.r.t conv:
    # w_i = conv_i^{p-1} / (S^{(p-1)/p})
    # Avoid division by zero.
    if S == 0:
        w = np.zeros_like(conv)
    else:
        w = conv ** (p - 1) / (S ** ((p - 1)/p))
    # gradient w.r.t a:
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n-1:n-1+n]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_continuation_custom(a_start, schedule, maxiter_per_lam, deadline, fft_len_conv, fft_len_g, n, use_pnorm=False):

    for param in schedule:
        if time.time() >= deadline:
            break
        for _ in range(1): # one run per param
            if use_pnorm:
                func = lambda y, n=n, fft_len_conv=fft_len_conv, fft_len_g=fft_len_g, p=param: objective_pnorm(y, n, fft_len_conv, fft_len_g, p)

"""
Propose a low‑score non‑negative step sequence.

Algorithm outline
-----------------
1.  Choose a problem size (n ≤ 4000) and initialise a random distribution
    on the probability simplex.
2.  Phase‑1 – Continuation L‑BFGS on a log‑sum‑exp smooth approximation
    of the max‑convolution.  A decreasing “lam” schedule together with
    many random restarts drives the solution close to a local optimum.
3.  Phase‑2 – Continuation L‑BFGS on an Lp‑norm smooth approximation
    (p = 2,4,…,256).  Large p gives a tighter approximation of the true
    max, further decreasing the objective.
4.  Phase‑3 – Projected sub‑gradient descent on the exact max‑convolution.
    A simple back‑tracking line‑search and many iterations improve the
    solution when the smooth approximations have already plateaued.
5.  Phase‑4 – Simulated‑annealing / hill‑climb with large random mass
    transfers to escape possible local minima.
6.  Phase‑5 – A final smoothing + fast hill‑climb (projected simplex) to
    polish the best sequence found.
All phases respect the constraints (non‑negative, sum = 1).  The code
splits the available time budget automatically and returns the best
sequence seen before the deadline.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score 2·n·max(a*a) for a already on the simplex (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed objectives
# ----------------------------------------------------------------------
def objective_logsumexp(y: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_g: int,
                        lam_factor: float):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) where objective = 2·n·smooth_max(conv).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def objective_pnorm(y: np.ndarray,
                    n: int,
                    fft_len_conv: int,
                    fft_len_g: int,
                    p: int):
    """
    Lp‑norm smooth approximation of max (p large → max).
    Returns (objective, gradient) with objective = 2·n·(||conv||_p).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    conv = np.maximum(conv, 0)       # numerical safety

    # Lp‑norm of the convolution vector
    conv_p = conv ** p
    S = conv_p.sum()
    if S == 0:
        f_val = 0.0
        w = np.zeros_like(conv)
    else:
        f_val = S ** (1.0 / p)
        w = conv ** (p - 1) / (S ** ((p - 1) / p))

    f = 2.0 * n * f_val

    # back‑propagation
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,

                       fft_len_conv: int,

                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs log] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs log] new best score={best_score:.6f} lam={lam:.5f}")

def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via projected sub‑gradient descent.
    """

    best_a = a.copy()

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size schedule (decreasing)
        step = 0.6 / np.sqrt(it + 1.0)

        # back‑tracking line‑search (up to 5 trials)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score

# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Simple hill‑climb that moves a random fraction of mass between two
    coordinates, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of mass from i to j
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """

    two coordinates and accepts worsening moves with a decreasing temperature.

    cur_score = score
    best_a = a.copy()
    best_score = score

        if i == j:

        # propose moving up to 15 % of the mass at i

        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)

        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:

def smoothing_step(a: np.ndarray, kernel_width: int = 7) -> np.ndarray:
    """
    Apply a moving‑average smoothing kernel and project back onto the simplex.
    """
    if kernel_width <= 1:
        return a
    kernel = np.ones(kernel_width) / kernel_width
    smooth = np.convolve(a, kernel, mode='same')
    proj = simplex_projection(smooth)
    return proj

def generate_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random start vector on the simplex using one of several heuristics."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'sine', 'gauss'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0)
        a /= a.sum()
    else:   # gaussian‑like bump
        center = n // 2
        sigma = rng.uniform(n / 12.0, n / 4.0)
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        a /= a.sum()
    return a
