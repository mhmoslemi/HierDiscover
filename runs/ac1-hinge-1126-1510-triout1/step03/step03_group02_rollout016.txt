"""
Improved step‑function search.

The algorithm proceeds in several stages:

1.   Determine a reasonable problem size (re‑use a previous best if
     available) and initialise a few candidate shapes:
     * uniform,
     * the previously found best,
     * a symmetric beta‑shape ((i+1)*(n‑i))**p  (p negative → “U” shape),
     * a symmetric exponential shape  exp(-α i)+exp(-α (n‑1‑i)).

     The best of those cheap candidates is kept.

2.   Refine the best candidate with a continuation L‑BFGS optimiser that
     minimises a smooth approximation of the max‑convolution.
     (The same schedule as the reference solution but a bit longer.)

3.   Perform a projected‑gradient descent on the *exact* objective:
     minimise   max_k (a*a)[k]   subject to a_i ≥ 0, Σ a_i = 1.
     In each iteration the index k achieving the current maximum is
     located, the sub‑gradient  g_i = 2·a_{k‑i} is formed (zero outside
     the valid range) and a simple step is taken followed by a
     projection onto the probability simplex.  The step size decays as
     1/√t.  This step is cheap (one FFT per iteration) and drives the
     max‑convolution down further than the generic hill‑climb.

4.   A final aggressive random hill‑climb (the same routine as in the
     reference code) uses any remaining time budget.

The whole search respects the 1000 s budget, prints progress and
returns the best non‑negative sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (full length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    This equals the value returned by evaluate_sequence for a normalised
    sequence.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    # Reference: https://arxiv.org/abs/1101.6081
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS utilities (copy of the reference implementation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (the log‑soft‑max coordinates).
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Random hill‑climb (same as reference)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10  # up to 10 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with low
    evaluation score.  The routine respects the 1000 s time budget,
    prints progress and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0  # small safety margin

    # ---------- problem size -------------------------------------------------
    # reuse a previous construction if possible
    n = 5000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = max(n, int(arr.size))

    fft_len = next_pow2(2 * n - 1)

    # ---------- initialise a few cheap candidates ---------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # previous best (if any)
    if prev is not None and len(prev) == n:
        a_prev = np.asarray(prev, dtype=float)
        if a_prev.sum() == 0:
            a_prev = np.full(n, 1.0 / n)
        else:
            a_prev = a_prev / a_prev.sum()
        s_prev = fast_score(a_prev, fft_len, n)
        if s_prev < best_score:
            best_a, best_score = a_prev, s_prev
            print(f"[init] using previous sequence, fast score = {s_prev:.6f}")

    # ---------- cheap parametric families -----------------------------------
    # 1) symmetric beta‑shape  ((i+1)*(n‑i))**p  ,  p < 0 → U‑shape
    idx = np.arange(n)
    def beta_shape(p):
        vals = ((idx + 1) * (n - idx)) ** p
        return vals / vals.sum()

    for p in np.linspace(-0.90, -0.05, 30):
        a = beta_shape(p)
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[beta] p={p:.3f} fast score = {s:.6f}")

    # 2) symmetric exponential shape  exp(-α i) + exp(-α (n‑1‑i))
    def exp_sym(alpha):
        vals = np.exp(-alpha * idx) + np.exp(-alpha * (n - 1 - idx))
        return vals / vals.sum()

    alphas = np.logspace(-5, 0.8, 30)  # up to ~6.3
    for a in alphas:
        seq = exp_sym(a)
        s = fast_score(seq, fft_len, n)
        if s < best_score:
            best_a, best_score = seq, s
            print(f"[exp] α={a:.6f} fast score = {s:.6f}")

    # ---------- refinement with continuation L‑BFGS -------------------------
    # start from the best found so far
    y_start = np.log(best_a + 1e-12)
    lam_factors = np.geomspace(0.5, 1e-5, num=25)
    y_opt = schedule_optimize(
        y_start=y_start,
        n=n,
        fft_len_conv=fft_len,
        fft_len_g=fft_len,
        lam_factors=lam_factors,
        maxiter_per_stage=500,
    )
    a_opt = softmax(y_opt)
    s_opt = fast_score(a_opt, fft_len, n)
    if s_opt < best_score:
        best_a, best_score = a_opt, s_opt
        print(f"[lbfgs] improved fast score = {s_opt:.6f}")

    # ---------- exact projected sub‑gradient descent ------------------------
    a_cur = best_a.copy()
    it = 0
    # constant that controls the initial step size; tuned empirically
    base_step = 0.35
    while time.time() < deadline - 5.0:
        conv = conv_fft(a_cur, fft_len, n)
        max_val = conv.max()
        k = int(conv.argmax())
        # sub‑gradient of the active max‑convolution entry
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idxs = np.arange(low, high + 1, dtype=int)
        rev = k - idxs
        g = np.zeros(n, dtype=float)
        g[idxs] = 2.0 * a_cur[rev]

        # diminishing step size
        eta = base_step / np.sqrt(it + 1.0)
        a_new = a_cur - eta * g
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= total

        s_new = fast_score(a_new, fft_len, n)
        if s_new < best_score:
            best_score, best_a = s_new, a_new.copy()
            print(f"[sgd] iter={it} new best fast score = {s_new:.6f}")

        a_cur = a_new
        it += 1

    # ---------- final aggressive hill‑climb ---------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=12000,
        )
        print(f"[final‑hill] fast score = {best_score:.6f}")

    # ---------- evaluate with the official routine -------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
