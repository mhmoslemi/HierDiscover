def compute_conv_fft(a):
    n = len(a)
    m = 2*n-1
    fft_len = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv

def compute_softmax_gradient(a, eps):
    n = a.shape[0]
    conv = compute_conv_fft(a)  # shape 2n-1
    # compute softmax weights
    # To avoid potential overflow, subtract max before exp
    exp_arg = eps * conv
    max_exp = np.max(exp_arg)
    w = np.exp(exp_arg - max_exp)
    w_sum = w.sum()
    w = w / w_sum
    # compute correlation gradient = 2 * (conv of a with w_rev) at positions
    w_rev = w[::-1]
    # convolution via FFT
    # Determine appropriate fft_len: length n + (2n-1) -1 = 3n-2
    m2 = n + (2*n - 1) - 1
    fft_len2 = 1 << ((m2-1).bit_length())
    A = np.fft.rfft(a, n=fft_len2)
    B = np.fft.rfft(w_rev, n=fft_len2)
    conv_rev = np.fft.irfft(A * B, n=fft_len2)[:m2]
    # extract g
    idx = 2*n-2 - np.arange(n)
    g = 2.0 * conv_rev[idx]
    return g

def compute_gradient(a, conv, eps):
    n = len(a)
    # Softmax weights
    exp_arg = eps * conv
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    w /= w.sum()
    w_rev = w[::-1]
    # Convolution via FFT (a and w_rev)
    # Length: n + (2n-1) - 1 = 3n-2
    m2 = n + 2*n - 2

    return grad

"""
Smooth‑max gradient descent on the probability simplex.

The objective to minimise is

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the expression is homogeneous we fix Σ a_i = 1 and work on the
simplex {a_i ≥ 0, Σ a_i = 1}.  Instead of the non‑smooth max we use a
soft‑max (log‑sum‑exp) with a temperature that is slowly lowered.
The gradient of the smooth approximation can be written in closed form
as a (cross‑)convolution and is evaluated with FFTs.  After each
gradient step we project back onto the simplex (Duchi et al. 2008).
A cheap stochastic “mass‑transfer’’ local search is applied at the end
of each run.  Several restarts with different lengths and initial
distributions are performed until the time budget expires; the best
sequence ever seen is returned.

The implementation respects the 1000 s limit, uses only numpy and
standard library imports and returns a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    L = next_pow2(m)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:m]
    return conv


def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return (full convolution a*a, its maximum entry)."""
    conv = conv_full_fft(a)
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Smooth‑max gradient step
# ----------------------------------------------------------------------
def smooth_gradient_step(
    a: np.ndarray,
    eps: float,
    step: float,
) -> tuple[np.ndarray, float]:
    """
    Perform one gradient step for the soft‑max approximation of max(conv).

    Returns the updated vector (projected onto simplex) and the current
    maximal self‑convolution value (used for scoring).
    """
    n = a.shape[0]

    # ----- compute full self‑convolution (FFT) -----
    conv = conv_full_fft(a)                     # length 2n‑1
    max_b = float(conv.max())

    # ----- soft‑max weights -----
    exp_arg = eps * conv
    max_exp = np.max(exp_arg)                   # for numerical stability
    w = np.exp(exp_arg - max_exp)
    w /= w.sum()                                # w_k = soft‑max weight

    # ----- gradient = 2 * (a  * w_rev) evaluated at appropriate indices -----
    w_rev = w[::-1]                             # length 2n‑1
    m_rev = 3 * n - 2                           # length of a * w_rev (full)
    L2 = next_pow2(m_rev)
    A_big = np.fft.rfft(a, n=L2)                # FFT of a at length L2
    B_big = np.fft.rfft(w_rev, n=L2)            # FFT of w_rev
    conv_rev = np.fft.irfft(A_big * B_big, n=L2)[:m_rev]

    # indices that give the cross‑correlation needed for the gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_rev[idx]

    # ----- gradient descent step + simplex projection -----
    a_next = project_simplex(a - step * grad)

    return a_next, max_b


# ----------------------------------------------------------------------
# Smooth‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    eps_start: float,
    eps_end: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run the smooth‑max descent (with a linearly decreasing temperature)
    and keep the best point seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (soft‑max smoothing)
        eps = eps_start + (eps_end - eps_start) * (it - 1) / max(1, max_iters - 1)

        # step‑size schedule (standard sub‑gradient style)
        step = step0 / np.sqrt(it)

        a, max_b = smooth_gradient_step(a, eps, step)

        score = 2.0 * n * max_b                     # Σa = 1 → denominator = 1
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if it % 200 == 0:
                print(f"[smooth it {it}] new best={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (mass transfer)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap random walk that moves a small amount of mass between two
    coordinates, accepting improvements (or occasional uphill moves).
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02  # initial acceptance temperature

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of the move
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)

        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # Metropolis acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score

        if s > 0.0:
            hb /= s
        else:
            raise ValueError
        global_best_seq = hb.copy()
        global_best_score = 2.0 * len(hb) * compute_max_conv(hb)
        print(f"[init] existing sequence score={global_best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Candidate lengths (a few short, medium and long sequences)
    # ------------------------------------------------------------------
    length_candidates = [2000, 4000, 8000, 12000, 20000]

    # Hyper‑parameters
    max_iters_per_restart = 15000
    step0 = 0.7
    eps_start = 0.5
    eps_end = 30.0

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    for n in length_candidates:
        while time.time() < deadline:
            # -------------------- initialise a starting point --------------------
            r = np.random.rand()
            if global_best_seq is not None and len(global_best_seq) == n and r < 0.3:
                # perturbed copy of the current global best
                a0 = np.array(global_best_seq, copy=True)
                a0 += 0.02 * np.random.randn(n)
                a0 = np.maximum(a0, 0.0)
                s = a0.sum()
                a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            elif r < 0.2:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)      # uniform
            elif r < 0.7:
                # power‑law decay
                p = np.random.uniform(0.45, 0.80)
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()
            else:
                # exponential decay
                lam = np.random.uniform(0.0002, 0.0010)
                idx = np.arange(n, dtype=np.float64)
                a0 = np.exp(-lam * idx)
                a0 /= a0.sum()

            # -------------------- decide the iteration budget --------------------
            time_left = deadline - time.time()
            if time_left < 0.2:
                break

            # rough per‑iteration cost (scales roughly linearly with n)
            est_time_per_iter = 0.0012 * (n / 4000.0)
            est_iters = max(1, int(time_left / est_time_per_iter * 0.7))
            it_budget = min(max_iters_per_restart, est_iters)

            # -------------------- smooth‑max descent --------------------
            seq, score = smooth_descent(
                a0,
                max_iters=it_budget,
                step0=step0,
                eps_start=eps_start,
                eps_end=eps_end,
                deadline=deadline,
            )

            # -------------------- optional local refinement --------------------
            if time.time() < deadline - 0.5:
                steps_ref = int(min(8000, (deadline - time.time()) / 0.0009))
                seq_ref, score_ref = local_refine(seq, steps_ref, deadline)
                if score_ref < score:
                    seq, score = seq_ref, score_ref

            # -------------------- update global best --------------------
            if score < global_best_score:
                global_best_score = score
                global_best_seq = seq.copy()
                print(f"[len={n}] new global best score={score:.6f} (iters={it_budget})")
                # Stop early if we have already reached the target.
                if global_best_score < 1.503:
                    deadline = time.time()  # force termination of outer loops
                    break

            # gradually shrink the base step size to aid fine‑tuning
            if step0 > 0.2:
                step0 *= 0.998

        # early exit if target reached
        if deadline <= time.time():
            break

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a /= a.sum()
        global_best_seq = a

    # Convert to plain Python list
    return [float(x) for x in global_best_seq.tolist()]
