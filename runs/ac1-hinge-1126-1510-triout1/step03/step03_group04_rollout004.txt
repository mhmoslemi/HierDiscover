"""
Propose a low‑value step‑function sequence.

Main ideas
----------
1. Initialise from the previous best or from a uniform distribution.
2. Perform several optimisation phases within the 1000 s budget:
   * Phase 1 – stochastic L‑BFGS on a smooth (log‑sum‑exp) approximation
               of the max‑convolution.
   * Phase 2 – projected sub‑gradient descent (plain Euclidean projection).
   * Phase 3 – simulated annealing for non‑local exploration.
   * Phase 4 – exponentiated‑gradient (mirror‑descent) refinement on the
               true max‑convolution.
   * Phase 5 – fast greedy hill‑climb to polish the final answer.
3. All convolutions are evaluated with FFT (O(n log n)).
4. Variables are kept on the probability simplex (∑a = 1, a≥0); the
   evaluation function therefore reduces to 2·n·max(a∗a).

The routine respects the supplied time budget, prints intermediate scores
and finally returns the best sequence as a plain list of Python floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))
    return conv_full[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f    : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # convolution of a_rev with weights w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (Phase 2)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method with Euclidean
    projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # convolution and max
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set mask (tolerance for numerical noise)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (Phase 3)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Accepts improving moves always, uphill moves probabilistically.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature as a small fraction of the current score
    T = max(cur_score * 0.02, 1e-12)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                # cooling step even on reject
                T *= decay
                it += 1
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        it += 1

    return best_a, best_score

# Exponentiated‑gradient refinement (Phase 4)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int,
                         rng: np.random.Generator):
    """
    Mirror‑descent on the simplex using an exponentiated‑gradient update.
    Works directly on the true max‑convolution (no smoothing).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    eta = 0.5               # initial step size
    it = 0

    while time.time() < deadline:
        # current convolution and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (as in sub‑gradient phase)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # multiplicative update with clipping to avoid under/overflow
        upd = -eta * g
        upd = np.clip(upd, -50, 50)        # prevent exp overflow
        a = a * np.exp(upd)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # simple schedule: shrink eta slowly
        it += 1
        if it % 2000 == 0:
            eta *= 0.90

        # occasional random jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + jitter, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (Phase 5)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb: move a small fraction of one coordinate
    to another while preserving the sum.  Accept only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the given time budget (default 1000 s) and
    returns the best sequence found as a list of floats.
    """
    # ------------------------------------------------------------------
    # Random generator and initial sequence
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Choose problem size (n)
    # ------------------------------------------------------------------
    # Use the previously discovered sequence if it exists, otherwise start uniform
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2500
            a0 = np.full(n, 1.0 / n)
        else:
            # we may enlarge the support a little to give the optimiser more freedom
            target_n = max(2500, init_seq.size * 2)
            n = int(min(target_n, 5000))               # cap to keep runtime reasonable
            if init_seq.size < n:
                # pad with zeros
                a0 = np.concatenate([init_seq, np.zeros(n - init_seq.size)])
                a0 = np.maximum(a0, 0.0)
                a0_sum = a0.sum()
                a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
            else:
                a0 = init_seq[:n]
                a0_sum = a0.sum()
                a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT length preparation (powers of two)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # Overall deadline
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on a smoothed max
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.38 * budget_s
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    iter_counter = 0

    while time.time() < phase1_end:
        lam = lam_schedule[iter_counter % len(lam_schedule)]

        # random start mode
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0003, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.006, 0.03)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # run L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 600,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {iter_counter}] lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # short hill‑climb to polish L‑BFGS result
        hc_deadline = min(phase1_end, time.time() + 0.30)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

        iter_counter += 1

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (longer)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.68 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.88 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] intermediate score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    phase4_end = deadline - 0.2 * budget_s   # reserve a sliver for final hill‑climb
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase4_end,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   n,
                                                   rng)
        print(f"[exp‑grad] after refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=3000)

    # Evaluate with the provided evaluator for reporting
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
