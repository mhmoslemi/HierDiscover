"""
Search for a low‑score non‑negative step sequence.

The algorithm combines:
 1. A quick random‑sparse exploration (binary support with equal masses).
 2. Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, with a decreasing smoothing parameter.
 3. A long projected sub‑gradient descent that directly minimizes the true
    max‑convolution.
 4. A simulated‑annealing hill‑climb that moves small fractions of mass.
 5. A final greedy hill‑climb for polishing.

All stages respect the simplex (non‑negative, sum = 1) and use
FFT‑based convolutions for O(n log n) speed.  The time budget (≈ 1000 s)
is split automatically; the best sequence found before the deadline is
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns real part truncated to needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation with simple line‑search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses back‑tracking line search to adapt the step size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search (simple back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(6):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # small move to avoid stall
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing lam schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            # unexpected failure – skip this lambda
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # short hill‑climb after each lambda can be helpful
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.03  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.88
            T = max(T, 1e-6)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random sparse initialisation
# ----------------------------------------------------------------------
def random_sparse_search(n: int,
                         fft_len: int,
                         budget: float,
                         rng,
                         k_min: int = 60,
                         k_max: int = 140):
    """
    Generate many binary‑support candidates with equal weight.
    Returns the best vector found within the allocated time.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    end = time.time() + budget
    while time.time() < end:
        k = rng.integers(k_min, k_max + 1)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[rand‑sparse] new best_score={best_score:.6f} (k={k})")
    return best_a, best_score

Search for a low‑score non‑negative step sequence.

The algorithm combines:
 1. A cheap random‑sparse exploration (binary support with equal masses).
 2. Continuation L‑BFGS on a log‑sum‑exp smoothed max‑convolution, with a
    decreasing smoothing parameter.
 3. A long projected sub‑gradient descent that directly minimises the exact
    max‑convolution.
 4. Simulated‑annealing moves that occasionally accept uphill steps.
 5. A final greedy hill‑climb for polishing.

All stages respect the simplex (non‑negative, sum = 1) and use FFT‑based

    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # small hill‑climb after each lam can be helpful
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.03
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.88
            T = max(T, 1e-6)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random sparse initialisation
# ----------------------------------------------------------------------
def random_sparse_search(n: int,
                         fft_len: int,
                         budget: float,
                         rng,
                         k_min: int = 60,
                         k_max: int = 140):
    """
    Generate many binary‑support candidates with equal weight.
    Returns the best vector found within the allocated time.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    end = time.time() + budget
    while time.time() < end:
        k = rng.integers(k_min, k_max + 1)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[rand‑sparse] new best_score={best_score:.6f} (k={k})")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size and an initial distribution
    # ------------------------------------------------------------------
    max_n = 2000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            init_seq[init_seq < 0] = 0.0
            if init_seq.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin
    random_phase_end = start_time + 0.15 * budget_s
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = deadline

    # ------------------------------------------------------------------
    # Phase 0 – cheap random sparse exploration
    # ------------------------------------------------------------------
    if time.time() < random_phase_end:
        a_rand, score_rand = random_sparse_search(n,
                                                  fft_len_conv,
                                                  random_phase_end - time.time(),
                                                  rng)
        if score_rand < best_score:
            best_score, best_a = score_rand, a_rand.copy()
            print(f"[phase0] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with several random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                    0.015, 0.008, 0.004, 0.002, 0.001]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'uniform'])
        if mode == 'random':
            a_start = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a_start = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.0)
            a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == 'edge':
            edge_frac = rng.uniform(0.006, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_start = np.full(n, a_center)
            a_start[:m_edge] = a_edge
            a_start[-m_edge:] = a_edge
        else:   # uniform
            a_start = np.full(n, 1.0 / n)

        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
