"""
Propose a low‑value step sequence.

The objective is  E(a)=2·n·max(a*a)/(sum a)^2 .
Because the metric is scale‑invariant we restrict the search to the
probability simplex (sum = 1, a_i ≥ 0).  The algorithm consists of
several cheap phases followed by a dedicated LP‑based refinement that
uses the linear programme described in the statement (maximise ∑g_j
subject to a·g convolution ≤ current bound).  After the LP phase we
polish the result with a sub‑gradient optimiser, a short simulated
annealing run and a final greedy hill‑climb.

All convolutions are performed with FFT (O(n log n)).
The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (true max)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Accepts improving moves always, uphill moves with a
    temperature‑dependent probability.  Temperature decays geometrically.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP step” from the paper)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(f,g).
    Row k corresponds to the coefficient sum_i f_i * g_{k-i}.
    """
    n = f.shape[0]
    # i indexes rows, j indexes columns
    i_idx = np.arange(n).reshape(-1, 1)          # column vector (n,1)
    j_idx = np.arange(n).reshape(1, -1)          # row vector (1,n)
    rows = (i_idx + j_idx).ravel()               # shape (n*n,)
    cols = np.repeat(np.arange(n), n)            # shape (n*n,)
    data = np.tile(f, n)                         # shape (n*n,)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_refinement(f: np.ndarray,
                  max_iters: int,
                  deadline: float,
                  tau: float = 0.2):
    """
    Perform a few LP‑based refinement steps.
    In each step we solve:
        maximise   1ᵀ g
        subject to conv(f,g) ≤ M ,  g ≥ 0
    where M = max(conv(f,f)).
    The solution g is mixed with f using a convex combination.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = fast_score(best_f, next_pow2(2 * n - 1), n)

    it = 0
    while it < max_iters and time.time() < deadline:
        # current bound
        conv_ff = full_conv(best_f, next_pow2(2 * n - 1))
        M = conv_ff.max()

        # build matrix A for conv(best_f, g)
        A = build_conv_matrix(best_f)
        b = np.full(2 * n - 1, M, dtype=float)

        # linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=float)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # LP failed – abort further LP steps
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            # trivial solution – nothing to improve
            break

        # normalise g to the same total mass as best_f
        g_scaled = g * (best_f.sum() / sum_g)

        # convex combination
        new_f = (1.0 - tau) * best_f + tau * g_scaled
        new_f = simplex_projection(new_f)

        new_score = fast_score(new_f, next_pow2(2 * n - 1), n)

        if new_score < best_score:
            best_score = new_score
            best_f = new_f
            print(f"[LP] iter {it} improved score={best_score:.6f}")
        else:
            # if no improvement, shrink the mixing factor
            tau = max(tau * 0.9, 0.01)

        it += 1

    return best_f, best_score

    phase4_end = deadline
    if time.time() < phase4_end:
        best_a, sa_score = simulated_annealing(best_a,
                                               best_score,
                                               fft_len_conv,
                                               n,
                                               phase4_end,
                                               rng)
        if sa_score < best_score:
            best_score = sa_score
            print(f"[SA] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (the remaining few seconds)
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a,
                                      best_score,
                                      fft_len_conv,
                                      n,
                                      deadline,
                                      max_iter=3000)

    # ------------------------------------------------------------------
    # Optional evaluation with the supplied evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
