"""
Search for a step‑function sequence with low evaluation score.
The algorithm combines:
  * a well‑known analytically‑motivated “arcsine” initialization,
  * a continuation L‑BFGS optimiser on a smooth‑max surrogate of max(conv),
  * aggressive greedy hill‑climbing,
  * a projected sub‑gradient refinement that directly reduces the current
    max‑convolution value,
  * many random restarts within the allotted time.
The search works on the probability simplex (sum = 1) via a soft‑max
parameterisation.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# elementary utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    # shift to make sum positive
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# convolution handling – using FFT for speed
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep only the real part (tiny imag due to round‑off)
    return np.real(conv)[: 2 * n - 1]

def conv_rev_vec(w: np.ndarray, a: np.ndarray, fft_len_g: int) -> np.ndarray:
    """
    Compute (a reversed) * w, returning the central n entries.
    Equivalent to the gradient of a*k wrt a when w holds the
    (sub‑)gradient of the smoothed max.
    """
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    rev = np.real(np.fft.ifft(fa_rev * fw))
    n = a.shape[0]
    return rev[n - 1 : n - 1 + n]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# objective with smooth‑max approximation and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    lam_factor is multiplied by the current max to obtain the smoothing
    temperature.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # now maximum is 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv values
    w = exp_shift / Z                     # shape (2n‑1,)

    # g_i = Σ_j a_j * w_{i+j}
    g_vec = conv_rev_vec(w, a, fft_len_g)   # length n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ----------------------------------------------------------------------
# optimisation helpers
# ----------------------------------------------------------------------
def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS problems with decreasing smoothing.
    Each stage starts from the solution of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 6000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    A move is kept only if the (exact) fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def flatten_max_subgradient(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    max_iters: int = 200,
    step_init: float = 0.05,
    decay: float = 0.97,
) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly reduces the current
    max‑convolution value.
    """
    step = step_init
    for it in range(max_iters):
        conv = conv_fft(a, fft_len_conv, n)
        M = conv.max()
        # mask of entries that are (near) maximal
        mask = (conv >= M - 1e-12).astype(float)

        # gradient of the max w.r.t. a (sub‑gradient)
        grad = conv_rev_vec(mask, a, fft_len_g)

        # make the gradient zero‑sum (projection onto the simplex tangent)
        grad_centered = grad - grad.mean()

        # gradient descent step
        a_tmp = a - step * grad_centered
        # project back onto simplex
        a = project_simplex(a_tmp)

        # decay step size
        step *= decay

        # optional early stop when improvement stalls
        if step < 1e-6:
            break
    return a

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "arc":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

"""
Search for a step‑function sequence with a low evaluation score.

The algorithm mixes several ideas:

* **Arcsine‑like initialization** – a closed‑form distribution
  a_i ∝ 1/√((i+½)(n‑i‑½)) yields a near‑flat self‑convolution,
  giving a good starting point.
* **Continuation L‑BFGS** – minimise a smooth‑max surrogate of
  max(conv(a,a)).  The smoothing temperature (lam) is decreased
  geometrically over many stages and each stage is allowed a large
  number of L‑BFGS iterations.
* **Aggressive hill‑climbing** – random moves that transfer a
  fraction of the mass from one index to another and keep only
  improvements.
* **Projected sub‑gradient flattening** – directly reduces the
  current maximal convolution entry by a sub‑gradient step,
  followed by projection onto the probability simplex.
* **Extensive random restarts** – the search loop repeats the above
  steps with various initialisations (random, exponential,
  power‑law, edge‑heavy, Sidon‑set, best‑so‑far, arcsine) until the
  time budget expires.

All optimisation works on the probability simplex (∑a_i = 1) via a
soft‑max parametrisation `a = softmax(y)`.  The final sequence is
returned as a plain list of non‑negative floats, ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# elementary utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max -> probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def conv_rev_vec(w: np.ndarray, a: np.ndarray, fft_len_g: int) -> np.ndarray:
    """Compute (reversed a) * w and return the central n entries."""
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    rev = np.real(np.fft.ifft(fa_rev * fw))
    n = a.shape[0]
    return rev[n - 1 : n - 1 + n]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# smooth‑max objective and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective 2·n·smooth_max(conv) and its gradient w.r.t. y.
    lam_factor is multiplied by the current max to obtain the smoothing
    temperature.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth maximum ----
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}
    g_vec = conv_rev_vec(w, a, fft_len_g)   # length n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ----------------------------------------------------------------------
# optimisation helpers
# ----------------------------------------------------------------------
def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS problems with decreasing smoothing.
    Each stage starts from the solution of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage,
                     "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 6000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def flatten_max_subgradient(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    max_iters: int = 200,
    step_init: float = 0.05,
    decay: float = 0.97,
) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly reduces the current
    maximal convolution entry.
    """
    step = step_init
    for _ in range(max_iters):
        conv = conv_fft(a, fft_len_conv, n)
        M = conv.max()
        # mask of (near‑)maximal entries
        mask = (conv >= M - 1e-12).astype(float)

        # sub‑gradient of the max w.r.t. a
        grad = conv_rev_vec(mask, a, fft_len_g)

        # make the gradient zero‑sum (tangent to the simplex)
        grad_centered = grad - grad.mean()

        # gradient step & projection onto simplex
        a_tmp = a - step * grad_centered
        a = project_simplex(a_tmp)

        step *= decay
        if step < 1e-7:
            break
    return a

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a pre‑softmax vector y for a given initialisation mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (cand * 2 not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "arc":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0
