"""
A two‑phase optimizer for the step‑function problem.

Phase 1 – cheap parametric families (power‑law, exponential, beta, arcsine, …)
and a short Adam run to obtain a reasonably flat convolution.

Phase 2 – a Frank‑Wolfe style linear‑programming (LP) oracle.
Given the current vector a (∑a = 1) we compute M = max(a∗a) and solve

    maximise   ∑ b_j
    subject to  (a ∗ b)_k ≤ M  for all k,
                b_j ≥ 0.

The solution b₀ is normalised to sum 1, mixed with a using a small
step t and projected back onto the probability simplex.  Repeating this
oracle step drives the supremum of the convolution down.

Finally a short Adam / sub‑gradient refinement and a cheap random hill‑climb
are performed.  The best sequence found (scaled to √(2 n)) is returned.

The optimiser respects the supplied time budget and prints intermediate
scores for monitoring.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (real part) of length 2·len(a)‑1."""
    fa = np.fft.fft(a, n=fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          phase: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  <w, a*a>  w.r.t. a.
    Implements  2·correlation(w, a)  using FFTs.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, n=fft_len)
    # FFT of reversed a (zero‑padded) = conj(FA) * phase_shift
    fa_rev = np.conj(fa) * phase
    fw = np.fft.fft(w, n=fft_len)
    corr = np.real(np.fft.ifft(fw * fa_rev))
    # central part (indices n‑1 … 2n‑2) contains the linear correlation
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of the true max."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    act = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv)
    if act.size:
        w[act] = 1.0 / act.size
    else:
        w[conv.argmax()] = 1.0
    return w


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   ∑ b_j
        s.t.       (a ∗ b)_k ≤ M   for all k,
                   b_j ≥ 0.
    Returns the optimal vector b (not normalised) or None if the LP fails.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # Build the dense constraint matrix A_ub where
    #   (A_ub @ b)_k = (a ∗ b)_k
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    c = -np.ones(n, dtype=np.float64)               # maximise sum(b)  ⇔ minimise -sum(b)
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(L, M),
                                 bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    # Helper to evaluate the current (unnormalised) vector.
    # We keep the vector on the probability simplex (∑=1) during optimisation.
    def _score_from_a(a_vec: np.ndarray) -> float:

        conv = _compute_convolution(a_vec, fft_len)

        _try_candidate(base, f"power p={p:.2f}")
        _try_candidate(base[::-1], f"power rev p={p:.2f}")

    # 3) Exponential
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        _try_candidate(exp_shape, f"exp λ={lam:.2f}")
        _try_candidate(exp_shape[::-1], f"exp rev λ={lam:.2f}")

    # 4) Double‑exponential (U‑shaped)
    for lam in np.linspace(0.0, 6.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        u_shape = left + left[::-1]
        _try_candidate(u_shape, f"U‑shape λ={lam:.2f}")

    # 5) Beta‑densities
    alphas = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0]
    betas = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta > 0):
                _try_candidate(beta, f"Beta a={a} b={b}")
                _try_candidate(beta[::-1], f"Beta rev a={a} b={b}")

    # 6) Arcsine‑like (classic U‑profile)
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    _try_candidate(arcsine, "arcsine")
    _try_candidate(arcsine[::-1], "arcsine rev")

    # If for any reason no candidate survived, fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score_from_a(best_a)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a smooth approximation of max.
    print("\n--- Phase 1: Adam on smooth‑max ---")
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.10
    max_adam_iters = 30000
    alpha_start, alpha_end = 4.0, 150.0
    it = 0
    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, fft_len)
        cur_max = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, fft_len)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        sc = _score_from_a(a_new)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            print(f"Adam it {it:5d}  improved score={sc:.8f}")

        a = a_new

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based Frank‑Wolfe oracle.
    print("\n--- Phase 2: LP oracle (Frank‑Wolfe) ---")
    max_lp_iters = 30
    t_mix = 0.25                     # mixing coefficient (1‑t keeps most of a)
    lp_it = 0
    while time.time() < deadline and lp_it < max_lp_iters:
        lp_it += 1
        conv = _compute_convolution(a, fft_len)
        M = conv.max()
        b0 = _solve_lp(a, M)
        if b0 is None:
            print("LP solve failed – exiting LP phase")
            break
        sum_b0 = b0.sum()
        if sum_b0 < 1e-12:               # degenerate solution
            print("LP produced near‑zero vector – stop LP")
            break
        g = b0 / sum_b0                  # normalise to ∑=1
        # convex combination
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = _project_simplex(a_new)

        sc = _score_from_a(a_new)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            print(f"LP it {lp_it:2d}  improved score={sc:.8f}")

        a = a_new

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the true max (as in the reference).
    print("\n--- Phase 3: sub‑gradient descent on true max ---")
    max_sub_iters = 60000
    sub_it = 0
    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1
        conv = _compute_convolution(a, fft_len)
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, phase, fft_len)
        step = 0.5 / np.sqrt(sub_it)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        sc = _score_from_a(a_new)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            print(f"Sub‑grad it {sub_it:5d}  improved score={sc:.8f}")

        a = a_new

    # ------------------------------------------------------------------
    # Phase 4 – cheap random hill‑climbing (tiny mass transfers).
    print("\n--- Phase 4: random local moves ---")
    a_cur = best_a.copy()
    while time.time() < deadline:
        i = np.random.choice(n, p=a_cur)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.02 * a_cur[i]   # up to 2 % transfer
        if delta <= 0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        a_new = _project_simplex(a_new)
        sc = _score_from_a(a_new)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            a_cur = a_new
            print(f"random move improved score={sc:.8f}")

    # ------------------------------------------------------------------
    # Final scaling: the objective is homogeneous, but the reference
    # solutions are usually scaled to √(2 n) before feeding evaluate_sequence.
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_a * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)          # just for logging
    elapsed = time.time() - start
    print(f"\n--- Finished (n={n}, time={elapsed:.1f}s) ---")
    print(f"Best raw score (2·n·max(conv)) = {best_score:.8f}")
    print(f"evaluate_sequence result = {final_score:.8f}")
    return final_seq
