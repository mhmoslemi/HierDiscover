def propose_candidate(seed=42, budget_s=1000):

    random.seed(seed)

    best_score = float('inf')

    def init_seq():
        p=0.6
        w=1.0
        a = (np.arange(1,n+1)**(-p))*w + (np.arange(n,0,-1)**(-p))*w
        a = a / a.sum()
        return a
    # initial f
    f = init_seq()
    # ensure sum=1
    f = f / f.sum()
    # compute initial score bound
    best_score = fast_score(f) # using fast_score defined earlier
    best_seq = f.copy()
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute convolution and max
        conv_f_f = np.fft.irfft(np.fft.rfft(f, n=next_pow2(2*n)) * np.fft.rfft(f, n=next_pow2(2*n)), n=next_pow2(2*n))[:2*n-1]
        M = conv_f_f.max()
        # active constraints
        eps = 1e-8
        active_idx = np.where(conv_f_f >= M - eps)[0]
        # ensure at least one constraint (maybe also add a few low ones)
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_f_f)])  # worst-case
        # build sparse A_ub for active constraints
        rows = len(active_idx)
        data = []
        indices = []
        indptr = [0]
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # slice f[i_start:i_end+1]
            slice_f = f[i_start:i_end+1]
            data.extend(slice_f.tolist())
            indices.extend(range(i_start, i_end+1))
            indptr.append(indptr[-1] + (i_end - i_start + 1))
        A_ub = sp.csr_matrix((data, indices, indptr), shape=(rows, n))
        b_ub = np.full(rows, M, dtype=np.float64)
        # solve LP: maximize sum g => minimize -sum g
        res = opt.linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': deadline - time.time()})
        if not res.success:
            # fallback: break
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # cannot improve further, maybe break
            break
        # scale to sum 1
        g = g0 / sum_g0
        # step size
        t = 0.2  # fixed step
        f_new = (1 - t) * f + t * g
        # ensure numeric sum 1 (project)
        f_new = f_new / f_new.sum()

            best_seq = f_new.copy()
            # print improvement

        f = f_new
    # final validation
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished. Best bound reported by evaluate_sequence = {final_score:.6f}")

"""
Propose a low‑bound step‑function.

The algorithm works in three stages within the time budget (default 1000 s).

1. **Broad stochastic search** – generate many random shapes (power‑law,
   exponential, two‑sided power‑law, beta‑pdf and logistic) for several
   moderate lengths (2000‑12000).  Each candidate is normalised to sum = 1
   and scored with a fast FFT‑based evaluator.  The best 10 candidates are
   kept.

2. **Momentum sub‑gradient refinement** – each elite candidate is refined
   by projected sub‑gradient descent on the simplex.  The sub‑gradient is
   built from *all* convolution entries that are (numerically) maximal.
   A Nesterov‑style momentum term and a diminishing step size are used,
   together with occasional Gaussian noise to escape plateaus.

3. **Linear‑programming “exchange” step** – on the final best candidate
   (if its length ≤ 3000) we solve a small LP that maximises the total
   mass under the current peak‑convolution constraint.  The LP solution
   yields a direction that provably cannot increase the peak, and a short
   convex combination step reduces the bound further.

The programme returns the best sequence found as a plain Python list.
All numbers are non‑negative, sum ≥ 0.01, and the total runtime never
exceeds the allotted budget.
"""

import time
import random
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute   2 * n * max(conv(seq, seq))   for a non‑negative sequence
    with sum(seq)=1.  Uses an FFT convolution (zero‑padded to a power of two)
    for speed.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # normalise – the algorithm always works with sum=1,
    # the returned score already equals the evaluation function.
    seq = seq / s
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf: a_i ∝ (i+1)^{α-1}(n-i)^{β-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    """a_i = 1/(1+exp((i-μ)/σ))."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 / (1.0 + np.exp((i - mu) / sigma))


# ----------------------------------------------------------------------
# Sub‑gradient refinement (momentum + occasional noise)
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    momentum: float = 0.9,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) sub‑gradient descent on the simplex.
    Returns the best sequence seen during the allotted time and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)                # ensure feasibility
    n = a.shape[0]
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    # pre‑compute FFT length (constant during the run)
    fft_len = next_pow2(2 * n)

    v = np.zeros_like(a)                  # momentum buffer
    it = 0
    no_improve = 0
    eps_active = 1e-12

    while time.time() < start_time + time_limit:
        it += 1

        # convolution of current a with itself
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # active set – all indices whose value is within eps of the maximum
        active = np.where(conv >= M - eps_active)[0]

        # sub‑gradient of the maximum (sum of contributions of all active indices)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # Nesterov‑style momentum / diminishing step size
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        # occasional random jitter to escape flat plateaus
        if no_improve >= 3000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score of the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming “exchange” step (optional, used only for modest n)
# ----------------------------------------------------------------------


def lp_exchange_step(seq: np.ndarray) -> np.ndarray:
    """
    Given a feasible sequence (sum=1, non‑negative) compute a direction
    that cannot increase the current peak M = max(conv(seq,seq)).
    This is done by solving the LP
        maximise   Σ g_j
        subject to (seq * g)_k ≤ M   for all k,
                   g_j ≥ 0.
    The solution g is normalised back to sum = 1 and we take a short
    convex combination with the current sequence.
    The function returns the new sequence (sum = 1).  For large n the
    LP would be too big, therefore the caller should only use this
    routine when n ≤ 3000.
    """
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    M = float(conv.max())

    # build the (2n‑1) × n constraint matrix A_ub where
    #   (A_ub @ g)[k] = Σ_i seq_i * g_{k-i}
    rows = 2 * n - 1
    data = []
    indices = []
    indptr = [0]
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        block = seq[i_start : i_end + 1]
        data.extend(block.tolist())
        indices.extend(range(i_start, i_end + 1))
        indptr.append(indptr[-1] + (i_end - i_start + 1))
    A_ub = sp.csr_matrix((data, indices, indptr), shape=(rows, n))
    b_ub = np.full(rows, M, dtype=np.float64)

    # linear programme: maximise Σ g_j  ⇔ minimise -Σ g_j
    res = opt.linprog(
        c=-np.ones(n),
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        return seq

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1.0 + 1e-12:
        return seq

    # normalise to sum=1 and take a small convex step
    g = g0 / sum_g0
    t = 0.1
    new_seq = (1.0 - t) * seq + t * g
    new_seq = new_seq / new_seq.sum()
    return new_seq

    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # 0)  Optional warm‑start from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        init_seq = generate_two_sided_power_law(3000, 0.55, 0.55)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad stochastic search
    # ------------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = ["power", "exp", "twosided", "beta", "logistic"]
    elite: list[tuple[float, np.ndarray]] = [(best_score, best_seq.copy())]

    # allocate ~55 % of time to random exploration
    rand_deadline = start + 0.55 * budget_s

    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 1.20)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 3e-2)          # λ roughly O(1/n)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.30, 1.20)
            p_r = random.uniform(0.30, 1.20)
            w_l = random.uniform(0.3, 3.0)
            w_r = random.uniform(0.3, 3.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            alpha = random.uniform(0.1, 3.0)
            beta = random.uniform(0.1, 3.0)
            seq = generate_beta_shape(n, alpha, beta)

        else:  # logistic
            mu = random.uniform(0.3 * n, 0.7 * n)
            sigma = random.uniform(0.05 * n, 0.3 * n)
            seq = generate_logistic(n, mu, sigma)

        # normalise and guard against pathological cases
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total

        sc = fast_score(seq)
        # keep global best
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time() - start:6.1f}s] new global best ({family}) "
                f"n={n} score={sc:.6f}"
            )

        # maintain a small elite pool (size ≤ 8)
        if len(elite) < 8:
            elite.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    # sort elite candidates by ascending score
    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2) Momentum sub‑gradient refinement on elite candidates
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted. Final score = {final_score:.6f}")
        return final_seq

    per_candidate = time_remaining / len(elite)

    for idx, (sc, seq) in enumerate(elite):
        if time.time() > deadline:
            break
        print(
            f"[refine] elite #{idx+1}/{len(elite)} start (score={sc:.6f}, n={seq.size})"
        )
        refined_seq, refined_score = refine_sequence(seq, per_candidate, time.time())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional LP‑exchange step (only for modest length)
    # ------------------------------------------------------------------
    if best_seq.size <= 3000:
        lp_iter = 3                      # a few cheap LP steps
        for i in range(lp_iter):
            best_seq = lp_exchange_step(best_seq)
            cur_score = fast_score(best_seq)
            if cur_score < best_score:
                best_score = cur_score
                print(f"[LP] iteration {i+1}: improved score → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
