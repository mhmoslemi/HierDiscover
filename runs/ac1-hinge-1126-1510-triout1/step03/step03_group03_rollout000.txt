Propose a step‑function (non‑negative heights) that minimizes  

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases under a 1000 s budget:

1. **Broad random exploration** – generate a few thousand candidates from
   several parametric families (power‑law, exponential, two‑sided power‑law,
   arcsine, Hann, Gaussian) with different lengths.  Keep the best one
   according to a fast FFT‑based score.

2. **Smoothed mirror‑descent** – starting from the best candidate, minimise a
   smooth approximation of the max‑convolution:
        f_α(a) = (1/α)·log( Σ_k exp(α·(a*a)_k) )
   The gradient of f_α is a weighted sum of the usual sub‑gradients, which
   can be computed by a single FFT‑based correlation.  An exponentiated‑gradient
   (mirror descent on the probability simplex) is performed with a slowly
   decaying step size and a schedule that increases α from a tiny value to
   ≈30, thereby focusing on the true max as optimisation proceeds.

3. **Fine‑grained random hill‑climbing** – for the remaining time a cheap
   two‑coordinate perturbation is applied repeatedly; any move that improves
   the fast score is kept.

The best sequence found is finally evaluated with the official
`evaluate_sequence` function and returned as a Python list of floats.
"""
import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT.
    The sequence is assumed to be non‑negative and already L1‑normalised
    (sum(seq)=1).  If the sum is too small a huge value is returned.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{-1/2}: a_0 = 1, recurrence a_i = a_{i‑1}·(2i‑1)/(2i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_hann(n: int) -> np.ndarray:
    """Discrete Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """
    Gaussian window centred in the middle of the array.
    sigma_frac = sigma / n   (typical range 0.07 … 0.30)
    """
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)
