"""
Propose a step‑function that minimizes the evaluate_sequence objective.

Algorithm
---------
1. **Length choice** – pick a fairly large n (≈ 20 k) to give the optimiser
   many degrees of freedom while keeping FFTs cheap (next power‑of‑2 = 65536).

2. **Simple families** – initialise the search with a handful of analytic
   shapes (arcsine, power‑law, exponential, double‑exponential, Beta‑densities).
   The best of those (lowest max‑convolution) becomes the starting point.

3. **Adam on a smooth max** – replace the hard max by a soft‑max
   `log‑sum‑exp(α·conv)`.  The gradient of the smoothed objective is
   `2·correlation(soft_weights, a)`.  Adam updates are performed on the
   probability simplex (Euclidean projection).  The smoothing parameter α
   grows exponentially from 1 to 500, so the algorithm gradually focuses on
   the true supremum.

4. **Sub‑gradient descent on the true max** – once α is large the
   iteration switches to a genuine sub‑gradient of `max(conv)`, taken as a
   uniform weight on the active set (all lags whose value is within a tiny
   tolerance of the maximum).  The step size decays like 1/√t.

5. **Random hill‑climbing + simulated annealing** – cheap mass transfers
   between random coordinates are tried; an improvement is always kept,
   otherwise a worsening move may be accepted with a temperature‑controlled
   probability.  This helps escape shallow plateaus.

6. **Final scaling** – the best normalized vector (L1‑norm = 1) is scaled
   to have L1‑norm = √(2 n) as required by `evaluate_sequence`.

The routine respects the time budget (default 1000 s) and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of <weights, a*a> w.r.t. a.
    Implements 2·correlation(weights, a) using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns a plain Python list of floats (already scaled for evaluate_sequence).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # keep a safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length (larger n gives more freedom)
    # --------------------------------------------------------------
    try:
        # if a previous construction exists, use its length as a lower bound
        n0 = len(height_sequence_1)  # may raise NameError
    except Exception:
        n0 = 0
    # target length – 20 k is a sweet spot for the 1000 s budget
    n = max(20000, n0)
    print(f"Working with n = {n}")

    # FFT helper
    L = 1 << ((2 * n - 1).bit_length())  # smallest power of two ≥ 2 n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # --------------------------------------------------------------
    # 2. Generate a pool of simple analytic families
    # --------------------------------------------------------------
    xs = np.arange(n, dtype=np.float64)
    best_seq = None
    best_max = float("inf")

    def _try_candidate(cand, name):
        nonlocal best_seq, best_max
        if time.time() > deadline:
            return
        cand = cand / cand.sum()
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  new best ({name})  max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    print("--- Phase 0: simple analytic families ---")
    # arcsine (Beta(0.5,0.5))
    x_mid = (xs + 0.5) / n
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    _try_candidate(arcsine, "arcsine")

    # power‑law family
    for p in np.linspace(0.0, 2.0, 21):
        power = (xs + 1) ** p
        _try_candidate(power, f"power p={p:.2f}")

    # exponential family
    for lam in np.linspace(0.0, 5.0, 21):
        exp_shape = np.exp(-lam * xs)
        _try_candidate(exp_shape, f"exp λ={lam:.2f}")

    # double‑exponential (U‑shape)
    for lam in np.linspace(0.0, 5.0, 11):
        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        _try_candidate(u_shape, f"U‑shape λ={lam:.2f}")

    # Beta‑densities (a,b) ∈ {0.3,0.5,1,2}
    alphas = [0.3, 0.5, 1.0, 2.0]
    betas = [0.3, 0.5, 1.0, 2.0]
    for a in alphas:
        for b in betas:
            # compute in log‑space to avoid overflow for a,b<1
            log_shape = (a - 1) * np.log(x_mid) + (b - 1) * np.log(1.0 - x_mid)
            beta_shape = np.exp(log_shape - np.max(log_shape))  # shift for stability
            _try_candidate(beta_shape, f"Beta a={a} b={b}")

    if best_seq is None:
        # fallback: uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _compute_convolution(best_seq, L).max()
        print("  fallback to uniform distribution")

    # --------------------------------------------------------------
    # 3. Adam descent on smoothed max (log‑sum‑exp)
    # --------------------------------------------------------------
    print("--- Phase 1: Adam descent on smoothed max ---")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    max_adam_iters = 200000  # generous upper bound; we will stop early by time
    it = 0
    alpha_start, alpha_end = 1.0, 500.0  # exponential schedule

    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        # exponential schedule for α
        progress = it / max_adam_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate new point
        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"  Adam it {it:6d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 4. Sub‑gradient descent on the true max
    # --------------------------------------------------------------
    print("--- Phase 2: sub‑gradient descent on true max ---")
    a = best_seq.copy()
    sub_it = 0
    max_sub_iters = 120000
    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, phase, L)

        step_size = 0.5 / np.sqrt(sub_it)
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"  Sub‑grad it {sub_it:6d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 5. Random hill‑climbing with occasional uphill acceptance
    # --------------------------------------------------------------
    print("--- Phase 3: random hill‑climbing (simulated annealing) ---")
    a_current = best_seq.copy()
    cur_max = _compute_convolution(a_current, L).max()
    temp_start = 1e-3
    temp_end = 1e-6
    anneal_iters = 0
    while time.time() < deadline:
        # temperature decays slowly with number of attempts
        tau = (anneal_iters + 1) ** -0.5
        temperature = temp_start * tau + temp_end * (1 - tau)

        # choose source index weighted by its mass, destination uniformly
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # random transfer up to 5 % of the source mass
        delta = np.random.rand() * 0.05 * a_current[i]
        if delta <= 0:
            continue

        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        improve = max_new < cur_max
        if improve or (np.random.rand() < math.exp(-(max_new - cur_max) / temperature)):
            a_current = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max, best_seq = max_new, a_new.copy()
                print(f"  Hill‑climb it {anneal_iters:7d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        anneal_iters += 1

    # --------------------------------------------------------------
    # 6. Final scaling and output
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)  # just for logging
    elapsed = time.time() - start
    print(f"--- Finished (elapsed {elapsed:.1f}s): best score = {final_score:.6f} "
          f"(n={n})  after {it} Adam iters, {sub_it} sub‑grad iters, "
          f"{anneal_iters} hill‑climb attempts ---")
    return final_seq
