* Stage 3 – Polyak sub‑gradient descent on the true max.  The step size is
  chosen as (cur_max − best_max) / ||grad||² (or a diminishing fallback).
* Stage 4 – simulated‑annealing / hill‑climbing with Gaussian perturbations
  (projected back onto the simplex) to escape possible plateaus.
* The best sequence found (still normalised to sum = 1) is finally scaled to
  sum = √(2 n) as required by `evaluate_sequence`.

All heavy work (convolution and gradients) uses FFTs, giving an O(n log n)
per iteration.  The routine respects the supplied time budget (default
1000 s) and prints periodic progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are equal – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)‑1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <w, a*a>  w.r.t. a.
    Implements  2·correlation(w, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    # FFT of reversed a (zero‑padded) = conj(fft_a) * phase
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # centre part of correlation (indices n‑1 … 2n‑2)
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max (log‑sum‑exp)."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _active_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of almost‑maximal entries."""
    m = conv.max()
    tol = max(m * 1e-12, 1e-15)
    active = np.where(conv >= m - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    else:  # safety
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns a plain Python list (the final scaled sequence).
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for printing etc.
    deadline = start + budget_s - 3.0

    # --------------------------------------------------------------
    # 0.  Determine the length and initial vector
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the global namespace
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = len(init_seq)
        init_seq = init_seq / init_seq.sum()
        print(f"[init] Using global construction (n={n})")
    except Exception:
        # fallback: a modest size that fits comfortably in the time budget
        n = 4000
        init_seq = None
        print(f"[init] No global construction – using n={n}")

    # pre‑compute FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    phase = np.exp(2j * np.pi * np.arange(L) / L)   # for reversed FFT

    # --------------------------------------------------------------
    # 1.  Parametric sweep + random Dirichlet samples
    # --------------------------------------------------------------
    best_seq = None
    best_max = np.inf

    def _update_best(cand, name=""):
        nonlocal best_seq, best_max
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            extra = f" ({name})" if name else ""
            print(f"[sweep] New best{extra}: max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # start from global or uniform
    if init_seq is not None:
        _update_best(init_seq, "global")
    else:
        uniform = np.full(n, 1.0 / n)
        _update_best(uniform, "uniform")

    xs = np.arange(n, dtype=np.float64)

    # ---- power‑law -------------------------------------------------
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1.0) ** p
        for cand in (base, base[::-1]):
            cand = cand / cand.sum()
            _update_best(cand, f"power p={p:.2f}")

    # ---- exponential ------------------------------------------------
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):
            cand = cand / cand.sum()
            _update_best(cand, f"exp λ={lam:.2f}")

    # ---- double‑exponential (U‑shape) -------------------------------
    for lam in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        cand = u_shape / u_shape.sum()
        _update_best(cand, f"U‑shape λ={lam:.2f}")

    # ---- Beta densities ----------------------------------------------
    alphas = [0.5, 1.0, 2.0, 5.0]
    betas = [0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a - 1.0)) * ((1.0 - x_mid) ** (b - 1.0))
            if np.all(beta_shape > 0):
                for cand in (beta_shape, beta_shape[::-1]):
                    cand = cand / cand.sum()
                    _update_best(cand, f"Beta a={a} b={b}")

    # ---- arcsine (U‑profile) -----------------------------------------
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    for cand in (arcsine, arcsine[::-1]):
        cand = cand / cand.sum()
        _update_best(cand, "arcsine")

    # ---- random Dirichlet samples ------------------------------------
    # 500 samples – cheap and often find hidden valleys
    for _ in range(500):
        if time.time() > deadline:
            break
        # using concentration 0.7 to give slightly spiky vectors
        cand = np.random.dirichlet(np.full(n, 0.7))
        _update_best(cand, "dirichlet")

    # --------------------------------------------------------------
    # 2.  Adam on smoothed max
    # --------------------------------------------------------------
    print("[stage] Adam with smooth‑max")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    max_adam_iters = 80000
    it = 0
    alpha_start, alpha_end = 4.0, 400.0

    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"[adam] it {it:5d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 3.  Polyak sub‑gradient descent on the true max
    # --------------------------------------------------------------
    print("[stage] Polyak sub‑gradient descent")
    a = best_seq.copy()
    sub_iters = 60000
    sub_it = 0
    while time.time() < deadline and sub_it < sub_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        w = _active_weights(conv)
        grad = _gradient_from_weights(a, w, phase, L)

        # Polyak step size (if best_max already fixed, fall back)
        if best_max < np.inf:
            step_sz = (cur_max - best_max) / (np.linalg.norm(grad) ** 2 + 1e-12)
            # safeguard against pathological step sizes
            if step_sz <= 0 or not np.isfinite(step_sz):
                step_sz = 0.5 / np.sqrt(sub_it)
        else:
            step_sz = 0.5 / np.sqrt(sub_it)

        a_new = a - step_sz * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"[sub‑g] it {sub_it:5d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 4.  Simulated annealing / random hill‑climbing
    # --------------------------------------------------------------
    print("[stage] Simulated annealing")
    a_current = best_seq.copy()
    cur_max = best_max
    # temperature schedule: start moderate, exponential decay
    T = max(1e-3 * cur_max, 1e-7)
    sigma = 0.02                      # initial perturbation strength (relative)
    anneal_it = 0
    while time.time() < deadline:
        anneal_it += 1
        # Gaussian perturbation in log‑space to keep positivity
        log_a = np.log(a_current + 1e-30)
        log_a_new = log_a + np.random.randn(n) * sigma
        cand = np.exp(log_a_new)
        cand = _project_simplex(cand)

        conv_new = _compute_convolution(cand, L)
        max_new = conv_new.max()
        delta = max_new - cur_max

        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / (T + 1e-30))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a_current = cand
            cur_max = max_new
            # keep global best
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"[anneal] it {anneal_it:6d} improved max={cur_max:.8e}  score={2*n*cur_max:.6f}")

        # cooling
        T *= 0.9995
        sigma *= 0.9998
        # stop cooling when temperature becomes negligible
        if T < 1e-10:
            break

    # --------------------------------------------------------------
    # 5.  Final scaling to the norm required by evaluate_sequence
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    # The next line is only for logging – `evaluate_sequence` will be
    # available in the execution environment.
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] best score = {final_score:.6f}  (n={n})")
    except Exception:
        # during offline testing `evaluate_sequence` may be unavailable
        print("[final] evaluation not performed (function unavailable).")
    return final_seq
