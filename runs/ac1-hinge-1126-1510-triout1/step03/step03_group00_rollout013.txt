import scipy.sparse as sp

def build_conv_matrix(a):
    n = len(a)
    rows = 2*n-1
    cols = n
    data = []
    offsets = []
    for i, ai in enumerate(a):
        length = n - i
        if length <= 0:
            continue
        diag = np.full(length, ai)
        data.append(diag)
        offsets.append(i)
    A = sp.diags(data, offsets, shape=(rows, cols), format='csr')

"""
Hybrid optimizer for the step‑function max‑convolution problem.

1.  A continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution (several random restarts).
2.  Projected sub‑gradient descent with a back‑tracking line‑search.
3.  A Frank‑Wolfe / conditional‑gradient refinement: solve a linear program
    that maximises the ℓ1‑norm of a vector `g` while keeping the convolution
    `a * g` below the current worst value.  The solution is mixed with the
    current iterate.
4.  A short hill‑climb (mass transfer) for final polishing.

All steps respect the probability simplex (non‑negative, sum = 1) and use
FFT‑based convolutions for speed.  The algorithm runs for at most the given
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns real part truncated to needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search (back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(8):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # tiny move to avoid stalling
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear–programming refinement (conditional gradient)
# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) = a * g (full convolution).
    Row k corresponds to index k of the linear convolution.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    cols = n
    # each diagonal offset i (0 ≤ i < n) contains the same value a[i]
    data = [np.full(cols, ai, dtype=np.float64) for ai in a]
    offsets = list(range(n))
    A = sp.diags(data, offsets, shape=(rows, cols), format='csr')
    return A


def refine_lp(a: np.ndarray,
              deadline: float,
              fft_len: int,
              n: int,
              max_iters: int = 12,
              mix_alpha: float = 0.5):
    """
    One–shot Frank‑Wolfe / LP refinement.
    Repeatedly solves   max   sum(g)   s.t.   a * g ≤ M, g ≥ 0
    and mixes the normalised solution back into the current iterate.
    """
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current worst convolution value
        conv = full_conv(best_a, fft_len)
        M = conv.max()

        # construct constraint matrix a * g ≤ M
        A = build_conv_matrix(best_a)

        # Linear programme: maximise 1ᵀg  ⇔ minimise (‑1)ᵀg
        c = -np.ones(n, dtype=np.float64)          # minimise (‑1)ᵀg
        b = np.full(2 * n - 1, M + 1e-12)          # tiny slack for numerics
        bounds = [(0.0, None) for _ in range(n)]

        # Solve with HiGHS (fast interior‑point)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            print("[refine_lp] LP failed (iter %d):", it, res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Normalise to the simplex
        g_norm = g / sum_g

        # Try mixing – we keep the first mixing factor that yields improvement
        improved = False
        for alpha in [mix_alpha, mix_alpha * 0.5, mix_alpha * 0.25, mix_alpha * 0.125]:
            a_candidate = (1.0 - alpha) * best_a + alpha * g_norm
            cand_score = fast_score(a_candidate, fft_len, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                improved = True
                break

        if not improved:
            # If none of the mixes improve, exit early
            break

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (mass transfer) – final polishing
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """
    Greedy mass‑transfer hill‑climb (keeps sum = 1).
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move a small fraction (up to 5 %) of mass from i to j
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 2000                # length of the sequence (feel free to change)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # ------------------------------------------------------------------
    # Initial distribution (use previous best if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size > 0:
            if init_seq.size != n:
                # truncate / pad to required length
                if init_seq.size > n:
                    init_seq = init_seq[:n]
                else:
                    pad = np.full(n - init_seq.size, 0.0)
                    init_seq = np.concatenate([init_seq, pad])
            if init_seq.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient
    phase3_end = deadline                       # LP refinement + final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.4, 0.3, 0.2, 0.15,
                    0.10, 0.07, 0.04, 0.02, 0.01, 0.005]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        # random starting distribution
        mode = rng.choice(['random', 'exp', 'power', 'edge'])
        if mode == 'random':
            a_tmp = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:  # edge‑heavy heuristic
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
        a_tmp /= a_tmp.sum()

        # run continuation L‑BFGS
        a_opt, score_opt = lbfgs_continuation(a_tmp,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP based conditional gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        lp_deadline = phase3_end
        best_a, best_score = refine_lp(best_a,
                                       lp_deadline,
                                       fft_len_conv,
                                       n,
                                       max_iters=12,
                                       mix_alpha=0.5)
        print(f"[LP‑refine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=5000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function supplied by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq


# ----------------------------------------------------------------------
# Helper used by the L‑BFGS phase (same as the original script)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)        # logits for the simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # a quick hill‑climb after each lam to shake out residuals
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=800)
    return best_a, best_score
