"""
Hybrid Adam‑gradient descent on a smooth approximation of the L∞
self‑convolution norm, followed by cheap stochastic local refinement.
The optimisation works on the probability simplex (Σa_i = 1, a_i ≥ 0);
the final returned sequence is scaled by the optimizer itself, which
preserves the evaluation score because the objective is homogeneous.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max = Σ w_k (a*a)_k  w.r.t a.
    Implements g_i = 2·Σ_j w_{i+j} a_j   (indices outside range are ignored).
    Both a and w are non‑negative, Σ w = 1.
    """
    n = a.shape[0]
    # reverse w and convolve with a (FFT)
    w_rev = w[::-1]
    L = 1 << ((len(w_rev) + n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(A * W, n=L)[: (len(w_rev) + n - 1)]  # length 3n‑2
    # pick the slice that corresponds to Σ_j w_{i+j} a_j
    start = 2 * n - 2                     # highest index needed
    idx = np.arange(start, start - n, -1)  # start, start‑1, …, start‑n+1
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of max(conv(a,a))
# ----------------------------------------------------------------------
def adam_smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 12.0,
    step0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max of the self‑convolution.
    The smoothing parameter β is linearly increased from beta_start to
    beta_end during the run, turning the objective from a soft‑average
    to the true max.
    Returns the best sequence (on the simplex) encountered and its
    corresponding score 2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = float("inf")

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # pre‑compute schedule length to avoid division by zero
    if max_iters < 2:
        max_iters = 2

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- evaluate current solution -----
        conv = full_convolution(a)
        max_b = float(conv.max())
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[Adam it {it}/{max_iters}] new best score={score:.6f}")

        # ----- smoothing parameter β (soft‑max temperature) -----
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)

        # ----- soft‑max weights -----
        scaled = conv * beta
        shift = scaled.max()
        w = np.exp(scaled - shift)
        w = w / w.sum()

        # ----- gradient (smooth max) -----
        grad = weighted_gradient(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep on the simplex -----
        a = project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap random walk that moves a tiny amount of mass between two
    coordinates, accepting uphill moves with a slowly cooling temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * full_convolution(a).max()

    temperature = 0.015

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # choose a direction of mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # amount to move – up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * full_convolution(new).max()

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # cool down
        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score
