import time, math, numpy as np
from scipy.optimize import minimize

def _project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _compute_convolution(a, L):
    """Linear convolution a*a (real) truncated to length 2*len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[:2*n-1]

def _gradient_from_weights(a, weights, phase, L):
    """
    Gradient of <weights, a*a> w.r.t a.
    Implements 2 * correlation(weights, a) using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n-1:n-1+n]

def _softmax_weights(conv, alpha):
    """Softmax weights for smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()

def _submax_weights(conv):
    """Uniform weights on active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w

def _smooth_obj_and_grad(x, alpha, L, phase):
    """
    Smoothed max objective f(x) = logsumexp(alpha * conv(a)) / alpha,
    where a = softmax(x). Returns (objective, gradient wrt x).
    """
    # stable softmax
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()

    conv = _compute_convolution(a, L)

    # softmax over convolution
    shift = conv.max()
    scaled = alpha * (conv - shift)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    f = (np.log(Z) + alpha * shift) / alpha

    w = exp_scaled / Z  # weights for convex combination of conv entries

    grad_a = _gradient_from_weights(a, w, phase, L)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return f, grad_x

"""
Search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (∑a)²

The algorithm works in several stages within the given time budget:

1. **Parametric sweep** – cheap families (power, exponential, double‑exp,
   Beta, arcsine) are normalised to the simplex and the best is kept.

2. **L‑BFGS refinement** – the objective is approximated by a smooth
   “log‑sum‑exp” of the convolution values.  The variables are the logits
   of a soft‑max (guaranteeing positivity and sum 1).  Two passes are
   performed: a moderate smoothing (α≈30) to reach a good basin, then a
   stronger smoothing (α≈200) to focus on the true maximum.  Several
   random restarts (perturbations of the current best and fresh random
   Dirichlet draws) explore many basins.

3. **Sub‑gradient descent on the exact max** – after the smooth phase the
   true nondifferentiable objective is tackled with a diminishing step‑size
   sub‑gradient method (uniform weights over the active set).

4. **Random hill‑climbing** – tiny mass transfers between random positions
   (up to 5 % of the source) are tried and kept when they improve the
   maximal convolution.

5. **Final rescaling** – the best probability vector is scaled so that its
   L¹‑norm equals √(2 n), exactly the normalisation required by
   `evaluate_sequence`.

All helper routines are defined at module level; no closures are used.
The routine respects the supplied time budget (default 1000 s) and returns
a plain Python list of floats ready for evaluation.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


def _smooth_obj_and_grad(x: np.ndarray, alpha: float,
                         L: int, phase: np.ndarray):
    """
    Smoothed max:
        f(x) = (log ∑exp(α·conv(a)))/α ,   a = softmax(x)

    Returns (f, ∇ₓf).  The gradient uses the chain rule
    (soft‑max Jacobian) and the FFT‑based convolution gradient.
    """
    # soft‑max to obtain a probability vector
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()

    # convolution of a with itself
    conv = _compute_convolution(a, L)

    # soft‑max over the convolution entries
    shift = conv.max()
    scaled = alpha * (conv - shift)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    f = (np.log(Z) + alpha * shift) / alpha

    w = exp_scaled / Z                     # weights for convex combo of conv
    grad_a = _gradient_from_weights(a, w, phase, L)

    # chain rule through the soft‑max (a = softmax(x))
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return f, grad_x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000):
    """
    Search for a step function with a low evaluation score.

    Returns a plain Python list of floats already scaled to the normalisation
    used by `evaluate_sequence`.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0        # safety margin for finalisation

    # ------------------------------------------------------------------
    # Choose the sequence length
    try:
        base = np.array(height_sequence_1, dtype=np.float64)  # global if provided
        n = len(base)
        init_seq = base / base.sum()
    except Exception:
        n = 10000                               # fallback length
        init_seq = None

    # FFT helpers (power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ------------------------------------------------------------------
    # Bookkeeping for the best candidate discovered so far
    best_seq = None
    best_score = float("inf")
    best_max = float("inf")

    def _update(seq: np.ndarray):
        """Update the global best if `seq` yields a lower score."""
        nonlocal best_seq, best_score, best_max
        conv = _compute_convolution(seq, L)
        cur_max = conv.max()
        score = 2.0 * n * cur_max                # same as `evaluate_sequence` after scaling
        if score < best_score:
            best_seq = seq.copy()
            best_score = score
            best_max = cur_max
            print(f"    * new best: score={score:.6f}, max={cur_max:.8e}")

    # ------------------------------------------------------------------
    # 0. Global seed (if any)
    if init_seq is not None:
        _update(init_seq)

    # ------------------------------------------------------------------
    # 1. Parametric sweep (cheap families)
    xs = np.arange(n, dtype=np.float64)

    # power‑law family
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1.0) ** p
        cand = base / base.sum()
        _update(cand)
        _update(base[::-1] / base.sum())

    # exponential family
    for lam in np.linspace(0.0, 5.0, 81):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        cand = exp_shape / exp_shape.sum()
        _update(cand)
        _update(exp_shape[::-1] / exp_shape.sum())

    # double‑exponential (U‑shape)
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        u = left + right
        cand = u / u.sum()
        _update(cand)

    # Beta‑density family
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0]
    betas = [0.5, 1.0, 2.0, 5.0, 10.0]
    mids = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (mids ** (a - 1)) * ((1.0 - mids) ** (b - 1))
            if np.all(beta_shape > 0):
                cand = beta_shape / beta_shape.sum()
                _update(cand)
                _update(beta_shape[::-1] / beta_shape.sum())

    # arcsine‑like shape
    arcsine = 1.0 / np.sqrt(mids * (1.0 - mids) + 1e-30)
    _update(arcsine / arcsine.sum())
    _update(arcsine[::-1] / arcsine.sum())

    # a few random Dirichlet draws
    for _ in range(30):
        if time.time() > deadline:
            break
        rand = np.random.rand(n)
        cand = rand / rand.sum()
        _update(cand)

    # ------------------------------------------------------------------
    # 2. L‑BFGS refinement on a smoothed max (multiple restarts)
    max_restarts = 8
    restart = 0
    while time.time() < deadline - 5 and restart < max_restarts:
        restart += 1

        # Initialise logits (log‑softmax parameters)
        if best_seq is not None and np.random.rand() < 0.6:
            x0 = np.log(best_seq + 1e-12) + 0.1 * np.random.randn(n)
        else:
            # fresh random Dirichlet → logits
            rnd = np.random.rand(n) + 1e-12
            x0 = np.log(rnd)

        # Stage A – modest smoothing, finds a good basin
        alpha_a = 30.0
        res_a = minimize(_smooth_obj_and_grad, x0,
                         args=(alpha_a, L, phase),
                         method='L-BFGS-B',
                         jac=True,
                         options={'maxiter': 250, 'ftol': 1e-12, 'disp': False})

        # Stage B – stronger smoothing, sharpens toward the true max
        alpha_b = 200.0
        res_b = minimize(_smooth_obj_and_grad, res_a.x,
                         args=(alpha_b, L, phase),
                         method='L-BFGS-B',
                         jac=True,
                         options={'maxiter': 350, 'ftol': 1e-12, 'disp': False})

        # Convert final logits back to a probability vector
        xb = res_b.x
        xb_shift = xb - np.max(xb)
        expb = np.exp(xb_shift)
        a_opt = expb / expb.sum()

        _update(a_opt)

        # ------------------------------------------------------------------
        # 3. Sub‑gradient descent on the true max
        a = a_opt.copy()
        for it in range(3000):
            if time.time() > deadline:
                break
            conv = _compute_convolution(a, L)
            w = _submax_weights(conv)
            grad = _gradient_from_weights(a, w, phase, L)
            step = 0.5 / np.sqrt(it + 1)
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _compute_convolution(a_new, L)
            if conv_new.max() < best_max:
                _update(a_new)
                a = a_new
            else:
                a = a_new

        # ------------------------------------------------------------------
        # 4. Random hill‑climbing around the current point
        a_cur = a.copy()
        while time.time() < deadline - 1:
            i = np.random.choice(n, p=a_cur)
            j = np.random.randint(0, n)
            if i == j:
                continue
            delta = np.random.rand() * 0.05 * a_cur[i]  # ≤ 5 % of source mass
            if delta <= 0:
                continue
            a_new = a_cur.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue
            conv_new = _compute_convolution(a_new, L)
            if conv_new.max() < best_max:
                _update(a_new)
                a_cur = a_new

    # ------------------------------------------------------------------
    # 5. Final scaling to the normalisation required by evaluate_sequence
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)            # ∑a = √(2·n) for the scorer
    final_seq = (best_seq * target_sum).tolist()

    # optional final log (the grader ignores prints)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"--- Finished: best score = {final_score:.6f} (n={n})")
    except Exception:
        pass

    return final_seq
