def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000):
    conv = conv_fft(a, fft_len, n)
    K = np.argmax(conv)
    # compute contributions vector for K
    rev_a = np.zeros(n)
    start = max(0, K - (n-1))
    end = min(n-1, K)
    # i runs from start..end: a_i * a_{K-i}
    # define contribution per i as a_{K-i}
    # compute reversed array:
    g = np.zeros(n)
    for i in range(start, end+1):
        j = K - i
        g[i] = a[j]
        g[j] = a[i]  # also because product appears twice for i!=j
    # Use for selection.
    # Implement loops with random selection from top contributions.

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # sample source i with probability proportional to a[i] * g[i]
        probabilities = a * g
        sum_prob = probabilities.sum()
        if sum_prob == 0:
            break
        probs = probabilities / sum_prob
        i = rng.choice(n, p=probs)
        # sample dest j uniformly or biased to small product
        j = rng.choice(n)  # uniform
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.05
        if delta <= 0:
            continue
        # move delta from i to j
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return a, cur_score

The algorithm works on the probability simplex using a soft‑max

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector (sum=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector.
    Returns 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max coordinates y.

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for the convolution and the auxiliary
        convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width:
        λ = max(conv) * lam_factor.

    Returns
    -------
    f : float
        Smoothed approximation of 2·n·max(a*a).
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y`` (same shape as ``y``).
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max via log‑sum‑exp
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # gradient of each conv entry w.r.t. a:
    # (a * w) reversed convolution
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # we need only the n entries that correspond to indices 0..n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def generic_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    max_move_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb on the simplex (like the previous code).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move_frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves probability mass away from
    coordinates that contribute most to the current worst convolution peak.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute current convolution and index of the worst peak
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # contribution of each index to that peak: for a given k,
        # contribution of a_i is 2 * a_{k-i} (except when i = k-i).
        contrib = np.zeros(n)
        lower = max(0, k - (n - 1))
        upper = min(n - 1, k)
        # i runs over [lower, upper]
        idx = np.arange(lower, upper + 1, dtype=int)
        opp = k - idx
        contrib[idx] = best[opp]
        contrib[opp] = best[idx]   # double‑count, ok for weighting

        # source candidates = large contribution * large value
        src_weights = best * contrib
        if src_weights.sum() == 0:
            break
        src_weights /= src_weights.sum()
        i = rng.choice(n, p=src_weights)

        # destination candidates = small contribution (prefer places that hardly affect the peak)
        dst_weights = 1.0 / (1.0 + contrib)
        dst_weights /= dst_weights.sum()
        j = rng.choice(n, p=dst_weights)

        if i == j:
            continue

        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (for reproducibility).
    budget_s : int
        Total time budget in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – try a slightly larger n if time permits.
    # ------------------------------------------------------------------
    n = 5000
    # attempt to reuse a previously discovered construction
    try:
        prev = globals().get("height_sequence_1")
        if prev is not None:
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, int(arr.size))
    except Exception:
        pass

    # if we still have a lot of time left, bump n up a bit (more freedom)
    if time.time() + 200 < deadline:
        n = max(n, 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise with the uniform distribution (baseline)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.8, 0.001, num=30)   # smoother start → tighter finish
    maxiter_stage = 400                               # L‑BFGS iterations per stage

    iteration = 0

    # ------------------------------------------------------------------
    # multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 6.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.0005, 0.015)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # a few heavier edge buckets, smoother centre
            m_edge = max(1, int(n * rng.uniform(0.008, 0.025)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final deep refinement (tight continuation schedule)
    # ------------------------------------------------------------------
    fine_lam_factors = np.geomspace(0.002, 1e-6, num=25)
    for lam in fine_lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
