def compute_smoothed_max_and_gradient(a, lam):
    # a is np.ndarray length n, sum=1
    n = a.shape[0]
    # compute conv via FFT (full length M=2n-1)
    fft_len = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]  # real conv
    max_conv = conv.max()
    # use lam to compute softmax
    scaled = lam * (conv - max_conv)   # subtract max to avoid overflow
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp  # shape M

    # compute gradient via convolution with reversed w
    w_rev = w[::-1]
    # pad size for convolution length: use fft again
    L = 1 << ((len(w_rev) + n - 1).bit_length())
    A1 = np.fft.rfft(a, n=L)
    B1 = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(A1 * B1, n=L)[:len(w_rev) + n - 1]
    # extract appropriate slice
    M = len(w)
    # g_i = 2 * conv_rev[M - 1 - i]
    g = 2.0 * conv_rev[M - 1 - np.arange(n)]
    # smoothed max value = max_conv + (1/lam) log(sum_exp)
    smoothed = max_conv + (1.0/lam) * np.log(sum_exp)
    return smoothed, g, max_conv

def compute_grad_smoothed_max(a, lam):
    n = a.shape[0]
    # FFT conv of a with itself
    fft_len = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
    max_conv = float(conv.max())
    # compute w = softmax(lam*conv) with shift for numerical stability
    scaled = lam * (conv - max_conv)  # subtract max for stability
    exp_scaled = np.exp(scaled)
    sum_exp = float(exp_scaled.sum())
    w = exp_scaled / sum_exp  # shape M=2n-1

    # compute gradient: g_i = 2 * sum_j a_j w_{i+j}
    # compute w_rev and convolution
    w_rev = w[::-1]
    L = 1 << ((len(w_rev) + n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    w_fft = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(a_fft * w_fft, n=L)[:len(w_rev) + n - 1]
    M = len(w)
    # g = 2 * conv_rev[M-1 - np.arange(n)]
    g = 2.0 * conv_rev[M-1 - np.arange(n)]
    # smoothed max value (optional)
    smoothed = max_conv + (1.0/lam) * np.log(sum_exp)
    return smoothed, g, max_conv

            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
        temperature *= 0.9999

"""
Hybrid smooth‑max projected gradient descent with random restarts.

The objective 2·n·max(conv(a,a)) is homogeneous, so we keep a on the
probability simplex (a_i ≥ 0, Σa_i = 1).  Instead of the non‑smooth max,
we minimise a smooth approximation
    f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
and follow its gradient, which can be obtained efficiently via a single
FFT‑based convolution.  λ is increased exponentially during the run so
that the optimisation first moves globally (small λ) and later focuses
on the true peak (large λ).  After a projected gradient phase we apply
a cheap stochastic refinement (mass transfer between two coordinates)
to escape shallow minima.  Several independent restarts with different
initialisations and a few problem sizes are performed until the time
budget expires.  The best sequence observed is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of a full convolution a*a."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
# Smoothed max and its gradient (computed via FFT)
# ----------------------------------------------------------------------
def smoothed_max_and_grad(a: np.ndarray, lam: float):
    """
    Compute the smooth approximation of max(conv(a,a)) together with the gradient.
    Returns (smoothed_value, gradient, raw_max_conv).
    """
    n = a.shape[0]
    # ----- full self‑convolution (FFT) -----
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())

    # ----- soft‑max weights -----
    # shift by max for numerical stability
    shifted = lam * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    w = exp_shifted / sum_exp               # shape (2n‑1,)

    # ----- gradient = 2·(a * w) (cross‑correlation) -----
    # gradient_i = 2·Σ_j a_j·w_{i+j}
    w_rev = w[::-1]
    L = 1 << ((len(w_rev) + n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    w_fft = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(a_fft * w_fft, n=L)[: len(w_rev) + n - 1]
    M = len(w)
    grad = 2.0 * conv_rev[M - 1 - np.arange(n)]

    # ----- smooth max value -----
    smooth = max_conv + (1.0 / lam) * np.log(sum_exp)

    return smooth, grad, max_conv

# ----------------------------------------------------------------------
# Projected gradient descent on the simplex using the smooth max.
# ----------------------------------------------------------------------
def gradient_descent_smooth(a0: np.ndarray, max_iters: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the simplex using the
    smooth‑max surrogate.  λ grows from 5 to 200 exponentially.
    Returns the best sequence seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    a = a / a.sum()                     # enforce Σa = 1
    n = a.shape[0]

    # true objective (no smoothing)
    best_score = 2.0 * n * compute_max_conv(a)
    best_seq = a.copy()

    lam_start = 5.0
    lam_end = 200.0
    step0 = 0.7                         # initial step magnitude

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for λ
        lam = lam_start * (lam_end / lam_start) ** ((it - 1) / max_iters)

        # smooth surrogate and its gradient
        _, grad, max_conv = smoothed_max_and_grad(a, lam)

        # diminishing step size
        step = step0 / np.sqrt(it)

        # simple backtracking line‑search (max 5 reductions)
        cur_score = 2.0 * n * max_conv
        accepted = False
        for _ in range(5):
            cand = project_simplex(a - step * grad)
            cand_max = compute_max_conv(cand)
            cand_score = 2.0 * n * cand_max
            if cand_score <= cur_score or step < 1e-12:
                a = cand
                cur_score = cand_score
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # fall back to one projected step
            a = project_simplex(a - step * grad)
            cur_score = 2.0 * n * compute_max_conv(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[gd n={n} it={it} λ={lam:.1f} score={cur_score:.6f}]")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap random walk that moves a small amount of mass between two
    random coordinates, accepting improving moves (and occasional
    uphill moves).  Returns the best sequence and its true score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)
    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue
        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s
        new_score = 2.0 * n * compute_max_conv(new)
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # candidate lengths (larger sizes give a finer discretisation)
    length_candidates = [2000, 4000, 8000]

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main search loop: random restarts for each length.
    # ------------------------------------------------------------------
    for n in length_candidates:
        while time.time() < deadline:
            # ----- initialise a fresh starting point -----
            r = np.random.rand()
            if r < 0.35:
                # uniform start
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif r < 0.70:
                # random power‑law (decaying tail)
                p = np.random.uniform(0.45, 0.85)
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()
            else:
                # perturb the current global best if lengths match
                if global_best_seq is not None and len(global_best_seq) == n:
                    a0 = np.array(global_best_seq, copy=True)
                    a0 += 0.02 * np.random.randn(n)
                    a0 = np.maximum(a0, 0.0)
                    s = a0.sum()
                    if s == 0.0:
                        a0 = np.full(n, 1.0 / n)
                    else:
                        a0 /= s
                else:
                    # Dirichlet‑type random vector
                    a0 = np.random.gamma(1.0, 1.0, size=n)
                    a0 /= a0.sum()

            # ----- decide how many iterations we can afford -----
            time_left = deadline - time.time()
            if time_left < 0.2:
                break
            # rough estimate: ~0.0015 s per iteration for n≈4000
            est_iters = int(time_left / 0.0015)
            max_iters = max(200, min(est_iters, 2500))

            # ----- run projected smooth‑max gradient descent -----
            seq, score = gradient_descent_smooth(a0, max_iters, deadline)

            # ----- optional cheap stochastic refinement -----
            time_now = time.time()
            if deadline - time_now > 0.6:
                # a few thousand refinement steps are cheap
                ref_steps = int(min(6000, (deadline - time_now) / 0.0007))
                seq_ref, score_ref = local_refine(seq, ref_steps, deadline)
                if score_ref < score:
                    seq, score = seq_ref, score_ref

            # ----- update global best -----
            if score < global_best_score:
                global_best_score = score
                global_best_seq = seq.copy()
                print(f"[len={n}] new global best score={score:.6f}")
                # early stop if target is already achieved
                if global_best_score < 1.503:
                    deadline = time.time()  # force exit of outer loops
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a /= a.sum()
        global_best_seq = a

    # Convert to a plain Python list of floats as required by the harness.
    return [float(x) for x in global_best_seq.tolist()]
