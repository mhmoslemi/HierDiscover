"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
1. **Continuation L‑BFGS** on a smooth approximation of the max‑convolution.
   The smoothing parameter (a factor of the current max) is decreased
   progressively (0.6 → 0.001).  Several random restarts are performed
   within the first part of the time budget.

2. **Linear‑programming improvement**.
   Given the current distribution `f`, solve the LP

        maximise   sum(b)
        subject to b ≥ 0,
                   (f * b) ≤ M   (M = max(f*f))

   The LP tries to increase the total mass while keeping the cross‑convolution
   below the current worst value.  The optimal `b` is rescaled to have the same
   ℓ₁‑norm as `f` and mixed with `f` (convex combination).  This step can
   strictly reduce the max‑convolution and is repeated while time permits.

3. **Projected sub‑gradient descent** with a simple back‑tracking line search,
   directly minimising the true max‑convolution.

4. **Simulated‑annealing / random perturbations** to escape possible local minima.

5. **Hill‑climbing** (mass transfers between two coordinates) until the deadline.

All steps work on the probability simplex (non‑negative, sum = 1) and use FFT‑based
convolutions for O(n log n) speed.  The final sequence is returned as a list of
floats; it is automatically clipped by `evaluate_sequence` to the required
range.
"""

import time
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a already sums to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    """
    a = np.exp(y - np.max(y))
    a = a / a.sum()                     # softmax – maps y → simplex

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # a few hill‑climbs after each lam can be helpful
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline, max_iter=1500)
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with simple back‑tracking line search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses back‑tracking line search to adapt the step size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search (simple back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T *= 0.92
            T = max(T, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def lp_improve_step(f: np.ndarray, n: int, M: float, t: float,
                    fft_len_conv: int):
    """
    Solve the LP  maximise sum(b)  subject to b ≥ 0,
    (f * b) ≤ M   (full convolution).
    Return the mixed distribution (1‑t)·f + t·g where g = b / sum(b)
    and the new max‑convolution value.
    """
    # Build the convolution matrix A (size (2n‑1) × n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear programme
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]               # broadcasting M to each row
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # warm start with current f (feasible point)
    b.value = f
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-8, reltol=1e-8)
    except Exception as e:
        print("[LP] solver failed:", e)
        return f, M, False

    if b.value is None:
        return f, M, False

    b_val = np.maximum(b.value, 0.0)
    sum_b = b_val.sum()
    # No improvement possible if sum(b) ≤ 1 (cannot enlarge mass)
    if sum_b <= 1.0 + 1e-12:
        return f, M, False

    g = b_val / sum_b                # rescale to unit ℓ₁‑norm
    f_new = (1.0 - t) * f + t * g
    f_new = f_new / f_new.sum()      # numerical renormalisation
    conv = full_conv(f_new, fft_len_conv)
    M_new = conv.max()
    return f_new, M_new, True

    return final_seq
