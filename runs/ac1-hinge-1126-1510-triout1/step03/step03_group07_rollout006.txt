"""
Multistage search for a low‑score step function.

The algorithm explores a few alternative lengths (around the known good
length), builds several simple families (arcsine, power‑law,
exponential, random Dirichlet) and keeps the best normalized candidate.
For each length a short Adam optimisation on a smooth “log‑sum‑exp”
approximation of the max‑convolution is performed, followed by a
sub‑gradient descent with a back‑tracking line‑search on the true max.
The best length obtained is then refined with a long Adam run (α up to
500), another sub‑gradient phase and a cheap random hill‑climb.  All
updates stay on the probability simplex via an Euclidean projection.
The final sequence is scaled to the normalisation required by
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # fall‑back
        w[conv.argmax()] = 1.0
    return w


def _resample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq / seq.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, seq)
    return interp / interp.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine respects the supplied time budget and returns a plain
    Python list of floats suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore.
    try:
        base_seq = np.array(height_sequence_1, dtype=np.float64)  # may be defined globally
        n0 = len(base_seq)
    except Exception:
        base_seq = None
        n0 = 5600

    length_candidates = [
        max(2000, int(round(0.8 * n0))),
        int(round(1.0 * n0)),
        min(15000, int(round(1.2 * n0))),
    ]
    length_candidates = sorted(set(length_candidates))

    best_seq_global = None
    best_score_global = float("inf")
    best_n_global = None

    # ------------------------------------------------------------------
    # Helper to compute the objective for a vector that already sums to 1.
    def _score_of(a: np.ndarray, n: int) -> float:
        conv = _compute_convolution(a, 1 << ((2 * n - 1).bit_length()))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Phase 0 – try a few simple families for each length and keep the best.
    for n in length_candidates:
        if time.time() > deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        L = 1 << ((2 * n - 1).bit_length())
        phase = np.exp(2j * np.pi * np.arange(L) / L)

        # pool of initial candidates for this length
        cands = []

        # 1) arcsine (Beta(0.5,0.5))
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        arcsine = 1.0 / np.sqrt(x * (1.0 - x) + 1e-30)
        c = arcsine / arcsine.sum()
        cands.append(c)

        # 2) power‑law families
        for p in (0.2, 0.5, 1.0, 1.5, 2.0):
            pw = (np.arange(n, dtype=np.float64) + 1) ** p
            c = pw / pw.sum()
            cands.append(c)
            cands.append(c[::-1])

        # 3) exponential families
        for lam in (0.5, 1.0, 2.0, 3.0):
            expv = np.exp(-lam * np.arange(n, dtype=np.float64))
            c = expv / expv.sum()
            cands.append(c)
            cands.append(c[::-1])

        # 4) random Dirichlet samples (a few)
        for _ in range(3):
            rnd = np.random.rand(n)
            c = rnd / rnd.sum()
            cands.append(c)

        # 5) resample the known construction if available
        if base_seq is not None:
            c = _resample_seq(base_seq, n)
            cands.append(c)

        # evaluate each candidate, keep the best for this length
        best_cand_len = None
        best_score_len = float("inf")
        for cand in cands:
            cur_score = _score_of(cand, n)
            if cur_score < best_score_len:
                best_score_len = cur_score
                best_cand_len = cand.copy()
            if cur_score < best_score_global:
                best_score_global = cur_score
                best_seq_global = cand.copy()
                best_n_global = n
                print(f"   ** new global best (len={n})  score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # Short optimisation on the best candidate of this length
        a = best_cand_len.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.12
        max_adam = 20000
        for it in range(1, max_adam + 1):
            if time.time() > deadline:
                break
            conv = _compute_convolution(a, L)
            max_cur = conv.max()
            # slowly raise the softmax temperature
            alpha = 4.0 + (400.0 - 4.0) * (it / max_adam)
            w = _softmax_weights(conv, alpha)
            grad = _gradient_from_weights(a, w, phase, L)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = _project_simplex(a_new)

            conv_new = _compute_convolution(a_new, L)
            max_new = conv_new.max()
            if max_new < max_cur:
                score_new = 2.0 * n * max_new
                if score_new < best_score_global:
                    best_score_global = score_new
                    best_seq_global = a_new.copy()
                    best_n_global = n
                    print(f"   Adam (+{it}) improved to {score_new:.6f}")
            a = a_new

        # Sub‑gradient phase (short)
        a = best_cand_len.copy()
        for it in range(1, 30001):
            if time.time() > deadline:
                break
            conv = _compute_convolution(a, L)
            max_cur = conv.max()
            w = _submax_weights(conv)
            grad = _gradient_from_weights(a, w, phase, L)

            step = 0.5 / np.sqrt(it)
            a_new = a - step * grad
            a_new = _project_simplex(a_new)

            conv_new = _compute_convolution(a_new, L)
            max_new = conv_new.max()
            if max_new < max_cur:
                score_new = 2.0 * n * max_new
                if score_new < best_score_global:
                    best_score_global = score_new
                    best_seq_global = a_new.copy()
                    best_n_global = n
                    print(f"   Sub‑grad (+{it}) improved to {score_new:.6f}")
            a = a_new

    # ------------------------------------------------------------------
    # If something went wrong, fall back to a uniform vector.
    if best_seq_global is None:
        n = n0
        best_seq_global = np.full(n, 1.0 / n, dtype=np.float64)
        best_n_global = n

    # ------------------------------------------------------------------
    # Long refinement on the best length discovered.
    n = best_n_global
    a = best_seq_global.copy()
    L = 1 << ((2 * n - 1).bit_length())
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ---- Long Adam ----------------------------------------------------
    print(f"\n=== Long Adam refinement on n={n} ===")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.08
    max_adam = 80000
    for it in range(1, max_adam + 1):
        if time.time() > deadline:
            break
        conv = _compute_convolution(a, L)
        max_cur = conv.max()
        alpha = 4.0 + (500.0 - 4.0) * (it / max_adam)   # stronger smoothing
        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, L)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < max_cur:
            score_new = 2.0 * n * max_new
            if score_new < best_score_global:
                best_score_global = score_new
                best_seq_global = a_new.copy()
                print(f"   Long Adam it {it:5d} new best {score_new:.6f}")
        a = a_new

    # ---- Sub‑gradient with line‑search ---------------------------------
    print(f"\n=== Sub‑gradient with line‑search on n={n} ===")
    for it in range(1, 80001):
        if time.time() > deadline:
            break
        conv = _compute_convolution(a, L)
        max_cur = conv.max()
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, phase, L)

        step = 1.0 / np.sqrt(it)       # initial guess
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _compute_convolution(a_new, L)
            max_new = conv_new.max()
            if max_new <= max_cur:
                break
            step *= 0.5

        if max_new < max_cur:
            score_new = 2.0 * n * max_new
            if score_new < best_score_global:
                best_score_global = score_new
                best_seq_global = a_new.copy()
                print(f"   Sub‑grad LS it {it:5d} new best {score_new:.6f}")
            a = a_new
        else:
            # no improvement – keep the current point
            a = a

    # ---- Cheap random hill‑climbing ------------------------------------
    print("\n=== Random hill‑climbing ===")
    a_current = best_seq_global.copy()
    while time.time() < deadline:
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.03 * a_current[i]   # up to 3 % transfer
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        score_new = 2.0 * n * max_new
        if score_new < best_score_global:
            best_score_global = score_new
            best_seq_global = a_new.copy()
            a_current = a_new
            print(f"   Hill‑climb improved to {score_new:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the normalisation expected by `evaluate_sequence`.
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq_global * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)   # logging only
    print(f"\n--- Finished: final score = {final_score:.6f} with n = {n} ---")
    return final_seq
