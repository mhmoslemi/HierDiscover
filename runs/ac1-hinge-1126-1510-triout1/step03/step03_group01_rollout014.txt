"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1) using the soft‑max
  re‑parameterisation a = softmax(y).
* Initialise with a variety of hand‑crafted shapes (uniform,
  exponential, power‑law, edge‑biased and a quick grid search over
  a few exponent/power parameters).  The best among them is kept as
  a starting point.
* Repeatedly apply a continuation L‑BFGS optimisation on a
  smooth‑max approximation of  max(conv(a,a)).  The smoothing
  parameter λ is decreased geometrically in many stages (≈30).
* After each L‑BFGS run a fast greedy hill‑climb moves a small
  fraction of mass between random indices; a second “swap‑climb”
  exchanges whole values of two positions.  Both keep a move only
  if the exact fast_score = 2·n·max(a*a) improves.
* The whole loop runs until the time budget is exhausted, keeping the
  best sequence found.
* Finally a long refinement (still L‑BFGS with a very small λ and a
  larger iteration budget) followed by a final hill‑climb is performed.

The returned list consists of non‑negative floats (sum > 0) ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.

    lam = max(conv) * lam_factor  (smoothness parameter).
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]   # length n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    max_delta_frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Only keep a move if the exact fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to max_delta_frac of the mass at i
        delta = best[i] * rng.random() * max_delta_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def swap_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_swaps: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Greedy swap‑climb: exchange whole values of two indices.
    Accept only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_swaps > 0:
        max_swaps -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothness.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # The quadratic term of the objective is homogeneous, therefore
        # we keep the same lambda‑factor but use the explicit lam value.
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam / max(1e-12, lam)),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # concentrate a few percent of mass at the edges
        edge_frac = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums or candidate * 2 in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

        a /= a.sum()
        param_candidates.append(np.log(a + 1e-12))

    # exponential family a_i ∝ exp(-α i)
    for rate in np.linspace(0.0005, 0.025, 20):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        param_candidates.append(np.log(a + 1e-12))

    # evaluate the small grid and keep the best
    for y0 in param_candidates:
        a0 = softmax(y0)
        sc = fast_score(a0, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a0
            best_y = y0
            print(f"[grid] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    # a dense geometric schedule for the smooth‑max parameter
    lam_schedule = list(np.geomspace(0.6, 0.0005, num=30))

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    # --------------------------------------------------------------
    # Main stochastic search loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            # small Gaussian perturbation of the current best
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS (coarse → fine) -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=350,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive local refinement -----
        # first a mass‑move hill‑climb
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
            max_delta_frac=0.10,
        )
        # then a swap‑climb to re‑order the positions
        swap_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = swap_climb(
            best_a, best_score, fft_len_conv, n, swap_deadline, max_swaps=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final long‑run refinement
    # --------------------------------------------------------------
    final_lam_schedule = list(np.geomspace(0.2, 5e-5, num=40))
    for lam in final_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam / max(1e-12, lam)),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # ----------------------------------------------------------------
    # final greedy polishing (use whatever time remains)
    # ----------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
            max_delta_frac=0.10,
        )
        best_a, best_score = swap_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_swaps=3000
        )
        best_y = np.log(best_a + 1e-12)

    # ----------------------------------------------------------------
    # official evaluation and return
    # ----------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
