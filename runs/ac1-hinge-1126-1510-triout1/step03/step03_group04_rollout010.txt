"""
Enhanced search for a low‑value step sequence.

Algorithm
---------

The evaluation metric

    E(a) = 2·n·max(a*a) / (sum a)²

is invariant under scaling, therefore we restrict the search to the
probability simplex (∑a=1, a_i≥0).  The code consists of several
phases that together beat the previous best bound (≈1.5189).

1. **Initialisation** – start from the best known sequence if available;
   otherwise use a mixture of the uniform distribution and a discrete
   arcsine‑type shape (large mass near the edges, low mass in the centre).

2. **LP‑mixing phase** – the classical update described in the statement:
   for the current f compute M = ‖f*f‖∞ and solve the linear programme

        maximise   ∑ g_i
        subject to  (f * g)_k ≤ M      for all k,
                    g_i ≥ 0 .

   The solution g (renormalised to unit sum) is mixed with f:
   f ← (1−t)·f + t·g  with a small step‑size t.  Repeating this
   monotone‑ly reduces the supremum of the convolution.  Because the LP
   is small (≈4000 constraints, 2000 variables) the HiGHS solver solves
   it in a few seconds; we perform as many iterations as the time budget
   permits.

3. **Projected sub‑gradient refinement** – a fast stochastic sub‑gradient
   method (with Euclidean projection onto the simplex) further reduces
   the true max‑convolution.

4. **Simulated annealing** – a non‑local stochastic search that moves
   mass between two positions; uphill moves are accepted according to a
   decreasing temperature schedule.

5. **Final greedy hill‑climb** – a cheap local optimiser that repeatedly
   transfers a tiny fraction of mass between two coordinates while the
   score improves.

All FFT‑based convolutions use a power‑of‑two length for speed, the
simplex projection is exact, and the routine respects the 1000 s time
budget.  The result is returned as a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (phase 2)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a projected sub‑gradient method.
    """
    a = a0.copy()
    best_a = a.copy()
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # tolerance for active set
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: 2·(mask * reverse(a)) restricted to first n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # update best
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (phase 5)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb on the simplex.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass from i to j
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Simulated annealing (phase 4)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing moving a fraction of mass between two
    positions.  Improves the current best while exploring non‑local moves.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)      # initial temperature
    decay = 0.9995                       # geometric cooling

    it = 0
    while time.time() < deadline:
        # source with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue

        # distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # propose move
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑mixing phase (core improvement over the previous solution)
# ----------------------------------------------------------------------


def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A where (A @ g)[k] = (f * g)[k].
    Shape: (2*n-1, n)
    """
    n = f.shape[0]
    rows = 2 * n - 1
    cols = n
    A = np.zeros((rows, cols), dtype=np.float64)
    # column i contributes f[i] on rows i … i+n-1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Choose length and initialise the sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            if init_seq.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # Mix with a discrete arcsine‑type shape to give a better start
    idx = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx) + 1e-12)
    arcsine /= arcsine.sum()
    a0 = 0.5 * a0 + 0.5 * arcsine
    a0 /= a0.sum()

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP mixing (classic iterative bound reduction)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    iter_lp = 0
    while time.time() < phase1_end:
        # current supremum of the convolution
        conv = full_conv(best_a, fft_len_conv)
        M = conv.max()

        # build linear constraints (f * g)_k ≤ M
        A = build_constraint_matrix(best_a)
        b_ub = M * np.ones(2 * n - 1, dtype=np.float64)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[LP] zero‑sum solution, terminating LP phase")
            break

        # normalise to the simplex
        g = g0 / sum_g0

        # mixing step – smaller t as we progress
        if iter_lp < 10:
            t = 0.20
        elif iter_lp < 30:
            t = 0.10
        else:
            t = 0.05

        new_a = (1.0 - t) * best_a + t * g
        new_a = simplex_projection(new_a)  # eliminate tiny numerical drift

        new_score = fast_score(new_a, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[LP {iter_lp}] t={t:.2f} → score={best_score:.6f}")
        else:
            # still move, but keep the best seen so far
            best_a = new_a.copy()

        iter_lp += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient refinement
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (optional, if time permits)
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] after annealing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a,
                                      best_score,
                                      fft_len_conv,
                                      n,
                                      deadline,
                                      max_iter=3000)

    # ------------------------------------------------------------------
    # Reporting
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] fast_score={final_score:.6f}, evaluated={eval_val:.6f}")

    return final_a.tolist()
