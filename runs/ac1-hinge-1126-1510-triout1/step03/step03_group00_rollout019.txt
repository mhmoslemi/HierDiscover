assistantfinal**Solution Overview**

The score to minimise is  

```
score(a) = 2·n·max( a * a )          (a is non‑negative,  Σa = 1)
```

`a*a` is the linear convolution of the step heights with themselves.
The problem is non‑convex, however the following practical observations are useful:

*   `max(a*a)` is always attained at a lag that contains a sum of products
    `a_i·a_j`.  If we keep all individual heights `a_i ≤ √t` (where `t` is the
    current best value of `max(a*a)`) the contribution of any single product is
    at most `t`.  The difficulty is to avoid many products falling on the same
    lag.
*   A smooth approximation of the max – log‑sum‑exp – together with a
    continuation on the smoothing parameter allows efficient gradient‑based
    optimisation.
*   After a gradient step we can **tighten** the bound with a tiny linear‑
    programming (LP) sub‑problem: for the current vector `a` we maximise the
    ℓ₁‑norm of a second vector `g` under the *same* convolution bound.
    Adding a short convex combination `a←(1‑τ)a+τg` strictly reduces the
    maximal convolution.
*   Finally a few cheap projection‑based sub‑gradient steps and a short
    simulated‑annealing hill‑climb polish the result.

The implementation follows the same three‑phase schedule that was used
previously, but adds the LP refinement and uses a few more L‑BFGS
iterations / restarts.  All steps respect the probability simplex
(`a_i ≥ 0`, `Σa_i = 1`) and run completely inside the provided
`budget_s` seconds.

```python
"""
search for a low‑score non‑negative step sequence.

The algorithm consists of four phases

1.  Continuation L‑BFGS on a smooth approximation of max(a*a) (log‑sum‑exp)
    with many random restarts.
2.  A tiny linear‑programming refinement: maximise Σg under the same
    convolution bound and move a little towards the optimiser g.
3.  Projected sub‑gradient descent with a back‑tracking line search.
4.  Fast simulated‑annealing / hill‑climb for final polishing.

All phases respect the simplex (non‑negative, sum = 1) and use FFT‑based
convolutions for O(n log n) speed.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns objective 2·n·smooth_max(conv) and gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing parameter
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses back‑tracking line search to adapt the step size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search (simple back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T *= 0.92
            T = max(T, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast construction of the (2n‑1)×n convolution matrix for the LP step
# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Returns the dense matrix A with A[k, j] = a_{k‑j} (0 whenever the index is
    out of range).  Then A @ g = a * g (full convolution).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
# LP refinement – maximise Σg under the same convolution bound
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              n: int,
              fft_len: int,
              deadline: float,
              max_iters: int = 12,
              mix_factor: float = 0.20):
    """
    At most ``max_iters`` linear‑programming refinements.
    Each step solves:
        maximise Σg   subject to   a * g ≤ M   and   g ≥ 0,
    where M = max(a*a).  The optimiser g is normalised to sum = 1 and the
    current a is moved a little towards g.
    """
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # current bound
        conv_aa = full_conv(a, fft_len)
        M = conv_aa.max()

        # build constraints A g ≤ M
        A = build_conv_matrix(a)               # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M)

        # solve LP – minimise -Σg  (i.e. maximise Σg)
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            # LP failed – abort the refinement phase
            break
        g = res.x
        if g.sum() == 0.0:
            break
        g /= g.sum()                # project to simplex

        # convex combination and re‑evaluate
        a = (1.0 - mix_factor) * a + mix_factor * g
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing lam schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)                 # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # short hill‑climb after each λ can be useful
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score
```
