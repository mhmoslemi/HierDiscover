def subgradient_descent(y, n, fft_len, epochs=2000, step0=1.0):
    y_cur = y.copy()
    rng = np.random.default_rng()
    for t in range(epochs):
        a = softmax(y_cur)
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        # find argmax positions
        mask = conv == max_val
        w = mask.astype(float) / mask.sum()  # weight distribution
        # compute gradient wrt a:
        # g_vec = conv of a_rev and w, as earlier.
        # compute a_rev: reversed a
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        fw = np.fft.fft(w, fft_len)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_rev[(n-1):(n-1+n)]
        # gradient of max conv w.r.t y:
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)
        # step size schedule
        eta = step0 / np.sqrt(t+1)
        y_cur = y_cur - eta * grad_y
        # optionally add noise
        if t % 200 == 0:
            y_cur += rng.normal(scale=0.01, size=n)
    return y_cur
```

def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12
    while time.time() < deadline and max_iter > 0:
        # compute conv
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))
        # compute contributions
        contrib = np.zeros_like(best)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        contrib[i_vals] = best[i_vals] * best[j_vals]
        # select source and target
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if best[src] <= 0:
            # find next non-zero
            nonzero = np.where(best > 0)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero))
        delta = best[src] * rng.random() * max_move_frac
        if delta <= 0:
            continue
        best[src] -= delta
        best[tgt] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta
        max_iter -= 1
    return best, best_score
```

"""
Multi‑restart L‑BFGS with a finer λ‑schedule, enriched initialisations
(including Beta‑shaped seeds), a targeted hill‑climb that moves mass
away from the dominant autocorrelation peak, and a final sub‑gradient
descent on the exact max‑operator.  All steps respect the non‑negative
simplex constraint and are timed to stay within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming Σa = 1.
    This matches ``evaluate_sequence`` up to the scaling invariant.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max_{k}(a*a)_k and its gradient w.r.t.
    the log‑soft‑max parameters y (where a = softmax(y)).
    """
    a = softmax(y)

    # ---- a * a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation of L‑BFGS runs with a decreasing list of smoothing factors.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple hill‑climb (random moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Greedy random mass transfers on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (focus on the current convolution peak)
# ----------------------------------------------------------------------


def _contrib_to_k(a: np.ndarray, k: int) -> np.ndarray:
    """Return vector c_i = a_i * a_{k-i} where the index is in range."""
    n = a.shape[0]
    contrib = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i_vals = np.arange(i_min, i_max + 1)
    j_vals = k - i_vals
    contrib[i_vals] = a[i_vals] * a[j_vals]
    return contrib


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy moves that explicitly reduce the dominant convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        contrib = _contrib_to_k(best, k)
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))

        if best[src] <= 0.0:
            # pick a random non‑zero source if the true argmax vanished
            nz = np.where(best > 0)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz))

        delta = best[src] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[src] -= delta
        best[tgt] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[src] += delta
            best[tgt] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑operator (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    max_iters: int = 2000,
) -> np.ndarray:
    """
    Simple diminishing‑step sub‑gradient descent on the true max.
    Operates on the log‑softmax parameters y.
    """
    y_cur = y.copy()
    rng = np.random.default_rng()
    step0 = 1.0

    for t in range(max_iters):
        if time.time() >= deadline:
            break
        a = softmax(y_cur)

        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()
        mask = conv == max_val
        w = mask.astype(float) / mask.sum()  # weight over the arg‑max positions

        # gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        fw = np.fft.fft(w, fft_len_g)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]

        # chain rule through softmax
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        eta = step0 / np.sqrt(t + 1)
        y_cur = y_cur - eta * grad_y

        if (t + 1) % 500 == 0:
            # small jitter to escape flat regions
            y_cur += rng.normal(scale=0.005, size=n)

    return y_cur


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes
    ``evaluate_sequence``.  The routine respects the 1000 s time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # problem size – reuse a previously found construction if available
    # ------------------------------------------------------------------
    n = 5000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = max(n, int(arr.size))

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise best candidate (prefer the stored construction)
    # ------------------------------------------------------------------
    if prev is not None:
        best_a = np.asarray(prev, dtype=float)
        if best_a.shape[0] != n:
            if best_a.shape[0] < n:
                best_a = np.concatenate([best_a, np.zeros(n - best_a.shape[0])])
            else:
                best_a = best_a[:n]
        best_a = np.maximum(best_a, 0.0)
        total = best_a.sum()
        best_a = best_a / total if total > 1e-12 else np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # a finer λ‑schedule (down to ~5e‑5) and a moderate iteration budget
    lam_factors = np.geomspace(0.6, 5e-5, num=30)
    maxiter_stage = 300
    iteration = 0

    # ------------------------------------------------------------------
    # multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(
            ["random", "exp", "power", "edge", "beta", "best"]
        )

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "beta":
            # Beta‑distribution shape on the discrete grid
            alpha = rng.uniform(0.5, 5.0)
            beta = rng.uniform(0.5, 5.0)
            x = np.linspace(0, 1, n)
            a0 = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.02, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # ----- targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- generic hill‑climb (polish) -----
        if time.time() < deadline - 1.0:
            best_a, best_score = hill_climb(
                best_a,
                best_score,
                fft_len,
                n,
                deadline,
                max_iter=4000,
            )
            best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final deep refinement (L‑BFGS with the same fine λ‑schedule)
    # ------------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final‑LBFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sub‑gradient polishing on the exact max‑operator
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y_final = subgradient_descent(
            best_y,
            n,
            fft_len,
            fft_len_g,
            deadline,
            max_iters=2000,
        )
        best_a = softmax(y_final)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[subgrad] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
