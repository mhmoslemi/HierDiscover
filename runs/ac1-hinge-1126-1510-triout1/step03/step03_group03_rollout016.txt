"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a three‑stage strategy:

1. **Broad random search** – generate candidates from several analytic families
   (power‑law, exponential, two‑sided power‑law, arcsine, cosine window,
   Gaussian window and Beta‑distribution).  Keep the best (lowest fast_score).

2. **Smoothed max gradient descent** – start from the best candidate and
   minimise a smooth approximation of the max‑convolution:
       S_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
   The gradient of S_β w.r.t. a is
       g = 2·(w ⋆ a)     where   w_k ∝ exp(β·(a*a)[k])
   (⋆ is the linear cross‑correlation).  We perform an exponentiated‑gradient
   (multiplicative) update a_i ← a_i·exp(−η·g_i) and renormalise onto the
   probability simplex.  The temperature β is increased slowly, the step‑size
   η decays, and occasional jitter is added to avoid stagnation.

3. **Simple post‑processing** – try a few heuristic transforms (power‑law
   flattening, Gaussian smoothing) of the final vector and keep the best.
   The final vector is returned as a plain Python list.

All operations respect the 1000 s budget; the routine prints intermediate
scores so that progress can be monitored.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using FFT‑based convolution.  The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for the random initialisation phase
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Discrete version of the arcsine density:
    a_0 = 1, a_i = a_{i-1}·(2i-1)/(2i).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann window (non‑negative, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian window centred at the middle of the vector."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_distribution(n: int, p: float, q: float) -> np.ndarray:
    """
    Discretised Beta(p,q) density on [0,1] using mid‑point rule.
    p,q > 0.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x ** (p - 1.0)) * ((1.0 - x) ** (q - 1.0))
    return a


# ----------------------------------------------------------------------
# Core convolution helpers used in the smoothed‑max descent
# ----------------------------------------------------------------------


def self_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT of length L."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * a.shape[0] - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a)[i] = Σ_j w_{i+j}·a_j,
    returned for i = 0 … len(a)-1.
    """
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Fw * np.conj(Fa), n=L)
    return corr_full[: a.shape[0]]

    # FFT lengths (powers of two for speed)
    L_conv = next_pow2(2 * n)
    L_corr = next_pow2(3 * n)

    # Hyper‑parameters
    eta0 = 5.0
    tau = 5000.0               # controls decay of η
    beta = 0.5                 # initial temperature
    beta_max = 80.0
    beta_factor = 1.015        # geometric increase per 1000 iterations
    eval_interval = 2000
    jitter_interval = 8000
    jitter_scale = 0.01

    iteration = 0
    while time.time() < deadline - 2.0:
        iteration += 1

        # ----- (a*a) and the “soft‑max” weights w_k -----
        conv_full = self_convolution(a, L_conv)   # length 2n‑1
        max_conv = conv_full.max()
        # soft‑max on the convolution values (more focus as β grows)
        exp_arg = beta * (conv_full - max_conv)   # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max (g_i = 2·(w⋆a)_i) -----
        corr = cross_correlation(w, a, L_corr)    # length n, linear cross‑correlation
        g = 2.0 * corr

        # ----- multiplicative (exponentiated) update -----
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # ----- keep numbers strictly positive, re‑normalise onto simplex -----
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- jitter to escape possible plateaus -----
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum <= 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

        # ----- slowly raise the temperature -----
        if iteration % 1000 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # ----- occasional evaluation of the true (non‑smooth) objective -----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] iter {iteration}, new best score={cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Small post‑processing tweaks (flattening, Gaussian smoothing)
    # ------------------------------------------------------------------
    candidates = [best_seq]

    # ---- power‑law flattening (raise to a power < 1) ----
    for p in (0.90, 0.85, 0.80):
        cand = np.power(best_seq, p)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
            candidates.append(cand)

    # ---- Gaussian smoothing of the final vector ----
    for sigma_frac in (0.02, 0.03, 0.04):
        sigma = sigma_frac * n
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        kernel = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        kernel = kernel / kernel.sum()
        cand = np.convolve(best_seq, kernel, mode="same")
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum
            candidates.append(cand)

    # ---- pick the best among all post‑processed candidates ----
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[post‑proc] improved score to {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
