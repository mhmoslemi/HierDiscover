def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5

    # init sequence
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        init = generate_arcsine_like(8000)

    init = np.maximum(init,0.0)
    if init.sum()==0:
        init = np.full(init.shape,1.0/init.size)
    else:
        init = init/init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] n={best_seq.size}, score={best_score}")

    n = best_seq.size
    fft_len = next_pow2(2*n)

    # hyperparameters
    active_thresh = 5e-3  # 0.5%
    eps = 1e-12
    lr0 = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eval_interval = 200
    jitter_interval = 4000
    jitter_scale = 0.02

    # start from best_seq but also randomize each restart
    iteration = 0
    # keep log parameters
    log_a = np.log(best_seq + eps)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    while time.time() < deadline:
        iteration += 1

        # exponentiate and renorm
        a = np.exp(log_a)
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a/total

        # compute conv via FFT
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2*n-1]
        max_val = conv_full.max()
        # active indices
        active = np.where(conv_full >= max_val * (1.0 - active_thresh))[0]

        # gradient accumulation
        g = np.zeros_like(a)
        # iterate active indices
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # gradient w.r.t log a
        grad_log = g * a

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_a = log_a - step

        # occasional jitter
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a,0.0,None)
            a = a / a.sum()
            log_a = np.log(a + eps)

        # evaluate periodically
        if iteration % eval_interval == 0:
            cur_score = 2 * n * max_val  # sum=1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time()-start:.1f}s] iter {iteration}, new best {cur_score:.6f}")

        # time check
        if time.time() > deadline:
            break

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished. final_score={final_score}")
    return final_seq

def next_pow2(x):

    p = 1
    while p < x:

    Fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv)   # sum(seq)=1, so denominator is 1

        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
    except Exception:
        init = generate_arcsine_like(8000)

    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.full_like(init, 1.0 / init.size, dtype=np.float64)
    else:
        init = init / init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] n={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random family search (≈ 50 % of the time budget)
    # ------------------------------------------------------------------
    phase1_deadline = start + budget_s * 0.5
    length_choices = [4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "power_offset",
        "beta",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
    ]

    while time.time() < phase1_deadline:
        n = random.choice(length_choices)
        fam = random.choice(families)

        if fam == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law_offset(n, p, offset=0.0)

        elif fam == "power_offset":
            p = random.uniform(0.30, 0.80)
            off = random.uniform(0.0, 10.0)
            cand = generate_power_law_offset(n, p, offset=off)

        elif fam == "beta":
            alpha = random.uniform(0.2, 2.5)
            beta = random.uniform(0.2, 2.5)
            cand = generate_beta_shape(n, alpha, beta)

        elif fam == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif fam == "arcsine":
            cand = generate_arcsine_like(n)

        elif fam == "cosine":
            i = np.arange(n, dtype=np.float64)
            cand = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))   # Hann window

        elif fam == "gaussian":
            sigma = random.uniform(0.07, 0.30) * n
            mid = (n - 1) / 2.0
            i = np.arange(n, dtype=np.float64)
            cand = np.exp(-0.5 * ((i - mid) / sigma) ** 2)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total < 1e-12:
            continue
        cand = cand / total

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] random best ({fam}) n={n} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Continuous refinement (Adam + active‑set sub‑gradient)
    # ------------------------------------------------------------------
    n = best_seq.shape[0]
    fft_len = next_pow2(2 * n)

    # Adam hyper‑parameters
    eta0 = 0.20          # base learning rate for log‑variables
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Active‑set tolerance (relative to current peak)
    active_rel = 5e-3    # 0.5 %
    eval_interval = 200
    jitter_interval = 4000
    jitter_scale = 0.02  # 2 % of current coefficient magnitude

    # Initialise optimisation state from the best discrete candidate
    log_a = np.log(best_seq + 1e-12)    # work in log‑space to guarantee positivity
    m = np.zeros_like(log_a)           # first moment estimate
    v = np.zeros_like(log_a)           # second moment estimate

    it = 0
    last_imp = time.time()

    while time.time() < deadline:
        it += 1

        # ---- rebuild a and re‑normalise ----
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- self‑convolution via FFT ----
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        max_val = conv_full.max()
        active_idx = np.where(conv_full >= max_val * (1.0 - active_rel))[0]

        # ---- active‑set sub‑gradient ----
        g = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            jdx = k - idx
            g[idx] += 2.0 * a[jdx]

        # chain rule to obtain gradient w.r.t. log‑variables
        grad_log = g * a

        # ---- Adam update on log‑variables ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_a = log_a - step

        # ---- occasional jitter to escape plateaus ----
        if it % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            log_a = np.log(a + 1e-12)

        # ---- periodic evaluation & bookkeeping ----
        if it % eval_interval == 0:
            cur_score = 2.0 * n * max_val   # sum(a)=1 ⇒ denominator = 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_imp = time.time()
                print(
                    f"[{time.time() - start:.1f}s] iter {it}, new best {cur_score:.6f}"
                )

        # ---- simple restart if we have been idle for a while ----
        if it % 20000 == 0:
            if time.time() - last_imp > 30.0:
                # perturb around the current best and reset moments
                perturb = np.random.randn(n) * 0.01
                log_a = np.log(best_seq + 1e-12) + perturb
                m.fill(0.0)
                v.fill(0.0)
                print(
                    f"[{time.time() - start:.1f}s] restart after {time.time() - last_imp:.1f}s idle"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)           # type: ignore[name-defined]
    print(f"Finished. final_score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
```
This implementation diversifies the search space with several parametric families, then refines the best candidate using an Adam optimiser on the log‑variables together with an active‑set sub‑gradient of the convolution peak. The algorithm respects the 1000 s budget and returns a list of non‑negative heights ready for `evaluate_sequence`.
