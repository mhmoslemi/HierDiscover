def incremental_simulated_annealing(a_start, conv_start, n, deadline, rng):
    cur_a = a_start.copy()
    cur_conv = conv_start.copy()
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9997  # slower cooling
    iteration = 0
    reconv_interval = 5000

    L = len(cur_conv)

    while time.time() < deadline:
        i = rng.integers(0, n)
        ai = cur_a[i]
        if ai < 1e-12:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # propose delta
        alpha = rng.random() * 0.5   # up to 50% move
        delta = ai * alpha
        if delta <= 0:
            continue

        old_ai = cur_a[i]
        old_aj = cur_a[j]
        # store old conv values
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j

        conv2i_old = cur_conv[idx2i]
        conv2j_old = cur_conv[idx2j]
        convij_old = cur_conv[idxij]

        # compute new conv values
        new_conv2i = conv2i_old - 2*delta*old_ai + delta*delta
        new_conv2j = conv2j_old + 2*delta*old_aj + delta*delta
        new_convij = convij_old + 2*delta*(old_ai - old_aj) - 2*delta*delta

        # apply
        cur_conv[idx2i] = new_conv2i
        cur_conv[idx2j] = new_conv2j
        cur_conv[idxij] = new_convij

        cur_a[i] = old_ai - delta
        cur_a[j] = old_aj + delta

        new_max = cur_conv.max()
        new_score = 2.0 * n * new_max

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert changes
            cur_a[i] = old_ai
            cur_a[j] = old_aj
            cur_conv[idx2i] = conv2i_old
            cur_conv[idx2j] = conv2j_old
            cur_conv[idxij] = convij_old

        # temperature schedule
        T *= decay
        iteration += 1

        # periodic full recompute to avoid drift
        if iteration % reconv_interval == 0:
            cur_conv = full_conv(cur_a, next_pow2(2*n-1))
            cur_score = 2.0 * n * cur_conv.max()
            # optional update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            # optionally reheat if T too low
            if T < best_score * 1e-6:
                T = max(best_score * 0.02, 1e-8)

    return best_a, best_score

"""
Hybrid optimizer for the step‑function benchmark.

The evaluation metric E(a)=2·n·max(a*a)/(sum a)² is scale‑invariant,
so we work on the probability simplex (∑a_i=1, a_i≥0).

Algorithm
---------

0. **Initialisation** – use the previously discovered sequence (if any)
   or a uniform vector of length ~2000 and normalise it.

1. **L‑BFGS on a smoothed max** – minimise a log‑sum‑exp surrogate of the
   true max using several random starts and a continuation of the smoothing
   parameter λ.  The variables are the logits y that are mapped to the simplex
   by a soft‑max.  The best point found is kept.

2. **Projected sub‑gradient descent** – directly minimise the true max
   (the objective is 2·n·max(conv)) using a simple sub‑gradient step and an
   Euclidean projection onto the simplex.

3. **Fast incremental simulated annealing** – perform a huge number of
   “mass‑transfer” moves (move a fraction of the mass from one coordinate to
   another).  The convolution is updated *incrementally* (only three entries
   of a*a change) so that each trial costs O(1) plus a cheap max‑scan.
   A temperature schedule allows occasional uphill moves.

4. **Final greedy hill‑climb** – a short polishing phase that repeatedly
   transfers a small fraction of mass between two positions while keeping
   the sum fixed.

All phases respect the 1000 s budget.  The routine returns the best
non‑negative step sequence found (as a list of floats) ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    a = softmax(y)                     # map to simplex

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution of a_rev with w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated annealing on the true max
# ----------------------------------------------------------------------


def incremental_simulated_annealing(a_start: np.ndarray,
                                    conv_start: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    rng: np.random.Generator):
    """
    Very fast SA that updates the convolution a*a incrementally.
    Only three entries of the convolution change after a mass‑transfer move,
    so each trial is O(1) plus a cheap max‑scan.
    """
    cur_a = a_start.copy()
    cur_conv = conv_start.copy()
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.05, 1e-8)   # start a bit hotter than the basic SA
    decay = 0.9998                  # slower cooling
    it = 0
    reconv_interval = 5000          # re‑compute full convolution occasionally

    L = len(cur_conv)

    while time.time() < deadline:
        # pick a source index proportional to its mass (more likely to move)
        i = rng.choice(n, p=cur_a)
        ai = cur_a[i]
        if ai < 1e-14:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction of the source mass
        frac = rng.random() * 0.5               # up to 50 % of the source mass
        delta = ai * frac
        if delta <= 0.0:
            continue

        old_ai = cur_a[i]
        old_aj = cur_a[j]

        # indices in the convolution that will change
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        # store old convolution values for revert
        conv2i_old = cur_conv[idx2i]
        conv2j_old = cur_conv[idx2j]
        convij_old = cur_conv[idxij]

        # incremental update (see analysis for derivation)
        new_conv2i = conv2i_old - 2.0 * delta * old_ai + delta * delta
        new_conv2j = conv2j_old + 2.0 * delta * old_aj + delta * delta
        new_convij = convij_old + 2.0 * delta * (old_ai - old_aj) - 2.0 * delta * delta

        # apply tentative changes
        cur_conv[idx2i] = new_conv2i
        cur_conv[idx2j] = new_conv2j
        cur_conv[idxij] = new_convij
        cur_a[i] = old_ai - delta
        cur_a[j] = old_aj + delta

        new_max = cur_conv.max()
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] = old_ai
            cur_a[j] = old_aj
            cur_conv[idx2i] = conv2i_old
            cur_conv[idx2j] = conv2j_old
            cur_conv[idxij] = convij_old

        # temperature schedule
        T *= decay
        it += 1

        # occasional full recomputation to avoid drift
        if it % reconv_interval == 0:
            cur_conv = full_conv(cur_a, next_pow2(2 * n - 1))
            cur_score = 2.0 * n * cur_conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
            # re‑heat a little if temperature got too small
            if T < best_score * 1e-6:
                T = max(best_score * 0.05, 1e-8)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found (as a list of floats).  Intermediate progress is printed.
    """
    # ------------------------------------------------------------------
    # Random generator and initial sequence
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            if init_seq.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT preparations (powers of two)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.20 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.40 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.85 * budget_s   # incremental SA
    # final hill‑climb uses whatever time is left

    # ------------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS (few restarts, continuation on λ)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    iter_l = 0
    while time.time() < phase1_end:
        lam = lam_schedule[iter_l % len(lam_schedule)]

        # random start mode
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # L‑BFGS on the smoothed surrogate
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {iter_l}] lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # brief hill‑climb to polish the L‑BFGS result
        hc_deadline = min(phase1_end, time.time() + 0.25)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=800)
        iter_l += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fast incremental simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # initialise convolution once for the SA routine
        conv0 = full_conv(best_a, fft_len_conv)
        best_a, best_score = incremental_simulated_annealing(best_a,
                                                             conv0,
                                                             n,
                                                             phase3_end,
                                                             rng)
        print(f"[incremental SA] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=3000)

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (for reporting)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
