def solve_max_sum_b(f, M):
    n = len(f)
    # A_ub shape (2n-1, n)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        # column i has coefficient a_i if 0 <= k-i < n -> i in [max(0, k-(n-1)), min(k,n-1)]
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # For i in lo..hi, index for b is k-i
        # But we are constructing row corresponding to conv f*b = sum_i f_i * b_{k-i}
        # Coef for b_j is a_{k-j} if 0 <= k-j < n else 0.
        # So we can fill using reversed f segment.
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
        # Actually b indices (i) from lo..hi, coefficient is f[k-i].
    # Now objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Upper bound constraints: A_ub @ b <= M
    b_ub = np.full(rows, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Hybrid LP‑driven improvement + simple gradient refinement.

The evaluation objective

    score(a) = 2·n·max(conv(a,a)) / (sum(a)²)

is scale‑invariant, therefore we keep all candidates on the probability
simplex (a_i ≥ 0, Σ a_i = 1).  For a fixed a the quantity

    M(a) = max(conv(a,a))

defines a convex feasible set for a second vector b:

        (a * b)[k] ≤ M(a)   for all k,
        b ≥ 0.

The linear program

        maximise   Σ b_i
        subject to (a * b) ≤ M(a),   b ≥ 0

produces a direction that is guaranteed to increase the total mass
while keeping the convolution bound unchanged.  After solving the LP we
mix the new b into a (convex combination) and repeat.  Each LP step
strictly reduces M(a) or leaves it unchanged, which drives the score
down.

To obtain a fine‑grained local optimum we occasionally run a tiny
projected sub‑gradient descent (the same routine used in the previous
state‑of‑the‑art) and a cheap stochastic “mass‑transfer” refinement.
Both stages are cheap compared to the LP and help escape shallow
plateaus.

The whole loop runs until the time budget (≈1000 s) expires and the best
sequence seen so far is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as the previous implementation)
# ----------------------------------------------------------------------
def subgradient_step(a: np.ndarray, step0: float, it: int) -> np.ndarray:
    """One sub‑gradient step with diminishing step size."""
    n = a.shape[0]
    conv = full_convolution(a)
    max_idx = int(np.argmax(conv))
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[max_idx - idx]
    step = step0 / np.sqrt(it + 1)
    return project_simplex(a - step * grad)


# ----------------------------------------------------------------------
# Light stochastic local refinement (mass transfer)
# ----------------------------------------------------------------------
def local_refine(a: np.ndarray, steps: int, deadline: float) -> np.ndarray:
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * np.max(full_convolution(a))
    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        score = 2.0 * n * np.max(full_convolution(cand))
        if score < best_score or np.random.rand() < np.exp((best_score - score) / temperature):
            a = cand
            if score < best_score:
                best_score = score
                best = cand.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_i   s.t.   (a * b) ≤ M,   b ≥ 0
# ----------------------------------------------------------------------
def solve_max_mass_lp(a: np.ndarray, M: float, deadline: float):
    """
    Returns the optimal b (as a 1‑D ndarray) or None on failure.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz matrix that maps b → a * b  (full convolution)
    # First column: a[0], a[1], …, a[n‑1], 0, …, 0   (length rows)
    # First row : a[0], 0, …, 0                     (length n)
    c = np.concatenate((a, np.zeros(n - 1)))
    r = np.concatenate((a[:1], np.zeros(n - 1)))
    A = toeplitz(c, r)          # shape (rows, n)

    # Linear programme: maximise 1ᵀ b  ⇔ minimise –1ᵀ b
    c_obj = -np.ones(n, dtype=np.float64)
    ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS solver (available via SciPy)
    # Give the solver a tiny time budget to avoid overruns.
    remaining = deadline - time.time()
    options = {"presolve": True}
    if remaining > 0:
        options["time_limit"] = remaining

    res = linprog(c_obj, A_ub=A, b_ub=ub, bounds=bounds,
                  method="highs", options=options)

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    `evaluate_sequence` score.

    The algorithm mixes three ideas:
      1. LP‑driven “mass‑inflation” that strictly reduces the maximal
         self‑convolution bound.
      2. A few projected sub‑gradient steps for smooth descent.
      3. A cheap stochastic refinement that swaps tiny amounts of mass
         between coordinates.

    The best sequence observed before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives the same asymptotic constant but a richer search
    # space; n≈2000 is a good compromise for the LP step (≈8 M entries).
    # ------------------------------------------------------------------
    try:
        n0 = len(height_sequence_1)            # noqa: F821
        n = max(1500, min(2500, n0))
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initial candidate (uniform is a safe start).
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Global bookkeeping
    best_seq = a.copy()
    best_score = float("inf")

    # Parameters for the different phases
    subgrad_step0 = 0.7
    subgrad_iters = 300          # few iterations after each LP
    refine_steps_per_round = 2000
    mixing_factor = 0.25         # convex combination weight for (a,b) update
    mixing_decay = 0.995         # slowly reduce mixing factor

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Evaluate current sequence
        # --------------------------------------------------------------
        conv = full_convolution(a)
        M = float(conv.max())
        score = 2.0 * n * M          # because Σa = 1

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[iter {iteration}] new global best = {score:.6f}")

        # --------------------------------------------------------------
        # 2) Linear programming step
        # --------------------------------------------------------------
        b = solve_max_mass_lp(a, M, deadline)
        if b is None:
            # LP failed – fall back to a pure random perturbation
            a = project_simplex(a + 0.02 * np.random.randn(n))
            continue

        # Normalise b to the simplex (the LP maximises the *sum*, but we
        # later blend b with a, so we keep the shape only).
        b = np.maximum(b, 0.0)
        if b.sum() == 0.0:
            # degenerate case – restart from a random point
            a = project_simplex(np.random.rand(n))
            continue
        b /= b.sum()

        # Convex combination  a ← (1‑t)·a + t·b
        a = (1.0 - mixing_factor) * a + mixing_factor * b
        a = project_simplex(a)   # re‑project to eliminate round‑off

        # Reduce the mixing factor over time (fine‑tuning phase)
        mixing_factor = max(0.02, mixing_factor * mixing_decay)

        # --------------------------------------------------------------
        # 3) Tiny sub‑gradient descent run (helps to follow the smooth
        #    direction suggested by the LP).
        # --------------------------------------------------------------
        for it in range(subgrad_iters):
            if time.time() >= deadline:
                break
            a = subgradient_step(a, subgrad_step0, it)
        # --------------------------------------------------------------
        # 4) Light stochastic refinement
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            a = local_refine(a, refine_steps_per_round, deadline)

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list of floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
