def lp_step(f, M, t):

    n = len(f)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    bounds = [(0, None)]*n

        # short hill‑climb after each L‑BFGS

        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,

"""
Improved search for a low‑value step sequence.

The metric is
    E(a)=2·n·max(a*a)/(sum a)²     (scale‑invariant)
so we restrict the search to the probability simplex (sum = 1).

Algorithm
---------

1. **Initialisation** – Load the previous best sequence if it exists,
   otherwise start from a uniform distribution (≈2000 entries).  Several
   random Dirichlet starts are tried and the best is kept.

2. **LP‑based refinement** – For the current candidate f compute M=max(f*f).
   Build the convolution matrix A such that (A⋅b)[k]=∑_i f_i b_{k‑i}.
   Solve the linear programme
        maximise   1ᵀb
        s.t.       A b ≤ M ,   b ≥ 0 .
   The LP tries to increase the total mass while respecting the same
   maximum‑convolution bound.  After solving we rescale b to unit‑sum
   and move a fraction t towards it (convex combination).  Repeating this
   step drives the maximal convolution down.

3. **Smoothed max optimisation (L‑BFGS)** – Minimise a log‑sum‑exp
   approximation of the max using L‑BFGS with a continuation schedule
   for the smoothing factor λ (called *lam_factor* below).  Several
   randomised starting patterns are used.

4. **Projected sub‑gradient optimisation** – Directly minimise the true
   max‑convolution using a diminishing‑step sub‑gradient method with
   Euclidean projection onto the simplex.

5. **Simulated annealing** – Non‑local stochastic search that swaps
   mass between positions, accepting uphill moves with a temperature that
   cools geometrically.

6. **Final greedy hill‑climb** – Fast local polishing that transfers a
   small fraction of mass between two coordinates while keeping the sum
   fixed.

All phases respect the overall 1000 s time budget.  The best sequence
found is stored globally as ``height_sequence_1`` for reuse in later
runs and is also returned as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a distribution already summing to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1} (Wang‑Carreira‑Perpiñán)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp and its gradient w.r.t. y.
    Returns (f, grad_y) where f = 2·n·smooth_max(conv).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Directly minimise 2·n·max(a*a) using a sub‑gradient method."""
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Greedy mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """Simple simulated annealing."""
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “alternating LP” step)
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Matrix A such that (A @ b)[k] = Σ_i f[i] * b[k‑i],
    i.e. the linear map b ↦ f * b (full convolution).
    """
    n = f.shape[0]
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return scipy.sparse.csr_matrix((data, (rows, cols)),
                                   shape=(2 * n - 1, n))


def lp_refine(a: np.ndarray,
              deadline: float,
              t0: float = 0.5,
              max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement.

    Starting from a (Σa=1), repeatedly:
        – compute M = max(a*a)
        – solve maximise Σb   s.t. (a*b) ≤ M , b≥0
        – scale b to unit sum and move a fraction t towards it
    The process strictly reduces the maximal convolution.
    """
    n = a.shape[0]
    best_a = a.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    t = t0

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = np.convolve(best_a, best_a)
        M = conv.max()

        A = build_conv_matrix(best_a)
        ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σb  → minimise –Σb
        bounds = [(0, None)] * n

        # Solve the LP (HiGHS interior‑point)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            break

        g = b_opt / sum_b          # rescale to unit‑sum
        new_a = (1.0 - t) * best_a + t * g
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()       # protect against numerical drift

        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
            print(f"[LP‑refine] iter={30 - max_iter} score={best_score:.6f}")

        # slowly shrink the mixing factor – this helps convergence
        t = max(t * 0.9, 0.05)

    return best_a, best_score
