Projection onto simplex:

def proj_simplex(v):
    # ensure sum(v)=1 and v>=0
    # v is a 1-D numpy array
    # algorithm O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v))+1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Now gradient w.r.t a for exact max:

```
def subgrad_max_conv(a, conv, max_val):
    # conv = full convolution a*a
    # max_val = conv.max()
    # w indices where conv == max_val
    w = np.zeros_like(conv)
    w[conv == max_val] = 1.0
    # compute conv of rev(a) with w using fft
    n = len(a)
    fft_len = next_pow2( len(w) + n - 1 )
    # but we can compute using np.fft
    fa_rev = np.fft.fft(a[::-1], fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.fft.ifft(fa_rev * fw).real
    g = 2 * conv_rev[n-1 : n-1 + n]  # shape (n,)
    return g
```

Note that w might have multiple max positions; this aggregates equal weight on each due to w=1 per index; gradient becomes sum of contributions from all maxima, which is a valid subgradient (since subgrad of max is convex hull of gradients of active functions). """
Optimisation of a non‑negative step sequence for the objective

    2·n·max(conv(a,a))   (with sum(a)=1)

The algorithm combines three ideas:

1. **Smooth L‑BFGS optimisation** – a soft‑max re‑parameterisation of
   the probabilities `a = softmax(y)` and a log‑sum‑exp (smooth‑max)
   surrogate for `max`.  A continuation schedule drives the smoothing
   factor to zero.

2. **Exact sub‑gradient descent on the true max** – after the smooth
   phase we use the exact (non‑smoothed) objective.  The sub‑gradient
   of `max(conv)` w.r.t. `a` is obtained by a single FFT convolution,
   and each step is projected back onto the probability simplex.

3. **Greedy hill‑climbing polishing** – a fast local move that swaps a
   small fraction of probability mass between two coordinates whenever
   the true fast score improves.

All heavy work is performed with FFTs, the code keeps track of the time
budget (default 1000 s) and returns the best sequence found before the
deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. the
    unconstrained vector y (pre‑softmax).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z   # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_schedule: list[float],
                      maxiter_per_stage: int = 300) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    Accept the move only if the true fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgrad_max_conv(a: np.ndarray,
                     mask: np.ndarray,
                     fft_len_g: int,
                     n: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) w.r.t. a.
    `mask` is a 0/1 vector of length 2·n‑1 indicating the active
    (max‑attaining) convolution entries.
    """
    # convolution of reversed a with mask
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(mask, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = 2.0 * conv_rev[n - 1: n - 1 + n]   # factor 2 for ordered pairs
    return g


def generate_initial_a(mode: str,
                       n: int,
                       rng: np.random.Generator) -> np.ndarray:
    """Generate a starting probability vector for a given heuristic mode."""
    if mode == "random":
        a0 = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a0 = np.full(n, 0.0)
        a0[:m_edge] = 1.0
        a0[-m_edge:] = 1.0
    elif mode == "sidon":
        # greedy Sidon set (pairwise sums distinct)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums:
                    conflict = True
                    break
            if conflict or (cand * 2) in sums:
                continue
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
            # stop when we have roughly sqrt(2n) elements
            if len(chosen) >= int(np.sqrt(2 * n)):
                break
        a0 = np.zeros(n)
        if chosen:
            a0[chosen] = 1.0 / len(chosen)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        raise ValueError(f"unknown mode '{mode}'")
    # Normalise – protect against all‑zero vectors
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()
    return a0

    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------

    n = 5000
    try:

    # FFT lengths – powers of two for speed

    # --------------------------------------------------------------

    best_a = np.full(n, 1.0 / n)         # sum = 1
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)      # soft‑max representation
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # Continuation schedule – geometric decay of the smoothing factor
    lam_schedule = list(np.geomspace(0.5, 0.001, num=12))

    while time.time() < deadline:
