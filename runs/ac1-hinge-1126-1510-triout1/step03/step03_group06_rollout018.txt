def smooth_max_and_grad(a, alpha):
    conv = np.convolve(a,a)
    # compute softmax weights
    scaled = conv * alpha
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)  # for numerical stability

def adam_optimize(a0, max_iters, deadline, alpha_start, alpha_end, step0):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2=0.999; eps=1e-8
    best_seq = a.copy()
    best_score = float('inf')
    # schedule for alpha: e.g., use exponential or linear increase.
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1)
        smooth_max, grad, conv = smooth_max_and_grad(a, alpha)
        # compute true score
        max_b = conv.max()
        score = 2*len(a)*max_b
        if score < best_score:
            best_score = score; best_seq=a.copy()
            print(f"[Adam iter {it}] new best {score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps) * m_hat
        a = project_simplex(a - step)
    return best_seq, best_score

def fft_convolve(a,b):
    n = len(a) + len(b) - 1
    fft_len = 1 << ((n-1).bit_length())
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    res = np.fft.irfft(A*B, fft_len)
    return res[:n]

def smooth_max_and_gradient(a, alpha):
    # a is 1D np.ndarray sum=1.
    conv = fft_convolve(a, a)  # full conv length m = 2n-1
    # compute stable softmax
    scaled = conv * alpha
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    weight = exp_scaled / np.sum(exp_scaled)
    # gradient via convolution of weight with a reversed
    a_rev = a[::-1]
    grad_full = fft_convolve(weight, a_rev)  # length m+n-1
    n = len(a)
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return weight, grad, conv

def adam_optimize(a0, max_iters, step0, alpha_start, alpha_end, deadline):
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()
    # optional: ensure sum=1.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_score = float("inf")
    # not compute alpha schedule each iter p
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it-1) / (max_iters-1) if max_iters>1 else alpha_start
        # compute weight, grad, conv
        weight, grad, conv = smooth_max_and_gradient(a, alpha)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[Adam n={n} it={it}] new best={score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a - step)
    return best_seq, best_score

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
* Choose a few candidate lengths (based on any existing solution and a
  default of 4000–8000).
* For each length run a cheap 1‑D parametric sweep over power‑law exponents
  `a_i ∝ (i+1)^{-p}` and keep the best shape.
* Starting from that shape (or from a uniform / perturbed best sequence)
  run a projected Adam optimiser on the probability simplex.
  The optimiser minimises a smooth approximation of the maximal
  self‑convolution:
      smooth_max_α(a*a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
  The gradient of this smooth max can be written as a convolution and is
  obtained efficiently by FFTs.
* After the Adam run, a lightweight stochastic local refinement moves a
  small amount of mass between random coordinates and accepts improving
  moves (with occasional uphill moves).  This helps escape shallow minima.
* All candidates are kept on the simplex (Σa_i = 1, a_i ≥ 0) using the
  Duchi‑et‑al. projection.
* The best sequence seen before the time budget expires (default 1000 s)
  is returned as a plain Python list of floats.

The combination of a parametric warm‑start, a smooth‑max Adam descent and
a final random refinement gives better scores than the previous pure
sub‑gradient method and typically reaches the target bound 1.503 within
the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real inputs)
# ----------------------------------------------------------------------
def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, fft_len)
    return conv[:n]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    conv = fft_convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Smooth‑max of the convolution and its gradient (for Adam)
# ----------------------------------------------------------------------
def smooth_max_and_gradient(a: np.ndarray, alpha: float):
    """
    Return (softmax weights, gradient of smooth max, full convolution).

    The smooth max of the convolution is
        (1/α)·log Σ_k exp(α·(a*a)_k) .
    Its gradient w.r.t. a is
        grad_i = 2·Σ_k w_k·a_{k-i},
    where w_k are the softmax weights.
    """
    conv = fft_convolve(a, a)                # length 2n‑1
    scaled = conv * alpha
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)  # stability
    w = exp_scaled / np.sum(exp_scaled)       # softmax over k

    # gradient via convolution with the reversed vector
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)        # length (2n‑1)+n‑1
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return w, grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth‑max surrogate)
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    alpha_start: float,
    alpha_end: float,
    deadline: float,
):
    """
    Projected Adam minimising the smooth‑max surrogate of max(a*a).

    Returns the best sequence and its true (non‑smooth) score.
    """
    a = a0.copy()
    a = a / a.sum()                     # enforce Σa = 1
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the smoothing parameter α
        if max_iters > 1:
            alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1)
        else:
            alpha = alpha_start

        _, grad, conv = smooth_max_and_gradient(a, alpha)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[Adam n={n} it={it}] new best={score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a - step)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic refinement (similar to the earlier code)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float):
    """
    Random walk moving a tiny amount of mass between two coordinates.
    Accepts improvements and, with a small probability, uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a fraction of the source mass to the destination
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
        delta = np.random.uniform(0.0, 0.07) * a[src]   # up to 7 % of src
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: generate a decreasing power‑law vector and normalise it
# ----------------------------------------------------------------------
def power_law_vector(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a
