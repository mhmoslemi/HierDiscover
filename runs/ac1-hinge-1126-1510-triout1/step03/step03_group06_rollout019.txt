"""
Improved search for a low‑evaluation step‑function sequence.

The objective 2·n·max(conv(a,a)) / (sum(a)**2) is scale invariant, so the
search is performed on the probability simplex (a_i ≥ 0, Σa_i = 1).
A smooth approximation of the max‑operator (soft‑max) provides a
differentiable surrogate.  At each iteration we

  1. compute the full self‑convolution a*a,
  2. build soft‑max weights w_k ∝ exp(γ·(conv_k−max)),
  3. obtain the gradient of the smoothed objective:
        ∇ = (w * a)[0:n] + a                         (convolution)
  4. take a projected (Euclidean) gradient step with a
     diminishing stepsize η_t = η₀/√t,
  5. re‑project onto the simplex.

The procedure is run for several lengths (≈2k‒12k) and many random
re‑starts.  After the smooth‑gradient phase a cheap stochastic local
refinement (random mass moves with a Metropolis acceptance rule) is
applied.  The best sequence found before the time budget expires is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution using FFT (real‑valued inputs)
# ----------------------------------------------------------------------
def full_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution of two real vectors (full length).
    Uses a power‑of‑two FFT length for efficiency.
    """
    n = u.shape[0] + v.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())
    U = np.fft.rfft(u, n=fft_len)
    V = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(U * V, n=fft_len)[:n]
    return conv


# ----------------------------------------------------------------------
# Convolution helpers used by the original code (kept for compatibility)
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return a*a (full) and its maximum entry."""
    conv = full_convolve(a, a)
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper returning only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex (soft‑max approximation)
# ----------------------------------------------------------------------
def smoothed_gradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    gamma: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex using a soft‑max surrogate
    for the convolution peak.

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (will be re‑normalised to sum‑one).
    max_iters : int
        Upper bound on the number of iterations (the loop also checks time).
    step0 : float
        Base step size (scaled by 1/√iteration).
    gamma : float
        Soft‑max temperature (larger → closer to true max).
    deadline : float
        Unix timestamp after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        Sequence (probability vector) with the lowest observed score.
    best_score : float
        Corresponding evaluation score.
    """
    a = a0.copy()
    a = a / a.sum()                       # enforce Σa = 1
    n = a.shape[0]

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # -------- self‑convolution and peak ----------
        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b

        # keep track of the best vector seen so far
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[sgd n={n} it={it}] new best {score:.6f}")

        # -------- soft‑max weights ----------
        # values are ≤ 0; the largest entry gets weight 1
        w = np.exp(gamma * (conv - max_b))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback – uniform weights (should never happen)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # -------- gradient of the smoothed max ----------
        # grad = (w * a)[:n] + a   (see derivation in the docstring)
        grad = full_convolve(w, a)[:n] + a

        # -------- gradient step on the simplex ----------
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (same as the previous version)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """Random walk that moves a tiny amount of mass between coordinates."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score
