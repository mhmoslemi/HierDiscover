    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))

"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Broad random exploration** – a few thousand candidates are drawn from
   several families (power‑law, exponential, two‑sided power‑law, arcsine,
   Hann, Gaussian) with a range of lengths.  The best candidate (lowest fast
   score) is retained.

2. **Soft‑max weighted exponentiated‑gradient descent** – starting from the
   best candidate we iteratively reduce the peak of the self‑convolution.
   At each iteration we compute the full convolution, build a soft‑max
   distribution over its entries (parameter ``alpha``) and take the gradient
   of the smooth objective.  A momentum term (exponential moving average)
   is used, the update is multiplicative (mirror‑descent on the simplex)
   and the step‑size decays slowly.  Occasionally a tiny random jitter is
   added to avoid stagnation.  The whole search respects a 1000 s budget
   and returns the best sequence found.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast (FFT‑based) evaluation of
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    seq is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generators for the random‑exploration phase
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_hann_window(n: int) -> np.ndarray:
    """Standard Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian bell centred in the middle; sigma = sigma_frac·n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # keep a safety margin

    # --------------------------------------------------------------
    # 0️⃣  Load an initial seed (previous best if available)
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
    except Exception:
        # fallback: an arcsine‑like sequence of moderate length
        init = generate_arcsine_like(8000)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s > 0:
        init = init / s
    else:
        init = np.full_like(init, 1.0 / init.size, dtype=np.float64)

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Broad random exploration (≈ 200 s)
    # --------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000, 14000]
    families = ["power", "exp", "twosided", "arcsine", "cosine", "gaussian"]
    while time.time() < deadline - 200.0:

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()

    a = best_seq.copy()

    tau = 30000.0             # decay time‑scale for η
    beta = 0.90               # momentum coefficient
    m = np.zeros_like(a)      # momentum buffer

    alpha = 50.0              # start with a smooth objective
    alpha_max = 500.0
    alpha_inc = (alpha_max - alpha) / 80000.0   # approx linear increase

    eval_interval = 2000      # how often we evaluate the fast score
    jitter_interval = 5000    # frequency of jitter injections
    jitter_scale = 0.005      # relative amplitude of jitter

    while time.time() < deadline - 1.0:

        # ----- full self‑convolution (FFT) -----
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries -----
        max_c = conv_full.max()
        # increase sharpness gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # stable soft‑max
        exps = np.exp(alpha * (conv_full - max_c))
        w = exps / exps.sum()

        # keep only the K largest weights (they dominate the soft‑max)
        if w.size > K:
            idx = np.argpartition(w, -K)[-K:]
        else:
            idx = np.arange(w.size)

        # ----- gradient of the soft‑max objective -----
        g = np.zeros_like(a)
        for k in idx:
            wk = w[k]

            # part where i <= k  (contribution a[k‑i])
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_start <= i_end:

        m = beta * m + (1.0 - beta) * g

        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * m)

        # ----- projection back onto the simplex -----
        a_sum = a.sum()
        if a_sum <= 0 or not np.isfinite(a_sum):

            a = np.clip(a, 0.0, None)
            a_sum = a.sum()

    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
