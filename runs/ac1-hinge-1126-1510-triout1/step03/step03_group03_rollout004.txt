   a variety of parametric families (power‑law, exponential, two‑sided,
   arcsine, cosine‐window, Gaussian, Beta).  The best K candidates (by a fast

   is improved with a mirror‑descent style update
        a_i ← a_i·exp(−η·g_i)

   Instead of a single peak we use the *active set* of all indices whose

3. **Final verification** – the best sequence found is returned as a plain
   Python list and evaluated with the provided `evaluate_sequence`.

The search respects the total time budget (default 1000 s) and prints

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Very fast approximation of the true objective.
    It computes   2·n·max(conv(seq,seq)) / (sum(seq)**2)
    with an FFT‑based convolution.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Discrete coefficients of (1‑z)^{-1/2}.  a_i ∼ 1/√i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann window (cosine‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred window, sigma as a fraction of n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on (0,1).
    α,β > 0.  Values near the boundaries are regularised with a tiny epsilon.
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a

                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
                j_range = k - i_range
                # contribute 2*a_j (diagonal term counted only once is negligible)
                g[i_range] += 2.0 * a[j_range]

            # ----- Adam‑style moments -----
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # ----- exponentiated‑gradient step -----
            step = eta_base * m_hat / (np.sqrt(v_hat) + eps)
            a = a * np.exp(-step)

            # ----- projection onto the simplex (simple renormalisation) -----
            sum_a = a.sum()
            if sum_a <= 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / sum_a

            # ----- occasional jitter to avoid stagnation -----
            if t_iter % jitter_interval == 0:
                noise = np.random.randn(n) * jitter_scale * a
                a = a + noise
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

            # ----- periodic evaluation -----
            if t_iter % eval_interval == 0:
                cur_score = fast_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start:.1f}s] rank {rank} iter {t_iter} "
                          f"new global best fast_score={cur_score:.6f}")

                # also keep the locally best sequence for this candidate
                if cur_score < init_score:
                    candidate_pool[rank] = (cur_score, a.copy())

            # Very cheap early‑stop if the active set collapses to a single index
            # and the gradient becomes almost zero.
            if np.linalg.norm(g) < 1e-8:
                break

    # --------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
