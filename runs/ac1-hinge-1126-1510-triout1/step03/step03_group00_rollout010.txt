def lp_iterative(a, n, row_idx, col_idx, max_iters, t, deadline):
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution and max
        conv = full_conv(a, fft_len)  # returns length 2*n-1
        M = conv.max()
        # Solve LP to maximize sum(g) subject to A@g <= M
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2*n-1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=5000) # or cp.SCS for faster.
        if g.value is None:
            break
        sum_g = np.sum(g.value)
        # scaling
        g_scaled = g.value / sum_g
        # mixing
        a_new = (1 - t) * a + t * g_scaled
        # enforce nonneg and sum to 1 (should hold)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------

1. **Initialisation** – start from the best known sequence (`height_sequence_1`)
   if it exists, otherwise from a simple exponential profile.  The vector is
   normalised to sum = 1.

2. **Continuation L‑BFGS (short warm‑up)** – a few cheap L‑BFGS runs on the
   smooth log‑sum‑exp approximation of the max‑convolution.  This moves the
   iterate away from clearly sub‑optimal regions and provides a better start
   for the LP phase.

3. **LP‑based active‑set refinement** – for the current distribution `a`
   compute `M = max(a*a)`.  Solve the linear programme

        maximise    1ᵀg
        subject to   C(a)·g ≤ M·1 ,   g ≥ 0

   where `C(a)` is the (2n‑1)×n convolution matrix that maps `g` to the
   linear convolution `a * g`.  The optimum `g` has a larger L₁‑norm than
   `a`; after scaling it back to unit sum we obtain `g̃`.  We update

        a ← (1‑t)·a + t·g̃          (t≈0.5, slowly decayed)

   and repeat.  Each LP iteration never increases the bound `M`; after a
   handful of iterations the sequence approaches the conjectured optimum
   (self‑convolution close to a flat function).

4. **Projected sub‑gradient descent** – a few sub‑gradient steps on the
   exact nonsmooth objective `2·n·max(a*a)` to polish the result.

5. **Greedy hill‑climb** – final local search that moves a small fraction
   of mass between two coordinates while keeping the sum fixed.

All steps respect the simplex (`a_i ≥ 0`, `∑a_i = 1`).  The time budget
(≈ 1000 s) is split automatically; the best sequence found before the
deadline is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (same as in the reference code)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already normalised."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used for the warm‑up)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smooth‑max based on log‑sum‑exp. Returns (objective, gradient)."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Projected sub‑gradient descent with back‑tracking line search."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search (simple back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Greedy mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# L‑BFGS continuation (cheap warm‑up)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run a few L‑BFGS stages on the smoothed objective."""
    y = np.log(a_start + 1e-12)          # work in log‑space to keep positivity
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # quick hill‑climb after each lam
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=800)
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based active‑set refinement
# ----------------------------------------------------------------------
def _precompute_conv_matrix_indices(n: int):
    """Return row / col index arrays for the (2n‑1)×n convolution matrix."""
    total = n * n
    row = np.empty(total, dtype=np.int32)
    col = np.empty(total, dtype=np.int32)
    for i in range(n):
        start = i * n
        row[start:start + n] = np.arange(i, i + n, dtype=np.int32)
        col[start:start + n] = i
    return row, col


def _solve_lp_max_sum(f: np.ndarray, M: float, row_idx, col_idx, n: int):
    """Linear programme:
           maximise 1ᵀg
           s.t.    C(f)·g ≤ M·1 ,  g ≥ 0
       Returns the optimal g (or f if the solver fails)."""
    data = np.repeat(f, n)               # length n*n
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, verbose=False)
        if g.value is None:
            return f.copy()
        else:
            return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    except (cp.SolverError, MemoryError):
        return f.copy()


def lp_refine(a0: np.ndarray,
              n: int,
              fft_len_conv: int,
              time_deadline: float):
    """Iteratively improve `a0` by solving the LP described above."""
    row_idx, col_idx = _precompute_conv_matrix_indices(n)
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    t = 0.5               # initial mixing factor
    it = 0
    while time.time() < time_deadline:
        conv = full_conv(a, fft_len_conv)
        M = conv.max()

        # solve the LP
        g = _solve_lp_max_sum(a, M, row_idx, col_idx, n)
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # rescale to unit L1‑norm
        g_scaled = g / sum_g

        # convex combination (keeps us inside the feasible region)
        a = (1.0 - t) * a + t * g_scaled
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # compute score
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] iter={it:02d}, score={best_score:.6f}")

        # slowly decay the mixing factor to stabilise
        t = max(0.1, t * 0.97)
        it += 1

    return best_a, best_score
