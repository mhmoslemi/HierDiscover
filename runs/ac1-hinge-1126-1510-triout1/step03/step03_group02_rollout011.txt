def simulated_annealing(a, cur_score, fft_len, n, deadline, max_iter=20000, initial_T=0.01, cooling=0.9995):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    T = initial_T
    rng = np.random.default_rng()
    i_range = np.arange(n)
    while time.time() < deadline and max_iter>0:
        max_iter-=1
        # propose move
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue
        # random delta up to fraction of src mass
        max_move = cur[src] * 0.2
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cur[src] -= delta
        cur[dst] += delta
        # ensure non-negativity
        if cur[src] < 0:
            # adjust
            cur[dst] += cur[src] # subtract negative
            cur[src] = 0.0
        # renorm? sum stays same
        # compute new score
        new_score = fast_score(cur, fft_len, n)
        Δ = new_score - cur_score
        if Δ < 0 or np.exp(-Δ / max(T,1e-12)) > rng.random():
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[dst] -= delta
            cur[src] += delta
        T *= cooling
    return best, best_score

"""
Propose a low‑score step‑function using a blend of
  • smooth‑max L‑BFGS continuation (softmax‑parameterisation)
  • aggressive, max‑convolution‑aware hill‑climbing
  • a short simulated‑annealing phase for escaping shallow minima.

The code keeps everything on the probability simplex (∑aᵢ = 1, aᵢ ≥ 0).
Fast objective = 2·n·max(a∗a) is used for all internal optimisation;
the final list is evaluated with the supplied ``evaluate_sequence``.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a must have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector a
    (∑a = 1).  Returns 2·n·max(a∗a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution (a = softmax(y)) and its gradient.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run the optimiser over a decreasing sequence of
    smoothing factors λ = max_c·lam_factor.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb that is aware of the worst convolution shift
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,

    deadline: float,
    max_iter: int = 20000,

    """
    Greedy hill‑climb that moves mass away from the indices that
    contribute most to the current worst convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.20               # up to 20 % of source mass per move

    idx = np.arange(n)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current worst shift
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each index to that shift
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[idx[mask]] * best[partner[mask]]

        total = contrib.sum()
        if total == 0.0:
            src = rng.integers(n)
        else:
            prob_src = contrib / total
            src = rng.choice(n, p=prob_src)

        # target index – prefer indices with *small* contribution
        inv = 1.0 / (contrib + 1e-12)
        inv[src] = 0.0                     # never move mass back to src
        inv_sum = inv.sum()
        if inv_sum == 0.0:
            tgt = rng.integers(n)
        else:
            prob_tgt = inv / inv_sum
            tgt = rng.choice(n, p=prob_tgt)

        if src == tgt:
            continue

        delta = best[src] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        # apply move
        best[src] -= delta
        best[tgt] += delta

        # numerical safety
        if best[src] < 0.0:
            best[tgt] += best[src]
            best[src] = 0.0

        # re‑normalise (sum is theoretically unchanged but floating point drift exists)
        total = best.sum()
        if not np.isclose(total, 1.0):
            best /= total

        new_score = fast_score(best, fft_len, n)
        if new_score + 1e-12 < best_score:
            best_score = new_score               # keep the improvement
        else:
            # revert
            best[tgt] -= delta
            best[src] += delta
            # re‑normalise again
            total = best.sum()
            if not np.isclose(total, 1.0):
                best /= total

    return best, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing phase (helps escape shallow minima)
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    initial_T: float = 0.01,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Classic SA on the simplex.  Moves are simple mass transfers.
    """
    cur = a.copy()
    cur_score = cur_score
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = initial_T

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or cur[src] <= 0.0:
            continue

        # propose a transfer up to 20 % of the source mass
        delta = cur[src] * rng.random() * 0.20
        cur[src] -= delta
        cur[dst] += delta

        # safeguard non‑negativity
        if cur[src] < 0.0:
            cur[dst] += cur[src]
            cur[src] = 0.0

        # re‑normalise (tiny drift)
        total = cur.sum()
        if not np.isclose(total, 1.0):
            cur /= total

        new_score = fast_score(cur, fft_len, n)
        Δ = new_score - cur_score

        if Δ < 0.0 or np.exp(-Δ / max(T, 1e-12)) > rng.random():
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert the move
            cur[dst] -= delta
            cur[src] += delta
            # re‑normalise again
            total = cur.sum()
            if not np.isclose(total, 1.0):
                cur /= total

        T *= cooling

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low
    evaluate_sequence() score.  Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size – try to reuse any existing construction
    # --------------------------------------------------------------
    try:
        prev = globals().get("height_sequence_1")
        if prev is not None:
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 2000:
                n = int(arr.size)
            else:
                n = 5000
        else:
            n = 5000
    except Exception:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise best sequence (prefer a stored construction)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_arr = np.asarray(globals()["height_sequence_1"], dtype=float)
        init_arr = np.maximum(init_arr, 0.0)
        if init_arr.sum() > 0:
            best_a = init_arr / init_arr.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 0.001, num=30)   # smoother to sharper
    maxiter_stage = 400                               # per λ stage

    iteration = 0

    # --------------------------------------------------------------
    # multi‑restart L‑BFGS loop
    # --------------------------------------------------------------
    while time.time() < deadline - 15.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturbed current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb (quick) -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final deep L‑BFGS refinement (a few extra passes)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] λ={lam:.5g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # simulated annealing + a last targeted climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        # allocate most of the remaining budget to SA
        sa_deadline = min(deadline, time.time() + 30.0)
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            sa_deadline,
            max_iter=30000,
            initial_T=0.02,
            cooling=0.9994,
        )
        best_y = np.log(best_a + 1e-12)

    if time.time() < deadline:
        # final polishing with the convolution‑aware climb
        final_deadline = deadline
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=15000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
