def compute_conv(seq):
    n = len(seq)
    fft_len = next_pow2(2*n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    return conv

def compute_gradient(seq, beta):
    n = len(seq)
    conv = compute_conv(seq)
    # softmax weights
    maxc = conv.max()
    # shift for numerical stability
    exp_weights = np.exp(beta * (conv - maxc))
    sum_weights = exp_weights.sum()
    w = exp_weights / sum_weights
    # compute gradient via convolution with w_rev
    w_rev = w[::-1]  # length 2n-1
    fft_len = next_pow2(n + (2*n-1) - 1) = next_pow2(3*n - 2)
    # Convolve a and w_rev
    fa = np.fft.rfft(seq, n=fft_len)

"""
Propose a step‑function (non‑negative sequence) with a low value of

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the discrete convolution.
The algorithm consists of three phases:

1. **Broad random sampling** – generate many candidates from several
   parametric families (power‑law, exponential, two‑sided power‑law,
   Beta‑shape and a sparse random support).  All candidates are
   normalised to sum = 1 and the best few are kept.

2. **Smooth‑max Adam refinement** – for each elite candidate we minimise
   a smooth approximation of the `max` (a log‑sum‑exp) using the Adam
   optimiser on the probability simplex.  The gradient of the
   smoothed objective is obtained efficiently by a pair of FFT‑based
   convolutions.  The smoothing parameter is slowly increased so that
   the optimiser first moves in a well‑behaved landscape and finally
   focuses on the true peak.

3. **Final verification** – the best sequence found is returned as a
   plain Python list and its exact score is printed using the supplied
   `evaluate_sequence` function.

The whole search respects the 1000 s budget; if the time expires the
current best sequence is returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)²)
    using an FFT‑based convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (families used in the random stage)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(−p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(−lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float,
    w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(−p_left) + w_right·(n−i)^(−p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha−1)·(n−i)^(beta−1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sparse_random(n: int, nz: int) -> np.ndarray:
    """
    Randomly choose `nz` positions (nz ≤ n) and assign i.i.d.
    exponential values, then normalise.
    """
    nz = min(nz, n)
    seq = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=nz, replace=False)
    seq[idx] = np.random.exponential(scale=1.0, size=nz)
    return seq


# ----------------------------------------------------------------------
# Convolution / gradient utilities
# ----------------------------------------------------------------------


def conv_self(seq: np.ndarray) -> np.ndarray:
    """Self‑convolution of `seq` (length 2·n−1) via FFT."""
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def gradient_smooth_max(seq: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth‑max objective
        Lβ(seq) = (1/β)·log Σ_k exp(β·conv_k)
    Returns (grad, current_max_conv).  The gradient is
        ∂Lβ/∂a_i = 2· Σ_j w_{i+j}·a_j,
    where w_k = exp(β·conv_k)/Σ_l exp(β·conv_l).
    The computation uses two FFT‑based convolutions.
    """
    n = seq.shape[0]

    # 1) self‑convolution
    conv = conv_self(seq)
    max_conv = float(conv.max())

    # 2) soft‑max weights (shifted by max for numerical stability)
    exp_w = np.exp(beta * (conv - max_conv))
    w = exp_w / exp_w.sum()                # length = 2·n−1

    # 3) convolution a * w_rev  (w_rev is w reversed)
    w_rev = w[::-1]                        # also length 2·n−1
    fft_len = next_pow2(3 * n)            # enough for n + (2n−1) − 1
    fa = np.fft.rfft(seq, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    conv2 = np.fft.irfft(fa * fw, n=fft_len)   # length fft_len

    # extract the slice [n‑1 … 2·n‑2] and reverse it
    slice_mid = conv2[n - 1: 2 * n - 1][::-1]
    grad = 2.0 * slice_mid

    return grad, max_conv


# ----------------------------------------------------------------------
# Adam‑style refinement with smooth‑max
# ----------------------------------------------------------------------


def refine_sequence_adam(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smooth‑max objective.
    Returns the best sequence encountered and its *exact* fast_score.
    """
    a = project_simplex(seq.astype(np.float64).copy())
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_score = fast_score(a)

    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1
        # linearly increase the smoothness parameter β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 1500.0)

        # gradient of the smooth‑max surrogate
        grad, _ = gradient_smooth_max(a, beta)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays slowly with the iteration count
        lr = lr0 / (np.sqrt(it) + 1.0)

        update = - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a + update
        a = project_simplex(a_candidate)

        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Occasional jitter to escape flat plateaus
        if no_improve >= 2000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact evaluation of the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score

    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = ["power", "exp", "twosided", "beta", "sparse"]

            seq = generate_power_law(n, p)

        elif family == "twosided":
            p_l = random.uniform(0.10, 1.20)
            p_r = random.uniform(0.10, 1.20)

        elif family == "beta":

        s = seq.sum()
        if s < 1e-12:

        seq = seq / s

        sc = fast_score(seq)

        # keep global best

            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] **new global best** ({family}) n={n} score={sc:.6f}")

        # maintain an elite pool (size ≤ 8)

        else:
