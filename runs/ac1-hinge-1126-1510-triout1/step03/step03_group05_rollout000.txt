#!/usr/bin/env python
"""
Hybrid stochastic + gradient + simulated‑annealing search for a step‑function
that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

under a_i ≥ 0.  The algorithm consists of three phases:

1. Broad random search over several parametric families (power‑law,
   exponential, two‑sided power‑law, Beta‑pdf) and a few hand‑crafted
   shapes.  The best ~10 candidates are kept.

2. For each elite candidate we run two refinements:
   * a momentum sub‑gradient descent on the probability simplex
     (the same idea as the reference solution, but with a slightly
     more aggressive step‑size schedule);
   * an exponentiated‑gradient (mirror‑descent) refinement with an
     Adam‑style adaptive step‑size.

3. A cheap simulated‑annealing local search is applied to the current
   global best to escape possible local minima.

All operations respect the simplex constraint (non‑negative, sum = 1)
and use FFT‑based convolution for fast evaluation.  The final sequence
is validated with the provided ``evaluate_sequence`` function before
being returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2 * n * max(conv(seq, seq)) / (sum(seq)**2)
    using FFT.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families for the broad stochastic phase
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sin_shape(n: int, power: float = 1.0) -> np.ndarray:
    """
    a_i ∝ sin(π·(i+0.5)/n) ^ power.
    This concentrates mass at the ends when power < 0,
    and in the centre when power > 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    if power != 1.0:
        vals = np.sign(vals) * (np.abs(vals) ** power)
    return np.maximum(vals, 0.0)

# Gradient based refinements
# ----------------------------------------------------------------------


def subgradient_refine(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    momentum: float = 0.9,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Momentum sub‑gradient descent on the simplex (very similar to the
    reference implementation, but with a slightly more aggressive
    learning‑rate schedule).
    Returns the best sequence encountered and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)

    fft_len = next_pow2(2 * n)
    best_seq = a.copy()
    # objective for normalized sum=1 is simply 2*n*max(conv)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())

    v = np.zeros_like(a)
    it = 0
    no_improve = 0
    eps = 1e-12

    while time.time() < start_time + time_limit:
        it += 1
        # current convolution & max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # sub‑gradient of the peak (all active indices)
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # momentum + diminishing step‑size
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        # occasional random jitter to escape plateaus
        if no_improve >= 3000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    best_score = fast_score(best_seq)
    return best_seq, best_score


def exp_gradient_refine(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement.
    Uses an Adam‑style adaptive learning‑rate for the sub‑gradient.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = a / a.sum()  # ensure simplex
    n = a.shape[0]
    fft_len = next_pow2(2 * n)

    best_seq = a.copy()
    best_score = fast_score(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    it = 0

    while time.time() < start_time + time_limit:
        it += 1
        # compute convolution & max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # sub‑gradient of max (all active positions)
        active = np.where(conv >= M - 1e-12)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # Adam update on the sub‑gradient
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # exponentiated step (mirror descent on simplex)
        eta = eta0 / (np.sqrt(it) + 1.0)
        update = -eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(update)          # multiplicative update
        # renormalise to the simplex
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a.sum()

    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local search (tiny but sometimes helpful)
# ----------------------------------------------------------------------


def anneal_refine(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    init_temp: float = 0.12,
    final_temp: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing on the simplex: repeatedly move a
    random fraction of mass from one coordinate to another and accept
    according to a temperature schedule.
    """
    a = seq.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a.sum()

    best = a.copy()
    best_score = fast_score(best)
    n = a.shape[0]

    while time.time() < start_time + time_limit:
        # linear temperature schedule
        elapsed = time.time() - start_time
        temp = init_temp * (1.0 - elapsed / time_limit) + final_temp * (elapsed / time_limit)

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if a[i] < 1e-12:
            continue

        # move a random fraction of a[i] to a[j]
        frac = np.random.rand() * 0.3   # up to 30 % of the source mass
        delta = a[i] * frac
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # evaluation
        prop_score = fast_score(a_prop)

        if prop_score < best_score:
            best_score = prop_score
            best = a_prop.copy()
            a = a_prop
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(prop_score - best_score) / max(temp, 1e-12)):
                a = a_prop

    return best, best_score

            print(f"[refine] anneal improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the supplied evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
