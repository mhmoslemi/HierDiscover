"""
Hybrid optimizer for the self‑convolution bound.

1.  Parameter sweep of a power‑law family  a_i ∝ i^(‑p)  (golden‑section
    search) to obtain a good starting shape for a given length n.

2.  Long projected sub‑gradient descent on the simplex.
    The sub‑gradient is taken from all (near) active peaks of the
    convolution, not just a single one, which yields a smoother descent.
    A simple momentum term is used and the step size decays as 1/√t.

3.  A cheap simulated‑annealing refinement that moves a small amount of
    mass between random coordinates; occasional uphill moves allow the
    algorithm to escape shallow local minima.

The routine explores a few candidate lengths (≈2000, n, 2 n) and keeps the
best sequence seen before the time budget expires.  All numbers are kept
non‑negative and are normalised to sum to 1, which makes the objective
scale‑invariant.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return full linear convolution a*a and its maximal entry."""
    n = a.shape[0]
    L = 2 * n - 1
    fft_len = 1 << ((L).bit_length())          # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:L]
    return conv, float(conv.max())


def evaluate_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) / (sum(a)^2) (identical to evaluate_sequence)."""
    n = a.shape[0]
    conv, mx = compute_conv_and_max(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * mx / (s * s)


# ----------------------------------------------------------------------
# Power‑law family and golden‑section search for the exponent p
# ----------------------------------------------------------------------
def powerlaw_sequence(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def search_powerlaw(n: int, deadline: float) -> np.ndarray:
    """Find a good exponent p for a_i ∝ i^{-p} by golden‑section."""
    phi = (1 + 5 ** 0.5) / 2.0
    lo, hi = 0.30, 1.00               # plausible range
    best_seq = None
    best_score = np.inf
    for _ in range(30):
        if time.time() >= deadline:
            break
        p1 = hi - (hi - lo) / phi
        p2 = lo + (hi - lo) / phi
        seq1 = powerlaw_sequence(n, p1)
        seq2 = powerlaw_sequence(n, p2)
        s1 = evaluate_score(seq1)
        s2 = evaluate_score(seq2)
        if s1 < s2:
            hi = p2
            if s1 < best_score:
                best_score, best_seq = s1, seq1
        else:
            lo = p1
            if s2 < best_score:
                best_score, best_seq = s2, seq2
    if best_seq is None:
        best_seq = powerlaw_sequence(n, 0.55)
    return best_seq


# ----------------------------------------------------------------------
# Active‑set projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------
def active_subgradient_descent(a0: np.ndarray,
                               step0: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Long run of projected sub‑gradient descent using all near‑active peaks."""
    a = a0.copy()
    a = a / a.sum()                         # enforce Σa = 1
    n = a.shape[0]

    best_seq = a.copy()
    best_score = evaluate_score(a)

    v = np.zeros_like(a)                    # momentum term
    beta = 0.9

    it = 0
    while time.time() < deadline:
        it += 1
        conv, max_b = compute_conv_and_max(a)
        cur_score = 2.0 * n * max_b          # Σa = 1 ⇒ denominator =1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[sgd n={n} it={it}] new best={best_score:.6f}")

        # ------- active set (all peaks within 0.01 % of the maximum) -------
        tol = max(max_b * 1e-4, 1e-12)
        active = np.where(conv >= max_b - tol)[0]

        # ------- gradient from the active set -------
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution 2·a_{k‑idx}
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active))

        # ------- step size (1/√t) and momentum update -------
        step = step0 / np.sqrt(it)
        v = beta * v - step * grad
        a = project_simplex(a + v)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass between coordinates."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = evaluate_score(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # direction of the move
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # amount to move – up to 10 % of the source mass
        delta = np.random.uniform(0.0, 0.1) * a[src]
        if delta <= 0.0:
            continue
        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s
        new_score = evaluate_score(new)

        # accept if better, otherwise with Metropolis probability
        if new_score < best_score or np.random.rand() < math.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score
