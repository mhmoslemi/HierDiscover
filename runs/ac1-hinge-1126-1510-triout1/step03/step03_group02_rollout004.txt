    decreased geometrically (≈30 stages).  This stage strongly reduces the
    objective.

3.  Perform a *targeted* greedy hill‑climb that repeatedly moves a small
    fraction of mass from the index that contributes most to the current
    maximum of the auto‑convolution to a low‑value index.  The convolution is
    updated incrementally (no FFT per move), allowing many inexpensive moves
    and escaping shallow local minima.

The process is repeated with several random restarts until the time budget
runs out, after which a final polishing L‑BFGS pass and a longer hill‑climb
are executed.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1).

    Returns 2·n·max(a*a) which equals the value returned by
    ``evaluate_sequence`` when ``sum(a)==1``.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑softmax vector `y`.

    The smooth maximum is
        max_c + λ·log(∑exp((c−max_c)/λ)),
    with λ = lam_factor·max_c.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                       # d smooth_max / d conv_i
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: np.ndarray,
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation.
    The smoothing factor λ is reduced according to ``lam_factors``.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 20000,
    move_frac: float = 0.02,
    rng: np.random.Generator = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that specifically attacks the current peak of the
    auto‑convolution.

    * Choose the index `i_src` that contributes most to the maximal
      convolution value `k_max`.
    * Move a fraction `move_frac` of its mass to a low‑value index `j_tgt`.
    * Update the convolution incrementally (no FFT) and keep the move only
      if the objective improves.
    """
    if rng is None:
        rng = np.random.default_rng()

    best_a = a.copy()
    best_score = cur_score

    # initial convolution (FFT – cheap, done once)
    fa = np.fft.fft(best_a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    while time.time() < deadline and max_moves > 0:
        max_moves -= 1

        # location of the present maximum of the convolution
        k_max = int(np.argmax(conv))

        # indices that participate in this convolution entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range

        # contribution of each pair (i, j)
        contrib = best_a[i_range] * best_a[j_range]
        src_idx = int(i_range[np.argmax(contrib)])

        # choose a low‑mass target index (biased toward small values)
        eps = 1e-12
        inv_weights = 1.0 / (best_a + eps)
        inv_weights[src_idx] = 0.0                       # never pick the source
        probs = inv_weights / inv_weights.sum()
        tgt_idx = int(rng.choice(n, p=probs))

        # amount to transfer
        delta = move_frac * best_a[src_idx]
        if delta <= 0.0:
            continue

        # incremental convolution update:
        #    conv_new = conv + 2*delta*(shift(tgt) - shift(src))
        conv_new = conv.copy()
        conv_new[src_idx : src_idx + n] -= 2.0 * delta * best_a
        conv_new[tgt_idx : tgt_idx + n] += 2.0 * delta * best_a

        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            # accept move
            best_score = new_score
            best_a[src_idx] -= delta
            best_a[tgt_idx] += delta
            conv = conv_new
            # occasional re‑normalisation to avoid drift
            if max_moves % 1000 == 0:
                total = best_a.sum()
                if total != 0.0:
                    best_a /= total
        # else reject – keep `best_a` and `conv` unchanged

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats (its sum will be ≈ 1).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # problem size – reuse previous construction if possible
    # ------------------------------------------------------------------
    n = 5000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 500:
            n = max(n, int(arr.size))

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise a probability vector
    # ------------------------------------------------------------------
    if prev is not None and isinstance(prev, (list, tuple, np.ndarray)):
        a0 = np.asarray(prev, dtype=float)
        if a0.size != n:
            if a0.size < n:
                a0 = np.pad(a0, (0, n - a0.size), constant_values=0.0)
            else:
                a0 = a0[:n]
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 0.001, num=24)   # smoother → tighter
    maxiter_stage = 300

    iteration = 0

    # ------------------------------------------------------------------
    # multi‑restart optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a_start = np.exp(-rate * np.arange(n))
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.4, 2.0)
            a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.05)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_start = np.full(n, a_center)
            a_start[:m_edge] = a_edge
            a_start[-m_edge:] = a_edge
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.03, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 1.0)  # ~1 s per restart
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=5000,
            move_frac=0.02,
            rng=rng,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final deep refinement with the same continuation schedule
    # ------------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-16, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.5g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=20000,
            move_frac=0.01,
            rng=rng,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = (best_a / best_a.sum()).tolist()   # force exact normalisation
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
