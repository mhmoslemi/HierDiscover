"""
Hybrid optimizer for the step‑function evaluation problem.

The goal is to minimise  E(a)=2·n·max(a*a)/(sum a)².
The objective is scale‑invariant, therefore we restrict the search to the
probability simplex (∑a_i=1, a_i≥0).  The algorithm consists of three
stages:

1. **L‑BFGS on a smooth approximation** – many random restarts of a
   log‑sum‑exp (smooth‑max) surrogate, projected onto the simplex via
   a soft‑max parametrisation.  This quickly finds good basins.

2. **Linear‑programming refinement** – for the current vector a,
   solve the LP   maximise   1ᵀb
   subject to   a∗b ≤ M,   b≥0,
   where M = max(a∗a).  The optimal b has a larger ℓ₁‑norm than a.
   Mixing a and b as  a' = (a+b)/(1+∑b)  (which keeps the sum = 1)
   provably reduces the peak of the self‑convolution.  The LP is
   solved with SciPy’s HiGHS solver; we repeat this step as long as
   time permits.

3. **Projected hill‑climb** – a fast greedy mass‑transfer procedure
   that locally improves the exact objective 2·n·max(a∗a).  This
   finishes the search.

All phases respect a user‑supplied time budget (default 1000 s) and the
final sequence is returned as a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector already summing to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. a
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Direct minimisation of 2·n·max(a*a) via a simple diminishing‑step
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (simple mass transfers)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2500):
    """
    Greedy transfer of a small fraction of mass between two positions.
    Accept only moves that improve the exact score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, M: float):
    """
    Solve the LP  maximise 1ᵀb
    subject to   a∗b ≤ M,  b ≥ 0.
    Returns the new vector a' = (a+b)/(1+∑b) and the achieved σ=∑b.
    If the LP fails or yields σ≤1, None is returned.
    """
    n = a.shape[0]
    # Toeplitz matrix that maps b to a∗b
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([np.array([a[0]]), np.zeros(n - 1)])
    A = toeplitz(col, row)                # shape (2n‑1, n)

    ub = M * np.ones(2 * n - 1)

    c = -np.ones(n)                       # maximise sum(b)  ⇔ minimise -sum(b)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None, None

    b = res.x
    sigma = b.sum()
    if sigma <= 1.0 + 1e-12:              # no useful improvement
        return None, None

    a_new = (a + b) / (1.0 + sigma)       # keeps the sum exactly 1
    # Numerical safeguard
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new, sigma

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of heights found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Starting point – either the previously discovered sequence or uniform
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            if init_seq.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT lengths (powers of two) for speed
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradients / masks

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many cheap L‑BFGS restarts on a smoothed max
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.12 * budget_s
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    restart = 0
    while time.time() < phase1_end:
        lam = lam_schedule[restart % len(lam_schedule)]
        mode = rng.choice(['random', 'exp', 'power', 'edge'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {restart}] mode={mode} lam={lam:.3f} score={best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based refinement (the core improvement step)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.78 * budget_s
    a = best_a.copy()

    while time.time() < phase2_end:
        # current peak of the self‑convolution
        cur_maxconv = fast_score(a, fft_len_conv, n) / (2.0 * n)

        a_new, sigma = lp_refine(a, cur_maxconv)
        if a_new is None:
            # no further LP improvement possible
            break

        a = a_new
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] σ={sigma:.4f} score={best_score:.6f}")

        # safety break if improvement becomes tiny
        if sigma - 1.0 < 1e-4:
            break

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient polishing (optional, time‑aware)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sub, score_sub = subgradient_optimize(best_a,
                                                deadline,
                                                fft_len_conv,
                                                fft_len_g,
                                                n)
        if score_sub < best_score:
            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=4000)

    # ------------------------------------------------------------------
    # Report final evaluation (uses the provided evaluator)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    # Store for possible future runs
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
