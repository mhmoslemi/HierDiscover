    return np.convolve(seq, seq).max()

def normalize(seq, target_sum):
    seq = np.maximum(seq, 0)
    s = seq.sum()
    if s == 0:
        return np.full_like(seq, target_sum / len(seq))
    return seq * (target_sum / s)

def lp_update(f, target_sum, t_vals):
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    rows = 2*n-1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    c = -np.ones(n)
    b_ub = np.full(rows, M)
    bounds = [(0, None)]*n
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, M
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return None, M
    g = b_opt * (target_sum / sum_b)
    # convex combination steps
    for t in t_vals:
        f_new = (1 - t) * f + t * g
        M_new = np.convolve(f_new, f_new).max()
        if M_new < M - 1e-12:
            return f_new, M_new
    return f, M

"""
Propose a sequence via alternating linear‑programming updates
and greedy hill‑climbing.

The algorithm fixes the L1 norm of the step‑function to √(2 n) and
iteratively reduces the L∞ norm of its auto‑convolution:
1) start from a known good or random distribution;
2) compute the current maximal convolution value M;
3) solve a linear program that maximises the L1 norm of a new
   vector b under the linear constraints (a * b) ≤ M;
4) rescale b to the target norm and replace a by a convex
   combination (1‑t) a + t b with a decreasing step size t;
5) repeat until no improvement or time runs out;
6) finish with a fast greedy hill‑climb that moves small amounts
   of mass between entries while strictly decreasing the maximal
   convolution.
The final list is returned directly for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum entry of the auto‑convolution of ``seq``."""
    return np.convolve(seq, seq).max()

def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Force non‑negativity and scale to the supplied L1 norm."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        return np.full_like(seq, target_sum / seq.size)
    return seq * (target_sum / total)

def _lp_step(a: np.ndarray, target_sum: float, t_vals) -> tuple[np.ndarray, float]:
    """
    Perform one alternating‑LP update.

    Parameters
    ----------
    a : ndarray
        Current non‑negative vector (already normalised).
    target_sum : float
        Desired L1 norm (√(2 n)).
    t_vals : sequence of floats
        Candidate step sizes for the convex combination.

    Returns
    -------
    new_a : ndarray
        Updated vector (may be identical to the input if no improvement).
    new_score : float
        ``max(a*a)`` after the update.
    """
    n = a.size
    M = _conv_max(a)

    # --------------------------------------------------------------
    # Build the linear system A·b ≤ M, where A_{k,i} = a_i if
    # 0 ≤ k‑i < n, otherwise 0.  This is a dense (2n‑1)×n matrix.
    # --------------------------------------------------------------
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # fill column by column (vectorised inside the loop)
    for i in range(n):
        A[i : i + n, i] = a[i]

    # Linear program: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # fallback – return the unchanged vector
        return a, M

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return a, M

    # Rescale to the prescribed L1 norm
    g = b_opt * (target_sum / sum_b)

    # Try several step sizes; keep the first that strictly improves M
    for t in t_vals:
        candidate = (1.0 - t) * a + t * g
        new_M = _conv_max(candidate)
        if new_M < M - 1e-12:
            return candidate, new_M

    # No improvement found – signal stagnation by returning the original
    return a, M

def _hill_climb(a: np.ndarray, cur_score: float, deadline: float,
                max_moves: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing on the simplex.
    Moves a random fraction (≤10 %) of the mass from one index to another.
    Keeps a change only if it strictly lowers the maximal auto‑convolution.
    """
    n = a.size
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        max_moves -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = _conv_max(best)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    # Re‑normalise to avoid tiny drift caused by floating‑point noise
    target = np.sqrt(2.0 * n)
    best = _normalise(best, target)
    best_score = _conv_max(best)
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – try to use a fairly large n but stay within memory.
    # ------------------------------------------------------------------
    n = 2500
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise the vector a.
    # Prefer the previously discovered construction if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= n:
            a0 = arr[:n]
        else:
            # fallback to uniform
            a0 = np.full(n, target_sum / n)
    else:
        a0 = np.full(n, target_sum / n)

    a = _normalise(a0, target_sum)

    best_a = a.copy()
    best_score = _conv_max(best_a)
    print(f"[init] n={n} initial max(conv) = {best_score:.6f}")

    # Step‑size candidates for the convex update
    t_schedule = [0.5, 0.3, 0.2, 0.1, 0.05]

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimisation loop – alternating LP + occasional hill‑climb.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Occasionally restart from a fresh random distribution
        mode = rng.choice(["keep", "random", "exp", "power", "edge"])
        if mode != "keep":
            if mode == "random":
                a_start = rng.random(n)
            elif mode == "exp":
                rate = rng.uniform(0.001, 0.02)
                a_start = np.exp(-rate * np.arange(n))
            elif mode == "power":
                p = rng.uniform(0.3, 2.0)
                a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
            elif mode == "edge":
                m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
                edge_val = np.sqrt(n / 2.0) / n
                centre_val = 1.0 / np.sqrt(2.0) / n
                a_start = np.full(n, centre_val)
                a_start[:m_edge] = edge_val
                a_start[-m_edge:] = edge_val
            else:
                a_start = best_a
            a = _normalise(a_start, target_sum)

        # Perform a few LP updates (each reduces the max‑convolution)
        for _ in range(5):
            a_new, new_score = _lp_step(a, target_sum, t_schedule)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[iter {iteration}] mode={mode} new max(conv)={best_score:.6f}")
            # If the LP step did not improve the vector we break early
            if new_score >= _conv_max(a) - 1e-12:
                break
            a = a_new

        # Aggressive hill‑climb polishing – use up to half a second
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = _hill_climb(best_a, best_score, hc_deadline,
                                         max_moves=4000)
        a = best_a  # continue from the polished solution
        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _hill_climb(best_a, best_score, deadline,
                                         max_moves=8000)

    # ------------------------------------------------------------------
    # Return the sequence in plain Python list form.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Alternating linear‑programming (LP) updates + greedy hill‑climb.

The L1 norm of the step function is fixed to √(2 n).  Starting from a
non‑uniform distribution the algorithm repeatedly:
  1. computes M = max(a * a)   (auto‑convolution);
  2. solves   maximise Σ b  subject to b ≥ 0  and (a * b) ≤ M;
  3. rescales the solution to the target L1 norm and replaces a by a
     convex combination (1‑t) a + t b (trying several step sizes t);
  4. repeats until the maximal convolution cannot be reduced.
When the time budget is nearly exhausted a final greedy hill‑climb moves
small amounts of mass between entries while strictly decreasing the
maximal convolution.  The resulting list of non‑negative floats is
returned directly for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the linear convolution of ``seq`` with itself."""
    return np.convolve(seq, seq).max()


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Make ``seq`` non‑negative and scale it to have L1 norm = ``target_sum``."""
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        return np.full_like(seq, target_sum / seq.size)
    return seq * (target_sum / s)


def _lp_step(a: np.ndarray, target_sum: float, t_vals) -> tuple[np.ndarray, float]:
    """
    One alternating‑LP update.

    Parameters
    ----------
    a : ndarray
        Current non‑negative vector (already normalised).
    target_sum : float
        Desired L1 norm (√(2 n)).
    t_vals : sequence of step sizes to try for the convex combination.

    Returns
    -------
    new_a : ndarray
        Updated vector (may be identical to ``a`` if no improvement).
    new_score : float
        ``max(a*a)`` after the update.
    """
    n = a.size
    M = _conv_max(a)

    # Build the dense (2 n‑1) × n matrix A such that (A @ b)_k = (a * b)_k.
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a[i]

    # Linear program: maximise Σ b  ⇔  minimise –Σ b.
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # If the LP fails for any reason we simply return the unchanged vector.
        return a, M

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return a, M

    # Rescale to the prescribed L1 norm.
    g = b_opt * (target_sum / sum_b)

    # Try the supplied step sizes – keep the first that strictly lowers M.
    for t in t_vals:
        cand = (1.0 - t) * a + t * g
        new_M = _conv_max(cand)
        if new_M < M - 1e-12:
            return cand, new_M

    # No improvement found.
    return a, M


def _hill_climb(a: np.ndarray, cur_score: float, deadline: float,
                max_moves: int = 4000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing on the simplex.

    Randomly transfers up to 10 % of the source mass to another index.
    The move is kept only if it strictly decreases the maximal auto‑convolution.
    """
    n = a.size
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        max_moves -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10  # ≤10 % of source mass
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = _conv_max(best)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    # Re‑normalise to eliminate tiny drift caused by rounding.
    target = np.sqrt(2.0 * n)
    best = _normalise(best, target)
    best_score = _conv_max(best)
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The routine respects the 1000 s time budget and returns a list of floats
    directly usable by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – use a fairly large n but stay inside memory limits.
    # --------------------------------------------------------------
    default_n = 3000
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, np.ndarray)):
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = min(default_n, arr.size)
        else:
            n = default_n
    else:
        n = default_n

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial vector a.
    # Prefer a previously discovered construction; otherwise use an
    # edge‑biased distribution (known to beat the uniform case).
    # --------------------------------------------------------------
    if prev is not None and isinstance(prev, (list, np.ndarray)):
        arr = np.asarray(prev, dtype=float)
        if arr.size >= n:
            a0 = arr[:n]
        else:
            # Pad with a small uniform tail if the saved vector is too short.
            pad = np.full(n - arr.size, np.mean(arr))

    else:
        # Edge‑biased start (large values at the two ends, small in the centre).
        m_edge = max(1, int(n * 0.02))
        edge_val = np.sqrt(n / 2.0) / n
        centre_val = 1.0 / np.sqrt(2.0) / n

        a0[-m_edge:] = edge_val

    best_a = a.copy()
    best_score = _conv_max(best_a)
    print(f"[init] n={n} initial max(conv) = {best_score:.6f}")

    # Step‑size candidates for the convex update (large → small).
    t_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop: alternating LP updates + occasional hill‑climb.
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Occasionally restart from a fresh distribution to escape plateaus.
        mode = rng.choice(["keep", "random", "exp", "power", "edge"])
        if mode != "keep":
            if mode == "random":

            elif mode == "exp":

                a_start = np.exp(-rate * np.arange(n))
            elif mode == "power":

            elif mode == "edge":
                m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
                edge_val = np.sqrt(n / 2.0) / n
                centre_val = 1.0 / np.sqrt(2.0) / n
                a_start = np.full(n, centre_val)
                a_start[:m_edge] = edge_val
                a_start[-m_edge:] = edge_val
            else:
                a_start = best_a
            a = _normalise(a_start, target_sum)

        # Perform a short burst of LP updates (stop early if no improvement).
        inner_improved = False
        for _ in range(10):
            a_new, new_score = _lp_step(a, target_sum, t_schedule)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                inner_improved = True
                print(f"[iter {iteration}] mode={mode} LP improvement → max(conv)={best_score:.6f}")
            else:
                # No further progress from the LP with the current a.
                break

        # Aggressive hill‑climb polishing (use a fraction of the remaining time).
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = _hill_climb(best_a, best_score, hc_deadline,
                                         max_moves=6000)
        a = best_a
        iteration += 1

        # If a full LP burst produced no improvement, let the hill‑climb
        # try to shake things up before the next restart.
        if not inner_improved:
            continue

    # --------------------------------------------------------------
    # Final polishing hill‑climb (spend any remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _hill_climb(best_a, best_score, deadline,
                                         max_moves=12000)

    # --------------------------------------------------------------
    # Return the sequence in plain Python list form.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
