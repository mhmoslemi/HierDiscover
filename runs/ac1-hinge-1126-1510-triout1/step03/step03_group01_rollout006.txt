"""
Propose a low‑evaluation step sequence.

The search proceeds in three phases:

1. **Diverse deterministic starts** – we generate a catalogue of
   probability vectors on the simplex using a range of classical windows
   (exponential, power‑law, Tukey, Kaiser, Gaussian) and simple edge‑biased
   shapes.  Each candidate is scored with a fast FFT‑based evaluation
   (2·n·max(conv)) and the best is kept as a seed.

2. **Smooth‑max L‑BFGS refinement** – the seed is re‑parameterised with a
   soft‑max (unconstrained `y`‑vector).  A continuation schedule on the
   smoothing factor `lam_factor` drives the optimiser from a very smooth
   approximation of the max toward the true `max`.  Multiple random
   perturbations of the current best are performed while time permits.

3. **Greedy polishing** – a fast hill‑climb that repeatedly shifts a
   small fraction of mass between two indices is applied after each
   L‑BFGS run and finally until the budget runs out.

All operations respect the non‑negativity and sum‑normalisation constraints,
and the final sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import get_window

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum == 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep only the real part of the first 2*n‑1 entries
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the quantity 2·n·max(a*a) for a probability vector
    (sum == 1).  This is exactly the numerator of `evaluate_sequence` when
    the sum is 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. the
    unconstrained vector `y`.

    `lam_factor` controls the smoothing: a larger value yields a smoother
    approximation of the max.  The gradient is obtained via the chain rule
    through the soft‑max parametrisation.
    """
    # map to simplex
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient of smooth max ----------
    w = exp_shift / Z   # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    # this is a convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that aligns with a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    move_fraction: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb.  In each step a random fraction of the mass at
    index `i` is moved to a random index `j`.  The move is kept only if it
    strictly decreases `fast_score`.  The routine stops when the time budget
    expires or after `max_iter` successful attempts.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount of mass to move
        delta = best[i] * rng.random() * move_fraction
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_window_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a list of deterministic probability vectors based on classical
    window functions and simple parametric families.
    """
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # exponential decays (symmetrised)
    for lam in np.geomspace(0.001, 0.02, num=8):
        left = np.exp(-lam * np.arange(n))
        right = np.exp(-lam * np.arange(n - 1, -1, -1))
        a = left + right
        a = a / a.sum()
        cand.append(a)

    # power‑law (symmetrised)
    for p in np.linspace(0.2, 2.0, num=9):
        left = (np.arange(1, n + 1, dtype=float) ** -p)
        right = (np.arange(n, 0, -1, dtype=float) ** -p)
        a = left + right
        a = a / a.sum()
        cand.append(a)

    # Tukey windows (alpha from 0 to 1)
    for alpha in np.linspace(0.0, 1.0, num=7):
        w = get_window(("tukey", alpha), n, fftbins=False).astype(float)
        a = w / w.sum()
        cand.append(a)

    # Kaiser windows (beta from 0.5 to 12)
    for beta in np.linspace(0.5, 12.0, num=8):
        w = get_window(("kaiser", beta), n, fftbins=False).astype(float)
        a = w / w.sum()
        cand.append(a)

    # Gaussian windows (sigma as fraction of n)
    for sigma_frac in np.linspace(0.08, 0.32, num=6):
        sigma = sigma_frac * n
        w = get_window(("gaussian", sigma), n, fftbins=False).astype(float)
        a = w / w.sum()
        cand.append(a)

    # Edge‑heavy shapes (varying fraction of support at the ends)
    for edge_frac in np.linspace(0.01, 0.05, num=5):
        m_edge = max(1, int(edge_frac * n))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
        a = a / a.sum()
        cand.append(a)

    return cand


def generate_random_y(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random start in the soft‑max space (log‑probabilities)."""
    y0 = rng.standard_normal(n)
    return y0


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (larger n gives a better theoretical bound)
    # --------------------------------------------------------------
    n = 7000   # a compromise between quality and speed

    # If a previously discovered long sequence exists, reuse its length
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 2000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    print(f"[setup] Using n = {n}")

    # pre‑compute FFT lengths (next power of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Phase 1 – deterministic starts (window families)
    # --------------------------------------------------------------
    candidates = generate_window_candidates(n)

    # add a few random Dirichlet samples as extra seeds
    for _ in range(12):
        a_rand = rng.dirichlet(np.ones(n))
        candidates.append(a_rand)

    best_a = None
    best_score = np.inf

    for idx, a in enumerate(candidates):
        # ensure non‑negativity and normalise to sum 1
        a = np.maximum(0.0, a)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init] candidate {idx:03d} improved fast score = {best_score:.6f}")

    # Guarantee a fallback (uniform) if somehow nothing was kept
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len_conv, n)

    best_y = np.log(best_a + 1e-12)   # representation in soft‑max space

    # --------------------------------------------------------------
    # Phase 2 – stochastic L‑BFGS refinement with continuation
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 0.001, num=14))   # coarse schedule
    iteration = 0

    while time.time() < deadline:
        # Choose a perturbation strategy
        mode = rng.choice(["random", "best"])
        if mode == "best":
            # small Gaussian perturbation of the current best
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_random_y(n, rng)

        # L‑BFGS continuation
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # local greedy polishing
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 3 – final fine‑grained continuation and polishing
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-4, num=18))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # final greedy polishing with remaining budget
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Return the sequence (the harness will call evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception:
        # In the unlikely event that evaluate_sequence is unavailable
        print("[final] could not evaluate final score")
    return final_seq
