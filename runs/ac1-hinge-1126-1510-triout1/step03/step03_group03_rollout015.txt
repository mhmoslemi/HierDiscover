def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Phase 1: random search
    best_seq = None; best_score = float('inf')
    length_choices = [8000, 12000, 16000, 20000]
    families = ['beta', 'power', 'exp', 'two', 'gaussian', 'cosine']

    while time.time() < start + budget_s*0.3: # 30% time
        n = random.choice(length_choices)
        family = random.choice(families)
        if family == 'beta':
            α = random.uniform(0.1, 2.0)
            β = random.uniform(0.1, 2.0)
            i = np.arange(n, dtype=np.float64)
            seq = (i+1)**(α-1) * (n-i)**(β-1)
        elif family == 'power':
            p = random.uniform(0.2, 1.0)
            seq = generate_power_law(n, p)
        elif family == 'exp':
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        elif family == 'two':
            pl = random.uniform(0.2, 1.0)
            pr = random.uniform(0.2, 1.0)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        elif family == 'gaussian':
            sigma = random.uniform(0.08, 0.3) * n
            mid = (n-1)/2
            i = np.arange(n, dtype=np.float64)
            seq = np.exp(-0.5 * ((i-mid)/sigma)**2)
        elif family == 'cosine':
            i = np.arange(n, dtype=np.float64)
            seq = 0.5 * (1 - np.cos(2*np.pi*i/(n-1)))
        else:
            continue

        seq = np.maximum(seq, 0)
        s = seq.sum()
        if s <= 1e-12:
            continue
        seq /= s
        # symmetrize to ensure shape symmetry (optional)
        seq = (seq + seq[::-1]) / 2
        seq /= seq.sum()
        score = fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new random best {family} n={n} score={score:.6f}")

    # In case random search fails fallback to arcsine
    if best_seq is None:
        n = 12000
        best_seq = generate_arcsine_like(n)
        best_seq = best_seq / best_seq.sum()
        best_score = fast_score(best_seq)
        print(f"fallback arcsine length {n} score {best_score:.6f}")

    # Phase 2: gradient descent with active set
    a = best_seq.copy()
    n = a.shape[0]
    # optionally symmetrize a
    a = (a + a[::-1]) / 2
    a /= a.sum()

    fft_len = next_pow2(3*n)  # for gradient
    eta0 = 5.0
    tau = 5000.0
    beta = 0.03  # initial active set width
    beta_decay = 0.9995  # per iteration
    min_beta = 0.001
    jitter_interval = 4000
    jitter_scale = 0.005
    eval_interval = 1500
    iteration = 0

    while time.time() < deadline - 1.0:
        iteration += 1
        # compute self-convolution
        Fa = np.fft.rfft(a, n=fft_len)
        conv_self_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2*n - 1]
        max_val = conv_self_full.max()

        # active set indices
        thresh = max_val * (1.0 - beta)
        active_idx = np.where(conv_self_full >= thresh)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_self_full)], dtype=np.int64)

        # build h vector
        h = np.zeros(2*n - 1, dtype=np.float64)
        h[active_idx] = 1.0

        # compute gradient via convolution with reversed h
        h_rev = h[::-1]
        Fh = np.fft.rfft(h_rev, n=fft_len)
        conv_rev_full = np.fft.irfft(Fa * Fh, n=fft_len)[: 3*n - 2]
        # g_i = 2 * conv_rev[(2n-2) - i]
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * conv_rev_full[idx]

        # multiplicative update
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # symmetrize to keep shape symmetric
        a = (a + a[::-1]) / 2

        # normalize onto simplex
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / n)

        # decrease beta gradually
        beta = max(min_beta, beta * beta_decay)

        # jitter
        if jitter_interval and iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a += noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            a = (a + a[::-1]) / 2

        # evaluation
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] iter {iteration} new best {cur_score:.6f}")

    # final evaluation
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished. best eval = {final_score:.6f}")
    return final_seq

"""
Propose a low‑score step‑function (non‑negative heights) for the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has three stages:

1️⃣  **Initialisation** – try to reuse a previously computed solution
    (global variable ``height_sequence_1``) and fall back to an
    arcsine‑like distribution if it is unavailable.

2️⃣  **Broad random exploration** – generate many candidates from a
    variety of parametric families (power law, exponential, two‑sided
    power law, Gaussian, cosine, and the general Beta family) with
    different lengths.  The best candidate (lowest fast FFT‑based score)
    is kept.

3️⃣  **Active‑set multiplicative descent** – starting from the best
    candidate we iteratively lower the peak of the self‑convolution.
    In each iteration we:
        • compute the full self‑convolution,
        • select an *active set* of indices whose values are within a
          fraction ``beta`` of the current maximum,
        • compute the sub‑gradient of the (smoothed) max using a single
          FFT‑convolution with the indicator of the active set,
        • perform a multiplicative update ``a_i ← a_i·exp(−η·g_i)``,
        • symmetrise the sequence (the optimal shape is symmetric),
        • renormalise to the simplex.
    The step size ``η`` decays with the iteration counter, and ``beta``
    shrinks gradually so the algorithm first smoothes the whole
    convolution and later focuses on the true maximum.

    After the descent we perform a short random‑perturbation local
    search to escape possible plateaus.

The whole routine respects the 1000 s budget; intermediate progress
and the best score found are printed.  The returned list can be fed
directly into ``evaluate_sequence``.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT.
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))

# Families for random candidate generation
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  The unnormalised sequence behaves like
    1/√i and gives a near‑flat self‑convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    # Shift by 0.5 to avoid zero to the power of negative exponent
    i_shift = i + 1.0
    rev_shift = n - i
    seq = (i_shift ** (alpha - 1.0)) * ((rev_shift) ** (beta - 1.0))
    return seq


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian window with σ = sigma_frac·n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))

        # ---- enforce symmetry and normalise onto the simplex ----
        a = (a + a[::-1]) / 2.0
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

        # ---- gradually shrink the active‑set width ----
        beta = max(beta_min, beta * beta_decay)

        # ---- occasional jitter to avoid stagnation ----
        if jitter_interval and (iteration % jitter_interval) == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a = (a + a[::-1]) / 2.0

        # ---- periodic evaluation and bookkeeping ----
        if (iteration % eval_interval) == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] iter {iteration} new best {cur_score:.6f}")

        # ---- early termination if we already beat the target ----
        if best_score < 1.503:
            print(f"[{time.time() - start:.1f}s] target achieved (score={best_score:.6f})")
            break

    # ------------------ 3️⃣  Short random‑perturbation local search ------------------
    # Use the remaining time budget for cheap random tweaks.
    while time.time() < deadline - 1.0:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small random fraction of mass from i to j
        eps = 0.01 * np.random.rand()  # ≤1 %
        delta = eps * a[i]
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a.min() < 0.0:
            continue
        new_a = new_a / new_a.sum()
        # keep symmetry (helps stability)
        new_a = (new_a + new_a[::-1]) / 2.0
        new_a = new_a / new_a.sum()

        cur_score = fast_score(new_a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_a.copy()
            a = new_a  # continue local search from the improved point
            print(f"[{time.time() - start:.1f}s] perturbation improved to {cur_score:.6f}")

    # ------------------ 4️⃣  Final verification ------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # assumed to be pre‑imported
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
