def proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑evaluation step sequence using sub‑gradient descent on the
max‑convolution objective.  The algorithm works directly on a probability
vector (sum = 1) and repeatedly:

* computes the convolution a∗a,
* identifies the index with the largest value,
* takes a sub‑gradient of that max (which is just 2·a shifted by the
  arg‑max index),
* moves a in the negative gradient direction,
* projects back onto the probability simplex.

Random restarts, small Gaussian perturbations of the current best and a
lightweight greedy hill‑climb are interleaved.  The method runs until the
time budget expires and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is a constant vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random small fraction
    of mass from i to j (preserving Σa_i=1).  Keep a move only if the
    fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the max‑convolution objective.
    Each iteration:
        * compute the convolution,
        * locate its maximum index,
        * build the sub‑gradient (2·a shifted by that index),
        * take a step and project onto the simplex.
    A simple back‑tracking line search is used.
    """
    a = a_start.copy()
    conv = conv_fft(a, fft_len, n)
    max_c = conv.max()
    best_score = float(2.0 * n * max_c)
    best_a = a.copy()
    step = 0.2  # initial step size
    min_step = 1e-8

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current max‑convolution index
        k = int(conv.argmax())

        # sub‑gradient of the selected entry
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # back‑tracking line search
        eta = step
        while eta > min_step:
            a_new = project_simplex(a - eta * g)
            conv_new = conv_fft(a_new, fft_len, n)
            score_new = float(2.0 * n * conv_new.max())
            if score_new < best_score:
                a = a_new
                conv = conv_new
                best_score = score_new
                best_a = a_new.copy()
                break
            eta *= 0.5
        else:
            # Could not improve – shrink the global step size
            step *= 0.5
            continue

        # slowly decay the step size for stability
        step = max(step * 0.99, min_step)

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.

    Returns

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # problem size – larger n gives a finer search space
    n = 5000
    # If a previous construction is available and long enough, reuse its length.
    try:
        init_seq = globals().get("height_sequence_1", None)

    except Exception:
        pass

    fft_len = next_pow2(2 * n - 1)

    # start from the uniform distribution (baseline)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    iteration = 0
    modes = ["random", "best", "uniform"]

    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "random":
            a0 = rng.random(n)
            a0 = a0 / a0.sum()
        elif mode == "best":

            a0 += rng.normal(scale=0.02, size=n)

            a0 = a0 / a0.sum()
        else:   # uniform

        # --------------------------------------------------------------

        # --------------------------------------------------------------

            max_iter=2000,

        iteration += 1

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
