"""
Proposed candidate generator for the step‑function optimisation problem.

Algorithm
---------
1. **Simplex projection** – keep the coefficient vector on the
   probability simplex (non‑negative, sum = 1) after every update.
2. **Smooth max‑convolution gradient** – instead of the non‑smooth
   `max(conv(a,a))` we minimise a softmax approximation:
        m_β(a) = (1/β)·log Σ_k exp(β·conv_k(a)).
   Its gradient can be written in closed form and is computed by a pair
   of FFT‑based convolutions, yielding a cheap (≈ 1 ms) iteration for
   n≈2000.
   The smoothing parameter β is increased slowly, so the algorithm
   first explores a smooth landscape and then refines the true
   max‑objective.
3. **Projected gradient descent** – a diminishing step size
   `η_t = η₀ / √t` is used, followed by projection onto the simplex.
4. **Local stochastic refinement** – after the gradient phase a light
   simulated‑annealing walk moves a tiny amount of mass between two
   coordinates, accepting downhill moves and occasional uphill moves.
5. **Multiple restarts & length candidates** – the search is run for
   several sequence lengths (≈ 1500–4000) and random initialisations,
   keeping the best sequence ever seen.  The run stops as soon as the
   target score 1.503 is reached or the time budget expires.

The routine respects all constraints of the evaluation harness and
returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_max_conv(a: np.ndarray) -> float:
    """Exact maximal entry of the full linear convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(conv.max())


def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute the smoothed max of the self‑convolution and its gradient.

    Returns
    -------
    max_conv : float
        Exact maximum of a*a (used for the true objective).
    grad : np.ndarray
        Gradient of the smoothed max w.r.t. a (same length as a).
    """
    n = a.shape[0]
    # ----- exact convolution a*a -----
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]          # length 2n‑1
    max_conv = float(conv.max())

    # ----- soft‑max weights -----
    # w_k ∝ exp(beta * (conv_k - max_conv))
    exp_terms = np.exp(beta * (conv - max_conv))
    w = exp_terms / exp_terms.sum()                     # length 2n‑1, sums to 1

    # ----- gradient: 2 * (w ⋆ a) -----
    # cross‑correlation w ⋆ a = convolution(w, a_rev)
    a_rev = a[::-1]
    len_w = 2 * n - 1
    len_rev = n
    conv_len = len_w + len_rev - 1               # = 3n‑2
    L2 = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=L2)
    AR = np.fft.rfft(a_rev, n=L2)
    conv2 = np.fft.irfft(W * AR, n=L2)[:conv_len]

    # grad_j = 2 * Σ_k w_k * a_{k‑j}  → slice conv2[n‑1 : n‑1+n]
    grad = 2.0 * conv2[n - 1 : n - 1 + n]
    return max_conv, grad


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------
def smooth_descent(
    a0: np.ndarray,
    step0: float,
    beta0: float,
    beta_max: float,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run projected gradient descent on the simplex using the smoothed max.

    Parameters
    ----------
    a0 : np.ndarray
        Starting distribution (will be normalised to sum‑one).
    step0 : float
        Base step size (scaled by 1/√iter).
    beta0, beta_max : float
        Initial / final smoothing parameter.
    max_iters : int
        Upper bound on the number of iterations.
    deadline : float
        Unix timestamp after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        The best sequence seen during the run.
    best_score : float
        Corresponding exact evaluation score.
    """
    a = a0.astype(np.float64, copy=True)
    a = a / a.sum()                     # enforce Σa = 1
    n = a.shape[0]

    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    beta = beta0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_conv, grad = smooth_max_and_grad(a, beta)

        # step size schedule
        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

        # exact score (since max_conv is exact)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[smooth it {it:5d}] new best = {score:.6f}")

        # increase smoothing slowly
        if beta < beta_max:
            beta = min(beta_max, beta * 1.004)

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float):
    """
    Cheap random walk that moves a small amount of mass between two
    coordinates, accepting improvements (or occasional uphill moves).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02  # initial acceptance temperature

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # move up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # accept if better, otherwise with Metropolis probability
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # cool down slowly
        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score
