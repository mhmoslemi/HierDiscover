def adam_optimize(a0, max_iters, gamma_start, gamma_end, lr, deadline):
    n = a0.shape[0]
    y = np.log(a0 + 1e-30)  # small epsilon
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = a0.copy()
    best_score = float('inf')
    # precompute fft lengths
    fft_len_a = 1 << ((2*n - 1).bit_length())
    fft_len_corr = 1 << ((3*n -2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        # conv
        A = np.fft.rfft(a, fft_len_a)
        conv = np.fft.irfft(A * A, fft_len_a)[:2*n-1]
        # compute actual score
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute gamma
        gamma = gamma_start + (gamma_end - gamma_start) * (it - 1) / (max_iters - 1)
        # compute smooth weights
        z = gamma * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / exp_z.sum()

        A_rev = np.fft.rfft(a[::-1], fft_len_corr)
        W = np.fft.rfft(w, fft_len_corr)
        corr_full = np.fft.irfft(W * A_rev, fft_len_corr)[:len(w) + len(a) - 1]

"""
A smooth‑max Adam optimiser on the probability simplex.

The score to be minimised is

    2 * n * max(conv(a, a))          (∑a = 1, a_i ≥ 0)

Because the objective is homogeneous we work on the simplex.
A smooth approximation of the max (log‑sum‑exp) is used so that
the gradient is well‑defined.  The variables are the logits y,
a = softmax(y) guarantees the simplex constraints automatically.
Adam (with a slowly increasing “temperature” γ) drives the
distribution towards a shape whose self‑convolution is as flat
as possible.  For each candidate length we first perform a cheap
grid search over power‑law exponents, then run the Adam optimiser
from the best exponent, and finally apply a tiny stochastic
local refinement.  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length) using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation r_i = Σ_j w_{i+j} a_j   (0 ≤ i < n)
    realised as convolution of w with the reversed a.
    Returns the length‑n vector r.
    """
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a_rev, fft_len)
    full = np.fft.irfft(W * A, fft_len)[: w.shape[0] + a_rev.shape[0] - 1]
    n = a_rev.shape[0]
    # slice the n entries that correspond to i = 0 … n‑1
    start = n - 1
    return full[start:start + n]


# ----------------------------------------------------------------------
# Simple utility – score for a distribution that already sums to 1 -----
# ----------------------------------------------------------------------
def score_from_distribution(a: np.ndarray) -> float:
    """Exact score (no smoothing) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Grid search over power‑law exponents ------------------------------------
# ----------------------------------------------------------------------
def powerlaw_grid_search(n: int, p_grid) -> tuple[np.ndarray, float]:
    """Return the best power‑law vector among the supplied exponents."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    best_score = np.inf
    best_vec = None
    for p in p_grid:
        a = idx ** (-p)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = score_from_distribution(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
    return best_vec, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth‑max objective) -----------------
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    lr: float,
    gamma_start: float,
    gamma_end: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the logits y with a smooth‑max surrogate of max(conv).
    Returns the best concrete distribution (sum = 1) seen during the run.
    """
    n = a0.shape[0]

    # initialise logits (softmax is the simplex projection)
    eps = 1e-30
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # pre‑computed FFT lengths
    fft_len_a = 1 << ((2 * n - 1).bit_length())
    fft_len_corr = 1 << ((3 * n - 2).bit_length())

    best_seq = a0.copy()
    best_sc = np.inf

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- distribution from logits ---------------------------------
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # ---- exact convolution and current score -----------------------
        conv = full_convolution(a)                         # length 2n‑1
        cur_sc = 2.0 * n * np.max(conv)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            # occasional progress report
            print(f"[Adam n={n} it={it}] new best score={cur_sc:.6f}")

        # ---- smooth‑max weights ----------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * (it - 1) / (max_iters - 1)
        z = gamma * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stable softmax
        w = exp_z / exp_z.sum()            # weights sum to 1

        # ---- gradient of the smooth objective w.r.t. a -----------------
        #   grad_a = 4 * n * r   where r_i = Σ_k w_k a_{k‑i}
        r = cross_correlation(w, a[::-1], fft_len_corr)   # length n
        grad_a = 4.0 * n * r

        # ---- convert to gradient in log‑space (y) ----------------------
        #   ∂f/∂y = a ⊙ (∂f/∂a − ⟨a,∂f/∂a⟩)
        inner = np.dot(a, grad_a)
        grad_y = a * (grad_a - inner)

        # ---- Adam step ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Light stochastic refinement (mass transfer) -------------------------
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """Tiny simulated‑annealing style walk on the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_sc = score_from_distribution(a)

    temperature = 0.02
    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        sc = score_from_distribution(new)

        if sc < best_sc or np.random.rand() < np.exp((best_sc - sc) / temperature):
            a = new
            if sc < best_sc:
                best_sc = sc
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_sc
