    # Maintain velocity momentum on gradient.
    v = np.zeros_like(a)
    best_seq = a.copy()
    # compute initial best_score via fast_score
    best_score = fast_score(best_seq)
    it = 0
    while time.time() < start_time + time_limit:
        it += 1
        # compute convolution and max
        n=a.shape[0]; fft_len=next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        M = conv.max()
        cur_score = 2.0 * n * M
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute subgradient
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)

def refine_sequence(seq, time_limit, start_time, eta0=0.5, momentum=0.9):
    a = seq.copy()
    n = a.shape[0]
    a = project_simplex(a)
    best_seq = a.copy()
    # compute best_score initially
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())
    v = np.zeros_like(a)
    it = 0
    stagnation = 0
    while time.time() < start_time + time_limit:
        it += 1
        # compute conv and M
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # subgradient
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        # compute contributions for each active index
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # momentum
        v = momentum * v + (1 - momentum) * g
        # step size schedule
        eta = eta0 / (np.sqrt(it) + 1.0)
        # multiplicative update
        a = a * np.exp(-eta * v)
        # renormalize to sum 1
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / sum_a

        # random perturb if stagnation large
        if stagnation >= 5000:
            sigma = 0.02 / np.sqrt(n)
            a = a * np.exp(sigma * np.random.randn(n))
            a = a / a.sum()
            stagnation = 0

    # final best score via fast_score
    best_score = fast_score(best_seq)
    return best_seq, best_score

The algorithm has three stages:

1. **Initialisation** – try to reuse the previously best sequence
   ``height_sequence_1`` when it exists, otherwise fall back to a simple
   two‑sided power‑law shape.

   collection of parametric families (power‑law, exponential,
   two‑sided power‑law, Beta, Dirichlet and a sine shape).  Each candidate
   is normalised to sum = 1 and cheaply scored with an FFT‑based

   The sub‑gradient is the sum of contributions from *all* convolution
   entries that are (numerically) at the current maximum.  A momentum

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT convolution.  The sequence is assumed non‑negative and
    already normalised (sum = 1) – the denominator therefore equals 1.
    """
    n = seq.shape[0]
    # FFT length is a power of two ≥ 2·n to avoid circular effects
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

# Parametric shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(–p_right).
    Symmetric if the parameters coincide.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha‑1)·(n‑i)^(beta‑1) .
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector (Gamma trick)."""
    samples = np.random.gamma(alpha, 1.0, size=n)
    return samples


def generate_sin_shape(n: int) -> np.ndarray:
    """Simple symmetric bump a_i = sin(π·i/(n+1)).  Positive on (0,π)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.sin(np.pi * i / (n + 1))


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient refinement
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.5,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex with momentum.
    Returns the best sequence encountered and its exact fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)                     # ensure feasibility
    n = a.shape[0]
    fft_len = next_pow2(2 * n)

    # initialise best
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())
    best_seq = a.copy()

    v = np.zeros_like(a)                      # momentum accumulator
    it = 0
    stagnation = 0
    eps = 1e-12

    while time.time() < start_time + time_limit:
        it += 1

        # ----- current objective (already have conv from previous loop?) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ----- sub‑gradient of the peak (all active indices) ----------
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ----- momentum & step‑size ------------------------------------
        v = momentum * v + (1.0 - momentum) * g
        eta = eta0 / (np.sqrt(it) + 1.0)

        # ----- multiplicative update, then project onto simplex ----------
        a = a * np.exp(-eta * v)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # ----- occasional random perturbation to escape plateaus -----
        if stagnation >= 5000:
            sigma = 0.02 / np.sqrt(n)
            a = a * np.exp(sigma * np.random.randn(n))
            a = a / a.sum()
            stagnation = 0

    # final exact score for the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score


def noise_anneal_search(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    sigma0: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: multiply the current vector by random
    log‑normal noise, re‑project to the simplex and accept if the score
    improves.  The noise amplitude decays with time.
    """
    best_seq = seq.copy()
    best_score = fast_score(best_seq)
    n = seq.shape[0]
    it = 0
    while time.time() < start_time + time_limit:
        it += 1
        # geometric cooling of the noise amplitude
        sigma = sigma0 * (0.5 ** (it / 800.0))
        cand = best_seq * np.exp(sigma * np.random.randn(n))
        cand = cand / cand.sum()
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within the allowed time budget.
    Returns a plain Python list of non‑negative floats.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) Initialise (use previous best if available)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # fallback: modest two‑sided power law
        init_len = 6000
        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() <= 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad stochastic search (≈30 % of total budget)
    # ------------------------------------------------------------------
    rand_deadline = start + 0.3 * budget_s
    length_choices = [4000, 6000, 8000, 10000, 12000, 15000, 18000, 20000]
    families = ["power", "exp", "twosided", "beta", "dirichlet", "sin"]
    elite = [(best_score, best_seq.copy())]   # keep top‑5 later

    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 1.30)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-5, 8e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.05, 1.30)
            p_r = random.uniform(0.05, 1.30)
            w_l = random.uniform(0.2, 5.0)
            w_r = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            alpha = random.uniform(0.05, 1.70)
            beta = random.uniform(0.05, 1.70)
            seq = generate_beta_shape(n, alpha, beta)

        elif family == "dirichlet":
            alpha = random.uniform(0.01, 5.0)
            seq = generate_dirichlet(n, alpha)

        else:  # "sin"
            seq = generate_sin_shape(n)

        # Normalise and safeguard
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[rand] new global best ({family}) n={n} score={sc:.6f}")

        # maintain elite pool (≤5)
        if len(elite) < 5:
            elite.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    # sort elite by score (ascending)
    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2) Momentum exponentiated‑gradient refinement on elites
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 0:
        per_candidate = time_remaining / len(elite)
        for idx, (sc, seq) in enumerate(elite):
            if time.time() >= deadline:
                break
            print(f"[refine] elite #{idx + 1}/{len(elite)} len={len(seq)} start_score={sc:.6f}")
            refined_seq, refined_score = refine_sequence(
                seq, per_candidate, time.time()
            )
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final short noise‑annealing step (use any leftover time)
    # ------------------------------------------------------------------
    leftover = deadline - time.time()
    if leftover > 5.0:    # need at least a few seconds to be worthwhile
        best_seq, best_score = noise_anneal_search(
            best_seq, leftover, time.time()
        )
        print(f"[noise] post‑anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Final reported score = {final_score:.6f}")

    return final_seq
