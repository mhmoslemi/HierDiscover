"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two stages:

1️⃣ **Broad random sampling** – families of analytic shapes (power‑law,
   exponential, two‑sided power‑law, arcsine, cosine, Gaussian, Beta)
   with random parameters and several lengths are sampled.  The best
   candidate (lowest fast_score) is retained.

2️⃣ **Soft‑max driven exponentiated gradient descent** – starting from the
   best sample we minimise a smooth approximation of the peak of the
   self‑convolution.  For a temperature λ we form the soft‑max weights
   w_k ∝ exp(λ·conv_k).  The (sub‑)gradient of the smoothed objective is
   g = 2·corr(w,a) and we perform a multiplicative update
   a_i ← a_i·exp(−η·g_i) followed by renormalisation onto the probability
   simplex.  λ is increased slowly (approximation becomes sharper) and the
   step‑size η is decayed.  The current best vector is recorded whenever
   the fast_score improves.

Both stages respect the overall 1000 s budget.  The final sequence is
returned as a plain Python list of floats.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast (FFT‑based) evaluation of
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  The unnormalised sequence behaves like
    1/√i and gives a near‑flat self‑convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred at the middle, sigma expressed as a fraction of n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+1)/(n+1))^{α‑1} * (1‑(i+1)/(n+1))^{β‑1}.
    """
    x = (np.arange(1, n + 1, dtype=np.float64)) / (n + 1.0)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)  # safety for α,β<1 causing infinities at ends
    return a

# Main optimisation routine

def propose_candidate(budget_s: int = 1000, seed: int = 42) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
    except Exception:
        # fallback – a moderate‑size arcsine‑like sequence
        init = generate_arcsine_like(8000)

    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.full_like(init, 1.0 / init.size)
    else:
        init = init / init.sum()          # normalise to a probability simplex

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Randomised family exploration (≈15 % of the budget)
    # --------------------------------------------------------------
    random_phase_end = start + 0.15 * budget_s
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "exp",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
        "beta",
    ]

    while time.time() < random_phase_end:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_window(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)
            cand = generate_gaussian_window(n, sigma_frac)

        elif family == "beta":
            alpha = random.uniform(0.2, 1.5)
            beta = random.uniform(0.2, 1.5)
            cand = generate_beta_shape(n, alpha, beta)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new random best ({family}) "
                f"n={n} score={sc:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣  Soft‑max exponentiated‑gradient refinement (remaining budget)
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Length for FFTs: need to accommodate convolution of w (len 2n‑1) with a_rev (len n)
    fft_len = next_pow2(3 * n)

    # Hyper‑parameters (tuned empirically)
    eta0 = 0.5          # initial (multiplicative) step size factor
    tau = 5000.0        # decay time for η
    lambda0 = 5.0       # start temperature for soft‑max
    lambda_max = 200.0  # final temperature
    max_iters = 200000  # safety cap
    eval_interval = 2000
    jitter_interval = 8000
    jitter_scale = 0.005

    best_iter_score = best_score
    best_iter_seq = a.copy()

    it = 0
    while time.time() < deadline - 0.5 and it < max_iters:
        it += 1

        # ---- self‑convolution (FFT) ----
        Af = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

        # ---- soft‑max weights ----
        lam = lambda0 + (lambda_max - lambda0) * min(1.0, it / max_iters)
        # stability trick: subtract max before exponentiation
        shift = conv_full.max()
        w_raw = np.exp(lam * (conv_full - shift))
        w = w_raw / w_raw.sum()   # shape (2n‑1,)

        # ---- gradient via correlation (FFT) ----
        # a_rev = a[::-1]
        Afr_rev = np.fft.rfft(a[::-1], n=fft_len)
        Wf = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(Wf * Afr_rev, n=fft_len)[: 3 * n - 2]
        # extract the n‑length gradient (correlation slice)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # real values

        # ---- multiplicative update with decaying η ----
        eta = eta0 / (1.0 + it / tau)
        # clip the exponent to avoid under/overflow
        delta = -eta * grad
        delta = np.clip(delta, -50.0, 50.0)
        a = a * np.exp(delta)

        # ---- renormalise onto simplex (guarantees positivity) ----
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- occasional jitter to break symmetry ----
        if it % jitter_interval == 0:
            noise = (np.random.randn(n) * jitter_scale) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # ---- periodic evaluation on the cheap score ----
        if it % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_iter_score:
                best_iter_score = cur_score
                best_iter_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] iter {it}, new best score={cur_score:.6f}"
                )
            # also keep the overall best across both phases
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] >>> overall best improved to {cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
