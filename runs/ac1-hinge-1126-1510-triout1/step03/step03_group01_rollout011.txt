def project_onto_simplex(v):
    # ensures non-negative and sum = 1
    # algorithm by Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def solve_g_lp(a, M):

        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # Hill climb with more aggressive moves
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, deadline, max_iter=5000)
        best_y = np.log(best_a + 1e-12)

        # LP improvement step (if time)
        if time.time() < deadline - 5.0:
            M = np.max(conv_fft(best_a, fft_len_conv, n))
            g = solve_lp_for_g(best_a, M)
            if g is not None:
                g_sum = np.sum(g)
                if g_sum > 1e-12:
                    a_new = g / g_sum
                    score_new = fast_score(a_new, fft_len_conv, n)
                    if score_new < best_score:
                        best_score = score_new
                        best_a = a_new
                        best_y = np.log(best_a + 1e-12)
                        print(f"[{iteration}] LP improve score = {best_score:.6f}")

        iteration += 1

    # Final polishing with projected subgradient descent
    max_pg_iters = 4000
    lr = 0.1
    a = best_a.copy()
    score = best_score
    for pg in range(max_pg_iters):
        if time.time() >= deadline:
            break
        conv = conv_fft(a, fft_len_conv, n)
        k0 = int(np.argmax(conv))
        # compute subgradient
        grad = np.zeros_like(a)
        # fill grad based on k0
        start = max(0, k0 - (n-1))
        end = min(n-1, k0)
        # For i from start to end, j = k0 - i
        # grad[i] = 2 * a[j]
        for i in range(start, end+1):
            j = k0 - i
            grad[i] = 2.0 * a[j]
        # step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < score:
            a = a_new
            score = new_score
            # optionally increase lr slightly for faster descent
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.5
        if pg % 200 == 0:
            print(f"[pg {pg}] subgrad score = {score:.6f}")

    best_a = a
    best_score = score

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Propose a low‐evaluation step sequence.

Algorithm
---------
* Work on the probability simplex (sum = 1) using a soft‑max (y → a).
* Perform many random restarts and a continuation L‑BFGS with a
  geometrically decreasing smooth‑max parameter (lam) to minimise
  2·n·max(a∗a).
* After each L‑BFGS run apply an aggressive greedy hill‑climb that
  repeatedly transfers a random fraction of mass between two entries.
* When time permits, solve a linear programme
      maximise   Σ g_j
      subject to a∗g ≤ M ,   g ≥ 0
  (where M = max(a∗a)).  The optimal g has Σg>1, therefore the normalised
  vector ĝ = g/Σg yields a strictly smaller max‑convolution.
* Finally polish the best vector with a projected sub‑gradient descent
  that uses the exact sub‑gradient of the max‑convolution.
* Return the best a found (as a plain list of non‑negative floats).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained vector y (soft‑max parametrisation).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z

    # convolution a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random fraction of
    mass from i to j (keeping the sum = 1).  Keep a move only if the
    fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 15 % of the mass at i
        delta = best[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict:
                if candidate * 2 in sums:
                    conflict = True
                else:
                    chosen.append(candidate)
                    for c in chosen:
                        sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError("unknown mode")
    return y0


# ----------------------------------------------------------------------
# Linear‑programme improvement step (optional, requires cvxpy)
# ----------------------------------------------------------------------
def solve_lp_for_g(a: np.ndarray, M: float):
    """
    Solve   maximise Σ g_j   subject to   a ∗ g ≤ M , g ≥ 0 .
    Returns the optimal g as a NumPy array (or None on failure).
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(a, g)                     # length 2n‑1, linear in g
    constraints = [conv <= M]                # element‑wise inequality
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000, verbose=False)
    except Exception:
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)
