def compute_gradient_multi(a, mask):
    n = a.shape[0]
    # pad lengths for FFT
    full_len = len(mask) + n - 1  # = (2n-1) + n-1 = 3n - 2
    fft_len = 1 << ((full_len - 1).bit_length())
    A = np.fft.rfft(a[::-1], n=fft_len)
    M = np.fft.rfft(mask, n=fft_len)
    grad_full = np.fft.irfft(A * M, n=fft_len)[:full_len]
    grad = 2 * grad_full[n-1:n-1+n]
    return grad
```

"""
Hybrid grid‑search + multi‑peak projected sub‑gradient descent.

The score to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the expression is homogeneous we keep a on the probability simplex
(a_i ≥ 0, Σ a_i = 1).  First a cheap exhaustive search over a few
parameterised families (power‑law, exponential and mixtures with a uniform
tail) provides strong initial candidates.  Those seeds are refined by a
projected sub‑gradient method that follows *all* convolution peaks that are
close to the current maximum (rather than only the single active peak).  A
light simulated‑annealing local walk finishes each run.  Multiple independent
restarts and a few problem sizes (≈2k‑12k) are performed until the time
budget expires.  The best sequence ever observed is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a (shape 2n‑1) and its maximum entry.
    Uses an FFT of power‑of‑two length for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Multi‑peak projected sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def multi_peak_subgradient(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    top_k: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run a projected sub‑gradient descent that follows *all* peaks that are
    within a small fraction of the current maximal convolution value.
    """
    a = a0.copy()
    a = a / a.sum()                       # enforce Σa = 1

    best_seq = a.copy()
    best_score = float("inf")
    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[multi_subgrad it {it}] new best={score:.6f}")

        # --------------------------------------------------------------
        # Build a mask of the “almost‑active’’ peaks.
        # --------------------------------------------------------------
        delta = 0.02                         # consider peaks within 2 % of max
        thresh = max_b * (1.0 - delta)
        active = np.where(conv >= thresh)[0]

        # keep only the top‑k indices (to keep the step cheap)
        if active.size > top_k:
            sel = np.argpartition(conv[active], -top_k)[-top_k:]
            active = active[sel]

        # --------------------------------------------------------------
        # Sub‑gradient: sum of contributions of the selected peaks.
        # --------------------------------------------------------------
        grad = np.zeros_like(a)
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(i, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[i - idx]

        step = step0 / np.sqrt(it)          # diminishing step size
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap random walk that moves a small amount of mass between two
    coordinates, accepting improvements (or occasional uphill moves).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02                      # initial acceptance temperature
    eps = 1e-12

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction: move mass from src → dst
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta <= eps:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s <= eps:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # Accept if better, otherwise with Metropolis probability
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # slow cooling
        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score
