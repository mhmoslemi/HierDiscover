"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that
minimises the evaluation metric

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is deliberately different from the reference implementation:

1. **Large‐n arcsine start** – the discrete arcsine density
   `1 / sqrt(x (1-x))` is known to give a very flat self‑convolution.
   We try several lengths (up to 20000) because the asymptotic limit is
   `score = 2·sup (f∗f) = 1.5`; larger n gets us closer to the bound.

2. **Smoothed gradient descent** – replace the hard max by a soft‑max
   `log‑sum‑exp(α·conv)/α`.  The gradient of the smoothed objective is
   `2·correlation(weights, a)`.  We start with a small smoothing factor
   (`α≈2`) and increase it exponentially up to `α≈200`.  An Adam‑style
   update on the simplex (projection onto the probability simplex) is
   used; the learning rate decays as `1/√t`.

3. **Hard sub‑gradient phase** – once `α` is large we switch to the true
   max, using a uniform weight on the active set (all lags whose
   convolution value is within `tol` of the maximum).  The same
   correlation‑based gradient and a diminishing step size `η/√t` are used.

4. **Random hill‑climbing** – after the deterministic phases we perform
   many tiny mass transfers between random coordinates, accepting only
   improvements.  This helps to escape tiny plateaus.

All heavy work (convolution, correlation) is performed with FFTs
(`O(L log L)` per iteration).  The routine respects the supplied time
budget (`budget_s` seconds) and returns the best sequence found,
scaled to the normalisation required by `evaluate_sequence`.

The code contains no closures, no lambdas and only uses the standard
library together with NumPy.
"""

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <w, a*a>  w.r.t. a.
    Implements  2·correlation(w, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_w = np.fft.fft(w, n=L)

    # FFT of reversed a (zero‑padded) = conj(fft_a) * phase
    fft_a_rev = np.conj(fft_a) * phase

    # correlation = ifft(fft_w * fft_a_rev)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # extract the n entries that correspond to the gradient
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of the true max."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # fallback – should never happen
        w[conv.argmax()] = 1.0
    return w

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine respects the time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # try a few problem sizes; larger n gives a better asymptotic bound,
    # but each iteration becomes more expensive.
    n_candidates = [5000, 10000, 20000]      # feel free to adjust
    best_score = float("inf")
    best_seq = None
    best_n = None

    # ------------------------------------------------------------------
    for n in n_candidates:
        if time.time() > deadline:
            break

        # ==== initialise with the discrete arcsine density ====
        xs = np.arange(n, dtype=np.float64)
        x_mid = (xs + 0.5) / n
        a = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
        a /= a.sum()                     # L1‑normalisation (sum = 1)

        # FFT helpers ---------------------------------------------------
        L = 1 << ((2 * n - 1).bit_length())   # smallest power of two ≥ 2n‑1
        phase = np.exp(2j * np.pi * np.arange(L) / L)

        # ------------------------------------------------------------------
        # Phase 1: smoothed Adam‑style descent
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr0 = 0.15                     # base learning rate
        max_adam_iters = 25000
        alpha_start, alpha_end = 2.0, 200.0

        for it in range(1, max_adam_iters + 1):
            if time.time() > deadline:
                break

            conv = _compute_convolution(a, L)
            cur_max = conv.max()
            # Current (pre‑scaled) score = 2·n·cur_max
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                best_n = n
                print(f"[n={n}] Adam it {it:5d}  new best score={best_score:.6f}")

            # smooth weights

            w = _softmax_weights(conv, alpha)

            # gradient of smoothed objective
            grad = _gradient_from_weights(a, w, phase, L)

            # Adam update on the simplex

            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step_size * grad
            a_new = _project_simplex(a_new)
            a = a_new

        # ------------------------------------------------------------------
        # Phase 3: cheap random hill‑climbing (mass transfers)
        while time.time() < deadline:
            # pick a source proportionally to its mass (more mass → higher prob)
            i = np.random.choice(n, p=a)

                best_seq = a_new.copy()
                best_n = n

    if best_seq is None:

        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)

        print(f"--- Finished: best n={best_n} best pre‑scaled score={best_score:.6f} "
              f"elapsed={time.time() - start_time:.1f}s")

    return final_seq
