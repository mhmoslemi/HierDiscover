"""
search for a low‑value step‑function sequence.

The algorithm works on the probability simplex (sum = 1) using three
ingredients:

1. A continuation scheme that minimises a smooth approximation of the
   max‑operator (log‑sum‑exp) with L‑BFGS.  The smoothing parameter λ is
   decreased geometrically, each stage performing a modest number of
   L‑BFGS iterations.

2. A large amount of random restarts with diverse initialisations
   (uniform, exponential, power‑law, sinusoidal, edge‑biased and a
   perturbation of the current best).  The best vector found so far is
   kept between restarts.

3. Aggressive hill‑climbing on the simplex after each L‑BFGS run and a
   final fine‑grained gradient descent.  All candidates are symmetrised
   (a[i] = a[n‑1‑i]) before they are evaluated – symmetry is known to be
   beneficial for the autocorrelation problem.

The search runs for at most ``budget_s`` seconds (default 1000) and
returns the best non‑negative sequence found.  The returned list can be
passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fallback – should never happen
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    ``a`` must have length ``n``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # result is length fft_len; the linear part we need is the first 2·n‑1 terms
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector (sum = 1).
    This equals the value returned by ``evaluate_sequence`` when the
    sum of ``a`` is 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def smooth_max_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the self‑convolution and its gradient w.r.t. the
    unconstrained logits ``y`` (log‑soft‑max parameters).

    Parameters
    ----------
    y : ndarray
        Logits; the probability vector is ``a = softmax(y)``.
    n : int
        Length of the vector.
    fft_len_conv, fft_len_g : int
        FFT sizes for the two convolutions.
    lam_factor : float
        λ = max_c * lam_factor  (smooth‑max width).

    Returns
    -------
    f : float
        Smoothed objective  ``2·n·smooth_max(conv)``.
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y`` (same shape as ``y``).
    """
    # probability vector
    a = softmax(y)

    # ----- a * a -------------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max (log‑sum‑exp trick)
    shifted = conv - max_c            # maximum becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth‑max ------------------------------------
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # we compute the convolution a_rev * w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=smooth_max_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb on the simplex.
    Randomly moves a fraction of the mass from one coordinate to another
    and keeps the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta == 0.0:
            continue
        # propose move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise – tiny rounding errors may appear after the move
        s = cand.sum()
        if s <= 0:
            continue
        cand /= s
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
            # keep the improvement
        # else discard

    return best, best_score


def symmetrize(v: np.ndarray) -> np.ndarray:
    """
    Enforce a[i] = a[n-1-i] (mirror symmetry) and renormalise to sum = 1.
    """
    s = (v + v[::-1]) * 0.5
    total = s.sum()
    if total == 0.0:
        # backup to uniform
        return np.full_like(v, 1.0 / v.size)
    return s / total


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Wall‑clock time budget (seconds).  The search stops a few seconds
        before the deadline to leave room for the final evaluation.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    # use a larger dimension than before (more flexibility) but keep it
    # manageable for the time budget.
    default_n = 15000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                # keep the larger of the two lengths, but cap at 20000
                default_n = max(default_n, min(int(arr.size * 2), 20000))
        except Exception:
            pass

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise best candidate (uniform distribution)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    # smoother → finer schedule
    lam_factors = np.geomspace(0.5, 1e-5, num=25)
    maxiter_stage = 300  # L‑BFGS iterations per continuation stage

    iteration = 0

    # --------------------------------------------------------------
    # multi‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "sin", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n, dtype=float))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "sin":
            # half‑sine shape, strictly positive
            x = np.arange(n, dtype=float) + 0.5
            a0 = np.sin(np.pi * x / n)
            a0 += 1e-6  # avoid exact zeros
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        a_opt = symmetrize(a_opt)               # enforce mirror symmetry
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive local hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=6000,
        )
        # re‑symmetrise after hill‑climb
        best_a = symmetrize(best_a)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final deep refinement (continuation L‑BFGS with the same schedule)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=smooth_max_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-13, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_a = symmetrize(best_a)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final fine‑grained gradient‑descent polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        # start from the current best logits
        y = best_y.copy()
        step_size = 0.05
        # decreasing λ schedule (very small smoothing → almost exact max)
        fine_lams = np.geomspace(1e-5, 1e-7, num=12)
        for lam in fine_lams:
            for _ in range(30):
                if time.time() >= deadline:
                    break
                f_val, grad_y = smooth_max_and_grad(y, n, fft_len_conv, fft_len_g, lam)
                # simple back‑tracking line search in the log‑space
                for _ in range(4):
                    y_new = y - step_size * grad_y
                    a_new = softmax(y_new)
                    a_new = symmetrize(a_new)
                    new_score = fast_score(a_new, fft_len_conv, n)
                    if new_score < best_score:
                        y = y_new
                        best_a = a_new
                        best_score = new_score
                        step_size *= 1.1
                        break
                    else:
                        step_size *= 0.5

    # --------------------------------------------------------------
    # final polishing hill‑climb (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
        )
        best_a = symmetrize(best_a)
        best_score = fast_score(best_a, fft_len_conv, n)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # ``evaluate_sequence`` is supplied by the harness – we just call it.
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
