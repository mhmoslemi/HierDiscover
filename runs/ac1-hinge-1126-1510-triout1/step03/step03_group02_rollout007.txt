def proj_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    # Based on algorithm from https://arxiv.org/pdf/1101.6081.pdf
    # Works for 1-D arrays.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all nonpositive, return uniform? Actually projection onto simplex yields zero vector? Not possible sum=1?
        # In this case set to uniform.
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient–descent + hill‑climb + final L‑BFGS refinement.

The search works directly on the probability simplex (∑ a_i = 1, a_i ≥ 0).
1. Start from the best known construction (global `height_sequence_1`) or a
   simple heuristic (uniform, exponential, power‑law, edge‑biased).
2. Repeatedly:
   – Compute the auto‑convolution a∗a.
   – Identify the lag(s) where the convolution attains its maximum.
   – Form a sub‑gradient of the max‑operator (2·shifted copy of a) and
     take a projected‑gradient step on the simplex.
   – Occasionally perform a random restart to avoid stagnation.
   – Run a short greedy hill‑climb that moves a fraction of the mass from
     one coordinate to another only when the fast‐score improves.
3. When the time budget is almost exhausted, polish the solution with a
   continuation of L‑BFGS on a smooth approximation of the max‑operator.
4. Return the best non‑negative sequence found (its sum is 1, which does not
   affect the final evaluation because the objective is scale‑invariant).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a∗a) for a probability vector (sum = 1).
    This equals the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` is 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex (sum = 1, v ≥ 0).
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection; fall back to uniform
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_subgradient(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of f(a)=max_k (a∗a)_k at the active indices ``idxs``.
    For each active lag k we add the shifted copy of a, then average.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        j_range = k - i_range
        g[i_range] += a[j_range]
    if idxs.size > 1:
        g /= float(idxs.size)
    g *= 2.0                     # derivative of a_i a_{k-i}
    return g


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Soft‑max L‑BFGS utilities (final polishing)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (the log‑soft‑max coordinates).

    Returns
    -------
    f : float
        Smooth approximation of 2·n·max(a*a).
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y`` (same shape as ``y``).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z               # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_factors: list[float],
                      maxiter_per_stage: int = 200) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations with a decreasing smoothing factor.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-14,
                                "disp": False})
        y = res.x
    return y

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence with low evaluation score.
    The routine combines projected sub‑gradient descent, aggressive hill‑climbing,
    and a final L‑BFGS polishing phase.
    """

    deadline = time.time() + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------

    base_n = 5000
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= base_n:
        n = len(prev)
        a0 = np.asarray(prev, dtype=float)
    else:
        n = base_n
        a0 = np.full(n, 1.0 / n)

    # make sure a0 is a proper probability vector
    a0 = np.maximum(a0, 0.0)
    if a0.size != n:
        if a0.size > n:
            a0 = a0[:n]
        else:
            a0 = np.concatenate([a0, np.zeros(n - a0.size)])
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # gradient‑descent + hill‑climb loop
    # --------------------------------------------------------------
    iter_counter = 0
    eta0 = 5.0               # initial step‑size multiplier
    restart_prob = 0.02      # probability of a random restart each outer loop
    max_outer_iter = 10**9   # effectively unlimited; loop stops by deadline

    while time.time() < deadline - 5.0 and iter_counter < max_outer_iter:
        # --- optional random restart -------------------------------------------------
        if rng.random() < restart_prob:
            mode = rng.choice(["random", "exp", "power", "edge"])
            if mode == "random":
                a_tmp = rng.random(n)
            elif mode == "exp":
                rate = rng.uniform(0.001, 0.02)
                a_tmp = np.exp(-rate * np.arange(n))
            elif mode == "power":
                p = rng.uniform(0.3, 2.0)
                a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            else:          # edge‑biased
                m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
                a_edge = np.sqrt(n / 2.0) / n
                a_center = 1.0 / np.sqrt(2.0) / n
                a_tmp = np.full(n, a_center)
                a_tmp[:m_edge] = a_edge
                a_tmp[-m_edge:] = a_edge
            a_tmp = np.maximum(a_tmp, 0.0)
            a_tmp /= a_tmp.sum()
            cur_a = a_tmp
        else:
            cur_a = best_a.copy()

        # --- a few projected sub‑gradient steps ------------------------------------
        steps = 200
        # rough estimate of remaining time: assume ~0.01 s per step
        max_steps = min(steps, int((deadline - time.time()) / 0.012) )
        for step in range(max_steps):

            max_val = conv.max()
            idxs = np.where(conv >= max_val - 1e-12)[0]
            grad = compute_subgradient(cur_a, idxs)

            eta = eta0 / np.sqrt(iter_counter * max_steps + step + 1)

            cur_a = proj_simplex(raw)

        hc_deadline = min(deadline, time.time() + 0.6)
