"""
Propose candidate using linear-programming refinement.
...
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

def full_conv(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-2).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[:2*n-1]

def fast_score(a):
    n = a.shape[0]
    conv = full_conv(a)
    return float(2.0 * n * conv.max())

def build_toeplitz(a):
    n = a.shape[0]
    # Toeplitz matrix where column j is a shifted down by j
    c = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
    r = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(n-1, dtype=np.float64)])
    return toeplitz(c, r)

def lp_refine(a, deadline, max_iter=20):
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(a)
    eps = 1e-14
    it = 0
    while time.time() < deadline and it < max_iter:
        # Ensure strict positivity to avoid unbounded LP
        a_pos = np.maximum(a, eps)
        conv = full_conv(a_pos)
        M = conv.max()
        # Build Toeplitz constraint matrix
        # Using sparse could be better but dense works for n~2000.
        T = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            T[j:j+n, j] = a_pos
        # Linear program: maximize sum(b)  <=> minimize -sum(b)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed (success={res.success}), message: {getattr(res, 'message', '')}")
            break
        b = res.x
        s = b.sum()
        if s <= 1.0 + 1e-12:
            # No slack, cannot improve further
            break
        g = b / s
        # Evaluate candidate
        new_score = fast_score(g)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = g.copy()
            print(f"[LP] iter {it} improved score to {best_score:.6f} (sum_b={s:.6f})")
        # Update a to g for next iteration (full replace)
        a = g.copy()
        it += 1
    return best_a, best_score

def hill_climb(a, current_score, deadline, max_iters=2000):
    n = a.shape[0]
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5% transfer
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimizing the evaluation metric.
    The algorithm starts from the previously discovered best sequence (if any),
    then performs a series of linear‑programming refinements that exploit the
    linear bound `a * b ≤ M` (where `*` denotes convolution).  Each LP step tries
    to increase the total mass under the current maximal convolution `M`; after
    scaling the solution back to a probability simplex we replace the current

"""
Propose a candidate step sequence by linear‑programming refinement.

The routine starts from the best known sequence (if the global variable
`height_sequence_1` is present) or from a uniform distribution.
It repeatedly solves a linear programme

    maximise   Σ b_j
    subject to  b_j ≥ 0,
                (a * b)_k ≤ M   for all k,

where `a` is the current distribution, `*` denotes linear convolution and
`M = max(a * a)`.  The LP tries to increase the total mass while keeping the
convolution with the current iterate bounded.  The solution `b` is rescaled to
unit sum, mixed back into `a` (full, 75 % or 50 % steps are tried) and the
process repeats until no further improvement is possible or the time budget
expires.  A short greedy hill‑climb finishes the run.

All calculations are performed on the probability simplex (sum = 1); the
evaluation metric is `2·n·max(a*a)` which is scale‑invariant.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def full_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray) -> float:
    """Fast evaluation 2·n·max(a*a) for a distribution that already sums to 1."""
    n = a.shape[0]
    conv = full_conv(a)
    return float(2.0 * n * conv.max())


def lp_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 20) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.

    Repeatedly:
        * computes M = max(a*a)
        * solves   maximise Σ b_j  subject to  a*b ≤ M, b ≥ 0
        * rescales b to unit sum → g
        * tries mixing steps a ← (1‑t)·a + t·g  (t∈{0.5,0.75,1.0})
        * accepts the first move that improves the true score.

    Stops when the LP gives no slack (Σb ≤ 1) or when no mixing step reduces
    the score, or when the time budget runs out.
    """
    n = a.shape[0]
    eps = 1e-14                # ensures strict positivity for the LP
    best_a = a.copy()
    best_score = fast_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iter:
        # strictly positive copy (required to keep the LP bounded)
        a_pos = np.maximum(a, eps)

        # current maximal convolution
        M = full_conv(a_pos).max()

        # ------------------------------------------------------------
        # Build the Toeplitz matrix that represents the linear map
        #   b ↦ a_pos * b.
        # ------------------------------------------------------------
        # First column = a_pos padded with zeros; first row = [a0,0,…,0]
        c = np.concatenate([a_pos, np.zeros(n - 1, dtype=np.float64)])
        r = np.concatenate([np.array([a_pos[0]], dtype=np.float64),
                            np.zeros(n - 1, dtype=np.float64)])
        T = toeplitz(c, r)          # shape (2*n-1, n)

        # ------------------------------------------------------------
        # Linear programme: maximise Σ b_j   ⇔ minimise -Σ b_j
        # ------------------------------------------------------------
        c_obj = -np.ones(n, dtype=np.float64)   # minimise -Σb
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c_obj, A_ub=T, b_ub=b_ub,
                      bounds=bounds, method='highs')
        if not res.success:
            # LP failed (infeasible or unbounded) – stop the refinement.
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s <= 1.0 + 1e-12:
            # No slack – cannot increase the total mass any more.
            break

        # Rescale to the simplex.
        g = b / s

        # ------------------------------------------------------------
        # Try a few mixing factors and keep the first that improves.
        # ------------------------------------------------------------
        improved = False
        for t in (0.5, 0.75, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, eps)     # keep positivity
            cand /= cand.sum()                # stay on the simplex
            cand_score = fast_score(cand)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand.copy()
                improved = True
                print(f"[LP] iter {it:02d} t={t:.2f} → score {best_score:.6f} "
                      f"(Σb={s:.6f})")
                break
        if not improved:
            # No mixing step gave an improvement → stop.
            break

        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               deadline: float,
               max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb that repeatedly transfers a small random
    fraction of mass between two positions (keeping the sum fixed) and keeps
    any improving move.  Used as a final polishing phase.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05   # up to 5 % of source mass
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the change
            best[i] += delta
            best[j] -= delta
    return best, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    2. Normalise the vector to the probability simplex and ensure strict
       positivity (tiny epsilon) to keep the LP bounded.
    3. Perform a linear‑programming refinement loop (≈20 iterations or until
       the time budget is exhausted) that tries to increase the total mass
       under the current maximal convolution bound.
    4. Finish with a short greedy hill‑climb to clean up any remaining
       exploitable local moves.
    5. Return the best sequence as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Load or create the initial distribution
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000                     # keep the problem size manageable
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            total = init_seq.sum()
            if total <= 0.0:

    else:

    # ------------------------------------------------------------------
    # 2. Initial score
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a)
    print(f"[init] n={n}  initial score={best_score:.6f}")
