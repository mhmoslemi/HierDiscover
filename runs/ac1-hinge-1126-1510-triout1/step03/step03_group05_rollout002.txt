def compute_gradient(a, conv, beta):
    max_conv = conv.max()
    if beta > 0:
        shifted = conv - max_conv
        # Clip to avoid overflow issues: large negative values become zero after exp. We'll compute exp of min(beta*shifted, max_exponent)
        # Use np.exp with proper dtype.
        exp_vals = np.exp(beta * shifted)  # might overflow on huge positive? shift ensures max is zero, so exponent <= 0, safe.
        w = exp_vals / exp_vals.sum()
    else:
        # pure max: gradient concentrated on active indices => set w 1/m on active.
        eps = 1e-12
        active = np.where(conv >= max_conv - eps)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
    # now compute correlation: g = 2 * correlation of a with w for lags [0..n-1]
    n = a.shape[0]
    fft_len = next_pow2(3 * n - 2)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    Corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
    # Corr length = fft_len
    # The full correlation is the first (n + wlen - 1) = 3n-2 entries of Corr; beyond that maybe zero.
    # Get slice:
    start = len(w) - 1  # = 2n-2
    g = 2.0 * Corr[start : start + n]
    return g

def gradient_descent(seq, time_limit, start_time, beta0=5.0, beta_max=200.0, eta0=0.5, momentum=0.9):

    a = seq.copy()
    a = a / a.sum()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = fast_score(a)
    g_mom = np.zeros_like(a)
    iterations = 0
    no_improve = 0
    while time.time() < start_time + time_limit and iterations < 5000:
        iterations += 1
        conv = full_convolution(a)  # length 2n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # same as fast_score
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # compute beta schedule
        beta = beta0 + (beta_max - beta0) * min(1.0, (iterations - 1) / 2000.0)  # linear increase in first 2000 iters
        # compute gradient
        g = compute_gradient(a, conv, beta)
        # step size
        eta = eta0 / (np.sqrt(iterations) + 1.0)
        g_mom = momentum * g_mom + (1.0 - momentum) * g
        a_new = a * np.exp(-eta * g_mom)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / sum_a

        a = a_new

        # occasional random perturbation
        if no_improve >= 2000:
            sigma = 0.02 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    # final best evaluation using fast_score (maybe compute actual objective)
    best_score = fast_score(best_seq)
    return best_seq, best_score

def compute_gradient(a, conv, beta):
    n = a.shape[0]
    max_c = conv.max()
    shifted = conv - max_c
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()
    # correlation a with w for lags 0..n-1
    fft_len = next_pow2(3 * n - 2)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    Corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
    start = len(w) - 1  # 2n-2
    g = 2.0 * Corr[start : start + n]
    return g

"""
search for a low‑score step‑function (non‑negative sequence).

The algorithm works in three stages:

1. **Broad stochastic search** – generate many random shapes (power‑law,
   exponential, two‑sided power‑law, Beta‑pdf, Gaussian) of different
   lengths, normalise them to unit ℓ₁‑norm and keep the best few candidates
   (the “elite” set).

   projected multiplicative‑gradient method on the simplex.  The
   non‑differentiable max                                                                                                                                        (soft‑max of the convolution) is used as a smooth
   surrogate for the ℓ∞‑norm.  The temperature (β) of the soft‑max is
   annealed from a modest value up to a large one, turning the surrogate
   progressively into the true max.  Multiplicative updates keep the
   iterates non‑negative; after each step they are renormalised to stay on
   the probability simplex.

3. **Optional polishing** – a short random‑perturbation local search can

The best sequence found is finally evaluated with the supplied

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Cheap evaluation of 2·n·max(conv(seq,seq))/(sum(seq)**2)
    using an FFT based convolution.  Assumes seq is non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-½·((i‑μ)/σ)²) with centre μ = (n‑1)/2."""
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


# ----------------------------------------------------------------------
# Gradient computation (soft‑max surrogate of the max‑convolution)
# ----------------------------------------------------------------------


def compute_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. the vector a.
    Uses correlation a ⋆ w, where w_i ∝ exp(beta·conv_i).
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights (stable formulation)
    shifted = conv - max_c
    exp_vals = np.exp(beta * shifted)
    w_sum = exp_vals.sum()
    if w_sum == 0.0:
        # fall back to pure active‑set weighting
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w = exp_vals / w_sum

    # correlation a with w for lags 0 … n‑1
    fft_len = next_pow2(3 * n - 2)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
    start = len(w) - 1            # = 2·n‑2
    g = 2.0 * corr[start : start + n]   # gradient (same size as a)
    return g


# ----------------------------------------------------------------------
# Soft‑max gradient descent on the simplex
# ----------------------------------------------------------------------


def softmax_descent(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    eta0: float = 0.5,
    momentum: float = 0.9,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative gradient descent using a soft‑max surrogate.
    Returns the best sequence (ℓ₁‑normalised) found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = a / a.sum()                     # stay on the simplex
    n = a.shape[0]

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    g_mom = np.zeros_like(a)           # momentum buffer
    no_improve = 0

    it = 0
    while time.time() < start_time + time_limit and it < max_iters:
        it += 1

        # ----- current objective -----
        conv = full_convolution(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv   # identical to fast_score for sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- annealed temperature -----
        # linear increase from beta0 to beta_max during the first 2000 iters
        if it < 2000:
            beta = beta0 + (beta_max - beta0) * (it - 1) / 1999.0
        else:
            beta = beta_max

        # ----- gradient of the surrogate -----
        g = compute_gradient(a, conv, beta)

        # ----- stepsize (scaled by gradient norm) -----
        grad_norm = np.linalg.norm(g) + 1e-12
        eta = eta0 / (np.sqrt(it) * grad_norm)

        # ----- momentum update -----
        g_mom = momentum * g_mom + (1.0 - momentum) * g

        # ----- multiplicative update & projection to simplex -----
        a_new = a * np.exp(-eta * g_mom)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s
        a = a_new

        # ----- occasional random jitter to escape plateaus -----
        if no_improve >= 3000:
            sigma = 0.02 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    # final polishing with the cheap fast_score
    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Small random‑perturbation local search (optional polishing)
# ----------------------------------------------------------------------


def local_random_search(
    seq: np.ndarray, time_limit: float, start_time: float
) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing with Gaussian jitter."""
    a = seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    while time.time() < start_time + time_limit:
        sigma = 0.01 / np.sqrt(n)
        cand = best_seq + sigma * np.random.randn(n)
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # 0) Base candidate (possibly from a previous run)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # fallback: moderate two‑sided power‑law
        init_len = 8000
        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad random search – build an elite pool
    # ------------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000, 14000]
    families = ["power", "exp", "twosided", "beta", "gauss"]
    elite: list[tuple[float, np.ndarray]] = [(best_score, best_seq.copy())]

    rand_deadline = start + 0.3 * budget_s
    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 2.0)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-5, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.05, 2.0)
            p_r = random.uniform(0.05, 2.0)
            w_l = random.uniform(0.2, 5.0)
            w_r = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            alpha = random.uniform(0.05, 2.0)
            beta = random.uniform(0.05, 2.0)
            seq = generate_beta_shape(n, alpha, beta)

        else:   # "gauss"
            sigma = random.uniform(n / 30.0, n / 5.0)
            seq = generate_gaussian(n, sigma)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)

        # keep global best
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:.1f}s] new global best ({family}) n={n} score={sc:.6f}")

        # maintain an elite set (size ≤ 8)
        if len(elite) < 8:
            elite.append((sc, seq.copy()))
        else:
            # replace the worst elite entry if this one is better
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    elite.sort(key=lambda x: x[0])   # ascending by score

    # ------------------------------------------------------------------
    # 2) Soft‑max gradient descent on the elite pool
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0.0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted before refinement. Final score = {final_score:.6f}")
        return final_seq

    per_candidate = time_remaining / len(elite)

    for idx, (score, seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"[refine] elite #{idx+1} start (score={score:.6f}, len={seq.size})")
        refined_seq, refined_score = softmax_descent(
            seq,
            time_limit=per_candidate * 0.85,   # most of the budget for descent
            start_time=time.time(),
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:.1f}s] improved global best → {best_score:.6f}")

        # small polishing phase with random local search (remaining time)
        leftover = per_candidate * 0.15
        if leftover > 0.1:
            polished_seq, polished_score = local_random_search(
                refined_seq, time_limit=leftover, start_time=time.time()
            )
            if polished_score < best_score:
                best_score = polished_score
                best_seq = polished_seq.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:.1f}s] polishing improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
