import time
import random
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

def fast_score(seq):
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    # using FFT
    fft_len = next_pow2(2*n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (s*s))

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# shape generators
def generate_power_law(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    return idx ** (-p)

def generate_exponential(n, lam):
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)

def generate_two_sided_power_law(n, p_left, p_right, w_left=1.0, w_right=1.0):
    i = np.arange(1, n+1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

def generate_beta_shape(n, alpha, beta):
    i = np.arange(1, n+1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))

def generate_sine(n):
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)

def generate_triangle(n):
    half = n // 2
    left = np.linspace(0, 1, half, endpoint=False)
    right = np.linspace(1, 0, n - half)
    return np.concatenate([left, right])

    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    return float(conv.max())

        M = compute_max_conv(seq)
        # Build the sparse constraint matrix A (shape (2n-1, n))
        data = np.repeat(seq, n)   # length n*n
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        # inequality RHS
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        # linear objective: maximize sum(b)  => minimize -sum(b)
        c = -np.ones(n, dtype=np.float64)

        # allocate a small time slice for the LP to keep overall budget
        remaining = time_deadline - time.time()
        lp_options = {'presolve': True}
        # If remaining small, give it a time limit
        if remaining < 5.0:
            lp_options['time_limit'] = max(0.1, remaining * 0.9)

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options=lp_options)

        if not res.success:
            # LP failed; break and return current seq
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No improvement possible (unlikely)
            break

        g = b / sum_b
        # convex combination
        seq = (1.0 - t) * seq + t * g
        # keep strictly positive to avoid unbounded LP next round
        eps = 1e-12
        seq = np.maximum(seq, eps)
        seq = seq / seq.sum()

        # update best if improved
        cur_M = compute_max_conv(seq)
        cur_score = 2.0 * n * cur_M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()

        max_iter -= 1

    return best_seq, best_score

def refine_sequence(seq, time_limit, start_time, momentum=0.9, eta0=0.5):
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)
    fft_len = next_pow2(2*n)

    best_seq = a.copy()
    # compute initial objective
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    best_score = 2.0 * n * float(conv.max())

    v = np.zeros_like(a)
    it = 0
    no_improve = 0
    while time.time() < start_time + time_limit:
        it += 1
        # evaluate current
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # subgradient
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # momentum step
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        if no_improve >= 3000:
            sigma = 0.03 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    best_score = fast_score(best_seq)
    return best_seq, best_score

"""
search for a step‑function with a low value of `evaluate_sequence`.

The algorithm works in three stages:

1. **Random shape generation** – many candidates (power‑law,
   exponential, two‑sided power‑law, Beta, sine, triangular) of a
   moderate length (≈2500) are created, normalised to sum = 1 and
   scored quickly with an FFT based “fast_score”.  The best few
   (an elite set) are kept.

2. **LP‑based refinement** – for each elite candidate we run a few
   iterations of the linear‑programming step described in the prompt:

      maximise Σ b_i
      s.t.   b_i ≥ 0
             (f * b)[k] ≤ M   for all k,
      where M = max(f * f).

   The LP is solved with SciPy’s HiGHS solver.  The optimal b is
   normalised to a probability vector g and mixed with the current
   f (convex combination with weight t).  This drives the peak of the
   convolution down quickly.

3. **Momentum sub‑gradient polishing** – the best vector obtained from
   stage 2 is refined with the projected momentum sub‑gradient method
   used in the previous state‑of‑the‑art code.  The remaining time
   budget is spent here.

All steps respect the 1000 s time limit.  The final sequence is
returned as a plain Python list of non‑negative floats.
"""

import time
import random
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(–p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(–lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(–p_left) + w_right·(n‑i)^(–p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha‑1)·(n‑i)^(beta‑1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def generate_triangle(n: int) -> np.ndarray:
    """A simple symmetric triangular shape."""
    half = n // 2
    left = np.linspace(0, 1, half, endpoint=False)
    right = np.linspace(1, 0, n - half)
    return np.concatenate([left, right])


# ----------------------------------------------------------------------
# Convolution helper
# ----------------------------------------------------------------------


def max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k]."""
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” from the statement)
# ----------------------------------------------------------------------


def lp_iterate(
    seq: np.ndarray,
    t_mix: float,
    max_iter: int,
    deadline: float,
    row_idx: np.ndarray,
    col_idx: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑based refinement.
    Returns the best sequence found and its fast_score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_M = max_convolution(seq)
    best_score = 2.0 * n * best_M

    bounds = [(0, None)] * n   # b_i ≥ 0

    while time.time() < deadline and max_iter > 0:
        M = max_convolution(seq)

        # Build the sparse constraint matrix A for (f * b) ≤ M.
        data = np.repeat(seq, n)               # length n*n
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)      # maximise Σb_i  → minimise –Σb_i

        remaining = deadline - time.time()
        opts: dict = {"presolve": True}
        if remaining < 2.0:
            opts["time_limit"] = max(0.05, remaining * 0.9)

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method="highs", options=opts)

        if not res.success:
            # LP failed – give up further LP work for this candidate.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No improvement possible.
            break

        g = b / sum_b                     # re‑normalise to Σg_i = 1
        seq = (1.0 - t_mix) * seq + t_mix * g

        # keep a tiny positive floor to avoid unbounded LP in the next round
        eps = 1e-12
        seq = np.maximum(seq, eps)
        seq = seq / seq.sum()

        cur_M = max_convolution(seq)
        cur_score = 2.0 * n * cur_M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()

        max_iter -= 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Momentum sub‑gradient refinement (the method used previously)
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    momentum: float = 0.9,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected momentum sub‑gradient descent on the simplex.
    Returns the best sequence found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)               # ensure feasibility
    fft_len = next_pow2(2 * n)

    best_seq = a.copy()
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())

    v = np.zeros_like(a)
    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # sub‑gradient: contribution of all entries attaining the maximum
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # momentum update
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        # occasional random jitter to jump out of plateaus
        if no_improve >= 3000:
            sigma = 0.03 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    `evaluate_sequence`.  The algorithm follows the three‑stage plan
    described in the module docstring.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) Choose a moderate length – large enough to give a good bound,
    #    small enough for the LP solver to stay fast.
    # ------------------------------------------------------------------
    n = 2500

    # Pre‑compute the (row, col) index pattern for the Toeplitz matrix
    # representing the convolution constraints (f * b) ≤ M.
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    row_idx = np.repeat(i, n) + np.tile(j, n)          # shape (n*n,)
    col_idx = np.tile(j, n)                           # shape (n*n,)

    # ------------------------------------------------------------------
    # 1) Initial sequence (reuse a previous solution if available)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size < 2:
            raise ValueError
        if init_seq.size > n:
            # take a centred slice
            lo = (init_seq.size - n) // 2
            init_seq = init_seq[lo : lo + n]
    except Exception:
        init_seq = generate_two_sided_power_law(n, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    init_seq = init_seq / init_seq.sum()
    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Broad random search – keep a small elite set
    # ------------------------------------------------------------------
    families = ["power", "exp", "twosided", "beta", "sine", "tri"]
    elite = [(best_score, best_seq.copy())]   # (score, sequence)
    elite_size = 5

    # allocate ~30 % of the total budget for this stage
    rand_deadline = start + 0.3 * budget_s
    while time.time() < rand_deadline:
        fam = random.choice(families)
        if fam == "power":
            p = random.uniform(0.05, 1.5)
            seq = generate_power_law(n, p)
        elif fam == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        elif fam == "twosided":
            p_l = random.uniform(0.05, 1.5)
            p_r = random.uniform(0.05, 1.5)
            w_l = random.uniform(0.2, 5.0)
            w_r = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)
        elif fam == "beta":
            alpha = random.uniform(0.05, 1.5)
            beta = random.uniform(0.05, 1.5)
            seq = generate_beta_shape(n, alpha, beta)
        elif fam == "sine":
            seq = generate_sine(n)
        else:  # "tri"
            seq = generate_triangle(n)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        sc = fast_score(seq)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new global best ({fam}) score={sc:.6f}")

        # maintain elite list
        if len(elite) < elite_size:
            elite.append((sc, seq.copy()))
        else:
            worst = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst][0]:
                elite[worst] = (sc, seq.copy())

    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3) LP‑based refinement of the elite candidates
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[exit] time exhausted after random search – score={final_score:.6f}")
        return final_seq

    # Reserve ~60 % of the remaining budget for LP work, the rest for polishing.
    lp_budget = time_remaining * 0.6
    per_candidate = lp_budget / len(elite)
    t_mix = 0.5        # convex‑combination weight
    max_lp_iters = 12

    for idx, (sc, seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"[lp] elite #{idx + 1} (init score={sc:.6f})")
        eps = 1e-12
        seq = np.maximum(seq, eps)
        seq = seq / seq.sum()
        seq, lp_sc = lp_iterate(seq, t_mix, max_lp_iters,
                                time.time() + per_candidate,
                                row_idx, col_idx)
        if lp_sc < best_score:
            best_score = lp_sc
            best_seq = seq.copy()
            print(f"[lp] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Momentum sub‑gradient polishing (use whatever time is left)
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 0:
        refined_seq, ref_sc = refine_sequence(best_seq, time_remaining, time.time())
        if ref_sc < best_score:
            best_score = ref_sc
            best_seq = refined_seq.copy()
            print(f"[refine] final polish improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final verification with the supplied evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
    print(f"Finished. best reported score = {final_score:.6f}")
    return final_seq
