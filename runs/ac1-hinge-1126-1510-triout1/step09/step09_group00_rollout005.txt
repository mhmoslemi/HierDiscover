def generate_filtered_random(n, rng):
    a = rng.random(n)
    # Apply FFT smoothing: keep few low frequencies.
    ft = np.fft.rfft(a)
    # define decaying mask: ft[0] = ft[0] as is; for higher freq magnitude multiply by factor e^{-alpha * freq}
    alpha = rng.uniform(0.01, 0.5)
    freqs = np.arange(len(ft))
    mask = np.exp(-alpha * freqs)
    ft_s = ft * mask
    a_s = np.fft.irfft(ft_s, n=n)
    a_s = np.maximum(a_s, 0)
    a_s /= a_s.sum()
    return a_s

def adam_optimize(a0, deadline, fft_len_conv, fft_len_g, n):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        # compute subgradient at max indices
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # compute subgradient (like earlier)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # step size schedule: base_lr decays with sqrt(t) maybe
        base_lr = 0.1  # tuned
        step = base_lr / (np.sqrt(t) + 1)

        a_new = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        # projection onto simplex
        a = simplex_projection(a_new)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # optional print

    return best_a, best_score

"""
Advanced search for a non‑negative step sequence that minimizes

    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of several stages (all respecting the global 1000 s
deadline):

0. **Rich parametric sweep** – many analytically‑defined families (sqrt,
   inv‑sqrt, sine‑squared, linear, quadratic, filtered‑random, …) plus
   random convex combinations. The best candidate is kept.

1. **Random‑restart L‑BFGS on a smoothed max** – the true max is
   approximated by a log‑sum‑exp with a decreasing smoothing factor.
   Several restarts and a longer iteration budget increase the chance of
   escaping local minima.

2. **Adam‑style mirror descent on the true max‑convolution** – a
   projected Adam optimiser works directly with the nonsmooth max
   (sub‑gradient via the active set). This often fine‑tunes the solution
   found by L‑BFGS.

3. **Slow‑cooling simulated annealing** – stochastic hill‑climbing that
   moves a random fraction of mass between two (or three) indices;
   temperature decays as a square‑root. The move size is allowed to be
   as large as 50 % of the source mass.

4. **Final evaluation** – the best sequence found is returned as a list
   of floats. ``evaluate_sequence`` will turn it into the official score.

All stages log progress; the final score is printed before returning.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Computes the objective value   2·n·max(a*a)   for a
    sequence that is already normalised (∑a = 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # a lives on the simplex
    a = np.exp(y - np.max(y))
    a /= a.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. the convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Random‑restart L‑BFGS: for each lambda in lam_schedule the
    smoothed objective is minimised. The result of one stage is used as
    the start for the next.
    """
    # use logits to stay on simplex (softmax inside objective)
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:               # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Adam (first‑order) optimiser on the true max‑convolution
    (sub‑gradient) with projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    base_lr = 0.1

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # sub‑gradient for the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)

        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        step = base_lr / (np.sqrt(t_iter) + 1.0)
        a_new = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto the simplex
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len_conv: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that transfers a random fraction of mass
    (up to 50 %) between two (or three) indices. Acceptance follows a
    square‑root cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature – a noticeable fraction of the score
    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # square‑root cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # pick source index (must have positive mass)
        i = rng.integers(n)
        if a[i] <= 0.0:
            continue

        # choose destination(s)
        if rng.random() < 0.5:
            # move to a single destination
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # random fraction up to 0.5
            delta = a[i] * rng.random() * 0.50
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # split mass among two destinations
            j, k = rng.choice(n, size=2, replace=False)
            # ensure i differs from both
            if i == j or i == k:
                continue
            delta = a[i] * rng.random() * 0.30
            if delta <= 0.0:
                continue
            # split delta between j and k
            delta_j = delta * rng.random()
            delta_k = delta - delta_j
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta_j
            a_new[k] += delta_k

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0003, 0.03)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(100.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sqrt_diff":
        a = np.sqrt(i + 1) - np.sqrt(i)
    elif family == "inv_sqrt":
        a = 1.0 / np.sqrt(i + 1)
    elif family == "sin_sq":
        a = np.sin(np.pi * (i + 0.5) / n) ** 2
    elif family == "linear_inc":
        a = i + 1
    elif family == "linear_dec":
        a = n - i
    elif family == "quadratic":
        p = rng.uniform(0.5, 3.0)
        a = (i + 1) ** p
    elif family == "filtered_random":
        # random data filtered heavily in the frequency domain
        raw = rng.random(n)
        ft = np.fft.rfft(raw)
        alpha = rng.uniform(0.01, 0.5)
        freqs = np.arange(len(ft))
        mask = np.exp(-alpha * freqs)
        ft_f = ft * mask
        a = np.fft.irfft(ft_f, n=n)
        a = np.maximum(a, 0.0)
    elif family == "mix":
        # convex combination of two random families
        fam1 = rng.choice(["exp", "gauss", "cos", "logistic", "binary_sparse",
                           "sqrt_diff", "inv_sqrt", "sin_sq"])
        fam2 = rng.choice(["exp", "gauss", "cos", "logistic", "binary_sparse",
                           "sqrt_diff", "inv_sqrt", "sin_sq"])
        a1 = generate_family(fam1, n, rng)
        a2 = generate_family(fam2, n, rng)
        alpha = rng.uniform(0.0, 1.0)
        a = alpha * a1 + (1 - alpha) * a2
    else:
        # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int,
                     fft_len: int,
                     budget_s: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a cheap sweep over many random parametric families
    and return the best vector together with its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.15          # 15 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "sqrt_diff",
        "inv_sqrt",
        "sin_sq",
        "linear_inc",
        "linear_dec",
        "quadratic",
        "filtered_random",
        "mix",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Re‑use a previous construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] parametric sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # safety margin

    # allocate time for stages (percentages of total budget)
    phase1_end = start_time + 0.40 * budget_s
    phase2_end = start_time + 0.65 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    # remaining → annealing

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with finer schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06
    ]
    maxiter_per_lam = 2000                    # a bit more work per stage

    while time.time() < phase1_end:
        # Occasionally jitter the current best to stay in its basin
        if rng.random() < 0.25:
            a_start = best_a * rng.normal(1.0, 0.03, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimiser on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n)
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (kept as a safety net)
    # --------------------------------------------------------------
    # (same code as before but with a slightly larger horizon)
    if time.time() < phase3_end:
        # simple exponentiated‑gradient loop
        a = best_a.copy()
        best_local_a = a.copy()
        best_local_score = fast_score(a, fft_len_conv, n)
        it = 0
        while time.time() < phase3_end:
            # true max‑convolution
            fa = np.fft.fft(a, fft_len_conv)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

            max_val = conv.max()
            eps_val = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps_val).astype(np.float64)

            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len_g)
            fa_rev = np.fft.fft(a_rev, fft_len_g)

            eta = 0.5 / np.sqrt(it + 1.0)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0 or np.isnan(s):
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                print(f"[exp‑grad] new best fast_score={best_local_score:.6f}")

            it += 1

        if best_local_score < best_score:
            best_score, best_a = best_local_score, best_local_a.copy()
            print(f"[phase3] exp‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – long simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[phase4] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
