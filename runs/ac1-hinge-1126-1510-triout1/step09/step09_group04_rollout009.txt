def lbfgs_smooth_optim(a_start, lam_schedule, max_iter_per_lam, deadline, fft_len_conv, fft_len_g, n):
    y = np.log(a_start + 1e-12)  # logits
    best_a = a_start
    best_score = fast_score(best_a, fft_len_conv, n)
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter_per_lam, 'gtol': 1e-12, 'disp': False})
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score

"""
Hybrid multi‑stage optimiser for the step‑function evaluation problem.

Goal
----
Minimise  2·n·max(a*a) / (∑a)²   under a_i ≥ 0.
The algorithm respects the time budget (default 1000 s) and always keeps the
candidate on the probability simplex (∑a = 1, a_i ≥ 0).

Method
------
1. Initialise several diverse candidates (uniform, random, spectral,
   optional previous solution) and keep the best.
2. Perform many short random‑restart L‑BFGS runs on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The smoothing parameter
   ``lam`` is decreased gradually so that the optimiser approaches the
   true objective.
3. Refine the current best with a long L‑BFGS run using a finer ``lam``
   schedule.
4. Run exponentiated‑gradient descent on the *exact* max‑convolution
   (sub‑gradient of the hard max) while staying on the simplex.
5. Aggressive deterministic hill‑climb that repeatedly shifts a small
   fraction of mass from the coordinate that contributes most to the
   current maximal convolution entry towards a coordinate with the smallest
   partner value.
6. Final projected sub‑gradient descent with a back‑tracking line‑search
   for a few more seconds.
7. Return the best sequence found; a final call to the supplied
   ``evaluate_sequence`` checks validity.

The code is fully self‑contained, uses only NumPy / SciPy and prints
progress information.  It can be invoked directly as ``propose_candidate()``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen for a non‑empty vector, but guard anyway.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp smooth approximation of max(a*a) and its gradient w.r.t. logits."""
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # back‑prop through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed surrogate
# ----------------------------------------------------------------------


def lbfgs_smooth_optim(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS on a cascade of smoothing parameters.
    Returns the best vector (on the simplex) seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for softmax
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as exc:  # safeguard – continue with next lam
            print("[lbfgs] exception:", exc)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # classic 1/√t schedule
        eta = 1.0 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic fine‑grained hill‑climb
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  The step size is geometrically reduced.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: smallest partner value (skip i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size for next iteration
        else:
            step = max(step * 0.8, min_step)
    return a, score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard max) with back‑tracking line‑search
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len_conv: int,
                                   fft_len_g: int):
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        # convolution and maximal entry
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # initialise step size (decreasing like 1/√t)
        eta = 1.0 / np.sqrt(it + 1.0)

        improved = False
        # simple back‑tracking line‑search
        for _ in range(8):
            a_candidate = a - eta * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            eta *= 0.5

        if not improved:
            # if we could not improve, fall back to the current best
            a = best_a.copy()
        it += 1
    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    The routine follows the multi‑stage scheme described in the module
    docstring.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise from a previously known construction if available
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Interpolate to required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Gather a few diverse seeds
    # --------------------------------------------------------------
    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)
    candidates = [a0, a_spec, a_rand]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # Rough phase split (percent of total budget)
    phase1_deadline = start_time + 0.20 * budget_s   # many short L‑BFGS runs
    phase2_deadline = start_time + 0.45 * budget_s   # refined long L‑BFGS
    phase3_deadline = start_time + 0.70 * budget_s   # exponentiated‑gradient
    phase4_deadline = start_time + 0.85 * budget_s   # deterministic hill‑climb
    phase5_deadline = deadline                       # projected sub‑gradient

    # --------------------------------------------------------------
    # Phase 1 – random‑restart coarse L‑BFGS
    # --------------------------------------------------------------
    lam_schedule_coarse = [
        0.6, 0.45, 0.35, 0.25, 0.18,
        0.12, 0.08, 0.05, 0.03, 0.018,
        0.01, 0.006, 0.003, 0.001, 0.0005
    ]
    maxiter_coarse = 400

    while time.time() < phase1_deadline:
        # pick a start point
        choice = rng.random()
        if choice < 0.5:
            a_start = random_initial(n, rng)
        elif choice < 0.85:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.01), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_smooth_optim(
            a_start,
            lam_schedule_coarse,
            maxiter_coarse,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – refined (long‑run) L‑BFGS
    # --------------------------------------------------------------
    lam_schedule_fine = [
        0.5, 0.4, 0.3, 0.25, 0.2,
        0.15, 0.12, 0.10, 0.08, 0.06,
        0.04, 0.03, 0.02, 0.015, 0.01,
        0.008, 0.006, 0.004, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 0.00005, 0.00002
    ]
    maxiter_fine = 2000

    best_a, best_score = lbfgs_smooth_optim(
        best_a,
        lam_schedule_fine,
        maxiter_fine,
        phase2_deadline,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[Phase‑2] refined L‑BFGS score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (hard max)
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Phase‑3] exp‑gradient score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – fine deterministic hill‑climb (two passes)
    # --------------------------------------------------------------
    best_a, best_score = fine_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase4_deadline,
        init_step=0.20,
        min_step=1e-5,
    )
    print(f"[Phase‑4] hill‑climb pass‑1 score={best_score:.6f}")

    # second, finer pass
    best_a, best_score = fine_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase4_deadline,
        init_step=0.05,
        min_step=1e-7,
    )
    print(f"[Phase‑4] hill‑climb pass‑2 score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – projected sub‑gradient descent with line‑search
    # --------------------------------------------------------------
    if time.time() < phase5_deadline:
        best_a, best_score = projected_subgradient_optimize(
            best_a,
            phase5_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[Phase‑5] projected sub‑gradient score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
