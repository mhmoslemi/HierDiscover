"""
A practical optimizer for the step‑function evaluation problem.

The code builds on the four‑phase strategy previously used, but adds:
* a deterministic “central‑binomial’’ start (the coefficients of (1‑z)^‑½)
* longer optimisation schedules (more L‑BFGS/Adam/exp‑grad steps)
* a final tiny stochastic local‑search to polish the result.

All phases respect the 1000 s budget and keep the vector on the
probability simplex, so the final score is simply 2·n·max(conv(a,a)).
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'double_power']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # double_power – heavy edges, light centre
        beta = rng.uniform(0.5, 2.5)
        a_left = (np.arange(n // 2) + 1.0) ** (-beta)
        a_right = a_left[::-1]
        a = np.concatenate([a_left, a_right])
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Deterministic “central‑binomial’’ initialisation
# ----------------------------------------------------------------------


def central_initial(n: int) -> np.ndarray:
    """
    Coefficients a_i solving a*a = 1 for k = 0..n‑1,
    i.e. a_i = (1‑∑_{j=1}^{i‑1} a_j a_{i‑j}) / 2.
    The resulting vector is normalised to sum to 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # s = Σ_{j=1}^{i‑1} a_j a_{i‑j}
        s = np.dot(a[1:i], a[i - 1:0:-1]) if i > 1 else 0.0
        a[i] = (1.0 - s) / 2.0
        if a[i] < 0.0:                     # numerical safeguard
            a[i] = 0.0
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t. a_i (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source index
    to a destination index.  Convolution updates are performed incrementally.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)

    accepted = 0
    it = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger entry to smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic polishing (adds/removes a few percent of mass)
# ----------------------------------------------------------------------


def final_polish(a: np.ndarray,
                 best_score: float,
                 fft_len: int,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 attempts: int = 30):
    """
    Very small random perturbations that are kept only if they improve the score.
    """
    cur_a = a.copy()
    cur_score = best_score
    while time.time() < deadline:
        improved = False
        for _ in range(attempts):
            i = rng.integers(0, n)
            delta = cur_a[i] * (rng.random() - 0.5) * 0.04   # up to ±2 %
            if cur_a[i] + delta <= 0.0:
                continue
            cand = cur_a.copy()
            cand[i] += delta
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len)
            if cand_score < cur_score:
                cur_a, cur_score = cand, cand_score
                improved = True
                break
        if not improved:
            break
    return cur_a, cur_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 entries is a practical trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – try a deterministic central‑binomial start and a random start.
    # ------------------------------------------------------------------
    a_central = central_initial(n)
    a_random = random_initial(n, rng)

    best_a = a_central if fast_score(a_central, fft_len) < fast_score(a_random, fft_len) else a_random
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    phase1_end = start_time + 0.20 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # Adam (smoothed)
    phase3_end = start_time + 0.85 * budget_s   # Exp‑grad
    # remaining ⇒ hill‑climb + polish

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with a richer λ schedule.
    # ------------------------------------------------------------------
    lam_schedule = [
        0.4, 0.25, 0.15, 0.09, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003,
        0.0015, 0.0008, 0.0004, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        if rng.random() < 0.2:                         # jittered version
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smooth max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)

        lam_schedule_adam = [
            0.35, 0.20, 0.12, 0.07, 0.04,
            0.02, 0.01, 0.006, 0.003, 0.0015,
            0.0008, 0.0004, 0.0002, 0.0001
        ]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam,
            max_steps_per_lam=2500,
            deadline=phase2_end,
            n=n, fft_len=fft_len, lr0=0.13
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max.
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng,
            eta0=15.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy hill‑climbing (more aggressive, then refined).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # aggressive first pass
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline * 0.90,                 # keep a margin for polishing
            rng,
            max_fraction=0.22,
            recompute_every=800,
        )
        print(f"[hill‑climb] after aggressive pass score={best_score:.6f}")

        # final fine‑grained polishing
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.07,
            recompute_every=500,
        )
        print(f"[hill‑climb] after fine pass score={best_score:.6f}")

        # tiny stochastic polish
        best_a, best_score = final_polish(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            attempts=30
        )
        print(f"[polish] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
