"""
propose_candidate
=================

Search for a non‑negative step sequence that minimises

    score = 2·n·max(conv(a, a)) / (sum(a)²)

The algorithm mixes several optimisation ideas:

1. **Diverse initial pool** – uniform, spectral, triangular, Hann,
   Gaussian, ``arc‑sine`` (U‑shaped) and a few random draws.
2. **Smoothed‑max L‑BFGS** – minimise a log‑sum‑exp approximation of the
   max‑convolution.  A long schedule of decreasing smoothing parameters
   is used and many restarts are performed.
3. **Exponentiated‑gradient descent** on the exact max‑convolution,
   which keeps the vector on the probability simplex automatically.
4. **Projected sub‑gradient** with a diminishing step‑size.
5. **Simulated annealing** – large random mass transfers are accepted
   according to a temperature schedule.
6. **Targeted local move** – a greedy hill‑climb that moves a tiny
   fraction of mass from the indices that contribute most to the
   current peak to those that contribute least.
7. **Final polish** – a brief extra L‑BFGS run with an almost
   non‑smoothed objective.

All phases respect the global time budget (default 1000 s).  The best
vector found (normalised to sum = 1) is returned as a plain list of
floats; the external ``evaluate_sequence`` function will re‑scale it
as required.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (assumes Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a ( Σa = 1 )."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed‑max objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (linear, extracted from circular FFT)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of √(FFT of a uniform target) projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mix of simple stochastic families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Symmetric triangle, linearly rising then falling."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window (non‑negative, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    U‑shaped “arcsine” distribution:
        a_i ∝ 1 / √[(i+½)(n‑i‑½)]
    This mimics the continuous arcsine law on a discrete grid.
    """
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + eps)
    w[w < 0] = 0.0
    a = w / w.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS on the smoothed‑max objective.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of hard max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule – a bit slower decay than 1/√t
        eta = 0.5 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """
    Projected sub‑gradient descent with a diminishing step‑size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    base_step = 0.5

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # zero‑mean sub‑gradient (keeps the sum constant)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = base_step / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Simple SA: transfer a random fraction of mass between two indices.
    Accept worsening moves with probability exp(-Δ/T).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.15   # up to 15 % of source mass
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        score_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if score_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(score_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def targeted_mass_move(a: np.ndarray,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves a tiny fraction of mass
    from the index that contributes most to the current peak to the
    index that contributes least.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    for _ in range(max_iter):
        # full convolution
        fa = np.fft.rfft(best, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # contributions to the peak (ordered pair i, k_max-i)
        contrib = np.zeros(n)
        # vectorised: for i where k_max-i in range
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1)
        jdx = k_max - idx
        contrib[idx] = best[idx] * best[jdx]

        # pick source (largest contribution) and sink (smallest)
        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))

        if src == dst:
            # fallback random move
            src, dst = rng.integers(0, n, size=2)
            if src == dst:
                continue

        delta = best[src] * rng.random() * 0.08   # ≤ 8 % of source mass
        if delta <= 0.0:
            continue

        candidate = best.copy()
        candidate[src] -= delta
        candidate[dst] += delta

        score_cand = fast_score(candidate, fft_len, n)
        if score_cand < best_score:
            best = candidate
            best_score = score_cand
        else:
            # reject – keep the old vector
            pass
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)   # legacy safety

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                     # trade‑off between quality & speed
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a rich initial pool
    # --------------------------------------------------------------
    candidates = []

    # previous best (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(arcsine_init(n))

    # a few random draws
    for _ in range(4):
        candidates.append(random_initial(n, rng))

    # pick the best initial vector
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS (smoothed max)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.18 * budget_s
    # extended smoothing schedule (down to ~1e‑7 of max_c)
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06,
        1e-06, 5e-07, 2e-07, 1e-07
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_deadline:
        # diversify start points: mix current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.5
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (long)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase3_deadline,
                                                  n,
                                                  fft_len)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing (exploration)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=3e-4,
                                                 final_temp=5e-8,
                                                 max_moves=200000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 5 – Targeted mass‑move hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_mass_move(best_a,
                                                n,
                                                fft_len,
                                                rng,
                                                max_iter=30000)
        print(f"[targeted hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Optional final L‑BFGS polish (almost exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # one extra short L‑BFGS pass with a very tiny lambda
        extra_lam = [1e-08]
        best_a, _ = lbfgs_continuation(best_a,
                                       extra_lam,
                                       maxiter_per_lam=1200,
                                       deadline=deadline,
                                       fft_len_conv=fft_len,
                                       fft_len_g=fft_len,
                                       n=n)
        # recompute score after the last polish
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final L‑BFGS] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return result in the format expected by evaluate_sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
