def simulated_annealing_optimize(a_start, deadline, n, fft_len_conv, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    current_score = best_score
    # parameters
    # initial temperature set to maybe 0.01 (objective scale)
    # schedule: temperature = initial * ((deadline - now) / total)
    total_time = deadline - time.time()
    while time.time() < deadline:
        # compute temperature
        t_rem = deadline - time.time()
        temp = max(1e-8, t_rem / total_time)  # linear decreasing
        # choose random source and target
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # step size as proportion of a_i, e.g., delta = a[i] * (0.5 * (temp + 0.01))? Or random.

        a_new = a.copy()

        # ensure non-negative

        new_score = fast_score(a_new, fft_len_conv, n)

"""
Hybrid optimizer for the step‑function bound.
------------------------------------------------
The algorithm follows the successful pipeline from the
state‑of‑the‑art (coarse L‑BFGS, refined L‑BFGS,
exponentiated‑gradient, deterministic hill‑climb) but
adds two extra phases:

* A longer “fine” L‑BFGS sweep with a much finer
  λ‑schedule and more iterations.
* A simple simulated‑annealing stage that can
  escape tiny local minima that the gradient methods
  may miss, followed by a brief random‑perturbation
  local‑search loop.

All stages respect the 1000 s wall‑time budget and
work on the probability simplex (∑a = 1, a ≥ 0).  The
final sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomised spectral initialiser (random phase)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    # flat target spectrum
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)

    # random phase
    phase = rng.uniform(-np.pi, np.pi, size=amp.shape)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    A = sqrt_amp * np.exp(1j * phase)

    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of self‑convolution via log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y (a=softmax(y)).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation (coarse / refined)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS through a sequence of smoothing parameters.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent via exponentiated‑gradient
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Keeps a on the simplex automatically and uses a
    1/√t step‑size schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size (classic 1/√t schedule)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Simulated annealing optimisation (non‑gradient)
# ----------------------------------------------------------------------


def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len_conv: int,
                                 rng: np.random.Generator):
    """
    Simple annealing: repeatedly move a small amount of mass from a
    random source index to a random target index.  Accept worsening
    moves with probability exp(-Δ/T).  The temperature T decreases
    linearly from T0 to 0.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    cur_score = best_score

    # temperature schedule
    T0 = 0.02          # initial temperature (empirical)
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start
        T = max(1e-9, T0 * (1.0 - elapsed / total))

        # pick source and destination
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # max transferable mass (keep a[i] non‑negative)
        max_delta = min(a[i], 0.05)    # limit step size to 5 % of a[i]
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # (sum stays 1, non‑negativity guaranteed)

        new_score = fast_score(a_new, fft_len_conv, n)
        Δ = new_score - cur_score
        if Δ < 0.0 or rng.random() < np.exp(-Δ / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[SA] new best={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic fine hill‑climb
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with
    the smallest partner value.  Step size shrinks geometrically until
    min_step.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1)
        partner[idx_range] = a[max_idx - idx_range]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: smallest partner value (avoid picking i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renormalise (should stay 1 but guard against round‑off)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.8, min_step)
    return a, score

        a_pert = add_noise(best_a, sigma=rng.uniform(0.001, 0.008), rng=rng)
        # quick hill‑climb for a few iterations only
        sub_deadline = time.time() + 0.5  # half‑second dedicated
        a_pert, sc_pert = fine_hill_climb(
            a_pert,
            fast_score(a_pert, fft_len_conv, n),
            fft_len_conv,
            n,
            sub_deadline,
            init_step=0.15,
            min_step=5e-6,
        )
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"[Phase‑5‑perturb] improved score={best_score:.6f}")

    # Final deterministic hill‑climb on the current best vector
    best_a, best_score = fine_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase5_deadline,
        init_step=0.20,
        min_step=1e-5,
    )
    print(f"[Phase‑5] final fine hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
