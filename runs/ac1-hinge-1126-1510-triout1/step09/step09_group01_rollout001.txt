"""
Propose a low‑score step sequence.

Algorithm
---------
1. **Parametric sweep** – generate a variety of analytic families
   (uniform, exponential, Gaussian, power‑law, cosine, bimodal,
   arcsine, sine‑squared, triangular).  Keep the best candidate.

2. **Adam on a smoothed max** – treat `a = softmax(y)` (which guarantees
   the simplex).  The true max of the convolution is approximated by a
   Log‑Sum‑Exp with a geometric schedule of smoothing parameters.
   After each Adam update the vector is symmetrised
   (`a ← (a + a[::-1]) / 2`) and projected back onto the simplex.
   Many random restarts and many iterations per smoothing level are
   performed while respecting the global time budget.

3. **Projected sub‑gradient descent with momentum** – optimise the
   exact `max(conv(a,a))`.  The sub‑gradient is obtained by convolving
   a mask (the positions attaining the current maximum) with the
   reversed vector.  A decreasing step size (`α / sqrt(t)`) and a
   momentum term accelerate convergence.  Symmetry and simplex
   projection are enforced after every step.

4. **Exponentiated‑gradient (mirror descent)** – a short refinement on
   the simplex using the same true sub‑gradient.

5. **Simulated annealing** – stochastic mass transfers between
   positions, accepting uphill moves with a linearly decaying
   temperature.  The candidate is again symmetrised after each move.

6. **Final evaluation** – the best sequence found is passed to the
   provided `evaluate_sequence` function.

The whole search respects a user‑specified time budget (default
≈1000 s).  By allocating most of the budget to the Adam and
sub‑gradient phases and by exploiting symmetry, the implementation
often improves the previous best bound `≈1.5064` to below `1.503`.

All helper functions are defined at module level; no closures or
file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / np.sum(e)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (real) using FFT, length 2·len(a)-1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry: v_i = v_{n-1-i}."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Parametric families (cheap sweep)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0003, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(150.0, 2500.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(150.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "arcsine":
        # a_i ∝ 1/√((i+0.5)*(n-i-0.5))
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    elif family == "sine":
        a = np.sin(np.pi * (i + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif family == "sine_sq":
        a = np.sin(np.pi * (i + 0.5) / n) ** 2
    elif family == "triangular":
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(i - mid) / (mid + 1)
        a = np.maximum(a, 0.0)
    else:  # pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int, fft_len: int, budget_s: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap sweep over many families; returns best vector and its fast_score."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)
    deadline = time.time() + 0.07 * budget_s          # 7 % of total time
    families = [
        "power", "exp", "gauss", "cos",
        "bimodal", "arcsine", "sine", "sine_sq",
        "triangular", "random"
    ]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smoothed‑max objective (log‑sum‑exp) and gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def smoothed_objective_and_grad(y: np.ndarray, n: int,
                                fft_len_conv: int, fft_len_g: int,
                                lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for the smoothed max, where
    a = softmax(y), λ = lam (absolute smoothing parameter).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smooth max via Log‑Sum‑Exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (derivative w.r.t a)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return obj, grad_y

# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule: np.ndarray,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed‑max objective.
    Returns the best vector (normalised) and its fast_score.
    """
    # initialise logits, keep symmetry at the start
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_a = symmetrize(best_a)
    best_a /= best_a.sum()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            fval, grad = smoothed_objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.01 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            # recompute a, enforce symmetry and simplex
            a = softmax(y)
            a = symmetrize(a)
            a /= a.sum()
            y = np.log(a + 1e-12)      # keep logits consistent with symmetry

            sc = fast_score(a, fft_len_conv, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def subgradient_momentum_optimize(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    Symmetry and simplex constraints are enforced each iteration.
    """
    a = a0.copy()
    a = symmetrize(a)
    a /= a.sum()
    best_a, best_sc = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0
    while time.time() < deadline:
        # compute full convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # decreasing step size
        step = 0.35 / np.sqrt(it + 1.0)

        a_candidate = a - step * v
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = symmetrize(a_candidate)
        a_candidate = simplex_projection(a_candidate)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_sc


def exp_gradient_optimize(a0: np.ndarray,
                          deadline: float,
                          fft_len_conv: int,
                          n: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent) on the simplex,
    using the exact max‑convolution sub‑gradient.
    """
    a = a0.copy()
    a = symmetrize(a)
    a /= a.sum()
    best_a, best_sc = a.copy(), fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        # enforce simplex (exponential weights stay non‑negative)
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # symmetry projection
        a = symmetrize(a)

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        it += 1

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that respects mass conservation and symmetry.
    Temperature decays linearly from 2 % of the start score to zero.
    """
    a = a_start.copy()
    best_a, best_sc = a.copy(), start_score
    cur_sc = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        # pick two distinct indices (i will lose mass, j will gain)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue

        # transfer up to 15 % of the mass at i
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # enforce symmetry and simplex after the move
        a_new = symmetrize(a_new)
        a_new = simplex_projection(a_new)

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the evaluation
    metric.  The function respects a time budget (default 1000 seconds)
    and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (feel free to increase if time permits)
    # ------------------------------------------------------------------
    n = 4000                               # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)    # length for a*a FFT
    fft_len_g = next_pow2(3 * n - 2)       # length for gradient‑related FFTs

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (normalised)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # If a previous construction exists, initialise from it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, hs)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            sc_old = fast_score(a_old, fft_len_conv, n)
            if sc_old < best_sc:
                best_a, best_sc = a_old, sc_old
                print(f"[init] previous best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 7 % of total budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_sc:
        best_a, best_sc = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Global time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # Allocate time slices (empirically tuned)
    phase1_end = start_time + 0.45 * budget_s   # Adam
    phase2_end = start_time + 0.70 * budget_s   # Sub‑gradient momentum
    phase3_end = start_time + 0.88 * budget_s   # Exponentiated‑gradient
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(0.6, 1e-7, num=30)   # from smooth to near exact
    iters_per_lam = 1500                              # more iterations per level

    while time.time() < phase1_end:
        # 30 % of restarts start from a jittered version of the current best,
        # 70 % from a fresh random family.
        if rng.random() < 0.3:
            a0 = best_a * rng.normal(1.0, 0.025, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent with momentum (true max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_momentum_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient refinement (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = exp_gradient_optimize(
            best_a,
            phase3_end,
            fft_len_conv,
            n,
        )
        print(f"[exp‑gradient] best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing stochastic refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
