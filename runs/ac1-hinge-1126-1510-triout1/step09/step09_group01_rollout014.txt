"""
Step‐sequence optimizer for the metric

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a multi‑phase search that
   1. explores several parametric families,
   2. runs many Adam optimisations on a smoothed “log‑sum‑exp” surrogate,
   3. refines with projected sub‑gradient descent (momentum),
   4. performs mirror descent (exponentiated gradient),
   5. optional LP‑based “capacity‑filling’’ refinement (solves a linear program
      to push mass into under‑used convolution slots),
   6. finishes with a short simulated‑annealing hill‑climb.

All phases respect a global 1000 s deadline.  The best sequence found
is returned as a plain list of floats (already normalised to sum = 1, so
the final call to `evaluate_sequence` reproduces the true score).
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv(a,a)) under the assumption Σa=1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max surrogate (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_grad: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv entries (weights of the smooth max) ---------
    w = exp_shift / Z  # d smooth_max / d conv_i

    # ---- back‑propagate through convolution -------------------------------
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_grad)
    fa_rev = np.fft.fft(a_rev, fft_len_grad)
    conv_rev = np.real(np.fft.ifft(fw * fa_rev))

    # slice that corresponds to the original vector length
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- back‑propagate through softmax ----------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    lam_factors,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_grad: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max surrogate.
    Returns the best vector found (sum = 1) and its fast_score.
    """
    # initialise logits – softmax guarantees simplex feasibility
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_grad, lam)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            step = 0.05 / (np.sqrt(v_hat) + eps)  # a slightly larger base step
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_grad: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the *true* max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # ---- true convolution ------------------------------------------------
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- sub‑gradient of the max (linear w.r.t. a) -----------------------
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_grad)
        fa_rev = np.fft.fft(a_rev, fft_len_grad)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ---- momentum & projection -------------------------------------------
        v = beta * v + (1.0 - beta) * g
        step = 1.0 / np.sqrt(it + 1.0)          # more aggressive than before
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the simplex.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build sparse matrix A (size (2n‑1)×n) such that A @ b = conv(a,b).
    Each column j of A is a shifted copy of a.
    """
    n = a.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(2 * n - 1 - s, a[s]) for s in range(n)]
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j   subject to   a * b ≤ M   and   b ≥ 0.
    Returns the optimiser b (may be all‑zero on failure).
    """
    n = a.shape[0]
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise -Σb  →  maximise Σb
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        # fallback – return the current a (keeps the pipeline alive)
        print("[LP] linprog failed:", res.message)
        return a.copy()
    b = np.maximum(res.x, 0.0)
    return b


def lp_refine(
    a_start: np.ndarray,
    fft_len_conv: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    One‑step capacity‑filling refinement:
        solve a linear programme for g,
        mix g into the current a,
        keep the better of the two.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_score = fast_score(a, fft_len_conv, n)

    while time.time() < deadline:
        # current bound M
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # LP to push as much mass as possible without exceeding M
        b = solve_lp_for_g(a, M)
        if b.sum() == 0:
            break
        b_norm = b / b.sum()

        # convex combination (t = 0.5 works well in practice)
        t = 0.5
        a_new = (1 - t) * a + t * b_norm
        a_new = simplex_projection(a_new)

        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score - 1e-9:
            best_score, a = new_score, a_new
            print(f"[LP] improvement: fast_score={best_score:.6f}")
        else:
            # No progress → stop
            break

    return a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector drawn from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "triangular":
        # simple triangular hat
        a = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                            np.linspace(1, 0, n - n // 2)])
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap random exploration of several analytic families.
    Returns the best vector found and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + 0.15 * budget_s
    families = ["power", "exp", "gauss", "cos", "bimodal", "triangular", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step sequence.
    Returns the best list of floats before the global deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (feel free to adjust)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_grad = next_pow2(3 * n - 2)      # for gradient wrt y

    # ------------------------------------------------------------------
    # Baseline – uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 = a1 / a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # Phase split (percentages of total budget)
    phase0_end = start_time + 0.08 * budget_s
    phase1_end = start_time + 0.48 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    phase4_end = start_time + 0.95 * budget_s
    # remaining time → optional simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 8 % budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (≈ 40 % budget)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.7, 2e-5, num=30)      # finer schedule

        else:

            a_start,

            phase1_end,

        def simulated_annealing(a_start, start_score, fft_len, n, deadline, rng):
            a = a_start.copy()
            best_a_local, best_score_local = a.copy(), start_score
            cur_score = start_score

            T0 = 0.02 * start_score
            t0 = time.time()
            total_time = max(deadline - t0, 1e-6)

            while time.time() < deadline:
                remain = max(deadline - time.time(), 0.0)
                T = T0 * (remain / total_time)

                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                if a[i] == 0.0:
                    continue
                delta = a[i] * rng.random() * 0.15
                if delta <= 0.0:
                    continue

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

                new_score = fast_score(a_new, fft_len, n)
                diff = new_score - cur_score
                if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
                    a = a_new
                    cur_score = new_score
                    if new_score < best_score_local:
                        best_score_local, best_a_local = new_score, a_new.copy()
            return best_a_local, best_score_local

        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, deadline, rng
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
