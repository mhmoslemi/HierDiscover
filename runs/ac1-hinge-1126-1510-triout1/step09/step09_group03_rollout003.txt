def projected_subgradient_optimize(a_start, deadline, n, fft_len, rng, eta0=0.2):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    no_improve_cnt = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps= max_val*1e-12+1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = 2 * np.real(np.fft.ifft(fm * fa_rev))[n-1:n-1+n]
        # update
        eta = eta0 / np.sqrt(it+1)
        a_new = a - eta * grad
        # project onto simplex
        a_new = simplex_projection(a_new)
        # compute new score
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve_cnt = 0
        else:
            no_improve_cnt += 1
        a = a_new
        it += 1
        # occasional random perturb
        if no_improve_cnt > 1000:
            a = add_noise(a, sigma=0.005, rng=rng)
            no_improve_cnt = 0
    return best_a, best_score
```

"""
step‑function optimizer (≈ 1 000 s budget)

Goal: minimise   2·n·max(conv(a,a)) / (sum(a)**2)
               = 2·n·max(conv)   when sum(a)=1

Algorithm (four phases, all staying on the probability simplex)

1. **Smoothed L‑BFGS** – minimise a log‑sum‑exp smooth approximation of the
   max‑convolution.  A cascade of decreasing smoothing parameters (λ) is
   performed and several random restarts are used.

2. **Projected sub‑gradient descent** – true (hard) max‑convolution is used.
   The sub‑gradient g = 2·(a_rev * mask) is computed via FFT and the
   update a←a−ηg is projected back onto the simplex (Euclidean projection).

3. **Deterministic greedy descent** – repeatedly move a fraction of the mass
   from the “most harmful’’ index (largest gradient) to the “most helpful’’
   index (smallest gradient).  The maximal safe step δ is computed analytically
   so that the peak of the convolution never rises.

4. **Simulated‑annealing hill‑climb** – random mass transfers are accepted
   with a Metropolis criterion while a linear temperature schedule cools.
   The convolution is updated incrementally in O(n) time.

The length of the step‑function is taken larger (n=6000) than in the
reference implementation; a larger search space gives a better chance to
lower the ℓ∞‑norm of the self‑convolution.  All phases respect the global
time budget (default 1000 s) and the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation assuming sum(a)=1:
    score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Phase‑1 : smoothed L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the lam‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam, 'gtol': 1e-12,
                         'disp': False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 0.25,
):
    """Sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    no_improve = 0
    while time.time() < deadline:
        # convolution and the max‑mask
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # sub‑gradient step and simplex projection
        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        # evaluate
        score = fast_score(a_new, fft_len)

        # keep best
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional random shake‑up if stuck
        if no_improve >= 2000:
            a_new = add_noise(a_new, sigma=0.006, rng=rng)
            no_improve = 0

        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : deterministic greedy descent using analytical safe step size
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Move mass from the index with largest gradient to the smallest,
    using the maximal analytically safe δ that never raises the current peak."""
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # admissible δ to keep non‑negativity
        delta_src = a[src]

        # bound from entries that would increase beyond the current max
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92   # safety factor
        if delta <= 0.0:
            break

        # apply move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to keep numeric drift small
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # linear cooling: start temperature a modest fraction of the best score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()

    max_fraction = 0.18   # maximal fraction of src mass we may move
    recompute_every = 2500
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from larger entry to smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        # periodic exact recompute of the full convolution
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 6000 gives a richer search space while staying fast
    # ------------------------------------------------------------------
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if possible
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the new length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.12 * budget_s    # short but thorough L‑BFGS
    phase2_end = start_time + 0.65 * budget_s    # bulk of the time: sub‑gradients
    phase3_end = start_time + 0.90 * budget_s    # deterministic greedy
    # remaining → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS with several restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # With probability 0.3 perturb the current best, otherwise fresh start
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = projected_subgradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=0.25,
        )
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(
            best_a, phase3_end, n, fft_len,
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()['height_sequence_1'] = final_seq

    return final_seq
