"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer is a four‑phase pipeline tuned for the 1000 s budget.
Compared with the reference implementation it uses

* many more L‑BFGS iterations per smoothing level,
* several independent restarts in the heavy‑L‑BFGS phase,
* a slower, higher‑temperature annealing phase,
* a final “order‑fix” step that forces the sequence to be monotone
  (the reverse‑dot product – the dominant term of the max‑convolution –
  is then provably minimised by the rearrangement inequality).

All helper functions are top‑level; the only global state used is the
optional ``height_sequence_1`` from a previous run.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a, a)) for a vector that already
    satisfies sum(a)=1.  Uses FFT for the full linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "exp", "power", "bimodal", "edge", "random"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.015)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.1, 1.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        frac = rng.uniform(0.005, 0.02)
        m = max(1, int(n * frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.06 * n))
        peak = rng.uniform(0.6, 1.4) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def order_monotone(a: np.ndarray) -> np.ndarray:
    """
    Re‑order a vector to be monotone decreasing.
    This minimises the reverse‑dot product term a·a_rev, which is the
    dominant contributor to the max‑convolution.
    """
    a_sorted = np.sort(a)[::-1]
    a_sorted /= a_sorted.sum()
    return a_sorted


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns (value, gradient) suitable for L‑BFGS.
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution (a_rev * w) -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Run L‑BFGS for each λ in lam_schedule, always starting from the
    previous result.  Returns the best vector seen and its fast_score.
    """
    # initialise logits for the softmax parametrisation
    y = np.log(a_start + 1e-15)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        # recover a from logits
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) for the exact max‑convolution a*a.
    The sub‑gradient corresponds to the sum of contributions from all
    shifts attaining the maximal value (within numerical tolerance).
    """
    n = a.shape[0]
    # full convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam on the exact objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive‑momentum) using the true sub‑gradient.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)          # slowly shrinking step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (guarantees sum=1, non‑negative)
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a += rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb phase
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a simple simulated‑annealing rule.
    Moves a fraction of mass from high‑gradient indices to low‑gradient
    ones; also tries random swaps.  Temperature starts relatively high
    (≈ 10 % of the current score) and decays slowly.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule
    temperature = max(0.1 * cur_score, 1e-4)   # start a bit higher than before
    decay = 0.99995
    min_temp = 1e-9

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact sub‑gradient for the current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source / target indices using percentiles
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue
            frac = rng.uniform(0.08, 0.25)          # transfer 8‑25 % of source mass
            delta = cur_a[i] * frac
            if delta <= 0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # random swap (helps re‑ordering)
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # re‑normalise
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        # acceptance rule
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        # temperature decay
        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=600_000,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS (very low smoothing)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = np.logspace(np.log10(5e-6), np.log10(5e-8), num=8)[::-1]
        fine_maxiter = 1000
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final monotone‑ordering check (often improves the final bound)
    # ------------------------------------------------------------------
    ordered = order_monotone(best_a)
    ordered_score = fast_score(ordered, fft_len_conv, n)
    if ordered_score < best_score:
        best_a = ordered
        best_score = ordered_score
        print(f"[Final order] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose globally for possible warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
