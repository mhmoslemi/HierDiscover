"""
Proposes a non‑negative step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimisation pipeline:

1. Initialise from the best known construction (global ``height_sequence_1``)
   by interpolation to the working size ``n`` (default 4000).

2. A short L‑BFGS phase on a log‑sum‑exp smoothed objective
   (softmax‑parametrised probabilities).  A few random restarts give a
   good basin on the simplex.

3. Exponentiated‑gradient descent on the *true* max‑convolution.
   This works directly on the simplex and very quickly reduces the
   evaluation score.

4. A long simulated‑annealing hill‑climb.  Each iteration moves a small
   fraction of mass from one index to another, updates the convolution
   incrementally (O(n) work) and accepts worsening moves with a
   temperature that slowly cools.  The move size also shrinks with time.
   Periodic full recomputation of the convolution eliminates numerical drift.

5. (Optional) Symmetrisation of the best solution – averaging with the
   reversed sequence – followed by a brief final EG fine‑tune.

All phases respect the time‑budget (default 1000 s).  The final
sequence is returned as a plain list of floats; its true evaluation is
printed using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation: 2·n·max(conv(a,a))  (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Parameters
    ----------
    y : ndarray
        Logits; a = softmax(y) is the probability vector.
    n : int
        Length of a.
    fft_len : int
        Length of FFT (power of two ≥ 2·n‑1).
    lam_factor : float
        Controls the smoothing temperature (relative to the current max).
    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : ndarray
        Gradient of f w.r.t. y (same shape as y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t. a_i (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(a_start: np.ndarray,
                                   init_score: float,
                                   fft_len: int,
                                   n: int,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   max_fraction_start: float = 0.18,
                                   max_fraction_end: float = 0.01,
                                   init_temp_factor: float = 0.005,
                                   final_temp_factor: float = 1e-6,
                                   recompute_every: int = 5000):
    """
    Stochastic hill‑climb with simulated‑annealing acceptance.
    Moves a random fraction of mass from a source index to a destination
    index, updating the convolution incrementally.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)          # exact convolution (length 2n‑1)
    best_a = a.copy()
    best_score = init_score
    cur_score = init_score

    # temperature schedule (relative to the current score)
    T0 = max(init_temp_factor * init_score, 1e-12)
    Tf = max(final_temp_factor * init_score, 1e-15)

    start = time.time()
    it = 0
    accepted = 0

    diff = np.empty_like(conv)

    while time.time() < deadline:
        # progress based schedules
        elapsed = time.time() - start
        total = deadline - start
        prog = min(elapsed / total, 1.0)

        # linear cooling
        T = T0 * (1.0 - prog) + Tf * prog
        # linear shrink of move size
        max_frac = max_fraction_start * (1.0 - prog) + max_fraction_end * prog

        # pick distinct indices
        src = rng.integers(0, n)
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1
        if src == dst:
            continue
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        dE = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / max(T, 1e-12)):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_score = new_score
            accepted += 1

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        # occasional full recompute to avoid drift
        if (it + 1) % recompute_every == 0:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The algorithm combines a brief L‑BFGS phase,
    exponentiated‑gradient descent, and a long simulated‑annealing hill‑climb.
    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previously found construction if available.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # Phase partitions (empirically chosen)
    phase1_deadline = start_time + 0.15 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.45 * budget_s   # EG on true max
    phase3_deadline = deadline                      # SA hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – brief L‑BFGS with a few random restarts
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.006, 0.003]
    maxiter_per_lam = 500

    while time.time() < phase1_deadline:
        # 80 % fresh random starts, 20 % jittered best solution
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_deadline, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true max
    # --------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_deadline, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = simulated_annealing_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase3_deadline,
            rng,
            max_fraction_start=0.18,
            max_fraction_end=0.01,
            init_temp_factor=0.005,
            final_temp_factor=1e-6,
            recompute_every=6000,
        )
        print(f"[SA‑hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Symmetrisation + optional final EG fine‑tune
    # --------------------------------------------------------------
    sym = (best_a + best_a[::-1]) / 2.0
    sym /= sym.sum()
    sym_score = fast_score(sym, fft_len)
    if sym_score < best_score:
        best_a = sym
        best_score = sym_score
        print(f"[symmetrise] improved fast_score={best_score:.6f}")

    # one short EG pass to polish the symmetric solution
    if time.time() < deadline - 2.0:   # keep a couple of seconds for final eval
        best_a, best_score = exp_gradient_optimize(
            best_a, time.time() + 1.5, n, fft_len, rng, eta0=6.0
        )
        print(f"[final‑EG] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
