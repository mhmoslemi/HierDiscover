def random_hill_climb(a, score, fft_len, n, deadline, max_trials=100000):

"""
Optimisation of the step‑function bound 2·n·max(a*a)/(∑a)².

The pipeline follows several complementary stages:

1. **Initial population** – uniform, spectral, random and the
   previously discovered best sequence (if any) are normalised to the
   probability simplex.

2. **Coarse L‑BFGS** – many short random‑restart runs on a
   smoothed “log‑sum‑exp” surrogate of the max‑convolution.  The
   smoothing factor is gradually reduced.

3. **Refined L‑BFGS** – a long run on the best start from stage 2 with a
   finer λ‑schedule and many more iterations.

4. **Adam on the smoothed objective** – many cheap Adam steps on the
   same surrogate, with an ever‑decreasing smoothing parameter.  This
   often finds improvements missed by L‑BFGS.

5. **Sub‑gradient exponentiated‑gradient descent** – optimisation of the
   true non‑smooth objective, using a 1/√t step‑size schedule.

6. **Random hill‑climbing** – cheap Monte‑Carlo moves that transfer a
   small amount of mass from a “harmful’’ coordinate to a “helpful’’ one.
   The moves are accepted only if they improve the hard bound.

All stages keep the vector on the simplex (∑a = 1, a_i ≥ 0) and the
best sequence found so far is retained.  The routine stops when the
budget (default 1000 s) is exhausted and returns the best list of
floats for the external `evaluate_sequence` routine.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns objective value and gradient w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation (coarse / refined)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS through a sequence of smoothing parameters.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_start: float,
                  lam_end: float,
                  lam_decay: float,
                  epoch_len: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    The smoothing factor lam_factor is multiplied by max_c internally,
    so we schedule lam_factor from lam_start down to lam_end.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # periodically decay the smoothing factor
        if it % epoch_len == 0 and lam > lam_end:
            lam = max(lam * lam_decay, lam_end)

        # compute smoothed objective and gradient
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

        # Adam update (on logits)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning rate – a modest constant works well here
        lr = 0.05
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # optional early exit if improvement stalls
        if it % 2000 == 0:
            # tiny safeguard to avoid excessive loop time
            if time.time() >= deadline:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # classic 1/√t schedule, but with a larger prefactor
        eta = 0.5 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climbing (monotone improvement)
# ----------------------------------------------------------------------


def random_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from a random donor to a
    random recipient, accepting only moves that improve the hard bound.
    """
    best_a = a.copy()
    best_score = score
    it = 0

    while time.time() < deadline and it < max_iters:
        # choose a donor with probability proportional to its weight
        donor = rng.choice(n, p=best_a)
        if best_a[donor] < 1e-12:
            it += 1
            continue
        # choose a distinct target uniformly
        target = rng.integers(0, n - 1)
        if target >= donor:
            target += 1

        # move a modest fraction of the donor's mass
        delta = best_a[donor] * rng.uniform(0.001, 0.02)
        a_new = best_a.copy()
        a_new[donor] -= delta
        a_new[target] += delta

        # quick sanity: keep non‑negativity (should hold)
        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_a = a_new
            best_score = new_score
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                       # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from any previously found sequence, if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Gather a few diverse seeds
    # ------------------------------------------------------------------
    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)
    candidates = [a0, a_spec, a_rand]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split (approximate)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    phase1_deadline = start_time + 0.15 * budget_s   # coarse L‑BFGS
    phase2_deadline = start_time + 0.45 * budget_s   # refined L‑BFGS
    phase3_deadline = start_time + 0.70 * budget_s   # Adam on smoothed obj
    phase4_deadline = start_time + 0.85 * budget_s   # exponentiated‑gradient
    phase5_deadline = deadline                       # random hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart coarse L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule_coarse = [0.5, 0.35, 0.25, 0.18, 0.12,
                           0.08, 0.05, 0.03, 0.018, 0.01,
                           0.006, 0.003, 0.001, 0.0005]
    maxiter_coarse = 800

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.5:
            a_start = random_initial(n, rng)
        elif r < 0.85:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.01), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule_coarse,
            maxiter_coarse,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined L‑BFGS (single long run)
    # ------------------------------------------------------------------
    lam_schedule_fine = [
        0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02,
        0.015, 0.01, 0.008, 0.005, 0.003, 0.0015,
        0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002
    ]
    maxiter_fine = 3000

    best_a, best_score = lbfgs_continuation(
        best_a,
        lam_schedule_fine,
        maxiter_fine,
        phase2_deadline,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[Phase‑2] refined L‑BFGS score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = adam_optimize(
            best_a,
            lam_start=0.5,
            lam_end=0.00002,
            lam_decay=0.85,
            epoch_len=400,
            deadline=phase3_deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        print(f"[Phase‑3] Adam smoothed‑obj score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient descent (sub‑gradient on hard max)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase4_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Phase‑4] exp‑gradient score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – random hill‑climbing fine polishing
    # ------------------------------------------------------------------
    best_a, best_score = random_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase5_deadline,
        rng,
        max_iters=2_000_000,
    )
    print(f"[Phase‑5] random hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
