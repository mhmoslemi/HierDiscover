def simulated_annealing(a, conv, start_score, deadline, rng):
    best_a = a.copy()
    best_score = start_score
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = start_score
    n = a.shape[0]
    # Temperature schedule: start high -> log linear
    total_seconds = deadline - time.time()
    # We'll compute fraction of time elapsed
    # set initial temperature as e.g., 0.01 * best_score
    T0 = best_score * 0.05 # maybe 5% of best_score
    # cooling factor per second gamma = 0.99
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_seconds
        T = T0 * (1 - frac)  # linear cooling
        # random move
        src = rng.integers(0, n)
        if cur_a[src] <= 1e-12:
            continue
        dst = rng.integers(0, n-1)
        while dst == src:
            dst = rng.integers(0, n)
        # Choose delta as random fraction of source mass
        delta = cur_a[src] * rng.random() * (0.1 + 0.4*frac) # decreasing?
        # compute diff
        diff = np.zeros_like(cur_conv)
        diff[src:src + n] -= cur_a
        diff[dst:dst + n] += cur_a
        new_conv = cur_conv + 2.0 * delta * diff
        new_score = 2.0 * n * new_conv.max()
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            # accept
            cur_a[src] -= delta
            cur_a[dst] += delta
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        # optionally recompute conv exactly every many iterations
        if (elapsed % 30) < 0.01:   # maybe use iteration count
            cur_conv = np.convolve(cur_a, cur_a)
            cur_score = 2.0 * n * cur_conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
    return best_a, best_score

"""
min_convolution_max

Searches for a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The code runs for up to 1000 s and consists of four stages:

1. Random restarts of L‑BFGS on a log‑sum‑exp smoothed version of the max.
2. Adam optimisation on the same smoothed objective (cooling schedule).
3. Exponentiated‑gradient descent on the true max (sub‑gradient descent).
4. Simulated‑annealing hill‑climb that moves small amounts of mass between
   positions, sometimes accepting uphill moves.  This phase gets the bulk of
   the time budget and is the main source of improvements beyond the
   reference solution.

All stages keep the vector on the probability simplex (sum = 1).  The
final sequence is evaluated with the official ``evaluate_sequence`` function
and the best one found is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Surrogate score = 2·n·max(conv) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective & L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  Returns (objective, gradient) w.r.t. logits y,
    where a = softmax(y) is the current probability vector.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]  # grad wrt a before softmax

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam on smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            lr = lr0 / (np.sqrt(v_hat) + eps)
            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

# Exponentiated‑gradient descent on the true max
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Simulated‑annealing hill‑climb (final intensive phase)

def simulated_annealing_climb(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              rng: np.random.Generator):
    """
    Randomly moves a small fraction of mass between two positions.
    Accepts improving moves always and uphill moves with a temperature
    that linearly cools to zero.  The convolution is updated incrementally.
    Returns the best a seen and its fast_score.
    """
    # exact convolution at the beginning
    conv = np.convolve(a_start, a_start)
    cur_a = a_start.copy()
    cur_conv = conv.copy()
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule: start at ~5 % of current score, cool linearly
    T0 = cur_score * 0.05
    start_time = time.time()
    total_time = deadline - start_time
    # diff buffer allocated once
    diff = np.zeros_like(cur_conv)

    iteration = 0
    recompute_interval = 3000  # recompute exact conv to avoid drift

    while time.time() < deadline:
        # linear cooling factor
        remaining = max(0.0, deadline - time.time())
        frac = remaining / total_time
        T = T0 * frac + 1e-12

        # Choose a source (biased towards large entries) and a destination
        src = rng.choice(n, p=cur_a)          # more mass → higher chance
        if cur_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # ensure we move from larger to smaller entry
        if cur_a[src] < cur_a[dst]:
            src, dst = dst, src
        if cur_a[src] <= 0.0:
            continue

        # move a fraction that shrinks with temperature (larger moves early)
        max_frac = 0.25 * frac + 0.02          # from ~0.27 → 0.02
        delta = cur_a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of convolution (ignore O(delta^2) term)
        diff.fill(0.0)
        diff[src:src + n] -= cur_a
        diff[dst:dst + n] += cur_a
        new_conv = cur_conv + 2.0 * delta * diff
        new_score = 2.0 * n * new_conv.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill acceptance with Boltzmann probability
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur_a[src] -= delta
            cur_a[dst] += delta
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()

        iteration += 1
        if iteration % recompute_interval == 0:
            # correct drift from the ignored quadratic term
            cur_conv = np.convolve(cur_a, cur_a)
            cur_score = 2.0 * n * cur_conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

    return best_a, best_score

    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=12.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb (intensive refinement)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_climb(
            best_a, deadline, n, rng,
        )
        print(f"[SA hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
