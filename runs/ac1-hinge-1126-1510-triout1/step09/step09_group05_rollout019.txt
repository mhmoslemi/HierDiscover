"""
Multi‑phase optimiser for the step‑function evaluation problem.

Approach
--------
1.  Initialise a set of diverse deterministic candidates
    (uniform, Hann window, half‑sine, power‑law, exponential).
    Each candidate is projected onto the probability simplex.

2.  For every candidate run a continuation of L‑BFGS on the
    log‑sum‑exp smooth approximation of the max‑convolution.
    The schedule of smoothing parameters (λ) drives the solution
    towards the true non‑smooth objective.

3.  Continue with a projected Adam optimiser that uses the **exact**
    sub‑gradient of the max‑convolution (computed by FFT).

4.  A simulated‑annealing / hill‑climbing phase moves a small
    fraction of mass from indices that contribute heavily to the
    current maximal convolution towards indices that contribute
    little.  Both “transfer’’ and “swap’’ moves are attempted; a
    temperature schedule allows occasional uphill steps.

5.  A short fine‑tuned L‑BFGS run (very small λ) polishes the final
    vector.

All phases respect the 1000 s budget.  The best sequence found is
returned as a plain list whose entries sum to 1 (the external
``evaluate_sequence`` scales it internally, so this normalisation is
compatible with the evaluator).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the objective value 2·n·max(conv(a, a)) for a vector a that is
    already normalised to sum(a) = 1.  Uses an FFT to obtain the full linear
    convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point on the simplex (various heuristics)."""
    mode = rng.choice(["uniform", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        # exponential decay with a random rate
        rate = rng.uniform(0.0005, 0.005)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # a few heavy entries on both ends
        edge_frac = rng.uniform(0.005, 0.02)
        m = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        a[:m] = a[-m:] = np.sqrt(n / 2.0) / n
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def init_hann(n: int) -> np.ndarray:
    """Hann window (zero at the ends, single bump in the middle)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def init_half_sine(n: int) -> np.ndarray:
    """Half‑sine window: sin(π·i/(n‑1)), zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    a /= a.sum()
    return a


def init_power_law(n: int, p: float = 0.5) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^(-p).  p∈(0,∞) – smaller p → flatter."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def init_exponential(n: int, gamma: float = 0.0005) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-γ·i)."""
    a = np.exp(-gamma * np.arange(n))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                      fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed approximation of 2·n·max(conv(a,a)) where a = softmax(y).
    Returns (value, gradient) w.r.t. y.
    """
    # stable softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_schedule, maxiter_per_lam,
                       deadline: float, fft_len_conv: int,
                       fft_len_g: int, n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best vector found and its
    score.
    """
    y = np.log(a_start + 1e-12)  # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:
            print("[L‑BFGS] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """Return (max_conv, sub‑gradient) for the exact max‑convolution."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient = 2·(a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g

def subgradient_adam(a0: np.ndarray, deadline: float,
                     fft_len_conv: int, fft_len_g: int, n: int):
    """Projected Adam using the exact sub‑gradient of max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam update (step size decays with sqrt(it))
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter={it} new best={best_score:.6f}")

        # Insert occasional jitter to escape flat regions
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.005)
            a += rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb phase
# ----------------------------------------------------------------------


def annealing_hill_climb(a_start: np.ndarray, cur_score: float,
                         fft_len_conv: int, fft_len_g: int,
                         n: int, deadline: float,
                         max_iters: int = 250_000):
    """
    Aggressive hill‑climb with a simple simulated‑annealing acceptance rule.
    Returns the best vector seen.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9997
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        max_val, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # indices with high/low gradient contribution
        high_thr = np.percentile(grad, 80)
        low_thr = np.percentile(grad, 20)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue
            frac = rng.uniform(0.04, 0.18)
            delta = cur_a[i] * frac
            if delta <= 0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

# Main driver – required entry point

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser that returns a list of non‑negative floats
    (∑a_i = 1) minimising the external ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise best vector (either warm‑start or uniform)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate old solution to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)   # large enough for all convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global time budget
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2   # safety margin

    # Phase split (empirical)
    phase1_end = start + 0.30 * (budget_s - 0.2)   # heavy L‑BFGS
    phase2_end = start + 0.60 * (budget_s - 0.2)   # Adam
    phase3_end = start + 0.90 * (budget_s - 0.2)   # annealing
    # remainder → fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS on multiple deterministic starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.30, 0.20, 0.12, 0.07,
        0.04, 0.02, 0.01, 0.006, 0.003,
        0.0015, 0.0007, 0.0003, 0.0001, 5e-05,
    ]
    maxiter_per_lam = 600

    deterministic_initialisers = [
        init_uniform(n),
        init_hann(n),
        init_half_sine(n),
        init_power_law(n, p=0.4),
        init_exponential(n, gamma=0.0003),
        init_exponential(n, gamma=0.0010),
        init_power_law(n, p=1.2),
    ]

    # run L‑BFGS on each deterministic starter
    for init_vec in deterministic_initialisers:
        if time.time() >= phase1_end:
            break
        # a brief continuation for each starter
        a_opt, score_opt = lbfgs_continuation(
            init_vec,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS deterministic] new best={best_score:.6f}")

    # additional random restarts while time permits
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS random] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam on the true (non‑smooth) objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing / aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS (very small smoothing) for polishing
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06]
        fine_maxiter = 400
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose globally for possible warm‑starts in a later run
    globals()["height_sequence_1"] = final_seq

    return final_seq
