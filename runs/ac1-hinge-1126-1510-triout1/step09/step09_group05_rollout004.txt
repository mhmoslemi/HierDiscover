"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimiser works in five phases inside the 1000 s budget:

1.  **Shape‑search** – generate a short family of analytically‑motivated
    distributions (uniform, symmetric exponentials, symmetric beta /
    arcsine‑like shapes) and keep the best one according to the fast FFT‑based
    score.

2.  **Heavy L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The schedule of smoothing parameters drives the solution
    towards the true non‑smooth objective.

3.  **Projected Adam** – sub‑gradient descent on the exact (non‑smooth)
    objective with an adaptive‑moment update and simplex projection.

4.  **Aggressive simulated‑annealing hill‑climb** – repeatedly move a
    fraction of mass from indices with a large sub‑gradient to indices with a
    small sub‑gradient (and occasional swaps).  The temperature decays
    slowly, allowing many uphill moves.

5.  **Fine L‑BFGS** – a short continuation with very small smoothing to
    polish the final vector.

All phases respect the global time‑budget; the best vector found is returned
as a plain list (the external ``evaluate_sequence`` routine will normalise
the sum automatically).  The routine also stores the result in the global
``height_sequence_1`` variable for warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv(a, a))  –  a is already normalised to sum(a)=1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random point on the simplex (used for occasional restarts)."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Shape generators (used in Phase 0)
# ----------------------------------------------------------------------


def shape_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    a /= a.sum()
    return a


def shape_beta_symmetric(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta(p,p) density on the discrete grid.
    For p<1 the distribution is heavy at the ends (arcsine‑like).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid overflow/underflow for extreme p
    a = np.maximum(x ** (p - 1) * (1.0 - x) ** (p - 1), 1e-300)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns (value, gradient) for the L‑BFGS routine.
    """
    # stable softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a sequence of L‑BFGS continuations for the given smoothing schedule.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} → new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution + sub‑gradient (used by Adam / SA)
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) for the exact max‑convolution a⨂a.
    The sub‑gradient is the sum of the two symmetric contributions from all
    shifts attaining the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam on the exact objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the true sub‑gradient of the max‑convolution.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a simple decreasing step size (more aggressive than the previous 0.5/√it)
        step = 0.3 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.005)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb phase
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a simple simulated‑annealing acceptance rule.
    Moves mass from high‑gradient indices to low‑gradient ones, also tries
    swaps.  The temperature decays slowly (factor 0.9999) so many iterations
    can be performed inside the allocated time.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature – a fraction of the current score
    temperature = max(0.05 * cur_score, 1e-6)
    decay = 0.9999
    min_temp = 1e-9

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact max and sub‑gradient for the current vector
        max_val, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source / target sets based on percentiles of the gradient
        high_thr = np.percentile(grad, 92)
        low_thr = np.percentile(grad, 8)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.7 and high_idx.size and low_idx.size:
            # transfer a fraction of mass from a high‑gradient entry to a low‑gradient one
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.04, 0.22)   # 4‑22 % of the source mass
                delta = cur_a[i] * frac
                if delta > 0:
                    cand = cur_a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    continue
            else:
                continue
        else:
            # simple swap of two entries – helps re‑order the mass
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # maintain feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        # temperature decay
        temperature = max(temperature * decay, min_temp)
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – the required function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser for the step‑function sequence.
    Returns a list of non‑negative floats (sum = 1) that minimises the
    external ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a modest size keeps the inner loops cheap.
    # ------------------------------------------------------------------
    n = 4000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 0 – quick analytical shape search
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # fallback: uniform
    best_score = fast_score(best_a, fft_len_conv, n)

    print("[phase 0] starting shape search (uniform baseline)")

    # uniform already evaluated
    shape_candidates = []

    # exponential family
    for lam in np.logspace(np.log10(0.001), np.log10(0.02), num=9):
        shape_candidates.append(shape_exponential(n, lam))

    # symmetric beta / arcsine‑like family
    for p in np.linspace(0.2, 2.0, num=19):
        shape_candidates.append(shape_beta_symmetric(n, p))

    # evaluate all candidates
    for idx, a in enumerate(shape_candidates):
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[phase 0] candidate {idx} → new best={best_score:.6f}")

    print(f"[phase 0] best score after shape search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2  # safety margin

    # allocate portions of the total time (empirically chosen)
    phase1_end = start + 0.30 * (budget_s - 0.2)   # heavy L‑BFGS
    phase2_end = start + 0.60 * (budget_s - 0.2)   # Adam
    phase3_end = start + 0.90 * (budget_s - 0.2)   # annealing
    # remainder → final fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 900

    while time.time() < phase1_end:
        # occasional diversification to avoid premature convergence
        if rng.random() < 0.25:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score after Adam = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=800_000,
        )
        print(f"[Anneal] best_score after annealing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine‑tuned L‑BFGS (very small smoothing)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07, 2e-07]
        fine_maxiter = 600
        best_a, fine_sc = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_sc < best_score:
            best_score = fine_sc
            print(f"[L‑BFGS fine] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (for the competition harness)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts later
    globals()["height_sequence_1"] = final_seq

    return final_seq
