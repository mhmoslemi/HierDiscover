def hill_climb_guided(a, curr_score, fft_len, n, deadline, rng, max_fraction=0.08, recompute_every=2000, temperature_schedule=None):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    iteration = 0
    accepted = 0
    while time.time() < deadline:
        # compute gradient of max conv using mask
        fa = np.fft.fft(best_a, fft_len)
        conv_full = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv_full.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(np.float64)
        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n] # length n
        # compute src,dst based on gradient magnitudes: source with highest g (increase objective), destination with lowest g.
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst:
            # random fallback
            src = rng.integers(0,n)
            dst = rng.integers(0,n)
        if best_a[src] == 0:
            # choose another src
            nonzero = np.where(best_a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero))
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0:
            continue
        # incremental conv update similar to earlier
        diff = np.zeros_like(conv)
        diff[src:src+n] -= best_a
        diff[dst:dst+n] += best_a
        conv_new = conv + 2.0*delta*diff
        new_max = conv_new.max()
        new_score = 2.0*n*new_max
        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        else:
            # optional simulated annealing acceptance
            # temperature goes down linearly
            if temperature_schedule is not None:
                temp = temperature_schedule()
                if rng.random() < np.exp(-(new_score - best_score)/temp):
                    best_a[src] -= delta
                    best_a[dst] += delta
                    conv = conv_new
                    best_score = new_score
                    accepted += 1
        iteration+=1
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
    return best_a, best_score
```

"""
Improved search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The method expands on the previous four‑phase optimiser:

1. **Diverse random sampling** – a quick scan of many random points on the
   simplex, keeping the best few candidates for later refinement.

2. **Multi‑restart L‑BFGS on a log‑sum‑exp smooth approximation of max** –
   each kept candidate is optimised with a cascade of λ‑values.  The
   optimisation works on the soft‑max logits (a = softmax(y)), guaranteeing
   positivity and the simplex constraint.

3. **Adam on the same smoothed objective** – a longer fine‑tuning phase with a
   decreasing temperature schedule.

4. **Exact exponentiated‑gradient descent** – optimisation on the true
   max‑convolution using the sub‑gradient and a decaying step‑size.

5. **Guided greedy hill‑climb** – moves mass from indices with large
   sub‑gradient (most harmful for the max) to those with small sub‑gradient.
   The convolution is updated incrementally for speed and occasional full
   recomputation prevents drift.

The time budget (default 1000 s) is split among the phases; each phase stops
as soon as the deadline is reached.  The best sequence found is finally
evaluated with the supplied `evaluate_sequence` function and returned as a
plain Python list.
"""

import time
import math
import heapq
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv) for a probability vector (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'triangular', 'sinpow']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'triangular':
        # symmetric triangle, peak at centre
        x = np.arange(n, dtype=float) - (n - 1) / 2.0
        a = np.maximum(0, 1 - np.abs(x) * 2 / n)
    elif mode == 'sinpow':
        # sin^p shape, p in [1,8]
        p = rng.uniform(1.0, 8.0)
        t = np.linspace(0, math.pi, n)
        a = np.sin(t) ** p
    else:
        a = np.full(n, 1.0 / n)

    # optionally enforce symmetry (most good solutions are symmetric)
    if rng.random() < 0.3:
        a = (a + a[::-1]) * 0.5

    a += 1e-12  # avoid exact zeros which may hamper optimisers
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Returns (value, grad) for the smoothed objective:
        f(y) = 2·n·smooth_max(conv(softmax(y)))
    The smoothing is a log‑sum‑exp with scale λ = lam_factor·max(conv).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution:
    # d conv / d a = rev(a) * w   (linear convolution)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a (before softmax) lives on the central n entries
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation over a schedule of λ values
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result.  Returns the best a (as a probability vector) and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
            if steps % 300 == 0:
                print(f"[Adam] lam={lam:.5f} step={steps} best_score={best_score:.6f}")

        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Sub‑gradient descent on the exact max‑convolution, using a
    multiplicative (exponentiated) update and a 1/√t step‑size.
    Returns the best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
        if it % 500 == 0:
            print(f"[ExpGrad] it={it} best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Guided greedy hill‑climb (mass transfer guided by sub‑gradient)
# ----------------------------------------------------------------------


def hill_climb_guided(a: np.ndarray,
                      curr_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_fraction: float = 0.15,
                      recompute_every: int = 1200):
    """
    Greedy hill‑climb that always moves a fraction of mass from an index
    with a large sub‑gradient (harmful for the max) to an index with a small
    sub‑gradient.  Convolution updates are performed incrementally.
    """
    best_a = a.copy()
    best_score = curr_score
    # exact convolution for the first step
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)
    accepted = 0
    it = 0

    while time.time() < deadline:
        # --- compute sub‑gradient of the current max ---
        fa = np.fft.fft(best_a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv_full.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(np.float64)

        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # length n

        # choose source (large gradient) and destination (small gradient)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or best_a[src] <= 0.0:
            # fall back to a random move if the gradient choice is degenerate
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            if best_a[src] <= 0.0:
                continue

        # amount to transfer
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of convolution when moving delta from src to dst
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # occasional full recompute to control numeric drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
        if it % 5000 == 0:
            print(f"[HillClimb] it={it} best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the evaluation
    function.  Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 entries gives a good balance between quality and
    # speed (the algorithm works for any n, the constant in the objective
    # is asymptotically n‑independent).
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously stored construction if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # split the budget (empirically chosen)
    phase0_end = start_time + 0.15 * budget_s   # random sampling
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.68 * budget_s   # Adam
    phase3_end = start_time + 0.85 * budget_s   # Exp‑grad
    # remaining → hill‑climb

    # ------------------------------------------------------------------
    # Phase 0 – diverse random sampling, keep top‑K candidates
    # ------------------------------------------------------------------
    top_K = 12
    top_candidates = []   # min‑heap storing (-score, array)
    while time.time() < phase0_end:
        a = random_initial(n, rng)
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sample] new best fast_score={best_score:.6f}")

        # keep top candidates (negative score for max‑heap behaviour)
        if len(top_candidates) < top_K:
            heapq.heappush(top_candidates, (-sc, a.copy()))
        else:
            if sc < -top_candidates[0][0]:
                heapq.heapreplace(top_candidates, (-sc, a.copy()))

    # extract sorted list (best first)
    top_candidates = [item[1] for item in sorted(top_candidates, key=lambda x: -x[0])]

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS on a smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # pick a candidate – 70% from the elite list, 30% fresh jittered best
        if rng.random() < 0.3 and len(top_candidates) > 0:
            a_start = top_candidates[rng.integers(len(top_candidates))]
        else:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n)

        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.4, 0.25, 0.15, 0.08, 0.04,
                             0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam, max_steps_per_lam=1800,
            deadline=phase2_end, n=n, fft_len=fft_len, lr0=0.13)
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=12.0)
        print(f"[Exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – guided greedy hill‑climbing (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_guided(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.18,
            recompute_every=1500)
        print(f"[Hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
