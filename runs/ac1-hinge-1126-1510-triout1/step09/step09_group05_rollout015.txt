"""
Propose a non‑negative step function (a probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimisation is performed in four stages (all respecting a 1000 s
budget):

1. **Heavy L‑BFGS** on a log‑sum‑exp smoothed surrogate of the
   max‑convolution.  Several random restarts and a decreasing smoothing
   schedule drive the solution close to a (non‑smooth) local optimum.

2. **Linear‑programming improvement** – given the current vector `f`,
   solve the LP

        maximise   Σ g_j
        subject to   f * g ≤ M      (M = max(f * f))
                    g ≥ 0

   The optimal `g` concentrates mass where the current gradient is
   smallest.  A convex combination `f←(1‑t)f + t g` (with a small `t`)
   reduces the max‑convolution; the step is repeated a few times.

3. **Projected Adam** on the exact non‑smooth objective, using the true
   sub‑gradient of the max‑convolution.

4. **Fine L‑BFGS** with an extremely small smoothing parameter to polish
   the final vector.

The best sequence found is returned as a plain list.  The global variable
`height_sequence_1` is populated for warm‑starts in subsequent runs.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming a already satisfies sum(a)=1.
    Uses FFT to obtain the full linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """Smooth approximation of 2·n·max(conv) where a = softmax(y)."""
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
):
    """
    Returns (max_conv, subgradient) for the exact max‑convolution of a with itself.
    The sub‑gradient corresponds to the sum of contributions from all shifts
    that attain the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    # full convolution a * a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Projected Adam (adaptive moment) using the true sub‑gradient of the
    max‑convolution.  Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum

    return best_a, best_score

# Linear‑programming based improvement
# ----------------------------------------------------------------------


def lp_iterative_improvement(
    a_start: np.ndarray,
    init_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Given a vector `a`, iteratively solve the LP

        maximise Σ g_j   subject to   a * g ≤ M,  g ≥ 0

    where M = max(a * a).  After each LP, form a convex combination
    (1‑t) a + t g (t chosen from a small set) and keep the improvement.
    Returns the best vector found before `deadline`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = init_score
    best_max = best_score / (2.0 * n)

    iteration = 0
    while time.time() < deadline:
        # recompute current bound M
        M = best_max

        # Build convolution matrix A where A @ g = a * g
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A_dense = toeplitz(col, row)               # (2n‑1) × n dense
        A = csr_matrix(A_dense)                    # sparse for linprog

        # linear programme
        c = -np.ones(n)                            # maximise Σ g_j  → minimise -Σ g_j
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            print(f"[LP] iteration {iteration} produced zero‑mass g")
            break
        g = g0 / sum_g  # normalise to sum = 1

        # try a few mixing coefficients
        ts = [0.25, 0.5, 0.75, 1.0]
        improved = False
        for t in ts:
            cand = (1 - t) * a + t * g
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_max = best_score / (2.0 * n)
                best_a = cand.copy()
                a = cand.copy()
                improved = True
                print(f"[LP] it={iteration} t={t:.2f} new best={best_score:.6f}")
                break  # accept the first improving t

        if not improved:
            # no t gave improvement → stop the LP loop
            break

        iteration += 1
        if iteration >= 30:   # safety cap
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size keeps the inner loops cheap.
    # ------------------------------------------------------------------
    n = 2000  # chosen to keep LP inexpensive while still expressive

    # ------------------------------------------------------------------
    # Initialise from the previous best if available; otherwise uniform.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.1  # small safety margin

    # allocate phases
    phase1_end = start + 0.30 * (budget_s - 0.1)   # heavy L‑BFGS
    phase2_end = start + 0.65 * (budget_s - 0.1)   # LP improvements
    phase3_end = start + 0.90 * (budget_s - 0.1)   # Adam fine‑tune
    # remainder → fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS (multiple random restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # occasional diversification
        if rng.random() < 0.35:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming based improvement
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = lp_iterative_improvement(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase2_end,
            rng,
        )
        print(f"[LP] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_adam(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine‑tuned L‑BFGS (very small smoothing)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07]
        fine_maxiter = 800
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose globally for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
