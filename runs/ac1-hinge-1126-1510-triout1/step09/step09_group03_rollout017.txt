"""
Improved step‑function optimizer.

The optimizer works inside a 1000 s budget (configurable via *budget_s*).
It searches for a non‑negative vector a with ∑a = 1 that minimises

        2·n·max(conv(a,a))

where conv is the linear convolution.
The search proceeds in four stages:

1. **Diverse restarts of a smoothed L‑BFGS** – the true max is replaced
   by a log‑sum‑exp with a decreasing temperature (λ).  Thousands of short
   L‑BFGS runs explore many basins; the best basin is kept.

2. **Long Adam‑style exponentiated‑gradient descent** on the exact
   max‑convolution (sub‑gradient).  The step size decays as η₀/√t and a
   momentum term accelerates convergence.

3. **Deterministic safe moves** – mass is transferred from the index with
   the largest gradient to the one with the smallest, using the analytically
   maximal step that never raises the current peak.

4. **Fine‑grained simulated annealing** – random mass transfers are
   accepted with a Metropolis rule while the temperature is cooled linearly.

The algorithm also injects a small set of *block‑initialisations* (few
wide flat blocks) to escape the basin around the uniform distribution.
All phases keep the current best sequence and return it after the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) assuming a sums to 1 (fast surrogate for evaluate_sequence)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random probability vectors."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def block_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a vector consisting of a few flat blocks separated by gaps.
    The block widths are chosen so that all blocks fit comfortably.
    """
    # number of blocks – a random small integer
    K = rng.integers(4, 12)
    # allocate a generous amount of space for gaps:
    max_total_block = n // (2 * K + 1)
    if max_total_block < 5:
        return random_initial(n, rng)
    B = rng.integers(5, max_total_block + 1)          # block width
    gap = B + rng.integers(0, B)                     # gap ≥ B
    a = np.zeros(n, dtype=float)
    pos = 0
    for _ in range(K):
        if pos + B > n:
            break
        a[pos : pos + B] = 1.0 / (K * B)              # equal weight per block
        pos += B + gap
    # normalise – occasional rounding errors may make sum ≠ 1
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = random_initial(n, rng)
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    The optimisation variables are logits y; a = softmax(y) lives on the simplex.
    Returns (objective, gradient w.r.t. y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smooth max with log‑sum‑exp, temperature = lam
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]               # gradient wrt a

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_multi_restart(a_start: np.ndarray,
                        lam_sched,
                        maxiter_per_lam: int,
                        deadline: float,
                        fft_len: int,
                        n: int,
                        rng: np.random.Generator):
    """
    Run a cascade of L‑BFGS optimisations for a given temperature schedule.
    Returns the best vector found in this run.
    """
    y = np.log(a_start + 1e-12)          # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_adam(a_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      rng: np.random.Generator,
                      eta0: float = 8.0,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8):
    """
    Adam‑style exponentiated‑gradient descent on the exact L∞‑objective.
    Works directly in the probability simplex via multiplicative updates.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # Adam moments for the *log* update
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    t = 0
    while time.time() < deadline:
        # --- gradient of the hard max -------------------------------------------------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_loc = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_loc).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]      # size n

        # --- Adam on the *log* (mirrored) variables ---------------------------------
        t += 1
        eta_t = eta0 / np.sqrt(t)                  # learning‑rate decay

        m_t = beta1 * m_t + (1 - beta1) * (-g)      # note the minus sign (we descend)
        v_t = beta2 * v_t + (1 - beta2) * (g * g)

        m_hat = m_t / (1 - beta1 ** t)
        v_hat = v_t / (1 - beta2 ** t)

        step = eta_t * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated update
        exp_term = np.exp(np.clip(step, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # ----------------------------------------------------------------------
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # ----------------------------------------------------------------------
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic safe moves
# ----------------------------------------------------------------------


def deterministic_descent(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """Move mass from the index with largest gradient to the one with smallest."""
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution (same derivation as in phase‑2)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))            # mass to remove
        dst = int(np.argmin(grad))            # where to add
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # admissible δ  – source positivity
        delta_src = a[src]

        # bound from other entries that would increase (d > 0)
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        # safety factor 0.99 (a little more aggressive than the previous 0.92)
        delta = min(delta_src, delta_bound) * 0.99
        if delta <= 0.0:
            break

        # apply move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to keep numerical drift small
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (fine‑grained hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """Random mass transfers accepted with Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.12   # allow moving up to 12 % of the source mass
    recompute_every = 2500
    accepted = 0

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        accept = False
        if dE < 0.0:
            accept = True
        elif rng.random() < np.exp(-dE / temp):
            accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand elements give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – use a previous construction if available, otherwise a mix
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.20 * budget_s    # L‑BFGS restarts
    phase2_end = start_time + 0.70 * budget_s    # Adam‑style EG descent
    phase3_end = start_time + 0.90 * budget_s    # deterministic safe moves
    # remaining → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs on a smoothed objective
    # ------------------------------------------------------------------
    # temperature schedule – from coarse to very fine
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 5e-05, 2e-05, 1e-05,
        5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # 30 % chance to inject a block‑initialisation, otherwise a fresh random start
        if rng.random() < 0.30:
            a_start = block_initial(n, rng)
        else:
            a_start = random_initial(n, rng)

        # occasional small perturbation of the current best to escape the basin
        if rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)

        a_opt, score_opt = lbfgs_multi_restart(
            a_start, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n, rng
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam‑style exponentiated gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_adam(
            best_a, phase2_end, n, fft_len, rng, eta0=8.0
        )
        print(f"[Adam‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic safe mass transfers
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(
            best_a, phase3_end, n, fft_len
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the construction for possible subsequent runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
