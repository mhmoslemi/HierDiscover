"""
Propose a non‑negative step‑function (a list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimiser works in four phases inside the 1000 s budget:

1. **Heavy L‑BFGS** with a long smoothing schedule – many iterations per
   λ and frequent random restarts.  The smooth surrogate is the usual
   log‑sum‑exp of the convolution entries, differentiated analytically.

2. **Projected Adam** on the exact nonsmooth objective – a long sub‑gradient
   descent that respects the simplex constraint.

3. **Aggressive simulated‑annealing/hill‑climb** – moves a fraction of the
   mass from high‑gradient (dangerous) indices to low‑gradient ones.
   Temperature decays very slowly so that many uphill steps are still
   permitted.

4. **Fine‑tuned L‑BFGS** (very small λ) to polish the final vector.

All phases respect a global time‑budget and use only NumPy / SciPy.
The returned list is normalised to sum = 1 (any scaling would give the
same objective because the evaluation is homogeneous).
"""

import time
import numpy as np
from scipy.optimize import minimize
from numpy.fft import fft, ifft


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the exact external objective
        2·n·max(conv(a,a))
    assuming a already satisfies sum(a)=1.
    Uses FFT for the linear convolution.
    """
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "exp", "power", "edge", "bimodal", "triangular", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "triangular":
        # linear increase to centre then decrease
        x = np.arange(n, dtype=float) / (n - 1)
        a = 1.0 - np.abs(x - 0.5) * 2.0
        a += 0.01  # avoid exact zeros
    else:  # cosine
        x = np.arange(n, dtype=float) / (n - 1)
        a = np.cos(np.pi * (x - 0.5))
        a = np.maximum(a, 0.0) + 0.001
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smooth approximation of 2·n·max(conv) where a = softmax(y)."""
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS over the whole λ schedule, each stage started from the
    result of the previous stage.  Returns the best vector and its score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] lam={lam:.1e} new best={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Return (max_conv, subgradient) for the exact max‑convolution of a with itself.
    The sub‑gradient aggregates contributions from every shift attaining the
    maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2·(a_rev * mask)
    a_rev = a[::-1]
    fm = fft(mask, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_g = np.real(ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moment) using the exact sub‑gradient.
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)   # slowly decaying step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # simple Euclidean projection onto simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to kick out of plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb.  At each iteration a fraction of the mass is moved
    from a high‑gradient index to a low‑gradient one (or a random swap).
    Acceptance follows a simple Boltzmann rule with a very slow exponential
    temperature decay, allowing many uphill moves.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature and decay – receptive to the magnitude of the score
    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9998
    min_temp = 1e-12

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact max and sub‑gradient for the current vector
        max_val, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source and destination indices based on gradient quantiles
        high_thr = np.percentile(grad, 92)
        low_thr = np.percentile(grad, 8)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        # ---- choose a move -------------------------------------------------
        if rng.random() < 0.7 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue
            # amount of mass to move – adaptively smaller when temperature low
            max_move = cur_a[i] * 0.25
            delta = rng.uniform(0.001, max_move)   # avoid zero moves
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # maintain feasibility (non‑negative, sum = 1)
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        cand_score = fast_score(cand, fft_len_conv, n)

        # acceptance rule (Metropolis)
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – the required function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser for the step‑function sequence.
    Returns a plain Python list (sum ≈ 1) that minimises the external
    ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Warm‑start from a previous solution if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = simplex_projection(a0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2  # safety margin

    phase1_end = start + 0.38 * (budget_s - 0.2)   # heavy L‑BFGS
    phase2_end = start + 0.65 * (budget_s - 0.2)   # Adam
    phase3_end = start + 0.92 * (budget_s - 0.2)   # annealing
    # remainder → final fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS with many restarts and tighter iteration limits
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05,
        1e-05, 5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 1200   # a bit more than the previous 800

    while time.time() < phase1_end:
        # aggressive diversification
        if rng.random() < 0.45:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact objective (longer run)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing / hill‑climb (most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=1_000_000,   # allow a very large number of cheap moves
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS (very small λ, many iterations)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        fine_maxiter = 800
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (for debugging / logging)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_eval:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised", exc)

    # expose globally for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
