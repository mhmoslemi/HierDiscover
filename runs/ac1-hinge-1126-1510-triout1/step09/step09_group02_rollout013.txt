"""
Propose a low‑score step sequence for the evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑phase optimiser:

1. **L‑BFGS on a log‑sum‑exp smoothed max** – many random restarts,
   a cascade of decreasing smoothing parameters and a generous iteration budget.

2. **Adam optimisation** – the same smoothed objective with an exponentially
   decreasing temperature (λ) and a tuned learning‑rate schedule.

3. **Exact exponentiated‑gradient descent** on the true max‑convolution
   (sub‑gradient = mask of active entries).  The step‑size decays as 1/√t.

4. **Guided hill‑climb** – for a random pair (src,dst) we compute the exact
   change in the convolution and perform a ternary search for the best
   mass transfer δ (subject to 0 ≤ δ ≤ α·a_src).  This yields a near‑optimal
   improvement for each move.

5. **Fine random refinement** – add tiny Gaussian noise, re‑project onto the
   probability simplex and keep the move if it improves the score.

All phases respect the global 1000 s deadline.  The starting point is taken
from the global variable `height_sequence_1` (if present) and up‑sampled to
the working size `n = 8000`.  The final sequence is returned as a plain list
of floats; `evaluate_sequence` will be called afterwards for the official
score.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (FFT length ≥ 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) assuming sum(a) == 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:

    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (log‑sum‑exp max)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """Return (smooth objective, gradient) w.r.t. logits y where a = softmax(y)."""
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient wrt a_i (pre‑softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run a cascade of L‑BFGS optimisations for each λ in lam_schedule."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """Adam optimiser on the log‑sum‑exp smoothed max."""
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """Exponentiated‑gradient descent directly on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Guided hill‑climb with optimal line‑search (ternary search)
# ----------------------------------------------------------------------


def hill_climb_optimal(a: np.ndarray,
                       curr_score: float,
                       fft_len: int,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_moves: int = 3000,
                       max_fraction: float = 0.15):
    """
    Greedy hill‑climb that, for each randomly chosen (src,dst) pair,
    finds the optimal mass transfer δ by ternary search on the exact
    convolution update.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact linear convolution

    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # always move from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] <= 0.0:
            continue
        delta_max = best_a[src] * max_fraction
        if delta_max <= 0.0:
            continue

        # incremental convolution change for a unit transfer δ
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a

        # ternary search for the best δ ∈ [0, delta_max]
        lo, hi = 0.0, delta_max
        for _ in range(7):               # ~7 iterations give sufficient precision
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            max1 = np.max(conv + 2.0 * m1 * diff)
            max2 = np.max(conv + 2.0 * m2 * diff)
            if max1 < max2:
                hi = m2
            else:
                lo = m1
        delta_opt = (lo + hi) * 0.5

        new_conv = conv + 2.0 * delta_opt * diff
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-12:      # accept only strict improvements
            best_a[src] -= delta_opt
            best_a[dst] += delta_opt
            conv = new_conv
            best_score = new_score
            moves += 1
            # occasionally refresh conv from scratch to avoid drift
            if moves % 500 == 0:
                conv = np.convolve(best_a, best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fine random refinement (tiny Gaussian perturbations)
# ----------------------------------------------------------------------


def random_refinement(a: np.ndarray,
                      best_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator):
    """Add decreasing‑amplitude Gaussian noise and keep any improvement."""
    a_cur = a.copy()
    cur_score = best_score
    # schedule for noise magnitude: start at 0.02, decay to 1e‑5
    start_sigma = 0.02
    end_sigma = 1e-5
    total = max(deadline - time.time(), 1e-12)

    while time.time() < deadline:
        elapsed = deadline - time.time()
        # linear interpolation of sigma based on remaining time
        sigma = start_sigma * (elapsed / total) + end_sigma * (1 - elapsed / total)
        cand = add_noise(a_cur, sigma, rng)
        cand_score = fast_score(cand, fft_len)
        if cand_score < cur_score:
            a_cur = cand
            cur_score = cand_score
    return a_cur, cur_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step sequence with a very low evaluation score."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 entries gives a finer discretisation without a
    # prohibitive cost (FFT length stays at 16384).
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the previous best if available, otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # Adam
    phase3_end = start_time + 0.80 * budget_s   # Exp‑grad
    phase4_end = start_time + 0.95 * budget_s   # Guided hill‑climb
    # remainder → random refinement

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # 70 % fresh random starts, 30 % jittered versions of the current best
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.4, 0.25, 0.15, 0.08, 0.04,
                             0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam, max_steps_per_lam=2200,
            deadline=phase2_end, n=n, fft_len=fft_len, lr0=0.13,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=15.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Guided hill‑climb with optimal line‑search
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = hill_climb_optimal(
            best_a,
            best_score,
            fft_len,
            n,
            phase4_end,
            rng,
            max_moves=5000,
            max_fraction=0.18,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Fine random refinement (tiny Gaussian perturbations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_refinement(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
        )
        print(f"[refine] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
