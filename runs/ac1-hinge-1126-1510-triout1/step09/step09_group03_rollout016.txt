def subgradient(a, fft_len):
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[:2*n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val
```

def exp_gradient_optimize(a_start, deadline, n, fft_len, rng, eta0=5.0):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    t = 1
    while time.time() < deadline:
        grad, max_val = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(t)
        exp_term = np.exp(-eta * grad)
        # clip
        exp_term = np.clip(exp_term, np.exp(-50), np.exp(50))
        a = a * exp_term
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        t += 1
    return best_a, best_score

"""
Candidate generator for the step‑function minimisation task.

Approach
--------
1. **Parameter sweep** – cheaply explores several families of analytic
   sequences (pure exponentials, linear ramps, mixtures with the uniform
   distribution and power‑law decays).  Each candidate is normalised and
   evaluated with a fast FFT based score; the best one becomes the seed
   for the next phase.

2. **Exponentiated‑gradient descent** – mirrors a projected SGD on the
   simplex.  The true (non‑smoothed) max‑convolution is used to obtain a
   sub‑gradient; a multiplicative update ensures non‑negativity and the
   projection on the probability simplex preserves the normalisation.
   The step size decays like η₀/√t.

3. **Simulated‑annealing fine‑tuning** – random mass transfers between
   two indices are accepted according to a Metropolis rule with a linear
   cooling schedule.  Incremental updates of the convolution keep the
   routine fast.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned and stored in the global variable
`height_sequence_1` for possible reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the true max‑convolution objective and the
    current maximal convolution value.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force over a few low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # --- pure exponential -------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=200):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- linear decreasing ------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=50):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=150):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent on the true objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
):
    """Mirrored SGD (multiplicative update) on max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (exponentiated gradient)
        exponents = -eta * g
        exponents = np.clip(exponents, -50, 50)   # avoid overflow
        a = a * np.exp(exponents)

        # re‑project onto the simplex
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers with Metropolis acceptance and linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # initialise temperature as a small fraction of the current score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.20   # fraction of source mass moved per proposal
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        # linear cooling factor
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n) work)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if one exists.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split (roughly 5 % / 73 % / 22 % )
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a tiny safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.78 * budget_s
    # remaining time → simulated annealing

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng, eta0=6.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
