def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the first 2·len(a)-1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a sequence already normalised to ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries corresponding to the original vector (window of length n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y

def adam_optimize(
    a_start: np.ndarray,
    lam_factors,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max objective.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * g
        step = 0.8 / np.sqrt(it + 1.0)  # slightly larger initial step
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_score


def greedy_coordinate_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small mass from the index with
    the largest positive sub‑gradient to the index with the most negative.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # parameters
    epsilon_start = 0.02  # fraction of mass moved in first epochs
    epsilon_min = 0.0005
    decay_rate = 0.995  # exponential decay per iteration
    eps = epsilon_start

    while time.time() < deadline:
        # compute sub‑gradient for the current max
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12 * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i = int(np.argmax(g))  # moving mass away from here
        j = int(np.argmin(g))  # moving mass towards here
        if i == j:
            # gradient is flat – nothing to improve
            break

        delta = eps * a[i]
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re‑project to the simplex (numerical drift may appear)
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            a = a_new
        else:
            # reject move but still decay epsilon
            pass

        eps = max(eps * decay_rate, epsilon_min)

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that respects mass conservation.
    Temperature decays linearly to zero.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.05 * start_score  # higher initial temperature than baseline
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.03)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(300.0, 3000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(250.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(150.0, 600.0)
        sigma2 = rng.uniform(150.0, 600.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families.
    Returns the best vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.10
    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Use previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 10 % of total budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # Phase time slots (adjusted for the larger n)
    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart Adam on smoothed max
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.7, 1e-5, num=30)   # finer schedule
    iters_per_lam = 700                           # more iterations per level

    while time.time() < phase1_end:
        # Choose a start point: jittered copy of current best (most of the time)
        # or a fresh random family (occasionally).
        if rng.random() < 0.35:
            a_start = best_a * rng.normal(1.0, 0.025, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (true max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy coordinate descent (deterministic hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = greedy_coordinate_descent(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[greedy] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (stochastic refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
