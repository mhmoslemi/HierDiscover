def lbfgs_optimize(a_start, lam_schedule, maxiter_per_lam, deadline, fft_len, n):
    y = np.log(a_start+1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam, 'gtol':1e-12})
            y = res.x
            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception as e:
            pass
    return best_a, best_score

def simulated_annealing(a_start, start_score, deadline, rng):
    n = len(a_start)
    fft_len = next_pow2(2*n-1)
    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score
    # temperature schedule
    total = deadline - time.time()
    T0 = cur_score * 0.05  # initial temperature (5% of score)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total)
        fraction = (deadline - time.time()) / total
        T = T0 * fraction
        # random source biased by current values
        src = rng.choice(n, p=cur/cur.sum())
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if cur[src] < 1e-12:
            continue
        delta = cur[src] * rng.random() * 0.2
        if delta <= 0:
            continue
        # propose new distribution
        new = cur.copy()
        new[src] -= delta
        new[dst] += delta
        # compute score
        new_score = fast_score(new, fft_len)
        # acceptance
        if new_score < cur_score or (T>0 and rng.random() < math.exp(-(new_score - cur_score)/T)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        it += 1
        # optionally reduce max_fraction gradually
    return best, best_score

def hill_climb(a_start, start_score, deadline, rng):
    n = len(a_start)
    fft_len = next_pow2(2*n-1)
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)
    # increments
    max_fraction = 0.15
    accepted = 0
    while time.time() < deadline:
        # decaying max_fraction
        time_left = deadline - time.time()
        # reduce as we go:
        max_fraction = max(0.02, max_fraction * (time_left/(deadline-start_time)))
        # sample src weighted by a
        src = rng.choice(n, p=best/np.sum(best))
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best[src] < 1e-12:
            continue
        delta = best[src] * rng.random() * max_fraction
        if delta <= 0:
            continue
        # incremental update
        diff = np.zeros_like(conv)
        diff[src:src+n] -= best
        diff[dst:dst+n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        # recompute full conv occasionally
        if accepted and accepted % 1500 == 0:
            conv = np.convolve(best, best)
    return best, best_score

"""
search for a non‑negative step sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in four stages inside the given 1000 s budget:

1. A few short L‑BFGS runs on a log‑sum‑exp smoothed “max‑conv” objective.
   This quickly finds a good basin on the probability simplex.
2. Adam optimisation on the same smoothed objective with a decreasing
   smoothing parameter (λ) – it refines the solution aggressively.
3. Exponentiated‑gradient descent on the *true* max‑convolution (sub‑gradient
   descent) – an essentially parameter‑free method that keeps the iterates
   on the simplex.
4. A long greedy hill‑climb that moves a small fraction of mass from a
   heavy index to a lighter one; the move size shrinks with time.
   Convolution updates are performed incrementally for speed.

All stages keep the vector on the simplex, therefore the final evaluation
is simply `2·n·max(conv)`.  The best sequence found before the deadline is
returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled random walk on the simplex.
    Moves a random fraction of mass from a source index (chosen
    proportionally to its current weight) to a destination index.
    Accepts uphill steps with probability exp(‑Δ/ T).
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12                # initial temperature ~5 % of score
    max_frac = 0.20                               # max move fraction at start

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        # adapt move size
        max_frac = 0.02 + 0.18 * fraction

        # source biased by current mass, destination uniform
        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a fraction of mass from a heavy index
    to a lighter one, accepting only improvements.  Convolution updates
    are performed incrementally.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        # fraction shrinks from 0.20 → 0.02 as time runs out
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        # source weighted by its mass, destination uniform
        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # refresh the exact convolution occasionally to avoid drift
        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries works well for the time budget.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if available; otherwise uniform.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # small safety margin

    # Phase allocation (approximate)
    phase1_end = start_time + 0.10 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.20 * budget_s   # Adam (smooth)
    phase3_end = start_time + 0.70 * budget_s   # Exp‑grad
    # Remaining time → simulated annealing + hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – short L‑BFGS bursts from random starts
    # --------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 1e-5, num=12)

            deadline=phase2_end, n=n, fft_len=fft_len, lr0=0.13

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
