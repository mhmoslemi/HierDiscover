def simulated_annealing(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    temperature = 1.0 # will schedule down to 1e-4
    while time.time() < deadline:
        # randomly choose i!=j
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        # choose delta: maybe step fraction of a[i] or something
        # delta = rng.random() * min(a[i], step_max)??,
        max_delta = min(a[i], 0.1) # can't move more than 0.1 maybe
        delta = rng.random() * max_delta
        # propose new a
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        score_new = fast_score(a_new, fft_len, n)
        # Accept if better or with prob
        if score_new < best_score:
            # always accept obviously improves best
            best_score = score_new
            best_a = a_new.copy()
        # Accept according to Metropolis:
        if score_new < cur_score or rng.random() < np.exp(-(score_new - cur_score) / temperature):
            a = a_new
            cur_score = score_new
        # cooldown temperature gradually
        temperature = max(temperature * 0.9999, 1e-6) 
```

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)

def simulated_annealing(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    # initial temperature based on cur_score
    temperature = (cur_score - 1) * 0.5  # or some value
    # or start with 0.01
    while time.time() < deadline:
        # choose source index with non-zero mass
        src = rng.choice(np.where(a > 1e-12)[0])
        # choose dest index different
        dest = rng.integers(0, n)
        while dest == src:
            dest = rng.integers(0, n)
        # delta fraction
        max_delta = min(a[src], 0.1)  # limit to 0.1
        delta = rng.random() * max_delta
        # propose move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        # normalize (should sum to 1)
        # (mass moved, sum unchanged)
        # compute score
        new_score = fast_score(a_new, fft_len, n)
        # decide acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        # cooling
        temperature = max(temperature * 0.9995, 1e-6)
    return best_a, best_score

def targeted_mass_redistribute(a, n, fft_len, deadline):
    while time.time() < deadline:
        # compute conv and contributions
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k = np.argmax(conv)
        # compute partner array:
        partner = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        partner[idx] = a[k - idx]
        # find i_max = argmax a[i]*partner[i]; i_min = argmin partner[i] not equal to i_max
        contrib = a * partner
        i_max = np.argmax(contrib)
        # find index where partner smallest but not same
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]
        # move small delta from i_max to i_min
        delta = min(a[i_max] * 0.5, 0.01)  # maybe adapt
        if delta <= 1e-12:
            break
        a[i_max] -= delta
        a[i_min] += delta
        # renormalize
        a /= a.sum()
    return a

def projected_subgradient_descent(a_start, deadline, n, fft_len, fft_len_g, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it=0
    while time.time()<deadline:
        # compute conv & mask (same as exp_gradient)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - max_val*1e-12).astype(float)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad = 2 * np.real(np.fft.ifft(fm * fa_rev))[ (n-1):(n-1+n) ]
        # step size adapt
        eta = 0.5 / np.sqrt(it+1)
        # gradient descent step (projected)
        a_new = a - eta * grad
        # enforce nonneg, project onto simplex
        a_new = np.maximum(a_new, 0)

"""
propose_candidate: hybrid search for a low‑value step‑function.
- Collect a diverse set of initial vectors (uniform, triangular,
  cosine, spectral, random, sparse, previous best).
- Run a short exponentiated‑gradient (mirror‑descent) phase.
- Perform many short restarts of the same EG routine from noisy
  perturbations of the current best (exploration).
- Refine with a projected sub‑gradient descent on the true max‑convolution.
- Finish with a deterministic hill‑climb that moves a small fraction of
  mass from the most harmful coordinate toward the least harmful one.
All phases respect the overall 1000 s budget and keep the vector on the
probability simplex (∑a = 1, a ≥ 0).  The final sequence is returned as
a plain Python list.
"""

import time
import numpy as np
from math import log

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = v.astype(np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial vectors
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random probability vector (different flavours)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric triangular shape – peak in the centre."""
    i = np.arange(n)
    tri = np.minimum(i + 1, n - i)                     # 1,2,...,mid,...,2,1
    a = tri.astype(float)
    a /= a.sum()
    return a


def cosine_initial(n: int) -> np.ndarray:
    """Positive half‑sine shape."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n)                 # strictly positive on (0,π)
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Square‑rooted flat spectrum → smooth start vector."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def random_sparse(n: int, rng: np.random.Generator,
                  min_sz: int = 30, max_sz: int = 2000) -> np.ndarray:
    """Uniform mass on a random subset of coordinates."""
    t = rng.integers(min_sz, max_sz + 1)
    idx = rng.choice(n, size=t, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0 / t
    return a


def add_noise(a: np.ndarray, sigma: float,
              rng: np.random.Generator) -> np.ndarray:
    """Gaussian perturbation + re‑projection onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    if noisy.sum() == 0:
        return a.copy()
    return noisy / noisy.sum()

# Optimization primitives
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size (classic 1/√t schedule, slightly enlarged)
        eta = 1.5 / np.sqrt(it + 1.0)

        # multiplicative update, safe clipping
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

        # early exit if we are already below the target
        if best_score < 1.503:
            break
    return best_a, best_score


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len_conv: int,
                                   fft_len_g: int) -> tuple[np.ndarray, float]:
    """Projected (Euclidean) sub‑gradient descent on max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # convolution and mask around the current maximum
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (same as in exp‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size – slower decay for better precision
        eta = 1.0 / (np.sqrt(it + 1.0) * 2.0)

        a_new = a - eta * g
        # keep on the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = best_a.copy()
        else:
            a_new /= s

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1
        if best_score < 1.503:
            break
    return best_a, best_score


def fine_hill_climb(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.2,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb moving a small fraction of mass from the
    coordinate that contributes most to the maximal convolution entry
    to the coordinate with the smallest partner value.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] where defined
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # pick destination with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()  # re‑normalise (numerical safety)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
        else:
            step = max(step * 0.8, min_step)
    return a, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    fft_len_g = next_pow2(3 * n - 2)

    candidates = [np.full(n, 1.0 / n),                     # uniform

    # a few random sparse seeds

    if 'height_sequence_1' in globals():

    # pick the best of the crude seeds

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:

            best_a = cand.copy()
    print(f"[init] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin
