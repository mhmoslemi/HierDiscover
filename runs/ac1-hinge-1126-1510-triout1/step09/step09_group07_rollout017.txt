def lp_refine(a_start, deadline, n, max_iter=2000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    row_idx, col_idx = build_convolution_matrix_indices(n)
    # convert indices to int32 arrays
    # some may be large; but row + col dimension is 2n-1 = 7999 < 2^31

    iter = 0
    while time.time() < deadline and iter < max_iter:
        # compute M
        # conv computing
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        # Build A matrix data for this iteration
        data = a[col_idx]  # shape n**2
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        # Solve LP
        c = -np.ones(n)  # maximize sum => minimize negative sum
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        try:
            # Use method high-level
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                         method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] Exception", e)
            break
        if not res.success:
            print("[LP] Not successful:", res.message)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            print("[LP] Zero sum, break")
            break
        g = g0 / sum_g0
        # update with step t
        t = 0.1  # constant step; could use schedule
        a = (1 - t) * a + t * g
        # ensure positivity and sum=1
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        # compute score
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter {iter}, new best fast_score={best_score:.8f}, M={M:.8f}")
        # optional break if improvement small
        if iter > 0 and abs(best_score - score) < 1e-8:
            # maybe break only after many no improvement?
            pass
        iter += 1
    return best_a, best_score

"""
Minimal step‑function search (non‑negative sequence).

The objective to minimise is
    2·n·max(conv(a,a)) / (sum(a)²)

Because the value is scale‑invariant we fix sum(a)=1 and minimise
    2·n·max(conv(a,a))

Two complementary optimisation ideas are used:

* **Linear‑programming alternating step** – a classic scheme from the
  literature: for the current distribution f we compute M=max(f*f).
  Keeping M fixed we solve the LP

        maximise   1·g
        subject to f*g ≤ M , g ≥ 0

  (the convolution f*g is linear in g).  The LP solution g₀ is
  normalised to sum(g)=1 and the current distribution is updated by a
  small convex combination with g.  This operation provably never
  increases the maximal self‑convolution and quickly flattens the peak.

* **Smooth‑max L‑BFGS polishing** – after a few LP iterations we polish
  the result using a log‑sum‑exp smoothed version of the max‑convolution.
  The smooth objective is differentiable; we optimise it with L‑BFGS‑B
  while respecting the simplex via a soft‑max parametrisation.
  A continuation on the smoothing parameter (λ) drives the solution
  arbitrarily close to the exact max.

Both phases run inside the 1000 s budget (with generous time splits);
the best vector found is finally hill‑climbed with very small mass moves.
The returned list is evaluated with the supplied `evaluate_sequence`
function.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """A spectral‑style start – inverse FFT of a square‑root spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Hann window (non‑negative, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a, cur_score, fft_len, n, deadline, max_iter=15000):
    """Very small strict‑improvement greedy walk."""
    best = a.copy()
    best_score = cur_score
    rng_local = np.random.default_rng(123456)
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng_local.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng_local.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def build_conv_matrix_dense(f):
    """
    Build the dense (2n‑1)×n convolution matrix A such that A·g = f*g.
    Column i contains the constant value f[i] on rows i … i+n‑1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_refine(a_start, deadline, n, fft_len):
    """
    Alternating linear‑programming refinement.
    Returns the best a encountered and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    iteration = 0
    max_iter = 120  # generous but still well within time budget
    eps = 1e-12

    while time.time() < deadline and iteration < max_iter:
        # keep a strictly positive to avoid unbounded LP columns
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        # current maximal self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # linear constraints f * g ≤ M
        A = build_conv_matrix_dense(a)               # (2n‑1)×n
        c = -np.ones(n)                              # maximise sum(g)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                             bounds=bounds, method='highs',
                             options={'presolve': True})
        except Exception as e:
            print("[LP] exception:", e)
            break

        if not lp_res.success:
            print("[LP] failure:", lp_res.message)
            break

        g0 = lp_res.x
        s = g0.sum()
        if s <= eps:
            print("[LP] zero‑sum solution, abort.")
            break

        g = g0 / s               # normalise to sum=1

        # convex update – step size t can be constant (0.1 works well)
        t = 0.1
        a = (1.0 - t) * a + t * g

        # evaluate
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter {iteration:3d} → fast_score={best_score:.8f}")

        iteration += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy usage safety

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 2000                      # still large enough for a good bound
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # previous best (if supplied)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic starts
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n))

    # random draws
    for _ in range(4):
        candidates.append(random_initial(n, rng))

    # pick the best of the pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    if time.time() < phase1_deadline:
        best_a, best_score = lp_refine(best_a, phase1_deadline, n, fft_len)
        print(f"[LP] after refinement best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS polishing (smooth‑max continuation)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        lam_schedule = [
            0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
            0.05, 0.03, 0.018, 0.01, 0.006,
            0.003, 0.0015, 0.0008, 0.0004, 0.0002,
            0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
            2e-06, 1e-06
        ]
        maxiter_per_lam = 800
        best_a, best_score = lbfgs_continuation(best_a,
                                                lam_schedule,
                                                maxiter_per_lam,
                                                phase2_deadline,
                                                fft_len, fft_len,
                                                n)
        print(f"[L‑BFGS] after polishing best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        deadline,
                                        max_iter=25000)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence will do the final scaling)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
