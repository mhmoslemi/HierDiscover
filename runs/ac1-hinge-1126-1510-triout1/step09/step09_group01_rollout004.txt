"""
Multi‑phase search for a low‑score non‑negative step sequence.

The algorithm improves on the reference implementation by:

* Adding a richer set of parametric families for the cheap sweep.
* Using a symmetrisation step (a ← (a+a[::-1])/2) after every major
  optimisation phase – the optimum is invariant under reversal, and
  symmetry speeds up convergence.
* Running many Adam restarts with a finer λ‑schedule and more iterations.
* Fine‑tuning the best candidate with L‑BFGS‑B on the (almost) exact
  max‑convolution (λ ≈ 10⁻⁶·max).
* A longer projected sub‑gradient descent with momentum.
* A simulated‑annealing hill‑climb that redistributes mass among three
  positions instead of two.
* Careful time‑budget handling so the whole routine finishes well
  under the 1000 s limit.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0) and
the final score is obtained by calling the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT; returns 2·len(a)‑1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised to ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeroes – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[-i-1] and renormalise to sum 1."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s <= 0 or np.isnan(s):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_sym / s


# ----------------------------------------------------------------------
# Parametric families (used in the cheap sweep)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0003, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(150.0, 2500.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(150.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "beta":
        # a_i ∝ (i+1)^{α-1} (n-i)^{β-1}
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    elif family == "triangular":
        # symmetric triangle, higher in the centre
        i = np.arange(n, dtype=float)
        a = np.minimum(i + 1, n - i)
    elif family == "sinpow":
        p = rng.uniform(0.2, 5.0)
        a = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n) ** p
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0 or np.isnan(total):
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families.
    Returns the best vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.12  # a bit less than 12 % of total time
    families = ["power", "exp", "gauss", "cos", "bimodal", "beta",
                "triangular", "sinpow", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (linear, using zero‑padding)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # the part that corresponds to the original vector
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y

def adam_optimize(
    a_start: np.ndarray,
    lam_factors,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max objective.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps)   # a bit larger than before
            y -= step * m_hat

            a = softmax(y)
            a = symmetrize(a)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def lbfgs_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B fine‑tuning on the (almost) exact smoothed max.
    """
    if time.time() >= deadline:
        return a_start, fast_score(a_start, fft_len_conv, n)

    y0 = np.log(a_start + 1e-12)

    def func(y):
        """Return (objective, gradient) for the optimizer."""
        a = softmax(y)

        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv.max()
        lam = max_c * lam_factor + 1e-12

        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        smooth_max = max_c + lam * np.log(Z)
        f_val = 2.0 * n * smooth_max

        w = exp_shift / Z

        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        fw = np.fft.fft(w, fft_len_g)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]

        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        return f_val, grad_y

    options = {"maxiter": max_iter, "disp": False}
    res = minimize(func, y0, method="L-BFGS-B", jac=True, options=options)

    a_opt = softmax(res.x)
    a_opt = symmetrize(a_opt)
    return a_opt, fast_score(a_opt, fft_len_conv, n)


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # sub‑gradient mask

        # sub‑gradient of max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * g
        step = 0.9 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)
        a_candidate = symmetrize(a_candidate)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction of the mass among
    three positions. Temperature decays linearly to zero.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        # pick three distinct indices
        idx = rng.choice(n, size=3, replace=False)
        i, j, k = idx[0], idx[1], idx[2]

        # ensure we take mass from a position that actually has mass
        if a[i] == 0.0:
            continue

        # move a random fraction of a[i] into j and k
        delta = a[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        # split delta randomly between j and k
        split = rng.random()
        a_new[j] += delta * split
        a_new[k] += delta * (1.0 - split)

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < math.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size (feel free to increase n if you have spare time)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # initialise from possible previous construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 = a1 / a1.sum()
            a1 = symmetrize(a1)
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 12 % of total budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # allocate time slots (tunable)
    phase1_end = start_time + 0.45 * budget_s   # Adam + many restarts
    phase2_end = start_time + 0.70 * budget_s   # L‑BFGS fine‑tune
    phase3_end = start_time + 0.90 * budget_s   # sub‑gradient momentum
    # remaining time -> simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – many Adam restarts on a smoothed max objective
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.6, 2e-6, num=35)   # smoother to near‑exact
    iters_per_lam = 600

    while time.time() < phase1_end:
        # choose a start point: jitter best or fresh random family
        if rng.random() < 0.35:
            jitter = rng.normal(1.0, 0.02, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS‑B fine‑tuning on an almost exact max
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # use a very small lambda (≈1e‑6 of current max) to approximate true max
        lam_factor = 1e-6
        max_iter = 150  # a few L‑BFGS iterations are enough for a good tweak
        best_a, best_score = lbfgs_optimize(
            best_a,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factor,
            max_iter,
            phase2_end,
        )
        print(f"[L‑BFGS] fine‑tuned fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with momentum (true max)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
