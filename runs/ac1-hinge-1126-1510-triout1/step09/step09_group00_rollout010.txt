def propose_candidate(seed: int = 42, budget_s: int = 1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    n = 4000
    fft_len = next_pow2(2*n - 1)
    start = time.time()
    deadline = start + budget_s - 0.1
    # baseline uniform
    best_a = np.full(n, 1.0/n)
    best_score = fast_score(best_a, fft_len, n)
    print('[init] uniform fast_score=...', best_score)
    # parametric sweep binary masks
    sweep_deadline = start + 0.05 * budget_s
    while time.time() < sweep_deadline:
        density = rng.uniform(0.5, 0.9)
        mask = rng.random(n) < density
        if not mask.any():
            continue
        m = mask.sum()
        a = mask.astype(float) / m
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            best_mask = mask.copy()
            print(f'[binary sweep] new best fast_score={sc:.6f} density={density:.2f}')
    # if best_a from binary, we have best_mask; else compute mask from best_a
    if 'best_mask' not in locals():
        # convert best_a to mask approximating non-zero values > 1e-12
        best_mask = best_a > 1e-12
    # annealing
    anneal_deadline = deadline
    cur_mask = best_mask.copy()
    cur_score = best_score
    best_mask_local = cur_mask.copy()
    best_sc_local = cur_score
    # compute temperature
    T0 = max(0.05 * cur_score, 1e-6)
    t_start = time.time()
    total_anneal_time = max(anneal_deadline - t_start, 1e-6)
    # precompute list of all indices for convenience
    all_indices = np.arange(n)
    while time.time() < anneal_deadline:
        # compute a and conv
        m = cur_mask.sum()
        a = cur_mask.astype(float) / m
        conv = full_conv(a, fft_len)
        # find max index
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        # find indices i in support where partner also in support
        # compute partner indices
        partner = k_max - np.arange(n)
        # valid i within [0,n) and partner within [0,n)
        valid = (partner >= 0) & (partner < n)
        i_candidates = np.where(cur_mask & valid & cur_mask[partner])[0]
        if i_candidates.size == 0:
            # fallback to random source
            i_src = rng.choice(np.where(cur_mask)[0])
        else:
            i_src = rng.choice(i_candidates)
        # destination from complement
        non_support = np.where(~cur_mask)[0]
        if non_support.size == 0:
            break
        i_dst = rng.choice(non_support)
        # propose swap
        new_mask = cur_mask.copy()
        new_mask[i_src] = False
        new_mask[i_dst] = True
        # compute new score
        a_new = new_mask.astype(float) / m
        sc_new = fast_score(a_new, fft_len, n)
        diff = sc_new - cur_score
        # temperature schedule
        remain = max(anneal_deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_anneal_time)
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_mask = new_mask
            cur_score = sc_new
            if sc_new < best_sc_local:
                best_sc_local = sc_new
                best_mask_local = new_mask.copy()
                print(f'[anneal] new best fast_score={best_sc_local:.6f}')
        # occasional random restart if stuck
        if rng.random() < 0.001:
            # restart with random binary mask
            density = rng.uniform(0.5, 0.9)
            mask = rng.random(n) < density
            cur_mask = mask.copy()
            m = cur_mask.sum()
            cur_score = fast_score(cur_mask.astype(float) / m, fft_len, n)
    # final best vector
    best_a = best_mask_local.astype(float) / best_mask_local.sum()
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f'[final] evaluated_score={final_eval:.6f}')
    return final_seq

"""
Propose a step‑function sequence that minimizes

    2·n·max(a*a) / (sum(a)**2)

The algorithm is deliberately different from the previous baseline.
It focuses on *binary‑sparse* candidates (all non‑zero entries have the same
weight) because for a dense random support the convolution sum at any shift
behaves like a binomial distribution whose peak is ~1/(2 n), i.e. a fast
score ≈ 1.0 – far below the 1.506 obtained before.

Steps
-----
1. **Binary‑sparse sweep** – generate many random 0/1 masks with a
   density in [0.5, 0.9] (each 1 gets weight 1/|support|). Keep the best
   mask seen.

2. **Targeted simulated annealing** – starting from the best mask,
   repeatedly swap a “bad” index (one that contributes to the current
   maximal convolution bin) with a random zero‑position.  Swaps that lower
   the score are always kept; uphill moves are accepted with a slowly
   cooling temperature.

3. **Optional fine‑tuning** – a few exponentiated‑gradient steps on the
   resulting (now dense) vector can shave off tiny residuals.

The whole search respects a global deadline (default 1000 s).  The final
vector is returned as a plain Python list; `evaluate_sequence` is called
once at the end for the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    2·n·max(a*a) for a sequence already normalised (∑a = 1).
    a must be a 1‑D NumPy array of non‑negative numbers that sum to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def fast_score_mask(mask: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast score for a binary mask (0/1).  All active entries receive
    equal weight 1/|mask|, therefore
        max(a*a) = max_cnt / |mask|²
    where max_cnt is the maximal count of (i,j) with i+j=k.
    """
    m = mask.sum()
    if m == 0:
        return np.inf
    conv_cnt = full_conv(mask.astype(float), fft_len)  # counts, not weighted
    max_cnt = conv_cnt.max()
    return float(2.0 * n * max_cnt / (m * m))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Binary‑sparse random search + targeted annealing
# ----------------------------------------------------------------------


def binary_mask_sweep(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly generate binary masks with density in [0.5, 0.9] until
    ``deadline``.  Return the mask with the smallest fast_score.
    """
    best_mask = np.zeros(n, dtype=bool)
    best_score = np.inf
    while time.time() < deadline:
        density = rng.uniform(0.5, 0.9)
        mask = rng.random(n) < density
        # ensure at least one entry
        if not mask.any():
            mask[0] = True
        sc = fast_score_mask(mask, fft_len, n)
        if sc < best_score:
            best_score, best_mask = sc, mask.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} density={density:.2f}")
    return best_mask, best_score


def binary_targeted_anneal(
    mask: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that swaps a mask element that participates in the
    current maximal convolution bin with a random zero‑position.
    """
    cur_mask = mask.copy()
    cur_score = fast_score_mask(cur_mask, fft_len, n)
    best_mask = cur_mask.copy()
    best_score = cur_score

    # temperature schedule (square‑root cooling)
    T0 = max(0.05 * cur_score, 1e-6)
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    all_idx = np.arange(n)

    while time.time() < deadline:
        m = cur_mask.sum()
        if m == 0:
            break

        # --- compute convolution counts for the current mask ---
        conv_cnt = full_conv(cur_mask.astype(float), fft_len)
        k_max = int(conv_cnt.argmax())
        max_cnt = conv_cnt.max()

        # indices that contribute to the current maximal bin
        # i contributes iff mask[i] and mask[k_max-i] are both True
        partner = k_max - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        cand_src = np.where(cur_mask & valid & cur_mask[partner])[0]

        if cand_src.size == 0:
            # fallback: any active index
            cand_src = np.where(cur_mask)[0]

        src = rng.choice(cand_src)

        # pick a destination that is currently inactive
        inactive = np.where(~cur_mask)[0]
        if inactive.size == 0:
            break
        dst = rng.choice(inactive)

        # propose the swap
        new_mask = cur_mask.copy()
        new_mask[src] = False
        new_mask[dst] = True

        new_score = fast_score_mask(new_mask, fft_len, n)
        diff = new_score - cur_score

        # cooling temperature
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_mask = new_mask
            cur_score = new_score
            if new_score < best_score:
                best_score, best_mask = new_score, new_mask.copy()
                print(f"[anneal] new best fast_score={best_score:.6f}")

        # occasional random restart to avoid getting stuck
        if rng.random() < 0.001:
            density = rng.uniform(0.5, 0.9)
            cur_mask = rng.random(n) < density
            cur_score = fast_score_mask(cur_mask, fft_len, n)

    return best_mask, best_score


def exp_gradient_fine_tune(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> np.ndarray:
    """
    A short exponentiated‑gradient run (only when some time is left)
    to polish the final dense vector.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.3 / np.sqrt(it + 1.0)    # diminishing step
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0:
            a /= s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (feel free to change – larger n gives a lower theoretical bound)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – binary‑sparse random sweep (≈5 % of time)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 0.05 * budget_s
    mask, mask_score = binary_mask_sweep(n, fft_len, sweep_deadline, rng)
    if mask_score < best_score:
        best_score = mask_score
        best_a = mask.astype(float) / mask.sum()
        print(f"[phase0] binary sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – targeted simulated annealing on the best mask
    # --------------------------------------------------------------
    anneal_deadline = deadline
    mask, mask_score = binary_targeted_anneal(mask, fft_len, n, anneal_deadline, rng)
    a_from_mask = mask.astype(float) / mask.sum()
    if mask_score < best_score:
        best_score, best_a = mask_score, a_from_mask.copy()
        print(f"[phase1] annealing improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – optional fine‑tuning with exponentiated gradient (if time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        fine_deadline = deadline
        best_a = exp_gradient_fine_tune(best_a, fine_deadline, fft_len, n)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[phase2] fine‑tuned fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final check with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
