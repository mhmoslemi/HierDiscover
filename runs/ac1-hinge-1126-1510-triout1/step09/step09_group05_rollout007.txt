def exp_mirror_descent(a0, deadline, n, fft_len_conv, fft_len_g, max_iter=200000):
    a = a0.copy()
    y = np.log(a + 1e-12)  # log-probabilities
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    eta0 = 0.5
    while time.time() < deadline and it < max_iter:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        # adapt step size:
        eta = eta0 / np.sqrt(it + 1)
        y = y - eta * grad
        # shift for numeric
        y = y - np.max(y)
        a = np.exp(y)
        # Project onto simplex automatically by normalization
        a = a / a.sum()
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[ExpGrad] it={it} score={best_score:.6f}")
        it += 1
    return best_a, best_score

"""
Propose a non‑negative step‑function (vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear convolution.  The objective is invariant
under scaling of the vector, therefore the optimiser works on the
simplex {a ≥ 0, Σa = 1} and uses the fast surrogate 2·n·max(conv) (the
denominator is 1).

The optimiser runs inside the 1000 s budget in four stages:

1. **Heavy L‑BFGS** on a smooth log‑sum‑exp approximation of the
   max‑convolution.  A decreasing schedule of smoothing parameters
   drives the solution towards the true non‑smooth objective.

2. **Projected Adam** on the exact max‑convolution sub‑gradient.
   This stage refines the solution with an adaptive learning rate.

3. **Exponentiated‑gradient (mirror‑descent)** on the exact sub‑gradient.
   Updating the log‑probabilities (y = log a) respects the simplex
   constraints automatically and often escapes shallow local minima.

4. **Simulated‑annealing hill‑climb** for a final polish: moves a small
   fraction of mass from indices with a large sub‑gradient to those
   with a small one, accepting occasional uphill steps.

All stages are timed; the best vector seen at any moment is returned.
The implementation works for any length `n`; a relatively large value
(≈10 000) is used so that the theoretical lower bound 2·n·avg(conv)≈1 can
be approached.  The code makes no use of the file‑system, imports only
standard scientific packages and respects the 1000 s time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) under the assumption Σa = 1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random point on the simplex.
    A few hand‑crafted families are mixed to provide many different
    shapes (uniform, exponential, power‑law, edge‑heavy, bimodal…).
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "spike"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike – a single heavy entry + tiny background
        a = np.full(n, 1e-8)
        idx = rng.integers(n)
        a[idx] = 1.0 - (n - 1) * 1e-8
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Surrogate of 2·n·max(conv) using log‑sum‑exp.
    The variable y represents unnormalised logits; a = softmax(y) guarantees Σa=1.
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) for the exact max‑convolution of a with itself.
    The sub‑gradient corresponds to the sum of contributions from all shifts
    attaining the maximal value (within a very small tolerance).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam on the true (non‑smooth) objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the exact sub‑gradient of the max‑convolution.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)   # a slowly decaying step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to shake out of plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact objective
# ----------------------------------------------------------------------


def exp_mirror_descent(
    a0: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with the log‑entropy (i.e. exponentiated
    gradient).  The update uses the exact sub‑gradient of max(conv).
    """
    a = a0.copy()
    y = np.log(a + 1e-12)          # log‑probabilities
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    eta0 = 0.5

    while time.time() < deadline and it < max_iter:
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # simple diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        y = y - eta * grad
        # re‑normalise logits to keep numbers in a comfortable range
        y = y - np.max(y)

        a = np.exp(y)
        a = a / a.sum()            # automatic projection on the simplex

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[ExpGD] it={it} new best={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a simple simulated‑annealing acceptance rule.
    Moves mass from high‑gradient indices to low‑gradient ones, also tries
    swaps.  Returns the best vector found.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9995
    min_temp = 1e-9

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact sub‑gradient for the current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source / target indices based on the magnitude of the gradient
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.6 and high_idx.size and low_idx.size:
            # transfer a fraction of mass
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.04, 0.18)
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            # simple swap of two entries
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:

            a0 = np.interp(x_new, x_old, old)

            else:

    else:

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2      # tiny safety margin

    phase2_end = start + 0.80 * (budget_s - 0.2)   # exponentiated‑gradient
    # the remainder → annealing

    # ------------------------------------------------------------------
    # Phase 0 – heavy L‑BFGS (smooth surrogate)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.15, 0.09, 0.06, 0.035, 0.02,

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase0_end,
            fft_len_conv,
            fft_len_g,

        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Projected Adam (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = subgradient_adam(
            best_a, phase1_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient (mirror descent) on exact sub‑gradient
    # ------------------------------------------------------------------

            best_a,

            fft_len_conv,
            fft_len_g,
            max_iter=300_000,
        )
        print(f"[ExpGD] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing / hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            total_deadline,
            max_iters=500_000,
        )
        print(f"[Anneal] final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts later
    globals()["height_sequence_1"] = final_seq

    return final_seq
