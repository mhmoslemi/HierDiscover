"""
Propose a low‑score step sequence.

Algorithm
---------
* Use a moderately large problem size (n≈8000) – more variables give a
  finer trade‑off between the peak of the self‑convolution and the L1
  normalisation.
* Initialise a diverse pool (uniform, exponential, power‑law, Hann,
  Gaussian).  Keep the best candidate in the simplex (∑a=1).
* Phase 1 – short L‑BFGS runs on a log‑sum‑exp smoothed max‑convolution.
  A fine λ‑schedule drives the surrogate to the exact max.
* Phase 2 – long exponentiated‑gradient descent on the exact max‑convolution.
* Phase 3 – a linear‑programming refinement step that follows the
  “active‑set” idea from the reference: maximise the L1‑norm of a new
  vector g while keeping (a * g)≤M (M=current peak).  Mix the result with
  the current vector.
* Phase 4 – projected sub‑gradient descent, simple simulated annealing,
  and a final greedy hill‑climb.
* All phases respect a 1000 s total budget; the best vector found is
  returned.  The evaluate_sequence function (already imported) is used
  only for the final official score.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (∑a=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(1e-4, 5e-3)          # small decay → nearly uniform
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def power_init(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 1.8)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.10, 0.30) * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of a*a, returns (value, gradient w.r.t. logits y)."""
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z

    # back‑prop through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run L‑BFGS for each λ in the schedule."""
    y = np.log(a_start + 1e-12)        # stay in the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """Projected sub‑gradient descent."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    step0 = 0.3
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 80000):
    """Simple SA with small mass transfers."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    total_moves = max_moves
    move = 0
    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        score_candidate = fast_score(a_candidate, fft_len, n)

        if score_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(score_candidate - best_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               n: int,
               fft_len: int,
               deadline: float,
               max_iter: int = 20000):
    """Strictly improving random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng_local = np.random.default_rng(123456)
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng_local.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng_local.random() * 0.07
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set idea)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              n: int,
              fft_len: int,
              deadline: float,
              rng: np.random.Generator,
              top_k: int = 30,
              alpha: float = 0.45):
    """
    One LP refinement step:
    maximise sum(g) subject to g≥0 and (a*g) ≤ M on a small set of
    indices (the current active constraints).  Return a convex mixture
    (1‑α)·a + α·g_normalised.
    """
    # compute current peak
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    M = float(conv.max())

    # choose active indices – either the K largest values or those
    # within a tiny tolerance of the maximum
    if top_k >= len(conv):
        act_idx = np.arange(len(conv))
    else:
        # indices of the K largest entries
        act_idx = np.argpartition(conv, -top_k)[-top_k:]

    # ensure the extreme ends are present (they give a bound on every variable)
    extra = np.array([0, 2 * n - 2])
    act_idx = np.unique(np.concatenate([act_idx, extra]))

    K = len(act_idx)

    # build constraint matrix A (K × n)
    # each row k: (a * g)[k] = Σ_i a_i * g_{k‑i}
    rows = []
    for k in act_idx:
        # i runs over indices where both i and k‑i lie in [0,n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # number of terms = i_max‑i_min+1 (≤ n)
        cols = np.arange(k - i_max, k - i_min + 1, dtype=int)
        data = a[i_min:i_max + 1]
        rows.append(sp.csr_matrix((data, (np.zeros_like(cols), cols)),
                                  shape=(1, n)))
    A_ub = sp.vstack(rows).tocsr()

    # linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n)

    bounds = [(0.0, None) for _ in range(n)]

    # solve
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.full(K, M),
                      bounds=bounds, method='highs')
    except Exception as e:
        print("[LP] linprog exception:", e)
        return a, M

    if not res.success:
        # fallback – no improvement
        return a, M

    g = res.x
    # normalise g to the simplex (sum=1)
    if g.sum() > 0:
        g = g / g.sum()
    else:
        g = uniform_init(n)

    # convex combination
    a_new = (1.0 - alpha) * a + alpha * g
    a_new = simplex_projection(a_new)
    new_score = fast_score(a_new, fft_len, n)
    return a_new, new_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8000                     # larger than before – more flexibility
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidates
    # --------------------------------------------------------------
    candidates = []

    # 0. previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 1. uniform
    candidates.append(uniform_init(n))
    # 2. exponential (different decay rates)
    for _ in range(2):
        candidates.append(exponential_init(n, rng))
    # 3. power‑law
    for _ in range(2):
        candidates.append(power_init(n, rng))
    # 4. Hann and Gaussian
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n, rng))

    # pick best of the pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (short, many restarts)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        # draw a fresh start: blend best with a random vector
        rand_vec = random_initial(n, rng) if 'random_initial' in globals() else exponential_init(n, rng)
        alpha = rng.random() * 0.5
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming refinement (active‑set)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    while time.time() < phase3_deadline:
        best_a, new_score = lp_refine(best_a, n, fft_len,
                                      phase3_deadline, rng,
                                      top_k=30, alpha=0.45)
        if new_score < best_score:
            best_score = new_score
            print(f"[LP] improved fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.90 * budget_s
    if time.time() < phase4_deadline:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase4_deadline,
                                                  n,
                                                  fft_len)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-7,
                                                 max_moves=120000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score,
                                        n, fft_len,
                                        deadline,
                                        max_iter=30000)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the evaluate_sequence function will re‑scale)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
