def lp_refine(a_start, t_mix, active_tol, max_iters, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute self-convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        # active indices
        eps = max(active_tol, M*1e-12)
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])
        # Build constraint matrix A_ub
        k = active.size
        A = np.zeros((k, n), dtype=np.float64)
        for row_idx, t in enumerate(active):
            start = max(0, t - (n - 1))
            end = min(n-1, t)
            # j indexes
            j = np.arange(start, end+1)
            i = t - j
            A[row_idx, start:end+1] = a[i]
        b = np.full(k, M)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        # solve LP
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: uniform random perturbation
            a = a * (1 - t_mix) + t_mix * random_initial(n, rng)
            a /= a.sum()
            continue
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            continue
        g = g / sum_g
        # mix
        a = (1 - t_mix) * a + t_mix * g
        # ensure normalization
        a = a / a.sum()
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score

"""
Propose a low‑score step function.

Algorithm
---------
1. Generate a pool of diverse initial candidates (uniform,
   spectral, random Dirichlet‑like, triangular, Hann, Gaussian,
   plus interpolation of a previously found solution if available).
2. Run several short L‑BFGS runs on a log‑sum‑exp smoothed
   approximation of the max‑convolution.  The smoothing parameter
   (λ) is driven down to 10⁻⁶ of the current max, giving a very
   accurate gradient.
3. Perform long exponentiated‑gradient descent on the exact
   max‑convolution (multiplicative updates keep the simplex).
4. Refine the solution by the classic active‑set linear‑programming
   step described in the state‑of‑the‑art:  maximise Σ gⱼ
   subject to f * g ≤ M (M = max f * f) on the currently active
   shift positions only, then mix f←(1 ‑ τ)f + τg.
5. A short projected sub‑gradient phase and a final simulated‑annealing /
   greedy hill‑climb polish the result.
All phases respect a 1000 s overall budget and keep the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (Σ a = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    # avoid overflow / underflow
    shifted = np.clip(shifted / lam, -50, 50)
    exp_shift = np.exp(shifted)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

# Initialisation helpers

def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the
    FFT of a uniform target (length 2·n‑1), then project onto simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random/simple families on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                         # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window, shifted to be non‑negative."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # compute exact self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (decreasing)
        eta = 0.7 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """
    Projected sub‑gradient descent with a diminishing step size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    step0 = 0.3

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction with zero‑sum constraint
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 60000):
    """
    Simple SA: move a small fraction of mass between random indices.
    Accept worsening moves with probability exp(-Δ/T).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    total_moves = max_moves
    move = 0

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.10
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        score_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if score_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(score_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a_start: np.ndarray,
               cur_score: float,
               n: int,
               fft_len: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 20000):
    """
    Strictly improving greedy hill‑climb (small mass transfers).
    """
    best = a_start.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑programming refinement (state‑of‑the‑art step)
# ----------------------------------------------------------------------


def lp_active_set_refine(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         mix_factor: float = 0.15,
                         max_iter: int = 100):
    """
    Perform the active‑set LP refinement described in the reference.
    Repeatedly:
        * compute M = max(f*f)
        * collect active shift indices where f*f ≈ M
        * solve   maximise Σ gⱼ   s.t.   f * g ≤ M  on the active shifts, g ≥ 0
        * rescale g to Σ g = 1
        * mix  f ← (1‑τ) f + τ g   (τ = mix_factor)
    Stops when time expires or no improvement.
    Returns the best a found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active set (tight constraints)
        eps = max(1e-12, M * 1e-12)
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        # build constraint matrix A (rows = active shifts)
        k = active.size
        A = np.zeros((k, n), dtype=np.float64)
        for row_idx, t in enumerate(active):
            start = max(0, t - (n - 1))
            end = min(n - 1, t)
            # indices j = start … end
            j_idx = np.arange(start, end + 1)
            i_idx = t - j_idx               # corresponding i for f[i]
            A[row_idx, start:end + 1] = a[i_idx]

        b = np.full(k, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)    # maximise Σ gⱼ

        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # fall back to a tiny random perturbation and continue
            a = (1 - mix_factor) * a + mix_factor * random_initial(n, rng)
            a /= a.sum()
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            # numerical safety – perturb a little
            a = (1 - mix_factor) * a + mix_factor * random_initial(n, rng)
            a /= a.sum()
            continue

        g = g / sum_g
        # convex combination
        a = (1 - mix_factor) * a + mix_factor * g
        a /= a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # gradually shrink mixing factor to fine‑tune
            mix_factor = max(0.03, mix_factor * 0.97)

    return best_a, best_score

            x_new = np.linspace(0, 1, n)

            if s > 0:

    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------

        if sc_opt < best_score:

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (long)

    # ------------------------------------------------------------------
    # Phase 3 – Active‑set LP refinement (state‑of‑the‑art step)
    # ------------------------------------------------------------------

                                                  phase3_deadline,
                                                  n,
                                                  fft_len,
                                                  rng,
                                                  mix_factor=0.15,

                                                  phase4_deadline,
                                                  n,
                                                  fft_len)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing + final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < overall_deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 overall_deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=6e-4,
                                                 final_temp=5e-8,
                                                 max_moves=80000)
        print(f"[SA] best fast_score={best_score:.8f}")

    if time.time() < overall_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        n,
                                        fft_len,
                                        overall_deadline,
                                        rng,
                                        max_iter=30000)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence will re‑scale internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.8f}")
    except Exception as e:
        # In the unlikely event evaluate_sequence is not available,
        # just return the raw sequence.
        print("[final] evaluate_sequence not callable:", e)
    return final_seq
