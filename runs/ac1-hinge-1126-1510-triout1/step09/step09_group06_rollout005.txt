def lp_step(a, M, active_set, t):
    n = len(a)
    num_constraints = len(active_set)
    A = np.zeros((num_constraints, n))
    for row_idx, k in enumerate(active_set):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # iterate i
        for i in range(i_start, i_end + 1):
            j = k - i
            A[row_idx, j] += a[i]
    # Solve max sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(num_constraints, M), bounds=bounds, method='highs')
    if not res.success:

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a, M
    # scale to have sum 1

"""
Propose a low‑score step function.

The algorithm builds a diverse pool of initial candidates (previous best,
random families, beta‑shaped densities, etc.).  The best seed is polished
with a short L‑BFGS run on a smooth log‑sum‑exp surrogate of the max‑
convolution, followed by a brief exponentiated‑gradient phase.

The new ingredient is an **active‑set linear‑programming refinement**.
Given the current sequence `a` and its maximal self‑convolution value
`M = max(a*a)`, we solve a small LP:

    maximise   Σ b_i
    subject to b_i ≥ 0
               (a * b)_k ≤ M   for every k where (a*a)_k is (almost) maximal

The LP pushes mass to locations that are not “dangerous’’ with respect
to the current peak `M`.  After scaling the LP solution to unit sum we
mix it with the old sequence (convex combination).  Repeating this step
with a diminishing mixing weight monotonically lowers the objective.
A final greedy hill‑climb performs a last polish.

All operations respect the 1000 s budget and the function returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1  ⇒  score = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution (log‑sum‑exp)."""
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS over a schedule of smoothing parameters.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """Greedy hill‑climb: transfer a small random fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (various families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """Edge‑heavy shape."""
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    xs = np.linspace(0, 1, n)
    eps = 1e-12
    a = (xs + eps) ** (alpha - 1) * (1 - xs + eps) ** (alpha - 1)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """Explore a few parametric families and keep the best `keep` candidates."""
    candidates = []

    # Exponential family
    for c in np.logspace(-4, -0.5, num=12):
        a = exponential_shape(n, c)
        candidates.append((fast_score(a, fft_len, n), a))

    # Power‑law family
    for p in np.linspace(0.1, 4.0, num=15):
        a = powerlaw_shape(n, p)
        candidates.append((fast_score(a, fft_len, n), a))

    # Edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # Beta‑shape (edge‑heavy) family
    for alpha in np.linspace(0.05, 0.5, num=12):
        a = beta_shape(n, alpha)
        candidates.append((fast_score(a, fft_len, n), a))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # Keep the best distinct candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 8)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

# Active‑set LP refinement
# ----------------------------------------------------------------------


def build_active_matrix(a: np.ndarray, active_k: np.ndarray) -> np.ndarray:
    """
    Build dense matrix A where A[row, col] = a[i] with i+col = k
    for each k in active_k (row index).  Size = (len(active_k), len(a)).
    """
    n = a.shape[0]
    m = active_k.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_k):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over indices of a that participate
        for i in range(i_start, i_end + 1):
            j = k - i
            A[row, j] += a[i]
    return A


def lp_step(a: np.ndarray,
            M: float,
            active_k: np.ndarray,
            mix_weight: float,
            n: int,
            fft_len: int):
    """
    Perform one LP based refinement step.
    Returns new a and its max‑convolution value.
    """
    # Build constraint matrix for the *active* indices only
    A = build_active_matrix(a, active_k)

    # Linear program:
    # maximise   Σ b_i    ⇔ minimise  -Σ b_i
    # subject to   A @ b ≤ M
    #               b ≥ 0
    c = -np.ones(n, dtype=np.float64)                # minimise -sum(b)
    ub = np.full(A.shape[0], M, dtype=np.float64)   # RHS = M
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: keep current a
        return a, M

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a, M
    # Scale b to have unit sum (same normalisation as a)
    b = b / sum_b

    # Convex combination
    a_new = (1.0 - mix_weight) * a + mix_weight * b
    a_new = a_new / a_new.sum()

    # New max‑convolution
    fa = np.fft.rfft(a_new, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    M_new = conv.max()
    return a_new, M_new


def lp_iterative_improve(a: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         init_mix: float = 0.5,
                         min_mix: float = 0.02,
                         tol: float = 1e-8):
    """
    Repeated active‑set LP refinement until time expires or improvement stalls.
    Returns the best a encountered.
    """
    # initial convolution and score
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    mix = init_mix
    it = 0

    while time.time() < deadline:
        # Determine active set (indices where conv is (almost) maximal)
        eps = max(M * 1e-12, 1e-14)
        active_k = np.where(conv >= M - eps)[0]
        if active_k.size == 0:
            break

        a_new, M_new = lp_step(a, M, active_k, mix, n, fft_len)
        score_new = 2.0 * n * M_new

        if score_new < best_score - tol:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            M = M_new
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[: 2 * n - 1]
            # keep the mixing weight relatively large while we are improving
            mix = init_mix
        else:
            # No improvement – shrink the mixing weight
            mix = max(mix * 0.6, min_mix)
            if mix <= min_mix:
                # If we cannot make further progress, stop
                break

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                         # length of the step function (tunable)
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------
    # Gather initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # deterministic seeds
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # sweep families for extra diversity
    candidates.extend(sweep_parameter_families(n, fft_len, rng, keep=6))

    # ------------------------------------------------------------------
    # Keep the best initial seed
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – brief L‑BFGS polishing (diverse seeds)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.12 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 500

    seeds = [best_a]
    for _ in range(2):
        seeds.append(random_initial(n, rng))

    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len,
                                            n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (quick boost)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.30 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len)
        print(f"[exp‑grad] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – active‑set LP refinement (the new ingredient)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = lp_iterative_improve(best_a,
                                                  n,
                                                  fft_len,
                                                  phase3_deadline,
                                                  init_mix=0.5,
                                                  min_mix=0.02)
        print(f"[LP‑refine] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (as many moves as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=50000)
        print(f"[hill‑climb] final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
