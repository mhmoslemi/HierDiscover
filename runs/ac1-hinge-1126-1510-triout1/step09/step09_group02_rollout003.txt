        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

"""
Search for a low‑value step‑function sequence.

The optimizer combines three ideas:
1.  Randomly generated initial distributions (uniform, exponential,
    power‑law, Beta‑shaped, edge‑heavy, sparse) are refined by a *projected
    sub‑gradient descent* on the exact max‑convolution objective.
2.  Starting from the best candidate, a *gradient‑guided hill‑climb* moves
    mass from the index with the largest sub‑gradient to the one with the
    smallest, shrinking the step‑size over time.
3.  A final aggressive *incremental hill‑climb* (as in the reference code)
    explores the neighbourhood exhaustively.

All phases respect the 1000 s time budget, keep the vector non‑negative,
and maintain the simplex constraint (∑a = 1).  The best sequence found
before the deadline is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Return 2·n·max(conv(a,a)) for a probability vector a (∑a = 1).
    Uses FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a diverse distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'beta', 'edge', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        alpha = rng.uniform(0.3, 2.0)
        beta = rng.uniform(0.3, 2.0)
        x = (np.arange(n) + 0.5) / n
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    elif mode == 'edge':
        frac = rng.uniform(0.008, 0.04)
        m = max(1, int(frac * n))
        a = np.zeros(n, dtype=float)
        # distribute the total mass uniformly over the edge region
        edge_mass = 0.5 / m
        a[:m] = edge_mass
        a[-m:] = edge_mass
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        a = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a[idx] = vals
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Sub‑gradient utilities
# ----------------------------------------------------------------------


def compute_conv_and_mask(a: np.ndarray, fft_len: int):
    """Return full linear convolution a*a, its maximum and a mask of active entries."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(float)   # 1 where the max is attained
    return conv, max_val, mask


def compute_subgradient(a: np.ndarray, mask: np.ndarray, fft_len: int):
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    Returns an array of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fm = np.fft.fft(mask, fft_len)
    grad_full = np.real(np.fft.ifft(fa_rev * fm))
    # take the slice that corresponds to the gradient w.r.t. each a_i
    return 2.0 * grad_full[n - 1: n - 1 + n]


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eta0: float = 0.15,
                                   max_iter: int = 5000):
    """
    Projected (sub‑)gradient descent on the true max‑convolution.
    Keeps the vector on the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    while time.time() < deadline and it < max_iter:
        conv, max_val, mask = compute_conv_and_mask(a, fft_len)
        g = compute_subgradient(a, mask, fft_len)
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = simplex_projection(a_new)
        score = fast_score(a_new, fft_len)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill climbing
# ----------------------------------------------------------------------


def directed_hill_climb(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_step: float = 0.18,
                        min_step: float = 1e-6):
    """
    Repeatedly move a fraction of mass from the index with largest sub‑gradient
    to the index with smallest sub‑gradient.  Step size shrinks over time.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len)
    best_a = a.copy()
    best_score = cur_score
    step = init_step
    it = 0
    while time.time() < deadline:
        conv, max_val, mask = compute_conv_and_mask(a, fft_len)
        g = compute_subgradient(a, mask, fft_len)

        src = int(np.argmax(g))   # index where increasing a would most increase the max
        dst = int(np.argmin(g))   # index where increasing a would most decrease the max

        # ensure source still has some mass
        if a[src] <= 0.0:
            src = int(np.argmax(a))
        if src == dst:
            # fallback to a random pair
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
        delta = a[src] * step
        if delta < 1e-12:
            break

        a[src] -= delta
        a[dst] += delta

        new_score = fast_score(a, fft_len)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[direct] new best {best_score:.6f}")
        else:
            # undo and shrink step size
            a[src] += delta
            a[dst] -= delta
            step *= 0.93

        # gradual decay irrespective of acceptance
        step = max(step * 0.9999, min_step)
        it += 1
        if it % 5000 == 0:
            # occasional recomputation of the score to avoid drift
            cur_score = fast_score(a, fft_len)

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental hill climb (same idea as the reference implementation)
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a high‑mass
    index to a low‑mass one, using an O(n) incremental convolution update.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger entry to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries is a good balance of quality / speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if available; otherwise uniform
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] best_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – random starts + projected sub‑gradient descent
    # --------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = projected_subgradient_optimize(
            a_start, phase1_end, n, fft_len, rng,
            eta0=0.18, max_iter=3000
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[subgrad] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – gradient‑guided directed hill‑climb
    # --------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    a_dir, score_dir = directed_hill_climb(
        best_a, phase2_end, n, fft_len, rng,
        init_step=0.15, min_step=1e-7
    )
    if score_dir < best_score:
        best_score = score_dir
        best_a = a_dir.copy()
        print(f"[direct phase] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – intensive greedy hill‑climbing (incremental)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.18,
            recompute_every=1200,
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
