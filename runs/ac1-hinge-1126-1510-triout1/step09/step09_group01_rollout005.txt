"""
Improved search for a low‑score non‑negative step sequence.

The algorithm consists of four phases (time‑controlled):
1.  A cheap parametric sweep (including a Beta family) to obtain a good
    initial vector on the probability simplex.
2.  Many random‑restart Adam optimisations on a smoothed (log‑sum‑exp)
    approximation of the max‑convolution.  The smoothing factor is
    decreased geometrically from 0.5 down to 5e‑6, and each run performs
    a few hundred iterations per smoothing level.
3.  A long projected sub‑gradient descent with adaptive step‑size
    (momentum) on the *true* max‑convolution.  This removes the bias of
    the smoothed objective.
4.  A final simulated‑annealing hill‑climb that moves a small fraction
    of the mass between two positions, accepting occasional uphill
    moves while the temperature decays linearly to zero.

All phases respect the global deadline (default 1000 s).  The best
vector seen at any moment is returned as a plain list of floats.  The
search works on a vector of length 4000 (≈8 KB) and uses FFT‑based
convolutions for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (first 2·len(a)-1 entries)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv) for a vector already normalised to sum 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (including a Beta distribution)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "beta":
        # continuous Beta(α,β) on [0,1] discretised on n points
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    else:  # random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep_extended(n: int,
                              fft_len: int,
                              budget_seconds: float,
                              rng: np.random.Generator,
                              best_a: np.ndarray,
                              best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families (including beta).
    Returns the best vector and its fast_score.
    """
    deadline = time.time() + budget_seconds
    families = ["power", "exp", "gauss", "cos", "bimodal", "beta", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smoothed_objective_and_grad(y: np.ndarray,
                                n: int,
                                fft_len_conv: int,
                                fft_len_g: int,
                                lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution (a appears twice)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y

def adam_optimize(a_start: np.ndarray,
                  lam_factors: np.ndarray,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed‑max objective.
    Returns the best vector found and its fast_score.
    """
    # initialise logits (soft‑max guarantees the simplex)
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = smoothed_objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps)   # slightly larger base step
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_descent_momentum(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the *real* max‑convolution.
    Uses an adaptive step size and stops when the deadline expires.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)        # momentum buffer
    beta = 0.9
    step = 0.5                  # initial step length
    it = 0

    # Helper to compute current max‑convolution
    def conv_and_max(vec):
        fa = np.fft.fft(vec, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        return conv, conv.max()

    conv, cur_max = conv_and_max(a)
    cur_score = 2.0 * n * cur_max

    while time.time() < deadline:
        # sub‑gradient of the max (mask includes entries within eps of the max)
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # back‑propagation through convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # candidate step
        a_candidate = simplex_projection(a - step * v)

        # evaluate candidate
        conv_cand, max_cand = conv_and_max(a_candidate)
        cand_score = 2.0 * n * max_cand

        # Accept if we improve, otherwise shrink step
        if cand_score < cur_score:
            a = a_candidate
            conv = conv_cand
            cur_max = max_cand
            cur_score = cand_score
            step = min(step * 1.02, 1.0)       # slowly increase step after success
        else:
            step *= 0.6                         # aggressive reduction on failure
            if step < 1e-6:
                step = 0.5

        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()

        it += 1
        if it % 500 == 0:  # occasional progress report
            print(f"[subgrad] iter {it}, best fast_score={best_score:.6f}")

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that respects the total mass.
    Temperature decays linearly from 2 % of the start score to zero.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        # move a small random fraction of mass from i to j
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with the lowest possible
    evaluation score within the given time budget.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – feel free to adjust
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 10 % of total budget)
    # --------------------------------------------------------------
    phase0_end = start_time + 0.10 * budget_s
    best_a, best_score = parametric_sweep_extended(
        n,
        fft_len_conv,
        max(0.0, phase0_end - time.time()),
        rng,
        best_a,
        best_score,
    )
    print(f"[sweep] best fast_score after sweep={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on smoothed max (≈ 45 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_factors = np.geomspace(0.5, 5e-6, num=45)   # from smooth to almost exact
    iters_per_lam = 250

    while time.time() < phase1_end:
        # start either from a jittered version of the current best or a fresh random family
        if rng.random() < 0.35:
            a_start = best_a * (1.0 + rng.normal(0.0, 0.02, size=n))
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (≈ 30 % of budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    best_a, best_score = subgradient_descent_momentum(
        best_a,
        phase2_end,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
