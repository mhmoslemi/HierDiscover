"""...
Search for a low max self‑convolution step function.
The optimizer uses:
  * diverse parametric initialisations (uniform, exponential, power, cosine,
    semicircle, Gaussian, edge, sparse, bimodal)
  * smoothed L‑BFGS with continuation on the log‑sum‑exp smooth max.
  * exponentiated‑gradient (mirrored SGD) on the true max–convex objective.
  * deterministic safe mass transfers (optimal single‑pair greedy descent).
  * a final simulated‑annealing hill‑climb.

All phases respect the 1000 s time budget.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Computes 2·n·max(a*a) assuming a sums to 1.
    Uses FFT for O(n log n) speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def shape_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def shape_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def shape_edge(n: int, rng: np.random.Generator) -> np.ndarray:
    edge_frac = rng.uniform(0.006, 0.03)
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a /= a.sum()
    return a


def shape_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = n // 2
    width = rng.integers(int(0.02 * n), int(0.07 * n))
    peak = rng.uniform(0.5, 1.5) / n
    a = np.full(n, 1.0 / n)
    left = max(0, centre - width)
    right = min(n, centre + width)
    a[left:right] = peak
    a /= a.sum()
    return a


def shape_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    m = rng.integers(int(0.02 * n), int(0.07 * n))
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.random(m) + 1e-6
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


def shape_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raised‑cosine on a central window."""
    w_frac = rng.uniform(0.2, 1.0)
    w = max(2, int(w_frac * n))
    a = np.zeros(n)
    centre = n // 2
    half = w // 2
    start = max(0, centre - half)
    end = min(n, start + w)
    # generate cosine over [0, w-1]
    x = np.linspace(-1.0, 1.0, end - start)
    a[start:end] = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def shape_semicircle(n: int, rng: np.random.Generator) -> np.ndarray:
    """Semicircle window centred in the array."""
    w_frac = rng.uniform(0.2, 1.0)
    w = max(2, int(w_frac * n))
    a = np.zeros(n)
    centre = n // 2
    half = w // 2
    start = max(0, centre - half)
    end = min(n, start + w)
    idx = np.arange(start, end) - centre
    R = w / 2.0
    a[start:end] = np.sqrt(np.maximum(0.0, R ** 2 - idx.astype(float) ** 2))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def shape_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma_frac = rng.uniform(0.05, 0.3)
    sigma = sigma_frac * n
    x = np.arange(n) - n // 2
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector built from many possible styles."""
    mode = rng.choice(
        [
            'uniform',
            'random',
            'exp',
            'power',
            'edge',
            'bimodal',
            'sparse',
            'cosine',
            'semicircle',
            'gaussian',
        ]
    )
    if mode == 'uniform':
        a = shape_uniform(n)
    elif mode == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif mode == 'exp':
        a = shape_exponential(n, rng)
    elif mode == 'power':
        a = shape_power(n, rng)
    elif mode == 'edge':
        a = shape_edge(n, rng)
    elif mode == 'bimodal':
        a = shape_bimodal(n, rng)
    elif mode == 'sparse':
        a = shape_sparse(n, rng)
    elif mode == 'cosine':
        a = shape_cosine(n, rng)
    elif mode == 'semicircle':
        a = shape_semicircle(n, rng)
    else:  # gaussian
        a = shape_gaussian(n, rng)

    return a


# ----------------------------------------------------------------------
# Smoothed L‑BFGS core
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations for the lam‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam, 'gtol': 1e-12, 'disp': False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
):
    """Mirrored (exponentiated) SGD on the hard max‑convolution objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy descent using analytical safe step size
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Move a fraction of the mass from the index with the largest gradient
    to the index with the smallest gradient using the analytically maximal safe
    step size that never raises the peak."""
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # admissible δ to keep non‑negativity and not increase the peak
        delta_src = a[src]

        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92   # a safety factor <1
        if delta <= 0.0:
            break

        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to keep numerical drift small
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted by a Metropolis rule while temperature cools."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.15   # maximal fraction of src mass we may move per trial
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

"""Search for a low‑max self‑convolution step function.

The optimizer proceeds in four phases (time‑budget aware):
  1. Smoothed L‑BFGS with many diverse initialisations (uniform, exponential,
     power law, raised‑cosine, semicircle, Gaussian, edge, sparse, bimodal).
  2. Exponentiated‑gradient (mirrored SGD) on the true max‑convolution.
  3. Deterministic greedy descent that moves mass between the most/least
     harmful coordinates using the analytically maximal safe step size.
  4. Simulated‑annealing hill‑climb with incremental convolution updates.

All phases keep the current best vector and return it before the deadline.
The final sequence is evaluated with the provided ``evaluate_sequence``."""
import time

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) assuming a sums to 1 (FFT based, O(n log n))."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Shape generators for diverse initialisations
# ----------------------------------------------------------------------


def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def shape_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def shape_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def shape_edge(n: int, rng: np.random.Generator) -> np.ndarray:
    edge_frac = rng.uniform(0.006, 0.03)
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a /= a.sum()
    return a


def shape_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = n // 2
    width = rng.integers(int(0.02 * n), int(0.07 * n))
    peak = rng.uniform(0.5, 1.5) / n
    a = np.full(n, 1.0 / n)
    left = max(0, centre - width)
    right = min(n, centre + width)
    a[left:right] = peak
    a /= a.sum()
    return a


def shape_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    m = rng.integers(int(0.02 * n), int(0.07 * n))
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.random(m) + 1e-6
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


def shape_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raised‑cosine window centred in the array."""
    w_frac = rng.uniform(0.2, 1.0)
    w = max(2, int(w_frac * n))
    a = np.zeros(n)
    centre = n // 2
    half = w // 2
    start = max(0, centre - half)
    end = min(n, start + w)
    x = np.linspace(-1.0, 1.0, end - start)
    a[start:end] = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def shape_semicircle(n: int, rng: np.random.Generator) -> np.ndarray:
    """Semicircle window centred in the array."""
    w_frac = rng.uniform(0.2, 1.0)
    w = max(2, int(w_frac * n))
    a = np.zeros(n)
    centre = n // 2
    half = w // 2
    start = max(0, centre - half)
    end = min(n, start + w)
    idx = np.arange(start, end) - centre
    R = w / 2.0
    a[start:end] = np.sqrt(np.maximum(0.0, R ** 2 - idx.astype(float) ** 2))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def shape_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma_frac = rng.uniform(0.05, 0.3)
    sigma = sigma_frac * n
    x = np.arange(n) - n // 2
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random style and return a probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "cosine",
            "semicircle",
            "gaussian",
        ]
    )
    if mode == "uniform":
        a = shape_uniform(n)
    elif mode == "random":
        a = rng.random(n)
        a /= a.sum()
    elif mode == "exp":
        a = shape_exponential(n, rng)
    elif mode == "power":
        a = shape_power(n, rng)
    elif mode == "edge":
        a = shape_edge(n, rng)
    elif mode == "bimodal":
        a = shape_bimodal(n, rng)
    elif mode == "sparse":
        a = shape_sparse(n, rng)
    elif mode == "cosine":
        a = shape_cosine(n, rng)
    elif mode == "semicircle":
        a = shape_semicircle(n, rng)
    else:  # gaussian
        a = shape_gaussian(n, rng)
    return a


# ----------------------------------------------------------------------
# Smoothed L‑BFGS (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """Smooth objective based on log‑sum‑exp of the convolution.
    Returns (value, gradient) w.r.t. logits y where a = softmax(y)."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the lambda schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
):
    """Mirrored SGD (exponentiated gradient) on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy descent (single‑pair safe moves)
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Move a fraction of mass from the most harmful to the least harmful index,
    using the analytically maximal safe step size that never raises the peak."""
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # admissible δ
        delta_src = a[src]

        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92  # safety margin
        if delta <= 0.0:
            break

        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to curb drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted via Metropolis rule while temperature cools."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.15
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer search space
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.25 * budget_s  # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s  # exponentiated‑gradient
    phase3_end = start_time + 0.90 * budget_s  # deterministic greedy
    # remaining → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS with diverse starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
    ]
    maxiter_per_lam = 300

    while time.time() < phase1_end:
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=15.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy descent (analytic safe step)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(
            best_a, phase3_end, n, fft_len
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for a possible next run
    globals()["height_sequence_1"] = final_seq

    return final_seq
