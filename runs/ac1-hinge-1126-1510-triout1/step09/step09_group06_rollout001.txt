def fine_hill_climb(a, current_score, fft_len, n, deadline, max_iter=200000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0: 
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta as a small fraction maybe up to 0.5% of source mass.
        # This ensures fine adjustments.
        max_delta = best[i] * 0.005  # up to 0.5%
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0: continue
        best[i] -= delta
        best[j] += delta
        # compute new score
        new_score = fast_score(best, fft_len, n)
        # if improvement accept else undo
        if new_score < best_score - 1e-10:  # small tolerance
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Simple but aggressive multi‑phase optimizer for the step‑function
max‑convolution problem.

The algorithm builds a large and diverse pool of initial candidates
(random Dirichlet draws, exponential / power‑law families, edge‑heavy
shapes, a spectral init and the previous best if available).  The best
few seeds are polished by a continuation of L‑BFGS on a smooth
log‑sum‑exp approximation of the max‑convolution.  Afterwards a long
run of exponentiated‑gradient descent (which automatically stays on the
probability simplex) and a projected sub‑gradient descent are applied.
A second short L‑BFGS polishing and a fine‑grained greedy hill‑climb
complete the search.  All phases respect a shared 1000 s budget.

The final sequence is returned as a list of non‑negative floats.
"""
import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i]=a[-i-1]; keep sum=1."""
    b = (a + a[::-1]) / 2.0
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 0:
        b /= s
    else:
        b = np.full_like(b, 1.0 / b.size)
    return b


# --------------------------------------------------------------
# Fast score (sum is forced to 1, therefore 2·n·max(conv))
# --------------------------------------------------------------

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return the evaluation score for a already normalised to sum=1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient (for L‑BFGS)
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.3):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# --------------------------------------------------------------

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting from the result of the
    previous stage. Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a = symmetrize(a)                # enforce symmetry after each stage
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (larger initial step)
        eta = 1.5 / np.sqrt(it + 1.0)          # slightly larger than 1.0

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        a = symmetrize(a)                      # keep symmetry

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgrad_optimize(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     step0: float = 0.8):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses a larger initial step size than the previous implementation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that preserves the sum (projected)
        d = -(g - np.mean(g))

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)
        a_new = symmetrize(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000,
               max_frac: float = 0.10):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most max_frac of the source mass
        delta = best[i] * rng.random() * max_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def fine_hill_climb(a: np.ndarray,
                    current_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    max_iter: int = 5000):
    """
    Very fine hill‑climb: move at most 0.5 % of a source component.
    Used at the very end to shave off the last few 1e‑6 improvements.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        max_delta = best[i] * 0.005   # 0.5 %
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score + 1e-12 < best_score:   # tiny tolerance
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.05)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.1, 5.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.003, 0.02)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """
    Explore a few parametric families and return the best `keep` candidates.
    """
    candidates = []

    # Exponential family (more refined grid)
    for c in np.logspace(-6, -0.2, num=20):
        a = exponential_shape(n, c)
        score = fast_score(a, fft_len, n)
        candidates.append((score, a))

    # Power‑law family
    for p in np.linspace(0.05, 8.0, num=25):
        a = powerlaw_shape(n, p)
        score = fast_score(a, fft_len, n)
        candidates.append((score, a))

    # Edge‑heavy families (broader exploration)
    edge_fracs = [0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            score1 = fast_score(a1, fft_len, n)
            candidates.append((score1, a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            score2 = fast_score(a2, fft_len, n)
            candidates.append((score2, a2))

    # Random Dirichlet draws (varying concentration)
    for _ in range(40):
        alpha_scalar = rng.uniform(0.02, 2.5)
        a = rng.gamma(alpha_scalar, size=n)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        score = fast_score(a, fft_len, n)
        candidates.append((score, a))

    # Sort and keep the best distinct candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 9)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

    fft_len = next_pow2(2 * n - 1)

    candidates.append(a_prev)

    # spectral and a random initial draw
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # families sweep

    candidates.extend(family_best)

    # Additional random Dirichlet draws (different concentration)
    for _ in range(30):
        alpha_vec = rng.uniform(0.01, 1.0, size=n)
        a = rng.gamma(alpha_vec)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        candidates.append(a)

    # Evaluate the pool and keep the best few seeds
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on a few seeds
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.12 * budget_s
    lam_schedule = [0.6, 0.4, 0.3, 0.22, 0.15, 0.10,
                    0.07, 0.045, 0.027, 0.016, 0.009,
                    0.005, 0.003, 0.0015, 0.0008, 0.0004]
    maxiter_per_lam = 1200

    # choose a handful of diverse seeds (best + few random)
    seeds = [best_a]
    for _ in range(2):
        seeds.append(random_initial(n, rng))

    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = subgrad_optimize(best_a,
                                              phase3_deadline,
                                              n,
                                              fft_len,
                                              step0=0.9)
        print(f"[subgrad] best score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – second short L‑BFGS polishing (fine)
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.90 * budget_s
    if time.time() < phase4_deadline:
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                            lam_schedule[-8:],   # only the finest lambdas
                                            800,                # fewer iterations, fine tuning
                                            phase4_deadline,
                                            fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑2] new best score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – greedy hill‑climb (coarse then fine)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # coarse hill‑climb (larger moves)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=18000,
                                        max_frac=0.08)
        print(f"[hill‑climb‑coarse] score = {best_score:.8f}")

    # final fine hill‑climb (tiny moves)
    if time.time() < deadline:
        best_a, best_score = fine_hill_climb(best_a,
                                             best_score,
                                             fft_len,
                                             n,
                                             deadline,
                                             max_iter=12000)
        print(f"[hill‑climb‑fine] final refined score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (as a plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
