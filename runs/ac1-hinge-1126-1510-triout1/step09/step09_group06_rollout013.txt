```
def simulated_annealing(a_start, deadline, n, fft_len, init_temp=1e-3, final_temp=1e-8, max_steps=None):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    t0 = init_temp
    # we can use exponential schedule: T = t0 * (final_temp/t0) ** (t/total_steps)
    step = 0
    # compute total_steps = max_steps or (deadline - time.time()) / expected_time_per_iter.
    while time.time() < deadline and (max_steps is None or step < max_steps):
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        # propose delta as random fraction of a[i] (0 to min(0.2*a[i],0.01?)).
        max_delta = a[i]
        if max_delta == 0:
            continue
        delta = max_delta * rng.random() * 0.2  # up to 20%
        # ensure delta > small
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # renorm (should stay sum=1)
        # compute new score
        new_score = fast_score(a, fft_len, n)
        # acceptance
        if new_score < cur_score:
            cur_score = new_score
        else:
            # compute temperature based on step fraction
            # dynamic temperature decreasing
            # we compute fraction = step / total_steps if known else based on time.
            # else compute T = init_temp * (1 - (time.time() - start)/(deadline-start))
            # Accept with prob
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                a[i] += delta
                a[j] -= delta
                continue
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        step += 1
    return best, best_score
```

```
def simulated_annealing(a_start, start_time, deadline, n, fft_len, init_temp=1e-3):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    total_time = deadline - start_time
    # define exponential cooling schedule: T = init_temp * (0.1) ** ((time_elapsed) / total_time)
    # Or linear schedule.
    while time.time() < deadline:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        max_delta = a[i]
        if max_delta <= 0:
            continue
        # delta drawn from uniform up to maybe 0.1 of a[i]
        delta = max_delta * rng.random() * 0.2
        a[i] -= delta
        a[j] += delta
        # compute score
        new_score = fast_score(a, fft_len, n)
        dt = time.time() - start_time
        frac = dt / total_time
        T = init_temp * (1 - frac) + 1e-12 * frac  # linear decreasing to near zero
        if new_score < cur_score:
            cur_score = new_score
        else:
            # acceptance probability
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                a[i] += delta
                a[j] -= delta
                continue
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score
```

```
    mags = rng.random(half-2)  # random mags
```

```
"""Optimizes a non‑negative step function to minimise the l∞ norm of its self‑convolution.
The algorithm builds a diverse pool of candidates (parametric families,
random Dirichlet draws, random spectral designs) and keeps the best few.
From the best seed it runs several optimisation phases:
exponentiated‑gradient descent, projected sub‑gradient descent,
simulated‑annealing, and a final L‑BFGS refinement with a log‑sum‑exp
smooth approximation of the max‑convolution.  All phases respect a
global 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# helper functions
def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def fast_score(a, fft_len, n):
    # assumes sum(a) == 1
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# objective for L‑BFGS (smooth max)
def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z  # d smooth_max / d conv_i
    # back‑prop through convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam,
                       deadline, fft_len_conv, fft_len_g, n):
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]
        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score

def subgrad_optimize(a_start, deadline, n, fft_len, step0=0.5):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]
        d = -(g - np.mean(g))
        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)
        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

def simulated_annealing(a_start, start_time, deadline, n, fft_len,
                        init_temp=5e-2, final_temp=5e-6):
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    total_time = deadline - start_time
    while time.time() < deadline:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        # delta up to a fraction of source mass
        delta = a[i] * rng.random() * 0.15  # up to 15%
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        # re‑normalise (tiny drift)
        a_sum = a.sum()
        if a_sum != 0.0:
            a = a / a_sum
        new_score = fast_score(a, fft_len, n)
        # temperature schedule (linear)
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = init_temp * (1.0 - frac) + final_temp * frac
        if new_score < cur_score:
            cur_score = new_score
        else:
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_score = new_score
            else:
                # revert
                a[i] += delta
                a[j] -= delta
                a = a / a.sum()
                continue
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score

# shape families
def spectral_init(n):
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_initial(n, rng):
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def exponential_shape(n, c):
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

def powerlaw_shape(n, p):
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

def edge_shape(n, edge_frac, ratio, interior_zero=False):
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()

def beta_shape(n, alpha, beta):
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_phase_spectrum(n, fft_len, rng):
    L = fft_len
    half = L // 2 + 1
    # DC term (real)
    A = np.empty(half, dtype=complex)
    A[0] = 1.0
    # Random phases for the other coefficients (except Nyquist if even)
    phases = rng.random(half - 2) * 2.0 * np.pi
    A[1:-1] = np.exp(1j * phases)
    if L % 2 == 0:
        A[-1] = 1.0  # Nyquist
    a_full = np.fft.irfft(A, n=L)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def generate_initial_pool(n, fft_len, rng, keep=12):
    candidates = []
    # 1. prior sequence if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            if s > 0:
                a_prev = a_prev / s
            else:
                a_prev = np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2. uniform
    candidates.append(np.full(n, 1.0 / n))

    # 3. spectral init
    candidates.append(spectral_init(n))

    # 4. random initial draws
    for _ in range(30):
        candidates.append(random_initial(n, rng))

    # 5. exponential family sweep
    for c in np.logspace(-4, -0.3, num=20):
        candidates.append(exponential_shape(n, c))

    # 6. power‑law sweep
    for p in np.linspace(0.1, 4.0, num=25):
        candidates.append(powerlaw_shape(n, p))

    # 7. edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            candidates.append(edge_shape(n, ef, r, interior_zero=False))
            candidates.append(edge_shape(n, ef, r, interior_zero=True))

    # 8. beta family random draws
    for _ in range(40):
        alpha = 10 ** rng.uniform(-2, 1)   # between 0.01 and 10

        candidates.append(beta_shape(n, alpha, beta))

    # 9. random spectral draws
    for _ in range(50):

    # evaluate and keep the best `keep` distinct candidates
    scored = [(fast_score(a, fft_len, n), a) for a in candidates]
    scored.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in scored:
        h = round(sc, 10)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

"""
Optimises a non‑negative step function (length 4000) to minimise the
ℓ∞‑norm of its self‑convolution, i.e. minimise 2·n·max(a*a) with ∑a=1.
The programme builds a large, diverse pool of candidates (parametric
families, random Dirichlet draws, random spectral designs), keeps the
best few seeds and then runs several optimisation phases within a

1️⃣  Exponentiated‑gradient descent on the exact max‑convolution.
2️⃣  Projected sub‑gradient descent (larger step size).
3️⃣  Simulated‑annealing with mass‑transfer moves.
4️⃣  A final L‑BFGS polishing using a log‑sum‑exp smooth surrogate.

All phases respect the global time limit; the best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(self‑convolution) for a normalised vector
    (∑a = 1).  Uses an FFT of length `fft_len`.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS with a smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of a*a using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits `y` where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,

                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent (a ← a·exp(−η·g)) keeping a on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # active set

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgrad_optimize(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     step0: float = 0.5):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that respects the sum‑to‑1 constraint
        d = -(g - np.mean(g))
        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_time: float,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        init_temp: float = 5e-2,
                        final_temp: float = 1e-7):
    """
    Simple SA that repeatedly transfers a random fraction of mass from one
    coordinate to another.  Accepts worsening moves with a linearly
    decreasing temperature.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    total_time = deadline - start_time

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 15 % of the source mass
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta
        a = a / a.sum()                       # re‑normalise (tiny drift)

        new_score = fast_score(a, fft_len, n)

        # linear temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = init_temp * (1.0 - frac) + final_temp * frac

        if new_score < cur_score:
            cur_score = new_score
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur_score = new_score
            else:
                # revert
                a[i] += delta
                a[j] -= delta
                a = a / a.sum()
                continue

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score

# Parametric families and random constructions
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random distribution on the simplex (several preset shapes)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(‑c·min(i, n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i, n‑1‑i)+1)^‑p."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑distribution shape."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_phase_spectrum(n: int, fft_len: int,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Random spectral construction: random phases (unit magnitude) in the
    frequency domain, inverse‑FFT, then clip negative values and re‑normalise.
    """
    L = fft_len
    half = L // 2 + 1
    A = np.empty(half, dtype=complex)
    A[0] = 1.0
    phases = rng.random(half - 2) * 2.0 * np.pi
    A[1:-1] = np.exp(1j * phases)
    if L % 2 == 0:
        A[-1] = 1.0
    a_full = np.fft.irfft(A, n=L)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def generate_initial_pool(n: int, fft_len: int,
                          rng: np.random.Generator,
                          keep: int = 12):
    """
    Assemble a large set of diverse candidates and keep the best `keep`
    of them (after removing near‑duplicates).
    """
    candidates = []

    # 0. previous best (if any)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 1. uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2. spectral init
    candidates.append(spectral_init(n))

    # 3. random Dirichlet‑like draws
    for _ in range(30):
        candidates.append(random_initial(n, rng))

    # 4. exponential family sweep
    for c in np.logspace(-4, -0.3, num=20):
        candidates.append(exponential_shape(n, c))

    # 5. power‑law sweep
    for p in np.linspace(0.1, 4.0, num=25):
        candidates.append(powerlaw_shape(n, p))

    # 6. edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            candidates.append(edge_shape(n, ef, r, interior_zero=False))
            candidates.append(edge_shape(n, ef, r, interior_zero=True))

    # 7. random Beta shapes
    for _ in range(40):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(beta_shape(n, alpha, beta))

    # 8. random spectral designs
    for _ in range(50):
        candidates.append(random_phase_spectrum(n, fft_len, rng))

    # evaluate and keep the `keep` best distinct candidates
    scored = [(fast_score(a, fft_len, n), a) for a in candidates]
    scored.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in scored:
        h = round(sc, 10)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step sequence of length 4000 that minimises
    evaluate_sequence(...).  The function respects the supplied time budget
    and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to avoid overrunning the limit
    deadline = start_time + budget_s - 0.4

    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1️⃣  Build initial pool
    # ------------------------------------------------------------------
    init_pool = generate_initial_pool(n, fft_len, rng, keep=12)
    best_a = init_pool[0]
    best_score = fast_score(best_a, fft_len, n)
    for cand in init_pool[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Partition remaining budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_a.tolist()

    # rough allocation (tuneable)
    t_exp = 0.35 * remaining
    t_sub = 0.20 * remaining
    t_sa = 0.30 * remaining
    t_lbfgs = remaining - (t_exp + t_sub + t_sa)

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_dead = time.time() + t_exp
        best_a, best_score = exp_gradient_optimize(best_a, exp_dead,
                                                   n, fft_len, rng)
        print(f"[exp‑grad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + t_sub
        best_a, best_score = subgrad_optimize(best_a, sub_dead,
                                              n, fft_len, step0=0.9)
        print(f"[subgrad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_dead = time.time() + t_sa
        best_a, best_score = simulated_annealing(best_a,
                                                 time.time(),
                                                 sa_dead,
                                                 n,
                                                 fft_len,
                                                 init_temp=5e-2,
                                                 final_temp=1e-7)
        print(f"[SA] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final L‑BFGS polishing with a finer λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                        0.05, 0.03, 0.018, 0.01, 0.006,
                        0.003, 0.001, 0.0005, 0.0002,
                        0.0001, 5e-05, 2e-05, 1e-05, 5e-06]
        maxiter_per_lam = 800
        best_a, best_score = lbfgs_continuation(best_a,
                                                lam_schedule,
                                                maxiter_per_lam,
                                                deadline,
                                                fft_len,
                                                fft_len,
                                                n)
        print(f"[L‑BFGS final] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Return the best sequence
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
```
