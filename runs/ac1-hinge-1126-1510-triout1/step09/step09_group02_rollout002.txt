"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of several phases, each given a portion of the
overall time budget (default 1000 s):

1. **Multiple random restarts of L‑BFGS** on a log‑sum‑exp smoothed
   approximation of the max.  A schedule of decreasing smoothing
   parameters (λ) guides the optimizer into the true objective.

2. **Adam** on the same smoothed objective with a finer λ schedule.

3. **Exponentiated‑gradient descent** on the exact max‑convolution
   (sub‑gradient).  A diminishing step size η/√t is used.

4. **Annealed hill‑climbing** that moves a fraction of mass from a
   larger index to a smaller one, updating the convolution incrementally
   (including the ignored quadratic δ² terms).  The temperature is
   decreased linearly, allowing occasional uphill moves early on.

5. **Perturb‑&‑refine cycles**: as long as time permits the current
   best solution is slightly perturbed and a short Adam run is applied,
   followed by another short hill‑climb.

All iterates stay on the probability simplex (∑a = 1).  The best
sequence found before the deadline is returned as a plain list of
floats.  The code prints intermediate scores so the progress can be
monitored.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the objective when ∑a = 1:
       score = 2·n·max(conv(a,a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Smoothed objective and its gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  Parameters:
        y   – logits (real vector) such that a = softmax(y)
        n   – length of a
        fft_len – length of FFT (≥ 2·n‑1)
        lam_factor – controls the smoothing λ = lam_factor * max(conv)
    Returns (objective, gradient) w.r.t. y.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: ∂conv / ∂a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t. a (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(y_start: np.ndarray,

        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                # occasional log
                if steps % 250 == 0:
                    print(f"[Adam] λ={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # sub‑gradient of max(conv)

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Annealed hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_annealed(a: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_fraction_start: float = 0.25,
                        max_fraction_end: float = 0.02,
                        recompute_every: int = 1200):
    """
    Greedy hill‑climb that moves a random fraction of mass from a larger
    index to a smaller one.  The convolution is updated incrementally
    (including the quadratic δ² terms).  An annealing temperature
    allows occasional uphill moves early on.
    """
    best_a = a.copy()
    best_score = current_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    a_padded = np.concatenate([best_a, np.zeros(n - 1, dtype=float)])

    start = time.time()
    accepted = 0
    it = 0

    while time.time() < deadline:
        # Linear schedule for max move fraction
        t_frac = (time.time() - start) / (deadline - start)
        max_frac = (
            max_fraction_start * (1.0 - t_frac) + max_fraction_end * t_frac
        )

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # ensure we move mass from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] <= 0.0:
            continue

        delta = best_a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # ---------- incremental update ----------
        # linear part
        diff = a_padded[dst:dst + n] - a_padded[src:src + n]   # view, length n
        conv += 2.0 * delta * diff

        # quadratic corrections
        conv[2 * src] += delta * delta
        conv[2 * dst] += delta * delta
        conv[src + dst] -= 2.0 * delta * delta

        new_max = conv.max()

        # simple simulated‑annealing acceptance rule
        temperature = max(0.001, 0.1 * (1.0 - t_frac))  # decays to ~0
        accept = new_max < conv.max() or rng.random() < np.exp(
            -(new_max - conv.max()) / temperature
        )
        if accept:
            # commit move
            best_a[src] -= delta
            best_a[dst] += delta
            a_padded[src] = best_a[src]
            a_padded[dst] = best_a[dst]
            current_score = 2.0 * n * new_max
            if current_score < best_score:
                best_score = current_score
                best_a = best_a.copy()
            accepted += 1
        else:
            # revert incremental update
            conv -= 2.0 * delta * diff
            conv[2 * src] -= delta * delta
            conv[2 * dst] -= delta * delta
            conv[src + dst] += 2.0 * delta * delta

        # occasional full recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
            a_padded = np.concatenate([best_a, np.zeros(n - 1, dtype=float)])

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 5000 entries gives a good balance between resolution
    # and speed.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously saved sequence if available; otherwise
    # start from uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    phase1_end = start_time + 0.18 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.48 * budget_s   # Adam
    phase3_end = start_time + 0.68 * budget_s   # Exp‑grad
    phase4_end = start_time + 0.90 * budget_s   # Hill‑climb

    # ------------------------------------------------------------------
    # Helper lambda for quick re‑evaluation
    # ------------------------------------------------------------------
    def update_best(candidate):
        nonlocal best_a, best_score
        sc = fast_score(candidate, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[update] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – multiple restarts of L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.4, 0.3, 0.25, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.006, 0.0035, 0.002, 0.0012,
        0.0007, 0.0004, 0.0002, 0.0001
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # 70 % fresh random start, 30 % jittered best
        if rng.random() < 0.3:
            a_start = add_noise(best_a, sigma=rng.uniform(0.002, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        update_best(a_opt)

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.45, 0.30, 0.20, 0.12, 0.07,
                             0.04, 0.022, 0.012, 0.006, 0.003]
        best_a, _ = adam_optimize(
            y0,
            lam_schedule_adam,
            max_steps_per_lam=2000,
            deadline=phase2_end,
            n=n,
            fft_len=fft_len,
            lr0=0.13,
        )
        best_score = fast_score(best_a, fft_len)
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, _ = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=14.0,
        )
        best_score = fast_score(best_a, fft_len)
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – annealed hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = hill_climb_annealed(
            best_a,
            best_score,
            fft_len,
            n,
            phase4_end,
            rng,
            max_fraction_start=0.22,
            max_fraction_end=0.015,
            recompute_every=1500,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Perturb‑&‑refine cycles while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # tiny perturbation of the best solution
        a_pert = add_noise(best_a, sigma=rng.uniform(0.0005, 0.003), rng=rng)

        # short Adam run on the perturbed start
        y_start = np.log(a_pert + 1e-12)
        lam_sched = [0.04, 0.02, 0.008, 0.003, 0.001]
        a_refined, _ = adam_optimize(
            y_start,
            lam_sched,
            max_steps_per_lam=800,
            deadline=deadline,
            n=n,
            fft_len=fft_len,
            lr0=0.10,
        )
        # brief hill‑climb
        a_refined, _ = hill_climb_annealed(
            a_refined,
            fast_score(a_refined, fft_len),
            fft_len,
            n,
            deadline,
            rng,
            max_fraction_start=0.10,
            max_fraction_end=0.005,
            recompute_every=800,
        )
        update_best(a_refined)

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
