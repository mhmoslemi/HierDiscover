    best_S, best_max = S.copy(), max_count

    decay = 0.9995
    it = 0
    while time.time() < time_limit:

        out = rng.choice(list(S))
        # choose in index not in S
        # Since many not in S, we can random select until not in S
        while True:

        # Remove out contributions:

"""
Sparse‐subset optimisation for the max‑convolution objective.

The idea is to keep only *k* non‑zero entries (all equal to 1/k) and to
 minimise the maximal number of ordered pairs (i,j)∈S×S that give the
 same sum i+j.  For a binary indicator vector `ind` the convolution
 `ind*ind` contains exactly those pair counts.  By swapping a single
 element of the support we can update the counts in O(k) time, keep the
 current objective
        2·n·max(conv)/(k²)
 and accept the move with a Metropolis rule (simulated‑annealing).
 Several values of `k` are tried within the 1000 s budget and the best
 vector found is returned.

The algorithm is completely self‑contained, uses only NumPy and the
 provided ``evaluate_sequence`` routine and respects the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def initialise_subset(n: int, k: int, rng: np.random.Generator):
    """Return a random subset of size k (list of indices) and its mask."""
    subset = list(rng.choice(n, size=k, replace=False))
    mask = np.zeros(n, dtype=bool)
    mask[subset] = True
    return subset, mask


def counts_from_subset(subset: list[int], n: int) -> np.ndarray:
    """
    Full ordered‑pair count array for the current subset.
    Equivalent to np.convolve(ind, ind) where ind[i]=1 if i∈subset else 0.
    """
    ind = np.zeros(n, dtype=np.int32)
    ind[subset] = 1
    # ``np.convolve`` returns a view of length 2*n-1 (int64)
    return np.convolve(ind, ind)


def objective_from_counts(max_cnt: int, n: int, k: int) -> float:
    """Exact value of 2·n·max(conv)/(sum(a)²) for a uniform 1/k subset."""
    return 2.0 * n * max_cnt / (k * k)


def hill_climb_sparse(
    n: int,
    k: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float, int]:
    """
    Simulated‑annealing on a binary support of size ``k``.
    Returns the best mask, the corresponding objective value and the final
    maximum pair count.
    """
    # initialise random support
    subset, mask = initialise_subset(n, k, rng)
    counts = counts_from_subset(subset, n).astype(np.int64)
    max_cnt = int(counts.max())
    cur_obj = objective_from_counts(max_cnt, n, k)

    best_mask = mask.copy()
    best_obj = cur_obj
    best_max = max_cnt

    # temperature schedule – start with a small fraction of the current obj
    temperature = max(0.1, cur_obj * 0.05)
    decay = 0.9997

    it = 0
    while time.time() < deadline:
        # ---- propose a swap -------------------------------------------------
        out_el = rng.choice(subset)                # element to remove
        # choose a new element not currently present
        inn_el = rng.integers(n)
        while mask[inn_el]:
            inn_el = rng.integers(n)

        # ---- remove contributions of out_el (ordered pairs) ---------------
        for j in subset:                           # subset still contains out_el
            idx = out_el + j
            if j == out_el:
                counts[idx] -= 1                   # (out,out)
            else:
                counts[idx] -= 2                   # (out,j) and (j,out)

        # update data structures
        mask[out_el] = False
        subset.remove(out_el)                      # now len = k‑1

        # ---- add contributions of inn_el ----------------------------------
        for j in subset:                           # subset now = old without out_el
            idx = inn_el + j
            counts[idx] += 2                       # (inn,j) and (j,inn)
        counts[2 * inn_el] += 1                     # (inn,inn)

        mask[inn_el] = True
        subset.append(inn_el)                       # len back to k

        new_max = int(counts.max())
        new_obj = objective_from_counts(new_max, n, k)

        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            prob = np.exp(-(new_obj - cur_obj) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_obj = new_obj
            max_cnt = new_max
            if new_obj < best_obj:
                best_obj = new_obj
                best_max = new_max
                best_mask = mask.copy()
                print(
                    f"[k={k}] it={it:07d} new best obj={best_obj:.6f}",
                    flush=True,
                )
        else:
            # ---- revert the swap (undo what we just did) -------------------
            # subtract inn_el contributions
            for j in subset:                       # subset currently contains inn_el
                idx = inn_el + j
                if j == inn_el:
                    counts[idx] -= 1
                else:
                    counts[idx] -= 2
            mask[inn_el] = False
            subset.remove(inn_el)                  # back to k‑1 elements

            # add back out_el contributions
            for j in subset:                       # subset now = old without out_el
                idx = out_el + j
                counts[idx] += 2
            counts[2 * out_el] += 1
            mask[out_el] = True
            subset.append(out_el)                  # restore original state

        # temperature decay (slow)
        temperature = max(temperature * decay, 1e-12)
        it += 1

    return best_mask, best_obj, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.
    The method uses a sparse representation (k non‑zero entries) and a
    simulated‑annealing hill‑climber on the exact max‑convolution count.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – reuse the size of the previously found solution if
    # possible, otherwise default to 4000.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4000

    # total time budget (tiny safety margin)
    total_deadline = time.time() + budget_s - 0.2

    # ------------------------------------------------------------------
    # Try several values of k (support size).  Time is split evenly.
    # ------------------------------------------------------------------
    k_candidates = [400, 600, 800, 1000, 1200]            # < n
    time_per_k = (total_deadline - time.time()) / len(k_candidates)

    global_best_mask = None
    global_best_obj = np.inf
    global_best_k = None
    global_best_max = None

    for k in k_candidates:
        if time.time() >= total_deadline:
            break
        print(f"[phase] optimisation for k={k}", flush=True)

        deadline_k = time.time() + time_per_k
        mask, obj, max_cnt = hill_climb_sparse(n, k, deadline_k, rng)

        if obj < global_best_obj:
            global_best_obj = obj
            global_best_mask = mask
            global_best_k = k
            global_best_max = max_cnt
            print(
                f"[global] new best with k={k}: obj={global_best_obj:.6f}",
                flush=True,
            )

    # ------------------------------------------------------------------
    # Build the final sequence (uniform weight 1/k on the chosen indices)
    # ------------------------------------------------------------------
    if global_best_mask is None:                     # fallback – uniform
        final_seq = (np.full(n, 1.0 / n)).tolist()
    else:
        final_seq = np.zeros(n, dtype=float)
        final_seq[global_best_mask] = 1.0 / global_best_k
        final_seq = final_seq.tolist()

    # ------------------------------------------------------------------
    # Final external evaluation (for logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_score:.6f}", flush=True)

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
