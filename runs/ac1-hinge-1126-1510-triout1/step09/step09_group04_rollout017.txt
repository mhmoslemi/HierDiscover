"""
Improved search for a low‑value step‑function sequence.

The algorithm proceeds in several stages within the allowed 1000 s:

1️⃣  Initialisation – several diverse seeds (uniform, random, spectral,
    several random‑phase spectral vectors) are generated and the best
    one (lowest fast_score) is kept.

2️⃣  Coarse L‑BFGS – a log‑softmax parametrisation is optimised with a
    decreasing log‑sum‑exp smoothing parameter (lam).  The schedule
    starts with relatively large smoothing and ends at ≈1e‑4.  This
    rapidly finds a good basin.

3️⃣  Refined optimisation – the same L‑BFGS routine is run again with a
    finer lam‑schedule and many more internal iterations.

4️⃣  Exponentiated‑gradient (sub‑gradient) descent – the true
    max‑convolution objective is minimised directly for a long period.

5️⃣  Advanced hill‑climbing – at each step the index that contributes
    most to the current maximal convolution entry (donor) and the index
    with the smallest partner value (receiver) are identified.  A
    line‑search moves a small fraction of mass from donor to receiver
    while re‑evaluating the exact objective.  If no improvement is
    possible the step size is reduced.  Multiple donor/receiver pairs
    are examined per iteration (default 5 × 5) to escape shallow
    plateaus.

The final sequence is finally evaluated with the provided
`evaluate_sequence` routine and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_spectral_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random‑phase flat‑magnitude spectrum, projected onto the simplex.
    This often yields diverse non‑symmetric shapes.
    """
    L = next_pow2(2 * n - 1)
    # magnitude constant (1.0) – only the phase varies
    mags = np.ones(L)
    # generate random phases respecting Hermitian symmetry
    phases = np.zeros(L)
    # frequencies 1 .. L/2-1
    half = L // 2
    rand = rng.random(half - 1) * 2.0 * np.pi
    phases[1:half] = rand
    phases[half + 1:] = -rand[::-1]
    # handle Nyquist for even L
    if L % 2 == 0:
        phases[half] = 0.0
    # construct complex spectrum
    X = mags * np.exp(1j * phases)
    a_complex = np.fft.ifft(X)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns value and gradient w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (coarse / refined)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS through a sequence of smoothing parameters.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size (classic 1/√t schedule)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Advanced deterministic hill‑climb with line‑search
# ----------------------------------------------------------------------


def line_search_move(a: np.ndarray, i_d: int, i_r: int,
                     max_delta: float, cur_score: float,
                     fft_len: int, n: int,
                     max_iter: int = 12) -> tuple[np.ndarray, float]:
    """
    Perform a 1‑D golden‑section line search for the optimal amount of
    mass to move from i_d (donor) to i_r (receiver).  Returns the new
    vector and the achieved score.  If no improvement is possible,
    returns (None, cur_score).
    """
    if max_delta <= 0.0:
        return None, cur_score

    lo, hi = 0.0, max_delta
    best_a = None
    best_val = cur_score

    phi = (np.sqrt(5) - 1) / 2  # 0.618...

    # golden‑section search
    for _ in range(max_iter):
        m1 = lo + (1 - phi) * (hi - lo)
        m2 = lo + phi * (hi - lo)

        a1 = a.copy()
        a1[i_d] -= m1
        a1[i_r] += m1
        # sum is invariant but we re‑normalise to guard against round‑off
        a1 = a1 / a1.sum()
        s1 = fast_score(a1, fft_len, n)

        a2 = a.copy()
        a2[i_d] -= m2
        a2[i_r] += m2
        a2 = a2 / a2.sum()
        s2 = fast_score(a2, fft_len, n)

        if s1 < s2:
            hi = m2
            if s1 < best_val:
                best_val = s1
                best_a = a1
        else:
            lo = m1
            if s2 < best_val:
                best_val = s2
                best_a = a2

    if best_a is not None and best_val < cur_score - 1e-12:
        return best_a, best_val
    return None, cur_score


def advanced_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        top_k: int = 5,
                        init_step: float = 0.20,
                        min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Repeatedly relocate a small fraction of mass from the most harmful
    coordinates (largest a[i]·partner[i]) toward coordinates with the
    smallest partner value.  A line‑search determines the optimal amount
    of mass to move for each donor/receiver pair.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        conv = full_conv(a, fft_len)
        max_idx = int(np.argmax(conv))

        # build partner array for this max index
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution of each coordinate to the maximal entry
        contrib = a * partner

        # donor indices (largest contribution) and receiver indices (smallest partner)
        donors = np.argsort(contrib)[-top_k:]
        receivers = np.argsort(partner)[:top_k]

        improved = False
        best_move_a = a
        best_move_score = cur_score

        for i_d in donors:
            for i_r in receivers:
                if i_d == i_r:
                    continue
                max_delta = a[i_d] * step
                cand_a, cand_score = line_search_move(a, i_d, i_r,
                                                      max_delta,
                                                      cur_score,
                                                      fft_len,
                                                      n)
                if cand_a is not None and cand_score < best_move_score:
                    best_move_a = cand_a
                    best_move_score = cand_score
                    improved = True

        if improved:
            a = best_move_a
            cur_score = best_move_score
            # keep the current step size for the next iteration
        else:
            # no improving move – shrink the step
            step = max(step * 0.7, min_step)
            if step <= min_step:
                break
    return a, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (feel free to change n – larger n gives more freedom)
    # --------------------------------------------------------------
    n = 4000                      # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise candidates (use previous solution if available)
    # --------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # previous solution (if any)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.maximum(a_prev, 0.0)
            a_prev = a_prev / a_prev.sum()
            candidates.append(a_prev)

    # spectral deterministic start
    candidates.append(spectral_init(n))

    # random initialisations
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # random‑phase spectral initialisations
    for _ in range(5):
        candidates.append(random_spectral_initial(n, rng))

    # keep the best of the seeds
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # Rough allocation (feel free to tune)
    phase1_end = start_time + 0.12 * budget_s   # coarse L‑BFGS
    phase2_end = start_time + 0.45 * budget_s   # refined L‑BFGS
    phase3_end = start_time + 0.80 * budget_s   # exponentiated‑gradient
    phase4_end = deadline                       # advanced hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – coarse L‑BFGS (few iterations, aggressive smoothing)
    # --------------------------------------------------------------
    lam_schedule_coarse = [0.5, 0.35, 0.25, 0.18, 0.12,
                           0.08, 0.05, 0.03, 0.018, 0.01,
                           0.006, 0.003, 0.001, 0.0005]
    maxiter_coarse = 600

    while time.time() < phase1_end:
        # pick a random start (biased towards good candidates)
        if rng.random() < 0.5:
            a_start = random_initial(n, rng)
        elif rng.random() < 0.7:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.008), rng)
        else:
            a_start = spectral_init(n)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule_coarse,
            maxiter_coarse,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – refined L‑BFGS (more iterations, finer smoothing)
    # --------------------------------------------------------------
    lam_schedule_fine = [
        0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05,
        0.03, 0.02, 0.015, 0.01, 0.008, 0.005,
        0.003, 0.0015, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    maxiter_fine = 2000

    best_a, best_score = lbfgs_continuation(
        best_a,
        lam_schedule_fine,
        maxiter_fine,
        phase2_end,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[Phase‑2] refined L‑BFGS best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (long run)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Phase‑3] exp‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – advanced deterministic hill‑climb with line‑search
    # --------------------------------------------------------------
    best_a, best_score = advanced_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase4_end,
        top_k=5,
        init_step=0.20,
        min_step=1e-6,
    )
    print(f"[Phase‑4] advanced hill‑climb final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
