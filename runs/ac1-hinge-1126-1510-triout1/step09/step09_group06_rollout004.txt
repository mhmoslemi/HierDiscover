def adam_optimize(a_start, lambda_schedule, maxiter_per_lambda, deadline, fft_len, n):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    for lam in lambda_schedule:
        for it in range(maxiter_per_lambda):
            if time.time()>=deadline:
                break
            t+=1
            f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y = y - 0.01 * m_hat / (np.sqrt(v_hat) + eps)
    a = softmax(y)
    return a

"""
Propose a low‑score step function.

Algorithm
---------
1. Build a diverse initial pool (previous best, spectral, random draws,
   parametric families – exponential, power‑law, edge, triangular,
   Gaussian). Keep the best few as seeds.
2. Adam‑based continuation on a smoothed log‑sum‑exp approximation of
   the max‑convolution. The smoothing factor (λ) is reduced on a
   schedule, each stage performing several Adam iterations.
3. Exact exponentiated‑gradient descent on the true max‑convolution.
4. Projected sub‑gradient descent (larger initial step).
5. Two phases of greedy hill‑climbing:
   * ordinary random mass transfers,
   * a targeted move that reduces mass on indices contributing to the
     current maximal convolution entry.
All phases respect the global 1000 s budget and keep the best sequence
found. The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) assuming a already sums to 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution via log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_continuation(a_start: np.ndarray,
                     lam_factors,
                     max_iter_per_lambda: int,
                     deadline: float,
                     fft_len: int,
                     n: int):
    """
    Adam optimisation on the smoothed max‑convolution.
    Returns the best a and its fast_score encountered.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_base = 0.02
    t = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_factors:
        for it in range(max_iter_per_lambda):
            if time.time() >= deadline:
                break
            f, grad = objective_and_grad(y, n, fft_len, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y = y - step_base * m_hat / (np.sqrt(v_hat) + eps)

            # occasional evaluation
            if it % 120 == 0:
                a = softmax(y)
                sc = fast_score(a, fft_len, n)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        if time.time() >= deadline:
            break

    # final evaluation
    a = softmax(y)
    sc = fast_score(a, fft_len, n)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps a on the probability simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1

    return best_a, best_score


def subgrad_optimize(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     step0: float = 0.5):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses a larger initial step size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that preserves the sum
        d = -(g - np.mean(g))

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        # projection onto the simplex
        v = a_new - np.max(a_new)  # shift for stability
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a_new = np.maximum(v - theta, 0.0)
        else:
            a_new = np.full_like(a, 1.0 / n)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 40000):
    """
    Greedy hill‑climb: random mass transfer from i to j, accepting only
    strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def targeted_hill_climb(a: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 200000):
    """
    Targeted hill‑climb: identify the convolution entry attaining the
    maximum, pick a contributing pair (i,j) and move a small fraction
    of the larger coefficient to a random other position.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        # compute the convolution and the index(s) attaining max
        fa = np.fft.rfft(best, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        max_idxs = np.where(conv >= max_val - eps)[0]
        k = rng.choice(max_idxs)

        # choose a pair (i,j) with i+j = k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i = rng.integers(i_low, i_high + 1)
        j = k - i

        # pick the larger of the two to be the source
        src = i if best[i] > best[j] else j
        if best[src] <= 0.0:
            it += 1
            continue

        # target index (different from src)
        tgt = rng.integers(0, n)
        while tgt == src:
            tgt = rng.integers(0, n)

        # move a small fraction (≤ 2 %) of the source mass
        delta = best[src] * rng.random() * 0.02
        if delta <= 0.0:
            it += 1
            continue

        best[src] -= delta
        best[tgt] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta
        it += 1
    return best, best_score


def prune_small(a: np.ndarray, eps: float = 1e-9) -> np.ndarray:
    """Zero out entries below eps and renormalise to sum‑1."""
    a = np.where(a < eps, 0.0, a)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full_like(a, 1.0 / a.size)
    return a

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random draw from several simple families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    interior_zero forces interior mass to zero.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular (pyramid) shape."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = idx.astype(float)
    return a / a.sum()


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index range."""
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return a / a.sum()


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """
    Exhaustively explore a handful of parametric families and keep the
    best few distinct candidates.
    """
    candidates = []

    # Exponential family
    for c in np.logspace(-4, -0.5, num=12):
        a = exponential_shape(n, c)
        candidates.append((fast_score(a, fft_len, n), a))

    # Power‑law family
    for p in np.linspace(0.1, 4.0, num=15):
        a = powerlaw_shape(n, p)
        candidates.append((fast_score(a, fft_len, n), a))

    # Edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # Triangular
    a_tri = triangular_shape(n)
    candidates.append((fast_score(a_tri, fft_len, n), a_tri))

    # Gaussian with several sigmas
    for sigma in np.linspace(0.1 * n, 0.4 * n, num=6):
        a_gauss = gaussian_shape(n, sigma)
        candidates.append((fast_score(a_gauss, fft_len, n), a_gauss))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # Keep the best distinct candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 8)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluation score.
    Returns a list of floats that can be fed to `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                       # length of the step function
    fft_len = next_pow2(2 * n - 1) # power‑of‑two for fast FFTs

    # --------------------------------------------------------------
    # Gather a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best, if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2️⃣  Spectral and random seeds
    candidates.append(spectral_init(n))
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # 3️⃣  Sweep parametric families
    family_best = sweep_parameter_families(n, fft_len, rng, keep=6)
    candidates.extend(family_best)

    # --------------------------------------------------------------
    # Pick the best initial seed (fast_score)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – Adam continuation on smoothed max‑convolution
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lambda = 800

    seeds = [best_a] + [random_initial(n, rng) for _ in range(2)]
    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = adam_continuation(a_start,
                                          lam_schedule,
                                          maxiter_per_lambda,
                                          phase1_deadline,
                                          fft_len,
                                          n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len)
        print(f"[exp‑grad] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (larger step0)
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = subgrad_optimize(best_a,
                                              phase3_deadline,
                                              n,
                                              fft_len,
                                              step0=0.8)
        print(f"[subgrad] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy hill‑climbing (ordinary + targeted)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=100000)
        print(f"[hill‑climb] after generic hill‑climb = {best_score:.6f}")

        best_a, best_score = targeted_hill_climb(best_a,
                                                best_score,
                                                fft_len,
                                                n,
                                                deadline,
                                                max_iter=200000)
        print(f"[targeted] after targeted hill‑climb = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final cleanup and evaluation
    # --------------------------------------------------------------
    best_a = prune_small(best_a, eps=1e-9)
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
