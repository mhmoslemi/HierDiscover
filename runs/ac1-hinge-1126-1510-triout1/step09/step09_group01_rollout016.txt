assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It builds a good starting vector from many cheap parametric families,  
optimises it with a large‑scale Adam optimiser on a smooth‑max surrogate,  
then refines it with an aggressive projected sub‑gradient descent that
uses back‑tracking line‑search and occasional random restarts.  
A short simulated‑annealing phase is added at the end for stochastic
fine‑tuning.  The whole routine respects the 1000 s deadline and prints
progress information.

```python
"""
Propose a low‑score step sequence.

Algorithm
---------

1. **Cheap parametric sweep** (≈ 5 % time) – sample many simple
   families (exponential, centred exponential, Gaussian, cosine,
   bimodal, logistic, Dirichlet, random) and keep the best.

2. **Adam on a smoothed max** (≈ 35 % time) – minimise a
   log‑sum‑exp approximation of the max‑convolution.  The parameters
   are kept in the log‑softmax (guaranteeing the simplex).  A
   geometric schedule of smoothing factors drives the optimiser from a
   very smooth landscape to the true max.

3. **Projected sub‑gradient descent** (≈ 55 % time) – use the exact
   sub‑gradient of the max‑convolution, a momentum buffer and a
   back‑tracking line search.  If no improvement is seen for a while,
   the current solution is randomly jittered to escape possible local
   minima.

4. **Simulated annealing** (the last few seconds) – stochastic mass
   swaps with a linearly decaying temperature provide a final polish.

The best vector encountered at any moment is returned.  All operations
are carried out with NumPy and FFTs for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    out = e / e.sum()
    return out


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the first 2·len(a)-1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector already normalised to sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families used in the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    idx = np.arange(n, dtype=float)

    if family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * idx)

    elif family == "exp_centered":
        rate = rng.uniform(0.001, 0.02)
        mu = (n - 1) / 2.0
        a = np.exp(-rate * np.abs(idx - mu))

    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (idx - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)

    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((idx - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((idx - mu2) / sigma2) ** 2)
        a = a1 + a2

    elif family == "logistic":
        scale = rng.uniform(0.001, 0.02)
        mu = (n - 1) / 2.0
        a = 1.0 / (1.0 + np.exp(scale * (idx - mu)))

    elif family == "dirichlet":
        alpha = rng.uniform(0.05, 5.0)
        a = rng.gamma(alpha, size=n)

    elif family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = (idx + 1) ** p
        else:
            a = (n - idx) ** p

    elif family == "random":
        a = rng.random(n)

    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int, fft_len: int, budget_s: float, rng: np.random.Generator):
    """Cheap sweep over many random parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s
    families = [
        "exp",
        "exp_centered",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "dirichlet",
        "power",
        "random",
    ]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# ADAM optimiser on log‑sum‑exp (smooth max) objective
# ----------------------------------------------------------------------


def smooth_objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                              fft_len_g: int, lam_factor: float):
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑prop through convolution (a is softmax of y)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y


def adam_optimize(a_start: np.ndarray, lam_factors, iters_per_lam: int,
                  deadline: float, fft_len_conv: int, fft_len_g: int, n: int):
    """Adam optimisation on the smoothed‑max objective."""
    y = np.log(a_start + 1e-12)               # log‑softmax parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = smooth_objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.01 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score

# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, deadline: float,
                        fft_len: int, n: int, rng: np.random.Generator):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses a momentum buffer, back‑tracking line search and occasional
    random restarts if progress stalls.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    cur_score = best_score

    momentum = np.zeros_like(a)
    beta = 0.95                 # stronger momentum
    step = 2.0                  # initial step size
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # --- compute sub‑gradient of the max convolution -----------------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        improved = False
        # try a few shrink factors for robustness
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_candidate = simplex_projection(a - step * factor * momentum)
            sc = fast_score(a_candidate, fft_len, n)
            if sc < cur_score - 1e-12:          # strict improvement
                a = a_candidate
                cur_score = sc
                improved = True
                # enlarge step a bit after a successful move
                step = min(step * 1.2, 10.0)
                if sc < best_score:
                    best_score, best_a = sc, a_candidate.copy()
                break

        if not improved:
            # no progress – shrink step size
            step *= 0.5
            no_improve += 1

        # restart if stuck for a while
        if no_improve >= 3000:
            # jitter around the best known point
            jitter = 0.05 * rng.normal(1.0, 0.02, size=n)
            a = np.maximum(best_a * jitter, 0.0)
            a /= a.sum()
            momentum = np.zeros_like(a)
            step = 2.0
            no_improve = 0
            cur_score = best_score

        it += 1

    return best_a, best_score

# Simulated annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, start_score: float,
                        fft_len: int, n: int,
                        deadline: float, rng: np.random.Generator):
    """Stochastic hill‑climb that respects mass conservation."""
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.30   # allow larger moves
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
```
