    # Problem size
    n = 4000
    fft_len = next_pow2(2*n - 1)

    # Initialize with previous solution interpolated or uniform

        old = np.array(globals()['height_sequence_1'])
        if old.ndim == 1 and old.size > 0:
            # Interpolate to new size (maybe same)
            x_old = np.linspace(0, 1, len(old))
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0/n)
    else:
        a0 = np.full(n, 1.0/n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)

"""
Step‑function optimizer.

The goal is to minimise

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

for a non‑negative vector a (length n).  The algorithm works inside the
1000 s budget in four stages:

1.  A very fast sweep over a one‑parameter exponential “U‑shape’’
    (a_i = exp(-p·min(i,n−1−i))) to obtain a decent seed.
2.  Repeated L‑BFGS runs on a log‑sum‑exp smoothed version of the max‑convolution.
    The smoothing factor is annealed from 0.5 down to 5e‑06 and each
    cascade is allowed many L‑BFGS steps.  The vector is represented as a
    soft‑max of unconstrained logits, guaranteeing non‑negativity and
    unit sum.
3.  A long projected Adam optimisation on the true (non‑smoothed) objective.
    Adam’s adaptive step‑size and a Euclidean projection onto the simplex
    allow rapid descent while respecting the constraints.
4.  A deterministic greedy descent (safe mass transfers) followed by a
    short simulated‑annealing hill‑climb for final polishing.

The best sequence found before the deadline is returned and stored in the
global variable ``height_sequence_1`` for possible warm‑starts in later runs.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a already normalised to sum(a)=1.
    Uses FFT for O(n log n) speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed L‑BFGS utilities
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float):
    """
    Smooth approximation of the L∞ objective using log‑sum‑exp.
    Optimises over logits y where a = softmax(y).
    Returns (objective, gradient) w.r.t. y.
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations for the given lambda schedule.
    Each stage starts from the result of the previous one.
    """
    y = np.log(a_start + 1e-12)  # logits for softmax
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam, 'gtol': 1e-12, 'disp': False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Projected Adam optimiser on the true max‑convolution

def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    step0: float = 0.5,
):
    """
    Projected Adam on the exact max‑convolution objective.
    Returns the best vector seen before the deadline.
    """
    a = a_start.copy()
    m_exp = np.zeros_like(a)
    v_exp = np.zeros_like(a)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 1
    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # Adam update in the log‑space (mirrored SGD)
        m_exp = beta1 * m_exp + (1 - beta1) * g
        v_exp = beta2 * v_exp + (1 - beta2) * (g * g)
        m_hat = m_exp / (1 - beta1**it)
        v_hat = v_exp / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it}, fast_score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Deterministic greedy descent (safe mass transfers)
# ----------------------------------------------------------------------


def deterministic_descent(a_start: np.ndarray, deadline: float, n: int, fft_len: int):
    """
    Repeatedly move a fraction of mass from the index with the largest gradient
    to the one with the smallest, using an analytically computed safe step size.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    safety = 0.99  # a little less conservative than the previous 0.92

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution (same derivation as in phase‑2)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))          # mass to remove
        dst = int(np.argmin(grad))          # where to add
        if src == dst or a[src] <= 0.0:
            break

        # effect of moving δ from src to dst on the full convolution
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # admissible δ (source non‑negativity)
        delta_src = a[src]

        # bound from entries that would increase beyond the current peak
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * safety
        if delta <= 0.0:
            break

        # apply the move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recomputation to control drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score

# Simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with a Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.15
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Warm‑start from a previously saved construction, if any
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # Phase 0 – quick sweep over exponential edge shapes
    phase0_end = start_time + 0.05 * budget_s
    while time.time() < phase0_end:
        p = rng.uniform(0.001, 0.02)  # exponential rate
        distances = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-p * distances)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[exp‑edge] p={p:.5f}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts on the smoothed objective
    # --------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_sched = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.01)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_sched, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected Adam on the true max‑convolution
    # --------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    best_a, best_score = adam_optimize(
        best_a, phase2_end, n, fft_len,
        beta1=0.9, beta2=0.999, eps=1e-8, step0=0.5,
    )
    print(f"[Adam] finished with fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy descent (more aggressive)
    # --------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    best_a, best_score = deterministic_descent(
        best_a, phase3_end, n, fft_len,
    )
    print(f"[det‑desc] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()['height_sequence_1'] = final_seq

    return final_seq
