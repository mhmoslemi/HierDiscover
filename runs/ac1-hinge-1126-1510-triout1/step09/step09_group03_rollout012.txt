"""
Step‑function optimizer (≈ 1 000 s target)

The scoring function is

    score(a) = 2·n·max(conv(a,a))          with   ∑a = 1 , a_i ≥ 0

where ``conv`` is the *linear* convolution.
The global minimum is 2·n·(1/(2n‑1)) ≈ 1, i.e. a perfectly flat
self‑convolution.  The code below follows the successful four‑phase
strategy of the reference implementation but makes three systematic
improvements:

1. **Larger problem size** – we use n = 8000 (≈ twice the previous size);
   the theoretical lower bound becomes 1.00006, giving us more “room”
   for improvement.

2. **More aggressive smooth L‑BFGS** – a finer λ‑schedule and a larger
   maximum number of L‑BFGS iterations per λ (600 → 2000).  The smooth
   objective is the log‑sum‑exp approximation of ``max(conv)``; a dense
   schedule drives the solution to a very flat convolution.

3. **Extended deterministic refinement** – after the stochastic phases
   we run the analytic “mass‑transfer” descent for the rest of the
   budget (≈ 15 % of the time).  The step size is obtained by a
   binary‑search on the admissible δ, which is more accurate than the
   previous heuristic factor of 0.92.

All phases keep the globally best vector and return it after the
deadline.  The final sequence is a probability vector (∑a = 1) and
conforms to the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) – assumes ∑a = 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'edge', 'bimodal', 'sparse', 'fourier']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        frac = rng.uniform(0.006, 0.03)
        m = max(1, int(n * frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # fourier – random phases, then rectify
        phases = rng.random(n) * 2 * np.pi
        fourier = np.exp(1j * phases)
        a = np.real(np.fft.ifft(fourier))
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Phase‑1 – Smoothed L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    The optimisation variable is the log‑softmax vector y,
    i.e. a = softmax(y) lives on the simplex.
    Returns (objective, gradient) w.r.t. y.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smoothing parameter (lam) is a small fraction of the current peak
    lam_eff = max_c * lam + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                         # ∂ smooth_max / ∂ conv_i
    # back‑prop through the convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_conv = conv_rev[n - 1: n - 1 + n]    # size n

    # chain rule through softmax
    mean_g = np.dot(a, grad_conv)
    grad_y = a * (grad_conv - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS minimisations, each with a smaller smoothing
    parameter.  The variable is kept in log‑softmax space to avoid explicit
    projections.
    """
    y = np.log(a_start + 1e-12)               # log‑softmax coordinates
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Exponentiated‑gradient descent on the hard max‑conv.
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
):
    """
    Mirrored (exponentiated) sub‑gradient descent on the exact max‑conv.
    The step‑size follows ηₜ = η₀ / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution and mask of the (almost) active entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max (matrix‑vector product via FFT)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step‑size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated update (clipped exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score

def max_conv_given_a(a: np.ndarray) -> float:
    """Return max(conv(a,a))."""
    # uses same FFT length as elsewhere (next power of two)
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv.max()


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """
    Repeatedly move a fraction of the mass from the index with the largest
    gradient (most harmful for the current peak) to the index with the
    smallest gradient.  The maximal permissible δ is found by a short
    binary search, guaranteeing that the peak never rises.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # contribution of moving δ from src to dst:
        #   conv_new = conv + 2·δ·diff ,  diff = e_dst * a – e_src * a
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # binary search for the largest δ that does not increase the peak
        lo, hi = 0.0, a[src]
        for _ in range(30):                     # 30 iterations → ~1e‑9 precision
            mid = (lo + hi) / 2.0
            conv_test = conv + 2.0 * mid * diff
            if conv_test.max() <= max_val + 1e-14:
                lo = mid
            else:
                hi = mid
        delta = lo * 0.9999      # a tiny safety factor

        if delta <= 0.0:
            break

        # apply move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Classic Metropolis annealing with mass transfers."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.12                     # move at most 12 % of source mass
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of convolution
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        elif rng.random() < np.exp(-dE / temp):
            accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    start_time = time.time()
