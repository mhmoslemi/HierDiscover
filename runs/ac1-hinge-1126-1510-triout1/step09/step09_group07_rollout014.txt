def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return the exact objective for a probability vector a (sum=1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best={best_score}")

    deadline = start + budget_s - 0.5

    # Phase 1: L-BFGS heavy
    lam_schedule = np.logspace(-0.1, -10, num=30)  # from 0.8 down to 1e-10
    maxiter_per_lam = 3000
    while time.time() < deadline * 0.3 + start:   # allocate 30%

        a_opt, sc_opt = lbfgs_continuation(a0, lam_schedule, maxiter_per_lam, deadline*0.3+start, fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] new best {best_score}")

    # Phase 2: Alternate LP improvement
    # we will perform up to 50 iterations or time left
    def lp_step(a, M):
        # ensure positivity
        a_eps = np.maximum(a, 1e-12)
        c = np.concatenate([a_eps, np.zeros(n-1)])
        r = np.concatenate([a_eps[:1], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(c, r)[:2*n-1, :]
        # LP variable
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-09, reltol=1e-09)
        except cp.error.SolverError:
            return a
        if g.value is None:
            return a
        g0 = np.maximum(g.value, 0.0)
        sum_g = g0.sum()
        if sum_g == 0:
            return a
        return g0 / sum_g

    # Run LP alternating for some iterations
    while time.time() < deadline * 0.8 + start:
        M = np.max(np.fft.irfft(np.fft.rfft(best_a, fft_len)*np.fft.rfft(best_a, fft_len), fft_len)[:2*n-1])
        # computed exact max conv
        new_a = lp_step(best_a, M)
        # mix with t
        t = 0.3  # mixing weight
        mixed = (1 - t) * best_a + t * new_a
        mixed = mixed / mixed.sum()
        new_score = fast_score(mixed, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = mixed.copy()
            print(f"[LP] new best {best_score}")

    # Phase 3: Exponentiated gradient long run for remaining time
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(best_a, deadline, n, fft_len, fft_len, rng)
        print(f"[expgrad] final {best_score}")

    # Final hill climb
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a, deadline, n, fft_len, rng, init_temp=1e-4, final_temp=1e-9, max_moves=80000)
        print(f"[SA] final {best_score}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated = {final_eval}")
    return final_seq

"""
Propose a low‑score step function.
==================================
The function builds a diverse pool of initial distributions (uniform,
spectral, triangular, Hann, Gaussian, Dirichlet‑like).  It then
optimises the distribution by many rounds of smoothed‑max L‑BFGS
(gradients are obtained analytically via FFT), followed by a long
exponentiated‑gradient descent on the exact max‑convolution, and finishes
with an aggressive hill‑climb.  The whole search respects a 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum a = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return the exact nested evaluation for a probability vector a
    (∑a = 1).  The returned value is 2·n·max(a * a).
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Parameters
    ----------
    y : np.ndarray
        Logits; a = softmax(y) lives on the simplex.
    n, fft_len_conv, fft_len_g : int
        Sizes needed for FFT.
    lam_factor : float
        Relative smoothing strength (0 < lam_factor ≪ 1).
    Returns
    -------
    f, grad_y : float, np.ndarray
        Smoothed objective value and its gradient w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Heuristic spectral initialisation – inverse FFT of sqrt of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u = np.zeros(L)
    u[:target_len] = uniform
    S = np.fft.fft(u)
    sqrt_amp = np.sqrt(np.abs(S))
    sqrt_phase = np.angle(S) / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse Dirichlet‑like random vectors."""
    mode = rng.choice(['uniform', 'exponential', 'power', 'bimodal', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exponential':
        a = rng.exponential(scale=1.0, size=n)
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        frac = rng.uniform(0.005, 0.02)
        m = max(1, int(n * frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        l = max(0, centre - width)
        r = min(n, centre + width)
        a[l:r] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall – simple triangular window."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run a cascade of L‑BFGS optimisations for each smoothing factor
    `lam_factor` in `lam_schedule`.  The optimisation works in log‑space
    (logits y) and returns the best distribution found in this cascade.
    """
    y = np.log(a_start + 1e-12)          # interior point for softmax

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Update rule: a ← a·exp(−η·g) keeps a on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution a*a
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # diminishing step size
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a /= s
        else:                               # safety fallback
            a = best_a.copy()

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Simple strict‑improvement hill‑climb that transfers a tiny fraction
    of mass between random indices.
    """
    rng = np.random.default_rng(123456)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    """

    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                      # reasonable size for the budget
    fft_len = next_pow2(2 * n - 1)   # length for FFTs (power of two)

    # --------------------------------------------------------------
    # Build initial candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution is present, interpolate it
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))

    # a few Dirichlet‑like draws
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    # pick the best starter
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Heavy L‑BFGS polishing (30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    # a long schedule that drives the smoothing factor to near zero
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,

        2e-06, 1e-06, 5e-07, 2e-07, 1e-07
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_deadline:
        # blend the current best with a fresh random vector for diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.6
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Long exponentiated‑gradient descent (50 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s

        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len, fft_len,
                                                   rng)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

                                        best_score,
