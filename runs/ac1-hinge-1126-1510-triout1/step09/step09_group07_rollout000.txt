def transfer_step(a, g, step_frac):
    pos = np.where(g > 0)[0]
    neg = np.where(g < 0)[0]
    if len(pos)==0 or len(neg)==0:
        return a
    # choose random i from pos, j from neg
    i = rng.choice(pos)
    j = rng.choice(neg)
    delta = min(a[i] * step_frac, step_frac * 0.5) # limit
    if delta <= 1e-12: return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # ensure nonnegative (should hold)
    if a_new[i] < 0:
        # adjust delta to keep nonneg
        delta = a[i]
        a_new[i] = 0
        a_new[j] = a[j] + delta
    return a_new
```

"""
Propose a low‑score step function.

Algorithm
---------
* Build a diverse pool of initial sequences (windows, random Dirichlet,
  edge‑heavy, sinc‑squared, etc.).
* Pick the best from the pool (fast_score = 2·n·max(conv) for a
  probability vector).
* Phase 1 – many short L‑BFGS runs on a log‑sum‑exp smoothed version of
  the max‑convolution.  The smoothing factor (lam) is gradually reduced.
* Phase 2 – a long “exact‑max” exponentiated‑gradient descent (mirror
  descent on the probability simplex).  Step size decays as 1/√t.
* Phase 3 – deterministic mass‑transfer local search that moves a little
  mass from indices with positive sub‑gradient to those with negative
  sub‑gradient (the gradient of the current max‑convolution).
* Phase 4 – tiny greedy hill‑climb (strictly improving random pair
  transfers).
* The best sequence found is returned.  The sequence is a probability
  vector (sum = 1); `evaluate_sequence` rescales it automatically, so the
  returned score equals the internal fast_score.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import kaiser, blackman


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (used during optimisation)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective + gradient (for L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp smoothed version of max(conv) and its gradient.
    y are logits, a = softmax(y) is the probability vector."""
    a = softmax(y)

    # self‑convolution (full length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    w = kaiser(n, beta).astype(float)
    w = w - w.min()
    w /= w.sum()
    return w


def blackman_init(n: int) -> np.ndarray:
    w = blackman(n).astype(float)
    w = w - w.min()
    w /= w.sum()
    return w


def sinc_squared_init(n: int, sigma_factor: float = 0.20) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    x = (i - centre) / sigma
    w = np.sinc(x) ** 2
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random families on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # dirichlet‑like
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha, 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_factors,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """Run L‑BFGS for each smoothing factor in *lam_factors*.
    Returns best a and its fast_score."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # decreasing step size
        eta = 0.6 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def mass_transfer_search(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         max_iter: int = 30000):
    """Deterministic local search: move a little mass from
    indices with positive sub‑gradient to those with negative."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max‑convolution (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = rng.choice(pos)
        j = rng.choice(neg)

        # transfer a modest fraction of the mass at i
        delta = min(a[i] * 0.04, 0.02 * (g[i] - g[j]))
        if delta <= 0.0:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # protect against numerical negativity
        if a_candidate[i] < 0.0:
            a_candidate[j] += a_candidate[i]   # compensate
            a_candidate[i] = 0.0

        score = fast_score(a_candidate, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 20000):
    """Very small greedy hill‑climb (strictly improving random transfers)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[fine‑L‑BFGS] new best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent (exact max)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.65 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len,
            rng)
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic mass‑transfer local search
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.85 * budget_s
    if time.time() < phase4_deadline:
        best_a, best_score = mass_transfer_search(
            best_a,
            phase4_deadline,
            n,
            fft_len,
            rng,
            max_iter=50000)
        print(f"[mass‑transfer] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=25000)
        print(f"[final hill‑climb] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence will re‑scale)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
