"""
Multi‑phase optimizer for the step‑function evaluation problem.

The algorithm works in three (optional) stages within the given time budget:

1. **Heavy L‑BFGS** – many random restarts on a smooth log‑sum‑exp
   surrogate of the maximum convolution.  The smoothing parameter is
   driven down to near zero, and each restart is allowed many L‑BFGS
   iterations.

2. **Projected sub‑gradient descent** – using the exact sub‑gradient of
   the nonsmooth max‑convolution.  A simple diminishing‑step scheme with
   simplex projection is applied.

3. **Simulated‑annealing hill‑climb** – aggressive transfers of mass
   from high‑gradient to low‑gradient indices, with a temperature schedule
   that permits occasional uphill moves.

All phases respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list; the external
``evaluate_sequence`` routine will be used only once at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming a already satisfies sum(a)=1.
    Uses FFT for the linear convolution (full length 2·n‑1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

# Smoothed surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).

    Parameters
    ----------
    y : np.ndarray
        The log‑probability logits.
    n : int
        Length of the step function.
    fft_len_conv, fft_len_g : int
        FFT lengths for the convolution and for the back‑propagation step.
    lam_factor : float
        Relative smoothing factor (λ ≈ lam_factor * max(conv)).

    Returns
    -------
    f : float
        Smoothed objective (to be minimised).
    grad_y : np.ndarray
        Gradient w.r.t. y.
    """
    # ---- softmax (stable) ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution a * a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max w.r.t. convolution entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ---- back‑propagation through the convolution ----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through the softmax ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a sequence of L‑BFGS optimisations for each λ in ``lam_schedule``.
    The optimisation starts from ``a_start`` (converted to logits) and
    proceeds through the schedule, always keeping the best vector seen.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} → new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Return (max_conv, subgradient) for the exact max‑convolution a * a.
    The sub‑gradient is the sum of contributions from all shifts that
    attain the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def projected_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient method on the true (non‑smoothed) objective.
    Diminishing step size η_t = 1 / sqrt(t+1) is used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        step = 1.0 / np.sqrt(it + 1.0)          # classic diminishing step
        cand = a - step * grad
        cand = simplex_projection(cand)

        cur_score = fast_score(cand, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            print(f"[Subgrad] it={it} best={best_score:.6f}")

        a = cand
        it += 1
        # loosely limit iterations to avoid endless loops when time is short
        if it >= 20000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing with a temperature schedule.
    Moves mass from high‑gradient to low‑gradient indices and also tries
    random swaps.  The acceptance rule follows a simple Metropolis criterion.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9993
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        # exact max and sub‑gradient for current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.04, 0.18)               # 4‑18 % of source mass
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

# Main driver – required function

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    within the time budget.  Returns a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – double the previous length to give more “room”.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Warm‑start from a previously saved sequence if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    total_deadline = start_time + budget_s - 0.2  # tiny safety margin

    # allocate roughly 40 % to heavy L‑BFGS, 25 % to sub‑gradient,
    # 25 % to annealing, remaining to a final fine L‑BFGS.
    phase1_end = start_time + 0.40 * (budget_s - 0.2)
    phase2_end = start_time + 0.65 * (budget_s - 0.2)
    phase3_end = start_time + 0.90 * (budget_s - 0.2)

    # ------------------------------------------------------------------
    # Phase 1 – Heavy L‑BFGS with several random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        5e-01, 3e-01, 2e-01, 1.2e-01, 7e-02, 4e-02,
        2e-02, 1e-02, 5e-03, 2e-03, 1e-03, 5e-04,
        2e-04, 1e-04, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07,
    ]
    maxiter_per_lam = 1200  # more L‑BFGS iterations per λ than before

    restart_counter = 0
    while time.time() < phase1_end:
        # Occasionally diversify with a fresh random start
        if rng.random() < 0.35 or restart_counter < 3:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()

        a_opt, scr_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if scr_opt < best_score:
            best_score = scr_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

        restart_counter += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = projected_subgradient_descent(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Sub‑grad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Fine‑tuned L‑BFGS (very small smoothing) to polish
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        fine_maxiter = 600
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the external routine (once)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # The external evaluate_sequence is assumed to be pre‑imported.
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose for potential warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
