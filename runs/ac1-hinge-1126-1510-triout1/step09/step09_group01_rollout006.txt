"""
Propose a low‑score step sequence.

The algorithm works in two resolution levels.
A modest size (≈2000) is used for the bulk of the search – this makes
FFT‑based gradients cheap and allows many random restarts.  The best
candidate is then interpolated to the required length (4000) and
polished with a short sub‑gradient run.

Main phases (all time‑bound, total budget = budget_s seconds)

  0 – cheap parametric sweep (several simple families, e.g. power,
      exponential, Gaussian, bimodal, sparse random).
  1 – many random‑restart Adam optimisations on a smooth
      log‑sum‑exp approximation of max(conv(a,a)).
  2 – projected sub‑gradient descent with momentum on the true max.
  3 – exponentiated‑gradient (mirror descent) on the true max.
  4 – simulated‑annealing hill‑climb (mass moves between two sites).
  5 – up‑sample to the final length (4000) and run a final
      sub‑gradient refinement.

All iterates live on the probability simplex (∑a = 1, a_i ≥ 0) via
soft‑max or Euclidean projection.  The best sequence found at any
moment is returned and finally evaluated with the provided
evaluate_sequence() function.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used in the final version but kept for compatibility

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning 2·len(a)‑1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv(a,a)) for a vector already normalised to ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing temperature proportional to the current max
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # window of length n that corresponds to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    lam_factors,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed‑max objective.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # log‑space (soft‑max guarantees simplex)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.01 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)    # sub‑gradient of max

        # gradient w.r.t. a (through convolution with mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum & projection
        v = beta * v + (1.0 - beta) * g
        step = 1.0 / np.sqrt(it + 1.0)       # modest step size
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent on the simplex) on the true max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that conserves total mass.
    Temperature decays linearly to zero.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        # move a random fraction (up to 15 %) of the mass at i to j
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(5e-4, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(0.1 * n, 0.5 * n)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(0.05 * n, 0.5 * n)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, 0.45 * n)
        mu2 = rng.uniform(0.55 * n, n - 1)
        sigma1 = rng.uniform(0.05 * n, 0.15 * n)
        sigma2 = rng.uniform(0.05 * n, 0.15 * n)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "sym_power":
        # symmetric power‑law: a_i ∝ ((i+1)*(n-i))^p
        p = rng.uniform(-2.5, 2.5)
        idx = np.arange(n, dtype=float) + 1
        a = (idx * (n - idx + 1)) ** p
    elif family == "sparse":
        k = rng.integers(10, min(200, n))
        positions = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[positions] = rng.random(k)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families.
    Returns the best vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15
    families = [
        "power", "exp", "gauss", "cos", "bimodal", "sym_power", "sparse", "random"
    ]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


def upsample_to_length(a_small: np.ndarray, final_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    x_small = np.linspace(0.0, 1.0, a_small.shape[0])
    x_large = np.linspace(0.0, 1.0, final_n)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    final_n = 4000               # length required by the grader
    search_n = 2000              # a cheaper resolution for the heavy phases

    # pre‑compute FFT lengths for both resolutions
    fft_len_search = next_pow2(2 * search_n - 1)
    fft_len_search_g = next_pow2(3 * search_n - 2)

    fft_len_final = next_pow2(2 * final_n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a_search = np.full(search_n, 1.0 / search_n)
    best_score = fast_score(best_a_search, fft_len_search, search_n)
    print(f"[init] uniform (search_n={search_n}) fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Warm‑start from any previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0.0, 1.0, hs.size)
            x_new = np.linspace(0.0, 1.0, search_n)
            a_old = np.interp(x_new, x_old, hs)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            sc_old = fast_score(a_old, fft_len_search, search_n)
            if sc_old < best_score:
                best_a_search, best_score = a_old, sc_old
                print(f"[init] previous best (interpolated) fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # allocate time slots (percentages of total budget)
    phase0_end = start_time + 0.10 * budget_s
    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    # remaining time → simulated annealing + final refinement

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(search_n, fft_len_search, budget_s, rng)
    if sc_sweep < best_score:
        best_a_search, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart Adam on smoothed max (search space)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.8, 2e-5, num=30)   # geometric schedule
    iters_per_lam = 800                           # more work per smoothing level

    while time.time() < phase1_end:
        # initialise either by jittered current best or a fresh random family
        if rng.random() < 0.3:
            jitter = rng.normal(1.0, 0.025, size=search_n)
            a_start = best_a_search * jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", search_n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_search,
            fft_len_search_g,
            search_n,
        )
        if sc_opt < best_score:
            best_score, best_a_search = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (search space)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a_search, best_score = subgradient_optimize_momentum(
            best_a_search,
            phase2_end,
            fft_len_search,
            fft_len_search_g,
            search_n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (search space)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a_search, best_score = exp_gradient_optimize(
            best_a_search,
            phase3_end,
            fft_len_search,
            search_n,
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (search space)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a_search, best_score = simulated_annealing(
            best_a_search,
            best_score,
            fft_len_search,
            search_n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the required length and perform a short final polishing
    # ------------------------------------------------------------------
    best_a_final = upsample_to_length(best_a_search, final_n)

    # final polishing with a few thousand sub‑gradient steps (if time permits)
    remaining = deadline - time.time()
    if remaining > 5.0:  # a few seconds are enough for a short refinement
        fin_deadline = time.time() + min(remaining, 30.0)
        best_a_final, best_score_final = subgradient_optimize_momentum(
            best_a_final,
            fin_deadline,
            fft_len_final,
            next_pow2(3 * final_n - 2),
            final_n,
        )
        best_score = best_score_final
        print(f"[final‑sub‑grad] refined fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a_final.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
