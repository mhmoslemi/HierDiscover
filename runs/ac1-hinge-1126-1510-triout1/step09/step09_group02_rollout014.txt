def adaptive_exp_gradient_optimize(a_start, deadline, n, fft_len, rng,
                                 initial_eta=20.0, eta_decay=0.9):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n - 1]
        max_val = conv.max()
        # build mask close to max (tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # compute gradient g = 2 * (a_rev * mask) product
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad[n-1:n-1+n]  # shape n

        # base step size
        base_eta = initial_eta / np.sqrt(it + 1.0)
        # Generate candidate etas (geometric series)
        cand_factors = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
        best_local_a = None
        best_local_score = best_score
        for factor in cand_factors:
            eta = base_eta * factor
            exp_term = np.exp(np.clip(-eta * g, -50, 50))
            a_new = a * exp_term
            a_sum = a_new.sum()
            if a_sum <= 0:
                continue
            a_new /= a_sum
            score = fast_score(a_new, fft_len)
            if score < best_local_score:
                best_local_score = score
                best_local_a = a_new.copy()
        if best_local_a is not None:
            a = best_local_a
            # update global best
            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()
                # maybe print progress
                if it % 100 == 0:
                    print(f"[exp-grad] it={it} score={best_score:.6f}")
        else:
            # no improvement, maybe add small random perturbation
            a = add_noise(a, sigma=0.001, rng=rng)
        it += 1
    return best_a, best_score

def gradient_hill_climb(a_init, curr_score, n, fft_len, deadline, rng,
                       max_moves=500000):
    a = a_init.copy()
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(a, a)  # full conv exact length 2n-1
    # We'll maintain conv array.

    # compute active set tolerance
    # The fraction of mass moved gradually decreases.

    accepted = 0
    it = 0

    while time.time() < deadline and it < max_moves:
        # compute gradient for max conv (as before) to guide move
        max_val = conv.max()
        # mask for max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient w.r.t a (subgradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad[n-1:n-1+n]

        src = np.argmax(g)  # highest contribution
        dst = np.argmin(g)  # lowest contributor

        # if src weight too small, pick random src/dst
        if a[src] < 1e-12:
            src = rng.integers(0, n)

        # Choose delta fraction = min(a[src], max_move_fraction * a[src]) with decaying fraction
        max_frac = max(0.1, 0.5 / np.sqrt(it+1))
        delta = a[src] * max_frac * rng.random()
        if delta <= 0:
            it += 1
            continue

        # Update convolution incrementally
        # The effect of moving delta from src to dst:
        # New conv = conv + 2*delta*(diff) where diff is convolution of unit vectors.
        # Define diff as contribution matrix: unit vector at src negative, at dst positive.
        # We can compute diff vector of length 2n-1:
        # diff = np.zeros(2*n-1)
        # diff[src:src+n] -= a  (since before move a[src] includes original mass)...
        # Wait need to compute incremental effect: conv_new = (a - delta*e_src + delta*e_dst) * (a - delta*e_src + delta*e_dst)
        # Expand: conv_old - 2*delta * (a * e_src_rev) (???). Better use formula from previous code.

        # Use incremental update as done before:
        diff = np.zeros_like(conv)
        # source removal:
        diff[src:src+n] -= a
        # destination addition:
        diff[dst:dst+n] += a
        conv_new = conv + 2.0 * delta * diff

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            accepted += 1

        # occasionally recompute conv from scratch to reduce drift
        if it % 3000 == 0:
            conv = np.convolve(a, a)

        it += 1

    print(f"[gradient hill] total moves {it}, accepted {accepted}")
    return best_a, best_score

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds in four phases, all constrained by a 1000 s budget.

1. **Analytic initialisations** – a few deterministic shapes ( 1/√i,
   shifted variants, uniform, random) that are already close to the optimum.

2. **L‑BFGS on a log‑sum‑exp smoothed max** – a cascade of temperature
   schedules refines the shape while keeping the probabilities on the simplex.

3. **Adaptive exponentiated‑gradient descent on the true max** – at each
   iteration the exact sub‑gradient of the max‑convolution is computed.
   A small set of step‑sizes is tried and the best improvement is kept.
   This mirrors a line‑search and converges quickly.

4. **Gradient‑guided hill‑climbing** – the sub‑gradient identifies the
   index that contributes most to the current worst lag.  Mass is moved
   from that index to the index that contributes the least, with an
   exact incremental update of the convolution.  Accepted moves lower the
   objective; the step fraction shrinks with the iteration count.

The best sequence found is returned as a plain list of floats.  The code
uses only NumPy, SciPy and the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  Variables are logits y such that a = softmax(y).
    Returns (objective, gradient) with respect to y.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t. a_i (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient descent on the true max
# ----------------------------------------------------------------------


def adaptive_exp_gradient_optimize(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len: int,
                                  rng: np.random.Generator,
                                  eta0: float = 20.0):
    """
    Exponentiated‑gradient descent with a tiny line‑search over the step
    size.  Uses the exact sub‑gradient of the max‑convolution.
    Returns best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution and mask of active constraints
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the true max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # length n

        # base step size decays like 1/√t
        base_eta = eta0 / np.sqrt(it + 1.0)
        # try a handful of multipliers
        muls = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
        improved = False
        for m in muls:
            eta = base_eta * m
            exp_term = np.exp(np.clip(-eta * g, -50, 50))
            a_new = a * exp_term
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s
            score = fast_score(a_new, fft_len)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                # occasional progress report
                if it % 100 == 0:
                    print(f"[Exp‑Grad] it={it} score={best_score:.6f}")
                break
        if not improved:
            # shake a little to escape a plateau
            a = add_noise(a, sigma=0.0005, rng=rng)
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def gradient_hill_climb(a_start: np.ndarray,
                        start_score: float,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 600_000):
    """
    Greedy hill‑climbing guided by the sub‑gradient of the current worst
    lag.  Moves a small fraction of mass from the index with largest
    gradient contribution to the one with smallest contribution.
    Convolution is updated incrementally.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    conv = np.convolve(a, a)          # exact full convolution
    diff = np.zeros_like(conv)

    it = 0
    accepted = 0
    while time.time() < deadline and it < max_moves:
        # recompute the active mask and sub‑gradient
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        # avoid degenerate moves
        if src == dst or a[src] <= 1e-14:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        # fraction decays with the iteration count
        max_frac = max(0.05, 0.3 / np.sqrt(it + 1.0))
        delta = a[src] * max_frac * rng.random()
        if delta <= 0.0:
            it += 1
            continue

        # incremental update of convolution
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            accepted += 1

        # occasional full recomputation to curb accumulated rounding error
        if it % 2500 == 0:
            conv = np.convolve(a, a)

        it += 1
    print(f"[Hill‑Climb] iterations={it}, accepted={accepted}")
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    init_seqs = []

    # Analytic 1/√(i+β) shapes
    for beta in [0.0, 0.5, 1.0, 2.0]:
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1e-12)
        a /= a.sum()
        init_seqs.append(a)

    # Uniform
    init_seqs.append(np.full(n, 1.0 / n))

    # A few random starts
    for _ in range(3):
        a = rng.random(n)
        a /= a.sum()
        init_seqs.append(a)

    # If a previous run stored a sequence, use it as well
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            init_seqs.append(a)

    best_a = None
    best_score = np.inf

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smoothed max (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.30, 0.20, 0.12, 0.07, 0.04, 0.02,
                    0.01, 0.005, 0.002, 0.001, 5e-04,
                    2e-04, 1e-04, 5e-05, 2e-05, 1e-05]
    maxiter_per_lam = 900

    for a0 in init_seqs:

    # ------------------------------------------------------------------
    # Phase 3 – Adaptive exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
