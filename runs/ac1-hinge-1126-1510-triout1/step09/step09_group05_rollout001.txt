"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation follows four phases inside the 1000 s budget:

1. **Many short L‑BFGS runs** on a log‑sum‑exp smooth approximation of the
   maximum convolution.  Random restarts keep the search diverse.
2. **Projected sub‑gradient descent** (additive) and an **exponentiated‑gradient**
   variant (multiplicative) using the exact sub‑gradient of the max‑convolution.
   The two updates are tried alternately; the best improvement is kept.
3. **Aggressive simulated‑annealing hill‑climb** that moves mass from high‑gradient
   to low‑gradient indices.  The temperature decays slowly to allow escape
   from shallow plateaus.
4. **Final evaluation** by the provided ``evaluate_sequence`` routine.

All vectors are kept on the simplex (non‑negative, unit sum) so the objective
is simply ``2 * n * max(conv)``.  The code is self‑contained and uses only NumPy
and SciPy.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a moderately diverse point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "dirichlet"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.005, 0.03)
        m = max(1, int(edge_frac * n))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # dirichlet
        a = rng.dirichlet(np.ones(n))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Log‑sum‑exp smooth approximation of 2·n·max(conv).  The variable
    y holds unnormalised logits; a = softmax(y) is the current vector."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS for each λ in ``lam_schedule`` starting from ``a_start``."""
    y = np.log(a_start + 1e-12)  # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """Return (max_conv, subgradient) for a*a.  Sub‑gradient corresponds to the sum
    of contributions from all shifts attaining the maximum (within tolerance)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Sub‑gradient based refinements (additive & multiplicative)
# ----------------------------------------------------------------------


def projected_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Additive projected sub‑gradient descent (simple mirror descent)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        eta = 0.6 / np.sqrt(it + 1)   # step size schedule
        cand = a - eta * grad
        cand = simplex_projection(cand)
        cur = fast_score(cand, fft_len_conv, n)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            print(f"[PGD] it={it} new best={best_score:.6f}")
        # occasional jitter to leave plateaus
        if it % 2000 == 0 and it > 0:
            sigma = rng.uniform(0.0005, 0.003)
            cand = a + rng.normal(0.0, sigma, size=n)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
            a = cand
        else:
            a = cand
        it += 1
    return best_a, best_score


def exponentiated_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        eta = 6.0 / np.sqrt(it + 1)  # a relatively aggressive schedule
        exp_factor = np.exp(-eta * grad)
        cand = a * exp_factor
        s = cand.sum()
        if s == 0:
            cand = np.full_like(a, 1.0 / n)
        else:
            cand /= s
        cur = fast_score(cand, fft_len_conv, n)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            print(f"[EGD] it={it} new best={best_score:.6f}")
        a = cand
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Aggressive hill‑climb with a simple temperature schedule."""
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = cur_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9997
    min_temp = 1e-10

    it = 0
    while time.time() < deadline and it < max_iters:
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # use percentiles to pick source (high) / target (low) indices
        high_thr = np.percentile(grad, 92)
        low_thr = np.percentile(grad, 8)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.06, 0.22)
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative floats (sum = 1) that minimises the
    evaluation routine.  The function obeys the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 4000 balances quality & speed well
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start: take a few diverse candidates and keep the best
    # ------------------------------------------------------------------
    init_cands = []

    # use the previously found sequence if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            init_cands.append(a)

    # add a few random families
    for _ in range(12):
        init_cands.append(random_initial(n, rng))

    # also add a pure Dirichlet draw
    init_cands.append(rng.dirichlet(np.ones(n)))

    # keep the five best amongst them (fast_score is cheap)
    scored = [(fast_score(c, fft_len_conv, n), c) for c in init_cands]
    scored.sort(key=lambda x: x[0])
    best_a = scored[0][1].copy()
    best_score = scored[0][0]
    print(f"[init] n={n} initial best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.15  # small safety margin

    phase1_end = start + 0.38 * (budget_s - 0.15)  # heavy L‑BFGS
    phase2_end = start + 0.73 * (budget_s - 0.15)  # sub‑gradient refinements
    phase3_end = start + 0.95 * (budget_s - 0.15)  # annealing

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS continuations
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.1,
        0.05,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        5e-05,
        1e-05,
    ]
    maxiter_per_lam = 400

    while time.time() < phase1_end:
        # 30 % chance to restart from a fresh random point
        if rng.random() < 0.30:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected and multiplicative sub‑gradient refinements
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # split the remaining time roughly in half for the two variants
        mid = (time.time() + phase2_end) / 2.0

        # additive projected sub‑gradient
        best_a, best_score = projected_subgradient_descent(
            best_a, mid, fft_len_conv, fft_len_g, n
        )
        print(f"[PGD] best_score={best_score:.6f}")

        # multiplicative (exponentiated) gradient
        best_a, best_score = exponentiated_gradient_descent(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[EGD] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=500_000,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (mandatory)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
