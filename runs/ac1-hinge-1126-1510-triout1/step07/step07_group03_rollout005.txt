def optimize_sequence(a0, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    # iterate
    it = 0
    while time.time() < deadline:
        # compute gradient (subgradient of max conv)
        # we could compute mask and gradient g as earlier.
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient g
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)

def lp_refine(a, cur_score, n, fft_len_conv, fft_len_g, deadline):
    # compute conv, max M
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
    M = conv.max()
    # find active indices where conv close to M.
    eps = M * 1e-9 + 1e-12
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])
    # build sparse matrix A for active rows
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((len(active), n))
    rows = np.arange(len(active))
    for i in range(n):
        fi = a[i]
        if fi == 0:
            continue
        # For each active row k, we need column j = k - i
        # vectorized:
        # compute indices of active rows where k-i in 0..n-1
        # This can be computed with mask active_i = active - i
        col_idx = active - i
        valid = (col_idx >= 0) & (col_idx < n)
        if not np.any(valid):
            continue
        A[valid, i] = fi
    A = csr_matrix(A)
    # solve LP
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)]*n
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a, cur_score
    b_opt = np.clip(res.x, 0, None)
    sum_b = b_opt.sum()
    if sum_b <= 1 + 1e-8:
        return a, cur_score
    b_norm = b_opt / sum_b
    # evaluate new score
    new_score = fast_score(b_norm, fft_len_conv, n)
    if new_score < cur_score - 1e-8:
        print("[LP] improved from", cur_score, "to", new_score, "active", len(active))
        return b_norm, new_score
    else:
        return a, cur_score

"""
Advanced search for a low‑value step‑function sequence.

The algorithm follows a multi‑stage scheme:

1. **Initialisation** – take the provided best sequence (if any) or a
   random simplex point and normalise it.
2. **Gradient‑guided hill‑climb** – a fast deterministic local search
   that moves mass from indices with large sub‑gradient to those with
   small sub‑gradient (the “targeted hill‑climb” from the reference
   implementation).  Several 10⁴‑step iterations quickly drive the
   objective close to a local optimum.
3. **Active‑set linear‑programming refinement** – given the current
   distribution `a` and its maximal autocorrelation value `M`,
   we solve a linear program that maximises the total mass of a new
   distribution `b` under the *active* constraints
   `conv(a,b)[k] ≤ M` for those indices `k` where the convolution of
   `a` with itself is (almost) maximal.  The LP solution always yields
   `sum(b) ≥ sum(a)`.  After normalising `b` we blend it with `a`
   (convex mix) and keep the mixture only if it strictly improves the
   objective.  Repeating this step reduces the maximal autocorrelation
   further.
4. **Final polishing** – another round of the targeted hill‑climb.
5. **Exact evaluation** – the sequence is finally passed to the
   supplied ``evaluate_sequence`` function.

The combination of a very cheap gradient descent with a few exact LP
steps is able to push the score below the required 1.503 threshold
within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when ``a`` is already normalised
    (∑a = 1).  Returns ``2·n·max(a*a)``.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse point on the probability simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def subgradient_and_active_set(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """
    Return the convolution of ``a`` with itself, the sub‑gradient of the
    max‑convolution, and the list of active indices (where the
    convolution is within 1e-12 relative tolerance of the maximum).
    """
    n = a.shape[0]

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)   # active mask

    # sub‑gradient of the max‑convolution (size n)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

    # active indices (where conv is (almost) maximal)
    active = np.where(mask > 0)[0]

    return conv, max_val, g, active


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 50000) -> tuple[np.ndarray, float]:
    """Gradient‑guided hill‑climb that only accepts strict improvements."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # high / low gradient percentiles
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * 0.07   # up to 7 % of source mass
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection is not needed – the move conserves the sum and keeps
        # non‑negativity (δ ≤ best[i]).
        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1

    return best, best_score


def lp_refine(a: np.ndarray,
              cur_score: float,
              n: int,
              fft_len_conv: int,
              fft_len_g: int,
              deadline: float,
              max_iter: int = 5) -> tuple[np.ndarray, float]:
    """
    Active‑set linear‑programming refinement.
    Repeatedly solves a small LP that maximises the total mass of a new
    distribution ``b`` under the constraints ``conv(a,b) ≤ M`` for the
    currently active indices.  If a strictly better mixture is found,
    it is returned.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # 1) compute convolution and active set
        conv, M, _, active = subgradient_and_active_set(best, fft_len_conv, fft_len_g)
        if active.size == 0:
            break

        # 2) build sparse matrix representing conv(best, b) ≤ M for
        #    all active rows.
        A = lil_matrix((active.size, n), dtype=np.float64)
        # each column i contributes its weight `best[i]` to rows where
        # the index `k-i` lies inside the support of `b`.
        for i in range(n):
            fi = best[i]
            if fi == 0.0:
                continue
            col_idx = active - i
            valid = (col_idx >= 0) & (col_idx < n)
            rows = np.nonzero(valid)[0]
            if rows.size == 0:
                continue
            A[rows, i] = fi
        A = csr_matrix(A)

        # 3) solve LP: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(active.size, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            break

        b_opt = np.maximum(res.x, 0.0)
        sum_b = b_opt.sum()

        # if we cannot increase the total mass, stop
        if sum_b <= 1.0 + 1e-9:
            break

        # 4) normalise and try a convex mixture with the current best
        b_norm = b_opt / sum_b
        tau = 0.5                     # mixing fraction
        cand = (1.0 - tau) * best + tau * b_norm
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score + 1e-12 < best_score:        # strict improvement
            best = cand
            best_score = cand_score
            print(f"[LP] iteration {it} improved score to {best_score:.6f}")
        else:
            # No improvement – stop the LP refinement loop
            break

        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Starting point – use previously found sequence if available,
    # otherwise a random simplex point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # Interpolate to required length n
            old_n = init_seq.size
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = random_initial(n, rng)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – aggressive targeted hill‑climb
    # ------------------------------------------------------------------
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        deadline,
        max_iter=40000,
    )
    print(f"[hill‑climb 1] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – active‑set LP refinement + further hill‑climb,
    # repeated while time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # LP refinement (few inner iterations)
        best_a, best_score = lp_refine(
            best_a,
            best_score,
            n,
            fft_len_conv,
            fft_len_g,
            deadline,
            max_iter=5,
        )
        # short polishing hill‑climb after each LP pass
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[iteration] current best score = {best_score:.6f}")

        # occasional random perturbation to escape tiny plateaus
        if rng.random() < 0.07:
            sigma = rng.uniform(0.001, 0.006)
            noisy = best_a + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            noisy_score = fast_score(noisy, fft_len_conv, n)
            if noisy_score < best_score:
                best_a, best_score = noisy, noisy_score
                print(f"[noise] improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
