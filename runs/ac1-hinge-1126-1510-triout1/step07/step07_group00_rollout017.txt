"""
Improved step‑function search.

The algorithm is a more aggressive version of the reference implementation:

* A richer parametric sweep (power‑law, exponential and Dirichlet families).
* Longer L‑BFGS continuation with a finer λ‑schedule.
* Two gradient‑based refinements:
    – Projected sub‑gradient descent with momentum.
    – Adam optimisation on the smooth log‑sum‑exp surrogate.
* An annealed hill‑climb that moves a variable fraction of mass from
  the most harmful coordinates to the most helpful ones, occasionally
  accepting uphill moves.
* All phases respect a global 1000 s budget and keep the best sequence
  discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (return length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  Used for quick comparisons.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting from a_start.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-10, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            # failures are harmless – just skip this run
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ~ 1/√it
        step = 0.75 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth surrogate.
    Operates on logits y = log(a) (with a tiny offset to stay in interior).
    """
    y = np.log(a0 + 1e-12)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    while time.time() < deadline:
        t += 1
        f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.2)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        y = y - step * m_hat

        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def annealed_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb.
    Moves a random fraction of mass from a “bad” coordinate to a “good”
    coordinate; occasionally accepts uphill moves with probability exp(-Δ/T).
    Temperature decays linearly from T_start to T_end.
    """
    best = a.copy()
    best_score = current_score

    rng = np.random.default_rng()
    T_start = 0.5
    T_end = 1e-4
    it = 0

    while time.time() < deadline and it < max_iter:
        # linear temperature schedule
        T = T_start + (T_end - T_start) * (it / max_iter)

        # compute sub‑gradient mask (coordinates that take part in max conv)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # fraction of mass to move – random up to 10 %
        frac = rng.random() * 0.10
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        Δ = new_score - best_score
        if Δ < 0 or rng.random() < math.exp(-Δ / max(T, 1e-12)):
            # accept
            if new_score < best_score:
                best_score = new_score
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian perturbation and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "dirichlet", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "dirichlet":
        # draw concentration α from a wide range
        alpha = rng.uniform(0.05, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initial best (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # if a historic sequence is available, try it first
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            s1 = fast_score(a1, fft_len_conv, n)
            if s1 < best_score:
                best_a, best_score = a1, s1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – expanded parametric sweep
    # --------------------------------------------------------------
    # power‑law sweep (including reversed) + exponential + Dirichlet
    for p in np.linspace(0.0, 4.0, 17):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        a = ((np.arange(n, 0, -1, dtype=float) ** p))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    # exponential sweep
    for rate in np.linspace(0.001, 0.03, 12):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] exp rate {rate:.4f} -> {best_score:.6f}")

    # --------------------------------------------------------------
    # time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety gap

    # allocate portions of the time budget
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.95 * budget_s
    # remaining time for final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with finer λ‑schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # jitter around current best 30 % of the time
        if rng.random() < 0.3:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.020), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam on smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[adam] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – annealed deterministic hill‑climb
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = annealed_hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=25000
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final official evaluation (the environment will also call the same)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
