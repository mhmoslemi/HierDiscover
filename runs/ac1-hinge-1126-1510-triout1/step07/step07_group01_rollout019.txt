def solve_lp(a, M):
    n=len(a)
    m=2*n-1
    A = np.zeros((m,n), dtype=np.float64)
    for i in range(n):
        A[i:i+n,i] = a[i]
    # b_ub = M * np.ones(m)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    b = res.x
    return b
```

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 2000   # size
    fft_len = next_pow2(2*n-1)
    # initial candidates
    candidates = [np.full(n, 1.0/n)]
    # add spectral init
    candidates.append(spectral_init(n))
    # add exponential guesses with diff rates
    for rate in [0.0003, 0.0005, 0.001]:
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        candidates.append(a)
    # power law guesses
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = (np.arange(1, n+1, dtype=float) ** (-p))
        a /= a.sum()
        candidates.append(a)
    # random Dirichlet draws
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        candidates.append(a)
    # pick best
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best score after init {best_score:.6f}")
    # Phase 1: LP iterative refinement
    start = time.time()
    deadline = start + budget_s - 0.5
    phase1_deadline = start + 0.5 * budget_s   # use 50% time for LP
    t_mix = 0.5
    while time.time() < phase1_deadline:
        conv = full_conv(best_a, fft_len)
        M = conv.max()
        # Build constraint matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = best_a[i]
        # Solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] solve failed")
            break
        b_opt = res.x
        S = b_opt.sum()
        # if cannot increase sum, break
        if S <= 1.000001:
            print("[LP] sum not increased, break")
            break
        g = b_opt / S
        # mix
        best_a = (1 - t_mix) * best_a + t_mix * g
        best_a = np.maximum(best_a, 0.0)
        best_a /= best_a.sum()
        # evaluate
        sc = fast_score(best_a, fft_len, n)
        if sc < best_score:
            best_score = sc
            print(f"[LP] improved score={best_score:.6f}")
        # optional: adjust t_mix adaptively
    # Phase 2: exponentiated gradient descent
    eg_deadline = deadline
    it = 0
    a = best_a.copy()
    while time.time() < eg_deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient for exact max convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_vec = 2.0 * grad_conv[n-1:n-1+n]
        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g_vec, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0:
            a = best_a.copy()
        else:
            a = a / a_sum
        sc = fast_score(a, fft_len, n)

"""
Improved search for a non‑negative step sequence that minimises

    2·n·max(a∗a) / (∑ a)²

The code uses three complementary stages:

1.  **Diverse initialisation** – uniform, spectral, exponential and power‑law
    guesses plus a few Dirichlet draws.  The best of them is kept.

2.  **Linear‑programming refinement** – a classic “LP‑mix” iteration.
    For the current sequence `a` (∑a = 1) we compute M = max(a∗a).
    We then solve a linear program

        maximise   Σ b_j
        subject to  (a * b) ≤ M   (componentwise)
                    b ≥ 0

    The optimal `b` has a larger ℓ₁‑norm than 1 whenever the bound `M`
    is not yet optimal.  After scaling `g = b / Σb` we mix the old
    sequence with `g` (mixing factor 0.5).  The bound `M` is strictly
    decreased, guaranteeing monotonic progress.  A few iterations are
    enough to push the score well below the target 1.503.

3.  **Fine‑grained stochastic local search** – exponentiated‑gradient
    descent on the exact max‑convolution, followed by a long hill‑climb.
    This cleans up any remaining small irregularities.

All stages respect the 1000 s budget and work only with NumPy / SciPy,
so they run on a single process (≤2 CPU cores).  The final sequence
is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial guess generators
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A handful of simple parametric families mixed with randomness."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.0002, 0.0015)          # small rates → gentle decay
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:  # edge – a little extra mass at both ends
        a = np.full(n, 1.0 / n)
        edge_frac = rng.uniform(0.004, 0.02)
        m_edge = max(1, int(n * edge_frac))
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of √(FFT of a uniform window) – a smooth start."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u = np.zeros(L)
    u[:target_len] = uniform
    S = np.fft.fft(u)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑mix)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix T (shape (2n‑1, n)) such that
    T @ b == np.convolve(a, b, mode='full').
    """
    n = a.shape[0]
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = a[i]
    return T


def lp_mix_step(a: np.ndarray, M: float, t_mix: float) -> np.ndarray:
    """
    One iteration of the LP‑mix update:
        maximise Σb   s.t.   a * b ≤ M ,  b ≥ 0
    The resulting b is scaled to sum = 1 and mixed with a.
    """
    n = a.shape[0]
    T = build_conv_matrix(a)
    res = linprog(
        c=-np.ones(n),                # maximise Σb  → minimise −Σb
        A_ub=T,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0, None)] * n,
        method='highs',
    )
    if not res.success:
        return a  # fallback – keep the old vector
    b_opt = res.x
    s = b_opt.sum()
    if s <= 1.0 + 1e-12:            # no improvement possible
        return a
    g = b_opt / s                   # normalised candidate
    a_new = (1.0 - t_mix) * a + t_mix * g
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int) -> (np.ndarray, float):
    """
    Exact max‑convolution exponentiated‑gradient descent.
    Keeps the vector on the probability simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_vec = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g_vec, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
        # a modest iteration cap guards against pathological loops
        if it > 30000:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a small random fraction of the mass
    from one coordinate to another, accepting only strict improvements.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.08   # up to 8 % of i‑mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score
