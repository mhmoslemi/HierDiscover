"""
Search for a non‑negative step sequence that minimises
    2·n·max(a∗a) / (∑a)²

Algorithm
---------
1. Initialise a population of candidate vectors (uniform, spectral √‑init,
   random Dirichlet‑like).  Keep the best (sum normalised to 1).
2. Long exponentiated‑gradient descent on the *exact* max‑convolution.
   This stage rapidly drives the score from ≈2 down to ≈1.51.
3. Convex “LP‑refinement”:
   – For the current f compute M = max(f∗f).
   – Solve   maximise 1ᵀg   subject to   g≥0  and   (f∗g) ≤ M .
     The linear program is a Toeplitz‐matrix inequality.
   – Normalise g to sum = 1 and replace  f ← (1‑t)·f + t·g  (t≈0.5,
     gradually reduced).  This step is known to strictly decrease M.
4. Final greedy hill‑climb for a few thousand cheap moves.
5. Return the best vector found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full, length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a∗a) for a already normalised (∑a=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Initialise from the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (used as the fast “global” stage)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb (cheap final polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another. Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Linear‑programming refinement

def lp_refine_step(a: np.ndarray, max_conv: float, t: float) -> np.ndarray | None:
    """
    One Newton‑style refinement step:
        maximise 1ᵀg   subject to   g≥0, (a∗g) ≤ max_conv .
    Returns the mixed vector (1‑t)·a + t·g normalised to sum 1,
    or None if the LP fails.
    """
    n = a.shape[0]

    # Build the Toeplitz matrix T such that (T @ g) = a ∗ g (full).
    # First column: a followed by zeros, length 2n‑1.
    # First row: a[0] then zeros, length n.
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    T = toeplitz(col, row)                     # shape (2n‑1, n)

    b_ub = np.full(2 * n - 1, max_conv * (1.0 + 1e-12))
    c = -np.ones(n)                             # maximise 1ᵀg  → minimise –1ᵀg
    bounds = [(0.0, None)] * n

    # Solve the LP (highs is fast and robust for dense problems)
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return None

    # Normalise the direction and mix with the current a
    g = g0 / sum_g0
    a_new = (1.0 - t) * a + t * g
    a_new = a_new / a_new.sum()
    return a_new

    phase2_deadline = start_time + 0.80 * budget_s
    # 80 – 100 % : greedy hill‑climb
    phase3_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (global search)
    # ------------------------------------------------------------------
    if time.time() < phase1_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase1_deadline,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        t = 0.5                                 # initial mixing weight
        it_lp = 0
        while time.time() < phase2_deadline - 0.5:
            conv = full_conv(best_a, fft_len_conv)
            M = conv.max()
            a_new = lp_refine_step(best_a, M, t)
            if a_new is None:
                break
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new
                print(f"[LP #{it_lp}] new best score={best_score:.6f} t={t:.3f}")
                # keep the same step size for the next iteration
            else:
                # if no improvement, reduce the mixing factor
                t *= 0.7
                if t < 0.01:
                    break
            it_lp += 1

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=12000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
