def exp_grad_optimize(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    it=0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient via FFT
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_rev = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_rev[(n-1):(n-1 + n)]
        eta = 0.5 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()
        # record best
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best = a.copy()
        # occasional random noise to escape
        if it % 5000 == 0:
            a = add_noise(a, sigma=0.001, rng=rng)
        it += 1
    return best, best_score

    F(a) = 2·n·max(a*a) / (∑a)² .
It consists of three phases:

1. **Exploratory sub‑gradient descent** – many short, cheap runs of
   exponentiated‑gradient descent on the hard max‑convolution.
   Random restarts and occasional Gaussian perturbations help to
   escape shallow basins.

2. **Focused long run** – a single long exponentiated‑gradient descent
   starting from the best candidate found in phase 1.

3. **Fine‑grained greedy hill‑climb** – repeatedly transfer a small
   random fraction of the mass between two coordinates, keeping only
   improving moves.

All phases respect a global 1000 s deadline.  The best sequence found
is returned as a plain ``list[float]``; a final call to
``evaluate_sequence`` prints the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the official score when sum(a)=1.
    (The denominator is therefore 1 and can be ignored.)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = v.astype(np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse probability vector on the simplex.
    """
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'bimodal', 'edge', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent using the true max‑convolution.
    Returns the best vector seen before *deadline*.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # sub‑gradient of max

        # Gradient of the (hard) max‑convolution w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step‑size – 1/√t schedule, with a small prefactor
        eta = 0.6 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped exponent to stay safe)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()        # stays on the simplex

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # occasional jitter to shake out of tiny plateaus
        if (it + 1) % 4000 == 0:
            a = add_noise(a, sigma=0.001, rng=rng)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Small random mass transfers, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue

        # move up to 8 % of the source mass
        frac = rng.random() * 0.08
        delta = best[i] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8000                                 # larger than the baseline 4000
    fft_len = next_pow2(2 * n - 1)            # length for exact linear convolution

    # --------------------------------------------------------------
    # Initialise from a possible previous construction
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to new length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.55 * budget_s   # many short runs
    phase2_end = start_time + 0.90 * budget_s   # long focused run
    # remaining time will be spent on hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – many short exploratory runs
    # --------------------------------------------------------------
    while time.time() < phase1_end:
        # decide whether to perturb the current best or start from scratch
        if best_score < np.inf and rng.random() < 0.30:
            a_start = add_noise(best_a, sigma=rng.uniform(0.0005, 0.004), rng=rng)
        else:
            a_start = random_initial(n, rng)

        # give this run at most 8 seconds (or the remaining slice of phase 1)
        sub_deadline = min(phase1_end, time.time() + 8.0)

        a_opt, sc_opt = exp_gradient_descent(
            a_start,
            sub_deadline,
            n,
            fft_len,
            rng,
        )

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[phase‑1] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – a long, careful optimisation from the current best
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_descent(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
        )
        print(f"[phase‑2] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – intensive greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=50000,
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # evaluate_sequence is provided externally
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
