"""
Propose a low‑score step‑function.

The optimizer is a three‑phase pipeline:

1. **Heavy L‑BFGS on a smooth log‑sum‑exp surrogate of the
   max‑convolution**.
   – many random restarts,
   – a fine geometric schedule of the smoothing factor (lam) down to 5e‑6,
   – up to 2000 L‑BFGS iterations per λ.

2. **Exact sub‑gradient descent with an Adam‑style adaptive step**.
   The sub‑gradient is obtained from the active set of the maximal
   convolution entries.  The update is multiplicative (mirror‑descent
   on the simplex) and the step‑size decays like 1/√t.

3. **Simulated‑annealing hill‑climb**.
   A random fraction of mass is moved from one coordinate to another.
   Improving moves are always accepted; uphill moves are accepted
   with a temperature that linearly cools to zero.

After the three phases the best vector found is optionally up‑scaled
to a longer length (n=5000) by linear interpolation, refined by a
short Adam run, and the best of the two lengths is returned.

All stages respect the global time budget (default 1000 s) and print
progress information.  The returned list of floats is a non‑negative
step sequence that (empirically) improves the previous best bound.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a is on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Random initialisation utilities

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power',
         'edge', 'bimodal', 'sparse', 'sine', 'bell']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'sine':
        # a_i ∝ sin(π·(i+0.5)/n)  raised to a random power
        p = rng.uniform(0.5, 2.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:  # bell
        sigma = rng.uniform(0.12, 0.25)
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - centre) / (sigma * n)) ** 2)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient components that correspond to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its
    fast_score.
    """
    # work in log‑space to enforce positivity
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_adam(a_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      rng: np.random.Generator):
    """
    Mirror‑descent (exponentiated‑gradient) with Adam‑style adaptive step.
    Returns the best a seen before the deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # mask of active entries (within machine epsilon of the max)
        eps_rel = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_rel).astype(np.float64)

        # gradient of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # Adam moments
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # step size schedule – 1/√t scaled by a constant
        eta = 0.9 / np.sqrt(t)

        step = eta * (m_hat / (np.sqrt(v_hat) + eps))

        # multiplicative update on the simplex (mirror‑descent)
        exp_term = np.exp(np.clip(-step, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional progress report
        if t % 2000 == 0:
            print(f"[exp‑grad‑adam] it={t}, fast_score={best_score:.6f}")

    return best_a, best_score


def hill_climb_anneal(a: np.ndarray,
                      current_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      max_iter: int = 40000):
    """
    Simulated‑annealing hill‑climb on the simplex.
    Moves a random fraction of mass from i to j; uphill moves are
    accepted with a linearly cooling temperature.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # start temperature: a fraction of the current excess over the theoretical minimum (≈1)
    T0 = max(0.05, (best_score - 1.0) * 0.1)

    while time.time() < deadline and it < max_iter:
        frac = it / max_iter
        T = T0 * (1.0 - frac)          # linear cooling

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue

        # take up to 20 % of source mass
        delta = best[i] * rng.random() * 0.20
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # safeguard against negativity due to rounding
        if cand[i] < 0.0:
            it += 1
            continue

        conv = full_conv(cand, fft_len)
        new_score = 2.0 * n * conv.max()

        delta_score = new_score - best_score
        if delta_score < 0.0:
            best = cand
            best_score = new_score
        else:
            if T > 0.0 and rng.random() < np.exp(-delta_score / T):
                best = cand
                best_score = new_score

        it += 1
        if it % 10000 == 0:
            print(f"[hill‑climb] it={it}, fast_score={best_score:.6f}")

    return best, best_score

            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_adam(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
        )
        print(f"[exp‑grad‑adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb_anneal(
        best_a,
        best_score,
        fft_len,
        n,
        final_deadline,
        max_iter=80000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional second try with a longer sequence (n=5000)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 10.0:  # need a few seconds to be worthwhile
        n2 = 5000
        fft_len2 = next_pow2(2 * n2 - 1)
        # interpolate current best to the bigger grid
        a2 = np.interp(np.linspace(0, 1, n2),
                       np.linspace(0, 1, n),
                       best_a)
        a2 = np.maximum(a2, 0.0)
        a2 /= a2.sum()
        # short Adam run (half of the remaining time)
        sub_deadline = time.time() + time_left / 2.0
        a2, score2 = exp_gradient_adam(a2, sub_deadline, n2, fft_len2, rng)
        print(f"[larger n] n={n2}, fast_score={score2:.6f}")
        if score2 < best_score:
            best_a, best_score = a2, score2

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
