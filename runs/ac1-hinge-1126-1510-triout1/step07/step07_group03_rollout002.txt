"""
Propose a low‑score step‑function.

The search works on a relatively large array (n≈8 k) which gives a
theoretical lower bound of 1 for the evaluation function.
The algorithm consists of four stages:

1. **Parametric random search** – randomised beta / exponential /
   edge‑heavy shapes on the simplex.  The best candidate is kept.
2. **L‑BFGS continuation** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) while slowly decreasing the smoothing
   parameter λ.
3. **Projected sub‑gradient descent with momentum** – directly minimise
   the true non‑smooth objective using the sub‑gradient of the
   max‑convolution.
4. **Targeted gradient‑guided hill‑climb** – exploit the sub‑gradient
   to move mass from “high‑gradient’’ indices to “low‑gradient’’ ones
   with a strict improvement rule.

All steps respect the 1000 s budget and respect the non‑negativity /
simplex constraints required by ``evaluate_sequence``.
"""

from __future__ import annotations

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (output length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1).  Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).  Returns (objective, gradient wrt y).
    """
    # --- softmax (stable) ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth_max w.r.t. conv entries ---
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # --- back‑propagation through convolution ---
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # portion that corresponds to the gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through softmax ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:      # pragma: no cover – safety net
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] improved score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.95                   # strong momentum
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (size n)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size ~ 1/sqrt(it)
        step = 1.0 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb.
    Moves a modest fraction of mass from indices with a large sub‑gradient
    to indices with a small sub‑gradient, accepting only strictly
    improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # pick source from high‑gradient tail, target from low‑gradient tail
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # transfer up to 7 % of the source mass
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # else reject

        it += 1

    return best, best_score

# Parametric random generators
# ----------------------------------------------------------------------


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on the simplex (α,β > 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def exponential_distribution(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def edge_heavy_distribution(n: int, edge_frac: float) -> np.ndarray:
    """
    Put a higher constant value on a small fraction of the leading
    and trailing indices, rest is uniform.
    """
    a = np.full(n, 1.0 / n)
    m_edge = max(1, int(n * edge_frac))
    # heuristic heavy value that keeps total mass = 1
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a = a / a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "beta", "exp", "edge", "bimodal", "mixture"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "beta":
        alpha = rng.uniform(0.1, 2.0)
        a = beta_distribution(n, alpha, alpha)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = exponential_distribution(n, rate)
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        a = edge_heavy_distribution(n, edge_frac)
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
        a = a / a.sum()
    else:  # mixture – blend a beta shape with a uniform seed
        alpha = rng.uniform(0.1, 2.0)
        base = beta_distribution(n, alpha, alpha)
        a = (1.0 - 0.3) * np.full(n, 1.0 / n) + 0.3 * base
        a = a / a.sum()

    # tiny Gaussian jitter for diversity
    sigma = rng.uniform(1e-5, 5e-4)
    a = add_noise(a, sigma, rng)
    a = a / a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Using a larger n gives a better theoretical lower bound (1.0)
    n = 8192                     # power of two → fast FFTs
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from any previously found construction (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2        # safety margin

    # 25 % parametric random search, 30 % L‑BFGS, 25 % sub‑gradient,
    # remaining time for targeted hill‑climb.
    t_param_end = start_time + 0.25 * budget_s
    t_lbfgs_end = t_param_end + 0.30 * budget_s
    t_subgrad_end = t_lbfgs_end + 0.25 * budget_s
    t_hill_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – parametric random search
    # ------------------------------------------------------------------
    while time.time() < t_param_end:
        # pick a distribution type at random
        kind = rng.choice(["beta", "exp", "edge", "mixture"])
        if kind == "beta":
            alpha = rng.uniform(0.1, 2.0)
            a = beta_distribution(n, alpha, alpha)
        elif kind == "exp":
            rate = rng.uniform(0.001, 0.020)
            a = exponential_distribution(n, rate)
        elif kind == "edge":
            edge_frac = rng.uniform(0.004, 0.04)
            a = edge_heavy_distribution(n, edge_frac)
        else:  # mixture – blend a random shape with a uniform seed
            # start from uniform and add a modest bump
            base = random_initial(n, rng)
            mix = rng.uniform(0.2, 0.6)
            a = (1 - mix) * np.full(n, 1.0 / n) + mix * base
            a = a / a.sum()

        # very small Gaussian jitter for extra diversity
        sigma = rng.uniform(5e-5, 2e-3)
        a = add_noise(a, sigma, rng)
        a = a / a.sum()

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[param] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS continuation with decreasing smoothing
    # ------------------------------------------------------------------
    if time.time() < t_lbfgs_end:
        lam_schedule = [
            0.6, 0.4, 0.25, 0.15, 0.09,
            0.05, 0.03, 0.018, 0.01,
            0.006, 0.003, 0.0015, 0.0008,
            0.0004, 0.0002, 0.0001, 5e-05,
        ]
        maxiter_per_lam = 400
        best_a, opt_score = lbfgs_continuation(
            best_a,
            lam_schedule,
            maxiter_per_lam,
            t_lbfgs_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if opt_score < best_score:
            best_score = opt_score
            print(f"[lbfgs] after continuation score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < t_subgrad_end:
        best_a, sub_score = subgradient_optimize_momentum(
            best_a,
            t_subgrad_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sub_score < best_score:
            best_score = sub_score
            print(f"[sub‑grad] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < t_hill_end:
        best_a, final_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            t_hill_end,
            max_iter=120000,
        )
        if final_score < best_score:
            best_score = final_score
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluation error:", exc)
        final_eval = float("inf")

    return final_seq
