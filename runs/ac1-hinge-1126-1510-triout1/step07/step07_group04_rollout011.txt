def mse_objective_and_grad(y, n, fft_len_conv, fft_len_g, t_vec):

    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)

    conv = np.real(np.fft.ifft(fa * fa))[:2*n - 1]

    diff = conv - t_vec

    f = np.sum(diff**2)  # scalar

    # gradient w.r.t a: g = 4 * conv_rev(diff, a_rev). Please compute using FFT.

    fa_rev = np.fft.fft(a[::-1], fft_len_g)

    fd = np.fft.fft(diff, fft_len_g)

    grad_full = np.real(np.fft.ifft(fa_rev * fd))

    grad_a = 4.0 * grad_full[(n - 1) : (n - 1 + n)]

    # chain rule for softmax:

    mean_g = np.dot(a, grad_a)

    grad_y = a * (grad_a - mean_g)

    return f, grad_y

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    import numpy as np, time, scipy.optimize as opt

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Problem size
    default_n = 15000
    init_seq = globals().get('height_sequence_1', None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):

        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = max(default_n, int(tmp.size))
    n = default_n

    fft_len_conv = next_pow2(2*n - 1)
    fft_len_g = next_pow2(3*n - 2)

    # uniform target for conv
    t_vec = np.full(2*n - 1, 1.0 / (2*n - 1), dtype=float)

    # init a and y
    y = generate_initial_y('arcsine', n, rng)
    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, score={best_score:.6f}")

    # Phase 1: MSE L-BFGS
    phase1_deadline = start + 0.4 * budget_s  # 40% budget
    while time.time() < phase1_deadline:
        # maybe just run once

        y0 = y + rng.normal(scale=0.05, size=n)
        # minimize
        res = opt.minimize(fun=mse_objective_and_grad,
                           x0=y0,

                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1500, 'gtol': 1e-12, 'ftol': 1e-12, 'disp': False})
        y_opt = res.x

"""
Hybrid search for a low‑evaluation step‑function.

The algorithm works on the probability simplex (∑ a = 1) because the
objective 2·n·max(conv) is scale‑invariant.  The pipeline is:

1.  Initialise a with the discrete arcsine law (a_i ∝ 1/√(x·(1‑x))),
    which makes the self‑convolution almost uniform.

2.  Refine the distribution by minimising the L² distance between the
    convolution a∗a and the perfectly uniform target using L‑BFGS
    (the variables are the pre‑softmax logits y, the soft‑max guarantees
    positivity and the simplex constraint).

3.  Switch to a projected sub‑gradient method on the true
    max‑convolution objective to push the remaining peaks down.

4.  Finish with an aggressive greedy hill‑climb that moves a random
    fraction of mass between two indices whenever the fast score
    improves.

All heavy work (FFT‑based convolutions) is performed with NumPy’s fast
FFT, making the search feasible for n≈15 000 within the 1000 s budget.
The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective based on L2 distance to a uniform convolution
# ----------------------------------------------------------------------


def mse_objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    uniform_conv: np.ndarray,
) -> tuple[float, np.ndarray]:
    """
    Objective: sum_s ( (a*a)_s - uniform )^2  (L2 distance to a uniform
    self‑convolution).  Returns (value, gradient wrt logits y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    diff = conv - uniform_conv
    f = float(np.sum(diff ** 2))

    # ----- gradient wrt a -----
    # derivative of the L2 term: 2 * diff_s * d(conv_s)/da_i,
    # and d(conv_s)/da_i = 2 * a_{s-i} (if index in range)
    # => total factor 4.
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fd = np.fft.fft(diff, fft_len_g)
    grad_full = np.real(np.fft.ifft(fa_rev * fd))

    # central n entries correspond to ∂f/∂a_i (before the factor 4)
    grad_a = 4.0 * grad_full[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)

    return f, grad_y


def optimize_mse(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    uniform_conv: np.ndarray,
    maxiter: int = 1500,
) -> np.ndarray:
    """
    L‑BFGS optimisation of the L2‑uniformity objective.
    Returns the optimal logits y.
    """
    res = minimize(
        fun=mse_objective_and_grad,
        x0=y_start,
        args=(n, fft_len_conv, fft_len_g, uniform_conv),
        method="L-BFGS-B",
        jac=True,
        options={
            "maxiter": maxiter,
            "gtol": 1e-12,
            "ftol": 1e-12,
            "disp": False,
        },
    )
    return res.x


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    The most important mode is ``arcsine`` which approximates the
    continuous arcsine law whose self‑convolution is exactly uniform.
    """
    if mode == "arcsine":
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
        return y0

    # fallback: small random perturbation of arcsine
    arcsine_y = generate_initial_y("arcsine", n, rng)
    if mode == "perturb":
        y0 = arcsine_y + rng.normal(scale=0.05, size=n)
        return y0

    # other cheap patterns (may be useful for occasional restarts)
    if mode == "random":
        y0 = rng.standard_normal(n)
        return y0
    if mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    # default to arcsine
    return arcsine_y


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))  # index of maximal entry

        # Sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score
