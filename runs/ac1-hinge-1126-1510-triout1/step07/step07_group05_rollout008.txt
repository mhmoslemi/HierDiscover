"""
search for a non‑negative step function that minimizes
    2·n·max(a*a)/ (sum(a)^2)

The objective is homogeneous, therefore we keep the vector on the
probability simplex (sum = 1) and work directly with the fast score

    fast_score(a) = 2·n·max(conv(a,a))

The algorithm consists of three phases:

1. **Broad shape exploration** – generate a few deterministic families
   (power‑law, exponential, arcsine, product‑power, etc.) and keep the
   best one (few hundred fast evaluations).

2. **Continuation L‑BFGS** – minimise a smooth Log‑Sum‑Exp approximation
   of the max‑convolution.  The smoothing factor is decreased
   geometrically (≈70 stages).  The optimiser works on the soft‑max
   parametrisation `a = softmax(y)` which automatically enforces the
   simplex constraints.

3. **Projected sub‑gradient descent + aggressive hill‑climb** – use the
   true nonsmooth max‑convolution gradient, a diminishing step‑size and
   a final random mass‑transfer hill climbing until the time budget
   expires.

All heavy work is done with FFTs (O(n log n)).  The routine respects the
1000 s budget and prints intermediate scores.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max; input `logits` may be any real vector."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    `a` is assumed length `n`.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The weight `w` is uniform over all (almost) active convolution entries.
    A simple back‑tracking line search ensures monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # d(a_i a_j)/d a_i = 2 a_j

        # step size decreasing like 1/sqrt(it)
        eta = eta0 / math.sqrt(it)
        improvement = False
        for _ in range(8):                     # back‑track up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even the reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing with random mass transfers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20_000,
    max_move: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill climbing.
    Repeatedly move a random fraction of the source mass from i to j and
    keep the move only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Deterministic shape generators (return a normalised probability vector)
# ----------------------------------------------------------------------


def shape_power_sum(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a /= a.sum()
    return a


def shape_power_product(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p}  (symmetrised, zero at the ends)."""
    i = np.arange(n, dtype=float)
    prod = (i + 1) * (n - i)
    a = prod ** (-p)
    a /= a.sum()
    return a


def shape_exponential(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) + exp(-rate * (n-1-i))."""
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine: a_i ∝ 1/√(i+1) + 1/√(n-i)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    a /= a.sum()
    return a


def shape_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluate_sequence score.
    Returns a plain ``list`` of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[setup] n={n}")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic shape sweep
    # ------------------------------------------------------------------
    best_a = shape_uniform(n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform score = {best_score:.6f}")

    # list of shape generators to try
    # power‑law exponents
    for p in np.linspace(0.15, 2.0, 20):
        a = shape_power_sum(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a
    # product‑power exponents
    for p in np.linspace(0.15, 2.0, 20):
        a = shape_power_product(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a
    # exponential rates
    for rate in np.linspace(0.001, 0.05, 20):
        a = shape_exponential(n, rate)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a
    # arcsine (no parameter)
    a = shape_arcsine(n)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a

    print(f"[sweep] best fast score after shape sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (smooth max)
    # ------------------------------------------------------------------
    # Transform to the soft‑max parametrisation
    best_y = np.log(best_a + 1e-12)

    # A few random perturbations of the best start (to escape possible bad basins)
    n_perturb = 4
    for _ in range(n_perturb):
        y0 = best_y + rng.normal(scale=0.03, size=n)
        lam_sched = list(np.geomspace(0.5, 1e-10, num=70))
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_sched,
            maxiter_per_stage=900,
        )
        a_opt = softmax(y_opt)
        sc = fast_score(a_opt, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a, best_y = sc, a_opt, y_opt
            print(f"[lbfgs‑perturb] improved fast score = {best_score:.6f}")

    # One final, more precise L‑BFGS run from the current best
    lam_sched_fine = list(np.geomspace(0.3, 1e-12, num=80))
    y_opt = schedule_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_sched_fine,
        maxiter_per_stage=1000,
    )
    best_a = softmax(y_opt)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = y_opt
    print(f"[lbfgs‑final] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max objective
    # ------------------------------------------------------------------
    sub_deadline = deadline - 90.0          # reserve ~90 s for final HC
    if time.time() < sub_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.28,
            max_iters=1_000_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climb (use remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill‑climb] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=50_000,
        )
        print(f"[final hill‑climb] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to the format expected by the judge and evaluate.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
