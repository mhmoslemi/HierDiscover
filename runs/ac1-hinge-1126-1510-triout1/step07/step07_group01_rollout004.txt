"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.
The search combines:
    * random parametric families (beta‑like, exponential, sinusoid, edge‑heavy)
    * short local refinements with exponentiated‑gradient descent
    * a long exponentiated‑gradient run from the current best
    * a final greedy hill‑climb.
All phases respect the 1000 s budget and keep the vector on the probability
simplex.  The algorithm is deliberately different from the previous
implementation and focuses on fast parametric exploration before fine‑tuning.
"""

import time
import numpy as np
import heapq

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric shape generators
# ----------------------------------------------------------------------


def generate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex using a few simple
    parametric families (beta‑like, edge‑heavy, central‑peak, exponential,
    sinusoidal).  The result is always non‑negative and sums to 1.
    """
    i = np.arange(n, dtype=float)
    shape_type = rng.choice(
        ["uniform", "beta", "edge", "central", "exp", "sin"]
    )
    if shape_type == "uniform":
        a = np.full(n, 1.0 / n)

    elif shape_type == "beta":
        p = rng.uniform(0.0, 3.0)
        q = rng.uniform(0.0, 3.0)
        a = (i + 1.0) ** p * (n - i) ** q

    elif shape_type == "edge":
        p = rng.uniform(0.5, 4.0)
        a = (i + 1.0) ** p + (n - i) ** p

    elif shape_type == "central":
        p = rng.uniform(0.5, 4.0)
        a = ((i + 1.0) * (n - i)) ** p

    elif shape_type == "exp":
        r = rng.uniform(0.0, 0.015)
        a = np.exp(-r * i) + np.exp(-r * (n - 1 - i))

    else:  # sinusoidal
        b = rng.uniform(0.5, 3.0)
        a = np.sin(np.pi * (i + 0.5) / n) ** b

    # enforce non‑negativity, normalise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Sub‑gradient descent on the true max‑convolution using an
    exponentiated‑gradient update (keeps a on the simplex automatically).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # --- exact convolution a*a (linear, via FFT) ---
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # binary mask of max locations

        # gradient of the (hard) max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # exponentiated‑gradient step (step size decreasing)
        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 12000):
    """
    Random hill‑climb that moves a small random fraction of mass from one
    coordinate to another and accepts only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.12
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # ------------------------------------------------------------------
    # Return the result (evaluate for the official score)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
