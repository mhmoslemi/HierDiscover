"""
Propose a low‑evaluation step‑function.

The search works in three phases:

1.  Initialise a rich pool of analytic shapes (β(p,p), semicircle,
    uniform, random Gaussian, mixtures).  The pool is scored with an
    exact fast evaluator (2·n·max conv) and the K best candidates are
    kept.

2.  Continuation L‑BFGS on a smooth “log‑sum‑exp” surrogate of the
    max‑convolution.  The smoothing factor λ is reduced geometrically
    (≈30 stages) and each stage is given many L‑BFGS iterations.  The
    best result over the K candidates is retained.

3.  A fine‑tuning stage using an Adam optimiser on the same smoothed
    objective with an extremely small λ (≈ 10⁻¹²·max conv).  Finally a
    short greedy hill‑climb (random + targeted) polishes the solution.

All work is performed on the probability simplex (∑a_i = 1) using the
soft‑max parametrisation.  The routine respects the overall time budget
(≈ 1000 s) and returns a list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of the objective for a probability vector
    (∑ a_i = 1): 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic initialisers
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on (0,1); p>0.5 to avoid infinities at the ends."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle on {0,…,n‑1}."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑normalised random vector."""
    y = rng.standard_normal(n)
    a = np.exp(y)
    return a / a.sum()


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution objective 2·n·smooth_max and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # gradient of conv w.r.t. a (cross‑correlation)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Simple hill‑climbing helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: move a random fraction of the mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Greedy climb that preferentially moves mass away from the current
    maximum‑convolution contributors.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth log‑sum‑exp objective.
    Returns (y_opt, final_score).
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.3
    t = 0
    best_score = np.inf
    best_y = None

    while time.time() < deadline and t < max_iters:
        t += 1
        fval, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = step0 / (np.sqrt(t) + 1.0)   # simple decay
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 250 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_y = sc, y.copy()
    if best_y is None:
        best_y = y
        best_score = fast_score(softmax(y), fft_len_conv, n)
    return best_y, best_score

    lbgfs_deadline = start_time + 0.50 * budget_s

    best_a = None
    best_score = np.inf
    best_y = None

    print("[phase] L‑BFGS continuation")

    for init_sc, name, a0 in top_seeds:
        if time.time() > lbgfs_deadline:
            print("[phase] time exhausted")
            break
        print(f"[l‑bfgs] start {name:20s} (score={init_sc:.6f})")
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)
        print(f"[l‑bfgs] result {name:20s} -> {sc_opt:.6f}")

        if sc_opt < best_score:
            best_score, best_a, best_y = sc_opt, a_opt.copy(), y_opt.copy()
            print(f"[l‑bfgs] *new global best* {best_score:.6f}")

    if best_a is None:            # fallback (should not happen)
        best_a = uniform_init(n)
        best_y = np.log(best_a + 1e-12)
        best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Fine‑tuning with Adam (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.80 * budget_s
    print("[adam] start fine‑tuning")
    y_fine, fin_score = adam_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_factor=1e-12,
        max_iters=200_000,
        deadline=adam_deadline,
    )
    best_a = softmax(y_fine)
    best_score = fin_score
    best_y = y_fine
    print(f"[adam] finished with score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive random + targeted hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
            max_move=0.30,
        )
        print(f"[final] after random climb score = {best_score:.6f}")

    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
            max_move=0.20,
        )
        print(f"[final] after targeted climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore
        print(f"[final] evaluate_sequence = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return final_seq
