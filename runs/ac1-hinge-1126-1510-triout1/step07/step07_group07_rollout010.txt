"""
Propose a step‑function (non‑negative list) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex ∑ a_i = 1 (the objective is scale‑invariant).
* Build a large, diverse pool of analytic initialisations:
      – Beta(p,p) densities for many p∈[0.51,0.90] (U‑shaped to flatten the
        self‑convolution).
      – Uniform distribution.
      – Simple mixtures of a Beta shape with the uniform distribution.
* Keep the few (K≈5) most promising shapes according to a fast FFT‑based
  surrogate 2·n·max(conv(a,a)).
* For each of these candidates run a continuation L‑BFGS optimisation on the
  smoothed log‑sum‑exp surrogate of the max‑convolution.  The smoothing
  parameter λ is decreased geometrically from 5·max to 1e‑12·max.
* The best L‑BFGS result is then refined by a mirror‑descent (exponentiated
  gradient) sub‑gradient method on the exact objective.  The update is
  a_i ← a_i·exp(‑η·g_i) with g_i a sub‑gradient of the current maximal
  convolution entry; after each step the vector is re‑normalised onto the
  simplex.
* A short random‑restart / hill‑climbing phase is performed while time
  permits to escape local minima.
* The final sequence is returned as a plain list of floats; the provided
  evaluate_sequence() function is called once at the end to report the true
  score.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a of length 2*n-1 using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector
    (∑ a_i = 1):   2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp max) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (λ is a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # gradient of conv w.r.t. a (correlation a ↔ w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).

    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Analytic initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """β(p,p) density on (0,1); p≈0.5 gives the classic arcsine."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) sub‑gradient optimiser
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.8,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the true objective (max‑convolution).
    Update rule: a_i ← a_i·exp(‑η·g_i)  (g is a sub‑gradient of the current max).
    After each update we renormalise onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)

        g = np.zeros_like(a)
        # sub‑gradient of the max convolution entry
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        # exponentiated‑gradient step (log‑space update)
        loga = np.log(a + 1e-30) - eta * g
        a = np.exp(loga)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = uniform_init(n)
        else:
            a = a / a_sum

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Local random / targeted hill‑climbing (lightweight polish)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Implements the pipeline described in the module
    docstring and adds a finer exploration of Beta‑shapes together with
    exponentiated‑gradient refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – attempt to reuse a previously existing long sequence.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Build a rich pool of analytic starters.
    # ------------------------------------------------------------------
    candidates = []

    # Uniform baseline
    candidates.append(("uniform", uniform_init(n)))

    # Beta(p,p) for a fine grid of p-values (U‑shaped)
    for p in np.linspace(0.51, 0.90, 16):
        a = beta_init(n, p)
        candidates.append((f"beta_{p:.3f}", a))

    # Mixtures of the most U‑shaped beta (p≈0.51) with uniform
    beta_sharp = beta_init(n, 0.51)
    for w in [0.2, 0.4, 0.6, 0.8]:
        mixed = mixture_init(beta_sharp, uniform_init(n), w)
        candidates.append((f"betaU_mix_{w:.1f}", mixed))

    # If a historic sequence of the correct length exists, reuse it (normalised)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Quick scan – keep the most promising K candidates.
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:20s} fast score = {sc:.6f}")

    scored.sort()
    top_k = min(5, len(scored))

    best_a = None
    best_score = np.inf
    best_y = None

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (about 40 % of the total budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-12, num=30))
    maxiter_per_stage = 700
    l_bfgs_deadline = start_time + 0.40 * budget_s

    print("[phase] L‑BFGS continuation started")
    for rank in range(top_k):
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        print(f"[l‑bfgs] {name:20s} → score = {score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new global best = {best_score:.6f}")

        # a quick hill‑climb on the current best (still within L‑BFGS budget)
        if time.time() < l_bfgs_deadline:
            bc, bs = hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline=l_bfgs_deadline,
                max_iter=1500,
                max_move=0.25,
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] post‑hill‑climb improved to {best_score:.6f}")

        if time.time() >= l_bfgs_deadline:
            print("[phase] L‑BFGS time exhausted")
            break

    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Exponentiated‑gradient refinement (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    eg_deadline = start_time + 0.75 * budget_s
    print("[refine] exponentiated‑gradient descent starting")
    best_a, best_score = exp_gradient_descent(
        best_a, fft_len_conv, n, eg_deadline, eta0=0.8, max_iters=800_000
    )
    print(f"[refine] after EG best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random‑restart + short sub‑gradient polishing (remaining budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Small random perturbation of the current best
        perturb = rng.normal(scale=0.02, size=n)
        a0 = best_a + perturb
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:

        a0 = a0 / s
        # a brief exp‑gradient pass (≈10 s)
        sub_dead = time.time() + 8.0
        a_new, sc_new = exp_gradient_descent(
            a0, fft_len_conv, n, sub_dead, eta0=0.8, max_iters=150_000
        )
        if sc_new < best_score:
            best_a, best_score = a_new, sc_new
            print(f"[restart] improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
            max_move=0.30,
        )
        print(f"[final] hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return final sequence (list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] true evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return final_seq
