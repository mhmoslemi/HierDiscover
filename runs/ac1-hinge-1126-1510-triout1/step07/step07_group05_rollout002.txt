"""
Differential Evolution + Local refinement for minimizing the
evaluate_sequence objective.

The algorithm works in three phases:

1. **Population initialization** – generate a diverse set of
   probability vectors (sum=1, non‑negative) using several heuristics
   (uniform, random Dirichlet, exponential, power‑law, edge‑heavy,
   arcsine, etc.).  The previous best sequence (if any) is also injected.

2. **Differential Evolution** – evolve the population under the
   exact objective `fast_score` (maximal self‑convolution) using a
   classic DE/rand/1/bin scheme with projection onto the probability
   simplex after mutation/crossover.  This phase explores the search
   space aggressively while keeping the evaluation cheap (FFT‑based).

3. **Local refinement** – the best vector found by DE is polished by
   a continuation L‑BFGS on a Log‑Sum‑Exp smooth approximation of the
   max‑convolution, followed by a short stochastic hill‑climb.

The whole search respects the 1000 s budget; the final sequence is
returned as a plain list of non‑negative floats (already normalized,
so the external harness will compute the exact score directly).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max) + L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective and its gradient.

    f(y) = 2·n·smooth_max( a*a ),  a = softmax(y)
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv values
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indices corresponding to a_i coefficients
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return float(f), grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def gen_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def gen_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet (α=0.1)."""
    alpha = np.full(n, 0.1)
    a = rng.gamma(alpha, 1.0)
    a /= a.sum()
    return a


def gen_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """One‑sided exponential decay."""
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def gen_exponential_sym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    rate = rng.uniform(0.001, 0.02)
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a /= a.sum()
    return a


def gen_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law a_i ∝ i^{‑p}."""
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def gen_edge_heavy(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few heavy edge entries, light interior."""
    proportion = rng.uniform(0.01, 0.04)
    m_edge = max(1, int(n * proportion))
    a_edge = np.sqrt(n / 2.0) / n
    a_center = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, a_center)
    a[:m_edge] = a_edge
    a[-m_edge:] = a_edge
    a /= a.sum()
    return a


def gen_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like shape: 1/√i + 1/√(n‑i)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    a /= a.sum()
    return a


def gen_sidon_like(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy Sidon set – each sum appears at most once.
    We assign equal weight to the chosen positions.
    """
    chosen = []
    sums = set()
    for cand in range(n):
        conflict = False
        for c in chosen:
            if (cand + c) in sums or (2 * cand) in sums:
                conflict = True
                break
        if not conflict:
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
        if len(chosen) >= int(np.sqrt(n) * 1.5):
            break
    k = len(chosen)
    a = np.zeros(n)
    if k > 0:
        a[chosen] = 1.0 / k
    a /= a.sum()
    return a


def generate_initial_population(
    n: int,
    rng: np.random.Generator,
    popsize: int,
    init_seq: np.ndarray | None,
) -> tuple[list[np.ndarray], list[float]]:
    """Create an initial DE population and its fast scores."""
    modes = [
        gen_uniform,
        gen_random_dirichlet,
        gen_exponential,
        gen_exponential_sym,
        gen_powerlaw,
        gen_edge_heavy,
        gen_arcsine,
        gen_sidon_like,
    ]

    population = []
    scores = []

    for i in range(popsize):
        if i == 0 and init_seq is not None:
            a = init_seq.copy()
            a = a / a.sum()
        else:
            mode = rng.choice(modes)
            if mode in (gen_random_dirichlet, gen_sidon_like):
                a = mode(n, rng)
            elif mode in (gen_exponential, gen_exponential_sym, gen_powerlaw, gen_edge_heavy):
                a = mode(n, rng)
            else:
                a = mode(n)
        population.append(a)
    # placeholder scores – will be computed after FFT length known
    return population, scores

def de_optimize(
    pop: list[np.ndarray],
    pop_scores: list[float],
    n: int,
    fft_len: int,
    deadline: float,
    F: float = 0.5,
    CR: float = 0.9,
    max_no_improve: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Run a classic DE/rand/1/bin on the simplex‑projected population.
    Returns the best individual found and its fast_score.
    """
    rng = np.random.default_rng()
    popsize = len(pop)

    # initial scores
    for i in range(popsize):
        pop_scores[i] = fast_score(pop[i], fft_len, n)

    # keep track of the best
    best_idx = int(np.argmin(pop_scores))
    best_a = pop[best_idx].copy()
    best_score = pop_scores[best_idx]

    no_improve_counter = 0

    while time.time() < deadline and no_improve_counter < max_no_improve:
        improved_this_gen = False
        for i in range(popsize):
            # choose three distinct indices other than i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])

            # binomial crossover
            trial = pop[i].copy()
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = mutant[j]

            # project back onto the simplex
            trial = project_onto_simplex(trial)

            # evaluate
            trial_score = fast_score(trial, fft_len, n)

            # selection
            if trial_score < pop_scores[i]:
                pop[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
                    best_idx = i
                    improved_this_gen = True

        if improved_this_gen:
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # occasional diversity injection to avoid stagnation
        if no_improve_counter >= 50:
            # replace the worst 20 % of the population with fresh random members
            num_replace = max(1, popsize // 5)
            worst_idxs = np.argsort(pop_scores)[-num_replace:]
            for wi in worst_idxs:
                # random Dirichlet perturbation
                new_a = np.abs(rng.normal(size=n))
                new_a /= new_a.sum()
                pop[wi] = new_a
                pop_scores[wi] = fast_score(new_a, fft_len, n)
            no_improve_counter = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (mass transfer) – final polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # allow up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

    if time.time() < de_deadline:
        print("[DE] start")
        best_a, best_score = de_optimize(
            population,
            pop_scores,
            n,
            fft_len_conv,
            de_deadline,
            F=0.5,
            CR=0.9,
            max_no_improve=200,
        )
        print(f"[DE] finished → best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: Continuation L‑BFGS refinement (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print("[L‑BFGS] start")
        y_start = np.log(best_a + 1e-12)

        # More aggressive schedule – many stages, very small final λ
        lam_schedule = list(np.geomspace(0.5, 1e-12, num=80))
        y_opt = schedule_optimize(
            y_start,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] improved fast score = {best_score:.6f}")
        else:
            print(f"[L‑BFGS] no improvement (score = {score_opt:.6f})")
        # update y for possible later hill‑climb
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3: Final aggressive hill‑climb (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
