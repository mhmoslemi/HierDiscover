def smoothed_max_and_grad(y, n, fft_len_conv, fft_len_g, lam):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[:2 * n - 1]
    max_c = conv.max()
    # smoothing as before: compute shifted = conv - max_c; exp_shift = exp(shifted/lam); Z = sum(exp_shift); smooth_max = max_c + lam*log(Z)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    # gradient wrt conv
    w = exp_shift / Z   # derivative d(smooth_max)/d conv_i
    # compute gradient w.r.t. a: conv = a * a (linear convolution)
    # Use convolution of w with reversed a (like before)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    grad_a = conv_rev[(n-1) : (n-1 + n)]

    # chain rule through softmax:
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)

    #return f = smooth_max, grad_y (note final objective is 2*n * smooth_max)
    return smooth_max, grad_y

def simulated_annealing(a_start, start_score, n, fft_len, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    rng = np.random.default_rng()
    T0 = 0.1
    T_min = 1e-6
    total_steps_est = int(1e5)  # approximate max steps; will break early if out of time.
    start_time = time.time()
    for step in range(total_steps_est):
        if time.time() >= deadline:
            break
        # linear temperature schedule
        t = step / total_steps_est
        T = T0 * (1 - t) + T_min * t
        # choose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.1  # move up to 10% of source
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # acceptance probability
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                a = a_new
                cur_score = new_score
    return best_a, best_score

    # returns f and grad_y
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z

"""
Proposes a step‑function (non‑negative sequence) with a low evaluate_sequence
score.

The algorithm works in three phases:

1. **Diverse initialisation** – random, exponential, power‑law, beta‑type,
   arcsine, Dirichlet, Sidon‑set etc.  The vector is represented in logit
   space and normalised with a soft‑max (so the sum is always 1).

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of
   max(a*a) (log‑sum‑exp) using Adam.  The smoothing factor λ is reduced
   geometrically from 0.5·max(conv) down to ≈1e‑12·max(conv).  Each optimisation
   runs for a few thousand gradient steps, which is cheap because each step
   needs only two FFTs.

3. **Final stochastic hill‑climb (simulated annealing)** – moves a random
   fraction of mass from one index to another, accepting improving moves
   always and worsening moves with a temperature schedule.  The step size
   shrinks with the temperature.

Fast evaluation of the objective (2·n·max(conv)) is performed with an FFT‑based
convolution, allowing many cheap objective calls inside the optimisation loop.
The best sequence found is finally evaluated with the supplied
``evaluate_sequence`` function and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec
    grad_y = 2.0 * n * grad_y

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation Adam optimisation: run several stages with decreasing
    smoothing (lam_factor).  Each stage performs a bounded number of Adam
    steps.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # simple Adam with fixed learning rate; one pass of maxiter_per_stage steps
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.05
        for t in range(1, maxiter_per_stage + 1):
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # end of stage
    return y

# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting log‑it vector y for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        a0 = np.zeros(n)
        if chosen:
            a0[chosen] = 1.0 / len(chosen)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # Beta‑type distribution (U‑shaped when α,β<1)
        a = rng.uniform(0.05, 5.0)
        b = rng.uniform(0.05, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        a0 = (i ** (a - 1)) * ((n + 1 - i) ** (b - 1))
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a temperature schedule.
    Moves a random portion of mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    cur_score = start_score
    best_score = start_score
    rng = np.random.default_rng()

    # Annealing schedule parameters
    T_start = 0.12
    T_end = 5e-6
    # estimate a generous number of steps, but we will break when out of time
    approx_steps = 200_000

    for step in range(approx_steps):
        if time.time() >= deadline:
            break
        t_frac = step / approx_steps
        T = T_start * (1 - t_frac) + T_end * t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 8 % of the source mass
        delta = a[i] * rng.random() * 0.08
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert the move
                a[i] += delta
                a[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the smoothed‑max Adam stage
    # ------------------------------------------------------------------
    lam_schedule_main = list(np.geomspace(0.5, 1e-12, num=25))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "beta", "best"
    ]

    # ------------------------------------------------------------------
    # Phase 1 – many restarts with Adam optimisation
    # ------------------------------------------------------------------
    restart_deadline = deadline - 250.0   # keep ~4 minutes for final phases
    while time.time() < restart_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Adam optimisation with decreasing smoothing
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_main,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] mode={mode} → improved fast score = {best_score:.6f}")

        # brief hill‑climb to polish
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1800
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 2 – long simulated annealing using remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[simulated annealing] start")
        best_a, best_score = simulated_annealing(
            best_a, best_score, n, fft_len_conv, deadline
        )
        print(f"[simulated annealing] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
