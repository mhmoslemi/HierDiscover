"""
Optimise a non‑negative step‑function to minimise
`evaluate_sequence`.  The optimisation works on the probability simplex
(∑a_i = 1) – the objective is scale invariant – and proceeds in four
phases:

1. **Rich initialisation** – a pool of analytic families (symmetric /
   asymmetric Beta, inverse‑linear, log‑transform, Dirichlet draws,
   uniform and the historic sequence) is generated and the best few
   are kept.

2. **Continuation L‑BFGS** – a smoothed “log‑sum‑exp” surrogate of the
   max‑convolution is minimised using a decreasing smoothing schedule.
   Each stage runs a bounded number of L‑BFGS iterations.

3. **Exploratory random restarts** – while time permits new Dirichlet
   starts are refined with a short continuation schedule.

4. **Projected sub‑gradient descent with momentum** – a true sub‑gradient
   of the non‑smooth objective is followed (step size  η / √t) and the
   iterate is projected onto the simplex.  Momentum accelerates convergence.

5. **Aggressive hill‑climbing** – random and targeted greedy moves
   polish the final vector.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`),
and the total run time is limited by the 1000 s budget.  The returned
list of floats is suitable for the supplied `evaluate_sequence` routine.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is length n, fft_len ≥ 2*n‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a=1):
    2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and gradient
    w.r.t. the unconstrained vector y (softmax parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c  # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    corr = np.fft.irfft(fa_rev * fw, fft_len_g)

    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def inv_linear_init(n: int) -> np.ndarray:
    """U‑shaped density ~ 1/(i+1) + 1/(n-i)."""
    idx = np.arange(n, dtype=float)
    a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def log_transform_init(n: int) -> np.ndarray:
    """Density proportional to -log(x(1-x))."""
    x = (np.arange(n) + 0.5) / n
    a = -np.log(x * (1.0 - x) + 1e-12)
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def random_dirichlet_init(n: int, rng: np.random.Generator, alpha: float = 0.3) -> np.ndarray:
    """Samples a Dirichlet(α,…,α) vector."""
    if alpha <= 0.0:
        a = np.ones(n)
    else:
        g = rng.gamma(alpha, 1.0, size=n)
        a = g / g.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, renormalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return a / s

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple greedy random hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with simple momentum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        # sub‑gradient of max‑conv (ordered pairs counted twice)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        momentum = beta * momentum - eta * g
        a_new = a + momentum
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The algorithm follows the pipeline described above.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    # reuse a historic sequence if it is sufficiently long
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 8000)
        else:
            n = 6000
    else:
        n = 6000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # --------------------------------------------------------------
    # Build a diverse pool of initial shapes
    # --------------------------------------------------------------
    candidates = []

    # symmetric Beta with a wide range of p (including very U‑shaped)
    for p in [0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65, 0.70, 0.75]:
        candidates.append((f"beta_sym_{p:.2f}", beta_sym(n, p)))

    # a few random asymmetric Beta
    for _ in range(6):
        p = rng.uniform(0.2, 5.0)
        q = rng.uniform(0.2, 5.0)
        candidates.append((f"beta_asym_{p:.2f}_{q:.2f}", beta_asym(n, p, q)))

    # inverse‑linear U‑shape
    candidates.append(("inv_linear", inv_linear_init(n)))

    # log‑transform shape
    candidates.append(("log_transform", log_transform_init(n)))

    # random Dirichlet draws (sparse / concentrated)
    for a_nr in range(4):
        candidates.append(
            (f"dirichlet_{a_nr}", random_dirichlet_init(n, rng, alpha=0.25))
        )

    # uniform
    candidates.append(("uniform", np.full(n, 1.0 / n)))

    # historic sequence, if present
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Quick scan – keep only the most promising few
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:<20s} fast score = {sc:.6f}")

    scored.sort(key=lambda x: x[0])
    top_k = 5
    best_a = None
    best_score = np.inf
    best_y = None

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-9, num=30))
    maxiter_per_stage = 800
    phase_deadline = start_time + 0.45 * budget_s
    print("[phase] L‑BFGS continuation started")

    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:<20s} score = {best_score:.6f}")

        if time.time() >= phase_deadline:
            print("[phase] L‑BFGS time exhausted")
            break

    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_y = np.log(best_a + 1e-12)
        best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Random‑restart continuation (exploratory, ≈ 20 % of budget)
    # ------------------------------------------------------------------
    restart_deadline = start_time + 0.70 * budget_s

    except Exception as e:
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
