def adam_optimize_segments(y_init, deadline, n, seg_len, fft_len_conv, fft_len_g, rng):
    y = y_init.copy()
    m = y.shape[0]
    best_y = y.copy()
    best_score = np.inf
    best_a = None

    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m_t = np.zeros_like(y)
    v_t = np.zeros_like(y)

    it = 0

    # schedule for lam_factor: start high then decays
    # We'll predefine list of lam_factors corresponding to iteration ranges
    schedule = [
        (0, 5000, 0.5),
        (5000, 20000, 0.2),
        (20000, 80000, 0.08),
        (80000, 150000, 0.03),
        (150000, np.inf, 0.01)
    ]
    while time.time() < deadline:
        # Determine lam_factor based on it
        lam_factor = 0.01
        for (start, end, val) in schedule:
            if it >= start and it < end:
                lam_factor = val
                break

        f, grad_y = objective_and_grad_segments(y, n, seg_len, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        it += 1
        m_t = beta1 * m_t + (1 - beta1) * grad_y
        v_t = beta2 * v_t + (1 - beta2) * (grad_y ** 2)

        m_hat = m_t / (1 - beta1**it)
        v_hat = v_t / (1 - beta2**it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        y = y - step

        # compute actual score using fast_score (full a) for checking
        # For speed, don't compute each iteration; maybe every 1000 iters.
        if it % 500 == 0:
            # compute full a from y (via expand)
            prob = np.exp(y - np.max(y))
            prob = prob / prob.sum()
            h = prob / seg_len
            a = np.repeat(h, seg_len)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
                best_a = a.copy()
                print(f"[Adam] iter={it}, fast_score={best_score:.6f}")

    # Return best a (full) and best_score
    if best_a is None:
        # compute from final y
        prob = np.exp(y - np.max(y))
        prob = prob / prob.sum()
        h = prob / seg_len
        best_a = np.repeat(h, seg_len)
        best_score = fast_score(best_a, fft_len_conv, n)
    return best_a, best_score

def objective_and_grad_segments(y, n, seg_len, fft_len_conv, fft_len_g, lam_factor):
    # softmax
    max_y = np.max(y)
    e = np.exp(y - max_y)
    prob = e / e.sum()
    # heights per element:
    h = prob / seg_len
    a = np.repeat(h, seg_len)
    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
    max_c = conv.max()
    # smoothed max
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    # gradient w.r.t conv entries
    w = exp_shift / Z
    # backprop to a: conv_g = 2 * (a_rev * w) ??? use convolution with reversed a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_g_full = np.fft.irfft(fa_rev * fw, fft_len_g)
    # extract size n segment
    start = n - 1
    g_vec = 2.0 * conv_g_full[start : start + n]
    # aggregate per segment
    # sum g_i per segment
    # we can reshape since seg_len divides n
    g_sum = g_vec.reshape(-1, seg_len).sum(axis=1)  # shape (m,)
    # compute mean_g weighted by prob
    avg = np.dot(prob, g_sum)  # expectation of g_sum under prob
    # gradient wrt y: prob * (g_sum - avg) / seg_len
    grad_y = (prob * (g_sum - avg)) / seg_len
    return f, grad_y
```

def targeted_hill_climb(a, current_score, fft_len_conv, fft_len_g, n, deadline, max_iter=50000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute conv and mask
        fa = np.fft.rfft(best, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # subgradient
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        conv_g_full = np.fft.irfft(fa_rev * fm, fft_len_g)
        g_vec = 2.0 * conv_g_full[(n - 1) : (n - 1 + n)]
        # percentiles
        high_thr = np.percentile(g_vec, 80)
        low_thr = np.percentile(g_vec, 20)
        high_idx = np.where(g_vec > high_thr)[0]
        low_idx = np.where(g_vec < low_thr)[0]
        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score

def adam_optimize_segments(y_init, deadline, n, seg_len, fft_len_conv, fft_len_g, rng):
    y = y_init.copy()
    m = y.shape[0]
    best_y = y.copy()
    best_a_full = None
    best_score = np.inf

    # Adam hyperparameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_t = np.zeros_like(y)
    v_t = np.zeros_like(y)

    it = 0
    while time.time() < deadline:
        # schedule lam_factor decreasing
        lam_factor = max(0.001, 0.5 * (0.999 ** it))
        f, grad_y = objective_and_grad_segments(y, n, seg_len,
                                                fft_len_conv, fft_len_g,
                                                lam_factor)

        it += 1
        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * grad_y
        v_t = beta2 * v_t + (1 - beta2) * (grad_y ** 2)

        m_hat = m_t / (1 - beta1 ** it)
        v_hat = v_t / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # occasional evaluation of true max
        if it % 400 == 0:
            # compute full a
            max_y = np.max(y)
            e = np.exp(y - max_y)
            prob = e / e.sum()
            h = prob / seg_len
            a = np.repeat(h, seg_len)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a_full = a.copy()
                best_y = y.copy()
                print(f"[Adam] iter={it}, fast_score={best_score:.6f}")

    # final evaluation
    if best_a_full is None:
        max_y = np.max(y)
        e = np.exp(y - max_y)
        prob = e / e.sum()
        h = prob / seg_len
        best_a_full = np.repeat(h, seg_len)
        best_score = fast_score(best_a_full, fft_len_conv, n)
    return best_a_full, best_score

"""
Propose a step‑function sequence that minimizes the provided evaluation.

The search proceeds in two stages:

1. **Adam on a smoothed max objective**  
   The sequence is represented by a softmax over a modest number of
   “segment” variables (here 200 segments, each 20 points long).  The
   objective is a log‑sum‑exp approximation of the true max
   convolution, which is smooth and has a tractable gradient.
   Adam (adaptive‑momentum) is run for the bulk of the time budget,
   gradually lowering the smoothing factor.

2. **Targeted gradient‑guided hill climbing**  
   Starting from the Adam solution the full‑resolution vector is refined
   by moving a small fraction of mass from indices with a large
   sub‑gradient (active at the current maximum) to indices with a small
   sub‑gradient, accepting only strictly improving moves.

The final sequence is returned as a plain list of floats; it is then
checked by the external ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) (a is already normalised, ∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective for a *segment* representation
# ----------------------------------------------------------------------


def objective_and_grad_segments(
    y: np.ndarray,
    n: int,
    seg_len: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = piecewise‑constant (segments) derived from
    softmax(y).  Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax over segments -----
    max_y = np.max(y)
    e = np.exp(y - max_y)
    prob = e / e.sum()                     # ∑ prob = 1
    h = prob / seg_len                     # each element of a in a segment
    a = np.repeat(h, seg_len)              # length n (seg_len divides n)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- log‑sum‑exp smooth max -----
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. convolution entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through convolution (a*a) -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_g_full = np.fft.irfft(fa_rev * fw, fft_len_g)

    # g_i = d f / d a_i   (size n)
    start = n - 1
    g_vec = 2.0 * conv_g_full[start : start + n]

    # ----- aggregate gradient per segment -----
    g_sum = g_vec.reshape(-1, seg_len).sum(axis=1)       # shape (m,)
    avg = np.dot(prob, g_sum)                            # weighted mean

    # ----- chain rule through softmax -----
    grad_y = (prob * (g_sum - avg)) / seg_len            # ∂f/∂y

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the segment representation
# ----------------------------------------------------------------------


def adam_optimize_segments(
    y_init: np.ndarray,
    deadline: float,
    n: int,
    seg_len: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimizer on the smoothed‑max objective.
    Returns the best full‑resolution sequence and its fast_score.
    """
    y = y_init.copy()
    m = y.shape[0]

    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m_t = np.zeros_like(y)
    v_t = np.zeros_like(y)

    best_a = None
    best_score = np.inf
    it = 0

    while time.time() < deadline:
        # slowly reduce the smoothing factor
        lam_factor = max(0.001, 0.5 * (0.999 ** it))

        # smoothed objective and gradient
        _, grad_y = objective_and_grad_segments(
            y, n, seg_len, fft_len_conv, fft_len_g, lam_factor
        )

        # Adam update
        it += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad_y
        v_t = beta2 * v_t + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # evaluate true max every few iterations
        if it % 400 == 0:
            max_y = np.max(y)
            e = np.exp(y - max_y)
            prob = e / e.sum()
            h = prob / seg_len
            a = np.repeat(h, seg_len)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iter={it}, fast_score={best_score:.6f}")

    # final fallback if no evaluation happened
    if best_a is None:
        max_y = np.max(y)
        e = np.exp(y - max_y)
        prob = e / e.sum()
        h = prob / seg_len
        best_a = np.repeat(h, seg_len)
        best_score = fast_score(best_a, fft_len_conv, n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb on the full‑resolution vector
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb guided by the sub‑gradient of the current max‑convolution.
    Moves a modest fraction of mass from high‑gradient to low‑gradient indices,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute convolution and active mask
        fa = np.fft.rfft(best, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        conv_g_full = np.fft.irfft(fa_rev * fm, fft_len_g)
        g_vec = 2.0 * conv_g_full[(n - 1) : (n - 1 + n)]

        # choose source from high‑gradient tail, target from low‑gradient tail
        high_thr = np.percentile(g_vec, 80)
        low_thr = np.percentile(g_vec, 20)
        high_idx = np.where(g_vec > high_thr)[0]
        low_idx = np.where(g_vec < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # transfer a small fraction (up to 7 %) of the source mass
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                     # length of the step function
    seg_len = 20                 # each segment covers 20 points → 200 segments
    m = n // seg_len

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial sequence (use previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        old_n = init_seq.size
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, n)
        a0 = np.interp(x_new, x_old, init_seq)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Build segment‑level softmax initialisation
    # ------------------------------------------------------------------
    seg_idx = np.arange(n) // seg_len
    mass_per_seg = np.bincount(seg_idx, weights=a0, minlength=m)
    prob0 = mass_per_seg
    prob_sum = prob0.sum()
    if prob_sum == 0.0:
        prob0 = np.full(m, 1.0 / m)
    else:
        prob0 = prob0 / prob_sum
    y0 = np.log(prob0 + 1e-12)      # logits for the segment softmax

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on segment variables (large time slice)
    # ------------------------------------------------------------------
    deadline_adam = time.time() + 0.78 * budget_s   # keep some margin for final steps
    best_a, best_score = adam_optimize_segments(
        y0, deadline_adam, n, seg_len, fft_len_conv, fft_len_g, rng
    )
    print(f"[Adam] resulting fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Targeted gradient‑guided hill‑climb on full resolution
    # ------------------------------------------------------------------
    deadline_hc = time.time() + 0.20 * budget_s
    best_a, best_score = targeted_hill_climb(
        best_a, best_score, fft_len_conv, fft_len_g, n, deadline_hc, max_iter=60000
    )
    print(f"[hill‑climb] resulting fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation via the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
