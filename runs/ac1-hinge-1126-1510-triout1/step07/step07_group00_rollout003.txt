"""
Propose a low‑evaluation step sequence.

The search works on the probability simplex (∑a = 1) and follows
four stages inside the global time budget (default 1000 s).

0.  A cheap parametric sweep (powers, reverse powers, exponentials)
    to obtain a good initial candidate.
1.  Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation of
    the max‑convolution.  The smoothing parameter λ is driven down to
    10⁻⁵ to approach the true objective.
2.  An Adam‑style projected stochastic‑gradient method that uses the
    true sub‑gradient of the *max* convolution.
3.  A deterministic hill‑climb that repeatedly shifts a small
    fraction of mass from the coordinate with the largest sub‑gradient
    to the one with the smallest.
4.  A short simulated‑annealing phase that explores random mass‑transfer
    moves while allowing occasional uphill steps.

All phases respect a shared 1000 s deadline and keep the best sequence
found.  The final sequence is returned as a plain Python list; its
exact score is obtained with the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  Returns the numerator of the
    evaluation function, i.e. 2·n·max(conv).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, fft_len: int):
    """
    Returns (subgradient, max_conv).  The subgradient corresponds to the
    derivative of max(conv) w.r.t. the vector a (the factor 2 is included).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    grad = 2.0 * conv_g[n - 1 : n - 1 + n]   # sub‑gradient of max(conv)
    return grad, max_val


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam):
    """
    Smoothed version of 2·n·max(conv) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # λ is taken as a fraction of the current max value
    lam = max_c * lam + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """Run L‑BFGS over a decreasing schedule of smoothing parameters."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score


def adam_optimize(
    a0,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
    beta1=0.9,
    beta2=0.999,
    eps=1e-8,
    base_lr=0.2,
):
    """
    Adam‑style projected optimisation on the true max‑convolution.
    Works directly on the simplex (∑a = 1, a ≥ 0).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 1

    while time.time() < deadline:
        grad, _ = compute_subgradient(a, fft_len_g)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # element‑wise step (scaled by sqrt of variance)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = simplex_projection(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a,
    current_score,
    fft_len,
    n,
    deadline,
    max_iter=12000,
    max_frac=0.05,
):
    """
    Deterministic hill‑climb that moves a small fraction (≤ max_frac) of mass
    from the coordinate with the largest sub‑gradient to the one with the
    smallest.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        grad, _ = compute_subgradient(best, fft_len)
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        delta = best[i_hi] * rng.random() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def simulated_annealing(
    a_start,
    start_score,
    deadline,
    fft_len,
    n,
    init_temp=0.01,
    min_temp=1e-8,
):
    """
    Simple simulated‑annealing that randomly transfers a fraction of mass
    between two coordinates.  The temperature decays linearly with the
    remaining time.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        # linear temperature schedule
        remaining = deadline - time.time()
        t_frac = max(0.0, remaining / total_time)
        T = max(min_temp, init_temp * t_frac)

        # choose a donor with non‑negligible mass
        donors = np.where(a > 1e-12)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move up to 20 % of the donor mass
        delta = a[i] * rng.random() * 0.20
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_a, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise, clip to non‑negative and renormalise."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak

    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for sub‑gradient back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (use previous best if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = start_time + 0.05 * budget_s
    for p in np.linspace(0.0, 3.0, 13):
        # (i+1)^p  (increasing power‑law)
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        # reversed power‑law
        a = (np.arange(n, 0, -1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

        if time.time() >= sweep_deadline:
            break

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = [
        0.7,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        0.00005,
        0.00002,
        0.00001,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style projected sub‑gradient optimisation
    # --------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic gradient‑guided hill‑climb
    # --------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = gradient_guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            max_iter=20000,
        )
        print(f"[Hill‑climb] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing for final fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            fft_len_conv,
            n,
        )
        print(f"[Annealing] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided security‑checked function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
