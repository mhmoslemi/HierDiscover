"""
Propose a low‑score step‑function (non‑negative sequence).

Algorithm
---------
* Choose a problem size n (≈ 8000).  If a previous solution exists, keep it
  (capped at 10000) or enlarge to 8000.
* Maintain the best sequence found (initialised from the previous solution
  or from a uniform distribution).
* **Phase 0 – cheap parametric search**:  Sample many β‑distributions
  (discrete Beta(α,β) on {0,…,n‑1}) and keep the best one according to a
  fast FFT‑based score 2·n·max(conv).
* **Phase 1 – continuation Adam**:  Optimise the smooth log‑sum‑exp
  surrogate of the max‑convolution, starting from several heuristics
  (random logits, exponential, power‑law, edge‑heavy, arcsine, the best
  candidate and β‑samples).  The smoothing factor λ is decreased
  geometrically (continuation).
* **Phase 2 – projected sub‑gradient descent** on the true max‑convolution
  objective.  The sub‑gradient is the average of the gradients of all
  (almost) active sum indices.  A simple back‑tracking line‑search
  guarantees descent.
* **Phase 3 – aggressive hill‑climb / simulated annealing**:  Randomly
  move a fraction of mass between coordinates, accepting improving moves
  and, with a decaying temperature, occasional worsening moves to escape
  shallow local minima.
* Finally evaluate the exact score with the supplied ``evaluate_sequence``
  and return the best sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> Tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1)…(n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)              # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: List[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation Adam: for each smoothing factor λ (expressed as a
    multiplicative factor `lam_factor`) perform a few Adam updates.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)                 # slowly decaying learning rate
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a vector y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # more faithful discrete arcsine (Beta(0.5,0.5))
        i = np.arange(n, dtype=float) + 0.5
        a0 = 1.0 / np.sqrt(i * (n + 0.5 - i))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # Sample a Beta(α,β) distribution (α,β ∈ (0.1,3))
        α = rng.uniform(0.1, 3.0)
        β = rng.uniform(0.1, 3.0)
        x = (np.arange(n) + 0.5) / n          # points in (0,1)
        a0 = (x ** (α - 1)) * ((1 - x) ** (β - 1))
        a0 = np.maximum(a0, 1e-20)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "best":
        raise ValueError("mode 'best' should be handled by the caller")

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries whose value is very close to the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the averaged max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                         # ∂(max) / ∂a

        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):                         # back‑tracking line‑search
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # take the reduced step anyway (may be non‑improving)
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def simulated_annealing_hc(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 0.02,
    end_temp: float = 0.0001,
) -> Tuple[np.ndarray, float]:
    """Aggressive hill‑climb that also accepts worsening moves according
    to a linear annealing schedule."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0
    # total number of iterations we can afford
    max_iters = int(1e6)

    while time.time() < deadline and iter_no < max_iters:
        iter_no += 1
        # linear temperature decay
        t = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))  # dummy
        # simpler: compute proportion of time elapsed
        elapsed = time.time() - (deadline - (deadline - time.time()))
        # Instead use proportion of total budget
        # (we recompute each loop for clarity)
        proportion = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        temp = start_temp * (1 - proportion) + end_temp * proportion

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)

        if new_score < best_score:
            best = cand
            best_score = new_score
        else:
            # metropolis acceptance
            prob = np.exp(-(new_score - best_score) / (temp + 1e-12))
            if rng.random() < prob:
                best = cand
                best_score = new_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the best candidate.
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            # upscale/downscale by simple interpolation if sizes differ
            if tmp.size != n:
                x_old = np.linspace(0, 1, tmp.size)
                x_new = np.linspace(0, 1, n)
                best_a = np.interp(x_new, x_old, tmp)
                best_a = np.maximum(best_a, 0.0)
                best_a /= best_a.sum()
            else:
                best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric search over Beta(α,β) families.
    # --------------------------------------------------------------
    phase0_deadline = min(deadline, time.time() + budget_s * 0.15)
    while time.time() < phase0_deadline:
        α = rng.uniform(0.1, 3.0)
        β = rng.uniform(0.1, 3.0)
        x = (np.arange(n) + 0.5) / n
        a = (x ** (α - 1)) * ((1 - x) ** (β - 1))
        a = np.maximum(a, 1e-20)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a
            best_y = np.log(a + 1e-12)
            print(f"[beta] α={α:.3f}, β={β:.3f} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – continuation Adam optimisation.
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-10, num=80))
    adam_deadline = min(deadline, time.time() + budget_s * 0.45)
    iteration = 0
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform", "beta", "best"
    ]

    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        elif mode == "beta":
            # generate a β‑distribution and turn into logits
            α = rng.uniform(0.1, 3.0)
            β = rng.uniform(0.1, 3.0)
            x = (np.arange(n) + 0.5) / n
            a0 = (x ** (α - 1)) * ((1 - x) ** (β - 1))
            a0 = np.maximum(a0, 1e-20)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=15,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[adam {iteration:04d}] mode={mode} → fast score = {best_score:.6f}")

        # short greedy hill‑climb after each Adam run
        hc_deadline = min(adam_deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true objective.
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + budget_s * 0.30)
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.30,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb / simulated annealing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[anneal HC] start")
        best_a, best_score = simulated_annealing_hc(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            start_temp=0.02,
            end_temp=1e-5,
        )
        print(f"[anneal HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
