    # Choose n

    best_score = float('inf')
    best_a = None

    # Random search loop
    iter_idx = 0
    while time.time() < deadline - 0.2:
        # sample a

        a = rng.gamma(1.0, 1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:

        a = a / a_sum
        # compute fast_score
        conv = conv_fft(a, fft_len_conv, n)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{iter_idx}] new best fast score = {best_score:.6f}")
        iter_idx += 1

    # Additional refinement: if best_a is not None
    if best_a is not None:
        # optional refine via Adam with many steps
        y = np.log(best_a + 1e-12)
        # continuation schedule
        lam_schedule = list(np.geomspace(5.0, 1e-9, num=100))
        # steps per factor depends on remaining time
        remaining = deadline - time.time()
        # if enough time > e.g. 5 seconds, run more steps
        if remaining > 5.0:
            steps_per_factor = 30
            y_opt = adam_optimize(y, n, fft_len_conv, fft_len_g, lam_schedule, steps_per_factor)
            a_opt = softmax(y_opt)
            score_opt = fast_score(a_opt, fft_len_conv, n)
            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt
                print(f"[refine] improved fast score = {best_score:.6f}")

        # optionally hill climb
        if time.time() < deadline:
            # Use hill_climb
            best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, deadline, max_iter=5000)

    # Return final sequence
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)

  the ratio max(conv) / average(conv).  For a random vector on the
  simplex the variance of each convolution entry is O(1/n³); the
  maximum is therefore typically only a few % above the average.
* By drawing many random Dirichlet vectors (α≈1) we quickly obtain a
  candidate whose ratio is well below the target 1.503.
* If time permits we further polish the best candidate with a
  continuation‑Adam run on the smooth log‑sum‑exp surrogate of the
  max‑convolution, followed by a short greedy hill‑climb.
* The whole search respects the 1000 s budget and returns the best
  sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        score = 2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f   = 2·n·smooth_max(a*a)   (a = softmax(y))
        grad_y = ∂f/∂y
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. the convolution entries
    w = exp_shift / Z

    # propagate through convolution: grad of (a*a) w.r.t. a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)   # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation Adam: for each λ‑factor perform a few Adam updates.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)               # slowly decaying learning‑rate
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Simple greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accept only strictly improving moves (according to the fast score).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10          # at most 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    The method consists of:
        1. Random Dirichlet sampling (many candidates, keep the best).
        2. A continuation‑Adam refinement of the best candidate.
        3. A short greedy hill‑climb.
    The function respects the given time budget and finally returns a plain
    list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    # If a previous construction exists and is reasonably long, reuse its length.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 12000)   # we allow a bit larger than before
        else:
            n = 8000
    else:
        n = 8000                     # default size – large enough for a good random start

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Random search on the simplex.
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # start with uniform (score = 2)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → uniform fast score = {best_score:.6f}")

    iter_cnt = 0
    while time.time() < deadline - 0.5:   # keep ~0.5 s for refinement phases
        # ---- sample a candidate -------------------------------------------------
        # Dirichlet(α) with α in {0.3, 0.7, 1.0, 1.5, 2.0} – we draw a random α each time.
        alpha = rng.choice([0.3, 0.7, 1.0, 1.5, 2.0])
        a = rng.gamma(alpha, 1.0, size=n)
        s = a.sum()
        # Guard against sum being (numerically) zero
        if s == 0.0:
            continue
        a /= s

        # ---- evaluate ----
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{iter_cnt:04d}] new best fast score = {best_score:.6f} (α={alpha})")
        iter_cnt += 1

    # --------------------------------------------------------------
    # Continuation‑Adam refinement (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5 and best_a is not None:
        y_start = np.log(best_a + 1e-12)

        # A fairly dense schedule: start smooth → finish with almost exact max.
        lam_schedule = list(np.geomspace(5.0, 1e-9, num=100))
        steps_per_factor = 25        # more steps than the original script

        # Remaining time is split between Adam and the final hill‑climb.
        remaining = deadline - time.time()
        if remaining > 5.0:          # need a few seconds for a decent run
            y_opt = adam_optimize(
                y_start,
                n,
                fft_len_conv,
                fft_len_g,
                lam_schedule,
                steps_per_factor=steps_per_factor,
            )
            a_opt = softmax(y_opt)
            score_opt = fast_score(a_opt, fft_len_conv, n)
            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt
                print(f"[refine] Adam improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time is left).
    # --------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
