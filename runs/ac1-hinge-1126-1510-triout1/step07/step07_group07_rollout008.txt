"""
Propose a low‑evaluation step‑function.

The algorithm is a multi‑stage optimiser on the probability simplex
(∑ a_i = 1) that combines:

1. A broad initialisation pool (beta‑shapes, semicircle, uniform,
   Dirichlet draws and mixtures).
2. A continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
   max‑convolution. The smoothing parameter λ is decreased geometrically.
3. A projected sub‑gradient descent on the true max‑convolution objective.
4. Aggressive random / targeted hill‑climbing followed by a short
   simulated‑annealing polishing phase.

All stages respect a global 1000 s budget and the best discovered
vector is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the real part of the length (2*n-1) result.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a_i = 1):
    2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate of max‑convolution (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    λ = max(convolution) * lam_factor  (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """Continuation L‑BFGS with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """Symmetric β(p,p) pdf on (0,1); normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle, normalised."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def dirichlet_init(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and normalise to sum 1."""
    y = rng.gamma(alpha, 1.0, size=n)
    if y.sum() == 0.0:
        y = np.ones(n)
    return y / y.sum()


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Local optimisation helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that prefers moving mass away from the current max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)

        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 40000,
    init_temp: float = 1e-3,
    final_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing polish."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    # geometric temperature schedule
    while time.time() < deadline and it < max_iter:
        it += 1
        t = it / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            if diff < 0:
                best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  All stages respect the global 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 6000)  # keep it manageable
        else:
            n = 3000
    else:
        n = 3000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # --------------------------------------------------------------
    # Build a diverse pool of initialisations
    # --------------------------------------------------------------
    candidates = []

    # beta‑shapes (including U‑shaped variants)
    for p in np.linspace(0.35, 0.80, 10):
        a = beta_init(n, p)
        candidates.append((f"beta_{p:.2f}", a))

    # semicircle, uniform
    candidates.append(("semicircle", semicircle_init(n)))
    candidates.append(("uniform", uniform_init(n)))

    # Dirichlet draws with different concentration parameters
    for alpha in (0.15, 0.30, 0.60):
        a = dirichlet_init(n, rng, alpha=alpha)
        candidates.append((f"dirichlet_{alpha:.2f}", a))

    # Mixtures of a modestly U‑shaped beta with uniform
    beta_ref = beta_init(n, 0.45)
    for w in (0.2, 0.4, 0.6, 0.8):
        mixed = mixture_init(beta_ref, uniform_init(n), w)
        candidates.append((f"beta_uni_mix_{w:.1f}", mixed))

    # Historic solution (if any)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # --------------------------------------------------------------
    # Quick scan – keep the most promising few
    # --------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:22s} fast score = {sc:.6f}")

    scored.sort()
    top_k = 6
    best_a = None
    best_score = np.inf
    best_y = None

    # --------------------------------------------------------------
    # Continuation L‑BFGS phase
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-10, num=30))
    maxiter_per_stage = 600
    phase_deadline = start_time + 0.45 * budget_s
    print("[phase] L‑BFGS continuation started")

    for rank in range(min(top_k, len(scored))):
        if time.time() >= phase_deadline:
            print("[phase] time exhausted")
            break
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:22s} score = {best_score:.6f}")

        # a fast local hill‑climb to shake off possible tiny plateaus
        if time.time() < phase_deadline:
            bc, bs = hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline=phase_deadline,
                max_iter=1500,
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] post‑hill‑climb improved to {best_score:.6f}")

    # Fallback to uniform if nothing succeeded
    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Random‑restart continuation (exploratory)
    # --------------------------------------------------------------
    restart_deadline = start_time + 0.65 * budget_s
    restart_cnt = 0
    while time.time() < restart_deadline:
        restart_cnt += 1
        a0 = dirichlet_init(n, rng, alpha=0.4)
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule[:14],  # shorter schedule for quick restarts
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        sc = fast_score(a_opt, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] #{restart_cnt:03d} improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Projected sub‑gradient descent (major refinement)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 30.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a, fft_len_conv, n, subgrad_deadline, eta0=0.30, max_iters=600_000
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Aggressive polishing: hill‑climb + targeted + simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
            max_move=0.30,
        )
    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
            max_move=0.20,
        )
    if time.time() < deadline:
        print("[final] simulated annealing")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
            init_temp=5e-4,
            final_temp=1e-9,
        )

    # --------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
