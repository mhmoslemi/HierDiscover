def build_constraint_matrix(f, active_indices):
    n = len(f)
    K = len(active_indices)
    A = np.zeros((K, n), dtype=float)
    for row, k in enumerate(active_indices):
        # for each j (index in b), i = k - j
        # condition 0 <= i < n => k - n + 1 <= j <= k
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        i_start = k - j_end
        i_end = k - j_start
        # mapping: b[j_start:j_end+1] * f[i_start:i_end+1] sum
        A[row, j_start:j_end+1] = f[i_start:i_end+1]
    return A

"""
Propose a step‑function with a lower evaluation score.
The algorithm combines:
  * a quick scan of diverse analytic initialisations,
  * a small linear‑programming (LP) refinement that mimics the
    “mix‑with‑gradient‑bounded partner” idea from the paper,
  * long projected hill‑climbing (random + targeted) for polishing.

The LP step solves:
    max   sum(b_i)
    s.t.  (f * b)[k] ≤ M   for a selected set of convolution indices k,
          b_i ≥ 0,
where M = max(f * f).  The solution g = b/∑b is then mixed into the
current distribution f ← (1‑t) f + t g.  Repeating this quickly drives
down the maximal self‑convolution while keeping f on the probability
simplex (∑ f_i = 1, f_i ≥ 0).

The whole search respects a 1000 s budget and returns the best sequence
found.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a is length n, fft_len ≥ 2*n-1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (∑ a_i = 1): 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initialisers (all are normalised to sum = 1)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def beta_init(n: int, p: float, q: float = None) -> np.ndarray:
    """Beta(p,q) density on (0,1).  If q is None use the symmetric case."""
    if q is None:
        q = p
    x = (np.arange(n) + 0.5) / n
    # use safe computation of x^{p-1} (avoid zero division)
    a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = uniform_init(n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = uniform_init(n)
    else:
        a = a / a.sum()
    return a


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Hill‑climbing helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Moves mass away from indices that contribute most to the current max‑conv."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))
        # indices that pair to produce k
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contributors = idx[mask]
        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# LP‑based refinement utilities
# ----------------------------------------------------------------------


def build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Build matrix A where (A @ b)[idx] = (f * b)[k] for each k in ks.
    The entry A[row, j] = f[k - j] when 0 ≤ k - j < n, else 0.
    """
    n = f.shape[0]
    K = ks.size
    A = np.zeros((K, n), dtype=float)
    for row, k in enumerate(ks):
        # j range where k - j is a valid index of f
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        i_start = k - j_end
        i_end = k - j_start
        # assign the slice
        A[row, j_start : j_end + 1] = f[i_start : i_end + 1]
    return A


def lp_refine(f: np.ndarray, M: float, t: float, fft_len: int,
              max_active: int = 150) -> np.ndarray:
    """
    Perform one LP refinement step.
    Returns the mixed distribution (1‑t)·f + t·g,
    where g is obtained by solving the LP:
        maximise  ∑ g_i
        s.t.      (f * g)[k] ≤ M  for selected k,
                  g_i ≥ 0.
    """
    n = f.shape[0]
    # compute full convolution once to pick the active set
    conv_full = conv_fft(f, fft_len, n)
    # select up to `max_active` indices with largest values (including the max)
    if conv_full.size <= max_active:
        active_ks = np.arange(conv_full.size)
    else:
        # argpartition for largest
        idx = np.argpartition(conv_full, -max_active)[-max_active:]
        active_ks = np.sort(idx)  # sorting not strictly needed
    # build linear constraints
    A = build_constraint_matrix(f, active_ks)
    b_ub = M * np.ones(active_ks.size, dtype=float)
    c = -np.ones(n, dtype=float)  # minimise -sum(g)  → maximise sum(g)

    # bounds: g_i ≥ 0
    bounds = [(0.0, None)] * n

    # solve LP (HiGHS is fast for moderate sized problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        # fall back to uniform if something went wrong
        g = uniform_init(n)
    else:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s == 0.0:
            g = uniform_init(n)
        else:
            g = g / s

    # convex combination
    return (1.0 - t) * f + t * g

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score under the given 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = int(min(tmp.size, 12000))
        else:
            n = 8000
    else:
        n = 8000

    fft_len = next_pow2(2 * n - 1)
    print(f"[info] using n = {n}")

    # ------------------------------------------------------------------
    # Generate a diverse pool of initial shapes (all already normalised)
    # ------------------------------------------------------------------
    candidates = []

    # symmetric beta family
    for p in [0.55, 0.60, 0.65, 0.70, 0.75, 0.80]:
        candidates.append((f"beta_{p:.2f}", beta_init(n, p)))

    # asymmetric beta (p ≠ q) that might give a different tilt
    for p, q in [(0.55, 0.80), (0.60, 0.85), (0.70, 0.90)]:
        candidates.append((f"beta_{p:.2f}_{q:.2f}", beta_init(n, p, q)))

    # classic shapes
    candidates.append(("semicircle", semicircle_init(n)))
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("random", random_gaussian_init(n, rng)))
    candidates.append(("binary", binary_sparse_init(n, rng, density=0.15)))
    candidates.append(("bump", gaussian_bump_init(n, rng)))

    # mixtures of the best beta (p≈0.70) with uniform
    beta_ref = beta_init(n, 0.70)
    for w in [0.2, 0.4, 0.6, 0.8]:
        candidates.append((f"beta_unif_mix_{w:.1f}",
                           mixture_init(beta_ref, uniform_init(n), w)))

    # previous solution, if compatible
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Quick scan – keep only the most promising ones
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:20s} → fast score = {sc:.6f}")

    scored.sort()          # lower scores first
    top_k = 5              # stay a bit broader than before
    best_f = None
    best_score = np.inf

    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        if sc < best_score:
            best_score = sc
            best_f = a0.copy()
            print(f"[init] selecting {name} as start, score = {best_score:.6f}")

    if best_f is None:
        best_f = uniform_init(n)
        best_score = fast_score(best_f, fft_len, n)

    # ------------------------------------------------------------------
    # LP‑based refinement phase (≈ 60 % of time)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.60 * budget_s
    t_mix = 0.12                          # mixing coefficient
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 50:
        lp_iter += 1
        conv = conv_fft(best_f, fft_len, n)
        M = conv.max()
        # perform one LP update
        new_f = lp_refine(best_f, M, t_mix, fft_len, max_active=150)
        new_score = fast_score(new_f, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_f = new_f
            print(f"[lp] iter {lp_iter:02d} improved → score = {best_score:.6f}")
        else:
            # even if not improved, keep the new distribution for next iteration
            best_f = new_f
        if time.time() > lp_deadline:
            break

    # ------------------------------------------------------------------
    # Final polishing with aggressive hill‑climbing (≈ 35 % of time)
    # ------------------------------------------------------------------
    polish_deadline = start_time + 0.95 * budget_s
    cur_score = fast_score(best_f, fft_len, n)

    # Alternate random and targeted hill‑climbs
    while time.time() < polish_deadline:
        # random hill‑climb block
        best_f, cur_score = hill_climb(
            best_f, cur_score, fft_len, n, deadline=polish_deadline,
            max_iter=3000, max_move=0.25
        )
        # targeted hill‑climb block
        best_f, cur_score = targeted_hill_climb(
            best_f, cur_score, fft_len, n, deadline=polish_deadline,
            max_iter=2500, max_move=0.20
        )
        # break if we are very close to the deadline
        if time.time() > polish_deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # provided by harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation error:", e)
        final_val = None

    return final_seq
