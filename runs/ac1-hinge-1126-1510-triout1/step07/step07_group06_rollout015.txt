"""
A robust multi‑phase optimizer for the step‑function objective.

Phases
------
1.  Parameter sweep over a simple power‑law family (arcsine‑like) to obtain a
    good deterministic starting point.
2.  Continuation Adam on a log‑sum‑exp smoothing of the max‑convolution.
    The smoothing factor λ is decreased geometrically; many Adam steps are
    taken per λ to approach the true (non‑smooth) objective.
3.  Projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is averaged over all (almost) active constraints.
4.  (Optional, for moderate problem sizes) a linear‑programming refinement
    that minimizes the peak of the mixed convolution `f * g` under the current
    bound.  The LP is solved with SciPy's HiGHS solver.
5.  An aggressive random hill‑climb that transfers mass between coordinates.

All steps respect the non‑negative simplex constraint (sum = 1).  The final
sequence is returned as a plain list; the harness will evaluate it with the
provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (smooth objective, gradient w.r.t. logits y).

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # values ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # propagate through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation Adam optimiser.  For each smoothing factor λ (given as
    `lam_factor`), a fixed number of Adam updates are performed.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam_factor in lam_schedule:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor
            )
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)          # slowly decaying learning rate
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "best":
        raise ValueError("mode 'best' must be handled by caller")

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction of mass from i to j.
    Only strictly improving moves are accepted.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The sub‑gradient is averaged over all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):  # back‑tracking up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional for moderate n)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A such that (A @ g)[i] = (f * g)[i].
    Shape of A is (2*n-1, n).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i in range(2 * n - 1):
        j_min = max(0, i - (n - 1))
        j_max = min(n - 1, i)
        # f[i - j] is the coefficient for g_j
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            data.append(f[i - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(f: np.ndarray, M_f: float, max_iter: int = 5) -> np.ndarray:
    """
    Linear‑programming refinement:
        minimise M   subject to f * g ≤ M , g ≥ 0 , sum(g) = 1 .
    Returns the best feasible g (sum normalized) found.
    """
    n = f.shape[0]
    best_g = f.copy()
    best_M = M_f

    for _ in range(max_iter):
        A = build_convolution_matrix(f)                     # (2n-1, n)
        # Augment A with a column for the variable M (with coefficient -1)
        neg_one = -np.ones((A.shape[0], 1))
        A_ub = hstack([A, csr_matrix(neg_one)])           # shape (2n-1, n+1)
        b_ub = np.full(A.shape[0], M_f)

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Objective: minimise M (the last variable)
        c = np.zeros(n + 1)
        c[-1] = 1.0

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        x = res.x
        g = x[:n]
        M_new = x[-1]

        # Numerical clean‑up
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s <= 0.0:
            break
        g /= s

        # Compute the true peak of g * g
        true_M = conv_fft(g, next_pow2(2 * n - 1), n).max()
        if true_M + 1e-12 < best_M:
            best_M = true_M
            best_g = g.copy()
            f = g  # use the new g for the next iteration
        else:
            break

    return best_g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (sequence) with a low
    evaluate_sequence score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 8000)
            init_seq = tmp[:n]
        else:
            n = 5000
            init_seq = None
    else:
        n = 5000
        init_seq = None

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability vector).
    # ------------------------------------------------------------------
    if isinstance(init_seq, np.ndarray) and init_seq.size == n:
        best_a = np.maximum(init_seq, 0.0)
        if best_a.sum() < 1e-12:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a = best_a / best_a.sum()
    else:
        # Quick deterministic scan over a simple power‑law family
        best_a = None
        best_sc = np.inf
        for p in np.linspace(0.2, 2.0, 15):
            i = np.arange(n, dtype=float)
            a = (i + 1) ** (-p) + (n - i) ** (-p)
            a /= a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_sc:
                best_sc = sc
                best_a = a
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
        best_sc = fast_score(best_a, fft_len_conv, n)

    best_score = best_sc
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Continuation Adam.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-12, num=120))
    steps_per_factor = 30
    adam_deadline = min(deadline, time.time() + budget_s * 0.35)
    iteration = 0
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform", "best"
    ]

    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.01, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, steps_per_factor
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each Adam run
        hc_deadline = min(adam_deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + budget_s * 0.40)
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.30,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Optional LP refinement (only for moderate n).
    # ------------------------------------------------------------------
    if n <= 2000 and time.time() < deadline:
        print("[lp refine] start")
        M_current = conv_fft(best_a, fft_len_conv, n).max()
        refined_a = lp_refine(best_a, M_current, max_iter=5)
        refined_score = fast_score(refined_a, fft_len_conv, n)
        if refined_score < best_score:
            best_a, best_score = refined_a, refined_score
            print(f"[lp refine] improved → fast score = {best_score:.6f}")
        else:
            print("[lp refine] no improvement")
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 4 – Final aggressive hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final hill] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (the harness will compute the exact score).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
