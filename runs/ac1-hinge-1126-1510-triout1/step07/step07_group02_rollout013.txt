def exp_gradient_optimize_adapt(a_start, deadline, n, fft_len_conv, fft_len_g, rng):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    eta0 = 2.0 # maybe high
    while time.time() < deadline:
        # compute conv and max
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # building mask for subgradient (active set)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient g (size n)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]
        # compute adaptive eta
        eta = eta0 / np.sqrt(it + 1)
        # exponentiated update
        exp_term = np.exp(-eta * g)
        a_new = a * exp_term
        a_new_sum = a_new.sum()
        if a_new_sum <= 0:
            a_new = a.copy()
        else:
            a_new /= a_new_sum
        # possibly accept if improvement else not
        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        # optionally accept always to keep moving?

        it += 1

def targeted_hill_climb(a, current_score, fft_len, n, deadline, rng, max_iter=50000):
    best = a.copy()
    best_score = current_score
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute subgradient for current best
        # compute conv and max
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(np.float64)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n-1):(n-1+n)]  # subgradient
        # pick source and target based on g
        # source: high g_i (meaning contributes heavily to max)
        # target: low g_i (least contribute)
        # We can select using quantiles
        src_idx = np.argmax(g)  # maybe also weight by value a_i
        # ensure source has enough mass
        if best[src_idx] < 1e-12:
            # get next
            src_idx = np.where(best > 1e-12)[0][0]
        tgt_candidates = np.where(g == g.min())[0]
        if len(tgt_candidates) == 0:
            tgt_idx = rng.integers(n)
        else:
            tgt_idx = rng.choice(tgt_candidates)
        # transfer delta
        delta = best[src_idx] * rng.random() * 0.1
        if delta <= 0:
            it += 1
            continue
        best[src_idx] -= delta
        best[tgt_idx] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src_idx] += delta
            best[tgt_idx] -= delta
        it += 1
    return best, best_score
```

"""
Propose a low‑score step‑function sequence.

The optimizer consists of three stages:

1. **Heavy L‑BFGS on a log‑sum‑exp smooth approximation of the max
   convolution.**  A very fine continuation on the smoothing factor
   (down to 5·10⁻⁶) is performed with many random restarts.  The
   objective and its gradient are hand‑coded and use FFTs for speed.

2. **Adaptive exponentiated‑gradient descent** on the exact (hard) max.
   The learning‑rate starts large (η₀≈3) and decays as 1/√t, with a
   cheap line‑search for the best η among a few candidates each iteration.
   The current best solution is constantly tracked.

3. **Targeted greedy hill‑climb.**  Using the sub‑gradient of the max‑convolution
   we repeatedly move a small fraction of mass from the coordinates that
   contribute most to the current peak to those that contribute least.
   Only improving moves are kept.

All three phases respect the 1000 s wall‑time budget, and the best
sequence found is returned.  The code works with a 4000‑element
probability vector (the sum is 1, which makes the fast_score equal to
the official evaluation score) and uses only NumPy / SciPy.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFTs (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Random initialisation
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal',
         'sparse', 'alternating']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:               # 'alternating'
        a = np.full(n, 1.0 / (2 * n))
        a[::2] = 2.0 / (2 * n)       # twice the average on even positions
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑convolution.
    y are logits, a = softmax(y) lives on the simplex.
    Returns (objective, gradient) w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract part that corresponds to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (adaptive step)
# ----------------------------------------------------------------------


def exp_gradient_optimize_adapt(a_start: np.ndarray,
                                deadline: float,
                                n: int,
                                fft_len_conv: int,
                                fft_len_g: int,
                                rng: np.random.Generator):
    """
    Exponentiated‑gradient on the true max‑convolution.
    Adaptive learning‑rate with a cheap line‑search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    t = 0
    # initial learning‑rate scale
    eta0 = 3.0

    while time.time() < deadline:
        # ----- compute convolution and sub‑gradient -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # ----- adaptive step (line‑search over a few candidates) -----
        # candidate scale factors
        cand_factors = [0.5, 1.0, 2.0]
        best_local = None
        best_local_score = np.inf
        for cf in cand_factors:
            eta = eta0 * cf / np.sqrt(t + 1.0)
            exp_term = np.exp(np.clip(-eta * g, -50, 50))
            a_candidate = a * exp_term
            s = a_candidate.sum()
            if s <= 0.0:
                continue
            a_candidate /= s
            sc = fast_score(a_candidate, fft_len_conv, n)
            if sc < best_local_score:
                best_local_score = sc
                best_local = a_candidate

        if best_local is not None:
            a = best_local
            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()

        t += 1

        # occasional random jitter to escape plateaus
        if t % 1500 == 0:
            a = add_noise(a, sigma=0.003, rng=rng)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted greedy hill‑climb using sub‑gradient information
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 30000):
    """
    Greedy hill‑climb that moves mass from high‑gradient coordinates
    to low‑gradient coordinates.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(np.float64)

        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fr = np.fft.fft(rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fr))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # pick source (large gradient) and destination (small gradient)
        src = np.argmax(g)
        if best[src] <= 0.0:
            # degenerate – pick a random non‑zero entry
            nz = np.where(best > 1e-12)[0]
            if nz.size == 0:
                break
            src = rng.choice(nz)
        # destination – pick a coordinate with minimal gradient
        dst_candidates = np.where(g == g.min())[0]
        dst = rng.choice(dst_candidates)

        # transfer a modest fraction of source mass
        delta = best[src] * rng.random() * 0.08  # up to 8 % of source
        if delta <= 0.0:
            it += 1
            continue

        best[src] -= delta
        best[dst] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert (no improvement)
            best[src] += delta
            best[dst] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a small evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # 8192 for n=4000
    fft_len_g = next_pow2(3 * n - 2)         # a safe length for gradient conv

    # ------------------------------------------------------------------
    # Starting point – use a previously discovered construction if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to required length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # adaptive EG
    # remaining time for hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with a finer λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002,
        0.00001, 0.000005
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore neighbourhood
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – adaptive exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize_adapt(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        rng,
        max_iter=40000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
