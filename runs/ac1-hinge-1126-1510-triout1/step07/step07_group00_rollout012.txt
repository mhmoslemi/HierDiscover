def adam_optimize(y, n, fft_len, max_iter, deadline, lam_factor_init=0.4, lam_factor_final=0.001):
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a, best_score = None, np.inf
    for t in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # compute a from y via softmax
        a = softmax(y)

        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

        max_c = conv.max()
        # lambda schedule: linearly decreasing from lam_factor_init*max_c to lam_factor_final*max_c at final iteration
        lam = max_c * ((lam_factor_final) + (lam_factor_init - lam_factor_final)*(max_iter - t)/(max_iter-1))
        # compute smooth max
        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)  # safe?
        Z = exp_shift.sum()
        w = exp_shift / Z  # gradient factor
        # compute gradient w.r.t a
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fw * fa_rev))
        g_vec = conv_g[n-1:n-1+n]  # length n

        # gradient w.r.t y
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # learning rate schedule: might start with 0.1 and decay
        lr = 0.3 / np.sqrt(t)  # adjust
        y -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # track best actual score
        if t % 10 == 0:  # evaluate every 10 steps
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

def random_pair_transfer(a, score, fft_len, n, deadline, max_trials=20000):
    best = a.copy()
    best_score = score
    rng = np.random.default_rng()
    # We'll attempt random moves
    while time.time() < deadline and max_trials>0:
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i==j or best[i]==0: 
            max_trials -=1
            continue
        # transfer epsilon fraction from i to j
        eps = best[i] * rng.random() * 0.03
        new_a = best.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # ensure nonnegative
        # compute score using fast_score
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best = new_a
            best_score = new_score
        max_trials -=1
    return best, best_score

"""
search for a step‑function (non‑negative sequence) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (∑a)²

The objective is scale‑invariant; we thus keep ∑a = 1 (simplex).  The
algorithm consists of

  0. a few cheap parametric sweeps (power‑law, exponential) to get a
     decent starter,
  1. many short runs of Adam on a log‑sum‑exp smooth approximation of the
     max‑convolution (softmax‑parameterised a = softmax(y)),
  2. a long projected sub‑gradient descent with momentum on the *true*
     max‑convolution,
  3. a deterministic hill‑climb that moves a tiny amount of mass from the
     coordinate with the worst sub‑gradient to the best one,
  4. a final tiny Gaussian smoothing step (optional).

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a list of python floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑normalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a log‑sum‑exp smooth max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len: int,
    max_iter: int,
    deadline: float,
    lam_factors=None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the *smooth* max‑convolution.
    Returns best sequence (on the simplex) and its fast_score.
    """
    if lam_factors is None:
        lam_factors = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = None
    best_sc = np.inf

    # pre‑compute lengths for indexing
    half = n - 1

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        a = softmax(y)

        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv.max()
        # pick a smoothing factor (decreases over the run)
        idx = min((t - 1) * len(lam_factors) // max_iter, len(lam_factors) - 1)
        lam = max_c * lam_factors[idx] + 1e-12

        shifted = conv - max_c
        # stabilise the exponentials
        exp_shift = np.exp(np.clip(shifted / lam, -700, 700))
        Z = exp_shift.sum()
        w = exp_shift / Z

        # gradient of smooth max w.r.t. a (via back‑propagation through convolution)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_g[half : half + n]          # length n

        # chain rule through softmax
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        # Adam update (learning‑rate decays like 1/sqrt(t))
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.3 / (np.sqrt(t) + 1.0)   # simple schedule
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % 20 == 0:
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    # final fallback if nothing was recorded
    if best_a is None:
        a = softmax(y)
        best_a = a.copy()
        best_sc = fast_score(a, fft_len, n)

    return best_a, best_sc

# Projected sub‑gradient descent with momentum (true max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the *true* max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    half = n - 1
    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[half : half + n]

        # momentum update
        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)

        a_candidate = simplex_projection(a - step * v)
        cand_sc = fast_score(a_candidate, fft_len, n)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_sc

# Deterministic gradient‑guided hill‑climb

def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest sub‑gradient, keeping only
    improving moves.
    """
    best = a.copy()
    best_sc = current_score
    rng = np.random.default_rng()
    half = n - 1
    it = 0

    while time.time() < deadline and it < max_iter:
        # exact max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[half : half + n]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # move up to 5 % of the mass from worst to best
        delta = best[i_hi] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
            # keep the move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_sc

def gaussian_smooth(a: np.ndarray, sigma: float) -> np.ndarray:
    """Apply a mild Gaussian low‑pass filter in the frequency domain."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    freqs = np.fft.fftfreq(fft_len)
    gaussian = np.exp(-0.5 * (freqs / sigma) ** 2)
    smooth = np.real(np.fft.ifft(fa * gaussian))[:n]
    smooth = np.maximum(smooth, 0.0)
    s = smooth.sum()
    if s == 0.0:
        return a.copy()
    return smooth / s


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise from the global “height_sequence_1” if present, else uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to required length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, hs)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)   # uniform start

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len, n)
    print(f"[init] start fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps (power‑law, exponential, etc.)
    # ------------------------------------------------------------------
    # power‑law (including decreasing)
    for p in np.linspace(-2.0, 3.0, 26):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[sweep] power p={p:.2f} → {best_sc:.6f}")

    # exponential decay (left‑to‑right and reversed)
    for c in np.linspace(0.0005, 0.015, 30):
        a = np.exp(-c * np.arange(n))
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[sweep] exp c={c:.4f} → {best_sc:.6f}")

        a_rev = np.exp(-c * np.arange(n)[::-1])
        a_rev /= a_rev.sum()
        sc = fast_score(a_rev, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a_rev.copy()
            print(f"[sweep] rev‑exp c={c:.4f} → {best_sc:.6f}")

    # ------------------------------------------------------------------
    # time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # split the remaining budget
    phase1_end = start_time + 0.45 * budget_s   # Adam restarts
    phase2_end = start_time + 0.78 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.94 * budget_s   # final hill‑climb
    # the rest (≈0.06 × budget) for a tiny Gaussian smoothing pass

    # ------------------------------------------------------------------
    # Phase 1 – short Adam runs with random restarts
    # ------------------------------------------------------------------
    while time.time() < phase1_end:
        # 20 % of the restarts start from a noisy version of the current best
        if rng.random() < 0.2:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.012), rng=rng)
        else:
            a_start = random_initial(n, rng)

        y0 = np.log(a_start + 1e-12)

        # remaining time for this run
        sub_deadline = min(phase1_end, time.time() + 12.0)  # at most 12 s per run
        a_opt, sc_opt = adam_optimize(
            y0,
            n,
            fft_len,
            max_iter=8000,
            deadline=sub_deadline,
            lam_factors=[0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001],
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len, n
        )
        print(f"[sub‑gradient] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = gradient_guided_hill_climb(
            best_a, best_sc, fft_len, n, phase3_end, max_iter=30000
        )
        print(f"[hill‑climb] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional tiny Gaussian smoothing tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        smoothed = gaussian_smooth(best_a, sigma=0.015)   # very mild low‑pass
        sc_smooth = fast_score(smoothed, fft_len, n)
        if sc_smooth < best_sc:
            best_a, best_sc = smoothed, sc_smooth
            print(f"[smooth] improved fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # final evaluation with the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
