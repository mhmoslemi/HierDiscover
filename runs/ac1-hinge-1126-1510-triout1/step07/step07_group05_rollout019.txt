def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Choose n based on init_seq if available else default 8000.
    n_default = 8000
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, (list,tuple,np.ndarray)):
        arr = np.asarray(init_seq, dtype=float)
        if arr.ndim == 1:
            n = min(int(arr.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    # Use n around 8000 but maybe half.
    fft_len = next_pow2(2*n-1)

    # initial best = uniform as baseline
    best_a = np.full(n, 1.0/n)
    best_score = fast_score(best_a, fft_len, n)

    print(f"[init] n={n}, baseline fast_score={best_score:.6f}")

    # Phase 1: random sparse supports
    phase1_deadline = min(deadline, start + max(0, 0.35*budget_s))
    ratios = np.linspace(0.15, 0.85, 15)  # choose different densities
    num_samples = 0
    while time.time() < phase1_deadline:
        # pick random ratio
        p = rng.choice(ratios)
        m = max(2, int(p * n))
        # random unique indices
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / m  # sum=1
        score = fast_score(a, fft_len, n)
        num_samples += 1
        if score < best_score:
            best_score = score
            best_a = a
            print(f"[rand] density={p:.3f} m={m} → new best {best_score:.6f}")
    print(f"[rand] evaluated {num_samples} random candidates")

    # Phase 2: refined subgradient descent (like earlier)
    sub_deadline = deadline - max(0, 0.25*budget_s)  # Reserve 25% for final
    if time.time() < sub_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a, fft_len, fft_len, n,
            sub_deadline,
            eta0=0.3,
            max_iters=600_000,
        )
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # Phase 3: final local hill climbing (aggressive)
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=100_000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq

The score returned by `evaluate_sequence` is  

\[
\frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2}\quad\text{with }a_i\ge 0 .
\]

Since the denominator scales quadratically with a uniform scaling of the
vector, the problem is invariant under `a → c·a`.  
Consequently we can restrict ourselves to *probability vectors*  
`sum(a)=1`.  The task then becomes

*minimise* `2·n·max(conv(a,a))` *subject to* `a_i≥0,  Σ a_i = 1`.

A good shape for `a` is one that concentrates mass at positions that
produce few pairwise sums, while giving the remaining positions only
tiny weight.  Random sparse supports already give a much smaller peak
than the uniform distribution, and a simple projected sub‑gradient step
can further push the mass away from the most active convolution entries.

The algorithm works in three stages

1. **Diverse random starts** – generate many candidate vectors
   (uniform, exponential, power‑law, heavy‑edge, and, crucially,
   *random sparse supports*).  Keep the best one.
2. **Projected sub‑gradient descent** – optimise the true
   `max‑convolution` objective.  For the active (almost maximal)
   convolution entries the gradient points to the indices that most
   increase the peak; a decreasing step size together with projection
   onto the simplex drives the peak down.
3. **Aggressive hill‑climbing** – repeatedly move a random fraction of
   mass from one index to another and accept the move only when the
   fast upper‑bound (`2·n·max(conv)`) improves.

All three stages are time‑controlled; the whole search fits well inside
the 1000 s budget.

```python
"""
search for a low‑score non‑negative step function.

The algorithm consists of
  1. random initial candidates (including sparse binary supports);
  2. a long projected sub‑gradient descent on the exact max‑convolution;
  3. a final hill‑climbing phase with random mass transfers.

The best vector is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb: move a random fraction of mass from i to j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12  # up to 12 % of source mass per step

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The gradient is an average over all (almost) active convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # d/da_i of Σ a_i a_j

        # step size decreasing like 1/√it
        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):                     # back‑track up to 8 times
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # even the halved step did not improve – take the reduced step
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Random initial generators
# ----------------------------------------------------------------------


def random_initial_vector(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a candidate probability vector according to the given mode.
    The vector is always normalised to sum = 1.
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * d)
        a /= a.sum()

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()

    elif mode == "edge":
        # heavy mass at the two edges, tiny in the middle
        prop = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * prop))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
        a /= a.sum()

    elif mode == "sparse":
        # sparse binary support, equal mass on the support
        density = rng.uniform(0.15, 0.85)          # fraction of kept entries
        m = max(2, int(density * n))
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / m

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a = rng.gamma(alpha, 1.0)
        a /= a.sum()

    else:
        raise ValueError(f"unknown mode {mode}")

    return a.astype(float)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (uniform baseline)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → baseline fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – many random starts (≈ 35 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 0.35 * budget_s)
    modes = [
        "uniform",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "sparse",
        "dirichlet",
    ]
    trials = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        a_cand = random_initial_vector(mode, n, rng)
        score_cand = fast_score(a_cand, fft_len_conv, n)
        trials += 1
        if score_cand < best_score:
            best_score = score_cand
            best_a = a_cand.copy()
            print(f"[rand] mode={mode} → improved fast score = {best_score:.6f}")

        # tiny hill‑climb after each restart (≈ 0.5 s)
        hc_deadline = min(phase1_deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
    print(f"[rand] completed {trials} random starts, best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long projected sub‑gradient descent (≈ 35 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.30 * budget_s:
        sub_deadline = deadline - 0.30 * budget_s
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=500_000,
        )
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=50000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
```
