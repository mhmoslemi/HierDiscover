"""
Propose a low‑value step sequence.

The optimiser works in three stages (all fully deterministic apart from
controlled random perturbations):

1. **Heavy L‑BFGS continuation** – the objective
   2·n·max(a∗a) is smoothed with a log‑sum‑exp and minimised with L‑BFGS
   for a decreasing schedule of smoothing parameters λ.
   Several random restarts (including a restart interpolated from the
   previously best sequence ‹height_sequence_1›) are performed; the best
   result is kept.

2. **Adam‑style projected descent** – the true (non‑smoothed) max‑convolution
   is approximated by a smoothed version with a very small λ.  The
   gradient of the smoothed objective w.r.t. the probabilities a is
   computed analytically, an Adam update is taken and the iterate is
   projected back onto the simplex.  The step‑size adapts automatically
   and occasional Gaussian noise is injected to escape plateaus.

3. **Targeted gradient‑guided hill‑climb** – a sub‑gradient of the exact
   max‑convolution is obtained from the active constraints.
   Mass is transferred from indices with large gradient to indices with
   small gradient; the move size decays with the iteration count.
   Only strictly improving moves are kept.

All phases respect the 1000 s budget; the final candidate is evaluated
with the supplied ``evaluate_sequence`` function.  The algorithm is
designed to push the bound below the target 1.503 (the previous best
≈ 1.50658).
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a already normalised (∑a = 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random start on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient w.r.t. a
# ----------------------------------------------------------------------


def smooth_obj_and_grad_a(
    a: np.ndarray,
    lam: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray, np.ndarray]:
    """
    Compute the smoothed objective
        F(a) = 2·n·[ max(conv) + lam·log Σ exp((conv_i-max)/lam) ]
    and its gradient w.r.t. the probability vector a.
    Returns (F, grad_a, conv) where conv is the linear convolution a*a.
    """
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    shifted = conv - max_c
    # exponentials safely shifted; for very small lam values the
    # non‑max entries become essentially zero.
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution:
    #   ∂F/∂a = 2·n·2·(a_rev * w)_{central slice}
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]   # gradient of smooth_max

    # scale to the full objective (2·n factor)
    smooth_max = max_c + lam * np.log(Z)
    obj = 2.0 * n * smooth_max
    return obj, grad, conv


# ----------------------------------------------------------------------
# L‑BFGS continuation (optimises the smoothed objective via logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a with a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # stable softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in ``lam_schedule``, starting each stage from
    the previous result.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        # obtain probabilities
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Adam‑style projected descent on the smoothed objective
# ----------------------------------------------------------------------


def adam_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective with projection onto the simplex.
    The smoothing parameter λ is exponentially decayed from 1e-2 to 1e-6.
    """
    a = a0.copy()
    best_a = a.copy()
    cur_score = fast_score(a, fft_len_conv, n)
    best_score = cur_score

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05      # base learning rate
    lam_start = 0.01
    lam_end = 1e-6
    it = 0
    rng = np.random.default_rng()

    # we do not know the exact iteration budget, so we decay λ based
    # on the iteration index but never let it go below lam_end.
    while time.time() < deadline:
        # decay λ (fast at the beginning, then slowly)
        decay_factor = (lam_end / lam_start) ** (1.0 / max(1, int((deadline - time.time()) / 0.01)))
        lam = max(lam_start * (decay_factor ** it), lam_end)

        obj, grad, _ = smooth_obj_and_grad_a(a, lam, n, fft_len_conv, fft_len_g)

        # Adam update (note that grad is the gradient of the *smoothed* max)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = simplex_projection(a - step)
        cand_score = fast_score(a_candidate, fft_len_conv, n)

        # accept if we improve; otherwise keep the old a
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
        # occasional perturbation to escape flat regions
        if (it + 1) % 4000 == 0:
            if rng.random() < 0.25:
                sigma = rng.uniform(0.001, 0.006)
                a = add_noise(a, sigma, rng)
                a = a / a.sum()
                cur_score = fast_score(a, fft_len_conv, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        it += 1

    return best_a, best_score

def subgradient_maxconv(a: np.ndarray, fft_len_conv: int, fft_len_g: int, n: int):
    """
    Compute the exact max‑convolution value and a sub‑gradient of the
    max‑convolution (size n) using the active set mask.
    Returns (max_val, subgrad).
    """
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    subgrad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, subgrad, conv


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb using the exact sub‑gradient.
    Moves a modest fraction of mass from a “high‑gradient” index to a
    “low‑gradient” index and accepts only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        max_val, g, _ = subgradient_maxconv(best, fft_len_conv, fft_len_g, n)

        # Adaptive percentiles: become tighter as we progress
        p_high = max(85 - it // 5000, 55)   # shrink high threshold slowly
        p_low = min(20 + it // 5000, 45)    # raise low threshold slowly
        high_thr = np.percentile(g, p_high)
        low_thr = np.percentile(g, p_low)

        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] == 0.0:
            it += 1
            continue

        # move a fraction that shrinks with the iteration count
        delta_frac = 0.12 * np.exp(-it / 40000)   # max 12 % early, tapering
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Returns the best list of floats found before the deadline.
    """

    rng = np.random.default_rng(seed)

    # Problem size (power‑of‑two for FFT efficiency)
    # ------------------------------------------------------------------
    max_n = 4096                     # chosen power of two
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, max_n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(max_n, 1.0 / max_n)
        else:

    fft_len_conv = next_pow2(2 * n - 1)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.15  # small safety margin

    phase2_end = start + 0.71 * budget_s   # Adam descent

    # Phase 1 – heavy L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.30, 0.20, 0.13, 0.08,
        0.05, 0.03, 0.018, 0.010, 0.006,

        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected descent on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_descent(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted gradient‑guided hill‑climb (final polish)
    # ------------------------------------------------------------------
    final_deadline = deadline
    cur_score = fast_score(best_a, fft_len_conv, n)
    best_a, best_score = targeted_hill_climb(
        best_a,
        cur_score,
        fft_len_conv,
        fft_len_g,
        n,
        final_deadline,
        max_iter=120000,
    )
    print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # type: ignore  # provided by the platform
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
