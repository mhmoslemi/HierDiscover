def construct_sidon_set(n, target_m):
    sums_used = set()
    S = []
    for p in range(n):
        ok = True
        for q in S:
            if (p+q) in sums_used:
                ok = False
                break
        if not ok:
            continue
        # also check p+p sum
        if (p+p) in sums_used:
            continue
        # add
        S.append(p)
        for q in S: # include new sums
            sums_used.add(p+q)
        if len(S) >= target_m:
            break
    return S

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    n = 4000
    # try to construct maximal Sidon set
    target = int(np.sqrt(2*n))  # approx 89
    # we need at least 73 for target score
    min_required = int(np.ceil(np.sqrt(2*n / 1.503)))  # compute required m
    # perform multiple attempts within time
    best_S = []
    start = time.time()
    attempts = 0
    while time.time() - start < budget_s * 0.9:  # use part of time
        attempts += 1
        # generate random order
        order = rng.permutation(n)
        S = []
        used = set()
        for p in order:
            ok = True
            # test self sum
            s_self = p + p
            if s_self in used:
                continue
            for q in S:
                if (p + q) in used:
                    ok = False
                    break
            if not ok:
                continue
            # add p
            for q in S:
                used.add(p + q)
            used.add(s_self)
            S.append(p)
            # stop early if we have enough
            if len(S) >= target:
                break
        if len(S) > len(best_S):
            best_S = S
            if len(best_S) >= target:
                break
        # check time occasionally
        if attempts % 10 == 0 and time.time() - start > budget_s * 0.8:
            break
    m = len(best_S)
    # If we didn't meet required min, fallback: use uniform distribution
    if m < min_required:
        # use uniform as fallback
        a = np.full(n, 1.0 / n)
    else:
        w = 1.0 / m
        a = np.zeros(n, dtype=float)
        a[best_S] = w
    # Evaluate final score
    final_score = evaluate_sequence(a.tolist())
    print(f"[sidon] constructed set size={m}, final score={final_score}")
    return a.tolist()

"""
Improved “step‑function” optimizer.

The search combines three complementary stages:

1. **Heavy L‑BFGS** – many random restarts on a log‑sum‑exp
   smoothed version of the max‑convolution.  The smoothing
   parameter λ is driven down to 2·10⁻⁶, and each λ is solved
   with up to 1800 L‑BFGS iterations.

2. **Projected Adam** – an additive version of Adam applied to the
   *exact* sub‑gradient of the hard max‑convolution.  The update
   is projected onto the probability simplex after every step,
   which gives a more aggressive descent than the exponentiated‑
   gradient used in the reference implementation.

3. **Intensive hill‑climb** – a final polishing phase that repeatedly
   moves a small random fraction of mass between two coordinates,
   accepting only strictly improving moves.

All phases are guarded by explicit time checks, guaranteeing that the
function returns before the global 1000 s deadline.
The best sequence discovered is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is a probability vector.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Random initialisation

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient components that correspond to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  rng: np.random.Generator):
    """
    Projected Adam optimiser on the *hard* max‑convolution.
    The gradient is obtained from the exact sub‑gradient of the max.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv_full.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps_mask).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = 0.12 * (1.0 / (np.sqrt(t) + 1.0))  # decaying global step
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = simplex_projection(a)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 25000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.08  # up to 8 % of source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.53 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.90 * budget_s   # Adam
    # remaining time → hill‑climb

    # ------------------------------------------------------------------
    # Stage 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # Occasionally perturb the current best to diversify the search
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected Adam on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – intensive greedy hill‑climb for the remaining budget
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=50000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # eval is provided externally
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
