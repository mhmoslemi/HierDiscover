"""
Propose a step‑height sequence with a low evaluation score.

The algorithm works in three simple stages:

1.  Load a previously discovered good sequence (``height_sequence_1``)
    if it is present.  This already beats the uniform baseline.
2.  Run a very cheap random‑restart search on the simplex
    (sampling many random probability vectors).  The objective is
    proportional to the sum of squares, which can be evaluated in O(n)
    time without any FFT.
3.  Perform a tiny hill‑climb that moves a small amount of mass from a
    randomly chosen coordinate to another.  Only strictly improving
    moves are kept.  The loop stops when the global time budget is
    exhausted.

The final list of floats is returned.  ``evaluate_sequence`` will be
called by the outer driver to obtain the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_random(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point uniformly from the n‑dimensional simplex."""
    # Using exponential distribution gives a uniform sample on the simplex.
    y = rng.exponential(scale=1.0, size=n)
    return y / y.sum()


def score_simplex(a: np.ndarray, n: int) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) for a vector that lies on the
    simplex (∑a = 1).  By Cauchy‑Schwarz the convolution maximum equals
    Σ a_i², so we can evaluate the score as 2·n·Σ a_i².
    """
    return 2.0 * n * float(np.dot(a, a))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (the original script used n = 4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Candidate initialisation – try to reuse a known good solution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # If a previously discovered sequence is available, use it.
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        # Normalise it onto the simplex (the evaluation is scale‑invariant)
        hs = np.maximum(hs, 0.0)
        if hs.sum() == 0.0:
            hs = np.full_like(hs, 1.0 / hs.size)
        else:
            hs = hs / hs.sum()
        best_a = hs.copy()
        best_score = score_simplex(best_a, n)
        print(f"[init] using previous best score={best_score:.6f}")
    else:
        # Uniform distribution is the obvious baseline.
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = score_simplex(best_a, n)
        print(f"[init] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – massive random sampling on the simplex
    # ------------------------------------------------------------------
    max_random_samples = 20000
    samples_done = 0
    while time.time() < deadline and samples_done < max_random_samples:
        a = simplex_random(rng, n)
        sc = score_simplex(a, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[random] new best score={best_score:.6f} after {samples_done+1} samples")
        samples_done += 1

    # ------------------------------------------------------------------
    # Phase 2 – tiny deterministic hill‑climb from the best point
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline and it < 15000:
        # pick two distinct coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # move a small fraction of mass from i to j
        delta = best_a[i] * rng.random() * 0.04  # up to 4 % of the source mass
        if delta <= 0.0:
            it += 1
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = score_simplex(cand, n)
        if sc < best_score:
            best_score, best_a = sc, cand
            # report occasional improvements
            if it % 500 == 0:
                print(f"[hill‑climb] iter {it}, improved score={best_score:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Final evaluation (the official scorer will perform the same)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        # ``evaluate_sequence`` is supplied by the runner.
        final_val = evaluate_sequence(final_seq)
        print(f"[final] official score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluation failed:", exc)

    return final_seq
