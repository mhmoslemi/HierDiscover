"""
Propose a non‑negative step function with a low evaluate_sequence score.

The algorithm proceeds in several phases:

1. **Problem size** – Use a length around 8000 (larger than the previous 5000).
   The vector is always normalised to sum = 1, because the objective is
   scale‑invariant.

2. **Generate a diverse pool of seed distributions** – Uniform, exponential,
   power‑law, edge‑heavy, arcsine, beta‑distribution, and a few random
   Dirichlet draws.  The best few (by fast FFT‑based evaluation) become the
   starting points.

3. **Continuation Adam on a smooth log‑sum‑exp surrogate** – For each seed we
   minimise a smoothed version of 2·n·max(a∗a) using Adam with a geometric
   schedule for the smoothing factor.  The softmax parametrisation enforces
   non‑negativity and the simplex constraint.

4. **Projected sub‑gradient descent on the true max‑convolution** – Starting
   from the best Adam output we perform many projected sub‑gradient steps,
   averaging sub‑gradients over all (almost) active convolution entries and
   using a back‑tracking line‑search.

5. **Aggressive hill‑climbing** – A final greedy mass‑transfer loop explores
   random moves that strictly improve the fast objective.

All heavy work uses FFT‑based convolutions (O(n log n)).  The whole search
stops before the time budget expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        # fallback to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric seed distributions
# ----------------------------------------------------------------------


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) density on a discrete grid of length n."""
    x = (np.arange(n, dtype=float) + 0.5) / n  # centre of each bin
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def generate_initial_vectors(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of diverse probability vectors of length n."""
    vectors = []

    # 1. uniform
    vectors.append(np.full(n, 1.0 / n))

    # 2. arcsine-like
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    a /= a.sum()
    vectors.append(a)

    # 3. exponential decay
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    vectors.append(a)

    # 4. symmetric exponential (peak in middle)
    rate = rng.uniform(0.001, 0.02)
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a /= a.sum()
    vectors.append(a)

    # 5. power‑law
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    vectors.append(a)

    # 6. edge‑heavy
    m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
    a_edge = np.sqrt(n / 2.0) / n
    a_center = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, a_center)
    a[:m_edge] = a_edge
    a[-m_edge:] = a_edge
    a /= a.sum()
    vectors.append(a)

    # 7. U‑shaped (large at ends)
    mid = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - mid) / mid
    p = rng.uniform(0.5, 4.0)
    a = 1.0 + p * (dist ** 2)
    a /= a.sum()
    vectors.append(a)

    # 8–12. random beta shapes (α,β in [0.2,4])
    for _ in range(5):
        alpha = rng.uniform(0.2, 4.0)
        beta = rng.uniform(0.2, 4.0)
        a = beta_distribution(n, alpha, beta)
        vectors.append(a)

    # 13. Dirichlet sparse draw (α = 0.1)
    alpha_vec = np.full(n, 0.1)
    a = rng.gamma(alpha_vec, 1.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    vectors.append(a)

    # 14–18. random Gaussian logits (converted through softmax)
    for _ in range(5):
        y = rng.standard_normal(n)
        a = softmax(y)
        vectors.append(a)

    return vectors


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained logits y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv - max_c                # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv (weights for each entry)
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # propagate through convolution: grad_a = 2 * (a * w) (circular‑conv trick)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)         # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 20,
) -> np.ndarray:
    """
    Continuation using Adam on the smoothed objective.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)   # gentle decay
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: randomly move a fraction of mass from i to j.
    Accept only strictly improving moves (according to fast_score).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10          # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses the average of sub‑gradients over all (almost) active indices.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries within a tiny tolerance of the max
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec  # derivative of max conv

        eta = eta0 / np.sqrt(it)
        improved = False
        step = a - eta * grad
        step = project_onto_simplex(step)
        new_score = fast_score(step, fft_len_conv, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = step.copy()
            a = step
            improved = True
        else:
            # back‑track a few times
            for _ in range(8):
                eta *= 0.5
                step = a - eta * grad
                step = project_onto_simplex(step)
                new_score = fast_score(step, fft_len_conv, n)
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = step.copy()
                    a = step
                    improved = True
                    break
            if not improved:
                # even after back‑track, keep the last step (may be worse);
                # this helps escape shallow plateaus.
                a = step

    return best_a, best_score
