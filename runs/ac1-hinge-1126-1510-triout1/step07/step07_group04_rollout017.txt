def subgradient_descent(a_start, fft_len, n, deadline, eta0=0.25, max_iters=1_000_000):

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute gradient for indices within range
        idx = m - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2 * a[idx[valid]]
        # step
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        # project onto simplex
        a_new = project_onto_simplex(a_new)

def guided_hill_climb(a_start, cur_score, fft_len, n, deadline, max_iter=100000, alpha=0.2):
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    # precompute indices array
    indices = np.arange(n)
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        # gradient components = 2*a[m-i] where valid
        idx = m - indices
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2 * best[idx[valid]]
        # find source (high grad) and target (low grad)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            # maybe random move
            i_src, i_dst = rng.integers(0, n, size=2)
            if i_src == i_dst:
                continue
        delta = best[i_src] * alpha
        if delta <= 0:
            continue
        # produce candidate
        cand = best.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # ensure non-negative
        if cand[i_src] < -1e-15:  # numeric
            # adjust
            delta = best[i_src]
            cand[i_src] = 0.0
            cand[i_dst] = best[i_dst] + delta
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # occasional random perturbation to escape stagnation
        elif rng.random() < 0.001:
            # accept worse move with low prob to explore
            best = cand
            best_score = cand_score
    return best, best_score

    a = softmax(y)  # shape (n,)
    # conv
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
    max_c = conv.max()

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)

    f = 2.0 * n * smooth_max

"""
Hybrid optimisation for the step‑function minimisation problem.

The algorithm proceeds in four stages:

1.  Multi‑restart L‑BFGS on a log‑sum‑exp (smooth max) surrogate.
    Various simple initialisations are tried, the arcsine law is one of
    them.  The continuation schedule gradually reduces the smoothing
    parameter.

2.  Projected sub‑gradient descent on the true (non‑smoothed) objective.
    The step‑size decays like 1/√t.  This quickly drives the solution
    close to a (locally) optimal point.

3.  A *guided* hill‑climb that always moves mass from the index that
    contributes most to the current maximal convolution entry to the
    index that contributes the least.  The move size is a fraction of
    the source mass and only accepted if the objective improves
    (occasionally a worsening move is accepted to escape plateaus).

4.  A final short L‑BFGS refinement with extremely low smoothing,
    followed by a short random hill‑climb to use any remaining time.

All heavy convolution work uses NumPy’s real‑FFT (rfft/irfft) and the
search works on the probability simplex (∑a = 1) – the evaluation
function is scale‑invariant, so any normalisation is fine.  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (outputs a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    a = np.asarray(a, dtype=float)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
    2·n·max(conv)   (the denominator (∑a)² = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max (log‑sum‑exp) objective and its gradient w.r.t. the
    unconstrained vector y (the soft‑max parametrisation).
    """
    a = softmax(y)

    # ----- smooth max of the convolution -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv entries -----
    w = exp_shift / Z                     # shape (2n‑1,)

    # ----- derivative of conv w.r.t. a (reverse convolution) -----
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes include a variety of handcrafted patterns and a new
    ``arcsine`` law.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Simple greedy Sidon‑set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        idx = np.arange(n, dtype=float) + 0.5          # centre of each bin
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "two_edge":
        # mass concentrated near both ends, low interior plateau
        edge_frac = rng.uniform(0.08, 0.15)
        edge_len = max(1, int(edge_frac * n))
        a0 = np.full(n, 1.0 / n)                        # start uniform
        a0[:edge_len] = 0.5 / edge_len
        a0[-edge_len:] = 0.5 / edge_len
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i → j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = m - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def guided_hill_climb(
    a_start: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 100_000,
    alpha: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb: at each step move mass from the index that
    contributes most to the current maximal convolution entry to the
    index that contributes least.  Only improving moves are kept (with a
    tiny probability of accepting a worse move to escape plateaus).
    """
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    indices = np.arange(n)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # gradient contribution for each coordinate (ignoring the 2‑factor)
        idx = m - indices
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2 * best[idx[valid]]

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            # fallback to a random pair
            i_src, i_dst = rng.integers(0, n, size=2)
            if i_src == i_dst:
                continue

        delta = best[i_src] * alpha
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # safeguard against tiny negative rounding errors
        if cand[i_src] < -1e-15:
            # move the whole source mass instead
            delta = best[i_src]
            cand[i_src] = 0.0
            cand[i_dst] = best[i_dst] + delta

        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # occasional uphill move to escape flat regions
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows the four‑phase scheme described
    above and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # If a previous construction is available, reuse its length, otherwise
    # start with a moderately large n.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = int(tmp.size)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g    = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.clip(tmp, 0, None)
            if best_a.sum() == 0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        # arcsine distribution – a strong baseline
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        best_a = a0 / a0.sum()

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}  start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Multi‑restart L‑BFGS (≈40 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.7, 1e-9, num=35))
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "arcsine",
        "two_edge",
        "best",
    ]
    phase1_deadline = start_time + budget_s * 0.40
    iteration = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=500,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[lbfgs {iteration:04d}] mode={mode}  new best fast score = {best_score:.6f}")

        # brief random hill‑climb polish
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
            max_move=0.25,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈30 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 30.0, start_time + budget_s * 0.70)
    print("[phase2] projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.25,
        max_iters=800_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[phase2] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Guided hill‑climb (≈15 % of the budget)
    # ------------------------------------------------------------------
    guided_deadline = min(deadline - 20.0, time.time() + (deadline - time.time()) * 0.5)
    print("[phase3] guided hill‑climb")
    best_a, best_score = guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        guided_deadline,
        max_iter=60000,
        alpha=0.22,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[phase3] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final fine L‑BFGS refinement (≈10 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[phase4] final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=25))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=500,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine
            best_y = y_fine
            print(f"[phase4] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Aggressive random hill‑climb for any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[phase5] final random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=40000,
            max_move=0.30,
        )
        print(f"[phase5] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
