"""
Optimise a non‑negative step‑function (a probability vector) for the
objective

    score = 2 * n * max(conv(a, a))

where conv is the linear convolution and n = len(a).
The objective is scale‑invariant, therefore we work on the simplex
{a ≥ 0, Σa = 1}.

The optimiser uses several complementary stages that together explore a
large part of the search space within the 1000 s budget:

1. **Initial heuristics** – generate many candidate shapes (uniform,
   exponential, power‑law, arcsine, beta‑like, edge‑heavy, Sidon set,
   random Dirichlet).  Keep the best one.

2. **Continuation Adam** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor λ is
   decreased geometrically (≈ 70 → 120 stages).  Each stage performs
   several Adam updates (default 25).  Several random starts are
   performed; after each run a short greedy hill‑climb is applied.

3. **L‑BFGS refinement** – once a good point is found, run a limited
   L‑BFGS‑B optimisation on the same smoothed objective with a very
   small λ (≈ 10⁻⁶).  This yields a rapid local improvement.

4. **Projected sub‑gradient descent** – optimise the true nonsmooth
   objective using the averaged sub‑gradient of all (almost) active
   convolution entries.  A back‑tracking line‑search is used.

5. **Aggressive simulated‑annealing hill‑climb** – perturb the logits
   with Gaussian noise (temperature decreasing over time) and accept
   occasional worsening moves; intermittently perform a focused
   greedy mass‑transfer hill‑climb.

All stages respect the time budget; the best sequence found is returned
as a plain list of non‑negative floats.  The final value is printed via
`evaluate_sequence`.
"""

import time

# ──────────────────────────────────────────────────────────────────────
# Basic utilities
# ──────────────────────────────────────────────────────────────────────
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the exact objective for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ──────────────────────────────────────────────────────────────────────
# Smoothed max‑convolution (log‑sum‑exp) and its gradient
# ──────────────────────────────────────────────────────────────────────
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y = ∇_y f.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # max becomes 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1)…(2n‑2) correspond to i = 0…n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation Adam on the smoothed objective.
    For each smoothing factor lam_factor a fixed number of Adam steps is
    performed.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)   # simple decay
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ──────────────────────────────────────────────────────────────────────
# Heuristic initialisers (return *probability* vectors)
# ──────────────────────────────────────────────────────────────────────
def generate_initial_a(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector a for the given mode."""
    if mode == "random":
        a = rng.gamma(shape=1.0, scale=1.0, size=n)
        a /= a.sum()
        return a

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
        a /= a.sum()
        return a

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * d)
        a /= a.sum()
        return a

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        return a

    elif mode == "edge":
        # heavy mass on the two ends, width and height are randomised
        width = max(1, int(n * rng.uniform(0.008, 0.03)))
        heavy = rng.uniform(0.7, 4.0) * np.sqrt(n / 2.0) / n   # scale like previous edge heuristic
        centre = (1.0 - 2 * width * heavy) / (n - 2 * width)
        a = np.full(n, centre)
        a[:width] = heavy
        a[-width:] = heavy
        a = np.maximum(a, 0)
        a /= a.sum()
        return a

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a /= a.sum()
        return a

    elif mode == "beta":
        # a_i ∝ (t*(1-t) + eps)^{γ}, γ < 0 creates a U‑shape.
        t = np.linspace(0.0, 1.0, n)
        eps = 10 ** rng.uniform(-8, -2)          # tiny offset to avoid zero
        gamma = rng.uniform(-2.0, -0.1)          # heavy edges when |γ| large
        a = (t * (1.0 - t) + eps) ** gamma
        a = np.maximum(a, 0)
        a /= a.sum()
        return a

    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a = np.zeros(n)
        a[chosen] = 1.0 / k
        a /= a.sum()
        return a

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a = rng.gamma(alpha, 1.0)
        a /= a.sum()
        return a

    elif mode == "uniform":
        a = np.full(n, 1.0 / n)
        return a

    elif mode == "ushape":
        # quadratic U‑shape: large at the ends, small in the middle.
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a = 1.0 + p * (dist ** 2)
        a /= a.sum()
        return a

    else:
        raise ValueError(f"unknown mode {mode}")


# ──────────────────────────────────────────────────────────────────────
# Greedy hill‑climb on the *exact* objective (fast_score)
# ──────────────────────────────────────────────────────────────────────
def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.08  # up to 8 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ──────────────────────────────────────────────────────────────────────
# Projected sub‑gradient descent on the *true* max‑convolution objective
# ──────────────────────────────────────────────────────────────────────
def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    The sub‑gradient is the average of the gradients of all (almost) active
    convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # entries whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the averaged max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):   # back‑tracking up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # even the reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ──────────────────────────────────────────────────────────────────────
# Simulated‑annealing style random perturbations (optional)
# ──────────────────────────────────────────────────────────────────────
def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_sigma: float = 0.2,
    final_sigma: float = 0.002,
) -> tuple[np.ndarray, float]:
    """
    Random Gaussian perturbations of the logits with a simple Metropolis
    acceptance rule.  Temperature (σ) decays linearly to the deadline.
    """
    rng = np.random.default_rng()
    y_cur = np.log(a_start + 1e-12)
    best_y = y_cur.copy()
    best_score = score_start
    cur_score = score_start

    while time.time() < deadline:
        # linear schedule for σ
        remaining = deadline - time.time()
        total = deadline - (deadline - (deadline - time.time()))  # just for clarity
        sigma = final_sigma + (init_sigma - final_sigma) * (remaining / (deadline - time.time() + 1e-12))

        y_prop = y_cur + rng.normal(scale=sigma, size=n)
        a_prop = softmax(y_prop)
        prop_score = fast_score(a_prop, fft_len, n)

        if prop_score < cur_score:
            accept = True
        else:
            # acceptance probability
            T = sigma  # use σ as a proxy for temperature
            accept = rng.random() < math.exp(-(prop_score - cur_score) / (T + 1e-12))

        if accept:
            y_cur = y_prop
            cur_score = prop_score
            if cur_score < best_score:
                best_score = cur_score
                best_y = y_cur.copy()

    best_a = softmax(best_y)
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    default_n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = default_n
    else:
        n = default_n

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick scan of heuristic shapes.
    # ------------------------------------------------------------------
    shape_deadline = min(deadline, time.time() + budget_s * 0.06)
    shape_modes = ["arcsine", "beta", "edge", "exp", "power", "random", "sidon"]

        mode = rng.choice(shape_modes)
        a_cand = generate_initial_a(mode, n, rng)

        if cand_score < best_score:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------

    except Exception:  # pragma: no cover
        # In case the sandbox does not expose evaluate_sequence during testing.
        print("[final] evaluate_sequence not available – returning best candidate.")
