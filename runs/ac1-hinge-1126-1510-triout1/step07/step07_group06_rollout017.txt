"""
Propose a non‑negative step function (sequence) with a low `evaluate_sequence`
score.

The search proceeds in three phases:

1. **Parametric sampling** – a large number of simple families of
   probability vectors (power‑law, exponential, U‑shaped, mixtures, etc.) are
   generated, normalised and evaluated with a fast FFT‑based convolution.
   The best vector is kept.

2. **Continuation Adam optimisation** – the current best vector is refined
   by minimising a smooth log‑sum‑exp surrogate of the max‑convolution.
   The smoothing factor is decreased geometrically (continuation) and a few
   Adam passes are performed.

3. **Greedy hill‑climbing** – a final aggressive local search moves a random
   fraction of mass between coordinates, accepting only improving moves.
   The best vector found is returned as a plain list of floats.

All arithmetic is performed on the probability simplex (sum = 1); the
evaluation harness is scale‑invariant, so no extra normalisation is needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
          2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation using Adam.
    For each smoothing factor λ (expressed as a multiplicative factor `lam_factor`),
    perform a few Adam updates.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)          # slowly decreasing learning rate
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

# ----------------------------------------------------------------------


def generate_power(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    p = rng.uniform(0.2, 2.0)
    idx = np.arange(n, dtype=float)
    a = (idx + 1) ** (-p) + (n - idx) ** (-p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ exp(-rate·i) + exp(-rate·(n-1-i))."""
    rate = rng.uniform(0.0005, 0.02)
    idx = np.arange(n, dtype=float)
    a = np.exp(-rate * idx) + np.exp(-rate * (n - 1 - idx))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_us_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped: a_i ∝ (dist+ε)^{p}, with dist measured from the centre."""
    p = rng.uniform(0.5, 6.0)
    eps = 1e-6
    mid = (n - 1) / 2.0
    dist = np.abs(np.arange(n, dtype=float) - mid) / mid  # in [0,1]
    a = (dist + eps) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_mix(n: int, rng: np.random.Generator, mode1: str, mode2: str) -> np.ndarray:
    """Mix two elementary families with a random weight."""
    if mode1 == "power":
        a1 = generate_power(n, rng)
    elif mode1 == "exp":
        a1 = generate_exponential(n, rng)
    elif mode1 == "us":
        a1 = generate_us_shape(n, rng)
    else:
        a1 = np.full(n, 1.0 / n)

    if mode2 == "power":
        a2 = generate_power(n, rng)
    elif mode2 == "exp":
        a2 = generate_exponential(n, rng)
    elif mode2 == "us":
        a2 = generate_us_shape(n, rng)
    else:
        a2 = np.full(n, 1.0 / n)

    w = rng.random()
    a = w * a1 + (1.0 - w) * a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random_logits(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random logits → soft‑max."""
    y = rng.standard_normal(n)
    a = softmax(y)
    return a


def generate_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with a random concentration."""
    alpha = rng.uniform(0.01, 0.5)
    a = rng.gamma(alpha, 1.0, size=n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_sidon(n: int, rng: np.random.Generator) -> np.ndarray:
    """Greedy Sidon‑set‑based sparse vector."""
    chosen = []
    sums = set()
    for cand in range(n):
        conflict = False
        for c in chosen:
            if (cand + c) in sums or (2 * cand) in sums:
                conflict = True
                break
        if not conflict:
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
        if len(chosen) >= int(np.sqrt(n) * 1.5):
            break
    k = len(chosen)
    a = np.zeros(n)
    a[chosen] = 1.0 / k
    a /= a.sum()
    return a


def generate_perturb(best_a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small multiplicative perturbation of the current best."""
    noise = rng.normal(scale=0.02, size=best_a.shape)
    a = best_a * (1.0 + noise)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(best_a, 1.0 / best_a.shape[0])
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (reuse existing construction if possible)
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random parametric sampling
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, time.time() + budget_s * 0.40)
    families = [
        "power", "exp", "us", "mix_power_exp",
        "mix_power_us", "mix_exp_us", "random", "sidon", "dirichlet"
    ]

    while time.time() < phase1_deadline:
        mode = rng.choice(families)
        if mode == "power":
            a = generate_power(n, rng)
        elif mode == "exp":
            a = generate_exponential(n, rng)
        elif mode == "us":
            a = generate_us_shape(n, rng)
        elif mode == "random":
            a = generate_random_logits(n, rng)
        elif mode == "sidon":
            a = generate_sidon(n, rng)
        elif mode == "dirichlet":
            a = generate_dirichlet(n, rng)
        elif mode.startswith("mix_"):
            # parse the two components after 'mix_'
            parts = mode.split("_")[1:]  # e.g. ['power', 'exp']
            a = generate_mix(n, rng, parts[0], parts[1])
        else:
            a = best_a.copy()  # fallback

        # occasional perturbation of the current best
        if rng.random() < 0.05:
            a = generate_perturb(best_a, rng)

        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[sample] mode={mode} → new best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – continuation Adam optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(5.0, 1e-9, num=80)
    # a few passes of Adam; each pass sweeps the whole λ schedule
    adam_passes = 4
    for p in range(adam_passes):
        y_opt = adam_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=15,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[adam pass {p}] improved fast score = {best_score:.6f}")

        # short hill‑climb after each Adam pass (≈ 0.4 s)
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1200
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climb (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final hill] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
