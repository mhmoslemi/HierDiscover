"""
Hybrid optimiser for the step‑function min‑max convolution problem.

Key ideas
---------
* Work on the probability simplex (∑a = 1) – the objective is
  scale‑invariant: 2·n·max(conv).
* Use a continuation L‑BFGS on a smooth log‑sum‑exp surrogate of
  max(conv) with many restarts (different random / analytic initial
  shapes).  The optimisation variables are the unconstrained logits
  y so that a = softmax(y).
* Force symmetry a_i = a_{n‑1‑i} after every candidate is produced.
  The optimal solution is expected to be symmetric, and this reduces
  the effective search space.
* After the smooth passes run a projected sub‑gradient descent
  (still on the simplex) to optimise the true max‑conv objective.
* Finish with a *targeted* hill‑climb: locate the index m where the
  convolution is maximal, then move mass from indices that contribute
  most to that entry towards indices that contribute least.  The move
  is accepted only if the true objective improves.
* All phases respect a global time budget (default 1000 s).  The best
  sequence seen at any time is returned.

The routine is deliberately lightweight: it uses only NumPy and
SciPy (L‑BFGS) and the helper FFT‑based convolution implementation.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smooth surrogate of the max‑convolution objective and its gradient."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # make max → 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # reverse‑convolution to get ∂conv/∂a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a short L‑BFGS optimisation for each
    smoothing parameter in ``lam_schedule``.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting log‑probability vector for a given pattern."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # Discrete arcsine law: a_i ∝ 1/√(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb with symmetry enforcement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = symmetrize(a_new)      # keep symmetry
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true (non‑smoothed) objective.
    Symmetry is enforced after each projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        a_new = symmetrize(a_new)          # keep symmetry

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional progress report (optional)
        a = a_new

    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climb: always attack the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contributions to the max entry: prod_i = a_i * a_{m-i}
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros(n)
        contrib[mask] = best[idx[mask]] * best[counterpart[mask]]

        # probability to pick a source index (large contribution)
        if contrib.sum() > 0:
            p_src = contrib / contrib.sum()
        else:
            p_src = np.full(n, 1.0 / n)

        i = rng.choice(n, p=p_src)

        # probability to pick a destination index (small contribution)
        inv = 1.0 / (contrib + 1e-12)
        p_dst = inv / inv.sum()
        j = rng.choice(n, p=p_dst)

        if i == j:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = symmetrize(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
        # else reject (do nothing)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence().  The routine respects a 1000 s budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    default_n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 20000)   # allow larger if supplied
    n = default_n

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, symmetric)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}  start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Multi‑restart continuation L‑BFGS (≈55 % of budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.7, 1e-9, num=30))
    iteration = 0
    modes = [
        "random", "exp", "power", "edge", "binary",
        "bump", "harmonic", "arcsine", "best",
    ]
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = symmetrize(softmax(y_opt))
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode}  improved fast score = {best_score:.6f}")

        # brief hill‑climb polishing
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈30 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.40,
        max_iters=2_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Fine L‑BFGS refinement (very low smoothing)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] starting final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=25))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=800,
        )
        a_fine = symmetrize(softmax(y_fine))
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Aggressive targeted hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        final_hc_deadline = deadline
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
