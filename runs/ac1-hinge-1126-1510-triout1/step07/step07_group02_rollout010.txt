    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=[(0, None)] * n,

    b = res.x
    sum_b = b.sum()
    sum_a = a.sum()
    if sum_b <= 0:

    alpha = sum_a / sum_b
    g = alpha * b
    # mix with current a
    t = 0.5 # mixing parameter
    new_a = (1 - t) * a + t * g
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()

"""
Multi‑phase optimizer for the “step‑function” challenge.

The search works on a fixed length `n` (4000 by default).  It consists
of four stages:

1. **Heavy L‑BFGS** on a log‑sum‑exp smooth approximation of the
   max‑self‑convolution.  Several random restarts are performed and the
   smoothing parameter λ is driven down to 5·10⁻⁵.

2. **Exact exponentiated‑gradient descent** on the true ℓ∞‑norm of the
   self‑convolution (sub‑gradient of the hard max).

3. **Linear‑programming refinement** – from the current iterate `f`
   we maximise the total mass of a new vector `g₀` under the linear
   constraints `f∗g₀ ≤ M` (where `M` is the current peak of `f∗f`).
   After rescaling to the original ℓ¹–norm we mix `f` and the scaled
   `g` which provably reduces the peak.

4. **Greedy hill‑climb** for the remaining budget – small random
   transfers of mass are accepted only when they improve the fast
   score.

All stages respect the 1000 s time budget and the final sequence is
returned as a plain `list[float]`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the quantity 2·n·max(a*a) when `a` is a probability
    vector (i.e. sum(a)=1).  This is exactly the numerator of
    `evaluate_sequence`.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'double_edge']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'double_edge':
        # extra heavy mass at both ends, very sparse in the middle
        edge_frac = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1e-9)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                                   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑self‑convolution.
    Arguments
    ----------
    y : ndarray
        Logits; the actual probabilities are softmax(y).
    n : int
        Length of the sequence.
    fft_len_conv / fft_len_g : int
        FFT lengths used for convolution and for back‑propagation.
    lam_factor : float
        Determines the smoothing λ = lam_factor * max(conv) + eps.
    Returns
    -------
    f , grad_y : float , ndarray
        Smoothed objective value and its gradient w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution (real part)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient components that correspond to a (central n entries)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on the hard max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # sub‑gradient mask

        # gradient of the hard max w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size: η = η₀ / sqrt(t+1)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipping to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a modest random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10   # up to 10 % of source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Linear‑programming refinement
# ----------------------------------------------------------------------


def conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ b = a ∗ b
    (full linear convolution).  A is constructed as a sparse
    Toeplitz matrix using `scipy.sparse.diags`.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # each diagonal i (offset i) contains a[i] repeated n times
    data = [np.full(n, ai) for ai in a]
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    return A


def lp_refine_step(a: np.ndarray,
                   fft_len: int,
                   n: int,
                   deadline: float,
                   mix_factor: float = 0.5):
    """
    One LP‑based refinement:
    * maximise Σ b_j subject to a∗b ≤ M (M = max(a∗a)).
    * rescale the optimum to the original ℓ¹–norm.
    * mix the scaled vector with `a` using `mix_factor`.
    Returns the (possibly) improved distribution and its fast_score.
    """
    if time.time() >= deadline:
        return a, fast_score(a, fft_len, n)

    # current peak of the self‑convolution
    conv = np.convolve(a, a, mode='full')
    M = conv.max()

    # build constraint matrix A (sparse)
    A = conv_constraint_matrix(a)

    rows = 2 * n - 1
    b_ub = np.full(rows, M, dtype=np.float64)

    # objective: maximise Σ b  → minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    # time left for the LP solver
    remaining = deadline - time.time()
    # give the solver at most the remaining time (capped at 30 s)
    lp_time_limit = min(30.0, max(0.0, remaining))

    res = linprog(c,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True,
                           'time_limit': lp_time_limit})
    # free memory from the (large) matrix as soon as possible
    del A

    if not res.success:
        # LP failed – just return the current point
        return a, fast_score(a, fft_len, n)

    b_opt = res.x
    sum_b = b_opt.sum()
    sum_a = a.sum()
    if sum_b <= 0.0 or sum_a <= 0.0:
        return a, fast_score(a, fft_len, n)

    # scale to preserve the original ℓ¹–norm
    alpha = sum_a / sum_b
    g = alpha * b_opt

    # convex combination with the current iterate
    a_new = (1.0 - mix_factor) * a + mix_factor * g
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()

    new_score = fast_score(a_new, fft_len, n)
    old_score = fast_score(a, fft_len, n)
    if new_score < old_score:
        return a_new, new_score
    else:
        return a, old_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial sequence (interpolate if a previous construction exists)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.03            # tiny safety margin

    phase1_end = start_time + 0.45 * budget_s        # heavy L‑BFGS
    phase2_end = start_time + 0.80 * budget_s        # EG descent
    phase3_end = start_time + 0.95 * budget_s        # LP refinement
    # remaining time → hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # Occasionally perturb the best solution, otherwise fresh random start
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # we allow up to three LP refinements; each one is cheap compared
        # with the whole budget because we cap the solver at ~30 s.
        lp_deadline = phase3_end
        for step in range(3):
            if time.time() >= lp_deadline:
                break
            best_a, best_score = lp_refine_step(
                best_a,
                fft_len_conv,
                n,
                lp_deadline,
                mix_factor=0.6,          # a relatively strong move
            )
            print(f"[LP {step + 1}] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive greedy hill‑climb polishing
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=40000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the platform
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
