def gradient_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000):
    best=a.copy()
    best_score=cur_score
    rng = np.random.default_rng()
    for _ in range(max_iter):
        if time.time()>deadline: break
        # compute conv and gradient w.r.t a: we can approximate using smoothed derivative or using subgradient.
        conv = conv_fft(best, fft_len, n)
        max_val = conv.max()
        # compute active set and weight w:
        tol = max(1e-12, (max_val - conv.mean())*1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if len(active)>0:
            w[active] = 1.0 / len(active)
        else:
            w[np.argmax(conv)] = 1.0
        # gradient:
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        grad = conv_rev[(n-1):(n-1+n)] * 2.0
        # choose i = argmax(grad), j = argmin(grad)
        i = np.argmax(grad)
        j = np.argmin(grad)
        # compute delta fraction of source mass to move
        delta = best[i] * 0.05 # 5%
        if delta <= 0: continue
        best[i] -= delta
        best[j] += delta
        # evaluate fast_score
        new_score=fast_score(best, fft_len, n)
        if new_score<best_score:
            best_score=new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best,best_score
```

"""
Propose a step‑function with a low evaluate_sequence score.

Algorithm
---------
1. Choose length n (≈ 8000) – use the previous best if available, otherwise
   a fresh length.  Initialise a probability vector a (sum = 1) – either from
   the stored solution (padded/trimmed) or uniform.
2. **Phase 1 – many random restarts**  
   For the bulk of the time we run continuation L‑BFGS from a variety of
   heuristics (Gaussian logits, exponentials, power‑law, edge‑heavy, etc.).
   The smoothing factor λ is decreased geometrically (≈ 50 stages).  After each
   run a short hill‑climb improves the result.
3. **Phase 2 – fine L‑BFGS**  
   Starting from the best vector so far we run a longer continuation with a
   much finer λ‐schedule (≈ 80 stages, 1500 L‑BFGS iterations per stage).
4. **Phase 3 – gradient‑guided hill climb**  
   Using the exact sub‑gradient of the max‑convolution we repeatedly move a
   small fraction of mass from the index with the largest gradient to the
   one with the smallest gradient, accepting only improving moves.
5. Return the best vector (converted to a plain list) after evaluating it
   with the supplied ``evaluate_sequence`` function.

All operations respect the time budget (default 1000 s) and keep the
sequence non‑negative.  The code is fully self‑contained and makes no
filesystem or network accesses.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first 2·n‑1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective and gradient for L‑BFGS.

    f(y) = 2·n·smooth_max(a*a) , a = softmax(y)
    lam = max(conv)·lam_factor is the smoothing parameter.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c          # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weight for each convolution entry (∂smooth_max/∂conv_i)
    w = exp_shift / Z

    # propagate through convolution a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient wrt y (chain rule through soft‑max)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------
def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Hill‑climbing utilities
# ----------------------------------------------------------------------
def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb.  Move a random fraction of mass from i to j
    and accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # up to 12 % per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def gradient_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    frac_move: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb.  At each iteration we compute the sub‑gradient
    of the max‑convolution, pick the index with the largest (most positive)
    component as a source and the index with the smallest component as a sink,
    then move a small fraction of mass from source to sink if it improves the
    fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution and active set
        conv = conv_fft(best, fft_len_conv, n)
        max_val = conv.max()
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        # sub‑gradient wrt a
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        i_src = np.argmax(grad)   # most harmful location
        i_dst = np.argmin(grad)   # most beneficial location
        if i_src == i_dst or best[i_src] <= 0.0:
            continue

        delta = best[i_src] * frac_move * rng.random()
        if delta <= 0.0:
            continue

        # propose move
        best[i_src] -= delta
        best[i_dst] += delta

        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

    return best, best_score

            # enlarge to roughly double the old size (capped at 10000)

        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):

    best_score = fast_score(best_a, fft_len_conv, n)
