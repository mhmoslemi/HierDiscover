def grad_directed_hill_climb(a, cur_score, fft_len_conv, n, deadline, max_iter=2000, step_frac=0.02):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        # compute gradient (subgradient) of max conv
        conv = conv_fft(best, fft_len_conv, n)
        max_val = conv.max()
        # identify active indices as before
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size
        # compute gradient via convolution a_rev * w
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g) # fft_len_g must be passed; we can compute globally.
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        grad_vec = conv_rev[(n-1):(n-1+n)] * 2.0

        # find indices with largest positive and negative gradient
        i_pos = np.argmax(grad_vec)  # increasing a_i would increase max_c (bad)
        i_neg = np.argmin(grad_vec)  # decreasing a_i would increase max_c? Actually lower gradient means reducing cost if increase a_i? Wait gradient points direction of increase of max_c (since we are minimizing). So we want to move mass from indices with large positive gradient to those with large negative gradient (the latter would reduce max when increased). So i_pos is worst to increase, i_neg is best to increase.
        delta = best[i_pos] * step_frac
        if delta < 1e-15:
            continue
        best[i_pos] -= delta
        best[i_neg] += delta
        # ensure non-negativity
        if best[i_pos] < 0:
            # revert small portion: adjust delta = best[i_pos] before subtract (just set to zero) and add remainder to i_neg? But improbable due to step_frac <1.
            continue
        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_pos] += delta
            best[i_neg] -= delta
    return best, best_score

```
def simulated_annealing(a, cur_score, fft_len_conv, n, deadline, init_temp=0.01, final_temp=1e-6, decay=0.9995, max_move=0.1):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    T = init_temp
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        delta = cur[i] * rng.random() * max_move
        if delta == 0: continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len_conv, n)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            cur_score = new_score
            # update best if improved
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        # decay temperature
        T = max(final_temp, T * decay)
    return best, best_score
```

    # Phase 1: Adam with fine schedule
    lam_factors = list(np.geomspace(10.0, 1e-12, num=110))
    steps_per_factor = 24  # more steps
    # Determine time partition: allocate 0.55 of budget

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:

        # short hill climb after each Adam

        iteration += 1
    # Phase 2: projected subgradient descent (more aggressive)

        best_a,best_score = multi_active_subgrad(best_a, fft_len_conv, fft_len_g, n, phase2_deadline, eta0=0.5, max_iters=500_000)
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished, fast score={best_score:.6f}")
    # Phase 3: gradient guided hill climb
    phase3_deadline = min(deadline, time.time() + (budget_s * 0.10))
    if time.time() < phase3_deadline:
        print("[grad hill] start")
        best_a,best_score = grad_directed_hill_climb(best_a, best_score, fft_len_conv, n, phase3_deadline, max_iter=50000, step_frac=0.02)
        print(f"[grad hill] finished, fast score={best_score:.6f}")
    # Phase 4: simulated annealing final
    if time.time() < deadline:
        print("[sim anneal] start")
        remaining_time = deadline - time.time()
        # use 0.8 factor for anneal

        print(f"[sim anneal] finished, fast score={best_score:.6f}")
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score={final_val:.6f}")
    return final_seq

"""
Hybrid optimizer for the max‑self‑convolution problem.

The algorithm proceeds in four stages:
1. Continuation Adam on a log‑sum‑exp smoothing of the max‑convolution.
2. Projected sub‑gradient descent on the true max‑convolution objective.
3. Gradient‑guided coordinate hill‑climbing (move mass from indices that
   increase the objective to those that decrease it).
4. Simulated‑annealing style random mass moves for final fine‑tuning.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
and the objective is
    score = 2·n·max_k (a * a)_k,
which is exactly the value returned by ``evaluate_sequence`` after normalising.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained logits y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # --- convolution a*a ---
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max via log‑sum‑exp ---
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient w.r.t. conv ---
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # --- propagate through convolution (a * w) ---
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through soft‑max ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation Adam on the smoothed objective.  The smoothing factor
    ``lam_factor`` (a multiplicative factor) is gradually decreased.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor=lam
            )
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            lr = 0.5 / np.sqrt(t)          # slowly decaying step size
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

# Heuristic initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting logits vector y (pre‑soft‑max) for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set – entries whose value is within a tiny tolerance of the max
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # back‑tracking up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def grad_directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50_000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Coordinate‑wise hill‑climb guided by the sub‑gradient.
    At each step mass is moved from the index with the largest positive
    gradient (most harmful) to the index with the most negative gradient
    (most helpful), accepting the move only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute active weights (as in sub‑gradient)
        conv = conv_fft(best, fft_len_conv, n)
        max_val = conv.max()
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # sub‑gradient of max‑convolution
        a_rev = best[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        grad = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]

        i_pos = int(np.argmax(grad))      # biggest increase if we add mass here
        i_neg = int(np.argmin(grad))      # biggest decrease if we add mass here

        delta = best[i_pos] * step_frac
        if delta <= 0.0:
            continue

        best[i_pos] -= delta
        best[i_neg] += delta

        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_pos] += delta
            best[i_neg] -= delta

    return best, best_score


def simulated_annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    init_temp: float = 0.01,
    final_temp: float = 1e-7,
    decay: float = 0.9998,
    max_move: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style mass moves.
    A random fraction of mass is transferred from i to j; the move is
    accepted if it improves the fast score or with a Metropolis probability
    depending on the current temperature.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    T = init_temp
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len_conv, n)
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert
            cur[i] += delta
            cur[j] -= delta

        T = max(final_temp, T * decay)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low ``evaluate_sequence`` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (use existing construction if available)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
            best_a = tmp / np.sum(tmp)
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Continuation Adam (fine smoothing schedule)
    # --------------------------------------------------------------
    lam_factors = list(np.geomspace(10.0, 1e-12, num=110))
    steps_per_factor = 24                     # more inner steps per smoothing level
    phase1_deadline = min(deadline, time.time() + (budget_s * 0.55))
    iteration = 0
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform", "best"
    ]

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.01, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_factors, steps_per_factor
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[adam {iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each Adam run
        hc_deadline = min(phase1_deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (more aggressive)
    # --------------------------------------------------------------
    phase2_deadline = min(deadline, time.time() + (budget_s * 0.30))
    if time.time() < phase2_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            phase2_deadline,
            eta0=0.50,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Gradient‑directed coordinate hill‑climb
    # --------------------------------------------------------------
    phase3_deadline = min(deadline, time.time() + (budget_s * 0.10))
    if time.time() < phase3_deadline:
        print("[grad‑hill] start")
        best_a, best_score = grad_directed_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_deadline,
            max_iter=60_000,
            step_frac=0.02,
        )
        print(f"[grad‑hill] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing fine tuning (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("[sim‑anneal] start")
        best_a, best_score = simulated_annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,                     # use the global deadline
            init_temp=best_score * 0.01,
            final_temp=1e-7,
            decay=0.9998,
            max_move=0.05,
        )
        print(f"[sim‑anneal] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final exact evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
