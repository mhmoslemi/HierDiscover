"""
Optimiser for the “step‑function” challenge.

The algorithm works in three timed phases:

Phase 1 – many random restarts of L‑BFGS on a smooth log‑sum‑exp
approximation of the max‑convolution.  The smoothing parameter λ is
driven down to 5·10⁻⁶; each λ is solved with a few thousand L‑BFGS
iterations.

Phase 2 – exact sub‑gradient descent using an exponentiated‑gradient
update.  A tiny line‑search over a handful of step‑sizes is performed
at every iteration, guaranteeing that each update is improving (or
the step‑size is reduced).

Phase 3 – deterministic “gradient hill‑climb”.  The sub‑gradient of
the hard max‑convolution is computed, the coordinate with the largest
gradient loses a small fraction of its mass and the coordinate with the
smallest gradient receives it; the move is kept only if the fast
objective improves.

All phases respect a global 1000 s deadline and keep the vector on the
probability simplex (∑a = 1, a_i ≥ 0).  The best sequence found is
returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the official score (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter proportional to the current max
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through the convolution (conv = a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract the part that corresponds to the gradient w.r.t. a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple smoothing levels)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage.  Returns the best vector found
    and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_line_search(a_start: np.ndarray,

                             fft_len: int,
                             rng: np.random.Generator):
    """
    Sub‑gradient descent on the hard max‑convolution.
    Uses an exponentiated‑gradient update and a tiny line‑search over
    a set of step‑sizes, guaranteeing non‑worsening updates.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    # candidate step‑sizes (will be shrunk if no improvement)
    step_candidates = np.array([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])

    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        improved = False
        for eta in step_candidates:
            exp_term = np.exp(np.clip(-eta * g, -50, 50))
            a_new = a * exp_term
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved:
            # no candidate helped – shrink the step sizes and continue
            step_candidates = step_candidates * 0.6
            step_candidates = np.clip(step_candidates, 1e-4, None)
            # occasional random perturbation to escape plateaux
            if rng.random() < 0.02:
                sigma = rng.uniform(1e-4, 5e-4)
                a = add_noise(a, sigma, rng)

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill climbing (deterministic)
# ----------------------------------------------------------------------


def hill_climb_gradient(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 30000,
                        base_step: float = 0.02):
    """
    Greedy hill‑climb using the sub‑gradient of the hard max.
    In each iteration the coordinate with the largest gradient donates
    a small fraction of its mass to the coordinate with the smallest
    gradient; the move is kept only if the fast objective improves.
    """
    a = a_start.copy()
    best_score = start_score
    it = 0
    step_factor = base_step

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient (same as in exp‑gradient)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        i = np.argmax(g)   # donor (most responsible for the peak)
        j = np.argmin(g)   # receiver (least responsible)

        if i == j:
            break

        delta = step_factor * a[i]
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # safety – keep non‑negativity
        if a_new[i] < 0.0:
            a_new[i] = 0.0

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score - 1e-12:
            a = a_new
            best_score = sc
            # when we improve we can be a bit bolder
            step_factor = min(0.5, step_factor * 1.05)
        else:
            # reject – become more conservative
            step_factor *= 0.7
        it += 1

    return a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial sequence
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # Initialise from a previously stored construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global time budget
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # Phase splits (empirical)
    phase1_end = start_time + 0.55 * budget_s          # L‑BFGS
    phase2_end = start_time + 0.85 * budget_s          # EG + line‑search
    # Remaining time → deterministic hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002,
        0.00001, 5e-6
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # Occasionally perturb the current best, otherwise fresh start
        if best_score < np.inf and rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent with line‑search
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_line_search(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic gradient hill‑climb
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb_gradient(
        best_a,
        best_score,
        fft_len,
        n,
        final_deadline,
        max_iter=40000,
        base_step=0.02,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
