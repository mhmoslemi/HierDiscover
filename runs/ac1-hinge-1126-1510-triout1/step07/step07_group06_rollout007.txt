"""
Propose a non‑negative step function (sequence) with a low evaluate_sequence score.

Algorithm
---------
1. Choose a problem size `n` (default 8000 or the length of an existing
   construction).  All vectors are kept on the probability simplex
   (sum = 1) and are later returned unchanged – the harness evaluates the
   exact objective.
2. **Continuation Adam** on a smooth log‑sum‑exp approximation of the
   max‑convolution.  The smoothing factor λ = lam·max(conv) is decreased
   geometrically; for each λ a few Adam steps are performed.  Many random
   restarts (Gaussian, exponential, power‑law, edge‑heavy, arcsine, Sidon,
   Dirichlet, U‑shape, uniform and a perturbation of the current best)
   explore the search space.
3. **Exponential sub‑gradient descent** on the true objective.
   The sub‑gradient of `max(a*a)` is obtained by averaging the gradients
   of all (almost) active entries.  An exponentiated‑gradient (mirror‑descent)
   update `a ← a·exp(−η·g)` keeps the iterate non‑negative and is followed
   by a renormalisation onto the simplex.  The step size decays as
   `η = η₀/√t`.
4. **Simulated‑annealing hill‑climb**: random mass transfers between two
   coordinates are accepted if they improve the fast objective or with a
   probability `exp(−Δ/Temperature)`.  The temperature is slowly cooled.
5. The best vector found is returned as a plain list of floats.  The
   harness will compute the exact `evaluate_sequence` score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        # U‑shaped distribution: large weights at the two ends, small in the centre
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 12,
) -> np.ndarray:
    """
    Continuation using Adam.  For each smoothing factor λ (expressed as a
    multiplicative factor `lam_factor`) perform a small fixed number of Adam
    updates.  Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)  # simple decaying schedule
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Sub‑gradient utilities for the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_maxconv(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    tol: float = 1e-12,
) -> np.ndarray:
    """
    Compute a sub‑gradient of  max(a*a)  at the point a.
    Returns a vector of length n.
    """
    conv = conv_fft(a, fft_len_conv, n)
    max_val = conv.max()

    # active set – entries within a tiny relative tolerance of the maximum
    rel_tol = max(tol, (max_val - conv.mean()) * 1e-8)
    active = np.where(conv >= max_val - rel_tol)[0]

    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[np.argmax(conv)] = 1.0

    # gradient of the (averaged) max entry w.r.t. a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # d/d a of maxconv = 2 * g_vec
    return 2.0 * g_vec


def exponential_subgrad_descent(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact objective.
    Returns the best vector found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad = subgrad_maxconv(a, fft_len_conv, fft_len_g, n)

        eta = eta0 / np.sqrt(it)

        # exponentiated update – stays non‑negative
        a_new = a * np.exp(-eta * grad)
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:          # numerical safeguard
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= a_new_sum

        new_score = fast_score(a_new, fft_len_conv, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        # optionally keep moving even if not improved (helps escape plateaux)
        # here we simply keep the old a when no improvement

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (also used as a quick local optimiser)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def simulated_annealing_hc(
    a_start: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    init_temp: float = 0.02,
    cooling: float = 0.998,
    max_move: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.  Random mass moves are accepted if they
    improve the fast objective or with probability `exp(-Δ/Temp)`.
    The temperature decays geometrically.
    """
    a = a_start.copy()
    score = cur_score
    rng = np.random.default_rng()
    temp = init_temp

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        accept = False
        if new_score < score:
            accept = True
        else:
            prob = np.exp(-(new_score - score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            score = new_score
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        # cool
        temp *= cooling

    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    # Use an existing construction if it exists and is reasonably long.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 10000)
            best_a = np.maximum(tmp, 0.0)
            if best_a.sum() == 0.0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a /= best_a.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation Adam stage (exploratory)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-12, num=80))
    steps_per_factor = 25  # a bit more work per smoothing level

    adam_deadline = min(deadline, time.time() + budget_s * 0.45)
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform", "best"
    ]
    iteration = 0

    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=steps_per_factor,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb to polish the result of this Adam run
        hc_deadline = min(adam_deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1200
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Exponential sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + budget_s * 0.30)
    if time.time() < subgrad_deadline:
        print("[exp‑subgrad] start")
        best_a, best_score = exponential_subgrad_descent(
            best_a,
            n,
            fft_len_conv,
            fft_len_g,
            subgrad_deadline,
            eta0=0.5,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[exp‑subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[sim‑anneal] start")
        best_a, best_score = simulated_annealing_hc(
            best_a,
            best_score,
            n,
            fft_len_conv,
            deadline,
            init_temp=0.02,
            cooling=0.998,
            max_move=0.12,
        )
        print(f"[sim‑anneal] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the raw sequence – the harness will evaluate the exact score.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
