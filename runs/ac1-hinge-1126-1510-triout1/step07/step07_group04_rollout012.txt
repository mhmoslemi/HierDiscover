def simulated_annealing(a, score, fft_len, n, deadline):
    best_a = a.copy()
    best_score = score
    cur_a = a.copy()
    cur_score = score
    rng = np.random.default_rng()
    # Temperature schedule
    T0 = 0.1 * best_score  # initial temperature relative to score
    Tmin = 1e-6
    # Number of iterations approximate based on remaining time
    while time.time() < deadline:
        # time left
        # compute fraction of elapsed relative to total SA time
        # here we could compute t = (time.time() - start) / total_sa_time
        # and set temperature = T0 * (1 - t) + Tmin
        # propose new a
        # random pair i, j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # propose moving delta from i to j
        # delta up to cur_a[i] * rng.random() * move_factor
        move_factor = 0.2
        delta = cur_a[i] * rng.random() * move_factor
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # compute score
        new_score = fast_score(new_a, fft_len, n)
        if new_score < cur_score:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_a = new_a
                best_score = new_score
        else:
            # acceptance probability
            # Temperature schedule
            # compute temperature based on elapsed time
            # using linear decay
            # or exponential?
            accept_prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < accept_prob:
                cur_a = new_a
                cur_score = new_score
        # update temperature

def simulated_annealing_on_logits(y, score, n, fft_len, deadline):
    rng = np.random.default_rng()
    cur_y = y.copy()
    cur_a = softmax(cur_y)
    cur_score = score
    best_y = y.copy()
    best_score = score
    start = time.time()
    total_time = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        t = elapsed / total_time
        sigma = max(0.01, (1 - t) * 0.5) # gradually reduce
        # propose new y = cur_y + sigma * Normal(0,1)
        y_prop = cur_y + sigma * rng.standard_normal(n)
        a_prop = softmax(y_prop)
        score_prop = fast_score(a_prop, fft_len, n)
        # acceptance
        temp = max(1e-6, (1 - t) * 0.1 * best_score) # etc
        if score_prop < cur_score or rng.random() < np.exp(-(score_prop - cur_score) / temp):
            cur_y = y_prop
            cur_a = a_prop
            cur_score = score_prop
            if score_prop < best_score:
                best_y = y_prop
                best_score = score_prop
    return best_y, best_score

"""Hybrid search for low self‑convolution step function.

The algorithm proceeds in several phases:
1. Multi‑restart smoothed L‑BFGS (continuation) on the log‑sum‑exp
   approximation of the maximal convolution entry.
2. Projected sub‑gradient descent on the true objective.
3. Fine L‑BFGS with very small smoothing.
4. Aggressive hill‑climbing.
5. Simulated‑annealing on the soft‑max logits to escape local minima.

All phases respect the 1000 s wall‑clock budget and keep the
sequence on the probability simplex (non‑negative, sum = 1).  The final
sequence is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        # fallback to uniform distribution
        return np.full_like(y, 1.0 / y.size)
    return e / s

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # numerical noise may produce tiny imaginary parts
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    Implementation follows the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    with respect to the unconstrained logit vector y.

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a * a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z   # size 2n-1

    # reverse convolution to get gradient w.r.t a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # lam argument is not directly used in objective_and_grad;
        # it's passed through lam_factor indirectly via 'lam' variable.
        # Here we simply set lam_factor as a small constant; the schedule
        # influences the smoothing through the lam_factor inside the
        # objective (via lam = max_c * lam_factor). We keep lam_factor
        # constant and let the schedule affect max_c indirectly.
        # For simplicity we keep lam_factor = lam (treated as smoothing).
        # However the original code uses lam_factor = 0.30 and only
        # varies the smoothing through later parameter; we adopt the same
        # pattern but inject lam as a dummy argument to allow SciPy's
        # calling convention.
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Simple greedy Sidon‑set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # Discrete version of the arcsine law: a_i ∝ 1/√(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5          # centre of each bin
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))  # index of maximal entry

        # Sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        # contribution 2*a_j when i!=j, and a_i when i=j (handled by mask)
        # The convolution entry is sum_i a_i * a_{m-i}. Gradient w.r.t a_i is
        # a_{m-i} + a_{m-i} (if i != m-i) => 2*a_{m-i} for distinct pair,
        # and a_i for the diagonal term. Using mask and adding both works:
        g[mask] = 2.0 * a[counterpart[mask]]
        # correct diagonal double counting: when i == counterpart[i]:
        diag = (counterpart == idx)
        g[diag] = a[idx[diag]]

        # Diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def simulated_annealing(
    y_init: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the soft‑max logits.
    The temperature controls the amplitude of Gaussian perturbations
    and the Metropolis acceptance probability.
    """
    rng = np.random.default_rng()
    cur_y = y_init.copy()
    cur_a = softmax(cur_y)
    cur_score = fast_score(cur_a, fft_len, n)
    best_y = cur_y.copy()
    best_score = cur_score
    start = time.time()
    total = max(deadline - start, 1e-6)

    while time.time() < deadline:
        # progress from 0 to 1
        elapsed = time.time() - start
        progress = elapsed / total

        # Temperature for proposals – starts around 0.5 and decays to 0.01
        sigma = (1.0 - progress) * 0.45 + 0.01

        # Metropolis temperature – scales with current best score
        temp = max(1e-8, (1.0 - progress) * 0.12 * best_score)

        # Propose a new point
        y_prop = cur_y + sigma * rng.standard_normal(n)
        a_prop = softmax(y_prop)
        score_prop = fast_score(a_prop, fft_len, n)

        # Accept if better, otherwise with probability exp(-Δ/ temp)
        if score_prop < cur_score or rng.random() < np.exp(-(score_prop - cur_score) / temp):
            cur_y = y_prop
            cur_a = a_prop
            cur_score = score_prop
            if score_prop < best_score:
                best_y = y_prop.copy()
                best_score = score_prop
        # occasional reset to best to avoid drift
        if rng.random() < 0.001:
            cur_y = best_y.copy()
            cur_a = softmax(cur_y)
            cur_score = best_score
    return best_y, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows a multi‑phase scheme
    (continuation L‑BFGS, sub‑gradient descent, fine L‑BFGS,
    aggressive hill‑climb, simulated‑annealing) and returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size n
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    default_n = 5000
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 15000)  # cap to avoid huge FFTs
    n = default_n

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the probability simplex
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp.copy()
            best_a = np.maximum(best_a, 0.0)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

"""Hybrid search for a low self‑convolution step function.

The routine is divided into five phases:

1. Multi‑restart continuation L‑BFGS on a smooth log‑sum‑exp
   approximation of the maximal convolution entry.
2. Projected sub‑gradient descent on the true (non‑smoothed) objective.
3. Fine L‑BFGS refinement with very low smoothing.
4. Aggressive hill‑climb to polish the solution.
5. Simulated‑annealing on the soft‑max logits to escape possible
   local minima.

All stages respect a total wall‑clock budget of 1000 s.  The
sequence is kept on the probability simplex (non‑negative, sum = 1),
because the evaluation function is scale‑invariant.
The function returns the best sequence found as a plain list of
floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fall back to the uniform distribution
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2*n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=1} (Duchi et al.)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max and its gradient w.r.t. y.

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # reverse convolution to obtain ∂c/∂a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a short L‑BFGS for each λ in lam_schedule.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a pre‑softmax vector y for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # simple greedy Sidon set
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def simulated_annealing(
    y_init: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the logit vector y.
    Uses a Gaussian proposal whose scale decays with time, and a
    Metropolis acceptance rule.
    """
    rng = np.random.default_rng()
    cur_y = y_init.copy()
    cur_a = softmax(cur_y)
    cur_score = fast_score(cur_a, fft_len, n)
    best_y = cur_y.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-6)

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total

        # proposal amplitude (sigma) decays from ~0.45 to 0.01
        sigma = (1.0 - progress) * 0.45 + 0.01
        # Metropolis temperature (controls acceptance) decays similarly
        temp = max(1e-8, (1.0 - progress) * 0.12 * best_score)

        y_prop = cur_y + sigma * rng.standard_normal(n)
        a_prop = softmax(y_prop)
        score_prop = fast_score(a_prop, fft_len, n)

        if score_prop < cur_score or rng.random() < np.exp(-(score_prop - cur_score) / temp):
            cur_y = y_prop
            cur_a = a_prop
            cur_score = score_prop
            if score_prop < best_score:
                best_y = y_prop.copy()
                best_score = score_prop

        # occasional reset to the current best to avoid drift
        if rng.random() < 0.001:
            cur_y = best_y.copy()
            cur_a = softmax(cur_y)
            cur_score = best_score

    return best_y, best_score

        "best",
    ]
    phase1_deadline = start_time + 0.55 * budget_s
    iteration = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode}  improved fast score = {best_score:.6f}")

        # brief hill‑climb polishing
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1
        if best_score < 1.504:          # close to target – stop early
            print("[early] target bound reached during Phase 1")
            break

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈30 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.85 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=2_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Fine L‑BFGS refinement (very low smoothing)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] starting final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=30))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=500,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Aggressive hill‑climb (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_hc_deadline = deadline
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=30000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 5 – Simulated annealing (escape local minima)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[sa] simulated annealing")
        best_y, best_score = simulated_annealing(
            best_y,
            n,
            fft_len_conv,
            deadline,
        )
        best_a = softmax(best_y)

    # --------------------------------------------------------------
    # Return the final sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
    return final_seq
