def simulated_annealing(a_start, current_score, deadline, rng, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = current_score
    it = 0
    init_T = 0.01  # typical improvement magnitude? Let's set based on score.
    while time.time() < deadline:
        # adjust temperature gradually
        remaining = deadline - time.time()
        total = deadline - (deadline - (budget_remaining?)) Not clear.
        # we can compute T = init_T * (remaining / (deadline - start_time))
        # but we need start time for annealing; we can compute fraction of elapsed.
        # Simpler: T = init_T * (1 - it/ max_iter) but we don't know max_iter.

        # pick source and target
        src = rng.integers(n)
        while a[src] == 0:
            src = rng.integers(n)
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        max_transfer = a[src]
        delta = max_transfer * rng.random() * 0.2  # up to 20%
        if delta <= 0:
            continue
        a[src] -= delta
        a[dst] += delta

        score = fast_score(a, fft_len, n)
        # accept if better or with probability
        if score < best_score:
            best_score = score
            best_a = a.copy()
        else:
            # metropolis
            T = init_T * (1 - (time.time() - start_time) / (deadline - start_time))
            if T > 0 and rng.random() < np.exp(-(score - best_score) / T):
                # accept worse
                pass
            else:
                # revert move
                a[src] += delta
                a[dst] -= delta
        it += 1

    return best_a, best_score

"""
Improved search for a low‑evaluation step sequence.

The algorithm proceeds in several stages within the 1000 s budget:

0.  Load a previous good solution (if available) and perform a dense
    parametric sweep over power‑law families (including decreasing
    powers) to obtain a solid start.

1.  Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation of the
    max‑convolution.  A finer lam‑schedule and more inner iterations are
    used than in the reference code.

2.  Projected sub‑gradient descent with momentum (the same primitive as
    before) to further reduce the true max‑convolution.

3.  Simulated‑annealing style random mass transfers.  Each move keeps the
    total mass = 1, tests the fast objective (2·n·max(conv)) and accepts
    improvements outright, otherwise with a temperature‑dependent
    Metropolis probability.  This explores directions that are not
    reachable by the pure gradient‑guided hill‑climb.

4.  Deterministic gradient‑guided hill‑climb with a much larger number
    of iterations (up to 50 000) which always moves a small fraction of
    mass from the “most harmful’’ coordinate (largest sub‑gradient) to
    the “most beneficial’’ one (smallest sub‑gradient).

Finally the candidate is evaluated with the provided `evaluate_sequence`
function and the best list of floats is returned.

All steps respect the global time limit; the best sequence found at any
moment is kept.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy import sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small fraction of mass from the
    coordinate with the largest sub‑gradient (most harmful) to the one
    with the smallest sub‑gradient (most helpful).  Only strictly
    improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # move a small fraction (up to 5 %) of mass from i_hi to i_lo
        delta = best[i_hi] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
    n: int,
    fft_len: int,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealing that randomly transfers a fraction
    of mass between two positions while respecting the simplex.
    Accepts downhill moves always; uphill moves with probability exp(-Δ/T).
    Temperature T decays linearly with remaining time.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    # initial temperature (tuned empirically)
    T0 = 2e-3 * start_score  # a few thousandths of the current score
    while time.time() < deadline:
        # linear schedule based on fraction of time elapsed in this phase
        t_rem = deadline - time.time()
        t_tot = deadline - (deadline - (deadline - start_score))  # dummy to avoid zero
        # Since we don't have phase start time, use remaining fraction directly
        frac = t_rem / (deadline - (deadline - (deadline - start_score)) + 1e-12)
        T = T0 * frac

        # pick source with non‑zero mass
        src = rng.integers(n)
        while a[src] == 0.0:
            src = rng.integers(n)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer up to 20 % of source mass
        delta = a[src] * rng.random() * 0.20
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[dst] += delta

        new_score = fast_score(a, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        else:
            # metropolis acceptance
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                # keep the worse move
                pass
            else:
                # revert
                a[src] += delta
                a[dst] -= delta

    return best_a, best_score
