def guided_hill_climb(a, cur_score, fft_len, n, deadline, max_iters=20000, move_frac=0.1):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters >0:
        max_iters -=1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        # compute contribution weights for each index
        # w_i = a_i * a_{m-i} if m-i in range else 0
        indices = np.arange(n)
        counterpart = m - indices
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]
        # choose donor i with probability proportional to contrib * (maybe a_i)
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            break
        # donor selection: sample i with weight contrib
        i = rng.choice(n, p=contrib/total_contrib)
        # find potential recipients j that have low contribution
        # compute new contributions if moved to j (very expensive). Instead sample j uniformly.
        j = rng.integers(0,n)
        if i == j:
            continue
        # move some mass from i to j
        delta = best[i] * rng.random() * move_frac
        if delta <= 0:
            continue
        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative (i may become zero)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            # keep for later loops
    return best, best_score
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    n = 15000  # choose large n to reduce constant
    fft_len_conv = next_pow2(2*n - 1)
    fft_len_g = next_pow2(3*n - 2)

    # Phase0: evaluate baseline distributions
    best_a = None
    best_score = np.inf

    # Uniform
    a_uniform = np.full(n, 1.0/n)
    score = fast_score(a_uniform, fft_len_conv, n)
    best_a, best_score = a_uniform, score

    # Arcsine
    a_arcsine = arcsine_distribution(n)
    score = fast_score(a_arcsine, fft_len_conv, n)
    if score < best_score:
        best_a, best_score = a_arcsine, score

    # Beta family: loop alpha values
    alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    for alpha in alphas:
        a_beta = beta_distribution(n, alpha)
        sc = fast_score(a_beta, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a_beta, sc

    # Mix arcsine and uniform via lambda grid
    for lam in np.linspace(0.0, 1.0, 11):
        a_mix = lam * a_arcsine + (1 - lam) * a_uniform
        a_mix /= a_mix.sum()
        sc = fast_score(a_mix, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a_mix, sc

    # Also maybe mix arcsine with betaAlpha (some)
    for alpha in alphas:
        a_beta = beta_distribution(n, alpha)
        for lam in [0.25, 0.5, 0.75]:
            a_mix = lam * a_arcsine + (1 - lam) * a_beta
            a_mix /= a_mix.sum()
            sc = fast_score(a_mix, fft_len_conv, n)
            if sc < best_score:
                best_a, best_score = a_mix, sc

    # Logging
    print(f"[init] n={n} baseline best fast score = {best_score:.6f}")

    # Phase 1: L-BFGS restarts
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=50))
    max_restarts = 12
    iteration = 0
    while time.time() < deadline * 0.55 and iteration < max_restarts:
        # initial y: log of best + small perturbation
        y0 = np.log(best_a + 1e-12)
        y0 += rng.normal(scale=0.01, size=n)
        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=300)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            print(f"[L-BFGS {iteration}] improved fast score = {best_score:.6f}")
        iteration += 1

    # Phase 2: projected subgradient descent
    subgrad_deadline = deadline - 5.0
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(best_a, fft_len_conv, n, subgrad_deadline, eta0=0.4, max_iters=500000)
        print(f"[subgrad] end fast score = {best_score:.6f}")

    # Phase 3: guided hill climb
    if time.time() < deadline:
        print("[hill] aggressive climb")
        best_a, best_score = guided_hill_climb(best_a, best_score, fft_len_conv, n, deadline, max_iters=40000, move_frac=0.15)
        print(f"[hill] final fast score = {best_score:.6f}")

    # Final evaluate
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

def arcsine_distribution(n):
    idx = np.arange(n) + 0.5
    x = idx / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-15)
    a /= a.sum()
    return a
```

Beta distribution with parameter alpha (both shape parameters same) i.e., a_i ∝ x^{alpha-1} (1-x)^{alpha-1}. Using x for i+0.5/n.

```
def beta_distribution(n, alpha):
    idx = np.arange(n) + 0.5
    x = idx / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    a += 1e-30  # avoid zero
    a /= a.sum()
    return a
```

"""
Propose a step‑function with a low evaluation score.

The search works on the probability simplex (∑ a_i = 1) because the
objective 2·n·max(conv) is scale‑invariant.  The algorithm consists of:

1.  Build a set of simple analytic seeds (uniform, arcsine, Beta(α,α),
    a DPSS window).  Evaluate each with a fast FFT‑based score and keep
    the best one.

2.  Refine the current best seed with several multi‑restart L‑BFGS
    optimisations on a smooth log‑sum‑exp surrogate of the max‑convolution.
    The smoothing parameter is reduced geometrically (continuation).

3.  Projected sub‑gradient descent on the true non‑smooth objective
    (the max‑convolution) to escape any remaining bias introduced by the
    smoothing.

4.  A directed greedy hill‑climb that repeatedly moves a small fraction
    of the mass from indices that contribute heavily to the current
    maximal convolution entry to indices that contribute little.

The whole procedure respects the 1000 s budget, using a relatively large
length n≈15 000 (the larger the n the closer the asymptotic constant is to
its theoretical optimum).  The final sequence is returned as a plain list
of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a² = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (softmax parametrisation).

    lam_factor multiplies the current max‑convolution value to obtain the
    smoothing constant lam.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c            # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law a_i ∝ 1/√(x·(1‑x))."""
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    eps = 1e-15
    a = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) law: a_i ∝ x^{α‑1}(1‑x)^{α‑1}."""
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a += 1e-30          # avoid pure zeros
    a /= a.sum()
    return a


def dpss_distribution(n: int, nw: float = 2.5) -> np.ndarray:
    """Discrete Prolate Spheroidal Sequence (Slepian, order 0)."""
    try:
        from scipy.signal import windows
        a = windows.dpss(n, NW=nw, Kmax=1, sym=True)[0]
    except Exception:
        # fallback to uniform if DPSS not available
        a = np.full(n, 1.0 / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Local search helpers
# ----------------------------------------------------------------------


def guided_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 20000,
    move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves mass away from indices
    heavily contributing to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        # compute convolution and the maximal index
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contribution of each index to this maximal entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]

        total = contrib.sum()
        if total <= 0:
            # nothing to improve – break early
            break

        # choose a donor proportional to its contribution
        donor = rng.choice(n, p=contrib / total)

        # choose a receiver uniformly among indices that are *not* the donor
        rec = rng.integers(0, n)
        if rec == donor:
            continue

        # move a small random fraction of the donor's mass
        delta = best[donor] * rng.random() * move_frac
        if delta <= 0:
            continue

        candidate = best.copy()
        candidate[donor] -= delta
        candidate[rec] += delta

        new_score = fast_score(candidate, fft_len, n)
        if new_score < best_score:
            best = candidate
            best_score = new_score
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.35,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv)  (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows a four‑phase scheme and
    respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use a large n to get a better asymptotic constant
    # ------------------------------------------------------------------
    n = 15000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 0 – evaluate a few analytic seeds
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform
    best_score = fast_score(best_a, fft_len_conv, n)

    # arcsine
    a_arc = arcsine_distribution(n)
    sc_arc = fast_score(a_arc, fft_len_conv, n)
    if sc_arc < best_score:
        best_a, best_score = a_arc, sc_arc

    # beta family
    beta_alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    for a in (beta_distribution(n, a) for a in beta_alphas):
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc

    # DPSS (if available)
    a_dpss = dpss_distribution(n, nw=2.5)
    sc_dpss = fast_score(a_dpss, fft_len_conv, n)
    if sc_dpss < best_score:
        best_a, best_score = a_dpss, sc_dpss

    # Mixtures of arcsine and uniform
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * a_arc + (1 - lam) * best_a
        mix /= mix.sum()
        sc = fast_score(mix, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = mix, sc

    print(f"[init] n={n}  best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart continuation L‑BFGS (≈50 % of budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.6, 1e-9, num=45))
    max_restarts = 12
    restarts_done = 0
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline and restarts_done < max_restarts:
        # initialise y (pre‑softmax) with a small random perturbation
        y0 = np.log(best_a + 1e-12)
        y0 += rng.normal(scale=0.01, size=n)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=500,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            print(f"[L‑BFGS {restarts_done}] improved fast score = {best_score:.6f}")

        restarts_done += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.87 * budget_s)
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            subgrad_deadline,
            eta0=0.35,
            max_iters=800_000,
        )
        print(f"[subgrad] finished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive guided hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[hill] aggressive guided climb")
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iters=40_000,
            move_frac=0.12,
        )
        print(f"[hill] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
