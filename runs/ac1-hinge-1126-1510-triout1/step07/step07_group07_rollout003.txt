"""
Propose a low‑evaluation step function.

Algorithm outline
-----------------
* Work on the probability simplex (∑a_i = 1).  The objective used by
  ``evaluate_sequence`` is scale‑invariant, so only the shape of the
  distribution matters.
* Generate a rich set of symmetric analytic shapes (Beta(p,p) with a
  dense grid for p∈(0.5,0.9) and mixtures with the uniform distribution).
  Evaluate them quickly with an FFT‑based convolution and keep the best
  few.
* From each promising seed run a continuation L‑BFGS optimisation on a
  smoothed log‑sum‑exp surrogate of the max‑convolution.  The smoothing
  parameter λ is reduced geometrically (≈28 stages).  This gives a very
  accurate local optimum.
* Polish the L‑BFGS result with a long projected sub‑gradient descent
  on the true max‑convolution objective.  The sub‑gradient is taken
  only from the indices that attain the current maximum; after each
  step we project back onto the simplex and enforce symmetry
  a_i = a_{n‑1‑i}.
* Finally perform aggressive random / targeted hill‑climbs (mass transfer
  moves) while the time budget permits.
* The best sequence found is returned as a plain list of non‑negative
  floats.  All helper functions are top‑level (no closures) as required.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a must be length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the surrogate objective for a probability vector
    (∑a = 1):    2·n·max(conv(a,a))
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of 2·n·max(conv(a,a)).
    Parameters
    ----------
    y : ndarray
        Unconstrained logits (softmax yields a probability vector).
    n : int
        Length of a.
    fft_len_conv : int
        Power‑of‑two length for convolution FFT.
    fft_len_g : int
        Power‑of‑two length for the auxiliary correlation FFT.
    lam_factor : float
        λ = max(conv) * lam_factor (smoothing constant).
    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : ndarray
        Gradient w.r.t. y.
    """
    a = softmax(y)

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing of the max
    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z                    # shape: (2*n‑1,)

    # correlation of a with w gives ∂conv/∂a (factor 2 because conv is a*a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage solves a smooth problem with a fixed λ‑factor.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Analytic families of symmetric initialisations
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density discretised on {0,…,n‑1} and normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle (U‑shaped) on {0,…,n‑1}."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)            # ensure positivity
    return a / a.sum()


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return a / a.sum()


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, then re‑normalised."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

# Local search primitives
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        # keep symmetry – the move on the mirrored indices mirrors automatically
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    best = symmetrize(best)
    best = project_onto_simplex(best)
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from the current max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    best = symmetrize(best)
    best = project_onto_simplex(best)
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true (non‑smooth) objective.
    Uses the exact sub‑gradient that stems from the current maximal
    convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient: only the terms that contribute to the maximal entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The implementation follows the plan described in
    the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Prefer to keep the length of any historic solution, otherwise use
    # a reasonably large n (≈ 8000) which provides plenty of expressive
    # power without being too slow.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # 1) Scan a dense family of symmetric Beta(p,p) shapes, also mix them
    #    with the uniform distribution.  Keep the top few for full
    #    optimisation.
    # ------------------------------------------------------------------
    candidates = []

    # Dense grid for p
    p_grid = np.linspace(0.51, 0.90, 80)
    mix_weights = [0.0, 0.25, 0.5, 0.75, 1.0]        # 0 → uniform, 1 → pure beta
    for p in p_grid:
        beta = beta_init(n, p)
        for w in mix_weights:
            a = mixture_init(beta, uniform_init(n), w)
            a = symmetrize(a)
            a = project_onto_simplex(a)
            name = f"beta_{p:.3f}_mix_{w:.2f}"
            candidates.append((name, a))

    # Add a few hand‑crafted families for diversity
    candidates.append(("semicircle", symmetrize(semicircle_init(n))))
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("random_gauss", symmetrize(random_gaussian_init(n, rng))))
    candidates.append(("binary_sparse", symmetrize(binary_sparse_init(n, rng, density=0.15))))
    candidates.append(("gaussian_bump", symmetrize(gaussian_bump_init(n, rng))))

    # If a historic sequence exists, reuse it (after normalisation)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            a = symmetrize(a)
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Quick evaluation of all candidates (fast_score) and keep the best K.
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:28s} fast score = {sc:.6f}")

    # Keep only the most promising seeds (K = 6)
    scored.sort()
    top_k = 6
    best_a = None
    best_score = np.inf
    best_y = None

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the top seeds.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-9, num=28))
    maxiter_per_stage = 600
    phase_deadline = start_time + 0.45 * budget_s
    print("[phase] L‑BFGS continuation started")

    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        a_opt = symmetrize(a_opt)
        a_opt = project_onto_simplex(a_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:28s} score = {best_score:.6f}")

        # short hill‑climb polish while we still have time in this phase
        if time.time() < phase_deadline:
            bc, bs = hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline=phase_deadline,
                max_iter=1500,
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] post‑hill‑climb improved score = {best_score:.6f}")

        if time.time() >= phase_deadline:
            print("[phase] L‑BFGS time exhausted")
            break

    # ------------------------------------------------------------------
    # 3) Random‑restart continuation (exploratory) for the remaining budget.
    # ------------------------------------------------------------------
    random_restart_deadline = start_time + 0.70 * budget_s
    restart_counter = 0
    while time.time() < random_restart_deadline:
        restart_counter += 1
        a0 = symmetrize(random_gaussian_init(n, rng))
        y0 = np.log(a0 + 1e-12)
        # a shorter schedule for quick restarts
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule[:12],
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        a_opt = symmetrize(a_opt)
        a_opt = project_onto_simplex(a_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] #{restart_counter:03d} improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent – the heavy workhorse.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 15.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.35,
        max_iters=600_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Aggressive final polishing (random + targeted hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=20000,
            max_move=0.30,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=12000,
                max_move=0.20,
            )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 6) Final evaluation with the provided evaluate_sequence (exact score)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)      # type: ignore
        print(f"[final] exact evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
        final_val = None

    # Return the best sequence we have found.
    return final_seq
