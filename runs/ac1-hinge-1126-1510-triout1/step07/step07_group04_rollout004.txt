"""
Hybrid random‑search + projected sub‑gradient descent for the step‑function
minimisation problem.

The algorithm works on the probability simplex (Σa = 1) because the
objective 2·n·max(conv) is scale‑invariant.  The search consists of three
phases:

1.  Randomised parametrised families (arc‑sine–like, power laws and a
    mixture with the uniform distribution) are sampled for a range of
    lengths n.  The best candidate seen so far is kept.

2.  A projected sub‑gradient method is run on the best candidate.  The
    sub‑gradient of the maximal convolution entry is cheap to compute:
    for the index m where conv[m] is maximal the gradient is
    g_i = 2·a_{m−i} (zero when the index is out of range).  The update
    uses a diminishing step‑size η_t = η₀/√t and the Euclidean projection
    onto the simplex.

3.  An aggressive greedy hill‑climb moves a random fraction of mass from
    one index to another while accepting only improving steps.  This
    polishing phase uses the remaining time budget.

All heavy work (convolutions) is performed with NumPy’s FFT based on
`rfft`/`irfft`, which is fast for the problem sizes (n up to ~12 000) and
allows thousands of iterations within the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*len(a)-1) using real FFT.
    a is assumed 1‑D and length n.
    """
    # rfft on real input is roughly twice as fast as full complex FFT
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)   (the denominator sum_a**2 = 1).
    """
    n = a.shape[0]
    conv = conv_fft(a, fft_len)
    return float(2.0 * n * np.max(conv))


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Local optimisation helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    deadline: float,
    max_iter: int = 10_000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction of mass from i to j.
    Only accepting moves that strictly improve the fast_score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # fraction of the mass at i that we try to transfer
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).

    The sub‑gradient at the maximal entry m is:
        g_i = 2·a_{m−i}  if 0 ≤ m−i < n, else 0.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute the convolution and locate the maximal entry
        conv = conv_fft(a, fft_len)
        m = int(np.argmax(conv))

        # build the sub‑gradient
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

        # pre‑compute FFT length for this n
        fft_len = next_pow2(2 * n - 1)

        # ---- generate a candidate distribution ----
        # Discrete variable x in (0,1) at the centre of each bin
        idx = np.arange(n, dtype=np.float64) + 0.5
        x = idx / n

        # Exponent p controls the edge‑heaviness (p<0 → more weight at edges)
        p = rng.uniform(-0.70, 0.10)

        # Shape component (power‑law on x·(1−x))
        eps = 1e-15
        shape = (x * (1.0 - x) + eps) ** p

        # Mix with the uniform distribution; λ close to 1 keeps the shape
        lam = rng.random()
        a = lam * shape + (1.0 - lam) * np.ones_like(shape)

        # Normalise to a probability vector (sum = 1)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum

        # ---- evaluate ----
        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_n = n
            print(f"[phase1] new best: n={n}  score={best_score:.6f}")

    if best_a is None:
        # fallback – uniform distribution
        n = default_n
        best_n = n
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, next_pow2(2 * n - 1))
        print("[phase1] fallback to uniform distribution")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 30.0, start_time + 0.85 * budget_s)
    print("[phase2] projected sub‑gradient descent start")
    fft_len = next_pow2(2 * best_n - 1)
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len,
        subgrad_deadline,
        eta0=0.30,
        max_iters=1_500_000,
    )
    print(f"[phase2] finished, best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase3] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        print(f"[phase3] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Prepare and return the final sequence
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # Official evaluation (the harness provides evaluate_sequence)
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
