        v = beta2 * v + (1 - beta2) * (grad ** 2)

def precise_hill_climb(a, current_score, fft_len, n, deadline):
    best = a.copy()
    best_score = current_score
    while time.time() < deadline:
        # compute subgradient
        conv = full_conv(best, fft_len)
        max_val = conv.max()
        mask = (conv >= max_val - eps)  # could be multiple
        # compute gradient similarly but for each index
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n-1):(n-1+n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # binary search for max delta
        low = 0.0
        high = best[i_hi]  # can't exceed
        # ensure we have at least some improvement at delta=high? maybe not.
        # We'll search for best delta that reduces max conv.

        for _ in range(12):
            mid = (low + high) / 2.0

            a_tmp[i_lo] += mid

def simulated_annealing(a, current_score, fft_len, n, deadline):
    best = a.copy()
    best_score = current_score
    cur = a.copy()
    cur_score = current_score
    rng = np.random.default_rng()
    T0 = 0.01  # initial temperature relative to score
    while time.time() < deadline:
        t = (deadline - time.time()) / (deadline - start_time)  # fraction remaining?

        T = T0 * t
        # propose move: randomly choose i,j and delta fraction
        i = rng.integers(n)
        j = rng.integers(n-1)
        if i == j: continue
        delta = cur[i] * rng.random() * 0.05
        if delta == 0: continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        sc = fast_score(new, fft_len, n)
        if sc < cur_score or rng.random() < np.exp((cur_score - sc) / max(T, 1e-12)):
            cur = new
            cur_score = sc
            if sc < best_score:
                best = new.copy()
                best_score = sc
    return best, best_score

"""
Propose a low‑evaluation step sequence.

The algorithm explores the probability simplex (∑a = 1) with several
search phases:

  0. A quick parametric sweep (power‑law, exponential, sinusoid) to get a
     decent starter.
  1. Many random‑restart L‑BFGS optimisations on a smooth log‑sum‑exp
     approximation of the maximum convolution.
  2. Projected sub‑gradient descent with momentum (true max‑convolution).
  3. Adam optimisation on the smooth objective (more stable than plain
     L‑BFGS for long runs).
  4. A deterministic hill‑climb that moves mass from the coordinate with
     the largest sub‑gradient to the one with the smallest, using a binary
     line‑search to choose the optimal step size.
  5. A short simulated‑annealing fine‑tuning pass.

All phases respect a global wall‑clock budget (default 1000 s).  The best
sequence found (as a plain ``list``) is returned; its true score is printed
using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a distribution with sum 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a distribution (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Random initialiser utilities
# ----------------------------------------------------------------------


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "sin"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sin‑wave style
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.abs(np.sin(freq * np.arange(n) / n * np.pi + phase)) + 0.01
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ∝ 1/√t
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed log‑sum‑exp objective.
    """
    eps = 1e-12
    y = np.log(a0 + eps)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    # geometric schedule for the smoothing factor
    lam_sched = np.geomspace(0.6, 0.001, num=30)

    while time.time() < deadline:
        lam = lam_sched[it % len(lam_sched)]
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        # learning rate slowly decays with the iteration count
        lr = 0.6 / (np.sqrt(it + 1) + 1)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 10 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        it += 1

    return best_a, best_score


def precise_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves mass from the coordinate with the
    largest sub‑gradient to the one with the smallest, using a binary search
    for the optimal transfer amount.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the true max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # binary search for the best delta (up to 100 % of a[i_hi])
        lo, hi = 0.0, best[i_hi]
        best_local = best_score
        best_delta = 0.0

        # 12 iterations give sub‑machine precision for our purposes
        for _ in range(12):
            mid = (lo + hi) / 2.0
            trial = best.copy()
            trial[i_hi] -= mid
            trial[i_lo] += mid
            sc = fast_score(trial, fft_len, n)
            if sc < best_local:
                best_local = sc
                best_delta = mid
                lo = mid          # we can try moving even more
            else:
                hi = mid

        if best_delta > 0:
            best[i_hi] -= best_delta
            best[i_lo] += best_delta
            best_score = best_local
        else:
            # no improvement found – make a tiny random jiggle to escape
            i1, i2 = rng.integers(0, n, size=2)
            if i1 != i2 and best[i1] > 1e-12:
                delta = best[i1] * rng.random() * 0.02
                trial = best.copy()
                trial[i1] -= delta
                trial[i2] += delta
                sc = fast_score(trial, fft_len, n)
                if sc < best_score:
                    best = trial
                    best_score = sc

        it += 1

    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    A short simulated‑annealing pass that performs random mass transfers,
    accepting worsening moves with a temperature that decays to zero.
    """
    best = a.copy()
    best_score = current_score
    cur = a.copy()
    cur_score = current_score
    rng = np.random.default_rng()

    start = time.time()
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        T = max(1e-6, 0.02 * (1.0 - elapsed / total))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue

        delta = cur[i] * rng.random() * 0.03
        trial = cur.copy()
        trial[i] -= delta
        trial[j] += delta

        sc = fast_score(trial, fft_len, n)
        delta_score = sc - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            cur = trial
            cur_score = sc
            if sc < best_score:
                best = trial.copy()
                best_score = sc

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)          # for a*a
    fft_len_g = next_pow2(3 * n - 2)        # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate (uniform or previous best)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    for p in np.linspace(0.0, 3.0, 13):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        a = (np.arange(n, 0, -1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    for freq in [0.8, 1.5, 2.2]:
        phase = np.abs(np.sin(freq * np.arange(n) / n * np.pi)) + 0.01
        phase /= phase.sum()
        sc = fast_score(phase, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, phase.copy()
            print(f"[sweep] sin freq {freq:.2f} -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety gap

    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.95 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – many random‑restart L‑BFGS continuations
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        if rng.random() < 0.2:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.012), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a, phase3_end, fft_len, fft_len_g, n
        )
        print(f"[Adam] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic gradient‑guided hill‑climb with line search
    # ------------------------------------------------------------------
    best_a, best_score = precise_hill_climb(
        best_a, best_score, fft_len, n, deadline, max_iter=20000
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – short simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len, n, deadline
        )
        print(f"[simulated‑anneal] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
