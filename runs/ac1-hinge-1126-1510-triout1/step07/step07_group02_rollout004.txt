"""
Optimiser for the step‑function challenge.

Algorithm
---------
1. **Random parametric search** – generate many candidates from a few
   simple families (exponential, power‑law, Gaussian, bimodal Gaussian,
   edge‑heavy, fully random).  Each is normalised to a probability vector
   and scored with a fast FFT‑based convolution.  The best candidate is
   kept for the next stage.

2. **Adam optimisation on a smooth max‑convolution** – minimise a
   log‑sum‑exp approximation of the max‑convolution using the Adam
   optimiser on the logits of the soft‑max (so the simplex constraint is
   satisfied automatically).  A continuation on a decreasing smoothing
   factor (λ) drives the solution towards the true objective.

3. **Exact exponentiated‑gradient descent** – from the Adam result perform
   standard sub‑gradient descent on the hard max‑convolution using an
   exponentiated‑gradient (multiplicative) update which stays on the
   simplex.

4. **Greedy hill‑climbing** – final polishing by moving a small random
   fraction of mass between coordinates, accepting strictly improving
   moves.

All stages respect the global time budget (default 1000 s).  The code
uses only NumPy and SciPy and runs on a single thread (the FFTs are
vectorised).  The returned sequence is a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize  # kept for compatibility, not used directly


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective 2·n·max(a*a) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def candidate_exponential(rng: np.random.Generator, n: int) -> np.ndarray:
    r = rng.uniform(0.001, 0.999)
    a = np.power(r, np.arange(n, dtype=float))
    a /= a.sum()
    return a


def candidate_powerlaw(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(0.05, 4.0)
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)
    a /= a.sum()
    return a


def candidate_gaussian(rng: np.random.Generator, n: int) -> np.ndarray:
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.01 * n, 0.30 * n)
    xs = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def candidate_bimodal_gaussian(rng: np.random.Generator, n: int) -> np.ndarray:
    mu1 = rng.uniform(0, n - 1)
    mu2 = rng.uniform(0, n - 1)
    sigma1 = rng.uniform(0.01 * n, 0.20 * n)
    sigma2 = rng.uniform(0.01 * n, 0.20 * n)
    w = rng.uniform(0.0, 1.0)
    xs = np.arange(n, dtype=float)
    a = (w * np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2) +
         (1.0 - w) * np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2))
    a /= a.sum()
    return a


def candidate_edge_heavy(rng: np.random.Generator, n: int) -> np.ndarray:
    edge_frac = rng.uniform(0.001, 0.05)
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a /= a.sum()
    return a


def generate_random_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Pick one of the parametric families uniformly at random."""
    families = [
        candidate_exponential,
        candidate_powerlaw,
        candidate_gaussian,
        candidate_bimodal_gaussian,
        candidate_edge_heavy,
        random_initial,
    ]
    f = rng.choice(families)
    return f(rng, n)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – used by Adam/L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient pieces that correspond to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  max_iter_per_lambda: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimisation on the log‑sum‑exp smoothed max‑convolution.
    Returns the best vector found and its fast_score.
    """
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        # initialise logits from the current best
        y = np.log(best_a + 1e-12)
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        # a modest learning rate; it will be scaled automatically by Adam
        lr = 0.2

        for _ in range(max_iter_per_lambda):
            if time.time() >= deadline:
                break
            f, grad_y = objective_and_grad(y, n,
                                           fft_len_conv, fft_len_g,
                                           lam_factor=lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = -lr * m_hat / (np.sqrt(v_hat) + eps)
            y += step

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (hard max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on the true max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size: decreasing like 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10  # up to 10 % of source
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search consists of:
        * a rapid random‑parametric exploration,
        * Adam optimisation on a smoothed maximum,
        * exact exponentiated‑gradient descent,
        * a final greedy hill‑climb.

    The function returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and base initialisation
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # If a previous construction exists, interpolate it to the current size
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget splits (includes a tiny safety margin)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    phase0_end = start_time + 0.15 * budget_s   # random parametric search
    phase1_end = start_time + 0.70 * budget_s   # Adam optimisation
    phase2_end = start_time + 0.90 * budget_s   # exponentiated gradient
    # remaining time -> hill‑climb

    # ------------------------------------------------------------------
    # Phase 0 – random parametric search
    # ------------------------------------------------------------------
    while time.time() < phase0_end:
        cand = generate_random_candidate(rng, n)
        score = fast_score(cand, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[parametric] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation with decreasing λ
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    max_iter_per_lambda = 2000
    best_a, best_score = adam_optimize(
        best_a,
        lam_schedule,
        max_iter_per_lambda,
        phase1_end,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb (remaining budget)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=40000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
