"""
Optimise a non‑negative step function to minimise
    2·n·max(a*a) / (sum a)²

The algorithm combines many simple optimisation ideas:
  * Randomised simple‑x initialisations (Uniform, exponential, power, etc.).
  * A projected Adam‑style gradient descent on a *smooth* max‑convolution.
    The smoothing (log‑sum‑exp) is gradually reduced so the optimiser first
    sees a friendly landscape and then hones in on the true ℓ∞‑objective.
  * Heavy use of FFTs for fast convolutions and gradient computation.
  * A final deterministic, gradient‑guided hill‑climb that moves mass from
    indices with a large sub‑gradient to those with a small one.
  * Very light‑weight bookkeeping and early‑stop checks to stay well under the
    1000 s time budget.

The search is performed on the simplex (a_i ≥ 0, Σa_i = 1) by projecting
after each update, which guarantees that the final candidate can be fed
directly to ``evaluate_sequence`` without further post‑processing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence already on the simplex.
    Returns 2·n·max(a*a) .
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers and noise utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.005, 0.02)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max of the convolution and its gradient w.r.t. a
# ----------------------------------------------------------------------


def smooth_max_and_grad_a(
    a: np.ndarray,
    lam: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(a*a) using the log‑sum‑exp trick
    and return the gradient with respect to a (not the log‑itself).
    """
    # convolution a*a
    conv = full_conv(a, fft_len_conv)

    # log‑sum‑exp smoothing
    max_c = conv.max()
    shifted = (conv - max_c) / lam          # ≤ 0
    exp_shift = np.exp(shifted)            # between 0 and 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # derivative of smooth_max w.r.t. the convolution entries
    w = exp_shift / Z                       # shape (2n-1,)

    # back‑propagation through the convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

    return smooth_max, grad_a


# ----------------------------------------------------------------------
# Projected Adam‑style optimisation on the simplex
# ----------------------------------------------------------------------


def projected_adam(
    a_start: np.ndarray,
    lam_schedule: list[float],
    steps_per_lam: list[int],
    lr: float,
    beta1: float,
    beta2: float,
    epsilon: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex using the smoothed max‑convolution.
    Returns the best a seen and its fast_score.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    total_iters = sum(steps_per_lam)

    for lam, steps in zip(lam_schedule, steps_per_lam):
        for _ in range(steps):
            if time.time() >= deadline:
                return best_a, best_score

            _, grad = smooth_max_and_grad_a(
                a, lam, fft_len_conv, fft_len_g, n
            )

            # Adam update (on a directly)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

            a_candidate = simplex_projection(a - step)

            cur_score = fast_score(a_candidate, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_candidate.copy()

            a = a_candidate
            it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (deterministic, only improving moves)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb guided by the sub‑gradient of the current max‑convolution.
    Moves a modest fraction of mass from high‑gradient to low‑gradient
    indices, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        # current convolution and mask of active (max‑tight) indices
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (size n)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source / target via percentile‑based tails
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * 0.07  # up to 7 %
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score

        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns a list of floats (the height sequence) before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives a better theoretical bound,
    # but the optimisation becomes slower.  4000 is a good compromise.
    # --------------------------------------------------------------
    n = 4000

    # If a previous construction exists, up‑sample it; otherwise start uniform.
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # 80 % of the time is spent in the Adam‑style optimisation,
    # the remaining 20 % on the final hill‑climb.
    adam_deadline = start_time + 0.80 * budget_s

    # ------------------------------------------------------------------
    # Adam optimisation – several random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.008, 0.004, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05
    ]
    steps_per_lam = [2000] * len(lam_schedule)   # uniform budget per level

    while time.time() < adam_deadline:
        # Occasionally perturb the current best to explore a neighbourhood.
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = projected_adam(
            a_start,
            lam_schedule,
            steps_per_lam,
            lr=0.12,                # learning rate (tuned empirically)
            beta1=0.9,
            beta2=0.999,
            epsilon=1e-8,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
            deadline=adam_deadline,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final targeted gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        final_deadline,
        max_iter=120000,
    )
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Evaluate with the provided external function and return the list.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
