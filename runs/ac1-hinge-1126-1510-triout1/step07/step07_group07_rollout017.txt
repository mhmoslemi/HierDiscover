"""
Propose a low‑evaluation step‑function.

The objective is 2·n·max(conv(a,a)) with a normalized to sum = 1.
If a_i ∝ (i+½)^p (a power‑law on {0,…,n‑1}) then the
continuous limit gives a closed‑form peak
g_max = Γ(p+2)²/Γ(2p+2).  The discrete peak is
max{g_max/n, a_max²}.  By scanning a small set of integer exponents
we can find a distribution whose max‑convolution is far below the
previous best (~1.5).  After the scan a tiny random hill‑climb is
performed while time permits.

The routine works for any existing ``height_sequence_1`` – it re‑uses
its length (otherwise defaults to 5 000).  It returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) for a probability vector
    (∑ a_i = 1).  `fft_len` must be ≥ 2·len(a)-1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    with a brief random mass‑transfer hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Determine the problem size (use previous length if reasonable)

    init_seq = globals().get("height_sequence_1", None)

            n = 5000

    # FFT length for the convolutions (next power of two ≥ 2·n‑1)
    fft_len = next_pow2(2 * n - 1)

    print(f"[init] using n = {n}, fft_len = {fft_len}")

    # --------------------------------------------------------------
    # Scan a few integer exponents p and keep the best one.
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    best_p = None

    candidate_ps = list(range(1, 11))          # 1 … 10
