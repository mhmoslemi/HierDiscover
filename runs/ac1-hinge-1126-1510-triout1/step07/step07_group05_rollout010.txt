def grad_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000):
    # Compute gradient of max conv: 
    while time.time() < deadline and max_iter>0:
        # compute conv and active set
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        # active indices within tolerance
        tol = max(1e-12, (max_val - conv.mean()) * 1e-9)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0/active.size
        # compute gradient
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g) # store precomputed?
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n-1):(n-1+n)]
        grad = 2.0 * g_vec # gradient of max conv
        # find source idx with largest grad* a_i (maybe product)
        # choose source among indices with a_i > eps
        source_candidates = np.where(a > 1e-12)[0]
        if source_candidates.size == 0:
            break
        src = source_candidates[np.argmax(grad[source_candidates])]
        # choose target with smallest grad (could be zero)
        tgt = np.argmin(grad)
        # if both are same, break
        if src == tgt:
            break
        # Determine delta: up to fraction of src mass
        delta = a[src] * 0.1  # 10% maybe
        # Optionally try halving delta if no improvement
        for _ in range(5):
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta
            # keep positivity
            if a_new[src] < 0:
                delta = a_new[src]  # overshoot?
                continue
            new_score = fast_score(a_new, fft_len, n)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                break
            delta *= 0.5
        max_iter -= 1
    return a, cur_score

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence score.

Algorithm
---------
1. Choose a length n (≤10 000).  If a previous solution
   ``height_sequence_1`` exists we reuse its length,
   otherwise we start with n≈8000.
2. Keep the current best candidate on the probability simplex
   (sum = 1, all entries ≥0).
3. Main search (≈ 70 % of the time budget):
   * many random restarts → L‑BFGS optimisation of a
     log‑sum‑exp smoothing of the max‑convolution.
   * after each L‑BFGS run a short *gradient‑directed hill‑climb*
     moves mass from indices with a large sub‑gradient to indices
     with a small sub‑gradient (the true max‑convolution objective).
4. Refined phase (≈ 20 % of the budget):
   a longer L‑BFGS schedule with a very small smoothing factor.
5. Final phase (remaining time):
   an extended gradient‑directed hill‑climb until the deadline.
6. Return the best sequence found (already normalised – the harness
   calls ``evaluate_sequence`` which does the final scaling/clipping).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a vector that sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the exact objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators (logits before softmax)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Gradient‑directed hill‑climb (uses the true max‑convolution)
# ----------------------------------------------------------------------


def grad_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑directed hill‑climb.
    The sub‑gradient of the (almost) active max‑convolution entries is
    computed and mass is transferred from an index with a large gradient
    to one with a small gradient.  A tiny back‑tracking line‑search
    guarantees monotone improvement.
    """
    best_a = a.copy()
    best_score = cur_score
    eps = 1e-12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # ----- compute the (smoothed) sub‑gradient of max(conv) -----
        conv = conv_fft(best_a, fft_len_conv, n)
        max_val = conv.max()
        # consider all entries that are within a tiny tolerance of the max
        tol = max(eps, (max_val - conv.mean()) * 1e-9)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = best_a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                      # ∂max/∂a_i

        # ----- choose a source (large gradient) and a target (small) -----
        src_candidates = np.where(best_a > eps)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        tgt = np.argmin(grad)                  # could be the same as src
        if src == tgt:
            # pick the second‑best target if possible
            sorted_idx = np.argsort(grad)
            for cand in sorted_idx:
                if cand != src:
                    tgt = cand
                    break
        # ----- move a fraction of mass from src to tgt -----
        delta = best_a[src] * move_frac
        if delta < eps:
            continue

        # simple back‑tracking line‑search on delta
        improved = False
        for _ in range(6):
            new_a = best_a.copy()
            new_a[src] -= delta
            new_a[tgt] += delta
            # keep non‑negativity (tiny numerical errors are tolerable)
            if new_a[src] < -1e-15:
                delta *= 0.5
                continue
            new_a = np.maximum(new_a, 0.0)
            # renormalise – the move is mass‑conserving but rounding may drift
            new_a /= new_a.sum()
            new_score = fast_score(new_a, fft_len_conv, n)
            if new_score < best_score - 1e-13:
                best_a = new_a
                best_score = new_score
                improved = True
                break
            delta *= 0.5
        if not improved:
            # no improvement – try a different pair in the next iteration
            continue

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain ``list`` of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Schedules for continuation L‑BFGS.
    # ------------------------------------------------------------------
    # a relatively dense schedule early on, then a very fine schedule
    # for the final refinement.
    lam_schedule_main = list(np.geomspace(0.5, 1e-9, num=70))
    lam_schedule_refine = list(np.geomspace(0.5, 1e-12, num=120))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "best"
    ]

    # ------------------------------------------------------------------
    # First phase – many restarts with moderate L‑BFGS work.
    # ------------------------------------------------------------------
    while time.time() < deadline - 260.0:   # keep ~260 s for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_main,
            maxiter_per_stage=700,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] mode={mode} → improved fast score = {best_score:.6f}")

        # short gradient‑directed hill‑climb (≈ 0.7 s)
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = grad_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            hc_deadline,
            max_iter=1500,
            move_frac=0.15,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Second phase – refined sub‑gradient descent (more aggressive).
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 70.0
        print("[grad‑subgrad] start")
        best_a, best_score = grad_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            max_iter=500_000,
            move_frac=0.12,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[grad‑subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Third phase – final high‑precision L‑BFGS with tiny smoothing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print("[refine L‑BFGS] start")
        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_refine,
            maxiter_per_stage=900,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[refine L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive gradient hill‑climb (use whatever time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final grad‑HC] start")
        best_a, best_score = grad_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=30000,
            move_frac=0.10,
        )
        print(f"[final grad‑HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
