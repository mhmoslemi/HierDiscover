def max_conv_subgrad(a, fft_len):
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    grad = 2.0 * conv_g[n-1:n-1+n]
    return grad, max_val
```

"""
Propose a low‑evaluation step sequence.

This implementation improves on the previous baseline by:

1. **Parametric sweep over edge‑heavy families** – inverse‑power,
   double‑exponential and simple 1/(i+1) shapes.  The sweep quickly finds a
   good “U‑shaped’’ seed.

2. **Adam optimisation on a smooth log‑sum‑exp approximation** of the max‑
   convolution.  The variables are the logits of a soft‑max, guaranteeing
   non‑negativity and ∑a = 1.  The smoothing parameter λ is slowly reduced,
   turning the objective into the true maximum in the limit.

3. **Projected sub‑gradient descent with momentum** on the exact max‑convolution.

4. **Deterministic hill‑climb** that repeatedly moves a small fraction of mass
   from the coordinate with the highest sub‑gradient (most harmful) to the one
   with the lowest (most helpful).

5. **A final symmetrisation** (a←(a+a[::-1])/2) which never harms the objective
   for a self‑convolution and usually improves it.

6. **Optional simulated‑annealing fine‑tune** for escaping tiny local minima.

All phases respect a global time budget (default 1000 s).  The best sequence
found before the deadline is returned as a list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize  # used only for the optional L‑BFGS phase (may be omitted)


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the scaled objective 2·n·max(conv) when ∑a = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. the convolution entries
    w = exp_shift / Z        # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient entries that correspond to each a_j
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


# ----------------------------------------------------------------------
# Exact sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def max_conv_subgrad(a: np.ndarray, fft_len_g: int) -> tuple[np.ndarray, float]:
    """
    Returns a sub‑gradient of max(a*a) and the current maximal convolution value.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_g)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

    return grad, max_val


# ----------------------------------------------------------------------
# Simple noise injection
# ----------------------------------------------------------------------


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric families (edge‑heavy shapes)
# ----------------------------------------------------------------------


def make_inv_power(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-alpha}  (U‑shape, symmetric)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    prod = idx * (n - idx + 0.0)
    a = prod ** (-alpha)
    a /= a.sum()
    return a


def make_edge_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam*i) + exp(-lam*(n-1-i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def make_sum_inverse(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1) + 1/(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    a /= a.sum()
    return a


def make_power_family(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """a_i ∝ (i+1)^p  (or reversed)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        i = i[::-1]
    a = i ** p
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimization primitives
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp objective.
    The smoothing λ is decreased geometrically.
    """
    y = np.log(a_start + 1e-12)          # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                # learning rate for logits
    lam_factor = 0.30        # initial smoothing factor multiplier
    lam_decay = 0.998        # decay per iteration
    t = 0

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        # slowly decay smoothing
        lam_factor *= lam_decay
        f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def subgradient_momentum_opt(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        grad, _ = max_conv_subgrad(a, fft_len_g)

        # momentum
        v = beta * v + (1.0 - beta) * grad
        step = 0.7 / np.sqrt(it + 1.0)           # decreasing step

        a_candidate = simplex_projection(a - step * v)
        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb moving a small fraction of mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        grad, _ = max_conv_subgrad(best, fft_len)

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        delta = best[i_hi] * rng.random() * 0.05   # up to 5 % of the source mass
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    start_temp: float = 0.02,
    end_temp: float = 1e-5,
    max_moves: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Very light‑weight simulated annealing on the true max‑convolution.
    Randomly transfers a small fraction of mass; moves that worsen the
    objective may be accepted with a temperature‑dependent probability.
    """
    cur_a = a_start.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = start_score
    rng = np.random.default_rng()
    t_start = time.time()
    move = 0

    while time.time() < deadline and move < max_moves:
        # linear temperature schedule
        frac = (time.time() - t_start) / (deadline - t_start)
        temp = start_temp * (1.0 - frac) + end_temp * frac

        # pick two random indices (source, target)
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt:
            continue

        delta = cur_a[src] * rng.random() * 0.04   # ≤4 % transfer
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = simplex_projection(cand)

        cand_score = fast_score(cand, fft_len_conv, n)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp((cur_score - cand_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        move += 1

    return best_a, best_score


def symmetrize(a: np.ndarray) -> np.ndarray:
    """
    Returns the symmetric version (a+(a reversed))/2,
    projected back onto the simplex.
    """
    s = (a + a[::-1]) * 0.5
    s /= s.sum()
    return s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    evaluate_sequence(...).  The algorithm follows the phases described
    in the module docstring and respects the global time budget.
    Returns the best list of floats discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                      # length of the sequence
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Global time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    # ------------------------------------------------------------------
    # Helper to keep the global best
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def update_best(candidate):
        nonlocal best_a, best_score
        sc = fast_score(candidate, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[update] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Initialise from a possible previous construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, hs)
            a_interp = np.maximum(a_interp, 0.0)
            a_interp /= a_interp.sum()
            update_best(a_interp)

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (edge‑heavy families)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 0.10 * budget_s
    families = []

    # inverse‑power family
    for alpha in np.linspace(0.05, 3.0, 25):
        families.append(make_inv_power(n, alpha))

    # double‑exponential family

    # random seeds for robustness

    # evaluate all candidates (stop early if time runs out)
    for cand in families:
        if time.time() >= sweep_deadline:
            break
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        update_best(cand)

    # If nothing better than infinite (unlikely), fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    return final_seq
