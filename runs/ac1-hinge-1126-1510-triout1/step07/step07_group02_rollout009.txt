"""
Improved step-function optimizer.

The search consists of three stages:
1. Parameter‑grid exploration of several simple families (exponential,
   power‑law, linear, edge‑heavy).  The best candidate is kept.
2. A long Adam‑style mirror descent on the logits of a softmax
   parametrisation, using the exact max‑convolution sub‑gradient.
   The learning‑rate schedule is aggressive early on and gradually
   decays.
3. A greedy hill‑climb with occasional uphill moves (simulated‑annealing
   style) to polish the solution.

All stages respect the 1000 s deadline, update the best known sequence
and finally return it as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return the objective value 2·n·max(a*a) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector.  Several simple families are supported."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'linear', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'linear':
        a = np.linspace(1.0, 0.0, n, dtype=float)
    elif mode == 'edge':
        # heavy mass at both ends, small in the middle
        edge_frac = rng.uniform(0.005, 0.025)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = rng.uniform(0.5, 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.06 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.015 * n), int(0.06 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise to a probability vector and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of the max‑convolution.
    Returns (objective, gradient) with respect to logits y (a = softmax(y)).
    The smooth approximation uses a log‑sum‑exp with temperature λ.
    """
    a = softmax(y)

    # full convolution a*a (real)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w (via FFT)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (extract the n components that correspond to a)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run a continuation of L‑BFGS for each λ in lam_schedule.
    Returns the best probability vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  rng: np.random.Generator):
    """
    Adam‑style mirror descent on the logits of the softmax.
    Uses the exact sub‑gradient of the hard max‑convolution.
    """
    # initialise logits from probabilities
    y = np.log(a_start + 1e-12)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_y = y.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # exact max‑convolution and mask of active indices
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # gradient of the (hard) max w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_a = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # gradient w.r.t logits (softmax chain rule)
        mean_g = np.dot(a, g_a)
        g_y = a * (g_a - mean_g)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * g_y
        v = beta2 * v + (1 - beta2) * (g_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # learning rate schedule – start relatively large then decay
        lr = 0.5 / np.sqrt(t)  # 0.5 is a comfortable constant
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        a_new = softmax(y)
        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            best_y = y.copy()
    return best_a, best_score

def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000,
               temperature_start: float = 1e-2,
               temperature_end: float = 1e-6):
    """
    Greedy hill‑climb with a tiny simulated‑annealing component.
    A random mass transfer is accepted if it improves the score or
    with probability exp(−Δ / T) where T is a decreasing temperature.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        T = temperature_start * ((temperature_end / temperature_start) ** (it / max_iter))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.12  # up to 12 % of source
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # uphill acceptance with probability exp(-(new−old)/T)
            if rng.random() < np.exp(-(new_score - best_score) / max(T, 1e-12)):
                accept = True
        if accept:
            if new_score < best_score:
                best_score = new_score
        else:
            # revert move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

"""
Improved step‑function optimizer for the competition.

The search is split into four logical stages that respect the 1000 s
budget:

1. **Fast parametric scan** – evaluates a handful of simple families
   (power‑law, exponential, linear, edge‑heavy) on a grid of
   parameters and keeps the best probability vector.

2. **Heavy L‑BFGS continuation** – applies a smoothed log‑sum‑exp
   approximation of the max‑convolution and solves a series of
   increasingly accurate problems (λ decreasing).  Random restarts
   and small Gaussian perturbations of the current best help to escape
   local minima.

3. **Adam‑style mirror descent** – works directly on the logits of a
   softmax parametrisation and uses the exact sub‑gradient of the
   hard max‑convolution.  An Adam‑like adaptive schedule quickly
   refines the solution.

4. **Annealed greedy hill‑climb** – repeatedly moves a random
   fraction of mass from one coordinate to another; improvements are
   always accepted, but uphill moves are occasionally accepted with a
   temperature that decays to near‑zero, providing a final polishing
   step.

All stages monitor the global deadline and stop early if necessary.
The best sequence (a list of non‑negative floats) is returned and also
stored in ``height_sequence_1`` for possible reuse by subsequent runs.
"""

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Objective value 2·n·max(a*a) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector from a few simple families."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'linear', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'linear':
        a = np.linspace(1.0, 0.0, n, dtype=float)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.005, 0.025)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = rng.uniform(0.5, 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.06 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.015 * n), int(0.06 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – quick parametric scan + few random starts
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)

    # 0.1 s budget for deterministic families
    for p in np.linspace(0.2, 2.5, 23):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()

    for r in np.linspace(0.001, 0.02, 20):
        a = np.exp(-r * np.arange(n, dtype=float))
        a /= a.sum()
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()

    a = np.linspace(1.0, 0.0, n, dtype=float)
    a /= a.sum()
    s = fast_score(a, fft_len_conv, n)
    if s < best_score:
        best_score, best_a = s, a.copy()

    for frac in np.linspace(0.005, 0.025, 9):
        m_edge = max(1, int(n * frac))
        a = np.full(n, 1.0 / n)
        heavy = rng.uniform(0.5, 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a /= a.sum()
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # few pure random starts (≈5 % of total budget)
    while time.time() < start_time + 0.05 * budget_s:
        a0 = random_initial(n, rng)
        s = fast_score(a0, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a0.copy()

    print(f"[init] n={n}, best fast_score after quick scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS continuation on smoothed max‑convolution
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = [
        0.6, 0.45, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, s_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if s_opt < best_score:
            best_score, best_a = s_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam mirror descent on the exact max‑convolution
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.88 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Adam] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Annealed greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=60000,
            temperature_start=5e-3,
            temperature_end=1e-9,
        )
        print(f"[Hill‑climb] final fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # make the result available for later runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
