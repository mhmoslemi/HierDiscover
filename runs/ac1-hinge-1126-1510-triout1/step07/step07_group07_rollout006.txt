def powerlaw_init(n, alpha):
    idx = np.arange(n)
    # avoid zero division
    weight = (idx+0.5)**(-alpha) + (n-idx-0.5)**(-alpha)
    weight[weight < 0] = 0
    weight_sum = weight.sum()
    if weight_sum == 0:
        return uniform_init(n)
    return weight / weight_sum

"""
Propose a low‑evaluation step‑function.

The optimizer works on the probability simplex (sum = 1) and consists of
four phases

1.  A diverse pool of analytic initialisations (Beta, power‑law,
    semicircle, uniform, random Gaussian, sparse binary, Gaussian bump,
    mixtures and the historic sequence if available).  The best few are
    kept for full optimisation.

2.  Continuation L‑BFGS on a smoothed log‑sum‑exp surrogate of the
    max‑convolution.  The smoothing parameter λ is reduced geometrically.
    The optimiser runs several stages with up to 600 L‑BFGS iterations
    per stage.

3.  (Optional) short projected sub‑gradient descent on the true
    objective for a few hundred thousand steps.

4.  A long simulated‑annealing / hill‑climbing phase that repeatedly
    transfers a fraction of mass between two coordinates and accepts
    worsening moves with a temperature that decays geometrically.

All phases respect the global 1000 s budget.  The best sequence found
is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function when the vector is normalised
    (∑a_i = 1).  The returned value equals `2·n·max(conv(a,a))`.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # Correlation of a and w (gradient of conv w.r.t. a)
    # Reverse a for convolution
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # Chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Analytic initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """β(p,p) density on (0,1); p≈0.5 gives the classic arcsine."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def powerlaw_init(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+½)^‑α + (n‑i‑½)^‑α."""
    idx = np.arange(n, dtype=float)
    w = np.power(idx + 0.5, -alpha) + np.power(n - idx - 0.5, -alpha)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)                     # positive
    a = a / a.sum()
    return a


def binary_sparse_init(
    n: int, rng: np.random.Generator, density: float = 0.20
) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that tries to move mass away from current max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)

        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb.  A small fraction of mass is moved
    between two coordinates; worsening moves are accepted with probability
    exp(-Δ/T) where T decays geometrically from 0.2 to 1e-5.
    """
    best = a.copy()
    best_score = cur_score
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / max(1e-12, deadline - start)   # 0 → 1
        T = 0.2 * (1e-5 / 0.2) ** frac                 # geometric decay

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # accept with probability exp(-Δ/T)
            if rng.random() < np.exp(-(cand_score - best_score) / (T + 1e-12)):
                best = cand
                best_score = cand_score
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The function respects a total wall‑clock budget of
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – try to keep compatibility with a historic solution.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Build a rich pool of initial shapes.
    # ------------------------------------------------------------------
    candidates = []

    # Beta(p,p) – include the classic arcsine (p=0.5) and more extreme U‑shapes
    for p in [0.5, 0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85]:
        candidates.append((f"beta_{p:.2f}", beta_init(n, p)))
    for p in [0.45, 0.40, 0.35, 0.30]:
        candidates.append((f"beta_{p:.2f}", beta_init(n, p)))

    # Power‑law families
    for alpha in [0.5, 0.75, 1.0, 1.5, 2.0]:
        candidates.append((f"power_{alpha:.2f}", powerlaw_init(n, alpha)))

    # Classic shapes
    candidates.append(("semicircle", semicircle_init(n)))
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("random", random_gaussian_init(n, rng)))
    candidates.append(("binary", binary_sparse_init(n, rng, density=0.15)))
    candidates.append(("bump", gaussian_bump_init(n, rng)))

    # Mixtures of the best beta with uniform
    beta_ref = beta_init(n, 0.70)
    for w in [0.2, 0.4, 0.6, 0.8]:
        mixed = mixture_init(beta_ref, uniform_init(n), w)
        candidates.append((f"beta_uniform_mix_{w:.1f}", mixed))

    # Historic sequence (if length matches)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            candidates.append(("previous", tmp / tmp.sum()))

    # ------------------------------------------------------------------
    # Quick scan – keep only the most promising possibilities.
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:22s} fast score = {sc:.6f}")

    scored.sort()
    top_k = 5
    best_a = None
    best_score = np.inf
    best_y = None

    # ------------------------------------------------------------------
    # Continuation L‑BFGS (≈ 50 % of total budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-9, num=30))
    maxiter_per_stage = 600
    lbfgs_deadline = start_time + 0.50 * budget_s

    print("[phase] L‑BFGS continuation started")
    for rank in range(min(top_k, len(scored))):
        if time.time() >= lbfgs_deadline:
            print("[phase] L‑BFGS deadline hit")
            break
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:22s} score = {best_score:.6f}")

        # Brief random hill‑climb after each candidate (if time permits)
        if time.time() < lbfgs_deadline:
            bc, bs = hill_climb(
                best_a, best_score, fft_len_conv, n,
                deadline=lbfgs_deadline,
                max_iter=1500,
                max_move=0.15,
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] hill‑climb improved to {best_score:.6f}")

    # Fallback – should never happen
    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Random‑restart continuation (exploratory runs while time permits)
    # ------------------------------------------------------------------
    restart_deadline = start_time + 0.70 * budget_s
    restart_cnt = 0
    while time.time() < restart_deadline:
        restart_cnt += 1
        a0 = random_gaussian_init(n, rng)
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g,
            lam_schedule[:12],          # shorter schedule for quick restarts
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        s_opt = fast_score(a_opt, fft_len_conv, n)
        if s_opt < best_score:
            best_score, best_a, best_y = s_opt, a_opt.copy(), y_opt.copy()
            print(f"[restart] #{restart_cnt:03d} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.85 * budget_s
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a, fft_len_conv, n, subgrad_deadline,
            eta0=0.40,
            max_iters=300_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final annealing / hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] annealing hill‑climb")
        best_a, best_score = annealing_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, rng
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore – supplied by harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
        final_val = None

    return final_seq
