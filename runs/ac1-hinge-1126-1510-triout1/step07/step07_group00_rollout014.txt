"""
Propose a low‑evaluation step sequence.

The search proceeds in four phases.

1️⃣  A cheap sweep over several analytic families (power‑law,
    exponential, Gaussian, edge‑heavy) to obtain a decent starting
    distribution.

2️⃣  Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
    true ``max`` of the auto‑convolution.  The λ‑schedule is long and
    ends at 5·10⁻⁴ so that the smoothed objective is very close to the
    true non‑smooth one.

3️⃣  A population‑based evolutionary refinement.  From the current best
    point we generate many noisy perturbations, project them onto the
    simplex and keep any improvement.  The mutation strength σ decays
    over time (simulated‑annealing‑style) and a short local L‑BFGS
    “polish” is applied to each accepted offspring.

4️⃣  Deterministic gradient‑guided hill‑climbing that repeatedly moves a
    small fraction of mass from the coordinate with the largest
    sub‑gradient (most harmful) to the one with the smallest
    sub‑gradient (most helpful).  This final stage typically squeezes
    the last few percent of the objective.

All phases respect a global time budget (default 1000 s).  The function
returns the best list of floats found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  The function works in O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev needed for the
    # correlation that appears in the derivative)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def evolutionary_refinement(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple population based refinement.
    Repeatedly create noisy offspring, keep the best,
    and apply a short L‑BFGS polish after each improvement.
    """
    best = a_start.copy()
    best_score = fast_score(best, fft_len, n)

    # decay schedule for mutation strength
    sigma = 0.05  # initial standard deviation
    min_sigma = 0.001
    last_improve = time.time()

    # a short λ‑schedule for the polishing step
    lam_schedule = [0.3, 0.1, 0.04, 0.015, 0.006, 0.002]

    while time.time() < deadline:
        # reduce sigma slowly
        sigma = max(min_sigma, sigma * 0.995)

        # generate a small batch of candidates
        batch = 32
        improved = False
        for _ in range(batch):
            cand = add_noise(best, sigma=sigma, rng=rng)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                # polish the offspring with a few L‑BFGS steps (cheap)
                cand, cand_score = lbfgs_continuation(
                    cand,
                    lam_schedule,
                    maxiter_per_lam=200,
                    deadline=time.time() + 0.5,  # give it a tiny slice of time
                    fft_len_conv=fft_len,
                    fft_len_g=next_pow2(3 * n - 2),
                    n=n,
                )
                if cand_score < best_score:
                    best_score = cand_score
                    best = cand.copy()
                    improved = True
        if improved:
            last_improve = time.time()
        else:
            # if we have been stagnant for a while, increase sigma a little
            if time.time() - last_improve > 20.0:
                sigma = min(0.5, sigma * 1.5)
                last_improve = time.time()

    return best, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small amount of mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    Only strictly improving moves are retained.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # move a small random fraction (up to 5 %) from i_hi to i_lo
        delta = best[i_hi] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "gauss"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # gauss
        centre = n // 2
        sigma = rng.uniform(0.05 * n, 0.15 * n)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – use a power‑of‑two length for convenient FFTs
    # ------------------------------------------------------------------
    n = 4096                  # a little larger than the previous 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidate (uniform or a previously discovered one)
    # ------------------------------------------------------------------
    a0 = np.full(n, 1.0 / n)                 # uniform start
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to required length and renormalise
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc = fast_score(a1, fft_len_conv, n)
            if sc < best_score:
                best_a, best_score = a1, sc
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (power‑law / exponential / Gaussian)
    # ------------------------------------------------------------------
    for p in np.linspace(0.0, 3.0, 13):
        # increasing power‑law
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        # decreasing power‑law
        a = ((np.arange(n, 0, -1, dtype=float) ** p))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    # Gaussian family sweep
    for sigma_frac in [0.05, 0.10, 0.15, 0.20]:
        centre = n // 2
        sigma = sigma_frac * n
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] gauss sigma={sigma_frac:.2f} -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # Split the budget (adjustable)
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.90 * budget_s   # evolutionary
    # remaining time for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18,
        0.12, 0.08, 0.05, 0.03, 0.018,
        0.01, 0.006, 0.003, 0.001, 0.0005,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
    ]
    maxiter_per_lam = 800   # a bit more agressive than before

    while time.time() < phase1_end:
        # Occasionally jitter around the current best; otherwise fresh start
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – evolutionary / population based refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = evolutionary_refinement(
            best_a, phase3_end, fft_len_conv, n, rng
        )
        print(f"[evolution] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = gradient_guided_hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=20000
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
