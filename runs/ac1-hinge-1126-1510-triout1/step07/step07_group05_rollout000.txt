    best = a.copy()
    best_score = fast_score(best,...)
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        max_val = conv.max()
        # find indices k where conv near max
        active_sums = np.where(conv >= max_val - 1e-12)[0]
        k = rng.choice(active_sums)

        # i in [0,n-1], j = k-i

        j_vals = k - i_vals

        products = best[i_vals] * best[j_vals]

        i0 = i_vals[idx]

        # move mass delta from i0, j0 to two random other indices m1,m2
        delta = min(best[i0], best[j0]) * 0.1 * rng.random()

        # choose alternative indices not i0,j0

"""
Propose a step‑function with a low evaluate_sequence score.

Algorithm
---------
1. Choose problem size n (≈ 8k–10k).  If a previous sequence exists, reuse its
   length (capped at 10000).  Normalize to a probability vector (sum = 1).
2. Generate a diverse pool of starting points on the simplex (random,
   exponential, power‑law, edge‑heavy, arcsine, greedy Sidon, sparse Dirichlet,
   plus a slight perturbation of the current best).
3. For each start run a continuation L‑BFGS on a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor λ is reduced
   geometrically (≈ 60 stages).  Each stage performs a few hundred L‑BFGS
   iterations.
4. After the coarse L‑BFGS phase, run a refined L‑BFGS with a much smaller
   λ schedule (≈ 80 stages, more iterations) to get very close to the true max.
5. Perform a **targeted hill‑climb**: repeatedly locate a convolution entry at
   the current maximum, identify the pair (i,j) contributing the largest
   product a_i·a_j to that entry and move a small fraction of mass from i and
   j to other indices.  Accept only improvements; the move respects the simplex
   (project via Euclidean projection).  This directly attacks the active
   constraints and often yields a noticeable drop in the objective.
6. Return the best sequence found (as a plain list).  The external harness
   calls ``evaluate_sequence`` to obtain the final score.

The total runtime is limited to the given budget (≈ 1000 s).  The code
periodically checks the deadline and stops early if needed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Targeted hill‑climb
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that focuses on the active (maximum) convolution
    entry.  Moves a small amount of mass from the pair (i,j) that
    contributes the largest product to that entry and redistributes it
    elsewhere.  Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute convolution once per iteration
        conv = conv_fft(best, fft_len, n)
        max_val = conv.max()

        # pick a random index among those attaining (or almost attaining) the max
        # tolerance small enough to avoid stagnation
        tol = max(1e-14, (max_val - conv.mean()) * 1e-8)
        active_idxs = np.where(conv >= max_val - tol)[0]
        k = rng.choice(active_idxs)

        # all pairs (i, j) with i+j = k and 0<=i,j<n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals

        # product contributions
        products = best[i_vals] * best[j_vals]
        idx_max = np.argmax(products)
        i0 = i_vals[idx_max]
        j0 = j_vals[idx_max]

        # decide how much mass to move
        # take a fraction of the smaller of the two masses (or of the single mass)
        if i0 == j0:
            mass_src = best[i0]
            if mass_src <= 0:
                continue
            delta = mass_src * 0.08 * rng.random()  # up to ~8% of that entry
            if delta <= 0:
                continue

            # choose two distinct destinations different from i0
            dest_candidates = np.delete(np.arange(n), i0)
            if dest_candidates.size < 2:
                continue
            m1, m2 = rng.choice(dest_candidates, size=2, replace=False)

            new = best.copy()
            new[i0] -= delta
            # split delta evenly
            new[m1] += delta * 0.5
            new[m2] += delta * 0.5
        else:
            # both i0 and j0 have some mass
            max_transfer = min(best[i0], best[j0])
            if max_transfer <= 0:
                continue
            delta = max_transfer * 0.08 * rng.random()
            if delta <= 0:
                continue

            # choose two distinct destinations not equal to i0 or j0
            excluded = {i0, j0}
            dest_candidates = np.setdiff1d(np.arange(n), list(excluded))
            if dest_candidates.size < 2:
                continue
            m1, m2 = rng.choice(dest_candidates, size=2, replace=False)

            new = best.copy()
            new[i0] -= delta
            new[j0] -= delta
            new[m1] += delta * 0.5
            new[m2] += delta * 0.5

        # project back to simplex (helps with numerical drift)
        new = project_onto_simplex(new)

        new_score = fast_score(new, fft_len, n)
        if new_score < best_score - 1e-12:
            best = new
            best_score = new_score
        # else reject

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluate_sequence score.  Returns the sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate on the simplex.
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for continuation L‑BFGS
    # --------------------------------------------------------------
    lam_schedule_coarse = list(np.geomspace(0.5, 1e-8, num=55))
    lam_schedule_fine = list(np.geomspace(0.5, 1e-12, num=85))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "best"
    ]

    # --------------------------------------------------------------
    # Phase 1 – many restarts with moderate L‑BFGS work.
    # --------------------------------------------------------------
    while time.time() < deadline - 350.0:  # reserve ~350 s for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_schedule_coarse,
            maxiter_per_stage=700,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] mode={mode} → improved fast score = {best_score:.6f}")

        # short targeted hill‑climb after each restart (≈ 0.8 s)
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 2 – refined L‑BFGS with very small smoothing.
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print("[refine L‑BFGS] start")
        y_opt = schedule_optimize(
            y_start=best_y,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_schedule_fine,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[refine L‑BFGS] improved fast score = {best_score:.6f}")
        else:
            print("[refine L‑BFGS] no improvement")
    else:
        print("[refine L‑BFGS] skipped (time)")

    # --------------------------------------------------------------
    # Phase 3 – aggressive targeted hill‑climb until the deadline.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final targeted HC] start")
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=50000
        )
        print(f"[final targeted HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # The harness will call evaluate_sequence; we also compute it for logging.
    try:
        final_val = evaluate_sequence(final_seq)
    except Exception as e:
        print(f"[final] evaluate_sequence raised an exception: {e}")
        final_val = float("inf")
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
