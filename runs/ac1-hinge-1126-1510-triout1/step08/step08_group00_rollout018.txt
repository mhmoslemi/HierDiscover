from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_for_g(a, M):
    n = len(a)
    # Precomputed row_idx, col_idx
    data = np.repeat(a, n)
    A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)   # minimize -sum => maximize sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    return g
```

def lp_iteration(a, M, row_idx, col_idx, n, deadline):
    # Build data
    data = np.repeat(a, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return a, None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return a, None
    g_norm = g / sum_g
    # mixing
    t = 0.5
    a_new = (1 - t) * a + t * g_norm
    a_new = simplex_projection(a_new)   # enforce sum=1
    return a_new, fast_score(a_new, fft_len_conv, n)

```python
"""
Propose a low‑evaluation step sequence via a multi‑phase
search: cheap parametric sweep → L‑BFGS continuation on a
log‑sum‑exp smoothed max → linear‑programming based mixing
→ final simulated‑annealing fine‑tuning.
The LP step iteratively mixes the current sequence with a
solution of   max_{k}(f∗g)_k ≤ M   that maximises ∑g_j,
guaranteeing a non‑increasing max‑convolution value.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z  # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient and exponentiated‑gradient refinements
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the simplex using the true sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb that accepts uphill moves with a temperature schedule."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    it = 0
    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families (cheap sweep)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap sweep over many random parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    start = time.time()
    deadline = start + budget_s * 0.15

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based mixing step
# ----------------------------------------------------------------------


def precompute_conv_indices(n: int):
    """Return row and column index arrays for the Toeplitz convolution matrix."""
    i = np.arange(n, dtype=np.int32).repeat(n)
    j = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i + j  # shape (n*n,)
    cols = j
    return rows, cols


def lp_mix(
    a: np.ndarray,
    fft_len_conv: int,
    rows_idx: np.ndarray,
    cols_idx: np.ndarray,
    deadline: float,
    mix_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Solve the linear program that maximises ∑g_j subject to
    (a * g) ≤ M (element‑wise) and returns the mixed vector.
    """
    n = a.shape[0]
    # compute current bound M
    M = full_conv(a, fft_len_conv).max()

    # Build sparse matrix A where A[k, j] = a[k-j] (zero when out of range)
    data = np.repeat(a, n)          # length n*n
    A = csr_matrix((data, (rows_idx, cols_idx)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                 # minimise -∑g ⇒ maximise ∑g
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        # If the LP failed, just return the original vector.
        return a, fast_score(a, fft_len_conv, n)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, fast_score(a, fft_len_conv, n)

    g_norm = g / sum_g
    # Mix the current vector with the LP solution
    a_new = (1.0 - mix_factor) * a + mix_factor * g_norm
    # Numerical drift → project back onto simplex
    a_new = simplex_projection(a_new)
    return a_new, fast_score(a_new, fft_len_conv, n)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser for the step‑function evaluation problem.
    Returns a list of non‑negative heights that yields a low score.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate length keeps the LP tractable.
    # ------------------------------------------------------------------
    n = 2500
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidate (uniform, possibly upgraded with a historic solution)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # Interpolate historic sequence to the required length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] historic fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # Allocate phase boundaries (percentages of the total budget)
    phase_sweep_end   = start_time + 0.05 * budget_s
    phase_lbfgs_end   = start_time + 0.45 * budget_s
    phase_lp_end      = start_time + 0.70 * budget_s
    phase_anneal_end  = deadline                     # the rest

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    if time.time() < phase_sweep_end:
        a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
        if sc_sweep < best_score:
            best_a, best_score = a_sweep, sc_sweep
            print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (smooth max optimisation)
    # ------------------------------------------------------------------
    if time.time() < phase_lbfgs_end:
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03,
            0.018, 0.01, 0.006, 0.003, 0.0015, 0.0008, 0.0004,
            0.0002, 0.0001, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06
        ]
        maxiter_per_lam = 1200   # a bit more than the previous 800
        while time.time() < phase_lbfgs_end:
            if rng.random() < 0.25:
                a_start = best_a.copy()
                a_start = a_start * rng.normal(1.0, 0.02, size=n)
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()
            else:
                a_start = generate_family("random", n, rng)

            a_opt, sc_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase_lbfgs_end,
                fft_len_conv,
                fft_len_g,
                n,
            )
            if sc_opt < best_score:
                best_score, best_a = sc_opt, a_opt.copy()
                print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming mixing (non‑increasing max bound)
    # ------------------------------------------------------------------
    if time.time() < phase_lp_end:
        rows_idx, cols_idx = precompute_conv_indices(n)
        iter_lp = 0
        while time.time() < phase_lp_end and iter_lp < 40:
            a_candidate, score_candidate = lp_mix(
                best_a, fft_len_conv, rows_idx, cols_idx, phase_lp_end, mix_factor=0.5
            )
            if score_candidate < best_score:
                best_score, best_a = score_candidate, a_candidate.copy()
                print(f"[LP] iteration {iter_lp} new best fast_score={best_score:.6f}")
            else:
                # even if not improved we keep the candidate as the next state
                best_a = a_candidate
            iter_lp += 1

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < phase_anneal_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq

Now let's reason about correctness.

Potential issues:

- `lp_mix` uses `fft_len_conv` argument for M. It has parameter `fft_len_conv` that is passed to full_conv; that's fine.

- The `lp_mix` may be called many times; each time we compute M and build data matrix using a repeat of a. Works.

- The linprog solver may not behave well with very ill-conditioned problems due to scaling; but should succeed as constraints simple.

- The `precompute_conv_indices` generates rows and col arrays of shape n*n. For n=2500 => 6.25M entries; each int32 4 bytes -> rows ~25 MB, cols ~25 MB. Acceptable.

- We use `maxiter_per_lam = 1200` may increase runtime; but each L-BFGS call with 1200 iterations can be heavy. However we restrict total time for LBFGS phase (45% -> 450 sec). If each lam schedule has 24 lam values, each iteration executes a minimize call for up to 1200 iterations, which might be a lot (24*1200 = 28800 total inner iterations). """
Multi‑phase optimiser for the step‑function evaluation problem.

The search proceeds as follows:

1. Cheap parametric sweep – generate a few hundred simple families and keep
   the best candidate.

2. L‑BFGS continuation – minimise a log‑sum‑exp smoothed version of the
   max‑convolution using the soft‑max parametrisation (∑a_i = 1).  A
   schedule of decreasing smoothing factors drives the optimiser toward the
   true non‑smooth objective.

3. Linear‑programming mixing – given the current sequence f, solve the LP
   maximise ∑g_j subject to (f * g) ≤ M where M = max(f * f).  The solution
   g is mixed with f (convex combination) which provably does not increase
   the max‑convolution bound.  Repeating this step drives the bound down
   further.

4. Simulated‑annealing hill‑climb – stochastic mass‑transfer moves on the
   simplex, accepting uphill moves with a linearly‑cooling temperature to
   escape local minima.

The algorithm returns the best sequence found before the allotted time
budget (default 1000 s) expires.  It works for any length; we use 2500
elements to keep the LP tractable while still allowing a strong
improvement over the uniform baseline.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a sequence that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    it = 0
    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        it += 1
    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform a cheap sweep over many random parametric families and return
    the best vector and its score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    start = time.time()
    deadline = start + budget_s * 0.15

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing utilities
# ----------------------------------------------------------------------


def precompute_conv_indices(n: int):
    """Row/column indices for the Toeplitz matrix encoding (a * g)."""
    i = np.arange(n, dtype=np.int32).repeat(n)
    j = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i + j
    cols = j
    return rows, cols


def lp_mix(
    a: np.ndarray,
    fft_len_conv: int,
    rows_idx: np.ndarray,
    cols_idx: np.ndarray,
    mix_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Solve    maximise ∑g_j   subject to   (a * g) ≤ M   and   g≥0,
    where M = max(a*a).  Return the convex mixture (1‑t)a + t·g_norm.
    """
    n = a.shape[0]
    M = full_conv(a, fft_len_conv).max()

    data = np.repeat(a, n)  # coefficients a_i for each column entry
    A = csr_matrix((data, (rows_idx, cols_idx)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                     # minimise –∑g  ⇒  maximise ∑g
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        # fallback – return the unmodified vector
        return a, fast_score(a, fft_len_conv, n)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, fast_score(a, fft_len_conv, n)

    g_norm = g / sum_g
    a_new = (1.0 - mix_factor) * a + mix_factor * g_norm
    a_new = simplex_projection(a_new)
    return a_new, fast_score(a_new, fft_len_conv, n)

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser.  Returns a list of non‑negative heights that
    yields a low evaluation score.
    """
    rng = np.random.default_rng(seed)

    # --------------------
    # Problem size (2500 balances LP tractability & quality)
    # --------------------
    n = 2500
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------
    # Initial candidate (uniform, optionally upgraded with historic data)
    # --------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 = a1 / a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] historic fast_score={best_score:.6f}")

    # --------------------
    # Time bookkeeping
    # --------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    phase_sweep_end   = start_time + 0.05 * budget_s
    phase_lbfgs_end   = start_time + 0.45 * budget_s
    phase_lp_end      = start_time + 0.70 * budget_s
    phase_anneal_end  = deadline

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    if time.time() < phase_sweep_end:
        a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
        if sc_sweep < best_score:
            best_a, best_score = a_sweep, sc_sweep
            print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (smooth max optimisation)
    # ------------------------------------------------------------------
    if time.time() < phase_lbfgs_end:
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03,
            0.018, 0.01, 0.006, 0.003, 0.0015, 0.0008, 0.0004,
            0.0002, 0.0001, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06
        ]
        maxiter_per_lam = 1200
        while time.time() < phase_lbfgs_end:
            if rng.random() < 0.25:
                a_start = best_a * rng.normal(1.0, 0.02, size=n)
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()
            else:
                a_start = generate_family("random", n, rng)

            a_opt, sc_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase_lbfgs_end,
                fft_len_conv,
                fft_len_g,
                n,
            )
            if sc_opt < best_score:
                best_score, best_a = sc_opt, a_opt.copy()
                print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming mixing (non‑increasing max bound)
    # ------------------------------------------------------------------
    if time.time() < phase_lp_end:
        rows_idx, cols_idx = precompute_conv_indices(n)
        iter_lp = 0
        while time.time() < phase_lp_end and iter_lp < 40:
            a_candidate, score_candidate = lp_mix(
                best_a,
                fft_len_conv,
                rows_idx,
                cols_idx,
                mix_factor=0.5,
            )
            if score_candidate < best_score:
                best_score, best_a = score_candidate, a_candidate.copy()
                print(f"[LP] iteration {iter_lp} new best fast_score={best_score:.6f}")
            else:
                # keep the candidate as the new state to allow further movement
                best_a = a_candidate
            iter_lp += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (stochastic fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < phase_anneal_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
