"""
Optimiser for the step‑function attack.

The algorithm proceeds in four stages within the 1000 s budget:

1️⃣  Heavy L‑BFGS on a *smoothed* max‑convolution (log‑sum‑exp) with many
    random restarts and a fine λ‑schedule.

2️⃣  Precise sub‑gradient descent on the *hard* max‑convolution:
    the gradient of the active (max) constraints is computed via FFT,
    a line‑search on the simplex (Euclidean projection) is performed,
    and tiny random perturbations are added when progress stalls.

3️⃣  A short simulated‑annealing phase to escape possible local minima.

4️⃣  Final greedy hill‑climb (tiny mass transfers) for polishing.

All stages respect the non‑negative simplex ∑aᵢ = 1, which makes the
evaluation invariant under overall scaling.  The routine returns the
best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast version of evaluate_sequence – returns 2·n·max(a*a)
    (the denominator Σa = 1 is already enforced).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sparse', 'logit_normal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:                               # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max–convolution.  Parameter y are log‑softmax logits
    (a = softmax(y)).  Returns (value, gradient w.r.t. y).
    """
    a = np.exp(y - np.max(y))
    a /= a.sum()                     # softmax, guarantees simplex

    # full self‑convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12   # smoothing width
    shifted = conv_full - max_c       # ≤ 0
    exp_shift = np.exp(shifted / lam)  # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                 # ∂smooth_max/∂conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fw * fa_rev))

    # slice that corresponds to derivative w.r.t. each a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple λ)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous λ."""
    y = np.log(a_start + 1e-12)       # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# ----------------------------------------------------------------------
# Precise sub‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------
def refine_subgradient(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses a small back‑tracking line search on the simplex.
    """
    a = a_start.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # convolution
        conv = full_conv(a, fft_len)
        M = conv.max()
        eps = max(M * 1e-12, 1e-16)
        mask = (conv >= M - eps).astype(float)

        # gradient of the active constraints
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fr = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fm * fr))
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # try a few step sizes
        improved = False
        for eta in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
            a_cand = simplex_projection(a - eta * g)
            sc = fast_score(a_cand, fft_len, n)
            if sc < best_score:
                best_score = sc
                a = a_cand
                improved = True
                break
        if not improved:
            # occasional tiny random perturbation to escape plateaus
            sigma = rng.uniform(5e-5, 5e-3)
            a_cand = add_noise(a, sigma, rng)
            sc = fast_score(a_cand, fft_len, n)
            if sc < best_score:
                best_score = sc
                a = a_cand
        it += 1
    return a, best_score

# ----------------------------------------------------------------------
# Simulated annealing (light) – used as a safety net
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        max_local_transfer: float = 0.5):
    """Simple SA on the simplex (mass transfers + Gaussian perturbations)."""
    a_cur = a_start.copy()
    cur_score = fast_score(a_cur, fft_len, n)
    best_a = a_cur.copy()
    best_score = cur_score
    start = time.time()
    init_temp = cur_score * 0.02 + 1e-9

    while time.time() < deadline:
        frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (1.0 - frac)

        a_new = a_cur.copy()
        if rng.random() < 0.6:
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * max_local_transfer
                a_new[i] -= delta
                a_new[j] += delta
        else:
            sigma = rng.uniform(0.0005, 0.015)
            a_new = add_noise(a_new, sigma, rng)

        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

# ----------------------------------------------------------------------
# Greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 30000):
    """Tiny random mass transfers, accept only if they improve the score."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------

    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    # allocate ~55 % to L‑BFGS, ~25 % to precise sub‑gradient,
    # ~10 % to SA, remaining to hill‑climb
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    # the rest is for hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts

    lam_schedule = [

    maxiter_per_lam = 1400          # a bit more than the previous 1200

    while time.time() < phase1_end:
        # occasional perturbation of the current best to diversify
