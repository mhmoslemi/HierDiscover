"""
Propose a low‑score step‑function using a multi‑phase optimizer.

Phases
------
1. Light L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution.  The schedule of λ values is chosen to drive the
   smoothed objective towards the true L∞ objective.

2. Targeted projected‑gradient descent on the *hard* max‑convolution.
   At each iteration we compute a sub‑gradient (using the active set of
   maximal convolution entries), then move a fraction of probability
   mass from the index with the largest sub‑gradient to the index with
   the smallest one.  A simple back‑tracking line search guarantees an
   improvement, the iterate is kept on the probability simplex and the
   best solution seen so far is recorded.

3. Fine‑grained greedy hill‑climbing (the same incremental update as
   the reference implementation) to polish the final answer.

The whole search respects a 1000 s budget (the deadline is passed as an
argument).  The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT; returns length 2·len(a)‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return the objective 2·n·max(a*a) for a distribution a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# 1. L‑BFGS on a smoothed max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) with respect to the logits y
    where a = softmax(y).
    """
    # softmax → a on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # back‑propagate through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def subgradient_max_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute a sub‑gradient of max_i (a*a)_i.
    Returns a vector of length len(a) (non‑negative).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)  # shape 2n‑1
    # gradient = 2 * (mask * a_rev) convolved
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]
    return g, conv.max()


def targeted_gradient_descent(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator):
    """
    Move probability mass from indices with the largest sub‑gradient
    to those with the smallest.  A simple back‑tracking line search
    is used; if no improvement is found after several tries we add
    a little noise to escape a plateau.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # parameters
    move_frac = 0.5            # start by moving 50 % of the source mass
    min_move_frac = 0.001
    decay_rate = 0.9995        # slowly shrink the step size
    stagnation = 0             # counts iterations without improvement
    max_stagnation = 2000

    it = 0
    while time.time() < deadline:
        # sub‑gradient and current max
        g, cur_max = subgradient_max_conv(a, fft_len)

        # candidate source (large gradient) and destination (small gradient)
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dst = np.argmin(g)

        if src == dst:
            # nothing to move – break out
            break

        # try to move a fraction of src's mass
        delta = move_frac * a[src]
        improved = False
        for _ in range(6):   # at most 6 halvings
            if delta <= 1e-12:
                break
            a_tmp = a.copy()
            a_tmp[src] -= delta
            a_tmp[dst] += delta
            # No need to re‑normalise – sum stays 1
            score_tmp = fast_score(a_tmp, fft_len)
            if score_tmp < best_score - 1e-12:
                a = a_tmp
                best_score = score_tmp
                best_a = a.copy()
                improved = True
                break
            delta *= 0.5

        if improved:
            stagnation = 0
        else:
            stagnation += 1

        # adapt step size
        move_frac = max(min_move_frac, move_frac * decay_rate)

        # occasionally inject noise to escape flat regions
        if stagnation >= max_stagnation:
            a = add_noise(a, sigma=0.01, rng=rng)
            a = simplex_projection(a)
            stagnation = 0

        it += 1
        # safety: stop after many iterations even if time permits
        if it >= 200_000:
            break

    return best_a, best_score

# 3. Fast greedy hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)  # exact full convolution
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodic full recompute to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

    while time.time() < phase1_end:
        # 80 % of the time perturb the current best, 20 % fresh start
        if rng.random() < 0.8:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted projected gradient descent on the hard max
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = targeted_gradient_descent(
            best_a, phase2_end, n, fft_len, rng,
        )
        print(f"[targeted‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the best sequence globally for possible reuse
    globals()['height_sequence_1'] = final_seq

    return final_seq
