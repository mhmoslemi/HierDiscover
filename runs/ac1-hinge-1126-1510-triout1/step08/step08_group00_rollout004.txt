def random_dirichlet_search(n, fft_len, budget_seconds, rng):
    best_a = None
    best_score = np.inf
    # allocate time chunk: maybe use a fraction of total budget
    start = time.time()
    deadline = start + budget_seconds
    while time.time() < deadline:
        # sample concentration α log-uniform
        log_alpha = rng.uniform(-3, 2)  # α between 0.001 and 100
        alpha = 10**log_alpha
        # sample from gamma
        a = rng.gamma(alpha, 1.0, size=n)
        sum_a = a.sum()
        if sum_a == 0:
            continue
        a /= sum_a
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]
        # direction (negative subgradient)
        d = -g
        # line search
        step = 1.0
        improved = False
        cur_score = fast_score(a, fft_len_conv, n)
        while step > 1e-8:
            a_candidate = simplex_projection(a + step * d)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            # fallback small step
            step = 0.1
            a_candidate = simplex_projection(a + step * d)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a_candidate.copy()
        it += 1
    return best_a, best_score

"""
propose_candidate
=================

A condensed yet more aggressive optimisation pipeline that aims to beat the
previous best bound (~1.5064).  The algorithm proceeds in several stages
(under a single 1000 s budget):

1. **Cheap parametric sweep** – generate candidates from a few hand‑crafted
   families (power‑law, exponential, Gaussian, cosine, bimodal, random) and
   keep the best.

2. **Dirichlet random search** – draw many Dirichlet‑distributed vectors
   with a wide range of concentration parameters; these often give
   surprisingly flat autocorrelations.

3. **Smooth L‑BFGS continuation** – minimise a log‑sum‑exp smooth approximation
   of the max‑convolution using the soft‑max parametrisation.  A long
   schedule of decreasing smoothing parameters is used together with a high
   iteration budget.

4. **Adaptive sub‑gradient descent** – compute the exact sub‑gradient of the
   true max‑convolution and perform a line‑search on the simplex.  This
   refines the solution far beyond fixed‑step momentum updates.

5. **Simulated annealing** – stochastic mass transfers with a slowly cooling
   temperature help to escape any remaining local minima.

All phases respect the global deadline.  The best sequence found is finally
evaluated with ``evaluate_sequence`` and returned as a plain ``list`` of
non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    out = e / e.sum()
    return out


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector that is already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "random":
        a = rng.random(n)
    else:  # fallback
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int, fft_len: int, time_budget: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap sweep over many random parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + time_budget

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random Dirichlet search (another cheap global explorer)
# ----------------------------------------------------------------------


def dirichlet_search(n: int, fft_len: int, time_budget: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Search using Dirichlet vectors with a broad range of concentration parameters."""
    best_a = None
    best_score = np.inf

    start = time.time()
    deadline = start + time_budget
    while time.time() < deadline:
        # concentration α chosen log‑uniformly between 10⁻³ and 10²
        log_alpha = rng.uniform(-3.0, 2.0)
        alpha = 10.0 ** log_alpha
        a = rng.gamma(alpha, 1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    # fallback to uniform if nothing found (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len, n)
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30) -> tuple[float,
                                                                          np.ndarray]:
    """Smoothed max of a*a where a = softmax(y).  Returns (obj, grad_y)."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-14,
                                    "disp": False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adaptive sub‑gradient descent with exact line‑search
# ----------------------------------------------------------------------


def subgradient_descent_adaptive(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution with a
    simple back‑tracking line search.  The direction is the negative
    sub‑gradient; after each step we project back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    while time.time() < deadline:
        # ---------- compute sub‑gradient ----------
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # ---------- descent direction ----------
        d = -g                              # move opposite to sub‑gradient

        # ---------- back‑tracking line search ----------
        cur_score = fast_score(a, fft_len_conv, n)
        step = 1.0
        improved = False
        while step > 1e-8:
            a_candidate = simplex_projection(a + step * d)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a_candidate.copy()
                break
            step *= 0.5

        # if even the smallest step did not improve, take a tiny move
        if not improved:
            step = 0.05
            a_candidate = simplex_projection(a + step * d)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a_candidate.copy()
        # loop again
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (stochastic)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a temperature
    schedule.  Moves transfer a random fraction (up to 30 %) of the mass at
    a source index to a destination index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature = a modest fraction of the current score
    T0 = 0.03 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # Linear cooling based on remaining fraction of budget
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue

        # move a random fraction (up to 30 %) from i to j
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise with uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a previous construction is available, use it as a warm start
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – global cheap explorations
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # 15 % of total budget for the parametric sweep
    sweep_budget = 0.15 * budget_s
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, sweep_budget, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # another 10 % for Dirichlet random search
    dirichlet_budget = 0.10 * budget_s
    a_dir, sc_dir = dirichlet_search(n, fft_len_conv, dirichlet_budget, rng)
    if sc_dir < best_score:
        best_a, best_score = a_dir, sc_dir
        print(f"[sweep] dirichlet   best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping – split remaining budget among the remaining phases
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # give 45 % to L‑BFGS, 20 % to adaptive sub‑gradient, 15 % to annealing
    phase1_end = now + 0.45 * remaining
    phase2_end = phase1_end + 0.20 * remaining
    phase3_end = phase2_end + 0.15 * remaining
    # any leftover goes to the final annealing stage

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation (smooth optimisation)
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06
    ]
    maxiter_per_lam = 1500   # more inner iterations per λ

    while time.time() < phase1_end:
        # Randomly perturb the current best or start from a fresh random vector
        if rng.random() < 0.30:
            a_start = best_a.copy()
            a_start = a_start * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – adaptive projected sub‑gradient descent with line‑search
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent_adaptive(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (stochastic)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
