def smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    # gradient:
    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n-1):(n-1+n)]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y
```

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

        # mixing ratio decreasing

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

Efficient non‑negative step‑function optimizer.

The routine searches for a probability vector a (sum = 1) that
minimises 2·n·max(conv(a,a)).  It combines several optimisation
ingredients:

*   diverse initialisations (uniform, spectral, random, previous best)
*   continuation L‑BFGS on a log‑sum‑exp smoothed max‑convolution
    with a decreasing smoothing factor
*   a linear‑programming refinement step: for the current a,
    maximise ∑bⱼ subject to a∗b ≤ M (M = max(a∗a)).  The optimum b
    is mixed back into a, guaranteeing a monotone reduction of the
    max‑convolution bound.
*   exponentiated‑gradient descent on the exact max‑convolution
*   projected sub‑gradient descent (optional)
*   a final greedy hill‑climb for polishing.

The algorithm respects a 1000 s wall‑clock budget, checks the time
periodically and returns the best sequence found before the deadline.
All helper functions are defined at top level (no closures) and the
code only uses NumPy/SciPy.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def compute_conv(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full (linear) self‑convolution of a; returns length 2·n‑1 slice."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(conv) for a already normalised to sum = 1."""
    conv = compute_conv(a, fft_len, n)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # a*a (full)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (smooth → exact max)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule: list[float],
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Very unlikely, just continue
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        conv = compute_conv(a, fft_len, n)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print occasional progress
            if it % 500 == 0:
                print(f"[exp‑grad] iter={it} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (optional refinement)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on max(conv) with Euclidean projection
    onto the simplex after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    step0 = 0.3
    while time.time() < deadline:
        conv = compute_conv(a, fft_len, n)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # direction that reduces the max while preserving the sum
        d = -(g - g.mean())
        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (polish)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a small fraction of the mass from one entry to another.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------


def lp_refine_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise  ∑bⱼ   s.t.  a∗b ≤ M , bⱼ ≥ 0 .
    Returns the normalised optimiser b (sum = 1) or None on failure.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # first column of the Toeplitz matrix: a followed by zeros
    c = np.concatenate([a, np.zeros(m - n)])
    # first row: a₀ at position 0, rest zeros
    r = np.zeros(n)
    r[0] = a[0]
    A = toeplitz(c, r)                     # shape (m, n)

    # constraints A @ b <= M
    b_ub = M * np.ones(m)

    # maximise ∑bⱼ  → minimise -∑bⱼ
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    return b / s


def lp_refine(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based refinement.  Each iteration:
        * compute M = max(a∗a)
        * solve the LP for a direction b
        * mix a ← (1‑t)·a + t·b   with a diminishing step‑size
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        conv = compute_conv(a, fft_len, n)
        M = conv.max()
        b = lp_refine_step(a, M)
        if b is None:
            break

        # mixing coefficient – larger early, shrinking later
        t = 0.6 / np.sqrt(it + 1.0)
        a = (1 - t) * a + t * b
        a = simplex_projection(a)   # protect against numerical drift

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lp‑refine] iter={it} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation utilities
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the FFT
    of a uniform target (length 2·n‑1).  The result is projected onto
    the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # ------------------------------------------------------------------
    # reproducibility
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size (larger n generally allows a lower constant)
    # ------------------------------------------------------------------
    n = 4000                     # trade‑off between quality and speed
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # build a set of diverse starting points
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if present)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)
    candidates.append(a0)

    # 2️⃣  Spectral guess
    candidates.append(spectral_init(n))

    # 3️⃣  Random draws
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # pick the best of the crude candidates
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on a log‑sum‑exp smooth max
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.15,
        0.1,
        0.07,
        0.04,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        0.00005,
    ]
    a_lbfgs, sc_lbfgs = lbfgs_continuation(
        best_a,
        lam_schedule,
        maxiter_per_lam=1500,
        deadline=phase1_deadline,
        fft_len=fft_len,
        n=n,
    )
    if sc_lbfgs < best_score:
        best_score = sc_lbfgs
        best_a = a_lbfgs.copy()

    # --------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement (the new ingredient)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.45 * budget_s
    if time.time() < phase2_deadline:
        a_lp, sc_lp = lp_refine(best_a, phase2_deadline, fft_len, n)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()

    # --------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent on the exact max‑conv
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    if time.time() < phase3_deadline:
        a_exp, sc_exp = exp_gradient_optimize(
            best_a, phase3_deadline, n, fft_len, rng
        )
        if sc_exp < best_score:
            best_score = sc_exp
            best_a = a_exp.copy()
            print(f"[exp‑grad] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – (optional) projected sub‑gradient descent
    # --------------------------------------------------------------
    phase4_deadline = start_time + 0.90 * budget_s
    if time.time() < phase4_deadline:
        a_sub, sc_sub = subgradient_optimize(best_a, phase4_deadline, n, fft_len)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            print(f"[subgrad] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb for polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (converted to a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
