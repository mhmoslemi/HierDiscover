"""
Hybrid optimizer for the step‑function bound (improved version).

Features
--------
* Uses a larger support size (default n = 8000) which gives the optimiser
  many more degrees of freedom to flatten the self‑convolution.
* Phase 1 – many random restarts of L‑BFGS on a log‑sum‑exp smoothing of the
  max‑convolution.  The smoothing parameter λ is driven down to 1e‑5,
  allowing the optimiser to see a very sharp approximation of the true max.
* Phase 2 – long exponentiated‑gradient descent on the exact max‑convolution.
  The update a ← a·exp(−η g) stays on the probability simplex automatically.
* Phase 3 – two deterministic hill‑climbs:
    - “partner” hill‑climb (the same idea as the reference code).
    - “gradient” hill‑climb that moves mass from the coordinate with the
      largest positive gradient of the max‑entry to the one with the most
      negative gradient.
* All phases respect a global 1000 s budget, the deadline being split
  30 % / 60 % / 10 % across the three phases.
* The code interpolates a previous solution (if the variable
  ``height_sequence_1`` is present) to the new dimension, giving a warm
  start that often lies near a good local optimum.

The function ``propose_candidate`` returns the best list of floats found
before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a has been normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for softmax
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # learning rate η_t = 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Deterministic mass transfer from the coordinate that contributes most
    to the current maximal convolution entry to a coordinate with the
    smallest partner value. Accept only strict improvements.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # smallest partner (avoid i_max)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.8, min_step)

    return a, score


def gradient_hill_climb(a: np.ndarray,
                        score: float,
                        fft_len: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        init_step: float = 0.12,
                        min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate with the largest
    positive gradient of the max‑convolution to the one with the most
    negative gradient.  This directly follows the direction of steepest
    descent for the exact max‑objective.
    """
    step = init_step
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        i_pos = int(np.argmax(g))   # most positive gradient – we want to reduce a_i
        i_neg = int(np.argmin(g))   # most negative gradient – we want to increase a_i

        if g[i_pos] <= -g[i_neg] + 1e-14:
            # No clear descent direction; reduce step size and continue
            step = max(step * 0.7, min_step)
            continue

        delta = a[i_pos] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_pos] -= delta
        a_new[i_neg] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size (or gently increase)
            step = min(step * 1.03, init_step)
        else:
            step = max(step * 0.7, min_step)

    return a, score

# Random initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the
    FFT of a uniform target (length 2·n‑1).  The result is projected onto
    the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # ------------------------------------------------------------------
    # Random generator
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we use a larger support than the reference code.
    # ------------------------------------------------------------------
    n = 8000                      # length of the step function (feel free to change)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous solution if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # Additional diverse starts
    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    phase1_deadline = start_time + 0.30 * budget_s
    phase2_deadline = start_time + 0.80 * budget_s
    phase3_deadline = start_time + 0.96 * budget_s
    phase4_deadline = deadline  # final fine‑tuning

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS on smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.45:
            a_start = random_initial(n, rng)
        elif r < 0.85:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.015), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic hill‑climbs
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        # partner‑based hill climb (identical to the reference)
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_deadline,
            init_step=0.20,
            min_step=0.001,
        )
        print(f"[guided hill‑climb] intermediate score={best_score:.6f}")

    # Phase 4 – gradient‑based hill climb for a final polish
    if time.time() < phase4_deadline:
        best_a, best_score = gradient_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase4_deadline,
            init_step=0.12,
            min_step=1e-5,
        )
        print(f"[gradient hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # Make the result available for possible future runs
    # (the platform may store globals between invocations)
    globals()['height_sequence_1'] = final_seq

    return final_seq
