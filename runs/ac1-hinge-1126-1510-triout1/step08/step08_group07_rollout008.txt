def solve_lp_max_sum(a, M):
    n = len(a)
    rows = 2*n-1
    # Build matrix A as described. SciPy's linprog expects constraints: A_ub @ x <= b_ub.
    # Our constraint: conv = A @ b <= M * np.ones(rows). So A_ub = A, b_ub = M * np.ones(rows)
    # Also b >= 0 (bounds (0, None)).
    # Objective: maximize sum(b) -> minimize -sum(b)
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    # col for Toeplitz
    col = np.concatenate([a, np.zeros(rows - n)])
    row = np.concatenate([np.array([a[0]]), np.zeros(n-1)])
    A = toeplitz(col, row)   # shape (rows,n)
    # Maybe use np.float64.

    c = -np.ones(n)   # minimize -sum(b)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(rows)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return uniform maybe.
        print("[LP] warning: not successful", res.message)
        return np.full(n, 1.0/n)
    b_opt = res.x
    return b_opt

#!/usr/bin/env python
"""
Propose a low‑score step‑function sequence.

The search combines three complementary phases:

1️⃣  Frank‑Wolfe‑style updates:
    – At the current point `a` (sum = 1) compute the maximal
      convolution value `M = max(a * a)`.
    – Solve a linear programme that maximises the ℓ1‑norm of a new
      vector `b ≥ 0` under the linear constraints `a * b ≤ M`.
      (`a * b` is a linear convolution, built as a Toeplitz matrix.)
    – Normalise `b` to the simplex (`g = b / Σb`) and mix
      `a ← (1−t)·a + t·g` (few candidate mixing factors `t` are tried).
    – Repeat while time permits.  This is the exact analogue of the
      iteration described in the “state‑of‑the‑art” description and
      often yields a rapid drop of the objective.

2️⃣  Exponentiated‑gradient descent on the simplex (mirror descent)
    using the true sub‑gradient of the max‑convolution.  This phase
    refines the distribution and removes any remaining “flat‑spots”.

3️⃣  Deterministic hill‑climb with a tiny line‑search on the moved mass,
    identical to the polishing stage used previously.

All phases maintain non‑negativity and normalise the sequence to
Σa = 1, therefore the final score returned by `evaluate_sequence`
is scale‑invariant.  The total runtime is kept below the 1000 s budget
by explicit deadline checks.

The algorithm works for any length; the default `n = 4000` matches the
previous best construction but can be changed without breaking the code.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective when Σa = 1:
       2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase‑1: LP‑based Frank‑Wolfe update
# ----------------------------------------------------------------------


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP
        maximise Σ b_j
        s.t.   (a * b)_k ≤ M   for all k,
               b_j ≥ 0.
    Returns the optimal b (may be dense).  The convolution matrix
    (a * b) is built as a dense Toeplitz matrix; for n ≈ 4000 this
    fits comfortably into memory (~250 MiB).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Toeplitz matrix T where T @ b = a * b (linear convolution)
    col = np.concatenate([a, np.zeros(rows - n, dtype=np.float64)])
    row = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)])
    T = toeplitz(col, row)                     # shape (rows, n)

    # Linear programme data
    c = -np.ones(n)                            # minimise -Σb  ⇔ maximise Σb
    bounds = [(0.0, None)] * n
    # Slightly relax the RHS to avoid numerical infeasibility
    rhs = M * (1.0 + 1e-12) + 1e-12
    b_ub = np.full(rows, rhs, dtype=np.float64)

    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'quiet': True})
    if not res.success:
        # Fall back to the uniform vector – it is always feasible.
        print("[LP] warning: solver failed ({}) – use uniform".format(res.message))
        return np.full(n, 1.0 / n, dtype=np.float64)

    return res.x.astype(np.float64)


def lp_frank_wolfe_step(a: np.ndarray, fft_len: int, n: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform one Frank‑Wolfe‑style iteration:
        • compute current M,
        • solve the LP,
        • try several mixing factors and keep the best improvement.
    Returns the (possibly) updated a and its fast_score.
    """
    # current convolution & M
    conv = full_conv(a, fft_len)
    M = conv.max()

    # Solve LP (may be a bit expensive)
    b_opt = solve_lp_max_sum(a, M)

    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return a, fast_score(a, fft_len, n)

    g = b_opt / sum_b                     # normalised to Σg = 1

    # Try a handful of mixing coefficients.
    best_a = a
    best_score = fast_score(a, fft_len, n)

    for t in (0.25, 0.5, 0.75, 0.9):
        a_candidate = (1.0 - t) * a + t * g
        score_candidate = fast_score(a_candidate, fft_len, n)
        if score_candidate < best_score - 1e-12:
            best_score = score_candidate
            best_a = a_candidate

    # If none of the tried mixes improve, keep the original a.
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------


def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                          fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the exact
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution (size n)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # EG update
        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: Fine‑grained hill‑climb with line‑search
# ----------------------------------------------------------------------


def targeted_hill_climb_line_search(a: np.ndarray, current_score: float,
                                    fft_len: int, n: int,
                                    deadline: float,
                                    max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb that, for each selected (source, target)
    pair, tries a few transfer fractions and keeps the best strictly
    improving move.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    fractions = [0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (unsmoothed) max
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # pick source/high‑gradient and target/low‑gradient indices
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑normalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s
