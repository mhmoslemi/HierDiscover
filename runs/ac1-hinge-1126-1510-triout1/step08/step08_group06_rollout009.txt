"""
Propose a non‑negative step sequence with a low evaluation score.

Algorithm
---------
1. Initialise a distribution on the probability simplex.
   – use the previously best sequence if available,
   – otherwise start from a uniform distribution.
2. Very cheap parametric sweep (power‑law, exponential, Gaussian,
   beta‑like, sinusoidal) to obtain a reasonable starting point.
3. Refine the distribution with an ADAM optimiser on a *smooth*
   approximation of the max‑convolution (log‑sum‑exp).  The smoothing
   factor λ is decreased in stages (0.5 → 1e‑5) and many Adam steps
   (≈15 000) are performed while always projecting back onto the simplex.
4. A deterministic hill‑climb moves a small fraction of mass from the
   coordinate with the largest sub‑gradient (most harmful) to the one with
   the smallest sub‑gradient (most helpful).  Strictly improving moves are
   kept and the process stops when time runs out.
5. Finally evaluate the best sequence with the supplied ``evaluate_sequence``
   and return it as a plain list of floats.

All phases respect the global 1000 s budget.  The implementation is fully
self‑contained, uses only NumPy / SciPy and runs on a single thread.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
from scipy import sparse


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. a)
# ----------------------------------------------------------------------


def smooth_max_grad(a: np.ndarray, lam_factor: float, fft_len: int):
    """
    Compute (smooth_max, gradient) of 2·n·max(a*a) with log‑sum‑exp smoothing.
    Returns (smooth_max, grad_a) where grad_a is the gradient w.r.t. a.
    """
    n = a.shape[0]
    # full convolution of a with itself
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # weights d(smooth_max)/d conv_i
    w = exp_shift / Z

    # back‑propagation: gradient w.r.t. a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]

    # final objective 2·n·smooth_max
    obj = 2.0 * n * smooth_max
    return obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    iters_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of the max‑convolution.
    Returns the best a (sum‑to‑1) and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # learning‑rate scaling (will be reduced together with lam)
    base_lr = 0.2

    for lam_factor in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            # compute smooth objective and gradient
            obj, grad = smooth_max_grad(a, lam_factor, fft_len)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # step size – we shrink it for very small lambdas
            lr = base_lr * (lam_factor + 0.01)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = a - step
            a_candidate = simplex_projection(a_candidate)

            cur_score = fast_score(a_candidate, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_candidate.copy()

            a = a_candidate

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the most harmful coordinate (largest
    sub‑gradient) to the most helpful one (smallest sub‑gradient).  Only
    strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # true max‑convolution and sub‑gradient
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # try a few fractions and keep the best improving one
        best_local_score = best_score
        best_local_move = None
        for _ in range(3):
            frac = rng.random() * 0.20  # up to 20 %
            delta = best[i_hi] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i_hi] -= delta
            cand[i_lo] += delta
            new_score = fast_score(cand, fft_len, n)
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_move = (cand, new_score)

        if best_local_move is not None:
            best, best_score = best_local_move
        # else we did not find an improving move – stop early
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (can be tuned – 4000 gives a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(3 * n - 2)          # length sufficient for all convolutions

    # ------------------------------------------------------------------
    # Initialise from the previous best if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                old_x = np.linspace(0, 1, hs.size)
                new_x = np.linspace(0, 1, n)
                a0 = np.interp(new_x, old_x, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    def try_update(a: np.ndarray):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    # power‑law families
    for p in np.linspace(-2.0, 5.0, 15):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        try_update(candidate_exponential(n, lam, False))
        try_update(candidate_exponential(n, lam, True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 7):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # symmetric beta‑like shapes (covers U‑shaped and bell‑shaped)
    for alpha in np.linspace(0.2, 5.0, 12):
        try_update(candidate_beta_symmetric(n, alpha))

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update(candidate_sine_power(n, p))

    # a few random restarts (quick)
    while time.time() < start_time + 0.08 * budget_s:
        try_update(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Allocate time for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # Adam optimisation
    phase2_end = start_time + 0.85 * budget_s   # additional random‑restart Adam attempts
    # remaining time → deterministic hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – ADAM optimisation on smoothed max‑convolution
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-05
    ]
    iters_per_lam = 800

    while time.time() < phase1_end:
        # occasional jitter around the current best
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_schedule,
            iters_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[ADAM] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – additional Adam runs with a tighter schedule (more iterations)
    # ------------------------------------------------------------------
    extra_schedule = [0.00005, 2e-05, 1e-05, 5e-06]
    extra_iters = 1200

    while time.time() < phase2_end:
        a_start = add_noise(best_a, sigma=rng.uniform(0.0005, 0.008), rng=rng)
        a_opt, sc_opt = adam_optimize(
            a_start,
            extra_schedule,
            extra_iters,
            phase2_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[ADAM‑extra] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len,
        n,
        deadline,
        max_iter=50000,
    )
    print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
