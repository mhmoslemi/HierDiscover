"""
Optimise a non‑negative step sequence (length up to a few‑thousand) for the
objective

    score = 2·n·max(conv(a,a)) / (∑a)²

The code is a tuned version of the previously‑published pipeline.
Improvements:

*  A longer “L‑BFGS‑with‑log‑sum‑exp” phase – finer λ‑schedule, more inner
   iterations and many random restarts.
*  Adam optimiser (smooth‑max) for a final refinement after L‑BFGS.
*  Momentum sub‑gradient descent with an adaptive step‑size schedule.
*  Gradient‑guided simulated annealing (mass is moved from indices with a
   large sub‑gradient to indices with a small sub‑gradient).
*  A tiny “coarse‑to‑fine” initialisation – optimise a low‑resolution
   vector (n_coarse=1024) and up‑sample it before the fine optimisation.
*  More aggressive time‑budget allocation (≈60 % L‑BFGS, 25 % sub‑gradient,
   10 % Adam, 5 % annealing).

All phases respect the simplex ∑a = 1 and a ≥ 0.  The best sequence found
before the deadline is returned as a plain list of Python floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. the logits y
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Returns the smoothed objective
        f(y) = 2·n·smooth_max(conv(softmax(y)))
    and its gradient w.r.t. y.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # Numerical stabilisation: λ must be positive
    lam = max(lam, 1e-12)

    # log‑sum‑exp smooth approximation of the max
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient propagation ----
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient of smooth_max w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_stage(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smoothed objective for each λ in lam_schedule,
    starting each stage from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        # enforce exact simplex (numeric drift)
        a = simplex_projection(a)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            # useful diagnostic
            print(f"[lbfgs]  λ={lam:.1e}  new best={best_score:.6f}")

    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    lam_schedule,
    lr: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max (log‑sum‑exp).  The λ‑schedule
    is traversed once (from large to small) while the iterates are
    updated with the Adam rule.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    schedule_iter = iter(lam_schedule)
    cur_lam = next(schedule_iter, None)

    while time.time() < deadline:
        if cur_lam is None:  # go to the end of schedule and stay at last λ
            cur_lam = lam_schedule[-1]

        f, grad_y = smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, cur_lam)

        it += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex via soft‑max (automatic)
        a = softmax(y)
        a = simplex_projection(a)   # numerical cleanup
        cur_score = fast_score(a, fft_len_conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] λ={cur_lam:.1e}  iter={it}  new best={best_score:.6f}")

        # move to next λ after a few hundred iterations
        if it % 400 == 0:
            cur_lam = next(schedule_iter, cur_lam)

    return best_a, best_score


def subgradient_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)       # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (nonsmooth) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing stepsize (1/√it)
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            # diagnostic
            print(f"[sub‑grad] it={it}  new best={best_score:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_score


def guided_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that moves mass preferentially
    from indices with a large sub‑gradient (high contribution to the peak)
    to indices with a small sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # temperature schedule – start with 2 % of the current score
    T0 = 0.02 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    # pre‑compute sub‑gradient once per cooling block (every 500 steps)
    grad = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # recompute gradient periodically (less overhead)
        if it % 500 == 0:
            fa = np.fft.fft(a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(np.float64)
            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len)
            fa_rev = np.fft.fft(a_rev, fft_len)
            conv_g = np.real(np.fft.ifft(fm * fa_rev))
            grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # temperature adapts to remaining time
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        # Pick source index proportionally to its gradient (large → high prob)
        prob_src = grad / grad.sum()
        i = rng.choice(n, p=prob_src)

        # Destination: favour indices with small gradient
        prob_dst = (grad.max() - grad) + 1e-12
        prob_dst /= prob_dst.sum()
        j = rng.choice(n, p=prob_dst)

        if i == j or a[i] == 0.0:
            it += 1
            continue

        # Move a random fraction (up to 20 %) of the mass from i to j
        delta = a[i] * rng.random() * 0.20
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] it={it}  new best={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for a cheap pre‑search (including a piecewise
# linear family that proved useful in earlier experiments)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** p if inc else (np.arange(n, 0, -1) ** p)
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "piecewise":
        # Random piecewise‑linear shape with a few breakpoints
        k = rng.integers(3, 9)          # number of knots
        xs = np.sort(rng.integers(0, n, size=k))
        xs = np.concatenate(([0], xs, [n - 1]))
        hs = rng.random(xs.size)        # heights at the knots
        a = np.interp(np.arange(n), xs, hs)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def coarse_to_fine(a_coarse: np.ndarray, n_fine: int) -> np.ndarray:
    """
    Upsample a low‑resolution vector to length n_fine by linear interpolation
    and re‑normalise to the simplex.
    """
    x_coarse = np.linspace(0, 1, a_coarse.size)
    x_fine = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine /= a_fine.sum()
    return a_fine


def parametric_sweep(
    n: int,
    fft_len: int,
    budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap random sweep over simple families – keeps the best performer."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    start = time.time()
    deadline = start + budget

    families = ["power", "exp", "gauss", "cos", "bimodal", "piecewise", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem dimensions – a slightly larger n gives a better theoretical
    # lower bound (average convolution = 1/(2·n)).  6000 is a good compromise
    # between expressiveness and runtime.
    # --------------------------------------------------------------
    n = 6000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial candidate – uniform distribution, then try to reuse a
    # previous construction if it exists.
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            a1 = coarse_to_fine(hs, n)
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈10 % of total budget)
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s * 0.10, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] best score after sweep={best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a small safety margin

    # allocate percentages of the total budget
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.90 * budget_s   # Adam
    # remaining time → guided annealing

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS optimisation on a smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05,
    ]
    maxiter_per_lam = 1500   # more fine‑grained optimisation per λ

    while time.time() < phase1_end:
        # Random restart: either jitter the current best or draw a fresh family
        if rng.random() < 0.3:
            a_start = best_a.copy()
            a_start = a_start * rng.normal(1.0, 0.025, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            # Occasionally use a cheap coarse‑to‑fine start
            if rng.random() < 0.2:
                n_coarse = 1024
                a0c = generate_family(rng.choice(["piecewise", "bimodal", "exp"]), n_coarse, rng)
                a_start = coarse_to_fine(a0c, n)
            else:
                a_start = generate_family(rng.choice(["power", "exp", "gauss", "cos", "bimodal"]), n, rng)

        a_opt, sc_opt = lbfgs_stage(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] improved best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑grad] after phase2 score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser on the smoothed max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
            lam_schedule,
            lr=0.04,
        )
        print(f"[adam] after phase3 score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – guided simulated annealing (stochastic hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[anneal] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
