def simulated_annealing(a, best_score, best_a, deadline):
    cur_a = a.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    # temperature: start high = maybe 0.1 * best_score
    init_temp = 0.05 * best_score  # relative?
    final_temp = 1e-6 * best_score
    start = time.time()
    total_time = deadline - start
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # linearize temperature schedule:
        T = init_temp * (1 - elapsed/total_time) + final_temp * (elapsed/total_time)
        # choose source and destination:
        src = rng.choice(n, p=cur_a)  # weighted by a
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # step size: proportional to cur_a[src]
        max_delta = cur_a[src]
        # choose delta random in [0, max_delta * step_factor] where step_factor decays
        step_factor = max(0.1 * (1 - elapsed/total_time), 0.001) # fade
        delta = max_delta * rng.random() * step_factor
        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # renormalise (maybe small floating errors)
        # no need because total sum stays: a_new.sum() = cur_a.sum()
        # compute score:
        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        it += 1
    return best_a, best_score

"""
Hybrid optimizer with additional simulated‑annealing refinement.

The bound to minimise is
    R(a) = 2·n·max(a∗a) / (sum a)²
which is invariant to scaling, so we work on the probability simplex
(sum a = 1, a_i ≥ 0).  The algorithm consists of four phases:

1.  Random‑restart L‑BFGS on a log‑sum‑exp smoothed version of
    the max‑convolution.  This quickly finds a good basin.
2.  Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
3.  Simulated‑annealing with random mass transfers to escape possible
    local minima.
4.  Deterministic hill‑climb that moves mass from the “most harmful’’
    coordinate to the “least harmful’’ one (the same routine used
    previously).

All phases respect a common 1000 s budget; the best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of   R(a) = 2·n·max(a*a)   with   sum(a)=1.
    (The denominator is omitted because sum a is 1.)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the solution of the previous stage.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # diminishing step size
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                       # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Spectral initialization – inverse FFT of square‑root of the FFT
    of a uniform target.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small fraction of mass
    from the coordinate that contributes most to the current maximal
    convolution entry to the coordinate with the smallest partner.
    Accept only strict improvements, shrinking the step size when
    necessary.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size for the next move
        else:
            step = max(step * 0.8, min_step)

    return a, score


def simulated_annealing_optimize(a_start: np.ndarray,
                                 best_a: np.ndarray,
                                 best_score: float,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 fft_len: int,
                                 n: int) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing.
    A fraction of mass is moved from a source index (chosen proportionally
    to its current mass) to a uniformly random destination.
    Moves are accepted according to a decreasing temperature schedule.
    The best sequence seen is returned.
    """
    cur_a = a_start.copy()
    cur_score = fast_score(cur_a, fft_len, n)

    if cur_score < best_score:
        best_score = cur_score
        best_a = cur_a.copy()

    start = time.time()
    total = max(deadline - start, 1e-12)

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # temperature linearly decays from a fraction of the current score
        T = (0.1 * cur_score) * (1.0 - frac) + 1e-12

        # source proportional to current mass, destination uniform
        src = rng.choice(n, p=cur_a)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        max_delta = cur_a[src]
        # step size slowly shrinks
        step_factor = max(0.1 * (1.0 - frac), 0.001)
        delta = max_delta * rng.random() * step_factor
        if delta <= 0.0:
            continue

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new /= a_new.sum()   # re‑normalize to avoid drift

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

        iteration += 1
    return best_a, best_score

    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)

            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)

        if sc < best_score:

    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    start_time = time.time()

    phase3_deadline = start_time + 0.90 * budget_s   # annealing

    maxiter_per_lam = 800   # a bit tighter to stay within budget

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.4:
            a_start = random_initial(n, rng)
        elif r < 0.8:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.015), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (longer run)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[exp‑grad] score after phase 2 = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing to escape local minima
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = simulated_annealing_optimize(
            best_a,
            best_a,
            best_score,
            phase3_deadline,
            rng,
            fft_len_conv,
            n,
        )
        print(f"[annealing] best score after phase 3 = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
            init_step=0.20,
            min_step=0.001,
        )
        print(f"[final hill‑climb] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (real evaluation for safety)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
