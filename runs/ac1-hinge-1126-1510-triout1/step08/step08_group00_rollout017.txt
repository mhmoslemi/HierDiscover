"""
Propose a step‐function sequence with a low evaluation score.

The optimizer works in several stages:
1.  A cheap parametric sweep over many simple families (power, exp,
    Gaussian, cosine, Beta, sinusoid, bimodal, fully random) keeps the
    best candidate.
2.  Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The schedule of smoothing parameters is long and
    the inner iteration count is increased compared to the reference
    implementation.
3.  Projected sub‑gradient descent with momentum on the *true* max‑convolution
    (no smoothing) refines the best point.
4.  Exponentiated‑gradient descent (mirror descent on the simplex) adds
    another deterministic refinement.
5.  Finally a long simulated‑annealing hill‑climb (mass moves) tries to
    escape any remaining local minima.

The whole routine respects the 1000 s budget, uses only numpy / scipy,
and always returns a list of non‑negative floats (the sequence is kept
on the simplex ∑a_i = 1, so the evaluation function reduces to
2·n·max(a∗a)).  The algorithm is deliberately more aggressive than the
reference version – a larger problem size (n = 8000) and many more
iterations are used to push the score below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (⟹ simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the linear part (length 2·len(a)−1)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) – exact score when ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution) and its gradient
# ----------------------------------------------------------------------


def smooth_objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(a*a).  a = softmax(y) lives on the simplex.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter – proportional to the current max value
    lam = max_c * lam_factor + 1e-14

    # log‑sum‑exp around the maximum
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. the convolution entries
    w = exp_shift / Z                 # d smooth_max / d conv_i
    # back‑prop through the convolution (a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the central n entries affect the gradient w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # back‑prop through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Deterministic optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS for each λ in a decreasing schedule."""
    y = np.log(a_start + 1e-12)      # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] improvement -> {best_score:.6f}")
    return best_a, best_score


def subgradient_refine(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the *true* max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)    # momentum buffer
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size
        step = 0.6 / np.sqrt(it + 1.0)
        a_new = simplex_projection(a - step * v)

        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[sub‑grad] new best -> {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


def exp_gradient_refine(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent directly on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing learning rate
        eta = 0.3 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[exp‑grad] new best -> {best_score:.6f}")

        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climb with a linear cooling schedule.  A move transfers a
    random fraction (up to 15 %) of the mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature: a small fraction of the current score
    T0 = 0.02 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remaining = max(deadline - time.time(), 0.0)
        T = T0 * (remaining / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue

        # move a random fraction (≤15 %) of the mass from i to j
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-15)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] new best -> {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep (expanded set)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalized non‑negative vector drawn from a parametric family."""
    x = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        a = np.arange(1, n + 1, dtype=float) ** p if inc else np.arange(n, 0, -1, dtype=float) ** p

    elif family == "exp":
        rate = rng.uniform(0.0002, 0.03)
        a = np.exp(-rate * x)

    elif family == "gauss":
        sigma = rng.uniform(100.0, 2500.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)

    elif family == "cos":
        width = rng.uniform(100.0, n / 1.5)
        a = np.cos(np.pi * (x - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)

    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
        a = a1 + a2

    elif family == "beta":
        # Parameters α,β > 0
        alpha = rng.uniform(0.5, 5.0)
        beta = rng.uniform(0.5, 5.0)
        t = x / (n - 1)
        a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))

    elif family == "sin":
        power = rng.uniform(0.5, 5.0)
        a = np.sin(np.pi * x / (n - 1)) ** power
        a = np.maximum(a, 0.0)

    else:  # fallback – pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    sweep_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Rapid exploration of many random families.  Returns the best vector
    (already normalised) and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + sweep_time

    families = [
        "power", "exp", "gauss", "cos", "bimodal",
        "beta", "sin", "random"
    ]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] {fam} improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects a global time limit (budget_s seconds) and
    returns the best list of floats found before expiry.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we use a larger n than the reference (8000)
    # ------------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise with uniform distribution
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If a previous construction is available, interpolate it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] imported best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Allocate phases (percentage of total budget)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    phase0_end = start_time + 0.10 * budget_s   # cheap sweep
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.90 * budget_s   # exp‑gradient
    # remaining time → annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    sweep_budget = max(phase0_end - time.time(), 0.0)
    if sweep_budget > 0.0:
        a_sweep, sc_sweep = parametric_sweep(
            n, fft_len_conv, sweep_budget, rng
        )
        if sc_sweep < best_score:
            best_a, best_score = a_sweep, sc_sweep
            print(f"[sweep] overall best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation (more inner steps)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.10,
        0.07, 0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally jitter around the current best
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new overall best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_refine(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑grad] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_refine(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑grad] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[anneal] final best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
