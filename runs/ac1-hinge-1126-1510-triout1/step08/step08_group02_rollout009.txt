"""
Improved search for a low‑score step‑function.

The algorithm proceeds in several stages (all respecting the 1000 s budget):

1. **Initial population** – a few diverse starts (spectral root of uniform,
   a random Dirichlet‑like draw, and the previous best if available).

2. **Adam on a smoothed max‑convolution** – we optimise the logits of the
   probability vector with Adam.  The objective is a log‑sum‑exp
   (smooth‑max) of the self‑convolution; a decreasing smoothing factor
   drives the solution towards a flat (low‑max) convolution.

3. **Exponentiated‑gradient descent** – an exact‑max sub‑gradient step that
   keeps the iterate on the simplex automatically (multiplicative update).

4. **Projected sub‑gradient descent with momentum** – a cheap Nesterov‑style
   refinement on the true max‑convolution.

5. **Greedy hill‑climbing** – a final polish that moves a small random
   fraction of mass between coordinates only if the score improves.

All heavy convolution work uses FFTs (real FFT) for speed.
The routine keeps the best sequence found before the deadline and returns
it as a list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  init_lr: float = 0.20,
                  beta1: float = 0.90,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on logits y (softmax(y) = a).  The objective is a
    log‑sum‑exp (smooth‑max) of the self‑convolution.
    """
    y = np.log(a_start + 1e-12)              # logits
    m = np.zeros_like(y)                    # first moment
    v = np.zeros_like(y)                    # second moment
    t = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # ---- convolution ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        # smoothing factor – start relatively large and decrease with sqrt(t)
        lam_factor = max(0.02 / np.sqrt(t), 0.001)
        lam = max_val * lam_factor + 1e-12

        # ---- smooth‑max weighting (log‑sum‑exp) ----
        shifted = conv - max_val
        # clip exponent to avoid under/overflow
        exp_shift = np.exp(np.clip(shifted / lam, -50, 0))
        Z = exp_shift.sum()
        w = exp_shift / Z                       # d smooth_max / d conv

        # ---- gradient w.r.t. a (back‑prop through convolution) ----
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
        g_vec = conv_rev[(n - 1):(n - 1 + n)]
        g_vec = 2.0 * g_vec                     # derivative of a*a

        # ---- gradient w.r.t. logits (softmax chain rule) ----
        avg = np.dot(a, g_vec)
        grad_y = a * (g_vec - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y = y - init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional evaluation ----
        if (t & 0x3FF) == 0:   # every 1024 iterations
            score = fast_score(a, fft_len, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] it {t:6d}, score {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent.
    The update a ← a·exp(−η·g) keeps the iterate on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (larger at the beginning)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Projected sub‑gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------


def subgrad_nesterov_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int) -> tuple[np.ndarray, float]:
    """
    Fast projected sub‑gradient descent on the exact max‑convolution.
    Uses Nesterov momentum (sum(d)=0) for faster decay.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    momentum = np.zeros_like(a)
    beta = 0.9                     # momentum factor
    it = 0
    step0 = 0.30                  # initial step size (will decay)

    while time.time() < deadline:
        # ---- compute max‑convolution and exact active mask ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- sub‑gradient of the max ----
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # direction that reduces the max while keeping the sum zero
        mean_g = np.mean(g)
        d = -(g - mean_g)               # Σ d = 0

        # Nesterov momentum
        momentum = beta * momentum + d
        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * momentum
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polish
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 12000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a tiny random fraction of mass from one index
    to another and keep the move only if it strictly improves the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo move
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Root‑of‑uniform spectral initialisation, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diverse collection of Dirichlet‑like random vectors."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max‑convolution
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * budget_s
    if time.time() < phase1_deadline:
        best_a, best_score = adam_optimize(best_a,
                                            phase1_deadline,
                                            n,
                                            fft_len,
                                            rng,
                                            init_lr=0.20)
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (exact max)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = subgrad_nesterov_optimize(best_a,
                                                       deadline,
                                                       n,
                                                       fft_len)
        print(f"[subgrad‑Nesterov] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=25000)
        print(f"[hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
