"""
Multi‑phase optimizer for the step‑function evaluation problem.

The optimizer works on a probability vector a (all entries ≥0, sum = 1).  
For a given a we have

    score = 2·n·max(a∗a)          (fast_score)

which equals the quantity returned by ``evaluate_sequence`` after the
usual scaling.  The goal is to minimise *score*.

The method consists of four stages, all respecting a 1000 s budget:

1. **Explicit shape sweep** – a small 2‑parameter family of U‑shaped
   densities (inverse‑power laws) is sampled densely.  The best shape
   provides a strong deterministic starting point.

2. **Heavy L‑BFGS** – optimisation of a smooth log‑sum‑exp surrogate
   of ``max(a∗a)`` with a decreasing schedule of smoothing parameters.
   Many random restarts exploit the diverse initial points from stage 1.

3. **Projected Adam** – sub‑gradient descent on the true nonsmooth
   objective with Euclidean projection onto the simplex.

4. **Aggressive hill‑climb** – a coordinate‑wise greedy move that
   transfers a small fraction of mass from “high‑gradient” to
   “low‑gradient” indices.  Several candidate fractions are tried and
   only strictly improving moves are kept.

Finally the best vector is converted to a plain list and fed to the
external ``evaluate_sequence`` function for the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1).  Returns 2·n·max(a*a) .
    """
    max_b = full_conv(a, fft_len).max()
    return float(2.0 * n * max_b)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal",
         "inv_sqrt", "prod_inv"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "inv_sqrt":
        # a_i ∝ 1/√(i+1) + 1/√(n-i)
        i = np.arange(1, n + 1, dtype=float)
        a = 1.0 / np.sqrt(i) + 1.0 / np.sqrt(n - i + 1)
    else:  # prod_inv
        # a_i ∝ 1/[(i)*(n-i+1)]^β with β≈0.3 by default
        beta = 0.3
        i = np.arange(1, n + 1, dtype=float)
        a = (i * (n - i + 1)) ** (-beta)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moment estimation) using the true
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√t
        step = 0.7 / np.sqrt(it)          # a bit larger than the previous 0.5
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to shake out plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.008)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def targeted_hill_climb_aggressive(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb using the sub‑gradient to move mass from high‑gradient
    (source) indices to low‑gradient (target) ones.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution & active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        frev = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(frev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Choose source (large gradient) and target (small gradient)
        high_thr = np.percentile(g, 90)
        low_thr = np.percentile(g, 10)
        high_idx = np.where(g >= high_thr)[0]
        low_idx = np.where(g <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # try a few transfer fractions
        improved = False
        for frac in (0.04, 0.09, 0.15, 0.25):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
                improved = True
                break
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Explicit low‑dimensional shape families
# ----------------------------------------------------------------------


def shape_power_inv(alpha: float, n: int) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ i^{-α}+ (n‑i+1)^{-α}."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-alpha) + (n - i + 1) ** (-alpha)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def shape_prod_inv(beta: float, n: int) -> np.ndarray:
    """Symmetric product‑inverse: a_i ∝ (i·(n‑i+1))^{-β}."""
    i = np.arange(1, n + 1, dtype=float)
    w = (i * (n - i + 1)) ** (-beta)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser returning a list of non‑negative floats.
    The schedule is deliberately aggressive to push the score below the
    previous best (≈1.5065).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                                   # length of the step function

    # initialise from a previously found solution if present
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # simple linear interpolation onto the new size
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2           # safety margin
    # allocate portions of the budget
    phase_shape_end = start + 0.12 * (budget_s - 0.2)   # explicit shape sweep
    phase_lbfgs_end = start + 0.40 * (budget_s - 0.2)   # heavy L‑BFGS
    phase_adam_end = start + 0.80 * (budget_s - 0.2)    # long Adam
    phase_hill_end = start + 0.95 * (budget_s - 0.2)    # hill‑climb
    # remaining -> fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 0 – deterministic shape sweep (few‑parameter families)
    # ------------------------------------------------------------------
    print("[phase 0] starting shape sweep")
    # Explore power‑law family
    for alpha in np.linspace(0.05, 2.0, 24):
        if time.time() >= phase_shape_end:
            break
        cand = shape_power_inv(alpha, n)
        score = fast_score(cand, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[shape] power α={alpha:.3f} → score={best_score:.6f}")

    # Explore product‑inverse family
    for beta in np.linspace(0.05, 0.9, 18):
        if time.time() >= phase_shape_end:
            break
        cand = shape_prod_inv(beta, n)
        score = fast_score(cand, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[shape] prod β={beta:.3f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    if time.time() < phase_lbfgs_end:
        print("[phase 1] heavy L‑BFGS")
        lam_schedule = [
            0.6, 0.4, 0.25, 0.15, 0.10, 0.06,
            0.04, 0.025, 0.015, 0.008, 0.004,
            0.002, 0.001, 5e-04, 2e-04, 1e-04,
        ]
        maxiter_per_lam = 700
        while time.time() < phase_lbfgs_end:
            # Occasionally perturb the current best to diversify starts
            if rng.random() < 0.3:
                sigma = rng.uniform(0.0015, 0.015)
                a_start = best_a + rng.normal(0.0, sigma, size=n)
                a_start = np.maximum(a_start, 0.0)
                a_start /= a_start.sum()
            else:
                # Try a fresh random initialisation
                a_start = random_initial(n, rng)

            a_opt, score_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase_lbfgs_end,
                fft_len_conv,
                fft_len_g,
                n,
            )
            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Projected Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase_adam_end:
        print("[phase 2] Projected Adam")
        best_a, best_score = subgradient_adam(
            best_a,
            phase_adam_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase_hill_end:
        print("[phase 3] aggressive hill‑climb")
        best_a, best_score = targeted_hill_climb_aggressive(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase_hill_end,
            max_iter=80000,
        )
        print(f"[hill‑climb] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short fine‑tuned L‑BFGS (very small λ)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        print("[phase 4] fine L‑BFGS")
        fine_lam_schedule = [5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06]
        fine_maxiter = 500
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[fine L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (counts clipping, NaN checks, …)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
