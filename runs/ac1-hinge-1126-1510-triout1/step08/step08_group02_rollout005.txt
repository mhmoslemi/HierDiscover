"""
Propose a low‑score non‑negative step function.

Algorithm overview
------------------
1️⃣  Initialise a population of candidate vectors (previous best,
    a spectral guess, a few random Dirichlet‑type draws) – all
    normalised to sum = 1.

2️⃣  Short L‑BFGS polishing on a smooth log‑sum‑exp surrogate of the
    max‑convolution (many iterations, gradually decreasing smoothing).

3️⃣  Long accelerated sub‑gradient descent (momentum + simplex projection)
    on the exact max‑convolution.  The step‑size decays as 1/√t and a
    small amount of Gaussian jitter is injected every 10 k iterations
    to avoid stagnation.

4️⃣  Final L‑BFGS refinement (very small smoothing) followed by a greedy
    hill‑climb for polishing.

All stages respect a global 1000 s budget and keep the best vector
found.  The returned list is evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """
    Build a guess by taking the inverse FFT of the square‑root of the
    flat spectrum (length 2·n‑1).  The result is projected onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing parameter in `lam_schedule`,
    starting each stage from the result of the previous stage.
    Returns the best vector encountered and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Rare numerical trouble – skip this lam
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs]   new best {best_score:.6f} (lam={lam})")
    return best_a, best_score


def momentum_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator,
                              beta: float = 0.9,
                              step0: float = 0.5):
    """
    Accelerated (momentum) projected sub‑gradient descent on the exact
    max‑convolution.  Returns the best vector seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    vel = np.zeros_like(a)               # momentum buffer
    it = 0
    while time.time() < deadline:
        # compute full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (exact)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # momentum update (we move *opposite* to the sub‑gradient)
        vel = beta * vel + (1.0 - beta) * g
        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a - alpha * vel
        a_new = simplex_projection(a_new)

        # evaluate
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1

        # occasional jitter to shake off plateaus
        if it % 10000 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = a / a.sum()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Simple greedy hill‑climbing: transfer a random fraction of mass
    from one coordinate to another, accepting only strictly improving
    moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the source mass
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of coefficients found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)        # power‑of‑two for fast FFTs

    # --------------------------------------------------------------
    # Initialise candidates (previous best > random > spectral)
    # --------------------------------------------------------------
    # 1.  Previous best, if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(globals()['height_sequence_1'], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, prev)
                a0 = np.clip(a0, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            else:
                a0 = np.full(n, 1.0 / n)
        except Exception:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # 2.  Spectral initial guess
    a_spec = spectral_init(n)

    # 3.  Random Dirichlet‑type draws (a few)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------- Phase 1: L‑BFGS polishing -----------------
    phase1_deadline = start_time + 0.15 * budget_s   # ~150 s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 2500                           # more iterations than before

    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------- Phase 2: accelerated sub‑gradient -------------
    phase2_deadline = start_time + 0.70 * budget_s   # bulk of the time
    if time.time() < phase2_deadline:
        best_a, best_score = momentum_subgrad_optimize(best_a,
                                                       phase2_deadline,
                                                       n,
                                                       fft_len,
                                                       rng,
                                                       beta=0.9,
                                                       step0=0.8)
        print(f"[momentum‑subgrad] score={best_score:.6f}")

    # --------------------- Phase 3: final L‑BFGS with fine smoothing -----
    if time.time() < deadline:
        # use a tighter smoothing schedule (smaller lambdas) and many iters
        fine_lam = [0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
        fine_maxiter = 4000
        fine_deadline = deadline - 0.2 * budget_s
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                            fine_lam,
                                            fine_maxiter,
                                            fine_deadline,
                                            fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final‑L‑BFGS] new best score={best_score:.6f}")

    # --------------------- Phase 4: greedy hill‑climb --------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=25000)
        print(f"[hill‑climb] final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the final sequence (list of Python floats)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # sanity evaluate with the provided evaluation function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
