"""
Optimise a non‑negative step‑function for the objective

    score = 2·n·max(convolve(a, a)) / (sum(a)²)

The algorithm mixes several cheap global explorations with a heavy
L‑BFGS continuation on a smoothed max‑objective, a short
exponentiated‑gradient phase, and a simulated‑annealing hill‑climb.
All phases respect the simplex (a ≥ 0, Σa = 1) – the evaluation function
is scale‑invariant, so we keep Σa = 1 throughout.

Time‑budget (≈1000 s) is split automatically; the best sequence found
before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation for a normalised (Σa = 1) vector:
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (softmax parametrisation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- stable softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing bandwidth
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient wrt convolution entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution (FFT‑based) -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # indices (n‑1) … (n‑1+n‑1) correspond to ∂/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, warm‑starting each stage.
    Returns the best a (already normalised) and its fast_score.
    """
    # work in log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # occasional numeric failures – just skip this λ
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex (mirror descent).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- current convolution -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # ----- sub‑gradient of the max (size n) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- EG update -----
        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def simulated_annealing_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis annealing moving a tiny amount of mass between
    two randomly chosen positions.  Accept moves that lower the score,
    otherwise accept with probability exp(-Δ/Temp).  Temperature decays
    geometrically.
    """
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    it = 0
    T = best_score * 0.10          # initial temperature
    decay = 0.9995                 # per‑iteration cooling factor

    while time.time() < deadline and it < 500_000:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            it += 1
            continue

        # fraction of source mass to transfer (log‑uniform)
        f = np.exp(rng.uniform(np.log(1e-5), np.log(0.05)))
        delta = a[i] * f
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise (protect against tiny rounding drift)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            it += 1
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)
        dE = cand_score - best_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
        # geometric cooling
        if it % 1000 == 0:
            T *= decay
        it += 1

    return best, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    Several heuristic families are mixed to give a broad search space.
    """
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "power_shift",
            "front_heavy",
            "bimodal",
            "cosine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "random":
        a = rng.random(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))

    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))

    elif mode == "power_shift":
        p = rng.uniform(0.2, 2.5)
        shift = rng.uniform(0.0, 5.0)
        a = (np.arange(1, n + 1, dtype=float) + shift) ** (-p)

    elif mode == "front_heavy":
        # linear ramp from a large value down to zero over a random length
        L = rng.integers(int(0.03 * n), int(0.40 * n))
        a = np.zeros(n)
        a[:L] = np.linspace(1.0, 0.0, L)
        # small uniform noise to avoid exact zeroes
        a += rng.uniform(0.0, 0.001, size=n)

    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.3, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak

    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

    # shifted power‑law
    for p in np.linspace(0.3, 2.5, 8):
        shift = rng.uniform(0.0, 5.0)
        a = (np.arange(1, n + 1, dtype=float) + shift) ** (-p)
        a /= a.sum()
        candidate_list.append(a)

    # exponential sweep
    for rate in np.linspace(0.001, 0.015, 12):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        candidate_list.append(a)

    # front‑heavy linear ramps
    for frac in np.linspace(0.03, 0.4, 6):
        L = max(1, int(frac * n))
        a = np.zeros(n)
        a[:L] = np.linspace(1.0, 0.0, L)
        a += rng.uniform(0.0, 1e-4, size=n)   # tiny noise to avoid exact zeros
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidate_list.append(a)

    # evaluate the pool
    for idx, cand in enumerate(candidate_list):
        if time.time() >= deadline:
            break
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[sweep] new best from family #{idx} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation from many random starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
        5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08,
    ]
    maxiter_per_lam = 1200

    # allocate ~55 % of the total time to this phase
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline:
        # occasional perturbation of the current best solution
        if rng.random() < 0.30:
            sigma = rng.uniform(0.0005, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (global walk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_optimize(
            best_a, deadline, fft_len_conv, fft_len_g, n, rng
        )
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final ultra‑fine L‑BFGS polish (tiny smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [5e-09, 1e-09],
            maxiter_per_lam=800,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (the function that will be judged)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
