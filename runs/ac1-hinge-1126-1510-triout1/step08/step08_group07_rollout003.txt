def adam_optimize(y0, lam_schedule, max_iter_per_lam, deadline, fft_len_conv, fft_len_g, n):
    y = y0.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    beta1=0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(y); v = np.zeros_like(y)
    t = 0

    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            if time.time()>deadline:
                return best_a, best_score
            t += 1
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # Adam update
            m = beta1 * m + (1-beta1) * grad
            v = beta2 * v + (1-beta2) * (grad**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = 0.1 # adjust
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        # after each lam maybe rebalance step size? not needed.
    return best_a, best_score
```

import time
import numpy as np

def next_pow2(x):
    return 1 << ((x-1).bit_length())

def full_conv(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))
    return conv[:2*a.shape[0]-1]

def fast_score(a, fft_len_conv, n):
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    # stable softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()
    # convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z

    # backpropagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def random_initial(n, rng):
    mode = rng.choice(["uniform","random","exp","power","edge","bimodal","cosine"])
    if mode == "uniform":
        a = np.full(n, 1.0/n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n+1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0/n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0/n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:
        # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def add_noise(a, sigma, rng):
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def adam_optimize(y0, lam_schedule, max_iter_per_lam, deadline,
                  fft_len_conv, fft_len_g, n):
    y = y0.copy()
    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    base_lr = 0.5
    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = base_lr / (np.sqrt(t) + 1.0)  # decaying step
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def targeted_hill_climb_line_search(a, current_score,
                                    fft_len_conv, fft_len_g, n,
                                    deadline, max_iter=200000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]
    while time.time() < deadline and it < max_iter:
        # compute convolution and gradient mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source/target based on gradient
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]
        if high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        if i == j or best[i] <= 0.0:
            it += 1
            continue
        improved = False
        best_local = best_score
        best_delta = 0.0
        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local - 1e-12:
                best_local = cand_score
                best_delta = delta
                improved = True
        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local
        it += 1
    return best, best_score

    start_time = time.time()

    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initial best (maybe reuse previous construction)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)

            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0/n)
        else:
            a0 = np.full(n, 1.0/n)
    else:
        a0 = np.full(n, 1.0/n)

    best_a = a0.copy()

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # schedule of smoothing factors (lam_factor)
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,

            best_a, best_score,

    print(f"[final] evaluated score = {final_eval:.6f}")

"""
Propose a low‑score step function.

The search works in three stages:
  1️⃣  Many short Adam‑optimised restarts on a soft‑max
      parametrisation of the simplex.  The objective is a
      log‑sum‑exp (smooth) approximation of max (conv(a,a));
      the smoothing factor (lam) is decreased gradually.
  2️⃣  The best sequence found is polished by a gradient‑guided
      hill‑climb that transfers a tiny fraction of mass from
      “high‑gradient” indices to “low‑gradient” ones.
  3️⃣  The final sequence is evaluated with the supplied
      ``evaluate_sequence`` routine.

All steps respect non‑negativity and the unit‑sum simplex,
so the evaluation function is scale‑invariant.
A 1000 s budget is split (≈75 % Adam, ≈25 % hill‑climb) and
time‑checks are performed throughout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective when Σa = 1:
          2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    lam_schedule,
    max_iter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space.
    The objective is the smoothed max‑convolution (log‑sum‑exp)
    with a decreasing smoothing factor (lam_schedule).
    Returns the best simplex‑feasible sequence and its fast_score.
    """
    y = y0.copy()
    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    base_lr = 0.5

    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            lr = base_lr / (np.sqrt(t) + 1.0)  # a simple decay
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb.
    For each iteration a (source, target) pair is chosen from the
    high/low ends of the sub‑gradient and a few transfer fractions
    are tried; only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]

    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (smoothed) max
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local - 1e-12:
                best_local = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local

        it += 1

    return best, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the 1000 s time limit and returns the best list
    of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # Problem size (chosen to keep FFTs cheap while giving enough freedom)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Schedule of smoothing factors (lam_factor) for the log‑sum‑exp
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
    ]

    # ------------------------------------------------------------------
    # Phase 1 – many Adam restarts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.75 * budget_s
    max_iter_per_lam = 120                # few hundred gradient steps per lambda

    while time.time() < phase1_end:
        # Choose a diverse start point
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        y0 = np.log(a_start + 1e-12)

        a_opt, score_opt = adam_optimize(
            y0,
            lam_schedule,
            max_iter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[adam] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑guided hill‑climb for the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=200000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
