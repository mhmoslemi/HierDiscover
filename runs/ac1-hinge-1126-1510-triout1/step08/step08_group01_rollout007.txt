def targeted_hill_climb(a, curr_score, fft_len, n, deadline, rng, max_fraction=0.07, recompute_every=1500):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    it = 0
    accepted = 0
    while time.time()<deadline:
        # compute mask and gradient
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # gradient
        a_rev = best_a[::-1]
        grad_fft = np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len))
        g = 2.0 * np.real(grad_fft[n-1:n-1+n])
        # source, dest selection
        # compute contribution
        contrib = g * best_a
        src = int(np.argmax(contrib))
        dst = int(np.argmin(g))
        if src == dst:
            # fallback random
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
        if best_a[src] == 0.0:
            # fallback choose random
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            src, dst = (i,j) if best_a[i] > best_a[j] else (j,i)
        # delta fraction
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0: continue
        # update conv incrementally:
        diff = np.zeros_like(conv)
        diff[src:src+n] -= best_a
        diff[dst:dst+n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score:
            # accept
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        # periodic recompute to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
        it += 1
    return best_a, best_score

"""
Step‑function optimizer (≈1000 s budget).

The optimizer consists of four stages:

1. **Fast L‑BFGS** on a log‑sum‑exp smoothed version of the
   max‑convolution.  Several random restarts with a decreasing
   smoothing parameter quickly locate a good basin.

2. **Long exponentiated‑gradient descent** on the true ∞‑norm
   objective.  The update a←a·exp(‑η g) stays on the simplex and
   steadily reduces the peak of the autocorrelation.

3. **Targeted hill‑climbing**.  In each iteration the gradient of the
   current maximum entry of the convolution is computed (via FFT) and
   the algorithm moves a small fraction of mass from the index that
   contributes most to the peak to the index that contributes least.
   This focuses the search on the active constraints.

4. **Random greedy hill‑climb** (as a final polishing step).

All phases respect the global 1000 s deadline and keep the sequence
non‑negative.  The final list is returned and its quality is reported
with the official `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) (fast, assumes sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# 1️⃣ L‑BFGS on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → a on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a ↔ a_rev)
    a_rev = a[::-1]
    grad_fft = np.fft.ifft(np.fft.fft(w, fft_len) *
                           np.fft.fft(a_rev, fft_len))
    g_vec = np.real(grad_fft[n - 1: n - 1 + n])

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Exponentiated‑gradient descent (multiplicative)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        grad_fft = np.fft.ifft(np.fft.fft(mask, fft_len) *
                               np.fft.fft(a_rev, fft_len))
        g = 2.0 * np.real(grad_fft[n - 1: n - 1 + n])

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Targeted hill‑climbing (gradient‑guided mass transfer)
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        curr_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_fraction: float = 0.07,
                        recompute_every: int = 1500):
    """
    Greedy hill‑climb that always moves mass from the index that
    contributes most to the current maximal convolution entry to the
    index that contributes least.  The convolution is updated
    incrementally (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full conv
    accepted = 0

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max entry (sub‑gradient)
        a_rev = best_a[::-1]
        grad_fft = np.fft.ifft(np.fft.fft(mask, fft_len) *
                               np.fft.fft(a_rev, fft_len))
        g = 2.0 * np.real(grad_fft[n - 1: n - 1 + n])

        # source = index with largest contribution to the peak
        contrib = g * best_a
        src = int(np.argmax(contrib))
        # destination = index with smallest gradient (least effect)
        dst = int(np.argmin(g))

        if src == dst or best_a[src] == 0.0:
            # fall back to a random admissible move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if best_a[i] >= best_a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            if best_a[src] == 0.0:
                continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution using the linear term
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept the move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodically recompute the convolution exactly to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣ Random greedy hill‑climb (fallback / polishing)
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 2000):
    """Generic hill‑climb that moves a random fraction of mass."""
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points provides a good compromise.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – reuse a previously discovered sequence if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.20 * budget_s       # L‑BFGS (≈200 s)
    phase2_end = start_time + 0.70 * budget_s       # exp‑grad (≈500 s)
    phase3_end = start_time + 0.90 * budget_s       # targeted hill‑climb (≈200 s)
    # remaining time → random hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – fast L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # occasional perturbation of the current best, otherwise fresh start
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=10.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase3_end,
            rng,
            max_fraction=0.07,
            recompute_every=1500,
        )
        print(f"[targeted‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – random greedy hill‑climb for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.10,
            recompute_every=2000,
        )
        print(f"[random‑hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
