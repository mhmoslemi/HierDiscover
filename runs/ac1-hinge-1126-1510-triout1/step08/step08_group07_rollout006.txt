def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule, iters_per_lam, deadline):
    y = y0.copy()
    best_score = float('inf')
    best_y = y.copy()
    for lam in lam_schedule:
        alpha = 0.01
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        for t in range(iters_per_lam):
            if time.time() >= deadline:
                break
            f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # update
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y**2)
            m_hat = m / (1 - beta1**(t+1))
            v_hat = v / (1 - beta2**(t+1))
            y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)
            # project? Not needed as softmax ensures positivity.
            # Compute a and fast_score
            a = np.exp(y - np.max(y))
            a = a / a.sum()
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
        y = best_y.copy()
    best_a = np.exp(best_y - np.max(best_y))
    best_a = best_a / best_a.sum()
    return best_a, best_score

"""
TinyOptim – a multi‑resolution, multi‑phase optimiser for the max‑convolution
objective.

The algorithm proceeds in three stages:

1️⃣  **Coarse optimisation on a small problem** (n≈2 k).  We run a
    succession of L‑BFGS steps on a smoothed log‑sum‑exp objective,
    followed by a short Adam‑style optimisation and a few exponentiated‑gradient
    updates.  The best sequence found is kept.

2️⃣  **Upscaling** the best coarse solution to a larger size (n≈8 k) by linear
    interpolation, normalising onto the simplex and repeating the same
    multi‑phase optimisation (more L‑BFGS iterations, a longer Adam schedule,
    and a final hill‑climb).  The larger dimension gives the algorithm a
    finer “attack surface” and typically lowers the max‑convolution peak.

3️⃣  **Polishing** – a deterministic, gradient‑guided hill‑climb with a tiny
    line‑search over several transfer fractions, finally followed by a single
    ultra‑low‑lambda L‑BFGS polish.

All stages respect non‑negativity and the simplex constraint, so the final
score is scale‑invariant.  Time‑budget checks guarantee termination well
before the 1000 s limit; the best sequence found so far is always returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a normalised sequence (Σa = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ---- softmax (stable) ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. convolution entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ---- back‑propagation through the convolution ----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through softmax ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    # work in the log‑softmax (log‑probability) space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        # map back to simplex
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax variables for the smoothed max.
    """
    y = np.log(a_start + 1e-12)  # logits
    best_y = y.copy()
    best_score = fast_score(a_start, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # Adam hyper‑parameters
        alpha = 0.02
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(y)
        v = np.zeros_like(y)

        for t in range(iters_per_lam):
            if time.time() >= deadline:
                break
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (t + 1))
            v_hat = v / (1 - beta2 ** (t + 1))
            y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate
            a = np.exp(y - np.max(y))
            a = a / a.sum()
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
        # start next λ from the best found so far
        y = best_y.copy()
    best_a = np.exp(best_y - np.max(best_y))
    best_a = best_a / best_a.sum()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using a true sub‑gradient
    of the (non‑smoothed) max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # step size – diminishing
        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb that moves a small fraction of mass from a
    “high‑gradient” index to a “low‑gradient” index, keeping only strictly
    improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # fractions of the source mass to try
    fractions = [0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (non‑smoothed) max
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source and target indices based on gradient percentiles
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score

# Multi‑resolution optimiser
# ----------------------------------------------------------------------


def optimise_at_size(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Runs the full multi‑phase optimisation (LBFGS → Adam → exp‑gradient →
    hill‑climb) for a given problem size `n`.
    Returns the best sequence (on the simplex) and its fast_score.
    """
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if init_seq is not None:
        a0 = np.array(init_seq, copy=True)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[optimise] n={n}, initial fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS continuation with a long λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
    ]
    maxiter_per_lam = 1200  # a bit more than the default

    # run a few independent restarts within the remaining budget
    phase1_deadline = deadline - 0.3 * (deadline - time.time())
    while time.time() < phase1_deadline:
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = adam_optimize(
            best_a,
            lam_schedule,
            iters_per_lam=800,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = exp_gradient_optimize(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[exp‑gradient] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic hill‑climb with line‑search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=200000,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[hill‑climb] new best fast_score={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The optimiser works on a small problem first (n≈2 k) and then
    up‑scales to a larger one (n≈8 k) before a final polishing step.
    The best sequence discovered before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Small‑scale optimisation (fast exploration)
    # ------------------------------------------------------------------
    n_small = 2048
    small_deadline = start_time + 0.35 * budget_s
    small_best, small_score = optimise_at_size(
        n_small, small_deadline, rng,
        init_seq=globals().get("height_sequence_1", None)
    )
    print(f"[small] best fast_score={small_score:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger problem size for finer control
    # ------------------------------------------------------------------
    n_large = 8192
    # Linear interpolation of the best small solution onto the larger grid
    x_old = np.linspace(0, 1, small_best.size)
    x_new = np.linspace(0, 1, n_large)
    large_start = np.interp(x_new, x_old, small_best)
    large_start = np.maximum(large_start, 0.0)
    large_start /= large_start.sum() if large_start.sum() > 0 else 1.0

    large_deadline = start_time + 0.85 * budget_s
    large_best, large_score = optimise_at_size(
        n_large, large_deadline, rng, init_seq=large_start
    )
    print(f"[large] best fast_score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Final ultra‑low‑lambda L‑BFGS polish (acts directly on the original
    # evaluation objective, because the smoothed max essentially becomes the
    # true max when λ → 0)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fft_len_conv = next_pow2(2 * n_large - 1)
        fft_len_g = next_pow2(3 * n_large - 2)
        final_a, final_score = lbfgs_continuation(
            large_best,
            [1e-07],
            maxiter_per_lam=500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n_large,
        )
        if final_score < large_score:
            large_best, large_score = final_a.copy(), final_score
            print(f"[final L‑BFGS] new best fast_score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided external function
    # ------------------------------------------------------------------
    final_seq = large_best.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final evaluation] score = {final_eval:.6f}")

    # expose the sequence for potential reuse in a later run
    globals()["height_sequence_1"] = final_seq

    return final_seq
