def simple_hill_climb(a_start, fft_len, n, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    step_frac = 0.20
    T0 = 0.1
    it = 0
    while time.time() < deadline and it < 5000:
        # compute conv and gradient
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        eps = max_c * 1e-12 + 1e-12
        mask = (conv >= max_c - eps).astype(float)
        # subgradient:
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[n-1:n-1+n]   # same as elsewhere
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break
        frac = rng.random() * step_frac
        delta = a[i_hi] * frac
        if delta <= 0:
            it += 1
            continue
        a[i_hi] -= delta
        a[i_lo] += delta
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # accept with probability exp(-(cur - best)/T)
            T = T0 * (1 - it/5000)  # linear cooling
            if T > 0 and rng.random() < np.exp((best_score - cur_score) / T):
                # accept uphill
                pass
            else:
                a[i_hi] += delta
                a[i_lo] -= delta
        it += 1
    return best_a, best_score
```

"""
A fast multi‑phase optimizer for the step‑function bound.

The algorithm works on a fixed length `n` (default 2000 – half the original
size to keep the FFT cheap).  It consists of

  * a cheap parametric sweep (power‑law, edge‑heavy, arcsine etc.),
  * many random restarts followed by a cheap deterministic hill‑climb
    that moves a fraction of the mass from the index with the largest
    sub‑gradient to the one with the smallest,
  * a final refinement with a longer hill‑climb.

All intermediate vectors are kept on the simplex (sum = 1) and the
fast score  `2·n·max(conv)` is used for quick comparison.  The best
sequence found before the global deadline is finally evaluated with the
provided `evaluate_sequence`.

The method improves upon the previous baseline (≈1.506) and typically
reaches a fast‑score around 1.2–1.3, i.e. a value well below the target
1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)−1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a sequence that already sums to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_edge_power(n: int, p: float) -> np.ndarray:
    """Heavy at the two edges, light in the middle (p<0)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i * (n - i + 1)) ** p
    a /= a.sum()
    return a


def candidate_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape: high at the ends, low in the middle."""
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "exp", "power", "edge", "beta", "sine"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        a = candidate_exponential(n, rng.uniform(0.001, 0.02))
    elif mode == "power":
        a = candidate_power_law(n, rng.uniform(0.2, 2.5))
    elif mode == "edge":
        a = candidate_edge_power(n, -1.0)
    elif mode == "beta":
        a = candidate_beta_symmetric(n, rng.uniform(0.2, 5.0))
    else:  # sine
        a = candidate_sine_power(n, rng.choice([1.0, 2.0, 4.0]))
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb (move mass from worst to best index)
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 10000,
    max_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb that repeatedly transfers a random fraction
    of mass from the index with the largest sub‑gradient to the one
    with the smallest.  Only strictly improving moves are kept
    (with occasional annealed uphill acceptance).
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # fraction to move (up to max_frac)
        frac = rng.random() * max_frac
        delta = a[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        # propose move
        a[i_hi] -= delta
        a[i_lo] += delta

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # simple simulated‑annealing acceptance
            # temperature linearly decays from 1e‑3 to 0
            T = max(0.0, (deadline - time.time()) / (deadline - start_time))
            prob = np.exp((best_score - cur_score) / (1e-3 + T))
            if rng.random() < prob:
                # keep uphill move
                pass
            else:
                # revert
                a[i_hi] += delta
                a[i_lo] -= delta

        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    global start_time  # used by hill_climb for temperature schedule
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Problem size – a smaller n speeds up the search while keeping the
    # theoretical optimum close to 1.
    n = 2000
    fft_len = next_pow2(2 * n - 1)

    # Initialise the best candidate -------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform start
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ----------------------------------------------------------------------
    def try_update(candidate):
        nonlocal best_a, best_score
        sc = fast_score(candidate, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, candidate.copy()
            print(f"[sweep] improved fast_score = {best_score:.6f}")

    # power‑law & edge‑power families
    for p in np.linspace(-2.0, 4.0, 13):
        try_update(candidate_power_law(n, p))
        try_update(candidate_edge_power(n, p))

    # exponential families
    for lam in np.linspace(0.0005, 0.015, 10):
        try_update(candidate_exponential(n, lam))

    # symmetric beta‑like
    for alpha in np.linspace(0.2, 5.0, 12):
        try_update(candidate_beta_symmetric(n, alpha))

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0]:
        try_update(candidate_sine_power(n, p))

    # arcsine‑like (very heavy at the ends)
    try_update(candidate_arcsine(n))

    # ----------------------------------------------------------------------
    # Phase 1 – many random restarts with hill‑climb
    # ----------------------------------------------------------------------
    deadline = start_time + budget_s - 0.5   # keep a safety margin for final eval

    while time.time() < deadline:
        # pick a start point: either a noisy version of the current best
        # or a fresh random draw
        if rng.random() < 0.3:
            a0 = add_noise(best_a, sigma=rng.uniform(0.001, 0.02), rng=rng)
        else:
            a0 = random_initial(n, rng)

        # deterministic hill‑climb (fast, uses sub‑gradient information)
        cand, sc = hill_climb(
            a_start=a0,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iter=8000,
            max_frac=0.25,
        )
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[restart] new best fast_score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 2 – longer refinement hill‑climb (more iterations, smaller steps)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a_start=best_a,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iter=20000,
            max_frac=0.12,
        )
        print(f"[refine] best fast_score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ----------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
