"""
Propose a step sequence with a low evaluation score.

Algorithm Overview
------------------
1. **Coarse parametric sweep** over several analytic families
   (power‑law, reversed power‑law, exponential, reversed exponential,
   Gaussian, symmetric beta, sine‑power).  All candidates are projected
   onto the simplex and evaluated with the fast O(n log n) score.

2. **Random‑restart L‑BFGS** on the log‑sum‑exp smoothed max‑convolution.
   A decreasing schedule of smoothing parameters (λ) gradually sharpens
   the approximation to the true max‑convolution.

3. **Refined projected sub‑gradient descent**.
   At each iteration we compute the exact sub‑gradient of the true
   max‑convolution, form a zero‑sum descent direction
   `d = -(g - mean(g))` and perform a back‑tracking line‑search with a
   simplex projection.  If no improvement is possible we inject a tiny
   Gaussian perturbation to escape plateaus.  This stage runs for the
   majority of the time budget and typically yields the strongest
   improvements.

4. **Local mixing & fine hill‑climb**.
   Small convex combinations of the best sequence with the uniform
   distribution and with its reversed copy are tested.
   Finally a deterministic hill‑climb (moving a fraction of mass from
   the coordinate with the largest sub‑gradient to the one with the
   smallest) is applied for a few thousand steps.

5. **Final evaluation** using the provided `evaluate_sequence` function.

All phases respect a total wall‑clock budget of 1000 s (default).  The
function returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  The function works in O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best `a` found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Refined sub‑gradient descent with line search
# ----------------------------------------------------------------------


def refined_subgradient(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the true max‑convolution.
    Uses a zero‑sum descent direction `d = -(g - mean(g))`
    and a back‑tracking line search.  Occasionally adds tiny Gaussian
    noise to escape flat regions.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    no_improve = 0
    while time.time() < deadline:
        # ----- compute convolution and mask for max entries -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the max‑convolution -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- descent direction (zero‑sum) -----
        d = -(grad - grad.mean())

        # ----- back‑tracking line search with simplex projection -----
        step = 1.0
        improved = False
        for _ in range(10):
            a_candidate = simplex_projection(a + step * d)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score - 1e-12:  # strict improvement
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate.copy()
                improved = True
                no_improve = 0
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                # occasional jitter to get out of a flat plateau
                a = add_noise(a, sigma=0.0008, rng=rng)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (small transfers)
# ----------------------------------------------------------------------


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a random fraction (≤15 %) of the
    mass from the coordinate with the largest sub‑gradient to the one with
    the smallest.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        frac = rng.random() * 0.15
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previously known best if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – extended cheap parametric sweep
    # ------------------------------------------------------------------
    def try_update(a: np.ndarray):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    # Power‑law families
    for p in np.linspace(-3.0, 6.0, 25):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # Exponential families (forward and reversed) – finer grid
    for lam in np.linspace(0.0005, 0.03, 30):
        try_update(candidate_exponential(n, lam, reverse=False))
        try_update(candidate_exponential(n, lam, reverse=True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.10, 0.50, 15):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # Symmetric beta‑like shapes
    for alpha in np.linspace(0.1, 6.0, 20):
        try_update(candidate_beta_symmetric(n, alpha))

    # Sine‑power windows
    for p in [0.25, 0.5, 0.75, 1.0, 2.0, 3.0, 4.0]:
        try_update(candidate_sine_power(n, p))

    # Quick random initial restarts
    while time.time() < start_time + 0.07 * budget_s:
        try_update(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Time allocation for later phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s  # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s  # refined sub‑gradient
    phase3_end = start_time + 0.95 * budget_s  # mixing / hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
        0.0001,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # occasional jitter around the current best
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined projected sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = refined_subgradient(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[refined sub‑grad] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – mixing with uniform / reversed and final hill‑climb
    # ------------------------------------------------------------------
    # small convex combinations with the uniform distribution
    uniform = np.full(n, 1.0 / n)
    for w in np.arange(0.001, 0.101, 0.005):
        cand = (1.0 - w) * best_a + w * uniform
        cand /= cand.sum()
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[mix‑uniform] w={w:.3f} improved fast_score={best_score:.6f}")

    # mixture with the reversed version
    rev = best_a[::-1]
    for w in np.arange(0.001, 0.101, 0.005):
        cand = (1.0 - w) * best_a + w * rev
        cand /= cand.sum()
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[mix‑rev] w={w:.3f} improved fast_score={best_score:.6f}")

    # deterministic hill‑climb for a final polishing
    if time.time() < phase3_end:
        best_a, best_score = gradient_guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            max_iter=40000,
        )
        print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
