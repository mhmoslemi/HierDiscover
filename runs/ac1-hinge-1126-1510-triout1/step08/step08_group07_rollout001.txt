"""
Propose a low‑score step‑function sequence.

The algorithm works in four phases (all time‑bounded):
  1️⃣  Smoothed‑max L‑BFGS (continuation over a schedule of smoothing
      factors).  Starts from the previous best sequence (if any) and from
      several random simple shapes.  The softmax parametrisation guarantees
      a feasible point on the simplex.
  2️⃣  Mirror‑descent via exponentiated‑gradient on the true (unsmoothed)
      max‑convolution sub‑gradient.  A decreasing step‑size yields fast
      descent while staying on the simplex.
  3️⃣  Gradient‑guided hill‑climb with line‑search on a handful of mass‑transfer
      fractions.  Pairs (i→j) are chosen from the high/low tails of the
      sub‑gradient, and only strictly improving moves are kept.
  4️⃣  Light‑weight simulated‑annealing: random mass‑transfer proposals are
      accepted either if they improve the score or with a temperature‑dependent
      probability, allowing the search to escape shallow local minima.

The sequence length is taken larger (n=8000) than the previous
implementation; a finer discretisation permits a slightly lower limiting
constant for the max‑auto‑convolution, which is needed to break the
1.503 barrier.  All phases respect the non‑negativity and unit‑sum
normalisation, and the final sequence is evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the first 2·len(a)-1 entries
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quick evaluation of the objective when a sums to 1:
        score = 2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the autocorrelation a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y) for L‑BFGS.
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()                 # a ≥0, Σa = 1

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max                  # objective value

    # ----- gradient of smooth_max w.r.t. each convolution entry -----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution (a*a → a) -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to ∂/∂a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)            # ∂f/∂y

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each smoothing factor in lam_schedule,
    starting each stage from the solution of the previous stage.
    Returns the best a observed and its fast_score.
    """
    # work in log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as exc:           # safety net
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        # recover a from logits
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the exact
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- convolution a*a -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of max‑convolution (size n) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- exponentiated gradient update -----
        eta = 0.30 / np.sqrt(it + 1.0)   # larger initial step
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb: repeatedly transfer a fraction of mass
    from a high‑gradient index to a low‑gradient index and keep strictly
    improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # fractions spanning several orders of magnitude
    fractions = [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.12, 0.20]

    while time.time() < deadline and it < max_iter:
        # ----- current convolution & active mask -----
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the (unsmoothed) max -----
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- choose source / target from gradient tails -----
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta

            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-13:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    init_temp: float = 0.02,
    final_temp: float = 0.0002,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Light‑weight SA: randomly move a small amount of mass between two
    coordinates; accept improvements or, with probability exp(-Δ/Temp), accept
    uphill moves.  Temperature decays linearly with iteration count.
    """
    best = a.copy()
    best_score = current_score
    cur = a.copy()
    cur_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # linear temperature schedule
        t_frac = it / max_iter
        temp = init_temp * (1 - t_frac) + final_temp * t_frac

        # pick source and target
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            it += 1
            continue

        # random fraction of source mass (log‑uniform)
        frac = np.exp(rng.uniform(np.log(1e-5), np.log(0.1)))
        delta = cur[i] * frac
        if delta <= 0.0:
            it += 1
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # accept with probability exp(-(cand_score-cur_score)/temp)
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
        it += 1

    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diversified random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and project back onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation and a slightly
    # lower asymptotic constant, which is crucial for breaking the 1.503 barrier.
    # ------------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient convolutions

    # ------------------------------------------------------------------
    # Initialise from the previously discovered sequence (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget partition
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    phase4_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with decreasing smoothing
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1,
        0.05, 0.02, 0.01, 0.005,
        0.002, 0.001, 5e-04, 2e-04,
        1e-04, 5e-05, 2e-05, 1e-05,
        5e-06, 1e-06, 5e-07, 1e-07,
    ]
    maxiter_per_lam = 600

    # keep a few restarts but respect the time budget
    while time.time() < phase1_end:
        # Occasionally perturb the current best to increase diversity
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – gradient‑guided hill‑climb with line search
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=300000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_end,
            init_temp=0.015,
            final_temp=0.0005,
            max_iter=400000,
        )
        print(f"[sim‑anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function (scale‑invariant)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
