def adam_descent(a_start, deadline, n, fft_len, rng, eta=0.1, beta1=0.9, beta2=0.999, epsilon=1e-8):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 1
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    while time.time() < deadline:
        # compute gradient of max conv (subgradient)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12 * max_val).astype(np.float64)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1: n-1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        a = simplex_projection(a)
        # evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        t_iter += 1
    return best_a, best_score
```

def hill_climb_simulated(a, curr_score, fft_len, n, deadline, rng,
                         max_fraction=0.12, recompute_every=2000,
                         init_T=0.01, final_T=1e-5):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)
    start = time.time()
    total_time = deadline - start
    accept_count = 0
    while time.time() < deadline:
        # compute remaining time fraction
        t_rem = deadline - time.time()
        frac = t_rem / total_time
        T = final_T + (init_T - final_T) * frac
        # random move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0: continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0: continue
        # incremental update
        diff.fill(0.0)
        diff[src:src+n] -= best_a
        diff[dst:dst+n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        delta_score = new_score - best_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if np.exp(-delta_score / T) > rng.random():
                accept = True
        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            accept_count += 1
            if new_score < best_score:
                best_score = new_score
                # update global best if needed
                # keep best_a and best_score as current
            
            if accept_count % recompute_every == 0:
                conv = np.convolve(best_a, best_a)
    return best_a, best_score

"""
Propose a step‑function that tries to lower the evaluation score.
The algorithm works in three stages:

1.  Several short L‑BFGS runs on a log‑sum‑exp smoothed max‑convolution.
    This gives a good “flat” starting point.
2.  An Adam optimiser that minimises the *square error* between the
    self‑convolution and the constant value 1/(2 n‑1).  This explicitly
    flattens the whole convolution and therefore reduces the ratio
    max(conv)/average(conv).
3.  A greedy simulated‑annealing hill‑climb with incremental convolution
    updates.  Small amounts of mass are moved from source to destination
    indices; moves that improve the max are always accepted, otherwise they
    are accepted with probability exp(‑Δ/ T) where the temperature T cools
    linearly to zero.

All work is done on the simplex (∑a_i = 1) so the scale of the sequence
does not affect the final score.  The code respects the 1000 s budget,
prints intermediate progress and returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) – valid when ∑a_i = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def cosine_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped (high at the ends, low in the middle) cosine profile."""
    r = rng.uniform(0.4, 0.98)               # amplitude control
    i = np.arange(n)
    a = 1.0 + r * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise to a and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """Smoothed max‑convolution objective.  y are logits, a = softmax(y)."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous."""
    y = np.log(a_start + 1e-12)                # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()          # stay on simplex
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser that flattens the whole convolution (minimise L2 error)
# ----------------------------------------------------------------------


def adam_flatten_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta: float = 0.2):
    """
    Adam optimiser for the squared‑error objective:
        E = Σ_k ( (a*a)_k – c )²,  where c = 1/(2n‑1).
    The algorithm projects onto the simplex after each update.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    c = 1.0 / (2 * n - 1)            # target constant value of the convolution

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 1

    while time.time() < deadline:
        # a*a via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        diff = conv - c                     # residual

        # gradient of Σ diff² w.r.t. a
        a_rev = a[::-1]
        fd = np.fft.fft(diff, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_tmp = np.real(np.fft.ifft(fd * fa_rev))
        g = 4.0 * grad_tmp[n - 1: n - 1 + n]   # factor 4 from the chain rule

        # Adam update on the pre‑projection vector
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = simplex_projection(a)          # enforce non‑negativity & sum=1

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        t_iter += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_simulated(a: np.ndarray,
                         current_score: float,
                         fft_len: int,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_fraction: float = 0.12,
                         recompute_every: int = 1500,
                         init_T: float = 0.015,
                         final_T: float = 1e-5):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Moves that improve the max are always
    accepted; worsening moves are accepted with a temperature‑dependent
    probability (simulated annealing).  The convolution is updated
    incrementally (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a.copy()
    best_score = current_score
    conv = np.convolve(best_a, best_a)                 # exact convolution
    diff = np.zeros_like(conv)                         # reusable buffer

    start = time.time()
    total = deadline - start
    accepted = 0

    while time.time() < deadline:
        # linear cooling of the temperature
        t_rem = deadline - time.time()
        frac = t_rem / total
        T = final_T + (init_T - final_T) * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        d_score = new_score - best_score

        accept = False
        if d_score < 0.0:
            accept = True
        else:
            if np.exp(-d_score / T) > rng.random():
                accept = True

        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            accepted += 1
            if new_score < best_score:
                best_score = new_score

            if accepted % recompute_every == 0:
                conv = np.convolve(best_a, best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the global best if available, else a simple start
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.25 * budget_s       # L‑BFGS phase
    phase2_end = start_time + 0.75 * budget_s       # Adam‑flatten phase
    # remaining time → simulated‑annealing hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS runs with diverse starts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # 20 % chance to perturb the current best, otherwise a fresh start
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            # pick a random generation mode
            mode = rng.choice(['uniform', 'random', 'exp',
                               'edge', 'cosine', 'bimodal'])
            if mode == 'cosine':
                a_start = cosine_initial(n, rng)
            else:
                a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation that flattens the whole convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_flatten_descent(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta=0.2,
        )
        print(f"[Adam‑flatten] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_simulated(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
            init_T=0.015,
            final_T=1e-5,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
