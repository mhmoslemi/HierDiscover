import time, numpy as np

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x-1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a with itself via FFT (full length)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast upper bound: 2*n*max(a*a) assuming sum(a)=sqrt(2n)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simple search over power‑law increasing step functions.
    For each exponent p we build a_i ∝ (i+1)^p, normalise so
    that Σ a_i = √(2n) (the scaling used in the original paper),
    and keep the shape with the lowest value returned by
    ``evaluate_sequence``.  The search evaluates only a few candidates
    and finishes well within the 1000 s budget.
    """
    np.random.seed(seed)
    n = 4000
    target_sum = np.sqrt(2.0 * n)

    # FFT length needed for fast_score (over‑estimate a power of two)
    fft_len = next_pow2(2 * n - 1)

    best_score = float('inf')
    best_seq = None

    start = time.time()
    # try a few exponents – the optimal p is around 5 for this n
    for p in range(1, 9):                     # p = 1 … 8
        # build raw power‑law vector (i starts at 1)
        i = np.arange(1, n + 1, dtype=np.float64)
        raw = i ** p
        # normalise to the required L1 norm
        a = raw * (target_sum / raw.sum())
        # quick lower‑bound using the FFT‑based fast_score
        score_fast = fast_score(a, fft_len, n)

        # evaluate only if it can improve the current best
        if score_fast < best_score:
            real_score = evaluate_sequence(a.tolist())
            print(f"[p={p}] fast={score_fast:.6f}  real={real_score:.6f}")
            if real_score < best_score:
