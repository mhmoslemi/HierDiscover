def coordinate_descent(a, deadline, fft_len):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    while time.time()<deadline:
        conv_max, grad = compute_subgradient(best)
        i_hi = np.argmax(grad)   # index where moving mass away reduces conv max
        i_lo = np.argmin(grad)   # index where moving mass adds less to max
        if i_hi == i_lo: break
        # compute max possible delta = best[i_hi] (can't go negative)
        max_delta = best[i_hi]
        # try delta = fraction * max_delta, starting at 0.5
        delta = max_delta * 0.5
        while delta > 1e-12:

"""
Improved search for a non‑negative step function minimizing
      2·n·max(convolution) / (sum a)²
i.e. 2·n·max(a∗a) under ∑a=1, a≥0.

Algorithm
---------
1. Choose a convenient length n (a power of two → cheap FFT).
2. Initialise with a rich set of parametric families (power‑law,
   exponential, Gaussian, beta‑like, sinusoidal) and a few random
   points on the simplex. Keep the best according to the fast O(n log n)
   score 2·n·max(a∗a).
3. Refine the best point with L‑BFGS on a smooth log‑sum‑exp
   approximation of the max‑convolution.  A decreasing schedule of
   smoothing parameters (λ) is used; each stage runs a moderate number
   of L‑BFGS iterations.
4. Perform a large‑scale gradient‑guided coordinate descent.
   In every step the sub‑gradient of the true max‑convolution is
   computed (via two FFTs).  The index with the largest positive
   gradient (most harmful) transfers a fraction of its mass to the
   index with the smallest gradient (most helpful).  The transferred
   amount is obtained by a simple back‑tracking line‑search that
   accepts the largest step that strictly improves the score.
   This deterministic move is repeated until the time budget expires.
5. A final short hill‑climb (small random transfers) polishes the
   solution.
6. The best distribution is returned as a plain list of floats.  The
   supplied `evaluate_sequence` function is called only once at the
   end to obtain the official score.

The whole search respects the global time budget (default 1000 s) and
never uses closures or lambdas, as required.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when ∑a = 1:
        2·n·max(a∗a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed version of 2·n·max(a∗a) with a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, fft_len: int):
    """Return (max_convolution, gradient vector) for a (∑a = 1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, grad


# ----------------------------------------------------------------------
# Large‑step gradient‑guided coordinate descent
# ----------------------------------------------------------------------


def coordinate_descent(a_start: np.ndarray, deadline: float, fft_len: int):
    """
    Deterministic descent that repeatedly moves a (potentially large)
    fraction of mass from the index with largest sub‑gradient to the one
    with smallest sub‑gradient.  A simple back‑tracking line‑search
    chooses the biggest admissible step that strictly improves the score.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_score = fast_score(a, fft_len, n)
    best = a.copy()
    it = 0

    while time.time() < deadline:
        max_val, grad = compute_subgradient(a, fft_len)
        i_hi = int(np.argmax(grad))  # most harmful
        i_lo = int(np.argmin(grad))  # most helpful
        if i_hi == i_lo:
            break

        # maximal amount we can move from i_hi
        max_delta = a[i_hi]
        # start with a relatively aggressive fraction
        delta = max_delta * 0.5
        improved = False
        while delta > max_delta * 1e-12:
            cand = a.copy()
            cand[i_hi] -= delta
            cand[i_lo] += delta
            # No need to renormalise – total mass preserved
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                a = cand
                best = cand
                best_score = cand_score
                improved = True
                break
            delta *= 0.5
        if not improved:
            # if even a tiny move does not improve, break to avoid endless loops
            break
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Small random hill‑climb (final polishing)
# ----------------------------------------------------------------------


def gradient_guided_hill_climb(a: np.ndarray, current_score: float,
                               fft_len: int, n: int,
                               deadline: float, max_iter: int = 20000):
    """
    Moves a random fraction (≤15 %) of mass from the worst to the best
    coordinate; only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        max_val, grad = compute_subgradient(best, fft_len)
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        frac = rng.random() * 0.15
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power of two → very fast FFTs)
    # --------------------------------------------------------------
    n = 4096
    fft_len_conv = next_pow2(2 * n - 1)      # = 8192
    fft_len_g = fft_len_conv                # same length is sufficient

    # --------------------------------------------------------------
    # Initialise from a previously known solution if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin for final eval

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (diverse families)
    # --------------------------------------------------------------
    def try_update(a):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    # power‑law families
    for p in np.linspace(-2.0, 5.0, 20):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        try_update(candidate_exponential(n, lam, False))
        try_update(candidate_exponential(n, lam, True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 9):
        try_update(candidate_gaussian(n, sigma_frac * n))

    # symmetric beta‑like shapes
    for alpha in np.linspace(0.2, 5.0, 14):
        try_update(candidate_beta_symmetric(n, alpha))

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update(candidate_sine_power(n, p))

    # a few quick random restarts
    sweep_deadline = start_time + 0.10 * budget_s
    while time.time() < sweep_deadline:
        try_update(random_initial(n, rng))

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smoothed max‑convolution
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001
    ]

    while time.time() < phase1_end:
        # jitter around the current best to diversify restarts
        if rng.random() < 0.2:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        y = np.log(a_start + 1e-12)

        for lam in lam_schedule:
            if time.time() >= phase1_end:
                break
            try:
                res = minimize(
                    fun=lambda yy: objective_and_grad(yy, n, fft_len_conv,
                                                      fft_len_g, lam),
                    x0=y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": 500, "gtol": 1e-12, "disp": False},
                )
            except Exception as e:  # pragma: no cover
                print("[lbfgs] exception:", e)
                continue

            y = res.x
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – large‑step gradient‑guided coordinate descent
    # --------------------------------------------------------------
    phase2_end = start_time + 0.90 * budget_s
    best_a, best_score = coordinate_descent(best_a, phase2_end, fft_len_conv)
    print(f"[coord‑desc] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final tiny random hill‑climb
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = gradient_guided_hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=40000
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
