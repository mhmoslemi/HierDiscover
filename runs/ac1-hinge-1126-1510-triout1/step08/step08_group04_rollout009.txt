"""
Propose a low‑score step‑function sequence using a multi‑stage optimizer.

Stages
------
1. Heavy L‑BFGS on a log‑sum‑exp smoothed max‑convolution.
2. Exact exponentiated‑gradient descent on the true max‑convolution.
3. Simulated annealing with mass‑transfer and Gaussian noise moves.
4. Linear‑programming refinement: maximise the L1‑mass of a partner
   sequence g under the constraint (f * g) ≤ M, where M = max(f*f).
   After scaling g to the same L1‑norm as f we replace f and repeat.
5. Final greedy hill‑climb for polishing.

All stages respect the simplex (non‑negative entries, sum = 1) and are
timed so that the whole routine finishes within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) when a lies on the simplex.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        [
            'uniform', 'random', 'exp', 'power', 'edge',
            'bimodal', 'sparse', 'logit_normal'
        ]
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:                        # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        a = softmax(logits)

    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of the max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam)         # safe, ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # slice corresponding to gradient w.r.t. a (indices n‑1 … n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous λ.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for softmax parametrisation

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exact exponentiated‑gradient descent on the hard max‑convolution.
    The update a ← a·exp(−η·g) automatically stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size ~ 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        max_local_transfer: float = 0.5):
    """
    Simple simulated‑annealing on the simplex.
    Proposes two kinds of moves:
    * a mass transfer between two coordinates,
    * a global Gaussian perturbation (add_noise).
    Accepts moves according to a linearly decreasing temperature.
    """
    a_cur = a_start.copy()
    cur_score = fast_score(a_cur, fft_len, n)
    best_a = a_cur.copy()
    best_score = cur_score

    start = time.time()
    init_temp = cur_score * 0.02 + 1e-9

    while time.time() < deadline:
        frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (1.0 - frac)

        a_new = a_cur.copy()
        if rng.random() < 0.6:
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * max_local_transfer
                a_new[i] -= delta
                a_new[j] += delta
        else:
            sigma = rng.uniform(0.0005, 0.015)
            a_new = add_noise(a_new, sigma, rng)

        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        delta = new_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a modest random fraction of mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


def lp_refinement(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len_conv: int,
                  rng: np.random.Generator,
                  mix_factor: float = 0.5):
    """
    Linear‑programming refinement.
    For the current a, solve:
        maximise   ∑ b_j
        subject to   (a * b)_k ≤ M   for all k,
                     b_j ≥ 0,
    where M = max(a*a).  The solution b is rescaled to unit L1‑norm
    and blended with a (mix_factor).  The process repeats while time permits.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        # compute current max‑convolution bound M
        conv = full_conv(a, fft_len_conv)[: 2 * n - 1]
        M = conv.max()

        # Build the convolution matrix A (size (2n‑1) × n) in sparse format.
        # Row k, column j receives a_{k‑j} when 0 ≤ k‑j < n.
        offsets = -np.arange(n)                     # negative offsets produce the Toeplitz pattern
        diag_data = [np.full(2 * n - 1 - d, a[d], dtype=np.float64)
                     for d in range(n)]
        A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear programme: maximise sum(b)  ⇔ minimise -1·b
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve with HiGHS (efficient for large sparse problems)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=(0, None),
                          method='highs',
                          options={'presolve': True})
        except Exception:
            # If anything goes wrong, abort the LP phase.
            break

        # Clean up the large matrix ASAP
        del A

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No improvement possible – stop.
            break

        # Scale b back to unit L1‑norm.
        b_scaled = b / sum_b

        # Blend with the current a.
        a = (1.0 - mix_factor) * a + mix_factor * b_scaled

        # Evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Small safety: if improvement is negligible, exit.
        if np.abs(score - best_score) < 1e-12:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)   # used inside L‑BFGS gradients

    # Warm start from a previously stored construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # allocate portions of the budget
    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # exponentiated gradient
    phase3_end = start_time + 0.75 * budget_s   # simulated annealing
    phase4_end = start_time + 0.95 * budget_s   # LP refinement
    # remaining time → final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if best_score < np.inf and rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            phase3_end,
            n,
            fft_len_conv,
            rng,
        )
        print(f"[SA] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = lp_refinement(
            best_a,
            phase4_end,
            n,
            fft_len_conv,
            rng,
            mix_factor=0.5,
        )
        print(f"[LP] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – intensive greedy hill‑climb for the remaining budget
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=50000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
