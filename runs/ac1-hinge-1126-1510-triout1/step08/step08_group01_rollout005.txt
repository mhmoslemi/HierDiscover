"""
Step‑function optimizer (≈1 500 ms → target ≤ 1.503).

The optimizer works in three phases:

1. **Smoothed L‑BFGS** –  Optimizes a log‑sum‑exp smooth approximation of the
   max‑convolution.  Several random restarts are performed with a schedule of
   decreasing smoothing parameters λ.

2. **Exponentiated‑gradient descent** –  Directly minimises the true
   L∞‑objective.  The step size η_t = η₀ / √t decays slowly so that a large
   exploratory phase is followed by a fine‑tuned regime.

3. **Gradient‑guided hill‑climb** –  Uses the exact sub‑gradient of the
   max‑convolution to move probability mass from the “most harmful’’ index to
   the “most helpful’’ index.  The convolution is updated incrementally,
   occasionally recomputed exactly to avoid numerical drift.

The algorithm works on the probability simplex (∑a_i = 1); the evaluation
function is invariant to scaling, so no explicit normalisation is required.
The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) – the quantity returned by evaluate_sequence for sum=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability distribution."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Phase‑1 – smoothed L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS for each λ in lam_schedule, starting from a_start."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑2 – exponentiated‑gradient descent on the true max‑convolution

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """Exponentiated‑gradient descent with η₀/√t schedule."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clamp exponent)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1

    return best_a, best_sc

def gradient_guided_hill_climb(a: np.ndarray,
                               curr_score: float,
                               fft_len: int,
                               n: int,
                               deadline: float,
                               rng: np.random.Generator,
                               max_fraction: float = 0.15,
                               recompute_every: int = 2000):
    """
    Greedy hill‑climb that always moves mass from the index with the
    largest sub‑gradient (most harmful) to the one with the smallest
    sub‑gradient (most helpful).  Convolution is updated incrementally.
    """
    best_a = a.copy()
    best_sc = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    accepted = 0
    it = 0

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # choose source (largest g) and destination (smallest g)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # ensure source actually holds mass
        if best_a[src] <= 1e-15:
            # pick a random positive index as source
            pos = np.where(best_a > 1e-12)[0]
            if pos.size == 0:
                break
            src = int(pos[rng.integers(len(pos))])

        if src == dst:
            # random move to avoid stagnation
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        # move a random fraction of the source mass
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_sc = 2.0 * n * new_max

        if new_sc < best_sc - 1e-14:      # accept only genuine improvements
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_sc = new_sc
            accepted += 1

        # occasional exact recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
            best_sc = 2.0 * n * conv.max()
            # informative progress output
            print(f"[hill‑grad] iter {it:>6d}, score {best_sc:.6f}")

        it += 1

    return best_a, best_sc

    phase1_end = start + 0.15 * budget_s   # L‑BFGS (quick)
    phase2_end = start + 0.75 * budget_s   # EG descent (long)

    # --------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS with several restarts
    # --------------------------------------------------------------
    lam_schedule = [
        0.8, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 0.00005
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # Occasionally perturb the incumbent; otherwise start fresh
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=12.0,
        )
        print(f"[exp‑grad] fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – gradient‑guided hill‑climbing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = gradient_guided_hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.15,
            recompute_every=1800,
        )
        print(f"[grad‑hill] fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # store for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
