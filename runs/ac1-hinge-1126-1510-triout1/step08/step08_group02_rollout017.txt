def linear_program_refine(a, max_iter, deadline):
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv and max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        M = conv.max()
        # mask active constraints near M
        eps = M * 1e-12 + 1e-12
        active_idx = np.where(conv >= M - eps)[0]
        if len(active_idx) == 0:
            continue
        # Build constraint matrix
        A = np.zeros((len(active_idx), n))
        for ri, k in enumerate(active_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            if j_start <= j_end:
                # indices j_start..j_end inclusive
                # set A[ri, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
                idx_range = np.arange(j_start, j_end+1)
                A[ri, j_start:j_end+1] = a[k - idx_range]
        # solve LP: maximize sum b subject to A @ b <= M, b>=0
        c = -np.ones(n) # because scipy.linprog minimizes; we want maximize sum(b)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(len(active_idx), M), bounds=bounds, method='highs')
        if not res.success:
            continue
        b = res.x
        sumb = b.sum()
        if sumb <= 0:
            continue
        # Normalize b to sum=1
        b = b / sumb
        # Mix
        t = 0.1  # mixing factor
        a_new = (1 - t) * a + t * b
        # renorm sum to 1
        a_new = a_new / a_new.sum()
        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best = a_new.copy()
        a = a_new
    return best, best_score

"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1.  Small‑scale exploratory search (n_small = 1024) using many random
    initialisations and a few cheap exponentiated‑gradient runs.
    The cheapest candidates are also generated from a few parametric
    families (U‑shape, power law, double‑exponential, Beta‑pdf).

2.  Upsample the best small‑scale candidate to the full size (n = 4000).

3.  Refine the full‑size sequence with several optimisation phases:
    * L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
    * Long exponentiated‑gradient descent on the exact max‑convolution.
    * Projected sub‑gradient descent.
    * A linear‑programming “flat‑max” step (max‑convolution ≤ current M)
      that mirrors the LP described in the statement.
    * Final greedy hill‑climb for polishing.

All phases respect the overall time budget (≈ 1000 s).  The best
sequence found before the deadline is returned as a plain list of
floats.  The code works with the provided ``evaluate_sequence`` function
and does not perform any I/O.

"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv(a,a)) assuming sum(a)=1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Parameterised initialisers
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def spectral_init(n: int) -> np.ndarray:
    """Square‑root of uniform spectrum, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a /= a.sum()
    return a


def random_dirichlet_init(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw from Dirichlet(alpha,…,alpha)."""
    x = rng.gamma(alpha, 1.0, size=n)
    x /= x.sum()
    return x


def power_law_init(n: int, rng: np.random.Generator, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p) (symmetric U‑shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a /= a.sum()
    return a


def double_exp_init(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n-1-i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def beta_init(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Discretised Beta(a,b) distribution on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    a[a < 0] = 0.0
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of self‑convolution (log‑sum‑exp) with gradient."""
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run a cascade of L‑BFGS optimisations for decreasing smoothing."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of hard max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """Projected sub‑gradient descent (identical to the reference)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    step0 = 0.3
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        mean_g = np.mean(g)
        d = -(g - mean_g)          # sum(d)=0

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              max_iter: int = 30):
    """
    LP refinement step: solve
        max Σ b_j   s.t.   b ≥ 0,  a * b ≤ M
    where M is the current max‑convolution value.
    The solution is mixed with the current a.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active constraints (within numerical tolerance)
        eps = M * 1e-12 + 1e-12
        active_idx = np.where(conv >= M - eps)[0]
        if active_idx.size == 0:
            break

        # Build A_ub for the active rows
        A_rows = []
        for k in active_idx:
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            row = np.zeros(n, dtype=np.float64)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                row[j_start:j_end + 1] = a[k - idx]
            A_rows.append(row)
        A_ub = np.vstack(A_rows)

        # Linear program (maximise sum b  → minimise -sum b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=np.full(A_ub.shape[0], M),
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # If LP fails we simply break; further iterations unlikely to help.
            break

        b = res.x
        sumb = b.sum()
        if sumb <= 0.0:
            break
        b = b / sumb                     # enforce Σb = 1

        # Mix a and b (small step towards b)
        t = 0.15
        a_new = (1 - t) * a + t * b
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """Simple greedy hill‑climb on the simplex."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n_full = 4000
    fft_len_full = next_pow2(2 * n_full - 1)

    # ------------------------------------------------------------------
    # 0️⃣ Small‑scale exploratory search (fast)
    # ------------------------------------------------------------------
    n_small = 1024
    fft_len_small = next_pow2(2 * n_small - 1)

    best_small = uniform_init(n_small)
    best_small_score = fast_score(best_small, fft_len_small, n_small)

    start_time = time.time()
    phase0_deadline = start_time + 0.20 * budget_s   # ~200 s for the cheap phase

    # a collection of parameter ranges
    power_params = np.linspace(0.1, 2.5, 12)
    exp_params = np.linspace(0.0005, 0.005, 12)
    beta_params = np.linspace(0.3, 1.5, 12)

    while time.time() < phase0_deadline:
        # ---- random Dirichlet initialisations ----
        a = random_dirichlet_init(n_small, rng, alpha=rng.uniform(0.3, 2.0))
        score = fast_score(a, fft_len_small, n_small)
        if score < best_small_score:
            best_small_score, best_small = score, a.copy()

        # ---- parametric families ----
        p = rng.choice(power_params)
        a = power_law_init(n_small, rng, p)
        score = fast_score(a, fft_len_small, n_small)
        if score < best_small_score:
            best_small_score, best_small = score, a.copy()

        lam = rng.choice(exp_params)
        a = double_exp_init(n_small, lam)
        score = fast_score(a, fft_len_small, n_small)
        if score < best_small_score:
            best_small_score, best_small = score, a.copy()

        alpha = rng.choice(beta_params)
        beta = rng.choice(beta_params)
        a = beta_init(n_small, alpha, beta)
        score = fast_score(a, fft_len_small, n_small)
        if score < best_small_score:
            best_small_score, best_small = score, a.copy()

        # ---- brief exponentiated‑gradient polishing ----
        # a few iterations (≈ 300) on the small problem
        small_deadline = time.time() + 0.30   # 0.3 s per trial
        a_opt, sc_opt = exp_gradient_optimize(a, small_deadline,
                                              n_small,
                                              fft_len_small,
                                              fft_len_small)
        if sc_opt < best_small_score:
            best_small_score, best_small = sc_opt, a_opt.copy()

    print(f"[phase0] small‑scale best score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Upsample to full size and normalise
    # ------------------------------------------------------------------
    x_small = np.linspace(0, 1, n_small)
    x_full = np.linspace(0, 1, n_full)
    a_full = np.interp(x_full, x_small, best_small)
    a_full = np.clip(a_full, 0.0, None)
    a_full_sum = a_full.sum()
    if a_full_sum > 0.0:
        a_full = a_full / a_full_sum
    else:
        a_full = np.full(n_full, 1.0 / n_full)

    # ------------------------------------------------------------------
    # 2️⃣ Assemble a diverse pool of starting points
    # ------------------------------------------------------------------
    candidates = [
        a_full,
        uniform_init(n_full),
        spectral_init(n_full),
    ]

    # add some parametrised guesses at full size (cheap)
    for _ in range(3):
        p = rng.uniform(0.1, 2.5)
        candidates.append(power_law_init(n_full, rng, p))
        lam = rng.uniform(0.0005, 0.005)
        candidates.append(double_exp_init(n_full, lam))
        a = beta_init(n_full, rng.uniform(0.3, 1.5), rng.uniform(0.3, 1.5))
        candidates.append(a)

    best_a = None
    best_score = np.inf
    for cand in candidates:
        cand = cand / cand.sum()
        sc = fast_score(cand, fft_len_full, n_full)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] full‑size best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget distribution
    # ------------------------------------------------------------------
    time_now = time.time()
    deadline = time_now + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS continuation (polishing)
    # ------------------------------------------------------------------
    phaseA_deadline = time_now + 0.25 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 1200

    while time.time() < phaseA_deadline:
        a_start = random_dirichlet_init(n_full, rng, alpha=rng.uniform(0.3, 2.0))
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phaseA_deadline,
                                            fft_len_full, fft_len_full, n_full)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phaseB_deadline = time_now + 0.55 * budget_s
    if time.time() < phaseB_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phaseB_deadline,
                                                   n_full,
                                                   fft_len_full,
                                                   fft_len_full)
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected sub‑gradient descent
    # ------------------------------------------------------------------
    phaseC_deadline = time_now + 0.75 * budget_s
    if time.time() < phaseC_deadline:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phaseC_deadline,
                                                  n_full,
                                                  fft_len_full)
        print(f"[subgrad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – LP refinement (new ingredient)
    # ------------------------------------------------------------------
    phaseD_deadline = time_now + 0.90 * budget_s
    if time.time() < phaseD_deadline:
        best_a, best_score = lp_refine(best_a,
                                       phaseD_deadline,
                                       n_full,
                                       fft_len_full,
                                       max_iter=25)
        print(f"[LP‑refine] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_full,
                                        n_full,
                                        deadline,
                                        max_iter=30000)
        print(f"[hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (as plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
