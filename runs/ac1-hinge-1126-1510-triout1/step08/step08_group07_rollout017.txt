"""
A multi‑stage optimiser for the step‑function bound.

1️⃣  L‑BFGS on a log‑sum‑exp smoothed max‑convolution (continuation on the
    smoothing parameter).  Gives a good coarse solution fast.

2️⃣  Adam on the *true* (non‑smoothed) max‑convolution.  We keep the
    probabilities on the simplex via the soft‑max parametrisation.
    Adam’s adaptive step‑size helps to escape shallow plateaus.

3️⃣  Very cheap incremental local search:
    move a small amount of mass from a “high‑gradient’’ index to a
    “low‑gradient’’ index.  The convolution is updated in O(n) using the
    exact analytic formula for the effect of a mass transfer, so millions
    of trial moves can be examined within the time budget.

4️⃣  A final ultra‑fine L‑BFGS polish with an almost zero smoothing
    factor.

The sequence is always kept normalised to sum = 1 during optimisation;
the true evaluation is scale‑invariant, therefore at the end we simply
multiply by √(2 n) to obtain a valid candidate for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is normalised (Σa = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, chaining the result.
    Returns the best a (sum = 1) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # start in log‑softmax space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

    return best_a, best_score

def subgradient_max(a: np.ndarray, fft_len_g: int, eps_factor: float = 1e-9):
    """
    Compute a sub‑gradient of max(a*a) w.r.t. a,
    together with the current max value and the full convolution.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_g)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    eps = max_c * eps_factor + 1e-12
    mask = (conv >= max_c - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (2 * n - 1)]  # length n
    return g, max_c, conv


def adam_optimize(a_start: np.ndarray, deadline: float,
                  fft_len_conv: int, fft_len_g: int, n: int):
    """
    Adam on the true max‑convolution.  Works in the log‑softmax space,
    thereby keeping the probability simplex automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        it += 1
        a = np.exp(y - np.max(y))
        a = a / a.sum()

        g, max_c, _ = subgradient_max(a, fft_len_g, eps_factor=1e-9)
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = np.exp(y - np.max(y))
        a_new = a_new / a_new.sum()
        cur_score = fast_score(a_new, fft_len_conv, n)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[Adam] iter {it:,d} best_score={best_score:.6f}")

    return best_a, best_score

def local_move_search(a_start: np.ndarray, deadline: float,
                      fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Large‑scale hill‑climb using exact O(n) updates of the convolution.
    Moves a small fraction of mass from a high‑gradient index to a low‑gradient
    one and accepts only improving steps.
    """
    a = a_start.copy()
    conv = full_conv(a, fft_len)[: 2 * n - 1]

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    rng = np.random.default_rng()
    it = 0
    max_frac = 0.25                 # maximum fraction of the source mass to move

    while time.time() < deadline:
        # pick source index proportional to its mass (more likely to move)
        if a.sum() == 0:
            break
        i = rng.choice(np.arange(n), p=a)          # source
        j = rng.integers(0, n)                     # target (could be any)
        if i == j or a[i] <= 1e-14:
            continue

        delta = a[i] * rng.uniform(0.0, max_frac)
        if delta <= 0.0:
            continue

        # --- incremental convolution update ---------------------------------
        conv_new = conv.copy()
        # subtract contribution of removed mass
        conv_new[i:i + n] -= 2.0 * delta * a
        # add contribution of added mass
        conv_new[j:j + n] += 2.0 * delta * a
        # second‑order delta² terms
        d2 = delta * delta
        conv_new[2 * i] += d2
        conv_new[2 * j] += d2
        conv_new[i + j] -= 2.0 * d2
        # ---------------------------------------------------------------------

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-12:
            # accept move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[local] iter {it:,d} best_score={best_score:.6f}")
        it += 1

    return best_a, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    sum_target = np.sqrt(2.0 * n)          # scaling used by the official metric
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Starting sequence (use previously discovered one if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()                     # work on the simplex (sum = 1)
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time‑budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # generous portion for L‑BFGS (continues smoothing)
    phase1_end = start_time + 0.25 * budget_s
    # Adam on true max
    phase2_end = start_time + 0.55 * budget_s
    # Incremental local moves
    phase3_end = start_time + 0.80 * budget_s
    # final ultra‑fine L‑BFGS polish
    final_end = deadline

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation with decreasing smoothing
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # occasional random perturbation of the current best
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimiser on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap incremental hill‑climb (mass transfer)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = local_move_search(
            best_a, phase3_end, fft_len_conv, n
        )
        print(f"[local move] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final ultra‑fine L‑BFGS polish (tiny smoothing)
    # --------------------------------------------------------------
    if time.time() < final_end:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [1e-07],
            maxiter_per_lam=400,
            deadline=final_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided external function
    # --------------------------------------------------------------
    # Rescale to the standard norm √(2 n) before passing to evaluate_sequence
    final_seq = (best_a * sum_target).tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
