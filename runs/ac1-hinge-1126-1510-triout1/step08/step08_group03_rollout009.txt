def local_random_search(a_initial, best_score, fft_len, n, deadline):
    a = a_initial.copy()
    best_a = a.copy()
    best_score_local = best_score
    it = 0
    # temperature schedule: start high, decays
    while time.time() < deadline:
        it += 1
        # Choose source among top k contributions
        # compute conv and contributions each iteration anyway; maybe we compute once per loop
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)
        # compute partner vector for max_idx (like earlier)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]
        # compute contributions
        contrib = a * partner
        # choose source i_src as weighted random among top indices
        # probabilistically choose from indexes with high contrib
        probs = contrib / contrib.sum()
        src = rng.choice(n, p=probs)
        # choose target j randomly weighted by low partner (i.e., 1-partner)
        inv_partner = 1 - partner
        inv_partner = np.maximum(inv_partner, 0)
        probs_target = inv_partner / inv_partner.sum()
        tgt = rng.choice(n, p=probs_target)
        if src == tgt:
            continue
        eps_max = a[src]  # can't exceed
        # choose epsilon as random fraction * eps_max
        eps = eps_max * rng.uniform(0.001, 0.5)
        a_new = a.copy()
        a_new[src] -= eps
        a_new[tgt] += eps
        new_score = fast_score(a_new, fft_len, n)
        # acceptance:
        # use simple greedy: accept if better, else possible small probability based on temperature
        if new_score < best_score_local:
            best_score_local = new_score
            best_a = a_new.copy()
            a = a_new
            # optionally shrink step_range?
        else:
            # Simulated annealing with temperature decreasing
            # temperature = some schedule
            temp = max(1e-4, 0.1 * (deadline - time.time()) / (deadline - start_time))
            if rng.random() < np.exp(-(new_score - best_score_local) / temp):
                a = a_new
        # optional: periodically recompute best if not improving for many steps.

    return best_a, best_score_local

def random_local_search(a_init, score_init, fft_len, n, deadline, rng):
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = score_init
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and max_idx only if needed
        # we compute conv for each iteration anyway for score.
        # pick src = argmax contribution to current max entry.
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        # compute partner vector as earlier
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        partner[idx] = a[max_idx - idx]

        # contributions
        contrib = a * partner
        src = np.argmax(contrib)

        # target: pick index with smallest partner (excluding src)
        sorted_idx = np.argsort(partner)
        tgt = sorted_idx[0] if sorted_idx[0] != src else sorted_idx[1]

        # compute maximum epsilon that retains positivity
        eps_max = a[src]
        if eps_max <= 0:
            # maybe choose another source eventual
            continue

        # sample epsilon fraction
        fraction = rng.uniform(0.001, 0.5)  # random
        eps = eps_max * fraction

        # create new candidate
        a_new = a.copy()
        a_new[src] -= eps
        a_new[tgt] += eps

        # optionally, renormalize due to numerical errors
        # but sum unchanged.
        new_score = fast_score(a_new, fft_len, n)

        # Accept if improvement
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # with some probability accept uphill to escape
            T = max(1e-4, (deadline - time.time()) / (deadline - start_time) * 0.1)
            if rng.random() < np.exp(-(new_score - best_score) / T):
                a = a_new
    return best_a, best_score

"""
Hybrid optimizer for the step‑function bound.

The algorithm proceeds in four stages (total budget ≤ budget_s seconds):
1. Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution.  A decreasing schedule of smoothing parameters (λ)
   drives the solution close to the true objective.
2. Long exponentiated‑gradient descent on the exact max‑convolution.
3. Deterministic hill‑climb that moves a fraction of mass from the
   coordinate that contributes most to the current maximal convolution
   entry toward the coordinate with the smallest partner value.  The
   minimal step size is set very low (1e‑6) to allow fine‑grained
   refinements.
4. A final random‑local‑search phase.  Small random transfers of mass
   between a “harmful’’ source and a “helpful’’ sink are attempted;
   downhill moves are always kept, uphill moves are occasionally
   accepted (simulated‑annealing style) to escape shallow local minima.

All iterates stay on the probability simplex (non‑negative, sum = 1).
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a has been normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex, mixing several families.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the
    FFT of a uniform target (length 2·n‑1).  Result is projected onto simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Guided hill‑climb (deterministic improvement)
# ----------------------------------------------------------------------


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry to a coordinate with the smallest
    partner value.  The step size is halved whenever a move is not improving.
    """
    step = init_step
    while time.time() < deadline:
        # compute convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # choose a destination with the smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        # re‑normalise (numerical drift)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size for next move
        else:
            step = max(step * 0.8, min_step)
    return a, score

# Randomised local search (final refinement)
# ----------------------------------------------------------------------


def random_local_search(a_init: np.ndarray,
                        score_init: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform many tiny random transfers of mass (source→target) guided by the
    current maximal convolution entry.  Downhill moves are always accepted;
    occasional uphill moves are accepted with a temperature that decays to
    zero as the deadline approaches.
    """
    start_time = time.time()
    a = a_init.copy()
    best_a = a.copy()
    best_score = score_init

    while time.time() < deadline:
        # full convolution for the current a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner vector for the maximal index
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution of each coordinate to the maximal entry
        contrib = a * partner
        src = int(np.argmax(contrib))

        # destination: index with smallest partner (avoid src)
        sorted_idx = np.argsort(partner)
        tgt = int(sorted_idx[0] if sorted_idx[0] != src else sorted_idx[1])

        eps_max = a[src]
        if eps_max <= 0.0:
            # fallback to a random feasible move
            src = int(rng.integers(0, n))
            eps_max = a[src]
            if eps_max <= 0.0:
                continue
        # random fraction of the source mass (between 0.1 % and 30 %)
        eps = eps_max * rng.uniform(0.001, 0.30)

        a_new = a.copy()
        a_new[src] -= eps
        a_new[tgt] += eps

        # compute new score
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # simulated‑annealing acceptance
            remaining = max(deadline - time.time(), 0.0)
            total = max(deadline - start_time, 1e-9)
            temp = 0.05 * (remaining / total)  # linearly decaying temperature
            if temp > 0 and rng.random() < np.exp(-(new_score - best_score) / temp):
                a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                      # length of the step function
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial candidates (use previous best if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_deadline = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.70 * budget_s   # exponentiated‑gradient
    phase3_deadline = start_time + 0.95 * budget_s   # guided hill‑climb
    # remaining time → random local search

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with a refined λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06
    ]
    maxiter_per_lam = 800   # keep per‑λ iterations modest to allow many restarts

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.4:
            a_start = random_initial(n, rng)
        elif r < 0.8:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.015), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len,
            fft_len,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len,
            fft_len,
            rng,
        )
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – guided deterministic hill‑climb with fine granularity
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase3_deadline,
            init_step=0.20,
            min_step=1e-6,
        )
        print(f"[guided hill‑climb] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – randomised local search (fine‑grained refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_local_search(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
        )
        print(f"[random local search] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
