def fast_score(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_conv = conv.max()
    return 2.0 * n * max_conv

def simulated_annealing(a0, time_deadline, n, fft_len):

    best = a.copy()
    best_score = fast_score(a, fft_len)
    cur_score = best_score
    T0 = 0.01
    T = T0
    it = 0
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        # compute step size factor: maybe decays with iteration
        # Choose i, j
        if it % 100 == 0:
            # recompute gradient and select indices
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
            max_val = conv.max()
            mask = (conv >= max_val - max_val*1e-12).astype(np.float64)
            # gradient
            a_rev = a[::-1]
            grad = 2 * np.fft.irfft(np.fft.rfft(mask, fft_len) * np.fft.rfft(a_rev, fft_len), n=fft_len)[:2*n-1]
            grad_vec = grad[n-1: n-1 + n]   # gradient on a
            # source indices where grad large
            high_idx = np.where(grad_vec > np.percentile(grad_vec, 90))[0]
            low_idx = np.where(grad_vec < np.percentile(grad_vec, 10))[0]
            if high_idx.size > 0 and low_idx.size > 0:
                i = int(rng.choice(high_idx))
                j = int(rng.choice(low_idx))
            else:
                i = rng.integers(n)
                j = rng.integers(n)
        else:
            i = rng.integers(n)
            j = rng.integers(n)
        if i == j:
            continue
        # delta sample
        max_delta = a[i]
        if max_delta <= 0:
            continue
        # choose fraction
        frac = rng.uniform(0.001, 0.1) # maybe adjust
        delta = max_delta * frac
        # propose new a
        cand = a.copy()
        cand[i] -= delta

        # ensure nonneg
        if cand[i] < 0:

        cand_score = fast_score(cand, fft_len)

"""
Propose a step‑function sequence that dramatically improves the upper bound.

The idea is to use a simple deterministic distribution:
    a_i ∝ (n−i)^p   with a small integer exponent p (here p=5).

For a non‑negative vector a with Σa_i = 1 the evaluation function reduces to
