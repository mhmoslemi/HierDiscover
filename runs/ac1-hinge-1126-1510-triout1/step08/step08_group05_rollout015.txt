"""
The quality of a step‑function is measured by
    2·n·max(conv(a,a))   with   sum(a)=1 ,   a_i ≥ 0 .
(If a is scaled the ratio does not change.)

The following algorithm tries to lower the peak of the auto‑convolution.
It starts from the best known sequence (``height_sequence_1`` if it exists,
otherwise a uniform distribution) and iteratively improves it with a
linear‑programming “mass‑expansion” step that is guaranteed to keep the
convolution bound intact.  After each LP a convex combination of the old
sequence and the new feasible one is tested; the best improvement is kept.
The whole loop runs until the time budget (≈1000 s) is exhausted or no
further improvement is possible.

The LP implements
    maximise   Σ b_i                (i.e. make the total mass as large as possible)
    subject to   a * b  ≤  M          (a convolution bound)
                 b_i ≥ 0 .
Because a feasible solution is always b = a, the program is always feasible.
If the optimal Σb_i > 1 we can mix a and the normalised version of b and
strictly reduce the maximum convolution value.  Repeating this procedure
converges to a very tight upper bound.

All heavy work (FFT for convolutions, construction of the sparse constraint
matrix) is performed with NumPy / SciPy.  The routine finishes with a final
call to the provided ``evaluate_sequence`` and returns the best sequence it
found.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return conv_full


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (a*a)[k] using FFT."""
    return full_conv_fft(a, fft_len).max()


def build_constraint_pattern(n: int):
    """
    Pre‑compute the row and column indices of the sparse matrix that
    represents the linear constraints a * b ≤ M.
    For each pair (i,j) the coefficient of b_j in row (i+j) equals a_i.
    """
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    row_inds = (i[:, None] + j).ravel()          # shape (n*n,)
    col_inds = np.tile(j, n)                     # shape (n*n,)
    return row_inds, col_inds


def solve_lp_expansion(a: np.ndarray, M: float, row_inds, col_inds, n: int):
    """
    Solve the LP:
        maximise   Σ b_i
        subject to a * b ≤ M   (convolution bound)
                   b_i ≥ 0
    Returns the optimal b (or None on failure).
    """
    # data for the sparse matrix: each a_i is repeated n times (once per j)
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (row_inds, col_inds)), shape=(2 * n - 1, n))

    # maximise Σb_i  == minimise -Σb_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS works well with sparse matrices
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        print("[LP] failed:", res.message)
        return None
    return res.x

    Search for a step‑function (a non‑negative vector whose sum is 1)
    that minimises   2·n·max(conv(a,a)).
    Returns the best sequence found before the time deadline.
    """
    np.random.seed(seed)

    # Starting point – use the previously found best if available
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(globals()["height_sequence_1"], dtype=float)
        if a.ndim != 1 or a.size == 0:

    else:

    else:

        a = np.maximum(a, 1e-12)          # avoid exact zeros
        a = a / a.sum()                    # normalise to 1

    # -------------------------------------------------------------
    # Pre‑computations
    # -------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)
    row_inds, col_inds = build_constraint_pattern(n)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin
    best_a = a.copy()
    best_score = 2.0 * n * conv_max(best_a, fft_len)

    return final_seq
