"""
Multi‑phase optimizer for the step‑function evaluation problem.

Phases
------
1️⃣  Heavy L‑BFGS on a log‑sum‑exp surrogate of the max‑convolution.
    A decreasing schedule of smoothing parameters drives the solution
    to the true non‑smooth objective.

2️⃣  Exponentiated‑gradient (multiplicative) sub‑gradient descent on the
    exact max‑convolution.  This preserves non‑negativity automatically,
    works directly on the simplex and can perform many cheap iterations.

3️⃣  Fine‑grained L‑BFGS with very small smoothing to polish the best
    candidate found by the previous phases.

The routine respects a 1000 s budget, tries a few diverse random starts,
and finally calls the external ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when ``a`` is already normalised
    (∑a = 1). Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "linear", "quadratic"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "linear":
        # decreasing linear profile: a_i ∝ (n-i)
        x = np.arange(n)[::-1] + 1
        a = x.astype(float)
    else:  # quadratic (symmetric, zero at ends)
        t = np.linspace(0, 1, n)
        a = t * (1 - t)  # parabola shape, zero at both ends
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def grad_of_max(a: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int) -> np.ndarray:
    """
    Sub‑gradient of max(a*a) w.r.t. a.
    Returns a vector of length n.
    """
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return grad


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Multi‑phase optimisation for the step sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000  # decent size that fits comfortably in the time budget

    # initialise from a previous solution if it exists
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2          # safety margin

    phase1_end = start + 0.30 * (budget_s - 0.2)     # heavy L‑BFGS
    phase2_end = start + 0.78 * (budget_s - 0.2)     # exponentiated‑gradient descent
    # remaining time → fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS with a decreasing λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 800

    # a few random starts (the loop stops when the time budget for this phase expires)
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    it = 0
    while time.time() < phase2_end:
        grad = grad_of_max(a, n, fft_len_conv, fft_len_g)
        it += 1
        eta = 0.5 / np.sqrt(it)                # decreasing step size
        # multiplicative update (exponentiated gradient)
        exp_term = np.exp(-eta * grad)
        # avoid overflow / underflow
        exp_term = np.clip(exp_term, 1e-12, 1e12)
        a = a * exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum

        new_score = fast_score(a, fft_len_conv, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[EG] improved best score={best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 3 – fine‑tuned L‑BFGS (very small λ) to polish the result
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07]
        fine_maxiter = 500
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
