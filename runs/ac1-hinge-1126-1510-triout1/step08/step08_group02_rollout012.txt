def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math, sys
    from scipy.optimize import minimize
    
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    
    n = 4000
    fft_len = next_pow2(2 * n - 1)
    
    # Get initial a
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim ==1 and prev.size>0:
            # interpolate to n
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            a0 = np.interp(x_new,x_old,prev)
            a0 = np.clip(a0,0,None)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0/n)
    else:
        a0 = np.full(n, 1.0/n)
    # also generate spectral
    a_spec = spectral_init(n)
    # random
    a_rand = random_initial(n, rng)
    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_a = cand.copy()
            best_score = sc
    print(f"[init] best_score={best_score:.6f}")
    
    start_time = time.time()
    deadline = start_time + budget_s - 0.5 # safety
    
    # Phase L-BFGS: from best_a, spectral, random, limited restarts
    # allocate 200s
    phase1_deadline = start_time + 0.20 * budget_s
    lam_schedule = [0.5,0.3,0.2,0.12,0.07,0.04,0.025,0.015,0.008,0.004,0.002,0.001,0.0005,0.0002,0.0001,0.00005]
    maxiter_per_lam = 800  # keep moderate
    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam, phase1_deadline, fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt; best_a = a_opt.copy()
            print(f"[L-BFGS] new best={best_score:.6f}")
    
    # Phase 2: subgradient descent with backtracking line search
    phase2_deadline = start_time + 0.75 * budget_s
    a = best_a.copy()
    best_local_a = a.copy()
    best_local_score = best_score
    it = 0
    step0 = 0.6
    while time.time() < phase2_deadline:
        # compute convolution via FFT quickly
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_val = conv.max()

        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n-1):(n-1+n)]

        d = -(g - np.mean(g))

                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # a*a (full)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

"""
Propose a low‑score step function (non‑negative sequence).

Algorithm
---------
1. Initialise a distribution on the probability simplex (sum = 1):
   - Use the previously found best sequence (if available) interpolated
     to the current length 4000.
   - Add a spectral guess and a random Dirichlet‑like draw.
2. Run a long continuation of L‑BFGS on a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor
   (lam_factor) is decreased from 0.5 down to 1e‑7, and each stage is
   allowed many iterations.  The best point over all stages is kept.
3. Refine the best point with projected sub‑gradient descent on the
   exact max‑convolution.  A simple back‑tracking line search guarantees
   monotone improvement.
4. Final polishing with a greedy hill‑climb that moves a tiny random
   fraction of mass between two coordinates; the exact convolution
   (np.convolve) is used for the final evaluation.
5. Return the best sequence found before the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, Σx=1}.
    Uses the O(n log n) sorting algorithm.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):

    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    # stable log‑sum‑exp
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z           # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution (∂conv / ∂a = rev(a) * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally numerical problems appear – just skip this lam.
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the
    FFT of a uniform target (length 2·n‑1).  The result is projected onto
    the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 25000):
    """
    Greedy hill‑climb using the exact convolution (np.convolve) for
    evaluation.  Moves a random small fraction of mass from one index to
    another and keeps only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng_local = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng_local.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng_local.random() * 0.04
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        # exact evaluation (more accurate than FFT)
        conv = np.convolve(best, best)
        new_score = float(2.0 * n * conv.max() / (best.sum() ** 2))
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000                       # length of the step function
    fft_len = next_pow2(2 * n - 1)  # power‑of‑two for FFT speed

    # 1️⃣  Use previous best if available (interpolated to current length)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # 2️⃣  Spectral guess
    a_spec = spectral_init(n)

    # 3️⃣  Random guess
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – thorough L‑BFGS on a decreasing smoothing schedule
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 1500  # generous per‑lam budget

    # Single, well‑initialised run (starting from the best candidate)
    a_opt, sc_opt = lbfgs_continuation(best_a,
                                       lam_schedule,
                                       maxiter_per_lam,
                                       phase1_deadline,
                                       fft_len, fft_len,
                                       n)
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        print(f"[L‑BFGS] improved to {best_score:.6f}")

    # Allow a few additional random restarts if time permits
    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑restart] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with back‑tracking line search
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * budget_s
    a = best_a.copy()
    best_local_a = a.copy()
    best_local_score = best_score
    it = 0
    step0 = 0.6  # initial step magnitude

    while time.time() < phase2_deadline:
        # convolution and exact max (FFT for speed)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (max‑tight) constraints
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that preserves the sum (centered gradient)
        d = -(g - np.mean(g))

        # back‑tracking line search (ensure strict improvement)
        alpha = step0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(12):
            a_new = a + alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_local_score:
                a = a_new
                best_local_score = sc_new
                best_local_a = a_new.copy()
                improved = True
                break
            else:
                alpha *= 0.5
        # if no improvement, just keep the current a (step will shrink later)
        it += 1
        if it % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[subgrad] it={it}, best={best_local_score:.6f}, elapsed={elapsed:.1f}s")

    best_a = best_local_a
    best_score = best_local_score
    print(f"[subgrad] finished – best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (exact convolution for polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=25000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
