"""
Propose a step‑function sequence (non‑negative, sum > 0) that minimizes
the external ``evaluate_sequence`` objective.

The algorithm consists of four stages, each using a different optimisation
paradigm:

1. **Multi‑restart L‑BFGS** on a log‑sum‑exp (smooth) surrogate for the
   max‑convolution.  A gradually decreasing smoothing parameter λ drives the
   solution towards the true objective.  Many random restarts are performed
   until the allocated time slice is exhausted.

2. **Projected Adam** on the exact non‑smooth objective.
   The sub‑gradient of max‑convolution is computed via FFT, an Adam update
   is applied and the iterate is projected onto the probability simplex
   (∑a = 1, a ≥ 0).

3. **Fine‑grained hill‑climb** – random transfers of a tiny amount of
   mass from a “source’’ index (large sub‑gradient) to a “target’’ index
   (small sub‑gradient).  Only strictly improving moves are kept.

4. **Polish L‑BFGS** with an ultra‑small λ to obtain the final candidate.

All stages respect the overall 1000 s budget.  The best sequence found is
returned as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective under the normalisation Σa = 1.
    Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpinán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random simplex vector."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "trimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.025)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # trimodal
        thirds = n // 3
        a = np.full(n, 1.0 / n)
        a[:thirds] *= 1.5
        a[2 * thirds :] *= 0.8
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)          # shape (2n-1,)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the relevant slice corresponds to indices (n‑1) … (n‑1+n‑1)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # logits (ensures positivity)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the true sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- compute convolution and active mask -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)  # 1 where max is attained

        # ----- sub‑gradient of the max -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.7 / np.sqrt(it)          # a slightly larger initial step than before
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to bypass plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.008)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.clip(a, 0.0, None)
            if a.sum() > 0:
                a = a / a.sum()

    return best_a, best_score

# Aggressive hill‑climb (small mass transfers)
# ----------------------------------------------------------------------


def aggressive_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 120000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained greedy hill‑climb that moves a tiny fraction of mass
    from “high‑gradient’’ indices to “low‑gradient’’ ones.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution & mask of active constraints
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fr = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fr * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # select source / target based on quantiles
        src_thr = np.percentile(g, 85)
        tgt_thr = np.percentile(g, 15)
        src_idx = np.where(g >= src_thr)[0]
        tgt_idx = np.where(g <= tgt_thr)[0]

        if src_idx.size == 0 or tgt_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(src_idx)
            j = rng.choice(tgt_idx)

        if i == j or best[i] < 1e-12:
            it += 1
            continue

        # try a handful of transfer fractions, keep the first improvement
        improved = False
        for frac in (0.01, 0.03, 0.06, 0.10):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            # re‑normalisation is not needed – total mass unchanged
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best = cand
                best_score = new_score
                improved = True
                break
        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    projected Adam on the true non‑smooth objective, (c) performs a
    fine‑grained hill‑climb and (d) finishes with a short ultra‑precise
    L‑BFGS polish.
    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Problem size – a slightly larger n gives a better theoretical bound
    # ------------------------------------------------------------------
    n = 5000

    # a simple interpolation of it onto the new grid; fall back to uniform.

    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)

            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.15  # tiny safety margin

    # Phase 1 – many random‑restart L‑BFGS on a smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        5e-04,
        2e-04,
        1e-04,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
    ]
    maxiter_per_lam = 800

    restarts = 0
    while time.time() < lbfgs_deadline:
        # occasional perturbation of the current best to keep diversity
        if rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.010)
            a_start = best_a + rng.normal(0.0, sigma, size=n)
            a_start = np.clip(a_start, 0.0, None)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            lbfgs_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        restarts += 1
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(
                f"[L‑BFGS restart #{restarts}] new best score={best_score:.6f}"
            )

    print(f"[L‑BFGS phase completed] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < adam_deadline:
        best_a, best_score = subgradient_adam(
            best_a, adam_deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < hill_deadline:
        best_a, best_score = aggressive_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            hill_deadline,
            max_iter=200000,
        )
        print(f"[Hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final ultra‑fine L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07]
        fine_maxiter = 600
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the sequence for potential future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
