def hill_climb_adaptive(a: np.ndarray, curr_score: float,
                        fft_len: int, n: int,
                        deadline: float, rng: np.random.Generator,
                        max_fraction_start: float = 0.20,
                        max_fraction_end: float = 0.02,
                        recompute_every: int = 1500):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0
    start_time = time.time()
    while time.time() < deadline:
        # fraction decreases linearly towards end
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = max_fraction_start - (max_fraction_start - max_fraction_end) * (elapsed / total)
        # pick indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * frac
        if delta <= 0.0:
            continue
        # incremental update
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        # periodic full recompute for drift control
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
        it += 1
    return best_a, best_score

"""
Optimises a non‑negative step function to minimise the official evaluation
score.

The optimiser runs four independent phases inside the 1000 s budget:

1. **Broad random sampling** – explores a wide family of distributions
   (uniform, exponential, power‑law, edge‑heavy, bimodal, sparse, …) and
   keeps the best candidate.

2. **Short L‑BFGS on a smooth approximation** – runs a few restarts of
   L‑BFGS on a log‑sum‑exp smoothed version of the max‑convolution.  The
   optimisation works in the soft‑max (log‑softmax) parametrisation and
   respects the simplex constraint automatically.

3. **Exponentiated‑gradient descent on the true ∞‑norm** – uses a
   multiplicative update (`a ← a·exp(−η·∇)`) with a step‑size
   `η = η₀/√t`.  The gradient is taken with respect to a *relaxed* mask
   that contains all convolution entries within a small relative
   tolerance of the current maximum, which gives a richer descent
   direction than the pure max‑mask.

4. **Adaptive greedy hill‑climbing** – repeatedly moves a random
   fraction of the mass from a larger entry to a smaller one,
   updating the full convolution incrementally (O(n) per move).  The
   maximum allowed move fraction is decreased linearly from a large
   value (≈20 %) to a fine‑grained value (≈2 %) as the time budget is
   exhausted, allowing both coarse exploration and delicate refinement.

All phases keep track of the best sequence seen so far and the final
sequence is returned as a plain list of floats.  The implementation
conforms to the required API (`propose_candidate`) and uses only
standard scientific Python libraries.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score: 2·n·max(a*a) for a simplex vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          rel_eps: float = 1e-3):
    """
    Exponentiated‑gradient descent directly on the L∞‑objective.
    Uses a mask that contains all convolution entries within `rel_eps`
    of the current maximum.  Step size η = η₀ / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * rel_eps
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (soft) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adaptive greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_adaptive(a: np.ndarray,
                        curr_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_fraction_start: float = 0.20,
                        max_fraction_end: float = 0.02,
                        recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index. The allowed fraction shrinks linearly
    from `max_fraction_start` to `max_fraction_end` as time progresses.
    Convolution is updated incrementally (O(n) per move) and periodically
    recomputed exactly to avoid drift.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)                               # reusable buffer
    start = time.time()
    accepted = 0

    while time.time() < deadline:
        # Linear schedule for the move size
        elapsed = time.time() - start
        total = deadline - start
        frac = max_fraction_start - (max_fraction_start - max_fraction_end) * (elapsed / total)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # Periodically recompute to keep numerical error in check
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score

    # ------------------------------------------------------------------
