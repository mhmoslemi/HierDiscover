def simulated_hill_climb(a_start, deadline):
    a = a_start.copy()
    score = fast_score(a, fft_len_conv, n)
    best_a = a.copy()
    best_score = score
    it = 0
    while time.time() < deadline:
        # compute conv and contributions as earlier
        conv, max_val, mask = compute_conv_and_mask(a, fft_len_conv)
        # Determine the 'most harmful' indices: those with large a_i * partner_i (like earlier)
        # partner array for the max index? Actually use mask to locate all t that are max.
        # For each t among max indices, compute contributions.
        # Let's just pick i_max as index with largest a[i] * sum of partners for max indices? Simpler:
        # compute for each i: contribution_i = sum_{t where mask[t]==1} a[i] * a[t-i] (with proper domain)
        # This is heavy O(n*m). Instead just use the earlier directed approach:
        max_idx = np.argmax(conv)
        # Compute partner as in guided hill climb:
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        partner[idx] = a[max_idx - idx]
        contrib = a * partner
        # select i_source = argmax(contrib)
        i_src = np.argmax(contrib)
        # choose random i_target != i_src, maybe weighted by low partner
        # Weighted random choice based on small partner values:
        probs = 1.0 / (partner + 1e-12)
        probs[i_src] = 0
        probs = probs / probs.sum()
        i_tgt = rng.choice(n, p=probs)
        # transfer delta = min(a[i_src], step_size)
        step_frac = 0.02  # fraction of source mass
        delta = min(a[i_src] * step_frac, 0.01)  # not too large
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_tgt] += delta
        # re-normalize (should sum to 1 but adjust for float)
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, fft_len_conv, n)
        # acceptance with temperature
        T = max(1e-4, 0.1 * (deadline - time.time()) / (deadline - start_time))  # simple linear cooling
        if new_score < score or rng.random() < np.exp(-(new_score - score) / T):
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        it += 1
    return best_a, best_score

def simulated_hill_climb(a_start, deadline, n, fft_len, fft_len_g, rng, start_time):
    a = a_start.copy()
    score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = score
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_idx = np.argmax(conv)
        # compute partner array
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]
        # compute contributions
        contrib = a * partner
        i_src = np.argmax(contrib)
        # choose destination randomly weighted by low partner
        # same as earlier but random
        probs = 1.0 / (partner + 1e-12)
        probs[i_src] = 0
        probs = probs / probs.sum()
        i_dst = rng.choice(n, p=probs)
        # transfer fraction
        delta = min(a[i_src], 0.02 * a[i_src])  # small fraction
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        # acceptance with temperature
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        temperature = max(1e-5, 0.05 * (1 - elapsed/total))
        if new_score < score or rng.random() < np.exp(-(new_score - score) / temperature):
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        iteration += 1
    return best_a, best_score

"""
Hybrid optimisation for the step‑function bound (new version).

The routine builds on the previous hybrid approach but adds:
  • a richer set of initial candidates (including several parametric families);
  • a deeper L‑BFGS continuation with a finer λ‑schedule;
  • an improved exponentiated‑gradient descent that falls back to a
    projected‑gradient step when the multiplicative update does not help;
  • a simulated‑annealing hill‑climb that moves mass from the most
    “harmful’’ coordinate to a random low‑partner coordinate, accepting
    occasional worsening moves;
  • a final deterministic guided hill‑climb for polishing.

All phases respect the global time budget (default 1000 s) and operate
entirely on the probability simplex (∑aᵢ=1, aᵢ≥0).  The best sequence
found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (len = 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a lives on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial‑candidate factories
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the FFT of
    a uniform target (length 2·n‑1).  The result is projected onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Any exception – skip this λ
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    Occasionally falls back to a Euclidean projected‑gradient step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # mask of (nearly) maximal entries – a tiny tolerance helps stability
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # decreasing step size
        eta = 1.0 / np.sqrt(it + 1.0)

        # multiplicative EG update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a_new = a * exp_term
        s = a_new.sum()
        if s <= 0.0:
            # fall back to projected gradient if something went wrong
            a_new = simplex_projection(a - eta * g)
        else:
            a_new /= s

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        # accept only if not worsening; otherwise keep old a
        if score <= fast_score(a, fft_len_conv, n):
            a = a_new
        else:
            # try a simple projected‑gradient move as a rescue
            a_proj = simplex_projection(a - eta * g)
            proj_score = fast_score(a_proj, fft_len_conv, n)
            if proj_score < best_score:
                best_score = proj_score
                best_a = a_proj.copy()
            if proj_score < fast_score(a, fft_len_conv, n):
                a = a_proj
        it += 1
    return best_a, best_score


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: move a small fraction of mass from the coordinate
    that contributes most to the current maximal convolution entry to a
    coordinate with the smallest partner value.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination with smallest partner (avoid i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.8, min_step)
    return a, score


def simulated_hill_climb(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         fft_len_g: int,
                         rng: np.random.Generator,
                         overall_start: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb.
    Moves a small amount of probability mass from the most harmful
    coordinate (largest a[i]*partner[i]) to a random low‑partner coordinate.
    Occasionally accepts worsening moves to escape local minima.
    """
    a = a_start.copy()
    score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = score
    it = 0

    while time.time() < deadline:
        # current convolution and the index achieving the current maximum
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner values for this maximal offset
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution of each coordinate to the maximal entry
        contrib = a * partner
        i_src = np.argmax(contrib)

        # choose a destination biased towards small partner values
        probs = 1.0 / (partner + 1e-12)
        probs[i_src] = 0.0
        probs /= probs.sum()
        i_dst = rng.choice(n, p=probs)

        # transfer a modest fraction of the source mass
        frac = max(0.001, 0.02 * (1.0 - (time.time() - overall_start) / (deadline - overall_start)))
        delta = min(a[i_src] * frac, 0.01)
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)

        # linear cooling schedule
        elapsed = time.time() - overall_start
        total = deadline - overall_start
        temperature = max(1e-5, 0.05 * (1.0 - elapsed / total))

        accept = False
        if new_score < score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - score) / temperature):
                accept = True

        if accept:
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # occasional random restart to keep exploration alive
            if rng.random() < 0.001:
                a = add_noise(a, sigma=0.02, rng=rng)

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                     # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previous solution if available (interpolate)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 1:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.clip(a_start, 0.0, None)
            a_start /= a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Diverse initial population
    # ------------------------------------------------------------------
    candidates = [
        a_start.copy(),
        spectral_init(n),
        random_initial(n, rng),
        # extra parametric families
        np.exp(-rng.uniform(0.001, 0.02) * np.arange(n)),                     # exponential
        (np.arange(1, n + 1, dtype=float) ** (-rng.uniform(0.2, 2.5)))      # power law
    ]
    for i, c in enumerate(candidates):
        c = np.maximum(c, 0.0)
        c /= c.sum()
        candidates[i] = c

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    phase1_end = start_time + 0.25 * budget_s   # L‑BFGS heavy continuation
    phase2_end = start_time + 0.70 * budget_s   # Exponentiated‑gradient descent
    phase3_end = start_time + 0.90 * budget_s   # Simulated annealing hill‑climb
    # remaining time → final deterministic guided hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – deep L‑BFGS continuation with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-5, 1e-5
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        r = rng.random()
        if r < 0.5:
            a0 = random_initial(n, rng)
        elif r < 0.8:
            a0 = add_noise(best_a, sigma=rng.uniform(0.001, 0.01), rng=rng)
        else:
            a0 = spectral_init(n)

        a_opt, sc_opt = lbfgs_continuation(
            a0,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_hill_climb(
            best_a,
            phase3_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
            start_time,
        )
        print(f"[sim‑hill] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final directed hill‑climb (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_step=0.15,
            min_step=0.001,
        )
        print(f"[guided hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the reference routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
