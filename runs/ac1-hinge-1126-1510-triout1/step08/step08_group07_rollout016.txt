    best_a = None

    # Phase 3: Simulated annealing + hill climb

"""
Optimise a non‑negative step function to minimise the evaluation metric.
The algorithm consists of several complementary phases:

1️⃣  A pool of diverse initial distributions (uniform, exponential,
    power‑law, cosine bump, truncated Gaussian, edge‑heavy, bimodal …)
    is generated.  Each start is refined by a continuation of L‑BFGS on a
    smooth approximation of the max‑convolution (log‑sum‑exp).  The
    smoothing parameter λ is decreased progressively (0.5 → 1e‑8).

2️⃣  The best result is further improved by exponentiated‑gradient descent
    on the simplex (mirror descent) which uses a true sub‑gradient of the
    max‑convolution.

3️⃣  A long simulated‑annealing / hill‑climbing stage moves small amounts
    of mass from “high‑gradient’’ indices to “low‑gradient’’ ones.
    For each selected pair a tiny line‑search over several fractions of the
    source mass is performed; only strictly improving moves are kept.

4️⃣  A final ultra‑low‑λ L‑BFGS polish extracts any remaining numerical
    improvement.

All phases respect non‑negativity and the simplex constraint
(sum = 1) – the evaluation function is scale‑invariant, so the final
sequence can be passed directly to ``evaluate_sequence``.
The total running time is kept below the allotted 1000 s.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (len = 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is normalised (∑a = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ---- softmax (stable) ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. convolution entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ---- back‑propagation through the convolution ----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through softmax ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, carrying the result forward.
    Returns the best a (∑a=1) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # stay in log‑softmax space

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # optimisation failed – skip this λ
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true
    sub‑gradient of the (non‑smoothed) max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (size n)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # exponentiated‑gradient update
        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb.  For each iteration a source index with a
    large sub‑gradient and a low‑gradient target are chosen; several
    fractions of the source mass are tried and the best improving move
    is applied.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # fractions examined for each move – denser near zero
    fractions = [0.001, 0.003, 0.006, 0.010, 0.020,
                 0.040, 0.070, 0.10, 0.15, 0.25, 0.35]

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (unsmoothed) max
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # pick source / target based on gradient percentiles
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine", "gauss"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "cosine":
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    else:  # truncated Gaussian
        sigma = rng.uniform(0.25, 0.45) * n
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    The method builds a pool of diverse starts, refines each with a
    decreasing‑λ L‑BFGS continuation, then applies mirror‑descent,
    simulated‑annealing / hill‑climb and a final ultra‑low‑λ polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                   # length of the sequence
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise from a previously found sequence if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time‑budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    phase1_end = start_time + 0.35 * budget_s   # L‑BFGS heavy phase
    phase2_end = start_time + 0.60 * budget_s   # mirror descent
    phase3_end = start_time + 0.90 * budget_s   # annealing / hill‑climb
    # remaining time for final polish

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with decreasing λ
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 2e-06,
        1e-06, 5e-07, 2e-07, 1e-07,
    ]
    maxiter_per_lam = 400

    # build a modest pool of diverse starts
    start_pool = [a0]
    # add a few handcrafted shapes
    for sigma_factor in [0.28, 0.33, 0.38]:
        sigma = sigma_factor * n
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
        a /= a.sum()
        start_pool.append(a)

    # add a few completely random starts
    for _ in range(5):
        start_pool.append(random_initial(n, rng))

    # optimisation on each start
    for a_start in start_pool:
        if time.time() >= phase1_end:
            break
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient (mirror descent)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing + intensive hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        # ---- simulated annealing -------------------------------------------------
        cur_a = best_a.copy()
        cur_score = best_score
        temp_start = 1.0
        temp_end = 0.001
        total_sa_steps = 15000

        for step in range(total_sa_steps):
            if time.time() >= phase3_end:
                break
            # linear cooling
            T = temp_start + (temp_end - temp_start) * step / total_sa_steps

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or cur_a[i] <= 0.0:
                continue
            # move a random fraction of the source mass
            frac = rng.uniform(0.0, 0.12)
            delta = cur_a[i] * frac
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)

            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                cur_a = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = cur_a.copy()
                    print(f"[SA] step {step} new best {best_score:.6f}")

        # ---- deterministic hill‑climb -------------------------------------------
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=300000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final ultra‑low‑λ L‑BFGS polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [1e-08, 5e-09, 1e-09],
            maxiter_per_lam=600,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the external evaluator
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
