"""
Optimiser for the step‑function challenge.

The search works in three layers and repeatedly restarts while time
remains:
  1. A few short L‑BFGS runs on a log‑sum‑exp smoothing of the
     max‑convolution (helps to find a good basin).
  2. A long exponentiated‑gradient descent on the true max‑convolution
     (sub‑gradient descent on the L∞‑objective).
  3. A simulated‑annealing hill‑climber that moves a fraction of the
     mass from a large entry to a smaller one; the convolution is
     updated incrementally (O(n) per move) and periodically recomputed
     exactly.

The optimiser works with a variable problem size (default 8000) and
exploits the fact that the evaluation is scale‑invariant – the
distribution is kept normalised to sum‑one, so the fast_score
`2·n·max(a*a)` equals the official score.

The routine returns the best list of non‑negative heights found before
the 1000 s deadline.
"""

import time
import math
import numpy as np
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS on a smoothed version of max‑convolution (optional)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """Smooth approximation of 2·n·max(a*a) where a = softmax(y)."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = fft(a, fft_len)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                    # d smooth_max / d conv_i

    # back‑propagation through convolution (reverse a)
    a_rev = a[::-1]
    fw = fft(w, fft_len)
    fa_rev = fft(a_rev, fft_len)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS over a decreasing λ‑schedule."""
    try:
        from scipy.optimize import minimize
    except Exception:
        # scipy not available – skip this phase
        return a_start, fast_score(a_start, fft_len)

    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """Exponentiated‑gradient descent directly on max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = fft(a, fft_len)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len)
        fa_rev = fft(a_rev, fft_len)
        grad_conv = np.real(ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc

# Simulated‑annealing hill‑climber with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_anneal(a_start: np.ndarray,
                      start_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_fraction: float = 0.12,
                      recompute_every: int = 1500):
    """
    Greedy + annealing hill‑climb.
    Moves a random fraction of mass from a larger entry to a smaller one.
    The convolution is updated incrementally (O(n) per move) and
    periodically recomputed exactly.
    """
    cur_a = a_start.copy()
    cur_sc = start_score

    best_a = cur_a.copy()
    best_sc = cur_sc

    # exact convolution for the current a
    conv = np.convolve(cur_a, cur_a)          # length 2n‑1
    diff = np.zeros_like(conv)

    total_time = deadline - time.time()
    it = 0
    accepted = 0

    while time.time() < deadline:
        # temperature linearly cools to zero
        remaining = deadline - time.time()
        T = max(1e-8, (remaining / total_time) * (cur_sc - 1.0))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from larger to smaller entry
        if cur_a[i] < cur_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if cur_a[src] == 0.0:
            continue

        delta = cur_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff.fill(0.0)
        diff[src:src + n] -= cur_a
        diff[dst:dst + n] += cur_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_sc = 2.0 * n * new_max

        accept = False
        if new_sc < cur_sc:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_sc - cur_sc) / T):
                accept = True

        if accept:
            cur_a[src] -= delta
            cur_a[dst] += delta
            conv = conv_new
            cur_sc = new_sc
            accepted += 1

            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cur_a.copy()

        # periodic full recompute to avoid accumulation of numerical error
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(cur_a, cur_a)

        it += 1

    return best_a, best_sc

        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – annealed hill‑climb (intensive local search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_anneal(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.15,
            recompute_every=1200,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
```

This implementation expands on the previous three‑phase optimizer by:

* Using a larger problem size (`n = 8000`) for finer approximation of the optimal shape.
* Adding a temperature‑guided simulated‑annealing step in the hill‑climb phase to escape shallow local minima.
* Running several short L‑BFGS restarts with a more aggressive λ‑schedule.
* Allowing a slightly larger fraction of mass to be moved per hill‑climb step and periodically recomputing the full convolution to keep numerical errors in check.

All the helper functions are top‑level, no closures or file I/O are performed, and the routine respects the 1000 s budget while striving to beat the target bound.
