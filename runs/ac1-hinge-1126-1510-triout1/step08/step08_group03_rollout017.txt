def lowdim_objective_and_grad(theta, B, n, fft_len_conv, fft_len_g, lam):
    y = B @ theta
    a = softmax(y)  # shape n
    # compute convolution as before using FFT length
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n -1]
    max_c = conv_full.max()
    lam_val = max_c * lam + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_val)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_val * np.log(Z)
    f_obj = 2.0 * n * smooth_max
    # gradient wrt a
    w = exp_shift / Z  # d smooth_max / d conv
    # gradient conv wrt a: conv_rev = ifft(F_rev * fw) where F_rev = fft(reversed a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n-1):(n-1 + n)]
    # gradient of softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)  # shape n
    grad_theta = B.T @ grad_y   # shape K
    return f_obj, grad_theta

def simulated_annealing_optimize(a_start, deadline, n, fft_len_conv, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    t0 = 1.0  # initial temperature
    # schedule: T = t0 * (deadline - time_now) / total_time
    total = deadline - time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear cooling
        T = t0 * max(remaining / total, 0.001)
        # propose
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # propose shift
        # bound step size depending on a[i] and maybe remaining
        max_delta = a[i] * 0.1  # up to 10% of a[i]
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonneg
        if a_new[i] < 0:
            continue
        score_new = fast_score(a_new, fft_len_conv, n)
        accept = False
        if score_new < best_score:
            accept = True
        else:
            # acceptance probability
            prob = np.exp(-(score_new - best_score) / (T + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                # print improvement
                print(f"[SA] new best {best_score:.6f}")
    return best_a, best_score

"""
Hybrid search for a low‑max convolution step‑function.
1. Build a pool of diverse initial vectors (uniform, spectral, edge‑heavy,
   U‑shaped, Beta‑U distributions, random).
2. Low‑dimensional L‑BFGS on a smoothed log‑sum‑exp surrogate.
   The distribution is parameterised by a soft‑max of a few Fourier bases;
   this yields fast optimisation while still exploring the full
   4000‑dimensional space.
3. Long exponentiated‑gradient descent on the exact max‑convolution
   (sub‑gradient descent on the ℓ∞‑objective).
4. A brief simulated‑annealing stage that randomly moves a little mass
   between two coordinates, accepting occasional uphill steps.
5. Deterministic hill‑climb that always moves mass from the most
   damaging coordinate to the least damaging one.
All phases respect a total time budget (default 1000 s) and the final
sequence is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full (linear) convolution a*a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a already normalised (∑a=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'beta02', 'beta03', 'cos2'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'beta02':
        a = beta_initial(n, alpha=0.2)
    elif mode == 'beta03':
        a = beta_initial(n, alpha=0.3)
    else:  # cos² – U‑shaped (high on the edges, low in the centre)
        i = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * i / (n - 1)) ** 2
        a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """U‑shaped Beta(α,α) pdf sampled on a regular grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1) * np.power(1 - x + eps, alpha - 1)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Initialise via square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    if a.sum() <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Low‑dimensional (Fourier‑basis) L‑BFGS on a smoothed surrogate
# ----------------------------------------------------------------------


def lowdim_obj_and_grad(theta: np.ndarray,
                        B: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_g: int,
                        lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution (log‑sum‑exp) where y = B·θ and a = softmax(y).
    Returns (objective, gradient w.r.t. θ).
    """
    y = B @ theta
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    # gradient w.r.t. θ
    grad_theta = B.T @ grad_y
    return f, grad_theta


def lowdim_obj_wrapper(theta: np.ndarray,
                       B: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """Wrapper required by scipy.optimize (returns f, grad)."""
    return lowdim_obj_and_grad(theta, B, n, fft_len_conv, fft_len_g, lam_factor)


def lowdim_lbfgs_continuation(theta_start: np.ndarray,
                              B: np.ndarray,
                              lam_schedule,
                              maxiter_per_lam: int,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int) -> tuple[np.ndarray, float]:
    """
    Run low‑dimensional L‑BFGS for each smoothing factor in lam_schedule.
    Returns the best full‑size vector a found and its fast_score.
    """
    theta = theta_start.copy()
    best_a = None
    best_score = np.inf

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=lowdim_obj_wrapper,
                           x0=theta,
                           args=(B, n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[low‑dim L‑BFGS] exception:", e)
            continue
        theta = res.x
        a = softmax(B @ theta)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[low‑dim L‑BFGS] new best score={best_score:.6f}")
    if best_a is None:
        a = softmax(B @ theta)
        best_a = a
        best_score = fast_score(a, fft_len_conv, n)
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the hard max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Simulated annealing (random small mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len_conv: int,
                                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    A lightweight SA that moves a random fraction of mass between two
    coordinates.  The temperature decays linearly with the remaining time.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len_conv, n)
    best_a = a.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear cooling schedule, never below a tiny temperature
        T = max(remaining / total_time, 1e-4)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1  # ensure i != j

        # move at most 5 % of the source mass
        max_delta = a[i] * 0.05
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # safeguard against numerical drift
        if a_new[i] < 0.0:
            continue

        new_score = fast_score(a_new, fft_len_conv, n)

        if new_score < cur_score:
            accept = True
        else:
            accept_prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            accept = rng.random() < accept_prob

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[SA] new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (mass shift from most to least harmful)
# ----------------------------------------------------------------------


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry to a coordinate with the
    smallest partner value.  Accept only strict improvements.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep the step size for the next move
        else:
            step = max(step * 0.8, min_step)

    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
