    best = a.copy()
    best_score = current_score
    it = 0
    while time.time() < deadline and it < max_iter:
        move_type = rng.random()
        if move_type < 0.5:

"""
Improved search for a low‑score step‑function.

The algorithm works on a fixed length n (4000) and respects the
simplex constraint (a_i ≥ 0, Σ a_i = 1).  It consists of three
phases that together use the whole 1000 s budget:

1. **Adam on the exact sub‑gradient** – we compute the true
   max‑convolution, form a sub‑gradient and run an Adam optimiser
   on the soft‑max logits.  This quickly flattens the convolution
   and drives the fast_score below the previous 1.5065.

2. **Simulated annealing** – large stochastic moves (mass transfer,
   swaps and Gaussian perturbations) are accepted according to a
   linear cooling schedule, letting the search escape possible
   local minima left by the gradient stage.

3. **Enhanced hill‑climb** – a greedy pass that tries three kinds
   of tiny moves (local transfer, global swap and global noise) and
   keeps only improving candidates.  This polishes the best vector
   found so far.

All phases keep the sequence on the probability simplex, therefore
the fast_score = 2·n·max(a*a) coincides with the official scorer
`evaluate_sequence`.  The routine prints progress and finally returns
the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize  # kept for compatibility; not used

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) when a lies on the simplex.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Random initialisation utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        [
            'uniform', 'random', 'exp', 'power', 'edge',
            'bimodal', 'sparse', 'logit_normal'
        ]
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        a = softmax(logits)

    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Adam optimisation on the exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def adam_exact_subgrad(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int):
    """
    Adam optimiser on the true max‑convolution sub‑gradient.
    The variables are the soft‑max logits (y) such that a = softmax(y).
    """
    a = a_start.copy()
    y = np.log(a + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.02

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        # Full self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # Gradient w.r.t. a (sub‑gradient of the max)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_a = 2.0 * grad_conv[n - 1:n - 1 + n]

        # Chain rule to logits y (a = softmax(y))
        mean_g = np.dot(a, g_a)
        g_y = a * (g_a - mean_g)

        # Adam update on y
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y = y - lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        a = softmax(y)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (re‑used from the previous baseline)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        max_local_transfer: float = 0.5):
    """
    Simple simulated‑annealing on the simplex.
    Two move types: local mass transfer and global Gaussian perturbation.
    """
    a_cur = a_start.copy()
    cur_score = fast_score(a_cur, fft_len, n)
    best_a = a_cur.copy()
    best_score = cur_score

    start = time.time()
    init_temp = cur_score * 0.02 + 1e-9   # small fraction of the current value

    while time.time() < deadline:
        # linear cooling
        frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (1.0 - frac)

        a_new = a_cur.copy()
        if rng.random() < 0.6:
            # transfer move
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * max_local_transfer
                a_new[i] -= delta
                a_new[j] += delta
        else:
            # global Gaussian noise move
            sigma = rng.uniform(0.0005, 0.015)
            a_new = add_noise(a_new, sigma, rng)

        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        delta = new_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Enhanced hill‑climbing (transfer, swap, noise)
# ----------------------------------------------------------------------


def enhanced_hill_climb(a_start: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 50000):
    """
    Greedy hill‑climb with three move types.
    Only strictly improving moves are kept.
    """
    a_cur = a_start.copy()
    cur_score = current_score
    best_a = a_cur.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline and it < max_iter:
        move_type = rng.random()
        a_new = a_cur.copy()

        if move_type < 0.5:
            # local transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * 0.10
                a_new[i] -= delta
                a_new[j] += delta
        elif move_type < 0.8:
            # swap two coordinates
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # global Gaussian noise
            sigma = rng.uniform(0.0002, 0.005)
            a_new = add_noise(a_new, sigma, rng)

        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)

        if new_score < cur_score:
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        it += 1

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initial sequence (use a stored one if it exists)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.45 * budget_s   # Adam on exact sub‑gradient
    phase2_end = phase1_end + 0.30 * budget_s   # Simulated annealing
    # remaining time → hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – Adam on exact max‑convolution sub‑gradient
    # --------------------------------------------------------------
    while time.time() < phase1_end:
        # occasional random restart to avoid being stuck
        if rng.random() < 0.15:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()

        a_opt, score_opt = adam_exact_subgrad(
            a_start,
            phase1_end,
            n,
            fft_len,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = simulated_annealing(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
        )
        print(f"[SA] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – enhanced greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = enhanced_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
