def greedy_descent(a0, deadline, fft_len, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    step_factor = 0.05
    no_improve = 0

    while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k_max = np.argmax(conv)
        max_val = conv[k_max]

        # contributions for that k
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]
        i_max = np.argmax(contrib)

        # choose i_min as index with smallest a (or zero contributions)
        i_min = np.argmin(a)  # or from mask where contrib is small?
        if a[i_max] <= 0:
            break

        delta = a[i_max] * step_factor
        if delta <= 0:
            break
        # ensure not exceed a[i_max]
        delta = min(delta, a[i_max])

        # move mass
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        # compute new max
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[:2*n-1]
        max_new = conv_new.max()
        if max_new < max_val - 1e-12:
            a = a_new
            cur_score = 2 * n * max_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                # reset no_improve
                no_improve = 0
            else:
                no_improve = 0
        else:
            # no improvement, reduce step factor
            step_factor *= 0.9
            no_improve += 1
            if no_improve > 200:
                # random perturb
                a = a + np.random.default_rng().normal(0, step_factor, size=n)
                a = np.clip(a, 0, None)
                a /= a.sum()
                step_factor = max(step_factor, 1e-4)
                no_improve = 0

        # decay step factor over time
        if step_factor < 1e-4:
            step_factor = 1e-4

    return best_a, best_score

"""
Optimise a non‑negative step sequence for the supplied ``evaluate_sequence``.

Approach
--------
1.  Initialise from the best known construction (the global ``height_sequence_1``)
    if it exists, otherwise from a short parametric sweep over a few simple
    families (exponential, cosine, bimodal, …) – the sweep is cheap and gives
    a decent starting point.

2.  Refine the candidate with a *targeted* greedy descent that repeatedly
    reduces the current maximal convolution entry.
    * For the current maximal convolution index k we compute the contribution
      a[i]·a[k‑i] of each coefficient.
    * The coefficient with the largest contribution (i_max) is a natural
      source of the peak.
    * A small fraction of its mass is transferred to the index with the
      smallest coefficient (i_min).  This moves mass from the “hot” region
      towards a “cold” region and usually lowers the peak.
    * The step size (fraction of mass moved) is adapted automatically;
      successful moves increase it, otherwise it is shrunk.
    * An occasional uphill move is accepted with a tiny temperature‑based
      probability (simulated‑annealing style) to avoid being trapped in a
      shallow local minimum.

3.  The whole optimisation runs inside the global 1000 s budget.  The best
    sequence found at any time is finally evaluated with ``evaluate_sequence``
    and returned.

The method works directly with the exact max‑convolution (no smoothing) and
uses FFTs for speed – a single convolution of length 4000 costs only a few
hundred microseconds, allowing many thousands of greedy updates within the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a sequence that already satisfies ∑a = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0,∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = (np.arange(1, n + 1, dtype=float) ** p) if inc else (np.arange(n, 0, -1, dtype=float) ** p)
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:  # pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families.
    Returns the best vector (already normalised) and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15  # 15 % of overall budget

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy peak‑reduction optimiser (works on the true max‑convolution)
# ----------------------------------------------------------------------


def peak_reduction(
    a_start: np.ndarray,
    start_time: float,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from the coefficient that
    contributes most to the current maximal convolution entry towards the
    currently smallest coefficient.  The step size adapts and a tiny
    simulated‑annealing acceptance rule helps to escape shallow minima.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    rng = np.random.default_rng(12345)
    step = 0.05                     # initial fraction of mass to move
    min_step = 1e-6
    max_step = 0.5

    while time.time() < deadline:
        # ---- current convolution and its peak ---------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # contributions of each coefficient to the peak
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]

        i_max = int(np.argmax(contrib))            # worst offender
        i_min = int(np.argmin(a))                  # cheapest place to receive mass
        if i_max == i_min:
            # pick a random different low‑mass index
            low = np.where(a <= np.percentile(a, 20))[0]
            low = low[low != i_max]
            if low.size == 0:
                break
            i_min = int(rng.choice(low))

        # ---- propose a move ----------------------------------------------------
        delta = a[i_max] * step
        if delta < 1e-15:               # nothing left to move
            break
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        # ---- evaluate the proposal --------------------------------------------
        fa_new = np.fft.rfft(a_new, fft_len)
        conv_new = np.fft.irfft(fa_new * fa_new)[: 2 * n - 1]
        sc_new = 2.0 * n * conv_new.max()

        # ---- accept or reject --------------------------------------------------
        if sc_new < best_sc:
            # improvement – keep the move and slightly increase step size
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.1, max_step)
        else:
            # occasional uphill acceptance (simulated annealing)
            remain = max(deadline - time.time(), 0.0)
            # temperature decays linearly to 0
            T = best_sc * 0.01 * (remain / (deadline - start_time))
            if rng.random() < np.exp(-(sc_new - best_sc) / (T + 1e-12)):
                a = a_new
            # otherwise reject and shrink the step size
            step = max(step * 0.9, min_step)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Start from a known good construction if it is available
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to required length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len, n)
            if sc1 < best_sc:
                best_a, best_sc = a1, sc1
                print(f"[init] previous best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈15 % of budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len, budget_s, rng)
    if sc_sweep < best_sc:
        best_a, best_sc = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – greedy peak reduction (the heavy part of the budget)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    best_a, best_sc = peak_reduction(
        best_a,
        start_time,
        deadline,
        fft_len,
        n,
    )
    print(f"[peak‑reduction] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
