   start = time.time()

        elapsed = time.time() - start
        total = deadline - start
        T = T0 * (1 - elapsed/total)  # linear cooling

        # propose move:

def compute_subgradient(a, fft_len_conv, fft_len_g, n):
    # compute convolution via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    grad = 2.0 * conv_g[(n-1):(n-1+n)]   # size n
    return grad, max_val

def adam_optimize(y_start, deadline, n, fft_len_conv, fft_len_g, rng):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = np.exp(y - np.max(y))
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len_conv, n)
    # schedule for lam_factor
    lam_factor = 0.3
    while time.time() < deadline:
        t += 1
        # adapt lam_factor gradually: multiply by decay each iteration
        lam_factor = max(1e-7, 0.3 * (0.9995 ** t))
        f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        # Adam update
        m = beta1*m + (1-beta1)*grad_y
        v = beta2*v + (1-beta2)*(grad_y**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule
        lr = 0.05 / (np.sqrt(t) + 1e-8)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if t % 500 == 0:
            print(f"[Adam] iter {t}, best {best_score:.6f}")
    return best_a, best_score

"""
Propose a low‑score step‑function sequence.

Algorithm outline
-----------------
1️⃣  Multiple L‑BFGS runs on a smoothed log‑sum‑exp max objective.
    The smoothing factor (lam_factor) is decreased gradually.
2️⃣  Adam optimisation on the same smoothed objective (softmax logits) to
    fine‑tune the distribution.
3️⃣  Simulated‑annealing style random mass transfers that accept
    worsening moves with a slowly cooling temperature.
4️⃣  Deterministic gradient‑guided mass transfer (move mass from the
    highest sub‑gradient index to the lowest) as a final polish.
All phases keep the vector on the simplex (non‑negative, sum = 1) so the
evaluation is scale‑invariant and we can use the cheap `fast_score`
(`2·n·max(conv)`) for all intermediate checks.
The total running time is limited to the supplied budget (default 1000 s).
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the score for a normalised (∑a = 1) sequence:
       2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # -------- softmax (stable) ----------
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # -------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12  # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # -------- gradient wrt convolution ----------
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # -------- back‑propagation through the convolution ----------
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # -------- chain rule through softmax ----------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage
    from the result of the previous stage.  Returns the best a and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)                     # log‑softmax space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Adam on the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max‑objective in the log‑softmax
    space.  Returns the best a (∑a=1) and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    a = np.exp(y - np.max(y))
    a = a / a.sum()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    while time.time() < deadline:
        t += 1
        # slowly decay the smoothing factor
        lam_factor = max(1e-7, 0.30 * (0.9995 ** t))
        _, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = 0.05 / (np.sqrt(t) + 1e-12)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if t % 500 == 0:
            print(f"[Adam] iter {t}, best {best_score:.6f}")

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    cur_score: float,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    SA that moves a random fraction of mass from a random source index
    to a random target index, accepting worse moves with a temperature
    that linearly cools to zero.
    """
    cur = a_start.copy()
    cur_score = cur_score
    best = cur.copy()
    best_score = cur_score
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        T = 0.2 * (1.0 - elapsed / total)          # linear cooling

        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.uniform(0.001, 0.25)           # fraction of source mass
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                print(f"[SA] new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

    return best, best_score

def compute_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient of the true max‑convolution (size n) and the max value."""
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return grad, max_val


def gradient_mass_transfer(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Deterministic polishing: repeatedly move a fraction of mass from the
    index with highest sub‑gradient to the one with lowest, accepting the
    move only if it strictly improves the score.
    """
    a = a_start.copy()
    best_score = fast_score(a, fft_len_conv, n)
    best_a = a.copy()

    while time.time() < deadline:
        grad, _ = compute_subgradient(a, fft_len_conv, fft_len_g, n)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if grad[i_max] - grad[i_min] < 1e-12:
            break

        improved = False
        # try a cascade of fractions, starting with a large one
        for frac in (0.30, 0.15, 0.07, 0.03, 0.01):
            delta = a[i_max] * frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-12:
                a = cand
                best_score = cand_score
                best_a = cand.copy()
                improved = True
                break
        if not improved:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (feel free to change n – larger n gives a bigger search space)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Starting point – reuse a previously discovered sequence if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin
    phase1_deadline = start_time + 0.55 * budget_s   # heavy L‑BFGS phase
    phase2_deadline = start_time + 0.80 * budget_s   # Adam phase
    phase3_deadline = start_time + 0.95 * budget_s   # SA + polish

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS runs with decreasing smoothing
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6,
        2e-6, 1e-6, 5e-7, 2e-7,
        1e-7,
    ]
    maxiter_per_lam = 1200
    lbfgs_runs = 0

    while time.time() < phase1_deadline:
        # occasional perturbation of the current best to increase diversity
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        lbfgs_runs += 1
        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS #{lbfgs_runs}] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed max objective
    # --------------------------------------------------------------
    if time.time() < phase2_deadline:
        y_start = np.log(best_a + 1e-12)
        a_adam, score_adam = adam_optimize(
            y_start,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        if score_adam < best_score - 1e-12:
            best_score = score_adam
            best_a = a_adam.copy()
            print(f"[Adam] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing + deterministic gradient polish
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        # Simulated annealing
        a_sa, score_sa = simulated_annealing(
            best_a,
            best_score,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        if score_sa < best_score - 1e-12:
            best_score = score_sa
            best_a = a_sa.copy()
            print(f"[SA] new best {best_score:.6f}")

        # Gradient‑guided mass‑transfer polishing
        a_polish, score_polish = gradient_mass_transfer(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
        )
        if score_polish < best_score - 1e-12:
            best_score = score_polish
            best_a = a_polish.copy()
            print(f"[Polish] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided external function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
