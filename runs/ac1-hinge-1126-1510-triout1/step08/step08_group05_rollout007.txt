def hill_climb_guided(a_start, best_score, fft_len_conv, fft_len_g, n, deadline):
    a = a_start.copy()
    best_a = a.copy()
    score = best_score
    rng = np.random.default_rng()
    frac_list = [0.5, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    iteration = 0
    grad = None
    source_idx = None
    target_idx = None
    while time.time() < deadline:
        # recompute high/low indices every few iterations
        if iteration % 10 == 0:
            grad_vec = compute_subgradient(a, fft_len_conv, fft_len_g, n)
            # high and low percentiles
            high_thr = np.percentile(grad_vec, 90)
            low_thr = np.percentile(grad_vec, 10)
            high_idx = np.where(grad_vec >= high_thr)[0]
            low_idx = np.where(grad_vec <= low_thr)[0]
            if high_idx.size == 0:
                high_idx = np.arange(n)
            if low_idx.size == 0:
                low_idx = np.arange(n)
        # pick source and target
        i = rng.choice(high_idx)
        j = rng.choice(low_idx)
        if i == j:
            continue
        # try various fractions until improvement found
        improved = False
        for frac in frac_list:
            delta = a[i] * frac
            if delta <= 0:
                continue
            a[i] -= delta
            a[j] += delta
            new_score = fast_score(a, fft_len_conv, n)
            if new_score < score:
                score = new_score
                best_a = a.copy()
                improved = True
                break
            # revert move
            a[i] += delta
            a[j] -= delta
        # occasional random jitter
        if iteration % 2000 == 0:
            sigma = rng.uniform(0.0005, 0.005)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < score:
                score = cur_score
                best_a = a.copy()
        iteration += 1
    return best_a, score

   optionally improve it with a short L‑BFGS run on a smooth
   log‑sum‑exp surrogate of the max‑convolution.

2. **Heavy L‑BFGS** – run a continuation of L‑BFGS with a decreasing
   smoothing parameter λ.  This drives the solution close to a local
   optimum of the true objective.

3. **Guided stochastic hill‑climb** – repeatedly move a small
   fraction of mass from indices with a large sub‑gradient to indices
   with a small (or negative) sub‑gradient.  Fraction choices range
   from 0.5 down to 0.001, and occasional Gaussian jitter is added to
   escape plateaux.  All moves are projected onto the simplex
   (∑a = 1, a≥0).

The whole routine respects a 1000 s budget, prints intermediate
scores and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence that already satisfies
    sum(a) == 1.
    Returns 2·n·max(a*a) using an FFT‑based convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ---- softmax (stable) ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution a*a ----
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. convolution entries ----
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # any numerical issue – skip this λ
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, fft_len_conv: int, fft_len_g: int, n: int) -> np.ndarray:
    """
    Returns a sub‑gradient of the functional
        g(a) = max_k (a*a)[k]
    w.r.t. a (under the constraint sum(a)=1, a≥0).
    """
    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_val = conv.max()
    eps_mask = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps_mask).astype(np.float64)

    # back‑propagation through the masked convolution
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_g = np.fft.irfft(fa_rev * fm, fft_len_g)
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return grad


# ----------------------------------------------------------------------
# Guided stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_guided(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly transfers a fraction of mass from
    indices with large sub‑gradient to those with small sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score

    rng = np.random.default_rng()
    # a broad set of transfer fractions – larger ones help early,
    # smaller ones fine‑tune later.
    frac_list = [0.5, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    it = 0

    while time.time() < deadline:
        # recompute a high/low‑gradient split every few iterations
        if it % 10 == 0:
            g_vec = compute_subgradient(a, fft_len_conv, fft_len_g, n)
            high_thr = np.percentile(g_vec, 90)
            low_thr = np.percentile(g_vec, 10)
            high_idx = np.where(g_vec >= high_thr)[0]
            low_idx = np.where(g_vec <= low_thr)[0]
            if high_idx.size == 0:
                high_idx = np.arange(n)
            if low_idx.size == 0:
                low_idx = np.arange(n)

        i = rng.choice(high_idx)
        j = rng.choice(low_idx)
        if i == j:
            it += 1
            continue

        improved = False
        for frac in frac_list:
            delta = a[i] * frac
            if delta <= 0.0:
                continue
            # perform tentative transfer
            a[i] -= delta
            a[j] += delta
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                improved = True
                break  # keep the first improvement
            # revert move
            a[i] += delta
            a[j] -= delta
        # if none of the fractions helped we keep the unchanged a

        # occasional jitter to break out of tiny plateaux
        if it % 2000 == 0 and it > 0:
            sigma = rng.uniform(0.0005, 0.005)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Mixing helpers – convex combination with simple patterns
# ----------------------------------------------------------------------


def mix_with_simple_patterns(a: np.ndarray, score: float,
                             fft_len_conv: int, n: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Try convex combinations of the current best with a few simple
    patterns (uniform, exponential) and keep any improvement.
    """
    best_a = a.copy()
    best_score = score

    # uniform pattern
    uniform = np.full_like(a, 1.0 / n)

    # exponential pattern (the decay rate is tuned on the fly)
    rng = np.random.default_rng()
    for alpha in np.linspace(0.0, 0.2, 21):
        cand = (1 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        cur = fast_score(cand, fft_len_conv, n)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()

    # a few random exponential shapes
    for _ in range(5):
        rate = rng.uniform(0.001, 0.02)
        exp_pat = np.exp(-rate * np.arange(n))
        exp_pat /= exp_pat.sum()
        for alpha in np.linspace(0.0, 0.15, 16):
            cand = (1 - alpha) * best_a + alpha * exp_pat
            cand /= cand.sum()
            cur = fast_score(cand, fft_len_conv, n)
            if cur < best_score:
                best_score = cur
                best_a = cand.copy()
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimisation for the step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (the same size as the previous best)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient wrt max‑convolution

    # --------------------------------------------------------------
    # Initialise from the previously known best if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # A quick sweep of a few random starts (helps when the global
    # variable is poor or missing)
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for _ in range(5):
        cand = random_initial(n, rng)
        cur = fast_score(cand, fft_len_conv, n)
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation with a rich λ schedule
    # --------------------------------------------------------------
    phase1_deadline = start + 0.45 * (budget_s - 0.1)
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
        5e-06, 2e-06, 1e-06, 5e-07, 2e-07,
        1e-07, 5e-08
    ]
    maxiter_per_lam = 800

    # run L‑BFGS continuation (may finish early if time runs out)
    a_opt, score_opt = lbfgs_continuation(
        best_a,
        lam_schedule,
        maxiter_per_lam,
        phase1_deadline,
        fft_len_conv,
        fft_len_g,
        n,
    )
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[L‑BFGS heavy] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – guided stochastic hill‑climb (dominant phase)
    # --------------------------------------------------------------
    if time.time() < total_deadline:
        best_a, best_score = hill_climb_guided(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            total_deadline,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fine mixing with simple patterns (quick polishing)
    # --------------------------------------------------------------
    if time.time() < total_deadline:
        best_a, best_score = mix_with_simple_patterns(
            best_a,
            best_score,
            fft_len_conv,
            n,
            total_deadline,
        )
        print(f"[mixing] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the external function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the result for possible reuse by a later run
    globals()["height_sequence_1"] = final_seq

    return final_seq
