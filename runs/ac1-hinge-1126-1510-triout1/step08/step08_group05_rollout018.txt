def evaluate_sequence(sequence):
    # after checks, clamp negatives to zero, >1000 to 1000, compute convolution of sequence with itself.

"""
Propose a step‑function sequence that (approximately) minimizes the
evaluation function:

    2·n·max(convolution(a,a)) / (sum(a)^2)

because the denominator is invariant under scaling we may enforce
∑a = 1 and minimise 2·n·max(conv).  The optimiser works in three
phases:

1. **Fast parametric scan** – families of simple analytic shapes
   (one‑sided exponential and symmetric edge‑exponential) are examined
   on a coarse grid of the decay‑parameter `c` (where λ = c/n).  The
   best sequence found is kept as a warm start.

2. **Projected stochastic Adam** – using the exact sub‑gradient of the
   max‑convolution we run Adam (with Euclidean projection onto the
   simplex) for the remaining time.  This refines the previous candidate
   and is able to escape plateaux.

3. **Greedy random hill‑climbing** – a long‑running greedy search that
   repeatedly transfers a small random amount of mass from one index to
   another and accepts the move only if the (fast) objective improves.
   The step size is gradually reduced.

All phases respect the 1000 s budget, use only NumPy (and the external
`evaluate_sequence`), and finally return a list of non‑negative floats.
"""
import time
import math

import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1). Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (one‑sided & symmetric exponential)
# ----------------------------------------------------------------------


def exponential_one_sided(c: float, n: int) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def exponential_symmetric(c: float, n: int) -> np.ndarray:
    """
    a_i ∝ exp(-c·min(i,n-1-i)/n) – high at both edges,
    low in the centre. Normalised to sum = 1.
    """
    lam = c / n
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact sub‑gradient of the max‑conv)
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the true sub‑gradient of max(a*a).
    Returns the best sequence seen and its fast_score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max‑conv
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√t  (slightly larger start)
        step = 0.8 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # occasional jitter to avoid stagnation
        if it % 3000 == 0:
            sigma = rng.uniform(0.0005, 0.0030)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best, best_score


# ----------------------------------------------------------------------
# Random greedy hill‑climb (mass transfers)
# ----------------------------------------------------------------------


def random_hill_climb(
    a_start: np.ndarray,
    init_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly transfer a random fraction of mass from
    a random source index to a random target index, accepting only
    strictly improving moves.  The step size (transfer fraction) is
    gradually reduced as the deadline approaches.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = init_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        # linearly shrink the maximum transfer fraction
        frac_max = max(1e-5, 0.05 * (1.0 - (time.time() - deadline + (deadline - start_time)) /
                                    (deadline - start_time)))
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # pick a small fraction of the source mass
        frac = rng.uniform(0.001, frac_max)
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is unnecessary – sum is preserved and non‑negativity holds
        cur_score = fast_score(cand, fft_len, n)
        if cur_score < best_score:
            best = cand
            best_score = cur_score
            a = cand  # continue from the improved point
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function sequence to minimise
    `evaluate_sequence`.  The function respects the given time budget
    and returns the best list of floats found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                                 # length of the sequence
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Starting point – either the global variable or a uniform guess
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 1:
            # interpolate old candidate onto the current length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric search over exponential families
    # ------------------------------------------------------------------
    c_min, c_max = 0.5, 5.0
    num_grid = 120

    # one‑sided exponential
    for c in np.linspace(c_min, c_max, num_grid):
        if time.time() >= total_deadline:
            break
        a = exponential_one_sided(c, n)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # symmetric exponential (edges)
    for c in np.linspace(c_min, c_max, num_grid):
        if time.time() >= total_deadline:
            break
        a = exponential_symmetric(c, n)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    print(f"[grid] best score after parametric scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fine refinement (golden‑section) around the best c
    # ------------------------------------------------------------------
    def refine_one_sided(c_low, c_high, iters=30):
        phi = (1 + math.sqrt(5.0)) / 2.0
        low, high = c_low, c_high
        for _ in range(iters):
            if time.time() >= total_deadline:
                break
            c1 = high - (high - low) / phi
            c2 = low + (high - low) / phi
            s1 = fast_score(exponential_one_sided(c1, n), fft_len, n)
            s2 = fast_score(exponential_one_sided(c2, n), fft_len, n)
            if s1 < s2:
                high = c2
                if s1 < best_score:
                    best_score = s1
                    best_a = exponential_one_sided(c1, n)
            else:
                low = c1
                if s2 < best_score:
                    best_score = s2
                    best_a = exponential_one_sided(c2, n)
        return (low + high) / 2.0

    def refine_symmetric(c_low, c_high, iters=30):
        phi = (1 + math.sqrt(5.0)) / 2.0
        low, high = c_low, c_high
        for _ in range(iters):
            if time.time() >= total_deadline:
                break
            c1 = high - (high - low) / phi
            c2 = low + (high - low) / phi
            s1 = fast_score(exponential_symmetric(c1, n), fft_len, n)
            s2 = fast_score(exponential_symmetric(c2, n), fft_len, n)
            if s1 < s2:
                high = c2
                if s1 < best_score:
                    best_score = s1
                    best_a = exponential_symmetric(c1, n)
            else:
                low = c1
                if s2 < best_score:
                    best_score = s2
                    best_a = exponential_symmetric(c2, n)
        return (low + high) / 2.0

    # refine around the best found c (we simply use the whole interval again)
    if time.time() < total_deadline:
        best_c_one = refine_one_sided(c_min, c_max)
        print(f"[refine‑one] best c ≈ {best_c_one:.4f}")

    if time.time() < total_deadline:
        best_c_sym = refine_symmetric(c_min, c_max)
        print(f"[refine‑sym] best c ≈ {best_c_sym:.4f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected Adam on the exact objective (long run)
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        # allocate about half of the remaining budget to Adam
        adam_deadline = time.time() + 0.45 * (total_deadline - time.time())
        best_a, best_score = subgradient_adam(
            best_a, adam_deadline, fft_len, n
        )
        print(f"[Adam] improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy random hill‑climb for fine polishing
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        start_time = time.time()
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            total_deadline,
            max_iters=5_000_000,
        )
        print(f"[hill‑climb] final fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
