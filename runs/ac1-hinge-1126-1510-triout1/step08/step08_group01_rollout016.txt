"""
Multi‑phase optimizer for the step‑function challenge.

The goal is to minimise
    2·n·max(a*a)/(sum(a)²)
for a non‑negative vector a.  Because the objective is homogeneous,
we may restrict to probability vectors (sum = 1).  The optimizer works in
four stages:

1.  Several short L‑BFGS runs on a smooth log‑sum‑exp approximation of the
    max‑convolution.  This quickly finds a good basin.

2.  Long exponentiated‑gradient descent on the true max‑convolution
    (the “hard’’ objective).

3.  Targeted sub‑gradient hill‑climb: at each step we compute the
    sub‑gradient of the max‑convolution, pick a source index with large
    positive gradient and a destination with the smallest (most negative)
    gradient, move a small fraction of mass and accept the move only if
    the objective improves.

4.  A final fine‑grained random hill‑climb (as a safety net).

The whole routine respects a 1000 s budget, returns the best sequence
found and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a (sum = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# 1. L‑BFGS on a smooth approximation of the max‑convolution

def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                # shape (2n-1,)

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_phase(a_start: np.ndarray,
                lam_schedule,
                maxiter_per_lam: int,
                deadline: float,
                fft_len: int,
                n: int):
    """Run L‑BFGS over a schedule of smoothing parameters."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_phase(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       eta0: float = 10.0):
    """Exponentiated‑gradient descent with ηₜ = η₀ / √t."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        fm = np.fft.fft(mask, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Targeted sub‑gradient hill‑climb
# ----------------------------------------------------------------------


def subgradient_hill_climb(a_start: np.ndarray,
                           deadline: float,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator,
                           max_fraction: float = 0.08,
                           recompute_every: int = 500):
    """
    Hill‑climb that always moves mass from an index with a large
    positive sub‑gradient to one with the smallest (most negative)
    sub‑gradient.  The move is accepted only if it improves the
    objective.  Convolution is recomputed from scratch periodically
    to avoid drift.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # keep an exact convolution for fast updates
    conv = np.convolve(a, a)
    it = 0
    accepted = 0

    while time.time() < deadline:
        # recompute full convolution from time to time
        if it % recompute_every == 0:
            conv = np.convolve(a, a)

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # compute sub‑gradient of the current max
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        fm = np.fft.fft(mask, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # choose source (largest gradient) and destination (smallest)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] <= 1e-15:
            # fallback to a random move if pathological
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                it += 1
                continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        # evaluate candidate (full recompute – cheap enough)
        cand_score = fast_score(a_candidate, fft_len)

        if cand_score < best_score:
            # accept
            a = a_candidate
            best_a = a_candidate.copy()
            best_score = cand_score
            conv = np.convolve(a, a)          # recompute after acceptance
            accepted += 1
        else:
            # occasional random noise to escape plateaus
            if rng.random() < 0.001:
                a = add_noise(a, sigma=0.002, rng=rng)
                a /= a.sum()
                conv = np.convolve(a, a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Fine‑grained random hill‑climb (fallback)
# ----------------------------------------------------------------------


def random_hill_climb(a_start: np.ndarray,
                      curr_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_fraction: float = 0.10,
                      recompute_every: int = 2000):
    """
    Simple greedy hill‑climb that moves a random fraction of mass from a
    source index to a destination index.  Convolution is updated
    incrementally (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a_start.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)           # exact full conv
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more flexibility (still cheap)
    # ------------------------------------------------------------------
    n = 8000                 # feel free to change; 8000 works fast
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previous global solution if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # linear interpolation to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split (roughly 10% L‑BFGS, 55% exp‑grad, 30% subgrad,
    # 5% final random hill‑climb)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.10 * budget_s
    phase2_end = phase1_end + 0.55 * budget_s
    phase3_end = phase2_end + 0.30 * budget_s
    # the rest goes to the final random hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – brief L‑BFGS runs on a smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # random perturbation of the current best with 30 % probability
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_phase(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_phase(
            best_a, phase2_end, n, fft_len, rng, eta0=15.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted sub‑gradient hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_hill_climb(
            best_a, phase3_end, n, fft_len, rng,
            max_fraction=0.07, recompute_every=300
        )
        print(f"[subgrad‑hill] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final random hill‑climb as a safety net
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
        )
        print(f"[final‑hill] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the sequence as a global for possible reuse in later runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
```

This script follows the required interface (`propose_candidate`) and contains all helper functions at top‑level.  It respects the 1000 s time budget, prints intermediate progress, and stores the final sequence in the global variable `height_sequence_1` for future runs.
