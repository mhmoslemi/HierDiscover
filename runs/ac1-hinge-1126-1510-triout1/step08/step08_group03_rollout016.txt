"""
Hybrid optimisation for the convolution‑max problem (aim: < 1.503).

The optimiser works in several stages inside a 1000 s budget.

0️⃣  Small deterministic grid‑search over a few parametric families
    (exponential, power‑law, Beta‑like) to get a good warm‑start.

1️⃣  Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation of the
    max‑convolution (more λ‑values and larger max‑iter than the reference).

2️⃣  Long exponentiated‑gradient descent on the exact max‑convolution
    (as in the reference code) – this gives a very good reduction of the
    peak value.

3️⃣  Gradient descent on the *L2* error between the convolution and the
    flat target 1/(2n‑1).  Flattening the whole convolution tends to lower
    the worst entry further.

4️⃣  Deterministic hill‑climb that moves mass from the most harmful
    coordinate (largest contribution to the current peak) to the least
    harmful one, with a finer step‑size schedule.

All steps keep the vector on the probability simplex (sum == 1, a_i ≥ 0)
by means of a Euclidean projection.  The best sequence found is finally
checked with ``evaluate_sequence`` and returned as a plain list of floats.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a on the simplex (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return 2.0 * n * conv.max()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initial families
# ----------------------------------------------------------------------


def exponential_family(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i , 0<r<1."""
    a = r ** np.arange(n, dtype=float)
    a /= a.sum()
    return a


def power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) ."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a /= a.sum()
    return a


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n  # mid‑point rule
    a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of simple random families (used for restarts)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Init via the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase * 0.5
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution using log‑sum‑exp."""
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt a (log‑sum‑exp part)
    w = exp_shift / Z                      # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # back‑prop through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def flatten_conv_gradient_descent(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len_conv: int,
                                 fft_len_g: int):
    """
    Gradient descent that minimises the L2 distance between the
    self‑convolution and the flat target 1/(2n‑1).
    Uses Euclidean projection onto the simplex after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    target_val = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # error w.r.t. flat target
        err = conv - target_val

        # gradient of 0.5*||err||^2  w.r.t. a
        a_rev = a[::-1]
        fe = np.fft.fft(err, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        g = 2.0 * np.real(np.fft.ifft(fe * fa_rev))
        g = g[(n - 1):(n - 1 + n)]

        # simple decreasing step size
        eta = 0.5 / np.sqrt(it + 1.0)

        a = a - eta * g
        a = simplex_projection(a)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """Deterministic local improvement – move mass from worst to best."""
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        # partner[j] = a[max_idx - j] (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = int(np.argmax(contrib))
        # smallest partner that is not i_max
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = simplex_projection(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size (allows large moves early)
        else:
            step = max(step * 0.8, min_step)  # shrink step if not improving
    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000  # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Base candidates (previous best, spectral, uniform, parametric families)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # a small pool of deterministic families
    candidates = [a_start,
                  spectral_init(n),
                  np.full(n, 1.0 / n)]

    # add a few parametric families (grid search – cheap)
    for r in np.linspace(0.90, 0.9995, 6):
        candidates.append(exponential_family(n, r))
    for p in np.linspace(0.1, 2.0, 6):
        candidates.append(power_family(n, p))
    for a in [0.5, 1.0, 2.0]:
        for b in [0.5, 1.0, 2.0]:
            candidates.append(beta_family(n, a, b))

    # pick the best among the pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin
    phase1_deadline = start_time + 0.30 * budget_s
    phase2_deadline = start_time + 0.70 * budget_s
    phase3_deadline = start_time + 0.90 * budget_s
    phase4_deadline = deadline                    # remaining time

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with richer schedule and more restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.32, 0.22, 0.15, 0.10,
        0.07, 0.045, 0.028, 0.018, 0.011,
        0.007, 0.004, 0.0025, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.4:
            a_start = random_initial(n, rng)
        elif r < 0.8:
            a_start = np.copy(best_a)
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.012), rng=rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] global new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – flatten‑convolution gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = flatten_conv_gradient_descent(
            best_a,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[flatten‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic hill‑climb with fine step control
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
            init_step=0.12,
            min_step=1e-5,
        )
        print(f"[guided hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq


# ----------------------------------------------------------------------
# Helper for adding Gaussian noise (used in Phase 1)
# ----------------------------------------------------------------------


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_projection(noisy)
