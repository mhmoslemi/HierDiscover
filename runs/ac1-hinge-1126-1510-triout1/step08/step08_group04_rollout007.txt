def lp_optimize(a, M):
    b = cp.Variable(len(a), nonneg=True)
    conv = cp.conv(a, b)  # length 2n-1
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, max_iters=2000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    b_opt = b.value
    if b_opt is None:
        return None
    b_opt = np.maximum(b_opt, 0)  # numeric
    # Normalize to sum 1
    b_opt = b_opt / b_opt.sum()

"""
Propose a low‑score step‑function sequence.

Approach
--------
1. **Small‑scale LP refinement** – work on a modest size (n≈2000) and
   repeatedly solve a linear program

        max   ∑ b_j
        s.t.  b_j ≥ 0,
              (f * b)_k ≤ M   for all k,

   where `f` is the current sequence, `*` denotes linear convolution
   and `M = max(f*f)`.  After solving we rescale the LP result to the
   required L1‑norm `√(2n)` and mix it with `f`.  This step mimics the
   classic “alternating convex combination” method and quickly lowers
   the maximal self‑convolution.

2. **Upscaling** – linearly interpolate the best small sequence to the
   target length (n≈4000) and renormalise to the same L1‑norm.

3. **Exact exponentiated‑gradient descent** – minimise the true
   `max(f*f)` by a sub‑gradient step on the maximal convolution entry.
   The update `a ← a·exp(−η·g)` keeps the vector on the simplex; after
   each step we renormalise to `√(2n)`.

4. **Greedy hill‑climbing** – tiny random mass transfers between two
   coordinates are accepted only if they improve the maximal convolution.

The routine respects the 1000 s budget, prints progress, and finally
returns the list of heights.  The internal objective is simply the
maximum of the self‑convolution because the L1‑norm is fixed to
`√(2n)`, which makes the final `evaluate_sequence` value equal to that
maximum.
"""

import time
import numpy as np

# cvxpy is optional – the LP phase is used only if it is available.
try:
    import cvxpy as cp

    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False

# -------------------------------------------------------------
# Basic numeric utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def max_conv(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(full_conv(a, fft_len).max())


# -------------------------------------------------------------
# Linear‑programming refinement (small n)
# -------------------------------------------------------------


def lp_step_fixed_f(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max sum(b)   s.t.   b≥0,  (f*b)_k ≤ M   (all k).
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = f.shape[0]
    # Variable b
    b = cp.Variable(n, nonneg=True)
    # Linear convolution of constant f with variable b
    conv_expr = cp.conv(f, b)  # length 2n‑1
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        # ECOS is fast for dense problems of this size
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


# -------------------------------------------------------------
# Exact exponentiated‑gradient descent (large n)
# -------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
    scale: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Keeps the L1‑norm equal to ``scale`` at every step.
    Returns the best sequence and its max‑convolution value.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = max_conv(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # Full self‑convolution
        conv = full_conv(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-13 + 1e-13
        mask = (conv >= max_val - eps).astype(np.float64)

        # Sub‑gradient of max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # length n

        # Decaying step size (1/√t)
        eta = 1.0 / np.sqrt(it + 1.0)

        # Exponentiated‑gradient update (stay non‑negative)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term

        # Renormalise to the required L1‑norm
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a * (scale / s)

        cur_score = max_conv(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# -------------------------------------------------------------
# Greedy hill‑climbing polishing
# -------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    cur_score: float,
    deadline: float,
    fft_len: int,
    n: int,
    scale: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb: move a modest random fraction of mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # Move up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta
        # Sum is unchanged, but enforce exact normalization to avoid drift
        best = best * (scale / best.sum())

        new_score = max_conv(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
            best = best * (scale / best.sum())
        it += 1

    return best, best_score

def initialise_sequence(n: int, scale: float, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise a non‑negative sequence of length ``n`` with L1‑norm ``scale``.
    If a global variable ``height_sequence_1`` is present we interpolate it.
    Otherwise we start from the uniform distribution.

    if "height_sequence_1" in globals():

        if old.ndim == 1 and old.size > 0:

            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a = a * (scale / a.sum())
            return a

    # uniform start
    return np.full(n, scale / n)

        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq * (scale_large / best_seq.sum())
        best_score = max_conv(best_seq, fft_len_large)
        print(f"[Upscale] max_conv (large) = {best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # ---------------------------------------------------------
    phase2_end = start_time + 0.73 * budget_s
    if time.time() < phase2_end:
        best_seq, best_score = exp_gradient_optimize(
            best_seq,
            phase2_end,
            fft_len_large,
            n_large,
            scale_large,
        )
        print(f"[exp‑grad] best max_conv = {best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 3 – greedy hill‑climb polishing
    # ---------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = hill_climb(
            best_seq,
            best_score,
            deadline,
            fft_len_large,
            n_large,
            scale_large,
            max_iter=30000,
        )
        print(f"[hill‑climb] best max_conv = {best_score:.6f}")

    # ---------------------------------------------------------
    # Final evaluation with the official scorer
    # ---------------------------------------------------------
    final_seq = best_seq.tolist()
    final_eval = evaluate_sequence(final_seq)  # type: ignore  # provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
