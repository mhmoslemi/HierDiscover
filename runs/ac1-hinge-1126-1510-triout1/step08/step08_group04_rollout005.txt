    maxiter_per_lam = 500

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam, 'gtol': 1e-12, 'disp': False})

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_y = y.copy()
    a_opt = softmax(best_y)
    return a_opt, best_score

The evaluator rewards step functions that minimise  

\[
\text{score}= \frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a` is a non‑negative vector and `a*a` denotes its linear self‑convolution.
Because the objective is homogeneous we may work on the probability simplex  
\(\sum a_i = 1\) and simply minimise  

\[
F(a)=2\,n\;\max (a*a).
\]

The optimiser below is a hybrid of four well‑known techniques:

1. **Diverse initial population** – several parametrised families (uniform, exponential,
   power‑law, triangular, cosine, sparse, logit‑normal, …) are interpolated to the
   chosen length `n`.  
2. **Differential Evolution (DE)** – a simple population‑based global optimiser that
   respects the simplex constraint by projecting every trial vector onto the simplex.
   DE explores the space aggressively and rapidly finds good basins.
3. **Exponentiated‑gradient descent** – the exact (hard) max‑convolution gradient is
   used to perform multiplicative updates `a ← a·exp(−η·g)`.  
   This refines the DE solution on the true non‑smooth objective.
4. **Greedy hill‑climbing & short L‑BFGS polishing** – tiny mass transfers and a
   final L‑BFGS run on a smoothed log‑sum‑exp surrogate finish the optimisation.

The whole routine respects the 1000 s time budget, prints progress, and returns the
best sequence found as a plain Python list.

```python
"""
Hybrid optimiser for the step‑function “max‑convolution” objective.

* global search : Differential Evolution on the simplex
* local search  : exponentiated‑gradient descent on the exact max‑convolution,
                  followed by greedy hill‑climbing and a short L‑BFGS polish.
The routine respects a 1000 s budget and returns a list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(a*a) for a already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random initialisation utilities (many families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector using a variety of patterns."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "logit_normal",
            "triangular",
            "cosine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "logit_normal":
        logits = rng.normal(0.0, 1.0, size=n)
        a = softmax(logits)
    elif mode == "triangular":
        half = n // 2
        ramp = np.arange(1, half + 1, dtype=float)
        if n % 2 == 0:
            a = np.concatenate([ramp, ramp[::-1]])
        else:
            a = np.concatenate([ramp, ramp[-2::-1]])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == "cosine":
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)  # non‑negative on (0,π)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (used by the final L‑BFGS polish)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed version of max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c  # ≤ 0
    exp_shift = np.exp(shifted / lam)  # safe, ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient slice that corresponds to a (indices n‑1 … n‑1+n‑1)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_refine(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
):
    """Short L‑BFGS run with a decreasing smoothing schedule."""
    y = np.log(a_start + 1e-12)
    best_y = y.copy()
    best_score = fast_score(a_start, fft_len_conv, n)

    # aggressive schedule that drives lam → 0
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
        0.00002,
        0.00001,
    ]
    maxiter_per_lam = 500

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_y = y.copy()
    a_opt = softmax(best_y)
    return a_opt, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (hard max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
):
    """Multiplicative updates a ← a·exp(−η·g) stay on the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step size η ∝ 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (tiny transfers)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
):
    """Greedy local search – move a modest random fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Differential Evolution (global search)
# ----------------------------------------------------------------------


def differential_evolution(
    population: np.ndarray,
    pop_score: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    F: float = 0.8,
    Cr: float = 0.9,
):
    """Classic DE/rand/1/bin with simplex projection."""
    pop_size = population.shape[0]
    best_idx = np.argmin(pop_score)
    best_a = population[best_idx].copy()
    best_score = pop_score[best_idx]

    gen = 0
    while time.time() < deadline:
        for i in range(pop_size):
            # select three distinct individuals different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.maximum(mutant, 0.0)
            mutant = simplex_projection(mutant)

            # binomial crossover
            cross = rng.random(n) < Cr
            if not cross.any():
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = np.maximum(trial, 0.0)
            trial = simplex_projection(trial)

            score = fast_score(trial, fft_len, n)
            if score < pop_score[i]:
                population[i] = trial
                pop_score[i] = score
                if score < best_score:
                    best_score = score
                    best_a = trial.copy()

        gen += 1

        # occasional diversity injection
        if gen % 15 == 0:
            worst = np.argsort(pop_score)[-max(1, pop_size // 10) :]
            for idx in worst:
                population[idx] = random_initial(n, rng)
                pop_score[idx] = fast_score(population[idx], fft_len, n)

        if gen % 30 == 0:
            print(f"[DE] generation {gen}, best_score = {best_score:.6f}")

    print(f"[DE] finished after {gen} generations, best_score = {best_score:.6f}")
    return best_a, best_score

    final_eval = evaluate_sequence(final_seq)  # provided by the judge

    return final_seq
```
