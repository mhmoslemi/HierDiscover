Propose a candidate sequence for the step function challenge.

Algorithm:
1. Generate a diverse pool of *structured* initial distributions (uniform,
   triangular, Gaussian, Hann, cosine, exponential, bimodal).  They are
   normalized to have unit L1 norm.

   candidates (including random restarts) until the bulk of the time budget
   is spent.  This directly minimises the true L∞ convolution objective.
4. Refine the result with a greedy hill‑climber that updates the convolution
   incrementally (O(n) per move) and repeatedly recomputes the exact convolution.
5. A final simulated‑annealing phase explores occasional uphill moves to
   escape local minima.
6. Return the best sequence found; it is evaluated with the official
   `evaluate_sequence` function before being returned.

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    elif mode == 'random':

    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# --------------------------------------------------------------
# Structured initial sequences
# --------------------------------------------------------------

def uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def triangular_seq(n: int) -> np.ndarray:
    # symmetric linear ramp up then down
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - centre) / (centre + 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def hann_window_seq(n: int) -> np.ndarray:
    i = np.arange(n)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a

def gaussian_seq(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def exponential_seq(n: int, scale: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(i - centre) / scale)
    a /= a.sum()
    return a

def power_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 1  # 1..n
    a = (i ** (-p))
    a = a[::-1]  # make it decreasing towards the centre for symmetry
    a /= a.sum()
    return a

def bimodal_gaussian_seq(n: int, sigma: float, offset: int) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - (centre - offset)) / sigma) ** 2)
    a += np.exp(-0.5 * ((i - (centre + offset)) / sigma) ** 2)
    a /= a.sum()
    return a

def generate_structured_candidates(n: int, rng: np.random.Generator):
    """Yield a list of candidate arrays built from simple parametric families."""
    candidates = []

    # Uniform and triangular are cheap baselines
    candidates.append(uniform_seq(n))
    candidates.append(triangular_seq(n))
    candidates.append(hann_window_seq(n))

    # Gaussian family; sample a few sigma values logarithmically spaced
    for sigma in [n * f for f in [0.05, 0.10, 0.15, 0.20, 0.30]]:
        candidates.append(gaussian_seq(n, sigma))

    # Exponential family
    for scale in [n * f for f in [0.05, 0.10, 0.20, 0.30, 0.50]]:
        candidates.append(exponential_seq(n, scale))

    # Power‑law family
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(power_seq(n, p))

    # Bimodal Gaussians (two symmetric hills)
    for sigma, off in [(n*0.08, int(0.15*n)), (n*0.12, int(0.20*n))]:
        candidates.append(bimodal_gaussian_seq(n, sigma, off))

    # Add a handful of random vectors for diversity
    for _ in range(10):
        candidates.append(random_initial(n, rng))

    return candidates

# --------------------------------------------------------------
# L‑BFGS on a smooth approximation of the max‑convolution
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# --------------------------------------------------------------
# Exponentiated‑gradient descent on the hard L∞ objective
# --------------------------------------------------------------

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the L∞ objective.
    Uses step size η = η₀ / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# --------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# --------------------------------------------------------------

def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly to avoid drift.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)

    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

# --------------------------------------------------------------
# Simulated annealing refinement (optional)
# --------------------------------------------------------------

def simulated_annealing(a: np.ndarray,
                       curr_score: float,
                       fft_len: int,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       init_temp: float = 1e-3,
                       cooling: float = 0.998,
                       max_fraction: float = 0.12,
                       recompute_every: int = 2000):
    """
    Simple annealing that accepts uphill moves with a temperature schedule.
    Works on top of the incremental convolution representation.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)

    temp = init_temp
    it = 0
    while time.time() < deadline and temp > 1e-9:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # define directed move
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp(-(new_score - best_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
        # cooling
        temp *= cooling
        it += 1
        if (it % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
    return best_a, best_score

"""
Propose a candidate sequence for the step‑function challenge.

Algorithm
---------
1. Build a modest library of *structured* probability vectors (uniform,
   triangular, Hann window, Gaussians, exponentials, power‑law, bimodal
   Gaussians, plus a few random vectors).  All are normalised to sum to 1.
2. Keep the best of these candidates according to the fast FFT‑based
   score 2·n·max(a*a).
3. Run a short L‑BFGS optimisation on a smooth log‑sum‑exp surrogate,
   starting from the current best and from random perturbations.
4. Perform many exponentiated‑gradient descents on the true L∞ objective,
   each time with a different start (the global best or another library
   entry).  This directly minimises 2·n·max(a*a).
5. Refine the result with a greedy hill‑climber that updates the
   convolution incrementally (O(n) per move) and periodically recomputes
   it exactly.
6. A final simulated‑annealing phase accepts occasional uphill moves,
   helping to escape shallow local minima.
7. Return the best sequence found before the 1000 s deadline; the final

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    cand.append(triangular_seq(n))
    cand.append(hann_window_seq(n))

    for sigma in [n * f for f in [0.05, 0.10, 0.15, 0.20, 0.30]]:
        cand.append(gaussian_seq(n, sigma))

    for scale in [n * f for f in [0.05, 0.10, 0.20, 0.30, 0.50]]:
        cand.append(exponential_seq(n, scale))

    for p in [0.5, 1.0, 1.5, 2.0]:
        cand.append(power_seq(n, p))

    for sigma, off in [(n * 0.08, int(0.15 * n)),
                       (n * 0.12, int(0.20 * n))]:
        cand.append(bimodal_gaussian_seq(n, sigma, off))

    for _ in range(10):
        cand.append(random_initial(n, rng))

    return cand


# ----------------------------------------------------------------------
# L‑BFGS on a smoothed max‑convolution surrogate
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """Smooth surrogate (log‑sum‑exp) of the max‑convolution."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                      # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS for each λ in lam_schedule, keeping the best intermediate."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """Exponentiated‑gradient descent directly on 2·n·max(a*a)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.14,
                           recompute_every: int = 1200):
    """Greedy local search moving a random fraction of mass between two points."""
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)        # exact full convolution
    diff = np.zeros_like(conv)

    it = 0
    accepted = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(a: np.ndarray,
                       curr_score: float,
                       fft_len: int,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       init_temp: float = 1e-3,
                       cooling: float = 0.999,
                       max_fraction: float = 0.14,
                       recompute_every: int = 2000):
    """Accept occasional uphill moves with a geometric temperature decay."""
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)

    temp = init_temp
    it = 0
    while time.time() < deadline and temp > 1e-9:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        accept = False
        if new_score < best_score:
            accept = True
        else:
            prob = np.exp(-(new_score - best_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            if new_score < best_score:
                best_score = new_score

        temp *= cooling
        it += 1
        if (it % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ---- warm start from previous run, if available ------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = uniform_seq(n)
    else:
        a0 = uniform_seq(n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – inspect a portfolio of structured candidates
    # ------------------------------------------------------------------
    candidates = generate_structured_candidates(n, rng)

    for cand in candidates:
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[candidate] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – brief L‑BFGS runs (explore basins)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 500
    phase1_deadline = start_time + 0.15 * budget_s

    while time.time() < phase1_deadline:
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_deadline, fft_len, n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many exponentiated‑gradient descents (restarts)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    while time.time() < phase2_deadline:
        if rng.random() < 0.4:
            a_start = best_a.copy()
        else:
            a_start = candidates[rng.integers(len(candidates))].copy()

        a_opt, sc_opt = exp_gradient_optimize(
            a_start, phase2_deadline, n, fft_len, rng, eta0=10.0
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climbing for fine tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.14,
            recompute_every=1200,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (optional escape from shallow minima)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            init_temp=1e-3,
            cooling=0.999,
            max_fraction=0.14,
            recompute_every=2000,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
