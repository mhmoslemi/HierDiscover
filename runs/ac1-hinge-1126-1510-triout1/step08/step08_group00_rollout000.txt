iteration counter=0

while time < deadline:

   lam_factor = lam_factors[min(iteration // steps_per_lambda, len(lam_factors)-1)]

   compute a = softmax(logits)

   compute convolution using FFT: conv = full_conv(a, fft_len) # length 2n-1.

   max_c = conv.max()

   # compute smooth max using lam = max_c * lam_factor + eps.

   lam = max_c * lam_factor + eps

   shifted = conv - max_c

   exp_shift = np.exp(shifted / lam)  # maybe large exponent causing overflow? Use np.exp with stable shift.

   Z = exp_shift.sum()

   w = exp_shift / Z

   # backprop: compute gradient of smooth max w.r.t a (g_vec)

   # We need similar steps: conv = a * a, derivative w.r.t a_i equals 2*(a_rev convolution w). Implementation earlier used FFT.

   # compute a_rev = a[::-1]; fm = fft(w); fa_rev = fft(a_rev); conv_rev = ifft(fa_rev * fm) -> real; then g_vec = conv_rev[n-1:n-1+n]  # same as earlier.

   a_rev = a[::-1]

   fm = np.fft.fft(w, fft_len)

   fa_rev = np.fft.fft(a_rev, fft_len)

   conv_rev = np.real(np.fft.ifft(fa_rev * fm))

   g_vec = conv_rev[n-1:n-1+n]

   logits = logits - step * m_hat

   # compute fast_score

"""
Propose a step‐function sequence with a low evaluation score.

The algorithm consists of several phases, each respecting a global
time budget (default 1000 s).  It starts from the best known construction
(if any) and then:

1. Performs a cheap parametric sweep over simple families (power, exp,
   Gaussian, cosine, bimodal, random) to obtain a good initial guess.

2. Runs many random‑restart L‑BFGS optimisations on a *smoothed* max
   objective.  The smoothing parameter (a factor of the current peak)
   is gradually decreased (0.8 → 0.0005) and each restart is allowed a
   generous number of inner iterations.

3. Refines the solution with an Adam optimiser on the same smoothed
   objective, using a decaying learning‑rate and the same λ‑schedule.

4. Conducts a projected sub‑gradient descent with momentum on the
   true (non‑smooth) max‑convolution.  The step size decays as
   1/√t and the mask that defines the sub‑gradient is tightened.

5. Finishes with a simulated‑annealing hill‑climb that moves a small
   fraction of mass between indices, accepting uphill moves with a
   schedule that cools linearly to zero.

During all phases the best feasible sequence (sum = 1, non‑negative)
and its fast_score = 2·n·max(a∗a) are tracked.  At the end the sequence
is evaluated with the provided `evaluate_sequence` function and
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a∗a) for a sequence that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a∗a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    # stabilize exp
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_conv)
    fa_rev = np.fft.fft(a_rev, fft_len_conv)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_factors,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam_factor in lam_factors, starting each
    stage from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam_f in lam_factors:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, lam_f),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print intermediate improvement
            print(f"[lbfgs] improved fast_score={best_score:.6f}")
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
    lam_factors,
    init_lr: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max objective.
    The smoothing factor cycles through lam_factors.
    """
    # initialise logits
    logits = np.log(a_start + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = softmax(logits)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # pick smoothing factor (cycle)
        lam_f = lam_factors[it % len(lam_factors)]

        a = softmax(logits)

        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv.max()
        lam = max_c * lam_f + 1e-12
        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        w = exp_shift / Z

        # back‑prop
        a_rev = a[::-1]
        fm = np.fft.fft(w, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_rev = np.real(np.fft.ifft(fa_rev * fm))
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]

        mean_g = np.dot(a, g_vec)
        grad = a * (g_vec - mean_g)   # gradient wrt logits

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = init_lr / (np.sqrt(v_hat) + eps)

        logits -= step * m_hat

        # evaluate current score
        a = softmax(logits)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")

        it += 1

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # tighter mask for the true sub‑gradient
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ∝ 1/√t
        step = 0.8 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[sub‑grad] improved fast_score={best_score:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a linear cooling schedule.
    Mass is transferred between two positions while preserving total sum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature as a small fraction of the current score
    T0 = 0.02 * start_score
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    it = 0
    while time.time() < deadline:
        # temperature proportional to remaining time
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue

        # move a random fraction (up to 12 %) of the mass at i to j
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] new best fast_score={best_score:.6f}")

        it += 1

    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:  # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families; returns the best vector.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    start = time.time()
    deadline = start + budget_s * 0.15  # use ~15 % of total budget

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    return best_a, best_score

            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed objective
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            n,
            lam_factors,
            init_lr=0.12,
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase3_end,
            fft_len_conv,
            n,
        )
        print(f"[phase3] sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (stochastic)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[phase4] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
