"""
Search for a low‑score non‑negative step function.

Algorithm (≈1000 s budget)
----------------------------------------------
1. **Smoothed L‑BFGS** – Optimises a log‑sum‑exp smooth approximation of the
   max‑convolution. A short but aggressive run provides a good starting basin.
2. **Exponentiated‑gradient descent** – Direct sub‑gradient descent on the true
   L∞‑objective using multiplicative updates on the probability simplex.
3. **Additive projected sub‑gradient descent** – Same sub‑gradient but with
   additive updates and Euclidean projection onto the simplex (often escapes
   plateaus that multiplicative updates miss).
4. **Greedy hill‑climb** – Moves a small fraction of mass from a source index
   to a destination index while updating the convolution incrementally.
   Helps polish the final solution.

All phases respect the overall time budget and keep the sequence non‑negative
and normalised (∑aᵢ = 1).  The best sequence found is returned as a list of
floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Phase 1 – smoothed L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax to stay on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smoothed max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Runs a sequence of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a (as a probability vector) and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – exponentiated‑gradient descent on the hard L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Multiplicative (exponentiated) gradient descent on 2·n·max(a*a).
    Uses step size η₀/√t and projects back to the simplex each iteration.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Phase 3 – additive projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 0.5):
    """
    Additive sub‑gradient descent on the exact max‑convolution.
    Uses a simple 1/√t step‑size schedule and projects onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # full convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution (same expression as in EG)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # additive update (move opposite to the sub‑gradient)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * g
        a = np.maximum(a, 0.0)
        a = simplex_projection(a)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[subgrad] iter={it} new best fast_score={best_score:.6f}")
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.10,
                           recompute_every: int = 2000):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly to avoid drift.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)

    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Move mass from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodically recompute from scratch to avoid numerical drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

    phase1_end = start_time + 0.15 * budget_s       # L‑BFGS
    phase2_end = start_time + 0.45 * budget_s       # Exp‑grad
    phase3_end = start_time + 0.80 * budget_s       # Additive sub‑grad
    # remaining time -> hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – short L‑BFGS runs with random restarts
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=10.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – additive projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=0.5,
        )
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy hill‑climb for final polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
