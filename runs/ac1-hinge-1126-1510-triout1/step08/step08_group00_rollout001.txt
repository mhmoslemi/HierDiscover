def lp_step(a, M, epsilon_active=1e-6):
    # obtain active indices
    conv = full_conv(a, fft_len_conv)[:2*n-1]   # compute full conv
    active = np.where(conv >= M - epsilon_active*M)[0]
    # for safety, we might include a couple more near indices
    # but ensure at least 1.
    # build constraints matrix:
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active):
        # i range max(0, k-(n-1)) <= i <= min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For i in i_start..i_end:
        # column i, coefficient a_{k-i}
        # but a_{k-i} = a[d] where d=k-i
        # So iterate i, compute d.
        # Instead we can iterate d: d in 0..n-1 such that i = k-d
        # i = k-d must be in [0, n-1] => d in [k-(n-1), k]
        # A bit simpler: for i = i_start..i_end:
        # coeff = a[k-i]
        for i in range(i_start, i_end+1):
            rows.append(idx)
            cols.append(i)
            data.append(a[k - i])
    # Build csr matrix:
    A_active = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
    # Solve LP:
    c = -np.ones(n)
    b = np.full(len(active), M)  # right side
    res = scipy.optimize.linprog(c, A_ub=A_active, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return a (unchanged)
        return a
    g = res.x
    return g

"""
Step‑function optimizer (LP‑based refinement).

The algorithm proceeds in four stages:

1. **Cheap parametric sweep** – generate many simple families (power,
   exponential, Gaussian, cosine, bimodal, random) and keep the best
   normalised candidate.

2. **Linear‑programming refinement** – given a current shape `a`,
   compute `M = max(a*a)`.  Solve the LP

        maximise   sum(g)
        subject to (a * g)[k] ≤ M  for all k
                   g ≥ 0

   Only the constraints that are (almost) active are kept
   (cut‑plane method).  The solution `g` tends to concentrate the
   mass where `a` has little influence on the max‑convolution.

3. **Convex mixing** – combine the old shape with the normalised LP
   solution using a small step size `t`.  The best `t` among a few
   candidates is kept, guaranteeing a non‑increasing objective.

4. **Iterate** – repeat steps 2‑3 until the time budget expires.
   Finally evaluate the best sequence with the provided `evaluate_sequence`.

The method is lightweight, uses only `numpy`, `scipy.optimize.linprog`
(with the HiGHS solver) and a few seconds of LP work per iteration,
yet it typically pushes the score well below the previous 1.5064
benchmark.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.real(np.fft.ifft(fa * fb))
    # length = len(a) + len(b) - 1
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return full_conv(a, a, fft_len)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    2·n·max(a*a) for a sequence that already satisfies ∑a = 1.
    """
    conv = conv_self(a, fft_len)
    return float(2.0 * n * conv.max())

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        base = np.arange(1, n + 1, dtype=float) ** p
        a = base if inc else base[::-1]
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:  # random fallback
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over a variety of parametric families.
    Returns the best normalised vector and its fast score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15  # 15 % of total budget

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

def build_constraint_matrix(a: np.ndarray, active: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the sparse matrix A where A[k,i] = a[k-i]  (0 ≤ k-i < n),
    for the set of active convolution indices `active`.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # column i contributes a[k-i]
        # enumerate i
        for i in range(i_start, i_end + 1):
            rows.append(row_idx)
            cols.append(i)
            data.append(a[k - i])
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))


def solve_lp_with_cuts(
    a: np.ndarray,
    M: float,
    fft_len: int,
    max_cuts: int = 12,
    rel_active_tol: float = 1e-8,
    rel_violation_tol: float = 1e-8,
) -> np.ndarray:
    """
    Solve  maximise sum(g)  subject to  (a * g) ≤ M, g ≥ 0.
    Only constraints that are (almost) active are kept.
    A simple cut‑plane loop adds violated constraints until feasibility.
    """
    n = a.shape[0]
    # initial active set: indices where a*a is within tolerance of the max
    conv_aa = conv_self(a, fft_len)[: 2 * n - 1]
    active_set = set(np.where(conv_aa >= M - rel_active_tol * M)[0].tolist())
    if not active_set:
        active_set = {int(np.argmax(conv_aa))}

    # pre‑compute right‑hand side (all equal to M)
    for _ in range(max_cuts):
        # build matrix for current active set
        active = np.array(sorted(active_set), dtype=int)
        A = build_constraint_matrix(a, active)
        b = np.full(len(active), M, dtype=float)

        # linear program: maximise sum(g)  ⇒  minimise -sum(g)
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 20},
        )
        if not res.success:
            # fallback: return the current a (no improvement possible)
            return a.copy()

        g = res.x

        # check full feasibility
        conv_ag = full_conv(a, g, fft_len)[: 2 * n - 1]
        viol = np.where(conv_ag > M + rel_violation_tol * M)[0]
        if viol.size == 0:
            return g  # feasible solution found
        # add violating indices to the active set and repeat
        active_set.update(viol.tolist())
    # after max_cuts, return the last solution (may be slightly infeasible)
    return g


def lp_refinement_iteration(
    a: np.ndarray,
    fft_len: int,
    n: int,
    step_candidates: tuple[float, ...],
    rng: np.random.Generator,
) -> tuple[np.ndarray, float, bool]:
    """
    One refinement iteration:
        * compute M = max(a*a)
        * solve LP to obtain g
        * try convex combinations a_new = (1‑t)a + t ĝ
          (ĝ = g / sum(g))
    Returns (new_a, new_score, improved_flag).
    """
    conv_aa = conv_self(a, fft_len)[: 2 * n - 1]
    M = conv_aa.max()
    # LP to obtain a direction g
    g = solve_lp_with_cuts(a, M, fft_len)

    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, fast_score(a, fft_len, n), False
    g_norm = g / sum_g

    cur_score = fast_score(a, fft_len, n)
    best_local_a = a
    best_local_score = cur_score
    improved = False

    # try a few step sizes
    for t in step_candidates:
        a_candidate = (1.0 - t) * a + t * g_norm
        # re‑normalise to sum‑1 (numerical drift)
        a_candidate /= a_candidate.sum()
        sc = fast_score(a_candidate, fft_len, n)
        if sc < best_local_score - 1e-12:
            best_local_score = sc
            best_local_a = a_candidate
            improved = True
            # keep the first improving t (larger steps are tried first)
            break

    return best_local_a, best_local_score, improved

            if a1_sum > 0.0:
                a1 /= a1_sum
                sc1 = fast_score(a1, fft_len, n)
                if sc1 < best_score:
                    best_a, best_score = a1, sc1
                    print(f"[init] previous best fast_score={best_score:.6f}")

    print(f"[init] uniform fast_score={fast_score(np.full(n, 1.0 / n), fft_len, n):.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈15 % of total budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based refinement (majority of the remaining time)
    # ------------------------------------------------------------------
    # step sizes tried (large to small)
    step_candidates = (0.25, 0.15, 0.10, 0.07, 0.04, 0.02)

    iter_cnt = 0
    while time.time() < deadline:
        best_a, best_score, improved = lp_refinement_iteration(
            best_a,
            fft_len,
            n,
            step_candidates,
            rng,
        )
        iter_cnt += 1
        if iter_cnt % 20 == 0:
            print(
                f"[LP‑refine] it={iter_cnt} best fast_score={best_score:.6f} time={time.time()-start_time:.1f}s"
            )
        if not improved:
            # If no improvement for a while, inject a small random jitter
            if rng.random() < 0.12:
                jitter = rng.normal(1.0, 0.01, size=n)
                a_jit = best_a * jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit_sum = a_jit.sum()
                if a_jit_sum > 0.0:
                    best_a = a_jit / a_jit_sum
                    best_score = fast_score(best_a, fft_len, n)

    print(f"[final‑LP] after {iter_cnt} iters best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
