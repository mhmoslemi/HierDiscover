def hill_climb(a, current_score, n, fft_len_conv, deadline, max_iter=200000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute grad mask (with subgradient)
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(np.float64)
        # compute grad
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_conv) # maybe need separate len
        f_rev = np.fft.fft(rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(f_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
        # source high grad
        high_idx = np.where(grad > np.percentile(grad, 90))[0]
        low_idx = np.where(grad < np.percentile(grad, 10))[0]
        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        if i == j:
            it += 1
            continue
        # choose fraction
        # compute max delta limited by a[i] (can't go negative)
        max_delta = best[i]
        # try several fractions
        for frac in (0.005, 0.01, 0.02, 0.05, 0.1):
            delta = max_delta * frac
            if delta <= 0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            # re-normalize? sum already 1 because we transferred mass, no need.
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score - 1e-8:
                best = cand
                best_score = new_score
                break
        it += 1
    return best, best_score

"""
Propose a step‑function sequence that minimizes the evaluation function.

The optimizer works on a probability vector (sum = 1) of length *n*.
It mixes three simple but complementary ideas:

1. **Random restarts + Projected Adam** – a sub‑gradient method that
   directly uses the true max‑convolution.  Each restart runs for a
   short, bounded time so that many independent runs explore the
   landscape.

2. **Aggressive hill‑climb** – after Adam, a greedy move transfers a
   small amount of mass from indices with a large sub‑gradient to
   indices with a small sub‑gradient.  Several transfer fractions are
   tried; only strictly improving moves are kept.  The procedure runs
   until the global deadline is reached.

3. **Optional fine L‑BFGS** – at the very end a few L‑BFGS steps on a
   smooth log‑sum‑exp surrogate polish the best candidate.

All operations use FFT‑based convolutions, run on the simplex
(\sum a = 1, a ≥ 0) and respect the 1000 s budget.  The final list of
floats is returned ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective when a is normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sine
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # In practice L‑BFGS rarely raises, but we guard against it.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the true (non‑smoothed) objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moment estimation) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max (via convolution with the mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def targeted_hill_climb_aggressive(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 100000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that uses the sub‑gradient to move mass from
    high‑gradient indices to low‑gradient ones.  Several candidate
    transfer fractions are tried and only strictly improving moves are
    kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution & active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fr = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fr * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # pick source (high grad) and target (low grad)
        high_thr = np.percentile(g, 90)
        low_thr = np.percentile(g, 10)
        high_idx = np.where(g >= high_thr)[0]
        low_idx = np.where(g <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # try a few transfer fractions, keep the first strictly improving one
        max_delta = best[i]
        improved = False
        for frac in (0.005, 0.01, 0.02, 0.05, 0.10):
            delta = max_delta * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best = cand
                best_score = new_score
                improved = True
                break
        # if none improved we simply move on
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase search for a low‑value step‑function.
    1. Many short Adam runs from random restarts.
    2. Aggressive hill‑climb on the current best.
    3. Optional fine L‑BFGS polishing (tiny λ schedule).
    The whole routine respects the time budget and returns a list of
    non‑negative floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a richer search space
    # ------------------------------------------------------------------
    n = 7000  # can be increased if time permits
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial solution (global variable fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:  # keep a few seconds for polishing
        # random start, optionally blended with current best
        a_start = random_initial(n, rng)
        if rng.random() < 0.3:
            w = rng.uniform(0.2, 0.8)
            a_start = w * best_a + (1 - w) * a_start
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()

        # give each Adam run at most 30 s (or the remaining time)
        sub_deadline = min(deadline - 0.5, time.time() + 30.0)
        a_opt, score_opt = subgradient_adam(
            a_start, sub_deadline, fft_len_conv, fft_len_g, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive hill‑climb on the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, score_hc = targeted_hill_climb_aggressive(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=200000,
        )
        if score_hc < best_score:
            best_score = score_hc
            best_a = a_hc.copy()
            print(f"[Hill‑climb] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short fine‑tuned L‑BFGS (tiny smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06]
        fine_maxiter = 300
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
