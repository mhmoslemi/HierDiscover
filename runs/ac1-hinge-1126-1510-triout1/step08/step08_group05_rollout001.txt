def projected_adam(a_start, deadline, fft_len, fft_len_g, n):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        grad = subgradient_maxconv(a, fft_len, fft_len_g)
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)

        step = 0.5/np.sqrt(it)  # can tune
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a)
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Multi‑phase optimizer for the step‑function bound.

The search proceeds in four stages inside the 1000 s budget:

1. **Broad random sampling** – generate a large variety of candidate
   distributions (uniform, exponential, power‑law, edge‑heavy, bimodal,
   etc.) and keep the best one.

2. **Projected Adam** – a sub‑gradient method that works directly on the
   true non‑smooth objective.  The gradient of the max‑convolution is
   obtained by an FFT‑based back‑propagation, followed by Euclidean projection
   onto the probability simplex (∑a=1, a≥0).

3. **Gradient‑guided hill‑climb** – using the same sub‑gradient we move mass
   from “source’’ indices (large gradient) to “target’’ indices (small
   gradient).  Several transfer fractions are tried; only strictly improving
   moves are kept.

4. **Simulated annealing** – a simple Metropolis walk that transfers a
   random fraction of mass between two positions.  The temperature is
   linearly cooled until the deadline, allowing occasional uphill jumps to
   escape plateaus.

All phases respect the global time budget; the best sequence discovered
throughout the run is finally evaluated with ``evaluate_sequence`` and
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised (∑a = 1).
    Returns 2·n·max(a*a) .
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def subgradient_maxconv(a: np.ndarray, fft_len_conv: int, fft_len_g: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  with respect to a.
    Returns a vector g of length n.
    """
    n = a.shape[0]
    # full linear convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_g = np.fft.irfft(fa_rev * fm, n=fft_len_g)
    grad = 2.0 * conv_g[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def projected_adam(a0: np.ndarray, deadline: float,
                   fft_len_conv: int, fft_len_g: int,
                   n: int) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the exact sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad = subgradient_maxconv(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)          # diminishing step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def targeted_hill_climb_aggressive(a: np.ndarray,
                                   current_score: float,
                                   fft_len_conv: int,
                                   fft_len_g: int,
                                   n: int,
                                   deadline: float,
                                   max_iter: int = 50000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from high‑gradient indices to low‑gradient
    ones.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # sub‑gradient of the max‑convolution
        grad = subgradient_maxconv(best, fft_len_conv, fft_len_g)

        # extreme percentiles for source / target
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # try a few transfer fractions
        improved = False
        for frac in (0.02, 0.07, 0.15):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True
                break
        # occasional random jitter to get unstuck
        if not improved and it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            cand = best + rng.normal(0.0, sigma, size=n)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best = cand
                best_score = cand_score

        it += 1
    return best, best_score


def simulated_annealing(a0: np.ndarray,
                        start_score: float,
                        deadline: float,
                        fft_len: int,
                        n: int) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that transfers a random fraction of mass between
    two positions.  Temperature is linearly cooled.
    """
    cur = a0.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # Initial temperature – a fraction of the current score
    T_start = max(cur_score * 0.1, 1e-4)
    T_end = 1e-12
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        T = T_start * (1.0 - progress) + T_end * progress

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 1e-14:
            continue

        # random transfer, up to 20 % of the source mass
        frac = rng.uniform(0.001, 0.2)
        delta = cur[i] * frac

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = fast_score(cand, fft_len, n)

        d = cand_score - cur_score
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        it += 1
        if it % 20000 == 0:
            print(f"[SA] it={it}, best_score={best_score:.6f}")

        # periodic renormalisation to avoid drift
        if it % 5000 == 0:
            cur /= cur.sum()
            best /= best.sum()

    return best, best_score
