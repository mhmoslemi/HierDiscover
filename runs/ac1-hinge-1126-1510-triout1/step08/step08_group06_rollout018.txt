def hill_climb_optimal(a, current_score, fft_len, n, deadline):
    best = a.copy()
    best_score = current_score
    while time.time() < deadline:
        # compute gradient
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break
        # attempt to move all mass from hi to lo
        delta_max = best[i_hi]
        # if delta_max is zero, break
        if delta_max <= 0:
            break
        # binary search for best delta between 0 and delta_max
        lo = 0.0
        hi = delta_max
        # quick heuristic: try full move first
        cand = best.copy()
        cand[i_hi] = 0.0
        cand[i_lo] += delta_max
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
            continue
        # else try halving steps
        # iterate some steps
        for _ in range(8):
            mid = (lo + hi) / 2
            if mid < 1e-12:
                break
            cand = best.copy()
            cand[i_hi] -= mid
            cand[i_lo] += mid
            new_score = fast_score(cand, fft_len, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
                # try moving more? continue searching finer direction
                lo = mid
            else:
                hi = mid
        # if no improvement break
        # continue loop
    return best, best_score

"""
A multi‑phase search for a low‑value step sequence.

The algorithm works on the simplex {a_i ≥ 0, Σa_i = 1} because the
objective 2·n·max(conv(a,a)) / (Σa_i)^2 is invariant to scaling.
The pipeline is:

1.  A cheap sweep over several parametric families (power‑law,
    exponential, Gaussian, beta‑like, sinusoidal, …) to obtain a few
    good seeds.
2.  Random‑restart L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The best few seeds are refined.
3.  Simulated‑annealing style random perturbations (Gaussian noise
    + simplex projection) that accept only improvements.  This
    explores the neighbourhood of the current best.
4.  A deterministic hill‑climb that repeatedly moves a fraction of
    mass from the coordinate with the largest sub‑gradient (most
    harmful) to the one with the smallest sub‑gradient (most helpful).
    The move size is adaptively reduced by binary search.
5.  Final evaluation with the protected `evaluate_sequence` function.

All phases respect a global time budget (default 1000 s).  The
best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence already on the simplex.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
    rng: np.random.Generator,
    sigma_start: float = 0.03,
    sigma_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbation + simplex projection.
    Only improving moves are kept.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    cur_score = best_score

    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        cand = a + rng.normal(0.0, sigma, size=n)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
    return best, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves mass from the coordinate with the
    largest sub‑gradient to the one with the smallest, using a binary
    search on the amount of transferred mass.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        delta_max = best[i_hi]
        if delta_max <= 0.0:
            break

        # first try moving the whole mass
        cand = best.copy()
        cand[i_hi] = 0.0
        cand[i_lo] += delta_max
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
            it += 1
            continue

        # binary search for a useful fraction (max 8 refinement steps)
        lo = 0.0
        hi = delta_max
        for _ in range(8):
            mid = (lo + hi) / 2.0
            if mid < 1e-15:
                break
            cand = best.copy()
            cand[i_hi] -= mid
            cand[i_lo] += mid
            new_score = fast_score(cand, fft_len, n)
            if new_score < best_score:
                # accept this fraction and try moving more
                best = cand
                best_score = new_score
                lo = mid
            else:
                hi = mid
        it += 1
    return best, best_score

def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = np.arange(1, n + 1, dtype=float) ** (-p)
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

            a0 = np.maximum(a0, 0.0)

    # ------------------------------------------------------------------
    # Allocate time slices for the remaining phases

    phase1_end = start_time + 0.40 * budget_s   # L‑BFGS

    # Phase 1 – L‑BFGS refinement from top seeds
    # ------------------------------------------------------------------

        0.18,
        0.12,

        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,

    maxiter_per_lam = 1200

    for seed_a in top_seeds:
        if time.time() >= phase1_end:
            break
        a_opt, sc_opt = lbfgs_continuation(
            seed_a,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:

    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (random perturbations)
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(
            best_a,
            phase4_end,
            fft_len_conv,
            n,
            rng,
            sigma_start=0.03,
            sigma_end=0.001,
        )
        print(f"[SA] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        deadline,
        max_iter=40000,
    )
    print(f"[hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the protected `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
