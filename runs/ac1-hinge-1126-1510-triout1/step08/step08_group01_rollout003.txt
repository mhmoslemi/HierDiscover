    deadline = start_time + budget_s - 0.05

        x_old = np.linspace(0, 1, old.size)
        x_new = np.linspace(0, 1, n)
        f = np.interp(x_new, x_old, old)
        f = np.maximum(f, 0.0)
        sum_f = f.sum()
        if sum_f > 0:

        else:

"""
Search for a non‑negative step sequence minimizing

    2·n·max(convolution) / (sum a)²

The algorithm works in four phases inside the 1000 s budget:

1️⃣  *Bootstrap* – generate a few parametric shapes (triangular, cosine,
    exponential, power‑law) and keep the best one.

2️⃣  *L‑BFGS* – optimise a log‑sum‑exp smooth approximation of the
    max‑convolution.  Several random restarts and a schedule of smoothing
    parameters are used.

3️⃣  *Alternating LP* – given the current f, solve a tiny linear programme
    (active‑set of the largest convolution entries) for a direction g with
    f∗g ≤ M (M = max f∗f).  Mix f←(1‑t)f+t g with a line search on t;
    repeat while the maximum convolution decreases.

4️⃣  *Hill‑climb* – greedy incremental moves of mass between indices,
    updating the convolution in O(n) per move.

The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a).  Assumes sum(a)=1, which makes it equal to evaluate_sequence."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS on a smoothed max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → probability vector a
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.10,
                           recompute_every: int = 2000):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly to avoid drift.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

def build_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix B (|active| × n) where  B[k, j] = f[active_idx[k] - j]
    for all j such that the index is inside [0, n‑1].  This encodes
    (f * g)[k] = Σ_j B[k, j] g[j].
    """
    n = f.shape[0]
    m = active_idx.shape[0]
    B = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # vector of j values
        j = np.arange(j_start, j_end + 1, dtype=int)
        B[row, j] = f[k - j]
    return B

        bounds = [(0, None)] * n
        res = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If LP fails, jitter a bit and continue
            best_a = add_noise(best_a, sigma=0.001, rng=rng)
            continue

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            best_a = add_noise(best_a, sigma=0.001, rng=rng)
            continue

        # Normalise g to have unit sum (same scaling as best_a)
        g = g0 / sum_g0

        # Try several mixing coefficients, accept the first that reduces M
        improved = False
        for t in [0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            cand = (1 - t) * best_a + t * g
            # keep the sum exactly 1 (numerical drift)
            cand /= cand.sum()
            new_M = full_conv(cand, fft_len).max()
            if new_M < M - 1e-12:   # a genuine improvement
                best_a = cand
                best_score = 2.0 * n * new_M
                best_score = float(best_score)
                improved = True
                break

        if improved:
            print(f"[LP] iter {iteration}, new best score={best_score:.6f}")
        else:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.15:
                best_a = add_noise(best_a, sigma=0.003, rng=rng)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            fast_score(best_a, fft_len),
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
