"""
Search for a non‑negative step‑function sequence minimizing
evaluate_sequence.  Uses a continuation of smoothed L‑BFGS followed
by a projected sub‑gradient descent on the exact max‑convolution and
a final greedy hill‑climb.  Restarts are performed until the time
budget expires.
"""
import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n-1) using FFT of fixed length."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
):
    """Smooth‑max objective and its gradient w.r.t. the unconstrained vector y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z  # derivative wrt each conv entry

    # convolution a * w (only the part that contributes to gradient)
    # (a*g)[i] = sum_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_schedule,
                      maxiter_per_stage: int = 150):
    """Run a series of L‑BFGS‑B optimisations with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def subgradient_descent(a: np.ndarray, max_iters: int, deadline: float):
    """Projected sub‑gradient descent on max(a*a)."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    cur_a = a.copy()
    # initial max
    fa = np.fft.fft(cur_a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (cur_a.sum() ** 2)

    rng = np.random.default_rng()
    eta = 0.5

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # recompute conv and max index
        fa = np.fft.fft(cur_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        # try a back‑tracking step
        improved = False
        for _ in range(6):
            cand = project_to_simplex(cur_a - eta * g)
            fa_c = np.fft.fft(cand, fft_len)
            conv_c = np.real(np.fft.ifft(fa_c * fa_c))[: 2 * n - 1]
            max_c = conv_c.max()
            if max_c < max_val - 1e-15:
                cur_a = cand
                cur_max = max_c
                cur_score = 2.0 * n * cur_max / (cur_a.sum() ** 2)
                eta *= 1.1   # slightly increase step size after success
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # if we cannot improve with the current sub‑gradient,
            # perform a tiny random mass‑transfer to escape flat regions.
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = cur_a[i] * 0.02
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = project_to_simplex(cur_a)  # keep on simplex
            eta = max(eta, 0.01)
    return cur_a, cur_score

def gauss_hill_climb(a: np.ndarray, cur_score: float,
                     fft_len: int, deadline: float,
                     max_iter: int = 3000):
    """Greedy random mass‑transfer hill‑climb (similar to the previous code)."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Search for a non‑negative step‑function sequence that minimises
`evaluate_sequence`.  The algorithm combines three complementary phases:

1. **Continuation L‑BFGS** on a soft‑max parametrisation with a smooth
   approximation of the max‑convolution.  The smoothing parameter is
   gradually reduced (lam‑schedule).

2. **Projected sub‑gradient descent** that directly reduces the exact
   maximal entry of the auto‑convolution while staying on the probability
   simplex (sum = 1, a_i ≥ 0).

3. **Greedy hill‑climb** (random mass transfers) for final polishing.

Multiple random restarts are performed until the allocated time budget
expires.  The best sequence found is finally evaluated with the official
`evaluate_sequence` function and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of `a` with itself using a fixed‑size FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    """
    conv = conv_fft(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
):
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. the
    unconstrained vector `y`.  `lam_factor` is the relative smoothing width.
    """
    a = softmax(y)

    # ---- convolution a*a (FFT) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ---- smooth maximum of the convolution ----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # ∂ smooth_max / ∂ conv_i

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule,
    maxiter_per_stage: int = 150,
):
    """
    Run a continuation of L‑BFGS‑B optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_descent(a: np.ndarray, max_iters: int, deadline: float):
    """
    Projected sub‑gradient descent that directly reduces the maximal
    entry of the auto‑convolution of `a`.  The vector stays on the
    probability simplex throughout the run.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    cur_a = a.copy()
    rng = np.random.default_rng()
    eta = 0.5

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        # current max entry and its index
        fa = np.fft.fft(cur_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # sub‑gradient of the selected max entry
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        # back‑tracking line search
        improved = False
        for _ in range(6):
            cand = project_to_simplex(cur_a - eta * g)
            fa_c = np.fft.fft(cand, fft_len)
            conv_c = np.real(np.fft.ifft(fa_c * fa_c))[: 2 * n - 1]
            max_c = conv_c.max()
            if max_c < max_val - 1e-15:
                cur_a = cand
                eta *= 1.1
                improved = True
                break
            else:
                eta *= 0.5

        if not improved:
            # tiny random perturbation to escape flat spots
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = cur_a[i] * 0.02
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = project_to_simplex(cur_a)
            eta = max(eta, 0.01)

    # final score (fast version, sum = 1)
    final_score = fast_score(cur_a, fft_len)
    return cur_a, final_score


def gauss_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
):
    """
    Simple random mass‑transfer hill‑climb that always respects the simplex.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score
