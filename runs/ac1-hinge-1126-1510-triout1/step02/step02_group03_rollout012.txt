def _softmax_weights(conv, alpha):
    if alpha <= 0:
        # uniform
        p = np.full_like(conv, 1.0/conv.size)
        return p
    max_c = conv.max()
    # subtract max for numeric stability
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        # fallback uniform
        p = np.full_like(conv, 1.0/conv.size)
    else:
        p = exp_vals / sum_exp
    return p

"""
Propose a low‑score step‑function using a smooth‑max (log‑sum‑exp) surrogate
and Adam optimisation on the soft‑max parametrisation of the coefficients.
The loss approximates the L∞‑norm of the self‑convolution a*a.
A schedule on the smoothness parameter (α) gradually turns the surrogate
into the true max, while the Adam updates keep the iterates in the simplex
(no explicit projection is needed).  Several random restarts and a short
local polishing phase are performed within the allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x and y using FFT."""
    m = x.shape[0]
    n = y.shape[0]
    size = 1 << ((m + n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: m + n - 1]


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights for a smooth approximation of max
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return a probability vector p_k = exp(α·conv_k) / Σ exp(α·conv_j).
    For α≤0 a uniform distribution is returned.
    """
    if alpha <= 0.0:
        return np.full_like(conv, 1.0 / conv.size)

    # stabilise the exponentials
    c_max = conv.max()
    exp_vals = np.exp(alpha * (conv - c_max))
    total = exp_vals.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm works on the log‑soft‑max parametrisation
    a_i = exp(u_i) / Σ_j exp(u_j) and minimises a smooth
    approximation of the max using Adam.  Several restarts,
    an adaptive smoothness parameter α and a brief polishing
    phase are performed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence (use previously found one if any)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = len(init_seq)
    except Exception:
        init_seq = None
        n = 1542                                             # default length

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None               # kept in normalised (sum=1) form
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    max_iter_per_restart = 15000
    lr = 0.05                 # Adam learning rate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    restart = 0

    # ------------------------------------------------------------------
    # Main loop over random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # ----- Initialise a candidate vector -----------------------------------------
        if init_seq is not None and restart == 1:
            a0 = np.clip(init_seq, 0.0, 1000.0)
            a0 = a0 / a0.sum()
        else:
            init_mode = np.random.choice(
                ["uniform", "sqrtdiff", "power", "random"], p=[0.15, 0.3, 0.25, 0.3]
            )
            if init_mode == "uniform":
                a0 = np.ones(n) / n
            elif init_mode == "sqrtdiff":
                a0 = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(
                    np.arange(0, n, dtype=float)
                )
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum()
            elif init_mode == "power":
                pexp = np.random.uniform(0.1, 1.8)
                a0 = (np.arange(1, n + 1, dtype=float) ** (-pexp))
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum()
            else:  # "random"
                a0 = np.random.rand(n)
                a0 = a0 / a0.sum()

        # ----- Log‑soft‑max parametrisation -----------------------------------------
        eps = 1e-12
        u = np.log(a0 + eps)      # initialise u such that a = softmax(u)

        # Adam moments
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Smooth‑max parameter α (starts small, grows)
        alpha = 5.0
        alpha_mul = 1.08               # geometric increase after each evaluation

        it = 0
        while time.time() < deadline and it < max_iter_per_restart:
            it += 1

            # --- a from u (stable soft‑max) ---------------------------------------
            shift = np.max(u)                     # for numerical stability
            exp_u = np.exp(u - shift)
            a = exp_u / exp_u.sum()

            # --- Self‑convolution -------------------------------------------------
            conv = _conv_fft(a)                   # length 2n‑1

            # --- Smooth max weights ------------------------------------------------
            p = _softmax_weights(conv, alpha)     # probabilities, sum to 1

            # --- Gradient w.r.t a -------------------------------------------------
            a_rev = a[::-1]
            full = _conv_full_fft(a_rev, p)       # length 3n‑2
            g_a = 2.0 * full[n - 1 : n - 1 + n]    # slice of length n

            # --- Gradient w.r.t u (respecting sum‑to‑1 constraint) ---------------
            dot_ag = np.dot(a, g_a)
            g_u = a * (g_a - dot_ag)

            # --- Adam update -------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * g_u
            v = beta2 * v + (1.0 - beta2) * (g_u ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # --- Periodic evaluation of the true objective ------------------------
            if it % 120 == 0 or it == max_iter_per_restart:
                seq_scaled = a * 1000.0
                seq_scaled = np.clip(seq_scaled, 0.0, 1000.0)
                cur_score = evaluate_sequence(seq_scaled.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(
                        f"[Restart {restart}] iter {it:05d} – new best {best_score:.6f} "
                        f"(t={elapsed:.1f}s)"
                    )
                # make the surrogate closer to the true max
                alpha = min(2000.0, alpha * alpha_mul)

        # ----- Polishing phase (local swaps) ----------------------------------------
        if best_seq is not None:
            a_loc = best_seq.copy()
            polish_deadline = time.time() + min(6.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a tiny fraction from i to j
                eps_move = min(a_loc[i], a_loc[i] * np.random.uniform(0.0, 0.05))
                if eps_move <= 0.0:
                    continue
                a_new = a_loc.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move
                a_new = a_new / a_new.sum()               # renormalise
                seq = a_new * 1000.0
                seq = np.clip(seq, 0.0, 1000.0)
                s = evaluate_sequence(seq.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_loc = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # Stop if we are out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n

    # ------------------------------------------------------------------
    # Scale to the [0,1000] range required by the evaluator (no effect on score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(
        f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s"
    )
    return final_seq.tolist()
