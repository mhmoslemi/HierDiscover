"""
Propose a very low‑bound step‑function analytically.

The objective 2·n·max(conv(a,a)) / (sum(a)**2) is homogeneous:
if Σa = √(2 n) the denominator cancels and the score equals simply
max(conv(a,a)).  For a non‑negative vector on the simplex a (Σa=1)
the convolution value at the central index is

    C(p) = Σ_i a_i·a_{n‑1‑i}
          = (p+1)²·B(p+1,p+1) / n   (as n → ∞),

where a_i ∝ (i+1)ᵖ and B is the Beta‑function.
The corresponding score is

    score(p) = 2·n·C(p) ≃ 2·(p+1)²·B(p+1,p+1),

which is *decreasing* with p.  In particular:
    p = 1  →  score ≈ 4/3   (≈ 1.333)
    p = 2  →  score ≈ 0.6
    p = 3  →  score ≈ 0.23
so any p ≥ 1 already beats the target 1.503.

The routine below builds a family of power‑law sequences
a_i ∝ (i+1)ᵖ, rescales them to Σa = √(2 n) and returns the
one with the smallest evaluated score (using the supplied
`evaluate_sequence`).  The search is extremely cheap (a few
candidates) and finishes well within the 1000 s budget.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative step heights that (hopefully) yields a
    score below the target 1.503.

    The method:
      * Choose a length n (preferably the length of the previous best
        construction if it exists, otherwise 5000).
      * Try a few power‑law exponents p ∈ {0.5, 1, 1.5, 2, 2.5, 3, 4}.
      * For each p build a vector w_i = (i+1)ᵖ, normalise it to sum = 1,
        then scale by √(2 n) (the normalisation used by the reference
        code).
      * Evaluate the sequence with `evaluate_sequence`; keep the best.
    The returned list is plain Python floats, already clipped to the
    range accepted by `evaluate_sequence`.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # -------------------------------------------------------------
    # Determine a reasonable length.
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 5000                     # fallback length, fast to evaluate

    target_sum = math.sqrt(2.0 * n)   # the required Σa

    print(f"--- Building candidate of length n={n} (target sum={target_sum:.3f}) ---")

    # -------------------------------------------------------------
    # Helper: build a power‑law sequence for a given exponent.
    def build_sequence(p: float) -> np.ndarray:
        w = np.arange(1, n + 1, dtype=np.float64) ** p   # (i+1)^p
        a = w / w.sum()                                 # Σa = 1
        return a * target_sum                           # Σa = √(2 n)

    # -------------------------------------------------------------
    # Search over a handful of exponents.
    best_seq = None
    best_score = float("inf")
    # Exponents that give progressively lower theoretical bounds.
    exponents = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0]

    for p in exponents:
        if time.time() > deadline:
            break

        seq = build_sequence(p)
        # Clip to the safe range enforced by evaluate_sequence.
        seq = np.clip(seq, 0.0, 1000.0)

        # The evaluation is cheap (single convolution) and gives the true
        # objective value; we keep the best.
        score = evaluate_sequence(seq.tolist())
        print(f"p={p:4.1f} → score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # -------------------------------------------------------------
    # Fallback – uniform distribution – should never be needed.
    if best_seq is None:
        print("Fallback to uniform distribution.")
        best_seq = np.full(n, target_sum / n)

    print(f"--- Final best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
