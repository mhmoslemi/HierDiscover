def adam_optimize(a0, max_iters, n, deadline, lr=0.1, alpha_start=0.5, alpha_end=50.0, beta1=0.9, beta2=0.999, eps=1e-8):
   a = a0.copy()
   # ensure sum=1 via project simplex
   a = project_simplex(a)
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   best_seq = a.copy()
   best_score = float('inf')
   for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = compute_full_conv(a)
        max_conv = conv.max()
        score = 2 * n * max_conv
        if score < best_score:
             best_score = score
             best_seq = a.copy()
             print(f"[iter {it}] new best score={score:.6f}")
        # compute alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # compute weights
        shift = conv - max_conv
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()
        w = w_raw / w_sum
        # compute gradient (2 * conv_of(w, a_rev))

        grad = compute_weighted_grad(w, a_rev)  # returns vector length n
        # Adam update

        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # Project onto simplex
        a = project_simplex(a)
   return best_seq, best_score

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_size(min_len: int) -> int:
    """Return the next power‑of‑two ≥ min_len."""
    return 1 << (min_len - 1).bit_length()


def full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using a pre‑chosen FFT size."""
    A = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(A * A, n=size)
    return conv_full[: 2 * a.shape[0] - 1]


def weighted_gradient(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Compute 2 * (w * a_rev) restricted to the central n entries.
    w has length 2n‑1, a_rev is a[::-1] (length n).
    Returns vector of length n.
    """
    W = np.fft.rfft(w, n=size)
    AR = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * AR, n=size)[: 3 * a_rev.shape[0] - 2]
    n = a_rev.shape[0]
    # central slice: indices n‑1 … 2n‑2 (inclusive) → length n
    grad = 2.0 * conv[n - 1 : 2 * n - 1]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with a smooth‑max objective
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    n: int,
    deadline: float,
    lr: float = 0.02,
    alpha_start: float = 0.5,
    alpha_end: float = 80.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation that minimises the peak of a*a.

    Returns the best sequence (already normalised to sum 1) and its
    evaluation score.
    """
    # initialise
    a = project_simplex(a0.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    size = _fft_size(3 * n - 2)          # enough for both convolutions
    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current objective -----
        conv = full_convolution(a, size)
        max_conv = float(conv.max())
        score = 2.0 * n * max_conv

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[it {it:5d}] new best score = {score:.6f}")

        # ----- temperature for smooth‑max -----
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # ----- softmax weights over convolution entries -----
        # shift by max_conv for numerical stability
        w_raw = np.exp(alpha * (conv - max_conv))
        w = w_raw / w_raw.sum()          # normalised weights

        # ----- gradient of the smooth‑max -----
        a_rev = a[::-1]
        grad = weighted_gradient(w, a_rev, size)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- back onto the simplex -----
        a = project_simplex(a)

    return best_seq, best_score
