        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣  **Parametric sweep** – generate a few hundred simple families
    (power‑law, exponential, double‑exponential, cosine‑bump,
    Gaussian) and keep the best normalised vector.

2️⃣  **Multi‑restart Adam on a smoothed max** – for each restart
    the current vector is optimised with Adam using a soft‑max
    approximation of the supremum norm.
    The smoothing parameter α is increased linearly from 4 to 120.
    The best vector over all restarts is retained.

3️⃣  **Projected sub‑gradient descent on the true max** – after the
    Adam phase we switch to the exact objective (max of the
    convolution).  The sub‑gradient at the current maximiser is
    simply 2·a shifted by the maximising index.  A diminishing step
    size and Euclidean projection onto the probability simplex are
    used.

4️⃣  **Guided random mass‑transfer hill‑climbing** – finally we
    perform cheap stochastic moves that relocate a small
    fraction of mass from a large entry to a random entry; any
    improvement is kept.

All stages respect the global time budget (default 1000 s).  The
returned sequence is scaled to the normalisation used by the
reference code (sum = √(2 n)), but the evaluation function is scale
invariant.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max(conv(a,a)) using an FFT of length L."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv.max()

        a_new = project_simplex(a_new)

        new_max = _conv_max(a_new, L)
        if new_max < cur_max - 1e-14:                # accept only strict improvement
            a = a_new
            cur_max = new_max
            if new_max < best_obj:
                best_obj = new_max
                best_a = a_new.copy()
                print(f"  sub‑grad iter {t:4d} improved | max_conv={new_max:.8e} | score={2*n*new_max:.6f}")
        t += 1
        if t > 50000:                               # safeguard against infinite loop
            break

    # ------------------------------------------------------------------
    # 4.  Guided random hill‑climbing (cheap mass transfers)
    # ---------------------------------------------------------------
    print("--- Phase 4: Guided random moves ---")
    a_cur = best_a.copy()
    while time.time() < deadline:
        # source index – bias towards large entries
        src = np.random.choice(n, p=a_cur)
        dst = np.random.randint(0, n)
        if src == dst:
            continue
        # transfer up to 5 % of the source mass
        delta = np.random.rand() * 0.05 * a_cur[src]
        if delta <= 0:
            continue
        a_new = a_cur.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        if a_new[src] < 0:
            continue
        new_max = _conv_max(a_new, L)
        if new_max + 1e-15 < best_obj:
            best_obj = new_max
            best_a = a_new.copy()
            a_cur = a_new
            print(f"  random move improved | max_conv={new_max:.8e} | score={2*n*new_max:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the reference normalisation (sum = √(2·n))
    final_seq = (best_a * target_sum).tolist()
    # final log line (optional, does not affect the returned value)
    try:
        final_score = evaluate_sequence(final_seq)  # noqa: F821
        print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {t} sub‑grad steps ---")
    except Exception:
        # evaluate_sequence may not be present in the local namespace during testing
        pass

    return final_seq
