"""
Hybrid local‑search + occasional linear‑programming refinement.

The algorithm works on a probability vector (sum = 1) of length *n*.
Its goal is to minimise the quantity

    score = 2·n·max(conv(a,a))         (the denominator is fixed =1)

which is exactly the value returned by `evaluate_sequence` after the
normalisation step.

1. **Initialisation**
   * Start from the half‑order (central‑binomial) series, which already
     yields ~π/2 ≈ 1.571.
   * Perform cheap parametric sweeps (power‑law tilts, mixing with the
     uniform vector, symmetric Beta‑shapes) and keep the best of these.
   * If the global variable `height_sequence_1` exists, use it (after
     normalising) as an additional starting point.

2. **Main optimisation loop**
   * The loop runs until the time budget expires.
   * *Random pair exchange* – move a small portion of mass from one
     index to another; accept improvements always and worse moves with a
     Metropolis probability that cools down.
   * *Projected sub‑gradient step* – compute a sub‑gradient of the
     current maximal convolution entry, take a back‑tracking line‑search
     step and project back onto the simplex.
   * *LP refinement* (every `lp_interval` iterations):
        * Build the Toeplitz matrix `A` such that `A @ g = conv(g,cur)`.
        * Solve the linear programme  
              maximise   1ᵀg   subject to   A·g ≤ M,   g ≥ 0,
          where `M` is the current maximal convolution value.
        * Normalise the solution to the simplex and try a line‑search
          mixing `cur` with this direction; any reduction of `M`
          is accepted.
   * Occasionally add a small random jitter to escape plateaus.

3. **Book‑keeping**
   * The best sequence (lowest score) seen so far is stored and printed
     periodically.
   * At the end the best sequence is returned as a plain Python list.

The combination of cheap stochastic moves, principled gradient descent,
and occasional exact LP “mass‑reallocation” steps gives a much richer
exploration of the feasible set and typically improves the bound well
below the previous 1.552385 limit within the 1000‑second budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """Central‑binomial coefficients C(2i,i)/4**i, normalised to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of *seq*."""
    # np.convolve is already highly optimised (uses FFT for large arrays)
    return np.convolve(seq, seq).max()


def _score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) (sum(seq) == 1)."""
    return 2.0 * len(seq) * _conv_max(seq)


def _build_conv_matrix(seq: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(g, seq).
    Row k contains seq[0…] shifted so that a_j contributes to row i=j,…,j+n‑1.
    """
    n = len(seq)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = seq
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that achieves a
    low value of `evaluate_sequence`.  The routine combines cheap stochastic
    moves, projected sub‑gradient descent and a periodic linear‑programming
    refinement.
    """
    # ------------------------------------------------------------------
    # 0) setup & time budget
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s) - 0.5   # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    if n <= 0:
        n = 1000

    # ------------------------------------------------------------------
    # 2) initialise from a good starting point
    # ------------------------------------------------------------------
    # a) try to reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        ext = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ext.shape[0] == n:
            base = ext.copy()
            base /= base.sum()
        else:
            base = _half_order_series(n)
    else:
        base = _half_order_series(n)

    cur_seq = base.copy()
    cur_score = _score(cur_seq)
    cur_max = _conv_max(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2a) cheap parametric sweep – power‑law tilt
    # ------------------------------------------------------------------
    for delta in np.linspace(0.0, 1.0, 21):
        tilt = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * tilt
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            cur_seq, cur_score, cur_max = cand.copy(), sc, _conv_max(cand)
            print(f"[init] delta={delta:.3f} → score {sc:.6f}")

    # ------------------------------------------------------------------
    # 2b) mix with uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - alpha) * base + alpha * uniform
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            cur_seq, cur_score, cur_max = cand.copy(), sc, _conv_max(cand)
            print(f"[init] mix={alpha:.3f} → score {sc:.6f}")

    # ------------------------------------------------------------------
    # 2c) symmetric Beta‑shapes (a_i ∝ (i+1)^{p‑1} (n‑i)^{p‑1})
    # ------------------------------------------------------------------
    for p in np.linspace(0.1, 2.0, 20):
        idx = np.arange(1, n + 1, dtype=np.float64)
        cand = (idx ** (p - 1)) * ((n - idx + 1) ** (p - 1))
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            cur_seq, cur_score, cur_max = cand.copy(), sc, _conv_max(cand)
            print(f"[init] beta p={p:.3f} → score {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    move_prob = 0.6                      # probability of a random exchange
    init_temp = cur_max * 0.5
    final_temp = 1e-12

    step_g = 0.05                        # sub‑gradient step size
    step_g_min = 1e-8
    step_g_max = 0.5

    lp_interval = 2500                    # how often we run the LP refinement
    jitter_interval = 5000                # occasional random perturbation

    # try‑import the LP solver once (it is optional)
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception:                    # pragma: no cover
        have_lp = False

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        prog = elapsed / total_time
        temp = init_temp * (final_temp / init_temp) ** prog

        # --------------------------------------------------------------
        # 3a) random pair exchange
        # --------------------------------------------------------------
        if np.random.rand() < move_prob:
            i, j = np.random.choice(n, 2, replace=False)
            max_delta = min(cur_seq[i], 0.02)   # move at most 2 % of total mass
            if max_delta > 0:
                delta = np.random.rand() * max_delta
                cand_seq = cur_seq.copy()
                cand_seq[i] -= delta
                cand_seq[j] += delta
                cand_max = np.convolve(cand_seq, cand_seq).max()
                accept = False
                if cand_max < cur_max:
                    accept = True
                else:
                    # Metropolis acceptance for occasional uphill moves
                    prob = math.exp(-(cand_max - cur_max) / temp) if temp > 0 else 0.0
                    if np.random.rand() < prob:
                        accept = True
                if accept:
                    cur_seq = cand_seq
                    cur_max = cand_max
                    cur_score = 2.0 * n * cur_max
                    if cur_score < best_score:
                        best_score, best_seq = cur_score, cur_seq.copy()
                        print(f"[{iteration:6d}] Random move improved: {best_score:.6f}")

        # --------------------------------------------------------------
        # 3b) projected sub‑gradient descent on the active set
        # --------------------------------------------------------------
        else:
            conv = np.convolve(cur_seq, cur_seq)
            cur_max = conv.max()
            # active indices (within a tiny tolerance of the maximum)
            tol = 1e-12
            active = np.where(conv >= cur_max - tol)[0]

            grad = np.zeros_like(cur_seq)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                rev = k - idx
                grad[idx] += 2.0 * cur_seq[rev]
            if len(active) > 1:
                grad /= float(len(active))

            improved = False
            trial_step = step_g
            for _ in range(5):
                cand = cur_seq - trial_step * grad
                cand = _project_simplex(cand)
                cand_max = np.convolve(cand, cand).max()
                if cand_max < cur_max - 1e-14:
                    cur_seq = cand
                    cur_max = cand_max
                    cur_score = 2.0 * n * cur_max
                    step_g = min(step_g * 1.1, step_g_max)
                    improved = True
                    break
                trial_step *= 0.5
            if not improved:
                step_g = max(step_g * 0.7, step_g_min)
                if iteration % jitter_interval == 0:
                    sigma = 0.002
                    noise = np.random.randn(n) * sigma
                    jitter = cur_seq * (1.0 + noise)
                    jitter = np.clip(jitter, 0.0, None)
                    jitter /= jitter.sum()
                    cur_seq = jitter
                    cur_max = np.convolve(cur_seq, cur_seq).max()
                    cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score, best_seq = cur_score, cur_seq.copy()
                print(f"[{iteration:6d}] Sub‑grad improved: {best_score:.6f}")

        # --------------------------------------------------------------
        # 3c) occasional LP refinement (if SciPy is available)
        # --------------------------------------------------------------
        if (iteration % lp_interval == 0) and have_lp and (deadline - time.time() > 2.0):
            try:
                A = _build_conv_matrix(cur_seq)           # (2n‑1)×n
                M = cur_max
                c = -np.ones(n)                           # maximise sum(g)
                b = np.full(2 * n - 1, M, dtype=np.float64)
                # Small time‑limit to keep the overall run fast
                res = linprog(c, A_ub=A, b_ub=b,
                              bounds=(0, None), method='highs',
                              options={'time_limit': 5.0})
                if res.success:
                    g = res.x
                    sum_g = g.sum()
                    if sum_g > 0:
                        g_norm = g / sum_g                     # project to simplex
                        # try a few mixing factors – descending steps are cheap
                        for t in [0.5, 0.25, 0.125, 0.0625, 0.03125,
                                  0.015625, 0.0078125, 0.00390625, 0.001953125]:
                            cand = (1.0 - t) * cur_seq + t * g_norm
                            cand_max = np.convolve(cand, cand).max()
                            if cand_max < cur_max:
                                cur_seq = cand
                                cur_max = cand_max
                                cur_score = 2.0 * n * cur_max
                                if cur_score < best_score:
                                    best_score, best_seq = cur_score, cur_seq.copy()
                                    print(f"[{iteration:6d}] LP step t={t:.5f} improved score: {best_score:.6f}")
                                break
            except Exception:                      # pragma: no cover
                pass

        # --------------------------------------------------------------
        # 3d) periodic progress report
        # --------------------------------------------------------------
        if iteration % 3000 == 0:
            print(f"[{iteration:6d}] current best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final result
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Return a plain Python list to satisfy the external caller.
    return best_seq.tolist()
