"""
Mirror‑descent optimisation for the self‑convolution bound.

We work on the probability simplex (a_i ≥ 0, Σ a_i = 1) because the
evaluation function is homogeneous of degree 0.  At each iteration we
approximate the non‑smooth objective  max_k (a*a)[k] by a
log‑sum‑exp (smooth‑max) with temperature ε≈0.5·max_k (a*a)[k].
The gradient of the smooth‑max is a simple convolution that can be
computed in O(n²) with NumPy.  We then perform a multiplicative (exponentiated)
gradient step, which is the natural mirror‑descent update for the simplex,
followed by renormalisation.  The step‑size η is adapted on the fly:
if a step improves the true max‑convolution we increase η,
otherwise we shrink it and occasionally inject a small random jitter to
escape plateaus.  The best sequence found during the run is returned.

The routine accepts an optional starting point `height_sequence_1` (the
previous best construction) and a user‑supplied length `n` (default 1000).
It runs for at most `budget_s` seconds (default 1000) and prints
intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _half_order_series(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
