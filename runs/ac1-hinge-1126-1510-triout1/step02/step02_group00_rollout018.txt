"""
Propose candidate for the “step‑function” minimisation problem.

Algorithm
---------
1. Initialise a probability vector a (sum = 1).  If a previous solution
   `height_sequence_1` exists it is used (truncated / padded to 4000
   entries); otherwise we start from the uniform distribution.

2. Deterministic shape search.
   * Power‑law families   a_i ∝ (i+1)^(-p)   with p∈[0.45,0.65]
   * Exponential families a_i ∝ exp(-λ·i)   with λ∈[0.001,0.02]
   * Simple linear drop   a_i ∝ max(0, 1–s·i)
   All candidates are normalised to sum 1 and evaluated with a fast
   FFT‑based convolution.  The best candidate is kept.

3. L‑BFGS refinement on a smooth approximation of the max (log‑sum‑exp)
   with a decreasing smoothing factor schedule.  The optimisation is
   performed on logits y (softmax(y) → a) and the best result replaces
   the current best vector.

4. Projected sub‑gradient optimisation of the true max.  The method
   repeatedly computes the active set (indices where the convolution
   attains its current maximum) and takes a diminishing step in the
   opposite direction, projecting back onto the simplex.

5. Final greedy hill‑climb: repeatedly transfer a small random fraction
   of mass from one coordinate to another while the fast score improves.

All steps respect the time budget (default 1000 s).  The final sequence
is returned as a plain list of non‑negative floats; `evaluate_sequence`
is called once at the end for reporting.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via real FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast computation of 2·n·max(a*a) for a distribution that already sums to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution of a_rev with w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # central part of length n
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: 2·(mask * a_rev) restricted to n entries
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len_g)
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        conv_g = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # --------------------------------------------------------------
    # Initial vector (use the previous construction if available)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.array(globals()['height_sequence_1'], dtype=float)
        if init.ndim == 1 and init.size > 0:
            if init.size != n:
                # truncate or zero‑pad to length n
                a0 = np.zeros(n)
                m = min(init.size, n)
                a0[:m] = init[:m]
                a0 = np.maximum(a0, 0.0)
            else:
                a0 = np.maximum(init, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Deterministic shape search
    # --------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=float)

    # ---- power‑law families ----
    for p in np.linspace(0.45, 0.65, 41):      # step ≈0.005
        a = idx ** (-p)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[shape] power p={p:.4f} score={score:.6f}")

    # ---- exponential families ----
    for lam in np.linspace(0.001, 0.020, 40):
        a = np.exp(-lam * np.arange(n, dtype=float))
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[shape] exp λ={lam:.5f} score={score:.6f}")

    # ---- simple linear drop (optional) ----
    for s in np.linspace(0.0, 4.0 / n, 20):
        a = np.maximum(0.0, 1.0 - s * np.arange(n, dtype=float))
        if a.sum() == 0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[shape] lin slope={s:.8f} score={score:.6f}")

    # --------------------------------------------------------------
    # 2. L‑BFGS refinement on a smoothed max
    # --------------------------------------------------------------
    lam_schedule = [0.50, 0.30, 0.15, 0.08, 0.04]   # decreasing smoothing factor
    for lam in lam_schedule:
        if time.time() > deadline - 30:
            break
        y0 = np.log(best_a + 1e-12)
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200,
                                'gtol': 1e-9,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] lam={lam:.3f} score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient optimisation (true max)
    # --------------------------------------------------------------
    if time.time() < deadline - 10:

                                                  fft_len_conv,

    # --------------------------------------------------------------
    # 4. Final greedy hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        deadline,
                                        max_iter=5000)
        print(f"[final hill] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the candidate and report the official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence score = {final_eval:.6f}")
    return final_seq
