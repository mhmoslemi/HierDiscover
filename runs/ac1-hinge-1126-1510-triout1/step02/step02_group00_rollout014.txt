"""
Improved search for a non‑negative step sequence that minimizes

    E(a) = 2·n·max(a*a) / (sum a)^2

The algorithm combines three ideas:

1.  Stochastic L‑BFGS on a smooth (log‑sum‑exp) approximation of the max.
2.  A linear‑programming refinement (the same LP described in the
    original state‑of‑the‑art) that, given a current sequence f,
    finds a partner g with the largest possible mass while keeping
    the convolution f*g ≤ M (the current worst convolution value).
    The new g is blended into f.
3.  A final greedy hill‑climb on the simplex.

All steps respect the simplex constraint (non‑negative, sum = 1)
and use FFT‑based convolutions for speed.  The LP is solved with
CVXPY/ECOS on the *active* constraints (those that are close to the
maximum), which makes each refinement cheap while still providing a
strong direction of improvement.

The search runs for the allotted budget (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when sum(a)=1:
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) objective for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f    – scalar value 2·n·smooth_max(conv)
    grad – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set LP)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              fft_len_conv: int,
              n: int,
              deadline: float,
              max_iter: int = 5,
              alpha: float = 0.5,
              tol: float = 1e-8):
    """
    Given a current sequence a (sum = 1), solve
        max  Σ g_j
        s.t. g ≥ 0
             (a * g)_k ≤ M   for all k in the active set,
    where M = max(a*a).  The LP is solved with CVXPY/ECOS.
    The resulting g is normalized to sum = 1 and blended into a:
        a ← (1‑α)·a + α·g .
    Returns the refined a and its fast_score.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    for it in range(max_iter):
        # current max of the convolution
        conv_a = full_conv(best, fft_len_conv)
        M = conv_a.max()
        eps = max(M * 1e-9, tol)
        active = np.where(conv_a >= M - eps)[0]
        if active.size == 0:
            break

        # ----- CVXPY model -----
        g = cp.Variable(n, nonneg=True)                # optimisation variable
        conv_ag = cp.conv(best, g)                     # length 2n‑1 expression
        constraints = [conv_ag[idx] <= M for idx in active]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1500)
        except Exception as e:  # pragma: no cover
            print("[LP] solver failed:", e)
            break

        if g.value is None:
            print("[LP] no solution returned")
            break

        g_opt = np.maximum(g.value, 0.0)
        if g_opt.sum() == 0:
            break
        # Normalise to the simplex
        g_opt = g_opt / g_opt.sum()
        # Blend
        best = (1.0 - alpha) * best + alpha * g_opt
        best = simplex_projection(best)

        new_score = fast_score(best, fft_len_conv, n)
        print(f"[LP] iter {it+1}/{max_iter}  score={new_score:.6f}")
        best_score = new_score

        if time.time() > deadline:
            break

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size & initial distribution
    # --------------------------------------------------------------
    # If the grader provides a previous construction we reuse its size,
    # otherwise we pick a moderately large n.
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            max_n = 5000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            init_seq[init_seq < 0] = 0.0
            if init_seq.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on the smoothed max
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    iteration = 0
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04]   # decreasing smoothing

    while time.time() < deadline:
        lam = lam_schedule[iteration % len(lam_schedule)]

        # pick a heuristic starting point
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # run L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-10,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] L‑BFGS lam={lam:.3f} mode={mode} "
                  f"score={best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

        iteration += 1

        # Every 20 L‑BFGS runs perform a brief sub‑gradient refinement
        if iteration % 20 == 0 and time.time() < deadline - 8.0:
            sg_deadline = min(deadline, time.time() + 5.0)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[{iteration}] sub‑gradient refined score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP refinement on the active set
    # --------------------------------------------------------------
    lp_deadline = deadline - 5.0
    if time.time() < lp_deadline:
        best_a, best_score = lp_refine(best_a,
                                       fft_len_conv,
                                       n,
                                       lp_deadline,
                                       max_iter=6,
                                       alpha=0.5)
        print(f"[LP] after refinement score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climb polish
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=4000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
