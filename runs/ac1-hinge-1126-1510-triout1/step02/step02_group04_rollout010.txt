def project_simplex(v):
    """Project onto probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has two phases:

1. **Smooth‑max Adam optimisation** – the objective
   `max(conv)` is approximated by a log‑sum‑exp, gradients are
   obtained by a convolution with soft‑max weights and the parameters
   are kept on the simplex by a soft‑max re‑parameterisation.
   Several restarts (previous best, uniform, random Dirichlet,
   power‑law and sinusoidal shapes) are tried; the best result of all
   restarts is kept.

2. **Stochastic hill‑climbing fine tuning** – starting from the best
   simplex vector obtained above, we repeatedly move a small amount of
   mass from a source index *i* to a target index *j*.  The convolution
   can be updated analytically in O(n) time, so the new `max(conv)` can
   be evaluated cheaply.  A simple simulated‑annealing acceptance rule
   (always accept improvements, sometimes accept deteriorations)
   enables escaping shallow local minima.

Both phases respect the 1000 s wall‑time budget; progress is printed
so you can see intermediate bounds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions used by the Adam phase
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a vector that sums to 1)."""
    vmax = v.max()
    e = np.exp(v - vmax)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a, returned length = 2·len(a)‑1.
    Uses FFT of length L (power‑of‑two ≥ 2·len(a)-1) for speed.
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Full correlation w ⊛ a_rev (same length as the full convolution of
    w (length 2n‑1) with a_rev (length n)).  Implemented via FFT.
    """
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used only by the hill‑climb
# if one would like to re‑normalise – here we keep the sum exactly 1).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hill_climb(
    a_init: np.ndarray,
    conv_init: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    best_a: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Fine‑tune a simplex vector `a_init` by moving a small amount of mass
    between two positions while updating the convolution exactly.
    Returns the best vector found and its (unnormalised) objective value
    2·n·max(conv).
    """
    cur_a = a_init.copy()
    cur_conv = conv_init.copy()
    cur_score = 2.0 * n * cur_conv.max()

    # parameters for the stochastic search
    step = 0.5          # relative step size (d = step * a[i] * uniform)
    temperature = 1e-3
    step_decay = 0.9999
    temp_decay = 0.9999
    max_iters = 500_000   # hard limit; actual loop stops at the deadline

    it = 0
    while time.time() < deadline and it < max_iters:
        # --- pick source (i) proportionally to its current weight ---
        if np.all(cur_a <= 0):
            break
        i = np.random.choice(n, p=cur_a)
        if cur_a[i] <= 1e-15:
            it += 1
            continue

        # --- pick a distinct target (j) uniformly ---
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # --- propose a transfer of mass d from i to j ---
        d = step * cur_a[i] * np.random.rand()
        if d <= 0:
            it += 1
            continue

        # --- update the convolution exactly (O(n) work) ---
        conv_new = cur_conv.copy()
        # linear part: shift of the whole vector a
        conv_new[i : i + n] -= 2.0 * d * cur_a
        conv_new[j : j + n] += 2.0 * d * cur_a
        d2 = d * d
        conv_new[2 * i] += d2
        conv_new[2 * j] += d2
        conv_new[i + j] -= 2.0 * d2

        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score

        # --- acceptance rule (simulated annealing) ---
        accept = False
        if delta < 0:
            accept = True
        else:
            # Accept uphill moves with a probability that decays with T
            if temperature > 0:
                accept = np.random.rand() < math.exp(-delta / temperature)

        if accept:
            cur_a[i] -= d
            cur_a[j] += d
            cur_conv = conv_new
            cur_score = new_score

            # record global best
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[hill‑climb] iter {it:6d}  new best = {best_score:.6f}")

        # decay step size and temperature
        step = max(step * step_decay, 1e-5)
        temperature = max(temperature * temp_decay, 1e-10)

        it += 1

    return best_a, best_score
