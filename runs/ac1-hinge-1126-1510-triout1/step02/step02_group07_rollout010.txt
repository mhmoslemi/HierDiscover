    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = kwargs.get('n', 1000)
    # generate initial candidates
    # We'll try 4 restarts: half-order, uniform, random Dirichlet(0.5), random power law
    init_candidates = []
    init_candidates.append(_half_order_series(n))
    init_candidates.append(np.full(n, 1.0/n))
    # random Dirichlet (alpha=0.5)
    init_candidates.append(np.random.dirichlet(np.full(n, 0.5)))
    # power law seeds
    for p in [0.2, 0.5, 0.8, 1.0]:
        seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
        seq /= seq.sum()
        init_candidates.append(seq)

    # variables to keep best
    best_seq = None
    best_score = np.inf

    # parameters for Adam-like updates
    lr = 0.05   # learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # iterate
    # choose candidate cyclically
    cand_idx = 0
    iteration = 0
    while time.time() < deadline:
        # pick the current base
        a = init_candidates[cand_idx].copy()
        # reset Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0
        # local loop for this candidate, spend some time then switch
        local_deadline = time.time() + min(2.0, deadline - time.time())  # small chunk to avoid hogging any single restart? Instead maybe loop for certain number of iterations.
        # Actually we need to interleave restarts to avoid stagnation: we will run a number of internal iterations (e.g., 2000) then move to next candidate
        while time.time() < deadline and time.time() < local_deadline:
            t_iter += 1
            iteration += 1

            # compute convolution and score
            conv = np.convolve(a, a)
            cur_score = 2 * n * conv.max()
            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration}] new global best {best_score:.6f}")

            # compute softmax over conv for smooth approx
            max_conv = conv.max()
            beta = 10.0 + 0.001 * iteration   # slowly increase

"""
Hybrid smooth‑max/ADAM optimizer on the probability simplex.

The objective for a non‑negative sequence a (sum(a)=1) is
    score(a) = 2·n·max_k (a∗a)_k ,
where “∗” is the discrete convolution.  The score is invariant under
rescaling, so we keep the sequence on the simplex.

The algorithm proceeds as follows:

1. Initialise several diverse candidates (half‑order series, uniform,
   random Dirichlet, power‑law decays, and – if present – the previously
   discovered best sequence `height_sequence_1`).

2. For the current candidate we minimise a smooth approximation of the
   max‑operator:
       f_β(a) = log‑sum‑exp( β·(a∗a) ) .
   Its gradient is
       ∇f_β(a) = 2·corr( softmax_β(a∗a) , a ),
   i.e. twice the cross‑correlation of the soft‑max weights with a.
   The correlation and the convolution are computed with a single FFT
   (size ≥ 3·n) for speed.

3. An ADAM‑style update on the gradient is performed, followed by a
   projection onto the simplex (guaranteeing non‑negativity and unit
   sum).  The “temperature’’ β is slowly increased so that the
   approximation becomes sharper as optimisation proceeds.

4. Every few thousand iterations a tiny random jitter is added and the
   ADAM moments are reset, helping the search escape shallow plateaux.
   The best sequence ever seen is stored and printed.

The loop runs until the allotted time budget (≈ 1000 s) expires and the
best sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np


def _half_order_series(n: int) -> np.ndarray:
    """
    First n coefficients a_i = C(2i,i) / 4**i, normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1 }.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(
    seed: int = 42,
    budget_s: float = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function height sequence with a low
    `evaluate_sequence` score.  The routine mixes several initialisations
    and then runs a smooth‑max ADAM optimiser on the simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin so the caller can still stop cleanly
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1500))          # default length; larger gives more freedom

    # ------------------------------------------------------------------
    # Prepare a few diverse starting points
    # ------------------------------------------------------------------
    initial = []

    # (i) half‑order series (already good)
    initial.append(_half_order_series(n))

    # (ii) uniform distribution
    initial.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (iii) random Dirichlet (α = 0.5 gives a spiky start)
    initial.append(np.random.dirichlet(np.full(n, 0.5)))

    # (iv) simple power‑law decays
    for p in (0.2, 0.5, 0.8, 1.0):
        seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        seq /= seq.sum()
        initial.append(seq)

    # (v) previous best if it is present in the global namespace
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = np.clip(prev, 0.0, None)          # ensure non‑negative
                if prev.sum() > 0:
                    prev = prev / prev.sum()
                    initial.append(prev)
        except Exception:
            pass  # ignore malformed global variable

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # ADAM parameters
    lr = 0.05                # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Soft‑max temperature (smoothness of the max operator)
    beta = 10.0              # start relatively smooth
    beta_max = 3000.0
    beta_factor = 1.001      # slow exponential increase

    # Jitter settings
    jitter_interval = 2000   # iterations between random perturbations
    jitter_std = 0.001

    # ------------------------------------------------------------------
    # Helper: compute full convolution via a single FFT of length M
    # ------------------------------------------------------------------
    def fft_convolve_self(x: np.ndarray, M: int) -> np.ndarray:
        """
        Return the linear convolution of x with itself (length 2·len(x)-1)
        using FFT of length M (M ≥ 2·len(x)-1).
        """
        X = np.fft.fft(x, M)
        conv_full = np.fft.ifft(X * X).real
        L = 2 * x.size - 1
        return conv_full[:L]

    # ------------------------------------------------------------------
    # Main optimisation loop (cycling through the initial candidates)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    total_iters = 0
    cand_idx = 0

    # Pre‑compute the FFT length once (it depends only on n)
    M = 1 << ((3 * n - 2).bit_length())   # enough for convolution + correlation

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) pick a start and reset ADAM moments
        # ------------------------------------------------------------------
        a = initial[cand_idx].copy()
        cand_idx = (cand_idx + 1) % len(initial)

        # Ensure we are on the simplex (numerical safety)
        a = _project_simplex(a)

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        # Local time bucket (run a few thousand iterations on this start)
        local_deadline = min(deadline, time.time() + 2.0)

        while time.time() < local_deadline and time.time() < deadline:
            t += 1
            total_iters += 1

            # ------------------------------------------------------------------
            # 2) convolution, soft‑max weights, and gradient
            # ------------------------------------------------------------------
            # a∗a via FFT (length 2n‑1)
            conv = fft_convolve_self(a, M)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{total_iters:7d}] global best score: {best_score:.6f}")

            # soft‑max over convolution entries (smooth max)
            # prevent overflow by subtracting max before exponentiation
            exp_vals = np.exp(beta * (conv - cur_max))
            w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

            # correlation w ⋆ a via FFT (gradient = 2·(w⋆a))
            W = np.fft.fft(w, M)
            A = np.fft.fft(a, M)
            corr = np.fft.ifft(np.conj(W) * A).real   # length M, linear correlation
            grad = 2.0 * corr[:n]                     # take first n entries

            # ------------------------------------------------------------------
            # 3) ADAM update on the simplex
            # ------------------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = -lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a + step
            a = _project_simplex(a)

            # ------------------------------------------------------------------
            # 4) schedule updates
            # ------------------------------------------------------------------
            # increase soft‑max temperature, but cap it
            if beta < beta_max:
                beta *= beta_factor

            # occasional jitter to escape flat regions
            if total_iters % jitter_interval == 0:
                noise = np.random.randn(n) * jitter_std
                a = a * (1.0 + noise)
                a = np.clip(a, 0.0, None)
                a /= a.sum()               # renormalise onto the simplex
                m.fill(0.0)
                v.fill(0.0)

            # periodic progress report
            if total_iters % 2000 == 0:
                elapsed = time.time() - start_time
                print(
                    f"[{total_iters:7d}] elapsed {elapsed:.1f}s | "
                    f"current best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # This should never happen, but fall back to the half‑order series
        best_seq = _half_order_series(n)

    print(f"Finished after {total_iters} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
