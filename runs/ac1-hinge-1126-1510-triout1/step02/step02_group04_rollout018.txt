def project_simplex(v):
    # ensure v >= 0? algorithm handles negative.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

    # Renormalise exactly to target sum (very small numerical drift)
    s = best_seq.sum()
    if s > 0:

Proposes a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The method is a three‑phase optimiser:

1.  Multi‑restart Adam on the simplex (a = softmax(v)).  The non‑smooth
    max is approximated by a log‑sum‑exp with a β that grows from 5 to 200,
    and the learning‑rate decays slowly.  Several diverse initialisations
    (previous best, uniform, Dirichlet with different concentrations and a
    few power‑law shapes) are tried.

2.  After Adam, a hard‑max sub‑gradient descent is performed.
    The exact max‑index of the convolution is identified, the corresponding
    sub‑gradient is computed via FFT‑based correlation and a projected‑gradient
    step on the simplex is taken.  The step size decays, and a tiny
    simulated‑annealing acceptance rule lets the search escape shallow plateaus.

3.  Finally a simple hill‑climbing stage moves small amounts of mass from one
    index to another (preserving the sum) and keeps any improvement.  This
    exploits the cheap O(n log n) convolution via FFT.

All phases respect a 1000 s wall‑clock budget (a 5 s safety margin).  The
returned sequence is clipped to the limits enforced by ``evaluate_sequence``
and re‑normalised exactly to Σa = √(2 n).  The function works with the global
``height_sequence_1`` if it exists, otherwise falls back to a default length.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------------------- #

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    vmax = v.max()
    e = np.exp(v - vmax)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (sum = 1, v_i ≥ 0)."""
    n = v.shape[0]
    # Fast O(n log n) implementation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if len(rho) == 0:
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a (full, length 2·len(a)‑1)."""
    fa = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Full correlation (w ⋆ a_rev) used for gradient computation."""
    fw = np.fft.fft(w, n=L)
    frev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fw * frev))
    return corr
