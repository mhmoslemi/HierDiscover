"""
Hybrid simplex optimizer for the self‑convolution peak.

The algorithm mixes three classic strategies on the probability simplex:
1. Projected sub‑gradient descent on the active convolution peak.
2. Smoothed (log‑sum‑exp) gradient descent that gradually tightens the
   approximation to the max‑operator.
3. Multiplicative (exponentiated) updates on the active peak.

Each restart starts from a random power‑law (or a perturbed copy of the
current best) and runs the hybrid routine for as many iterations as the
time budget permits.  The best sequence seen across all restarts is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008, O(n))
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core hybrid optimizer
# ----------------------------------------------------------------------
def hybrid_optimize(a0: np.ndarray,
                    max_iters: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Hybrid convex optimisation for the peak of a*a.

    Returns the best sequence encountered and its score (the value of
    evaluate_sequence for a vector normalised to sum=1, i.e. 2*n*max_conv).
    """
    a = a0.copy()
    n = a.shape[0]

    # normalise to the simplex (invariant under scaling)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # FFT sizes (powers of two for speed)
    fft_size = 1 << ((2 * n - 1).bit_length())           # for a*a
    grad_fft_size = 1 << ((3 * n - 2).bit_length())     # for w * a_rev

    # hyper‑parameters
    step0_sub = 0.9          # sub‑gradient step prefactor
    step0_smooth = 0.30      # smoothed‑gradient step prefactor
    step0_exp = 0.05         # exponentiated‑gradient step prefactor

    # split the budget into three phases
    phase1 = int(max_iters * 0.30)               # sub‑gradient
    phase2 = int(max_iters * 0.70)               # smoothed
    # the rest (phase3) uses exponentiated updates

    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- full self‑convolution via FFT -----
        A = np.fft.rfft(a, fft_size)
        conv_full = np.fft.irfft(A * A, fft_size)[: 2 * n - 1]
        max_conv = float(conv_full.max())
        score = 2.0 * n * max_conv                     # sum(a) == 1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[hybrid it {it:5d}] new local best score={score:.6f}")

        # ----- decide which update to apply -----
        if it <= phase1:
            # ---- 1. Projected sub‑gradient on the active peak ----
            k = int(np.argmax(conv_full))
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k - idx]                # gradient of the peak
            step = step0_sub / np.sqrt(it)
            a = project_simplex(a - step * grad)

        elif it <= phase2:
            # ---- 2. Smoothed (log‑sum‑exp) gradient descent ----
            # gradually increase the sharpness parameter
            gamma = min(200.0, 10.0 + 0.05 * it)

            # softmax weights over convolution entries
            shifted = conv_full - max_conv                # stability shift
            w = np.exp(gamma * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * (w * a_rev)  (linear convolution)
            w_fft = np.fft.rfft(w, grad_fft_size)
            a_rev_fft = np.fft.rfft(a[::-1], grad_fft_size)
            grad_full = np.fft.irfft(w_fft * a_rev_fft, grad_fft_size)
            start = n - 1
            grad = 2.0 * grad_full[start:start + n]      # slice to length n

            step = step0_smooth / np.sqrt(it)
            a = project_simplex(a - step * grad)

        else:
            # ---- 3. Exponentiated (multiplicative) update on active peak ----
            k = int(np.argmax(conv_full))
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k - idx]                # same sub‑gradient
            step = step0_exp / np.sqrt(it)
            # multiplicative update stays non‑negative automatically
            a = a * np.exp(-step * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:

    return best_a, best_score

    # If a previous construction exists we reuse its length,
    # otherwise we start with a moderately large size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n < 1:
            n = 4000
    except Exception:
        n = 4000

    # safety – cap the length to keep the per‑iteration cost reasonable
    n = max(2000, min(n, 8000))

    # --------------------------------------------------------------
    # Global best across all restarts
    # --------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")
    target_score = 1.503  # stop early if we beat the target

    # parameters for inner optimisation
    max_iters_per_restart = 5000
    restart = 0

    while time.time() < deadline:
        restart += 1

        # ----- choose an initial point for this restart -----
        r = np.random.rand()
        if r < 0.25:
            # uniform start
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        elif r < 0.70:
            # random power‑law start
            p = np.random.uniform(0.45, 0.70)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a0 = idx ** (-p)
            a0 /= a0.sum()
        else:
            # perturbation of the current global best (if any)
            if global_best_seq is not None:
                a0 = np.array(global_best_seq, copy=True)
                # add small Gaussian noise and re‑project onto the simplex
                eps = 0.02 * np.random.randn(n)
                a0 = np.maximum(a0 + eps, 0.0)
                s = a0.sum()
                if s == 0.0:
                    a0 = np.full(n, 1.0 / n)
                else:
                    a0 /= s
            else:
                # fallback to a default power‑law
                p = 0.55
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()

        # ----- decide how many iterations we can afford -----
        time_left = deadline - time.time()
        # empirical cost ≈ 0.004 s per iteration for n≈4000
        it_budget = int(min(max_iters_per_restart, time_left / 0.004))
        if it_budget <= 0:
            break

        # ----- run the hybrid optimiser -----
        seq, local_score = hybrid_optimize(a0, it_budget, deadline)

        # ----- evaluate the obtained sequence with the official scorer -----
        # (the internal score equals 2·n·max_conv when Σa=1, but we call to be safe)
        from __main__ import evaluate_sequence  # imported by the harness
        eval_score = evaluate_sequence(seq.tolist())

        if eval_score < global_best_score:
            global_best_score = eval_score
            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new GLOBAL best score={eval_score:.6f} "
                f"(iters={it_budget})"
            )
            if global_best_score < target_score:
                print("Target achieved – exiting early.")
                break

        # occasional step‑size damping to avoid stagnation
        if restart % 7 == 0:
            # the inner optimiser uses its own schedules; we only
            # decay the random‑power‑law exponent range a little
            pass  # placeholder for future schedule tweaks

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, use a simple power‑law.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()

    # Return as plain Python list
    return global_best_seq.tolist()
