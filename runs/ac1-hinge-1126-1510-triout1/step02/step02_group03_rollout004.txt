    print(f"Returning sequence – best score {best_score:.6f} after {total:.1f}s")

"""
Propose a low‑peak step function.

The algorithm starts from the best known construction (if available) and
repeatedly improves it by solving a **linear program** that maximises the
total mass of a new candidate `g` under the constraint that the convolution
`a * g` does not exceed the current autocorrelation peak `M`.  The optimal
`g` is normalised to a probability vector and mixed with the current `a`
with a line‑search over several mixing ratios.  This operation is guaranteed
to be non‑increasing in the peak value `max(a*a)` (the quantity minimised
by `evaluate_sequence`).  After the LP phase a short stochastic polishing
stage (random mass transfers) is performed to fine‑tune the result.

All vectors are kept on the simplex (non‑negative, sum = 1), so scaling to
the `[0, 1000]` range required by the evaluator does not affect the score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal autocorrelation entry
# ----------------------------------------------------------------------
def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the max entry of a*a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_idx.size == 0:
        return g
    for k in active_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g /= float(active_idx.size)
    return g


# ----------------------------------------------------------------------
# Linear‑program to obtain a direction that maximises total mass
# ----------------------------------------------------------------------
def _solve_lp(a_vec: np.ndarray, M_val: float):
    """
    Solve:
        maximise   sum(g)
        subject to a_vec * g  (convolution)  ≤ M_val  (for all shifts)
                    g ≥ 0
    Returns the optimal g normalised to sum(g)=1,
    or None if the LP fails.
    """
    from scipy.optimize import linprog

    n = a_vec.shape[0]
    rows = 2 * n - 1

    # Build the A matrix: each column j contains a copy of a_vec shifted down by j rows
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a_vec

    b = np.full(rows, M_val, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)        # minimise -sum(g)  → maximise sum(g)
    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS solver
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    return g0 / s


# ----------------------------------------------------------------------
# Main entry point – called by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a small autocorrelation peak.
    The procedure combines an LP‑driven descent with a short stochastic
    polishing stage.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 3.0          # a few seconds safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence (reuse previous construction if possible)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
        a = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 1542                               # default length used earlier
        # a simple power‑law start
        p = np.random.uniform(0.1, 1.8)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))

    # Normalise to the simplex
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=np.float64)
    a = a / a.sum()                            # sum == 1

    # ------------------------------------------------------------------
    # Track the best sequence found so far
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # LP‑driven descent
    # ------------------------------------------------------------------
    iter_no = 0
    while time.time() < deadline:
        iter_no += 1

        # current autocorrelation peak
        conv = _conv_fft(a)
        M = conv.max()

        # solve the LP for a direction g (sum(g)=1)
        g = _solve_lp(a, M)
        if g is None:
            print("[LP] solver failed – terminating LP phase")
            break

        # try a few mixing ratios; keep the first improvement
        improved = False

            cand = (1.0 - alpha) * a + alpha * g
            # ensure numerical feasibility
            cand = _project_to_simplex(cand)

            if score < best_score:

                print(f"[LP {iter_no}] pure g → new best {best_score:.6f}")
                improved = True

        if not improved:
            # no progress – exit the LP loop
            print(f"[LP {iter_no}] no improvement, stop LP phase")
            break

        # safety check on time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final stochastic polishing (small random mass transfers)
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(6.0, deadline - time.time())

        a_new = a_local.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # keep sum exactly 1 (numerical drift may appear)
        a_new = _project_to_simplex(a_new)

        s = evaluate_sequence(a_new.tolist())
        if s < best_score:
            best_score = s
            best_seq = a_new.copy()
            a_local = a_new.copy()
            print(f"[Polish] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Scale to the [0, 1000] interval (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total = time.time() - start
    print(f"Returning sequence – best score {best_score:.6f} after {total:.1f}s")
    return final_seq.tolist()
