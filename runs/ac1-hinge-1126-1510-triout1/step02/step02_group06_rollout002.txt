    # Precompute next_pow2 length for conv

    iteration = 0
    last_improve_time = start
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n - 1]

        max_conv = conv.max()

        # step size schedule
        eta = eta0 / np.sqrt(iteration + 1)
        a_candidate = a - eta * grad
        # project onto simplex (ensuring non-neg sum=1)
        a = project_simplex(a_candidate)

assistantfinal**Solution Overview**

The objective is  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is invariant under scaling, we may restrict to the
probability simplex  

\[
\Delta=\bigl\{a\in\mathbb{R}^n_{\ge0}\;|\;\sum_i a_i=1\bigr\}.
\]

On the simplex the score reduces to `2*n*max(conv)`.  
The convolution `conv = a * a` is a quadratic form, therefore the
maximum is a smooth (convex) function of the *soft‑max* of the
convolution entries.  Using a smooth approximation of the max we can
apply projected (momentum) gradient descent on the simplex.  The main
steps are

1. **initialisation** – pick a length `n` (≈ 10 k–20 k) and a random
   shape (power‑law, exponential or two‑sided power‑law).  Normalise to a
   probability vector.

2. **smooth‑max gradient** – for the current `a`

   * compute `c = a * a` with an FFT,
   * build a soft‑max weight vector  

     \[
     w_k=\frac{\exp(\lambda\,(c_k-\max c))}{\sum_j \exp(\lambda\,(c_j-\max c))}
     \]
     (the subtraction of `max c` avoids overflow),
   * the gradient of the smooth objective is  

     \[
     \nabla f(a)=2\;(a\star w),
     \]

     where `⋆` is a convolution; it is obtained again with an FFT.

3. **projected momentum step**  

   \[
   v\leftarrow\beta v+(1-\beta)\nabla f(a),\qquad
   a\leftarrow\Pi_{\Delta}\bigl(a-\eta\,v\bigr),
   \]

   where `ΠΔ` is the Euclidean projection onto the simplex
   (implemented with the classic O(n log n) algorithm).

   The step size `η` decays like `1/√t`; the smoothness parameter
   `λ` grows slowly from a small value (broad soft‑max) to a larger
   value (focus on the true maximum).

4. **restarts & bookkeeping** – every few seconds we evaluate the true
   cheap score (`fast_score`).  When a better vector is found it is stored.
   If the algorithm makes no progress for a while we restart from the
   current best with a small random perturbation, thereby escaping possible
   plateaus.

5. **termination** – after the time budget the best vector found is
   returned as a plain Python list.  A final call to the provided
   `evaluate_sequence` gives the official score.

The algorithm explores many different lengths and shapes, but the heavy
lifting is performed by the smooth‑max gradient which uses only two FFTs
per iteration and therefore fits comfortably into the 1000 s budget while
delivering a tighter bound than the previous sub‑gradient method.

Proposed candidate generator for the step‑function minimisation task.

The algorithm combines a diverse random initialisation with a smooth‑max
gradient descent on the simplex (projected momentum).  The gradient of
the soft‑max of the self‑convolution is computed via two FFTs per

def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quickly evaluate the objective 2 * n * max(conv) for a probability
    vector (sum(seq) == 1).  The convolution is performed with an FFT
    padded to a power of two.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # Pad to power of two for speed
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum_i x_i = 1}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # the zero vector – project onto the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} – unnormalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) – unnormalised."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}
    – unnormalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

                            f"[{time.time() - start_time:.1f}s] "
                            f"global improvement: n={n}, score={cur_score:.6f}"
                        )
                # restart if stagnant for >30 s
                if time.time() - last_improve > 30.0:
                    noise = np.random.normal(scale=0.005, size=n)
                    a = best_local + noise
                    a = np.maximum(a, 0.0)
                    if a.sum() == 0.0:
                        a = np.full(n, 1.0 / n, dtype=np.float64)
                    else:
                        a = a / a.sum()
                    v = np.zeros_like(a)
                    last_improve = time.time()
        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Main search loop – several independent runs with different lengths
    # ------------------------------------------------------------------
    # Lengths we are willing to try (they are kept modest to stay fast)
    length_pool = [5000, 8000, 10000, 12000, 15000, 18000, 20000]

    # Continue launching new runs until the overall deadline is reached
    while time.time() < deadline - 5.0:   # keep a few seconds for final eval
        # choose a length (biased a little towards larger sizes)
        n = random.choices(length_pool, weights=[1, 1, 2, 2, 1, 1, 1])[0]

        # pick a random generating family
        family = random.choice(["power", "exp", "twosided"])

        if family == "power":
            p = random.uniform(0.40, 0.75)
            seq0 = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            seq0 = generate_exponential(n, lam)
        else:  # "twosided"
            p_l = random.uniform(0.40, 0.75)
            p_r = random.uniform(0.40, 0.75)
            w_l = random.uniform(0.3, 3.0)
            w_r = random.uniform(0.3, 3.0)
            seq0 = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        seq0 = np.maximum(seq0, 0.0)
        s = seq0.sum()
        if s < 1e-12:
            seq0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            seq0 = seq0 / s

        # allocate a modest time slice for this particular run
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining * 0.3)   # at most 30 s, or 30 % of what remains
        if slice_time < 5.0:
            # too little time left – break out and go to final evaluation
            break

        _, _ = optimise_one_run(n, seq0, slice_time)

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 10000
        n = 10000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]  # noqa: F821
    print(f"Finished. Official score = {final_score:.6f}")
    return final_seq
