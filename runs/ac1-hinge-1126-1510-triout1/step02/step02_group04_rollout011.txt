"""
Propose a new step‑function sequence.

The optimizer works in three phases:

1.  **Smooth‑max Adam** –  the sequence `a` (lying on the simplex Σa=1)
    is represented as a soft‑max of free parameters `v`.  The objective
    `2·n·max(a*a)` (the original bound under the normalisation
    Σa=√(2n)) is approximated by a smooth log‑sum‑exp of the convolution
    entries.  Adam (with bias‑corrected moments) updates `v`.  The
    smoothing factor `β` is increased exponentially from a modest value
    (≈5) to a large value (≈500) so that the optimisation finally sees
    the true max.

2.  **Extended runs & many restarts** –  the above optimisation is
    executed from several diverse initial‑vectors (previous best,
    uniform, Dirichlet draws, power‑law and exponential shapes).  Each
    run is allowed a large number of iterations (up to 30 000) limited
    only by the remaining time budget.

3.  **Local hill‑climbing** –  after the Adam phase the best sequence is
    further refined by random mass transfers between two positions.
    Each candidate is evaluated with an FFT‑based convolution; improvements
    are kept.  The step size is gradually reduced, and a simple simulated
    annealing acceptance rule allows occasional uphill moves.

All operations respect the evaluation routine’s clipping rules
(`0 ≤ height ≤ 1000`).  The final sequence is returned scaled to the
required sum Σa = √(2 n) so that `evaluate_sequence` returns the true
max‑convolution value.
"""

import time
import math
import numpy as np


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (length = 2·len(a)‑1)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.fft.ifft(fft_a * fft_a).real[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Full correlation w ⋆ a_rev (used for the gradient)."""
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.fft.ifft(fft_w * fft_rev).real
    return corr

                     budget_s: float = 1000,
                     **kwargs) -> list[float]:

    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The function respects the 1000 s time budget.
    """

    start_time = time.time()
    # leave a few seconds margin for the outer system
    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------------------
    # Length of the candidate (use the existing construction if present)
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 6000  # reasonable default

    target_sum = math.sqrt(2.0 * n)                # normalisation used by the reference
    print(f"--- Optimising a sequence of length n={n} (target sum={target_sum:.3f}) ---")

    # pre‑compute FFT lengths (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())      # for a⋆a
    L_corr = 1 << ((3 * n - 2).bit_length())      # for correlation with w

    # -------------------------------------------------------------
    # Helper: objective under the unit‑simplex (i.e. Σa = 1)
    def _objective_normed(a: np.ndarray) -> float:
        conv = _conv_full(a, L_conv)
        return 2.0 * n * conv.max()                # = 2·n·max(conv) because Σa = 1

    # -------------------------------------------------------------
    # Build a set of starting points on the simplex (all sum to 1)
    start_vectors = []

    print(f"--- Finished optimisation: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
