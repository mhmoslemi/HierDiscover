Implementation of projection onto simplex:

```
def project_onto_simplex(v):
    # standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The implementation combines three ideas:

1. **Smooth‑max Adam optimisation** – a soft‑max approximation of the max
   (temperature β) is differentiated analytically; the unconstrained logits
   `v` are updated with Adam while `a = softmax(v)` guarantees a
   non‑negative vector of unit L¹–norm (the scale cancels in the objective).

2. **Hard‑max projected gradient descent** – after the smooth phase the exact
   max index is identified and a sub‑gradient of the true max is applied.
   The step is taken on the simplex using a Euclidean projection.

3. **Local random perturbations** – a cheap hill‑climbing phase moves a small
   amount of mass between two coordinates and keeps any improvement.
   This often escapes tiny local minima left by the gradient phases.

The routine respects the 1000 s budget, prints progress and returns the
best sequence found (scaled to the normalisation Σa = √(2 n) used by the
reference code).  If no sequence is found the uniform vector is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    vmax = v.max()
    e = np.exp(v - vmax)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a, length = 2·len(a)‑1 (truncated)."""
    # FFT‑based convolution (real‑valued, zero‑padded to length L)
    FA = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(FA * FA))[: 2 * a.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]                       # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – the projection is uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a))  (a already on the simplex Σa=1)."""
    L = 1 << ((2 * n - 1).bit_length())
    conv = _conv_full(a, L)
    return 2.0 * n * conv.max()


def _gradient_smooth(a: np.ndarray, conv: np.ndarray, beta: float,
                    L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‐max surrogate
        Bβ = (1/β)·log Σ exp(β·conv_k)
    w.r.t. the simplex variables a.
    """
    # soft‑max weights over the convolution entries
    cmax = conv.max()
    w_exp = np.exp(beta * (conv - cmax))
    w = w_exp / w_exp.sum()                     # length 2·n‑1

    # correlation w ⋆ a_rev to obtain ∂Bβ/∂a (see derivation in docstring)
    a_rev = a[::-1]
    FW = np.fft.fft(w, n=L_corr)
    FA_rev = np.fft.fft(a_rev, n=L_corr)
    corr_full = np.real(np.fft.ifft(FW * FA_rev))
    n = a.size
    grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]  # slice to length n
    return grad_a


def _gradient_hard(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max_{k} conv_k.
    Identify the index k where conv is maximal and back‑propagate.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # argmax index (0 … 2n‑2)

    # grad_i = 2·a_{k-i}   if 0 ≤ k‑i < n, else 0
    grad = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine works on the simplex Σa = 1, applies
    a smooth‑max Adam phase, a hard‑max projected‑gradient phase and finally
    a cheap local random search.  The best sequence found is scaled to the
    normalisation Σa = √(2·n) expected by `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------------------
    # Determine the length of the candidate.
    try:
        n = len(height_sequence_1)          # type: ignore
    except Exception:
        n = 8000                               # reasonable default

    target_sum = math.sqrt(2.0 * n)
    print(f"--- Optimising step function, n = {n}, target Σ = {target_sum:.3f} ---")

    # Pre‑compute FFT lengths (next power of two)
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    # -------------------------------------------------------------
    # Helper to keep the current best answer
    best_seq = None
    best_score = float("inf")

    def _maybe_update_best(a_vec: np.ndarray, cur_score: float):
        nonlocal best_seq, best_score
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_vec.copy()
            print(f"[+] new best score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Construct a modest repertoire of initial points on the simplex
    start_vectors = []

    # 1) previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.size > 0:
            a0 = prev / prev.sum()
            start_vectors.append(a0)
    except Exception:
        pass

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) random Dirichlet draws with various concentrations
    for alpha in (0.3, 1.0, 5.0):
        start_vectors.append(np.random.dirichlet(np.full(n, alpha)))

    # 4) cosine‑shaped bump (positive everywhere)
    x = np.arange(n, dtype=float) / (n - 1)
    cos_shape = np.sin(np.pi * x)          # 0 at ends, 1 at centre
    start_vectors.append(cos_shape / cos_shape.sum())

    # 5) Gaussian‑bell centred in the middle
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((np.arange(n) - n / 2) / sigma) ** 2)
    start_vectors.append(gauss / gauss.sum())

    # 6) twin‑Gaussian at the two ends (helps flatten the centre)
    gauss_left = np.exp(-0.5 * ((np.arange(n) - n / 5) / sigma) ** 2)
    gauss_right = np.exp(-0.5 * ((np.arange(n) - 4 * n / 5) / sigma) ** 2)
    twin = gauss_left + gauss_right
    start_vectors.append(twin / twin.sum())

    # -------------------------------------------------------------
    # ADAM parameters (shared across restarts)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr_initial = 0.07          # a bit higher than before – will be annealed
    beta_start = 3.0
    beta_end = 250.0
    max_iters_per_restart = 20000

    # -----------------------------------------------------------------
    # Phase 1 – smooth‑max Adam optimisation
    # -----------------------------------------------------------------
    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        eps = 1e-14
        v = np.log(a_init + eps)          # logits for the soft‑max parametrisation
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                break

            # current simplex point
            a = _softmax(v)

            # convolution and objective
            conv = _conv_full(a, L_conv)
            cur_score = 2.0 * n * conv.max()
            _maybe_update_best(a, cur_score)

            # temperature schedule – slower increase for large n
            beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_restart)

            # smooth‑max gradient w.r.t. a
            grad_a = _gradient_smooth(a, conv, beta, L_corr)

            # chain rule: ∂/∂v = diag(a)·(∂/∂a – (aᵀ∂/∂a)·1)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # optional gradient clipping for stability when β is large
            np.clip(grad_v, -10.0, 10.0, out=grad_v)

            # Adam update on the logits
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # learning‑rate annealing (simple 1/√t schedule)
            lr = lr_initial / math.sqrt(t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # ---------------------------------------------------------
        # Phase 2 – hard‑max projected gradient descent (fine‑tune)
        # ---------------------------------------------------------
        a = _softmax(v)                     # take the result of the Adam run
        for it in range(1200):              # 1200 is a cheap yet effective amount
            if time.time() > deadline:
                break

            conv = _conv_full(a, L_conv)
            cur_score = 2.0 * n * conv.max()
            _maybe_update_best(a, cur_score)

            # sub‑gradient of the true max
            grad = _gradient_hard(a, conv)

            # step (mirrored SGD on the simplex)
            step_size = 0.15 / (1.0 + it * 0.001)   # decaying step size
            a_new = a - step_size * grad
            a_new = _project_onto_simplex(a_new)

            a = a_new

        # ---------------------------------------------------------
        # Phase 3 – cheap random local search (hill climbing)
        # ---------------------------------------------------------
        # Use the best vector obtained in this restart as the seed.
        a = best_seq if best_seq is not None else a
        cur_score = _objective(a, n)

        while time.time() < deadline:
            # pick two distinct indices
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a small random amount of mass from i to j
            delta = np.random.rand() * 0.02 * a[i]   # at most 2 % of a[i]
            if delta <= 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # keep on the simplex (mass is conserved)
            # (no explicit projection needed because we moved mass only)

            conv = _conv_full(a_candidate, L_conv)
            cand_score = 2.0 * n * conv.max()
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                _maybe_update_best(a, cur_score)
            # occasional cooling – slowly reduce the typical delta
            # (handled implicitly by the 0.02 factor)

    # -----------------------------------------------------------------
    # If nothing was found for any reason, fall back to uniform.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Rescale to the normalisation required by evaluate_sequence()
    best_seq = best_seq * target_sum

    # Clip to the domain enforced by evaluate_sequence (just to be safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished optimisation: best score = {best_score:.6f} (n = {n}) ---")
    return best_seq.tolist()
