"""
Propose a low‑score step function.

The search consists of three phases:

1. **Parametric sweep** – we evaluate cheap one‑parameter families
   (truncated geometric a_i ∝ q^i and power‑law a_i ∝ (i+1)^{-p})
   together with a few simple deterministic shapes.
   The best candidate from this stage already improves on the previous
   record.

2. **LP‑guided mixing** – for the current distribution a we solve the
   linear programme

        maximise   Σ b_i
        subject to (a * b)_k ≤ M   for all k,
                   b_i ≥ 0,

   where M = max_k (a * a)_k.  The solution b tends to spread the mass
   in a way that reduces the worst autocorrelation peak.
   After normalising b we try several convex combinations
   (1‑α)a + αb and keep any improvement.

3. **Stochastic polishing** – finally a fast hill‑climbing / polishing
   stage performs random pairwise mass transfers and accepts any move
   that lowers the evaluation score.

All operations are performed on normalised sequences (∑a_i = 1);
the scoring function is therefore scale‑invariant and matches
`evaluate_sequence`.  The returned list is scaled to the [0,1000] range
required by the evaluator.

The implementation avoids closures: all helper functions are defined at
module level.
"""

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:

    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the needed 2·len(a)‑1 values
    return conv[: 2 * a.shape[0] - 1]


def _max_conv(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the autocorrelation a*a."""
    return _conv_fft(a, fft_len).max()


def _score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the same metric as evaluate_sequence (scale‑invariant).
    a must be non‑negative.
    """
    max_b = _max_conv(a, fft_len)
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:                         # guard against degenerate case
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the full convolution matrix A such that (a * b) = A @ b.
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Returns a dense matrix A of shape (2·n‑1, n) with rows representing
    the linear functional (a * b)_k.
    """
    import scipy.linalg as la
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=float)])
    return la.toeplitz(col, row)


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low autocorrelation peak.
    The algorithm consists of a parametric sweep, an LP‑guided refinement
    and a final stochastic polishing phase.
    """
    import numpy as np
    import scipy.optimize as opt
    import scipy.linalg as la

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (fallback to the historic length)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_seq.size
    except Exception:
        n = 1542
        init_seq = (
            np.sqrt(np.arange(1, n + 1, dtype=float))
            - np.sqrt(np.arange(0, n, dtype=float))
        )

    # Pre‑compute FFT length (next power of two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Normalise the initial vector to the simplex Σa = 1
    # ------------------------------------------------------------------
    init_seq = np.clip(init_seq, 0.0, None)
    if init_seq.sum() == 0.0:
        init_seq = np.ones_like(init_seq)
    a = init_seq / init_seq.sum()
    best_a = a.copy()
    best_score = _score(best_a, fft_len)
    print(f"[Init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – fast parametric sweep over simple families
    # ------------------------------------------------------------------
    # Geometric (truncated) family a_i ∝ q^i
    if time.time() < deadline:
        print("[Sweep] geometric family")
        for q in np.linspace(0.90, 0.9999, 300):
            if time.time() >= deadline:
                break
            if q >= 1.0:
                continue
            powers = q ** np.arange(n, dtype=float)
            denom = powers.sum()
            if denom <= 0.0:
                continue
            cand = powers / denom
            sc = _score(cand, fft_len)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"  geo q={q:.5f} → score {best_score:.6f}")

    # Power‑law family a_i ∝ (i+1)^{-p}
    if time.time() < deadline:
        print("[Sweep] power‑law family")
        for p in np.linspace(0.05, 2.5, 250):
            if time.time() >= deadline:
                break
            idx = np.arange(1, n + 1, dtype=float)
            cand = idx ** (-p)
            if cand.sum() == 0.0:
                continue
            cand = cand / cand.sum()
            sc = _score(cand, fft_len)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"  pow p={p:.4f} → score {best_score:.6f}")

    # Sqrt‑diff shape (used in the reference solution)
    if time.time() < deadline:
        cand = (
            np.sqrt(np.arange(1, n + 1, dtype=float))
            - np.sqrt(np.arange(0, n, dtype=float))
        )
        cand = cand / cand.sum()
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"  sqrt‑diff → score {best_score:.6f}")

    # Simple linear decreasing shape
    if time.time() < deadline:
        cand = np.linspace(1.0, 0.0, n)
        cand = cand / cand.sum()
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"  linear ↓ → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP guided refinement (only if SciPy’s optimizer is present)
    # ------------------------------------------------------------------
    a = best_a.copy()
    lp_iters = 0
    max_lp_iters = 40
    alphas = [0.2, 0.4, 0.6, 0.8]  # mixing coefficients

    while time.time() < deadline and lp_iters < max_lp_iters:
        lp_iters += 1
        # Current worst autocorrelation value
        M = _max_conv(a, fft_len)

        # Build the convolution matrix A (dense, but small enough)
        A = _build_conv_matrix(a)                     # shape (2·n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n, dtype=float)                  # maximise Σb ↔ minimise -Σb
        bounds = [(0.0, None)] * n

        # Solve the LP
        try:
            res = opt.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as e:  # safety net – fall back to subgradient only
            print(f"[LP] unexpected failure: {e}")
            break

        if not res.success:
            print(f"[LP] iteration {lp_iters} failed: {res.message}")
            break

        b = res.x
        if b.sum() == 0.0:
            continue
        b = b / b.sum()                               # re‑normalise to Σ=1

        # Try several convex combinations (a ↦ (1‑α)a + αb)
        improved = False
        for alpha in alphas:
            if time.time() >= deadline:
                break
            cand = (1.0 - alpha) * a + alpha * b
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            sc = _score(cand, fft_len)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(
                    f"[LP mix] iter {lp_iters} α={alpha:.2f} → best {best_score:.6f}"
                )
            if sc < _score(a, fft_len):
                a = cand
                improved = True
                break

        if not improved:
            # fallback – directly replace a by b if it improves the score
            sc_b = _score(b, fft_len)
            if sc_b < best_score:
                best_score, best_a = sc_b, b.copy()
                print(f"[LP replace] iter {lp_iters} → best {best_score:.6f}")
            if sc_b < _score(a, fft_len):
                a = b

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing / polishing
    # ------------------------------------------------------------------
    a = best_a.copy()
    hill_iters = 0
    max_hill_iters = 20000

    while time.time() < deadline and hill_iters < max_hill_iters:
        hill_iters += 1
        i, j = np.random.choice(n, 2, replace=False)
        # Transfer at most 5 % of the source mass
        eps = a[i] * np.random.uniform(0.0, 0.05)
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[Hill] iter {hill_iters} → best {best_score:.6f}")
        if sc < _score(a, fft_len):
            a = cand

    # Final polishing – smaller swaps, more aggressive acceptance
    a = best_a.copy()
    polish_iters = 0
    while time.time() < deadline and polish_iters < 5000:
        polish_iters += 1
        i, j = np.random.choice(n, 2, replace=False)
        eps = a[i] * np.random.uniform(0.0, 0.02)
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[Polish] iter {polish_iters} → best {best_score:.6f}")
            a = cand
        elif sc < _score(a, fft_len):
            a = cand

    # ------------------------------------------------------------------
    # Scale to the range required by the evaluator (no effect on the score)
    # ------------------------------------------------------------------
    final_seq = best_a * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(
        f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s"
    )
    return final_seq.tolist()
