"""
Propose a low‑score step sequence.

The algorithm proceeds in three stages:

1️⃣  **Coarse family search** – a cheap sweep over several parametric
    families (exponential, power‑law, Gaussian) on the simplex.
    The best candidate is kept.

2️⃣  **Smooth‑max L‑BFGS refinement** – the soft‑max approximation of the
    max‑convolution is minimised using L‑BFGS.  Multiple short restarts
    (random perturbations of the current best) are performed while the
    smoothing parameter is slowly reduced.

3️⃣  **Projected sub‑gradient + greedy hill‑climb** – a true sub‑gradient
    method on the non‑smooth objective finishes the optimisation, and a
    final greedy mass‑transfer hill‑climb polishes the result.

All heavy convolution work is performed with FFTs (O(n log n)).  The
implementation respects the 1000 s budget, checks the time at every
major step and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Map ℝⁿ → probability simplex in a numerically stable way."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    This equals the value returned by ``evaluate_sequence`` up to the
    scaling invariance.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# --------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns
    -------
    f    : float      = 2·n·smooth_max(conv)
    grad : np.ndarray = gradient w.r.t. y (shape n)
    where ``smooth_max`` is a log‑sum‑exp approximation of max(conv).
    """
    a = softmax(y)                               # simplex point
    # -------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ------- smooth approximation of the max -------
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                 # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ------- gradient of smooth max -------
    w = exp_shift / Z                            # ∂smooth_max/∂conv_i
    # g_i = Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# --------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a simple diminishing‑step
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.6
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active indices (tolerance protects against round‑off)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient g = 2 * (mask ⋆ a_rev)  (restricted to first n entries)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        sc = float(2.0 * n * conv.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Greedy hill‑climb (keeps Σa = 1)
# --------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000,
               delta_frac: float = 0.05):
    """
    Simple greedy mass‑transfer hill‑climb.
    Moves a random fraction (≤ delta_frac) of mass from i to j and
    accepts the move if it improves the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to delta_frac of the current mass at i
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

        # brief hill‑climb to polish this L‑BFGS point
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1200,
                                        delta_frac=0.06)

        iteration += 1

        # every 15 iterations launch a short sub‑gradient refinement
        if iteration % 15 == 0 and time.time() < deadline - 5.0:
            sg_deadline = min(deadline, time.time() + 4.0)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[{iteration}] sub‑grad refine score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – long final hill‑climb
    # ---------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=5000,
                                    delta_frac=0.04)

    # ---------------------------------------------------------
    # Return result (as a plain Python list)
    # ---------------------------------------------------------
    final_seq = best_a.tolist()
    # final check – we call the provided evaluate_sequence once just to
    # make sure the output is admissible and to report the true score.
    try:
        final_eval = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluate_sequence -> {final_eval:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence crashed:", exc)

    return final_seq
