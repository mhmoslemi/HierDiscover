import numpy as np

def grad_smooth(a, beta):
    n = len(a)
    b = np.convolve(a,a)
    max_b = b.max()
    w = np.exp(beta*(b-max_b))
    w /= w.sum()
    a_rev = a[::-1]
    L = 1 << ((len(w)+len(a_rev)-1).bit_length())
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(Wf * Af, n=L)[:len(w)+len(a_rev)-1]
    grad = 2 * conv[n-1::-1][:n]
    return grad
```

"""
Gradient‑descent optimiser for the self‑convolution peak.

The algorithm minimises the smooth approximation

    Fβ(a) = (1/β)·log Σk exp(β·(a∗a)k)

which converges to the true peak maxₖ(a∗a)ₖ as β→∞.
For a given β the gradient is
    ∇Fβ(a) = 2·(w ⋆ a) ,   wₖ ∝ exp(β·(a∗a)ₖ)

where “⋆’’ denotes the linear cross‑correlation.
The weights w form a soft‑max over the convolution values, so the
gradient is a dense combination of all active constraints instead of a
single sub‑gradient.  After each gradient step we project the vector onto
the probability simplex {a≥0, Σa=1} using the O(n) algorithm of Duchi et
al., thus keeping the scaling invariant property of the original score.

A continuation schedule slowly raises β (from 0.5 up to ~30) so that the
optimiser first sees a smooth landscape and later focuses on the true
peak.  A simple diminishing step size (∝1/√t) is used; occasional random
jitter restarts the search when progress stalls.  The best sequence seen
before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine (smooth‑max gradient descent)
# ----------------------------------------------------------------------


def smooth_descent(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 30.0,
    step0: float = 0.5,
    stagnation_limit: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Run the smooth‑max gradient descent until `deadline`.

    Returns the best sequence encountered and its (unsmoothed) score
    2·n·max(conv(a,a)) (the quantity returned by evaluate_sequence for
    a normalised vector with sum = 1).
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT sizes (pre‑computed for speed)
    conv_len = 2 * n - 1                      # length of a∗a
    fft_len_conv = 1 << ((conv_len).bit_length())
    # length of w_rev * a in the gradient step: (2n‑1) + n - 1 = 3n‑2
    grad_len = conv_len + n - 1
    fft_len_grad = 1 << ((grad_len).bit_length())

    best_seq = a.copy()
    best_score = np.inf

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ---- full self‑convolution (linear, using FFT) ----
        A = np.fft.rfft(a, n=fft_len_conv)
        b = np.fft.irfft(A * A, n=fft_len_conv)[:conv_len]
        max_b = b.max()

        # current (unsmoothed) score – used for bookkeeping
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[iter {it:6d}] new best score = {cur_score:.6f}")

        # ---- schedule β (smoothness) ----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 20000.0)

        # ---- soft‑max weights over the convolution values ----
        # shift by max_b for numerical stability
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth‑max (2·(w ⋆ a)) ----
        w_rev = w[::-1]                     # reverse the weight vector
        Wf = np.fft.rfft(w_rev, n=fft_len_grad)
        Af = np.fft.rfft(a, n=fft_len_grad)
        conv = np.fft.irfft(Wf * Af, n=fft_len_grad)[:grad_len]
        # grad_i = 2·conv[m‑1‑i] with m = len(w_rev) = conv_len
        m_len = conv_len
        grad = 2.0 * conv[m_len - 1 :: -1][:n]   # slice first n entries

        # ---- diminishing step size (sub‑gradient style) ----
        step = step0 / np.sqrt(it)

        # gradient step followed by projection onto the simplex
        a = project_simplex(a - step * grad)

        # ---- occasional stochastic perturbation to escape plateaus ----
        if it - last_improve > stagnation_limit:
            # restart with a fresh random power‑law seed
            p = np.random.uniform(0.45, 0.70)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = idx ** (-p)
            a /= a.sum()
            last_improve = it
            print(f"[iter {it:6d}] stagnation – re‑initialising a new power‑law")

    return best_seq, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine runs a smooth‑max gradient descent
    (see above) for as many iterations as the time budget allows and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # Choose a working length.  If a previous construction exists we reuse it.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n < 5000:
            n = 8000
    except Exception:
        n = 8000

    # ---- initialise a random power‑law start ----
    p0 = np.random.uniform(0.45, 0.70)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a0 = idx ** (-p0)
    a0 /= a0.sum()

    # Run the optimiser
    best_seq, best_score = smooth_descent(a0, deadline)

    # Fallback – should never be needed, but keeps the contract safe
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a

    # Convert to plain Python list for the external harness
    return best_seq.tolist()
