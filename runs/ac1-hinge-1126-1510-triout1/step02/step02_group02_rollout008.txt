def _adam_optimize(a0, max_iters, deadline, verbose=False):
    n = len(a0)
    L = 1 << ((2*n-1).bit_length())
    phase = np.exp(2j * np.pi * np.arange(L) / L)
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps=1e-8
    lr0 = 0.12 (maybe larger)
    alpha_start, alpha_end = 2.0, 200.0
    best_a = a.copy()
    best_obj = _objective(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        cur_max = conv.max()
        # compute softmax weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()
        fft_w = np.fft.fft(w, n=L)
        fft_a_rev = np.conj(fft_a) * phase
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n-1 : n-1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)
        obj_new = _objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj

def _lp_refine(a_init, deadline, max_iters=5, t0=0.3):
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = _objective(a)
    t = t0
    for k in range(max_iters):
        if time.time() > deadline:
            break
        M = _conv_max(a)
        # Build A_ub
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for idx in range(rows):
            i_start = max(0, idx - (n-1))
            i_end = min(idx, n-1)
            i_vals = np.arange(i_start, i_end+1)
            j_vals = idx - i_vals
            A[idx, j_vals] = a[i_vals]
        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n)  # maximize sum b -> minimize -sum
        bounds = [(0, None)] * n
        # solve
        try:
            res = linprog(c=c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("LP solve error:", e)
            break

"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
1. **Initialisation** – if a previous construction (`height_sequence_1`) exists,
   use it, otherwise fall back to a moderate length (≈5000).

2. **Parametric sweep** – generate simple families (power‑law, exponential,
   U‑shaped) and keep the best normalised candidate.

3. **Random‑candidate sweep** – draw many Dirichlet / exponential / power‑law /
   U‑shaped vectors of various lengths.  The best of these becomes the start
   point for the next stage.

4. **Adam descent on a smooth max** – minimise the soft‑max approximation of
   `max(conv(a,a))` while staying on the probability simplex (Euclidean
   projection).  The schedule for the smoothing parameter `α` grows from a
   small to a large value so that the optimiser first sees a smooth landscape
   and then focuses on the true supremum norm.

5. **Linear‑programming refinement** – given the current vector `a`,
   solve a linear programme that maximises `∑ b_j` under the constraints
   `conv(a,b) ≤ M` where `M = max(conv(a,a))`.  Mix the solution `b` back
   into `a` (convex combination) and repeat a few times.  This is the step
   described in the reference iteration and often reduces the peak
   convolution further.

6. **Local hill‑climbing** – cheap random mass transfers between entries,
   accepting only improvements.  Performed until the time budget is exhausted.

All phases check the global deadline (1000 s – 5 s safety margin).  The final
sequence is scaled to the normalisation used by `evaluate_sequence`
(`sum = sqrt(2·n)`) and returned as a plain Python list of floats.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# optional import – LP refinement works only if SciPy is available
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:  # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Return max(conv(a, a)) assuming a sums to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective = 2·n·max(conv(a,a)) for a on the simplex (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return float('inf')
    a = a / s
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, max_iters: int,
                   deadline: float, verbose: bool = False):
    """
    Adam optimisation of the smooth‑max of the self‑convolution.
    Returns the best vector found (still on the simplex) and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    alpha_start, alpha_end = 4.0, 150.0

    best_a = a.copy()
    best_obj = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()

        # gradient of smooth‑max (2 * correlation)
        fft_w = np.fft.fft(w, n=L)
        fft_a_rev = np.conj(fft_a) * phase
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        obj_new = _objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"Adam iter {it:5d}: improved obj={obj_new:.8f}")

        a = a_new

    return best_a, best_obj

def _lp_refine(a0: np.ndarray, deadline: float,
               max_iters: int = 5, t0: float = 0.3):
    """
    Linear‑programming refinement described in the reference paper.
    Returns a possibly improved vector and its objective.
    """
    if not _HAS_LP:
        return a0, _objective(a0)

    a = a0.copy()
    n = a.shape[0]

    # LP becomes very costly for large n; fall back for n > 2500
    if n > 2500:
        return a, _objective(a)

    best_a = a.copy()
    best_obj = _objective(a)
    t = t0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        M = _conv_max(a)

        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_vals = np.arange(i_start, i_end + 1)
            j_vals = k - i_vals
            A[k, j_vals] = a[i_vals]

        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n)                     # maximise sum(b)  →  minimise -sum(b)
        bounds = [(0, None)] * n

        try:
            res = linprog(c=c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs')
        except Exception as e:               # pragma: no cover
            print("LP solve error:", e)
            break

        if not res.success:
            # no feasible direction – stop refining
            break

        b_opt = res.x
        if b_opt.sum() == 0:
            break

        b_opt = b_opt / b_opt.sum()
        a = (1.0 - t) * a + t * b_opt
        a = a / a.sum()                     # protect against numerical drift

        obj = _objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"LP iter {it}: improved obj={obj:.8f}")

        t *= 0.5   # decreasing step size

    return best_a, best_obj

def _local_hill_climb(a0: np.ndarray, max_moves: int, deadline: float):
    """
    Cheap random mass transfers, accepting only improvements.
    Returns the best vector found and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _objective(a)

    moves = 0
    while moves < max_moves and time.time() < deadline:

        a_new[target] += delta

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        moves += 1
