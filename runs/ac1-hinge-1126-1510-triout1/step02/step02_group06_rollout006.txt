"""
Propose a low‑score height sequence using a smooth‑max gradient descent.
The algorithm works on the simplex (non‑negative entries that sum to 1)
and repeatedly reduces a smooth approximation of max(conv(a,a))
by projected gradient steps computed with FFTs.
A softmax (log‑sum‑exp) weighting of the convolution entries yields a
differentiable surrogate for the peak; the weighting sharpens during the
run (α grows from 10 to ~200).  The gradient of this surrogate is
computed via an FFT‑based convolution, then the iterate is projected back
onto the probability simplex.  Several random restarts with different
lengths are performed until the time budget (≈1000 s) is exhausted.
The best sequence found is finally validated with `evaluate_sequence`.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2).
    The sequence is assumed to be non‑negative.  The function is scale‑
    invariant; we simply normalise to sum = 1 to avoid overflow.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    a = seq / s
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    m = float(conv.max())
    return 2.0 * n * m  # because sum(a)=1


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The method combines several random restarts,
    a smooth‑max surrogate for the peak of the self‑convolution and
    projected gradient descent on the simplex (FFT‑based for speed).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Determine a set of candidate lengths
    # ------------------------------------------------------------------
    candidate_lengths = [2000, 3000, 5000, 7000, 9000, 11000]
    # Trim lengths that are excessively large for the remaining budget
    candidate_lengths = [l for l in candidate_lengths if l > 0]

    # ------------------------------------------------------------------
    # 1️⃣ Global best holder (will be updated throughout the search)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣ Helper to obtain a starting sequence for a given length
    # ------------------------------------------------------------------
    def get_initial_seq(n: int) -> np.ndarray:
        """Return a non‑negative sequence of length n, sum = 1."""
        # Try to reuse the previous global solution if possible
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
            if prev.size == n:
                seq = prev.copy()
            elif prev.size > n:
                seq = prev[:n].copy()
            else:
                # pad with zeros and renormalise
                seq = np.concatenate([prev, np.zeros(n - prev.size, dtype=np.float64)])
        except Exception:
            # fallback: a modest power‑law
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-0.55)

        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            seq = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            seq = seq / seq.sum()
        return seq.astype(np.float64)

    # ------------------------------------------------------------------
    # 3️⃣ Main optimisation loop (random restarts with different lengths)
    # ------------------------------------------------------------------
    run_counter = 0
    while time.time() < deadline - 5.0:  # keep a few seconds for final check
        run_counter += 1

        # pick a length (biased towards larger ones)
        n = random.choice(candidate_lengths)

        # initialise a sequence (often a perturbed version of the global best)
        if best_seq is not None and random.random() < 0.6:
            # adapt the current best to the new length
            if best_seq.size == n:
                a = best_seq.copy()
            elif best_seq.size > n:
                a = best_seq[:n].copy()
            else:
                a = np.concatenate([best_seq, np.zeros(n - best_seq.size)])
            # tiny random jitter to escape shallow plateaus
            a += 0.001 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            a = get_initial_seq(n)

        # pre‑compute FFT length (large enough for all convolutions we need)
        L_fft = next_pow2(3 * n)  # 2n‑1 + n‑1 ≤ 3n‑2

        # hyper‑parameters for the smooth‑max surrogate
        eta0 = 0.5           # base step size
        alpha_start = 10.0   # start of softmax sharpness
        alpha_max = 200.0    # final sharpness
        max_iters = 20000    # safety ceiling (actual loop stops by time)

        iteration = 0
        while time.time() < deadline - 2.0:
            iteration += 1

            # ---- self‑convolution (FFT) ---------------------------------
            fa = np.fft.rfft(a, n=L_fft)
            conv_full = np.fft.irfft(fa * fa, n=L_fft)[: 2 * n - 1]
            max_conv = float(conv_full.max())

            # ---- smooth‑max (log‑sum‑exp) weight -------------------------
            # α grows linearly for the first few thousand steps
            alpha = alpha_start + (alpha_max - alpha_start) * min(1.0, iteration / 5000.0)
            # stability: shift by max to keep exponentials ≤ 1
            shift = max_conv
            exp_terms = np.exp(alpha * (conv_full - shift))
            weights = exp_terms / exp_terms.sum()  # softmax over all convolution entries

            # ---- gradient of the smooth‑max -----------------------------
            # gradient_i = 2 * Σ_k weights_k * a_{k-i}
            # This is a convolution of weights with the reversed a.
            a_rev = a[::-1]                     # length n
            fw = np.fft.rfft(weights, n=L_fft)   # length L_fft
            fa_rev = np.fft.rfft(a_rev, n=L_fft)
            grad_full = np.fft.irfft(fw * fa_rev, n=L_fft)
            # extract the slice that corresponds to indices 0 … n‑1
            grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

            # ---- projected gradient step -------------------------------
            eta = eta0 / np.sqrt(iteration)     # diminishing step size
            a_candidate = a - eta * grad
            a = project_simplex(a_candidate)    # stay on the simplex

            # ---- occasional evaluation ----------------------------------
            if iteration % 200 == 0 or iteration == 1:
                cur_score = 2.0 * n * max_conv  # sum(a)=1 ⇒ exact objective value
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:.1f}s] run={run_counter}, n={n}, "
                        f"iter={iteration}, new best={best_score:.6f}"
                    )

            # ---- early exit from inner loop if stagnation detected -----
            if iteration >= max_iters:
                break

    # ------------------------------------------------------------------
    # 4️⃣ Final verification with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple uniform sequence (should never happen)
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]  # noqa: F821
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
