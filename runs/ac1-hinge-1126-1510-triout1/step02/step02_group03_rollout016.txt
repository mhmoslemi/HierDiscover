"""
Projected Adam‑style gradient descent with soft‑max smoothing of the
autocorrelation peak.

The objective to minimise is 2·n·max(conv(a,a)) (the denominator is fixed
to 1 by projecting onto the probability simplex).  At each iteration we
compute the full convolution b = a*a, build a smooth approximation of the
max using a soft‑max weight vector w (β is slowly increased), and obtain
the gradient

    ∇ = 2 * Σ_j a_j w_{i+j}  (cross‑correlation of a and w)

via a single FFT‑based convolution.  An Adam update is performed, the
result is projected onto the simplex, and the best sequence seen so far is
kept.  Several random restarts (different initial shapes) are run within
the time budget, followed by a cheap local polishing phase that swaps
small amounts of mass between two positions.  The final sequence is
scaled to the [0,1000] range required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution using FFT (real input → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Convolution a*a using a pre‑computed FFT size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Helper: generic convolution of two real vectors using FFT
# ----------------------------------------------------------------------
def _conv_fft_generic(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of x and y (real) using the given FFT size."""
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    # length = len(x) + len(y) - 1
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Gradient of the (soft‑max) peak w.r.t. the coefficient vector
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          size_grad: int, m: int) -> np.ndarray:
    """
    Returns ∇ = 2 * Σ_j a_j w_{i+j}  for all i.
    Implemented as a convolution of a with w reversed.
    """
    # reverse w
    w_rev = w[::-1]
    # linear convolution of a (len n) with w_rev (len m)
    conv_full = _conv_fft_generic(a, w_rev, size_grad)
    # g[i] = conv_full[m‑1‑i]   (i = 0..n‑1)
    g = conv_full[m - 1::-1][: a.shape[0]]
    return 2.0 * g


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function using a projected Adam‑style
    gradient descent with a soft‑max surrogate for the peak, multiple
    random restarts and a final polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # Determine length of the sequence (reuse previous construction if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1542                     # default length used earlier

    # ------------------------------------------------------------------
    # Pre‑computed FFT sizes
    #   conv(a,a) needs size ≥ 2·n‑1
    #   conv(a,w_rev) needs size ≥ (n + (2·n‑1) - 1) = 3·n‑2
    # ------------------------------------------------------------------
    size_conv = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((3 * n - 2).bit_length())
    m = 2 * n - 1                     # length of the full autocorrelation vector

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    max_iter_per_restart = 30000      # generous budget per restart
    step0 = 0.5                       # initial step size (diminishing as 1/√t)
    beta_start = 5.0                  # start of soft‑max temperature
    beta_end = 30.0                   # final temperature
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    adam_eps = 1e-8

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ----- Choose an initial shape ---------------------------------
        init_mode = np.random.choice(
            ["sqrtdiff", "power", "random", "exp", "tri", "prev"]
        )
        if init_mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif init_mode == "power":
            p = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif init_mode == "exp":
            lam = np.random.uniform(0.001, 0.02)
            a = np.exp(-lam * np.arange(n, dtype=float))
        elif init_mode == "tri":
            # rising then falling linear ramp
            a = np.concatenate([np.linspace(0, 1, n // 2 + 1),
                                np.linspace(1, 0, n - n // 2)[1:]])
        elif init_mode == "prev":
            # use the previously found construction if it exists
            try:
                a = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            except Exception:
                a = np.random.rand(n)
        else:  # "random"
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n, dtype=float)
        a = a / a.sum()          # project onto the simplex (sum = 1)

        # ----- Adam state ------------------------------------------------
        m_adam = np.zeros_like(a)
        v_adam = np.zeros_like(a)

        # ----- Main optimisation loop for this restart -------------------
        for t in range(1, max_iter_per_restart + 1):
            if time.time() >= deadline:
                break

            # 1) full autocorrelation (a*a)
            b = _conv_fft(a, size_conv)                # length 2n‑1
            M = b.max()                                 # current peak

            # 2) soft‑max weights over the autocorrelation entries
            beta = beta_start + (beta_end - beta_start) * (t / max_iter_per_restart)
            # shift by the maximum for numerical stability
            exp_term = np.exp(beta * (b - M))
            w = exp_term / exp_term.sum()               # w sums to 1

            # 3) gradient of the soft‑max surrogate
            g = _gradient_from_weights(a, w, size_grad, m)

            # 4) Adam update (biased‑corrected)
            m_adam = adam_beta1 * m_adam + (1.0 - adam_beta1) * g
            v_adam = adam_beta2 * v_adam + (1.0 - adam_beta2) * (g * g)

            m_hat = m_adam / (1.0 - adam_beta1 ** t)
            v_hat = v_adam / (1.0 - adam_beta2 ** t)

            # 5) diminishing step (combined with Adam's adaptive scaling)
            step = step0 / math.sqrt(t)
            a_new = a - step * (m_hat / (np.sqrt(v_hat) + adam_eps))

            # 6) projection back onto the simplex (non‑negative, sum = 1)
            a_new = _project_to_simplex(a_new)

            # 7) cheap internal score (exactly the one returned by evaluate_sequence)
            score = 2.0 * n * M
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart}] iter {t:05d} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            a = a_new

        # ----- Polishing phase (random small mass transfers) ----------
        if best_seq is not None and time.time() < deadline:
            a_local = best_seq.copy()
            # keep polishing until the deadline approaches
            while time.time() < deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a small fraction from i to j
                delta = min(a_local[i], a_local[i] * np.random.uniform(0.0, 0.05))
                if delta <= 0.0:
                    continue
                cand = a_local.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_to_simplex(cand)
                # internal score (no extra call to evaluate_sequence)
                b_cand = _conv_fft(cand, size_conv)
                s = 2.0 * n * b_cand.max()
                if s < best_score:
                    best_score = s
                    best_seq = cand.copy()
                    a_local = cand
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # ----------------------------------------------------------------
        # Stop if we are out of time
        # ----------------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fb = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        best_seq = a_fb / a_fb.sum()

    # ------------------------------------------------------------------
    # Scale to the [0, 1000] range required by the evaluator (no effect on score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()
