"""
propose_candidate
-----------------

Searches for a non‑negative step function with a small autocorrelation
peak.  The algorithm follows a *continuation* strategy:

1.  The true objective `max(conv(a,a))` is approximated by a smooth
    soft‑max:  L_α(a)= (1/α)·log Σ_k exp(α·(a*a)_k).
    As α→∞ the smooth objective converges to the L∞‑norm.

2.  For a current smoothness parameter α we minimise L_α by a (projected)
    gradient step on the probability simplex (the coefficients are kept
    non‑negative and sum to 1).  The gradient of L_α is
        ∇L_α = 2·cross‑correlation( w , a )
    where w_k = exp(α·(a*a)_k) / Σ_j exp(α·(a*a)_j).

3.  α is increased linearly during the run, turning the problem gradually
    into the true max‑problem, while the step size η is decreased like
    1/√t (standard diminishing‑step sub‑gradient schedule).  Both an
    additive (Euclidean projection) and a multiplicative (exponentiated
    gradient) update are tried each iteration and the better one is kept.

4.  Several random perturbations of the best solution are performed as
    cheap “restarts”.  After the main phase a short polishing stage swaps
    small amounts of mass between two random positions (simulated‑annealing
    style) to squeeze out any remaining improvement.

The whole procedure runs until the supplied time budget (default 1000 s)
is exhausted and finally returns the best sequence found, scaled to the
[0, 1000] range required by `evaluate_sequence`.  The evaluation itself is
scale‑invariant, so the optimisation works on the unit‑simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution using the FFT (real → real)
# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using an FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: n + m - 1]

# ----------------------------------------------------------------------

    method, projected (and exponentiated) gradient descent, random restarts,

    # Determine the length of the sequence.
