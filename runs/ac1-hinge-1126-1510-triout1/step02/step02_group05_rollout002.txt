def run_eg(a0, max_iters):
    a = a0.copy()
    n = a.shape[0]
    fft_size = 1 << ((2*n-1).bit_length())
    best_seq = a.copy()
    best_score = compute_score(a, n, fft_size)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # conv
        A = np.fft.rfft(a, n=fft_size)
        b = np.fft.irfft(A * A, n=fft_size)[:2*n-1]
        max_b = b.max()
        score = 2.0 * n * max_b  # because sum(a)=1
        if score < best_score:
            best_score = score; best_seq = a.copy(); print...
        # gradient
        # find active peaks
        # tolerance maybe 1e-12
        idx_peaks = np.where(b >= max_b - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in idx_peaks:
            i_low = max(0, k-(n-1))
            i_high = min(k, n-1)
            i_range = np.arange(i_low, i_high+1)
            j_range = k - i_range
            grad[i_range] += 2.0 * a[j_range]
        # step size schedule
        eta = eta0 / np.sqrt(it)  # eta0 e.g., 0.4
        # update via exponentiated gradient:
        # to avoid overflow, subtract max exponent
        update = -eta * grad
        # clip update magnitude: maybe compute max abs, scale if >50
        max_abs = np.max(np.abs(update))
        if max_abs > 50:
            update = update * (50 / max_abs)
        w = np.exp(update)
        a = a * w
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
    return best_seq, best_score

def eg_descent(a0, max_iters, eta0, deadline):
    a = a0.copy()
    n = a.shape[0]
    fft_size = 1 << ((2 * n - 1).bit_length())
    best_seq = a.copy()

    A = np.fft.rfft(a, n=fft_size)

    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    # record start time
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # compute convolution full and max for gradient and score
        A = np.fft.rfft(a, n=fft_size)
        conv_full = np.fft.irfft(A * A, n=fft_size)[:2 * n - 1]
        max_b = conv_full.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[EG it {it}] new best score={score:.6f}")

        # compute gradient w.r.t. active peaks
        grad = compute_gradient(a, conv_full, max_b, tol=1e-12)

        # step size
        eta = eta0 / np.sqrt(it)

        # exponentiated gradient update
        # compute weight vector
        # avoid overflow: clip -eta*grad
        update = -eta * grad
        # clip for stability
        update = np.clip(update, -50, 0)  # ensure exponent <=0
        w = np.exp(update)
        a = a * w

        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

"""
Exponentiated‑gradient (mirror‑descent) optimiser for the self‑convolution peak.

The score to minimise (with sum(a)=1) is
    2 * n * max(conv(a, a))

The method works on the probability simplex:
    a_i ← a_i * exp( -η_t * ∂/∂a_i max(conv) )
    renormalise to sum 1.

A diminishing step size η_t = η0 / sqrt(t) is used.
The gradient of the active peak(s) is
    ∇_i  = Σ_{k∈active} 2 * a_{k-i}   (only where the index is valid).

Multiple restarts with random power‑law seeds, the current best, or uniform
initialisation are performed until the allotted wall‑time expires.  The
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np


def _fft_size_for_n(n: int) -> int:
    """Return the next power‑of‑two size sufficient for linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full(a: np.ndarray, fft_size: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2*n‑1)."""
    A = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(A * A, n=fft_size)
    return conv[: 2 * a.shape[0] - 1]


def _gradient_active_peaks(a: np.ndarray, conv_full: np.ndarray,
                          max_val: float, tol: float = 1e-12) -> np.ndarray:
    """
    Compute the (sub)gradient of the maximal entry of `conv_full`
    with respect to the vector `a`.  All peaks within `tol` of the
    maximum are treated as active.
    """
    n = a.shape[0]
    # indices where the convolution is (almost) maximal
    active = np.where(conv_full >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i = np.arange(i_low, i_high + 1, dtype=np.intp)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _eg_descent(a0: np.ndarray, max_iters: int,
                eta0: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex.

    Returns the best sequence encountered and its score
    (score = 2 * n * max(conv) because sum(a)=1).
    """
    a = a0.copy()
    n = a.shape[0]
    fft_sz = _fft_size_for_n(n)

    # initial score
    conv = _conv_full(a, fft_sz)
    max_b = conv.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    it = 0

    while it < max_iters:
        if time.time() >= deadline:
            break
        it += 1

        # convolution and score
        conv = _conv_full(a, fft_sz)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[EG it {it}] new best score={score:.6f}")

        # gradient of the active peak(s)
        grad = _gradient_active_peaks(a, conv, max_b, tol=1e-12)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (all components stay non‑negative)
        upd = -eta * grad
        # prevent under‑/overflow of exp()
        upd = np.clip(upd, -50.0, 0.0)
        a = a * np.exp(upd)

        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

    return best_seq, best_score


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine performs many random restarts and runs a mirror‑descent
    optimiser (exponentiated gradient) on each restart.  The best sequence
    found before the time budget expires is returned as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be present in the global namespace
        init_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init_seq.ndim != 1:
            raise ValueError
        # Ensure non‑negative entries
        init_seq = np.maximum(init_seq, 0.0)
        n = max(init_seq.size, 4000)   # we like a reasonably large size
        init_seq = init_seq / (init_seq.sum() + 1e-18)
        global_best_seq = init_seq.copy()
        # compute its score (sum is 1)
        fft_sz = _fft_size_for_n(n)
        conv = _conv_full(global_best_seq, fft_sz)
        global_best_score = 2.0 * n * conv.max()
        print(f"[init] starting from existing sequence, score={global_best_score:.6f}")
    except Exception:

        n = 8000
        global_best_seq = None

    # --------------------------------------------------------------
    # Main search loop (multiple restarts)
    # --------------------------------------------------------------
    restart = 0

        # ----------------------------------------------------------

        time_left = deadline - time.time()
        # Empirical estimate: ~1 ms per iteration on the chosen size

        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()
        print("[fallback] using simple power‑law seed")

    # Return a plain Python list as required by the harness
    return global_best_seq.tolist()
