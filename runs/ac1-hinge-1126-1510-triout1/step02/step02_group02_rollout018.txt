"""
Propose a step‑function (non‑negative vector) that yields a low value of

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of the following stages (all respect the 1000 s budget).

1. **Parameter sweep** – generate many simple analytic families
   (uniform, power‑law, exponential, Gaussian, half‑sine, triangular
   and a few “two‑block” shapes).  Each candidate is normalised to
   unit sum and its convolution max is evaluated with a cheap FFT.
   The best candidate found is kept.

2. **Coarse optimisation on a modest size** – the best candidate from
   step 1 is interpolated to a smaller length `n_small` (≈2000) and
   refined by Adam optimisation of a smooth‑max surrogate
   `log‑sum‑exp(α·conv) / α`.  Several random restarts improve the
   chance of escaping local minima.

3. **Upsampling & final Adam** – the refined small‑size vector is
   interpolated to the target length `n` (either the length of the
   supplied global solution or the default 6000) and a short Adam run
   with a larger smoothing parameter finishes the optimisation.

4. **Local random hill‑climbing** – cheap mass‑transfer moves are
   performed until the time budget is exhausted.  Any improvement is
   kept.

The final sequence is scaled to the normalisation used in the reference
code (`sum = sqrt(2·n)`).  All operations stay on the probability simplex
(`a_i ≥ 0`, `Σ a_i = 1`) so the evaluation function is scale‑invariant.

The implementation uses only NumPy and the supplied `evaluate_sequence`
function; it prints progress and returns the best sequence as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max(a * a) via FFT (a is assumed to sum to 1)."""
    fa = np.fft.fft(a, n=fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for a convolution vector."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    fft_len: int,
    max_iters: int = 20000,
    lr: float = 0.07,
    alpha_start: float = 2.0,
    alpha_end: float = 150.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    time_deadline: float = None,
):
    """
    Adam optimisation of the smooth‑max objective.
    Returns (best_a, best_conv_max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max(a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # phase factor that turns fft(a) into fft(a[::-1])
    phase = np.exp(2j * np.pi * np.arange(fft_len) / fft_len)

    it = 0
    while it < max_iters:
        if time_deadline is not None and time.time() > time_deadline:
            break
        it += 1

        # ----- current convolution -----
        fa = np.fft.fft(a, n=fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]

        # ----- soft‑max weights -----
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # ----- gradient (2 * correlation) -----
        # reverse‑fft of a (no extra FFT)
        fa_rev = np.conj(fa) * phase
        fw = np.fft.fft(w, n=fft_len)
        corr = np.real(np.fft.ifft(fw * fa_rev))
        grad = 2.0 * corr[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- Adam update on the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- record improvement -----
        conv_new = np.real(np.fft.ifft(np.fft.fft(a_new, n=fft_len) *
                                      np.fft.fft(a_new, n=fft_len)))[: 2 * a.shape[0] - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


# ----------------------------------------------------------------------
def _local_random_moves(
    a_start: np.ndarray,
    fft_len: int,
    time_deadline: float,
    max_moves: int = 10_000_000,
):
    """
    Cheap hill‑climbing: repeatedly move a small random fraction of mass
    from one index to another, keep any improvement.
    """
    a = a_start.copy()
    best_obj = _conv_max(a, fft_len)
    n = a.shape[0]

    # pre‑compute probability distribution for source selection (mass‑biased)
    while time.time() < time_deadline and max_moves > 0:
        max_moves -= 1
        i = np.random.choice(n, p=a)          # large entries more likely
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move up to 2 % of the source mass
        delta = np.random.rand() * 0.02 * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        conv_new = np.real(np.fft.ifft(np.fft.fft(a_new, n=fft_len) *
                                      np.fft.fft(a_new, n=fft_len)))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            a = a_new
            # optional: print progress
            print(f"  move improved | obj={obj_new:.8e} | score={2*n*obj_new:.6f}")

    return a, best_obj

    """

    # FFT length – at least 2·n‑1, power of two for speed

    def _maybe_update(candidate_raw: np.ndarray, label: str = ""):

        a = candidate_raw.astype(np.float64)
        if a.sum() == 0:
            return
        a = a / a.sum()
        obj = _conv_max(a, fft_len)
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"  new best ({label}) | obj={obj:.8e} | score={2*n*obj:.6f}")

    # 1a. Uniform (baseline)
    _maybe_update(np.ones(n), "uniform")

    # 1b. Power‑law (both monotone directions)
    exps = np.linspace(0.0, 2.5, 35)
    idx = np.arange(n, dtype=np.float64)
    for p in exps:
        if time.time() > deadline:
            break
        inc = idx ** p
        dec = inc[::-1]
        _maybe_update(inc, f"pow_inc_{p:.2f}")
        _maybe_update(dec, f"pow_dec_{p:.2f}")

    # 1c. Exponential decay / growth
    lambdas = np.linspace(0.0, 6.0, 35)
    for lam in lambdas:
        if time.time() > deadline:
            break
        expd = np.exp(-lam * idx)
        _maybe_update(expd, f"exp_{lam:.2f}")
        _maybe_update(expd[::-1], f"exp_rev_{lam:.2f}")

    # 1d. Gaussian (varying sigma)
    xs = (np.arange(n) - (n - 1) / 2.0) / n
    sigmas = np.linspace(0.05, 0.4, 20)
    for sigma in sigmas:
        if time.time() > deadline:
            break
        gauss = np.exp(-0.5 * (xs / sigma) ** 2)
        _maybe_update(gauss, f"gauss_{sigma:.3f}")

    # 1e. Half‑sine (zero at borders, peak in middle)
    sine = np.sin(np.pi * np.arange(n) / (n - 1))
    _maybe_update(sine, "sine")

    # 1f. Triangular (peak in centre)
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    _maybe_update(tri, "triangular")

    # 1g. Two‑block shape (blocks at both ends, gap in middle)
    # Block length varied from ~0.1 n to 0.45 n
    for frac in np.linspace(0.1, 0.45, 12):
        if time.time() > deadline:
            break
        L = int(frac * n)
        if 2 * L > n:
            continue
        block = np.full(L, 0.5 / L)          # each block carries ½ of the mass
        a = np.zeros(n)
        a[:L] = block
        a[-L:] = block
        _maybe_update(a, f"two_block_{frac:.2f}")

    # ------------------------------------------------------------------
    # If we managed to load a previous solution, inject it
    if init_global is not None:
        _maybe_update(init_global, "global_init")

    # ------------------------------------------------------------------
    # 2. Coarse optimisation on a smaller problem size
    # ---------------------------------------------------
    # Reduce size to accelerate gradient steps while preserving shape.

    # Perform a few random restarts of Adam on the small problem
    restarts = 5
    best_small_obj = _conv_max(best_small, fft_len_small)
    best_small_seq = best_small.copy()

    for r in range(restarts):
        if time.time() > deadline:
            break
        # Random perturbation of the current best (or completely random)
        if r == 0:
            a0 = best_small_seq.copy()
        else:
            # Dirichlet draw with concentration 0.7 → moderate sparsity
            a0 = np.random.dirichlet(0.7 * np.ones(n_small))
        a_opt, obj_opt = _adam_optimize(
            a0,
            fft_len_small,
            max_iters=25000,
            lr=0.07,
            alpha_start=3.0,
            alpha_end=130.0,
            time_deadline=deadline,
        )
        if obj_opt < best_small_obj:
            best_small_obj = obj_opt
            best_small_seq = a_opt.copy()
            print(f"  small‑restart {r} improved | obj={obj_opt:.8e}")

    # ------------------------------------------------------------------

    print(f"--- Phase 2b: upsample & final Adam (n={n}) ---")

    )
    if final_obj < best_obj:
        best_obj = final_obj
        best_seq = a_final.copy()
        print(f"  after upsample‑Adam | obj={final_obj:.8e} | score={2*n*final_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Local random hill‑climbing until the deadline
    print(f"--- Phase 3: random local moves (remaining time {deadline - time.time():.1f}s) ---")
    a_current, obj_current = _local_random_moves(best_seq, fft_len, deadline)

    best_obj = obj_current

    # ------------------------------------------------------------------

    final_seq = (best_seq * target_sum).tolist()
    # One final call to the provided evaluator just for logging

    print(f"--- Finished: best score = {final_score:.6f} (n={n}) ---")
