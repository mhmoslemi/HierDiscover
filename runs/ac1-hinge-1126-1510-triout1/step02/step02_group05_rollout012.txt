--- Implementation details for smoothed gradient descent ---

def smoothed_gradient_descent(a0, max_iters, step0, beta_start, beta_end, deadline):
    a = a0.copy()
    n = a.shape[0]
    # Normalize to simplex
    a = np.maximum(a,0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s

    # Precompute FFT sizes
    conv_a_size = 1 << ((2 * n - 1).bit_length())
    conv_grad_size = 1 << ((3 * n - 2).bit_length())

    best_seq = a.copy()
    best_score = float('inf')

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # self convolution via FFT
        A = np.fft.rfft(a, n=conv_a_size)
        b = np.fft.irfft(A * A, n=conv_a_size)[:2*n-1]
        max_b = b.max()
        # current score
        score = 2.0 * n * max_b   # sum=1
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[sgd iter {it}] new best score={score:.6f}")

        # softmax weights
        beta = beta_start + (beta_end - beta_start) * ((it-1) / (max_iters-1) if max_iters>1 else 1)
        # prevent under/overflow
        # compute w = exp(beta*(b - max_b)), then normalize
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w = w / w_sum

        # gradient: 2 * conv(a, w)
        # compute via FFT
        # size = conv_grad_size = next pow2( n + (2n-1) -1 )
        A = np.fft.rfft(a, n=conv_grad_size)
        W = np.fft.rfft(w, n=conv_grad_size)
        grad_full = np.fft.irfft(A * W, n=conv_grad_size)[:n + (2*n-1) - 1]  # length 3n-2
        grad = 2.0 * grad_full[:n]    # first n entries

        # step size
        step = step0 / np.sqrt(it)

        # update and project
        a = project_simplex(a - step * grad)

    return best_seq, best_score

"""
Hybrid projected sub‑gradient / smoothed‑max optimizer.

We minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

over non‑negative vectors a of length n (the evaluation function is
scale‑invariant, so we keep a on the probability simplex Σa=1).
Two complementary descent schemes are used:

*   Classic sub‑gradient descent that moves only in the direction of the
    active peak of the convolution.

*   Gradient descent on a smooth approximation of the ∞‑norm:
    max(b) ≈ (1/β)·log Σ exp(β·b).  The gradient of this surrogate is a
    weighted sum of the gradients of the individual quadratic terms and
    can be evaluated with a single FFT‑based convolution.

Random restarts, adaptive β schedules and simple step‑size decay are
employed.  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Return the full (non‑circular) convolution of x and y using FFT.
    ``size`` must be at least len(x)+len(y)-1 and preferably a power of two.
    """
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    c = np.fft.irfft(X * Y, n=size)
    return c[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Sub‑gradient descent (active peak only)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        max_iters: int,
                        step0: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the peak of the self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_size = next_pow2(2 * n - 1)

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full self‑convolution
        A = np.fft.rfft(a, n=fft_size)
        b = np.fft.irfft(A * A, n=fft_size)[: 2 * n - 1]
        max_b = b.max()
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best score={score:.6f}")

        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # grad[i] = 2 * a[k-i] for i_low ≤ i ≤ i_high
        grad = np.zeros_like(a)
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k - idx]

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a0: np.ndarray,
                              max_iters: int,
                              step0: float,
                              beta_start: float,
                              beta_end: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the log‑sum‑exp surrogate of the peak.
    The gradient can be written as 2·conv(a, w_rev) where w is the softmax
    over the self‑convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # FFT sizes (powers of two)
    size_a = next_pow2(2 * n - 1)          # for a*a
    size_grad = next_pow2(3 * n - 2)       # for a * w_rev

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- self‑convolution (full) ---
        A = np.fft.rfft(a, n=size_a)
        b = np.fft.irfft(A * A, n=size_a)[: 2 * n - 1]
        max_b = b.max()
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[smooth it {it}] new best score={score:.6f}")

        # --- soft‑max weights over the convolution entries ---
        beta = beta_start + (beta_end - beta_start) * ((it - 1) / (max_iters - 1) if max_iters > 1 else 1.0)
        # stability: subtract max_b
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # --- gradient = 2 * conv(a, w_rev) (first n entries) ---
        w_rev = w[::-1]                     # reverse for the required sum_{j} a_j w_{i+j}
        grad_full = full_conv_fft(a, w_rev, size_grad)          # length = 3n-2
        L = len(w_rev) - 1                                            # = 2n-2
        grad = 2.0 * grad_full[L - np.arange(n)]

        # --- projected update ---
        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score

    """

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    try:

        elif r < 0.80 and global_best_seq is not None:

            s = a0.sum()
            if s == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= s
        else:
            # exponential decay
            base = np.random.uniform(0.95, 0.999)
            a0 = base ** np.arange(n, dtype=np.float64)
            a0 /= a0.sum()

        # ------------------- decide how many iterations we can afford --
        time_left = deadline - time.time()

            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new global best score={score:.6f} "
                f"(iters={it_budget})"
            )

        # occasional step‑size tuning

            subgrad_step0 = max(0.3, subgrad_step0 * 0.9)
            smooth_step0 = max(0.2, smooth_step0 * 0.9)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        if remaining > 10.0:
