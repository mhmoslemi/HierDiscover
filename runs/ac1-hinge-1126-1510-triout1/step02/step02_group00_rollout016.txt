def objective_and_grad_support(y, pos, n, fft_len_conv, fft_len_g, lam_factor):
    a_small = softmax(y) # (s,)
    a = np.zeros(n)
    a[pos] = a_small
    # compute conv via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. a (full)
    w = exp_shift / Z # shape 2n-1
    # compute convolution of a_rev with w to get grad w.r.t. a at each index (as earlier)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = 2.0 * conv_rev[(n-1):(n-1+n)] # shape n
    # now chain through softmax for support entries:
    grad_a_pos = grad_a[pos]  # shape s
    mean_g = np.dot(a_small, grad_a_pos)
    grad_y = a_small * (grad_a_pos - mean_g)  # shape s
    return f, grad_y

"""
Improved search for a low‑peak self‑convolution step sequence.

The idea is to keep only a relatively small set of active positions
(“support”) and optimise the probabilities on that support.
If the support is chosen so that the number of ordered pairs (i,j) with
the same sum i+j is low, the convolution peak can be made very small.
For a given support the objective
    F(a) = 2·n·max_k (a * a)[k]          (sum(a)=1)
is smoothened with a log‑sum‑exp approximation and minimised by L‑BFGS.
After each L‑BFGS run a cheap greedy hill‑climb is applied to polish
the result.  The whole procedure is repeated with many random supports
until the time budget (≈1000 s) runs out.

The returned list is a non‑negative step sequence ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a distribution already normalised to sum=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed‑max objective for a *restricted* support
# ----------------------------------------------------------------------
def obj_grad_support(y: np.ndarray,
                     pos: np.ndarray,
                     n: int,
                     fft_len_conv: int,
                     fft_len_g: int,
                     lam_factor: float = 0.10):
    """
    Returns (f, g) where
    f = 2·n·smooth_max(conv)
    g = gradient w.r.t. the log‑weights y (size = len(pos)).
    The distribution a is zero except on `pos`, where
    a[pos] = softmax(y) and sum(a) = 1.
    """
    # ----- probability on the support -----
    a_small = softmax(y)                     # shape (s,)
    a = np.zeros(n, dtype=np.float64)
    a[pos] = a_small

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth approximation of the max -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. a -----
    w = exp_shift / Z                         # d smooth_max / d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = 2.0 * conv_rev[(n - 1):(n - 1 + n)]   # size n

    # ----- chain rule through the soft‑max (restricted to support) -----
    grad_a_pos = grad_a[pos]                  # size s
    mean_g = np.dot(a_small, grad_a_pos)
    grad_y = a_small * (grad_a_pos - mean_g)  # size s

    return f, grad_y

# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy transfer of a small fraction of mass from one coordinate
    to another while preserving the sum.  Moves are accepted only if they
    improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # take a tiny fraction (≤5 %) of mass from i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The algorithm:
        1.  Randomly sample a support set (≈√(2·n) positions).
        2.  Optimise the probabilities on that support with L‑BFGS
            using a smooth approximation of the max.
        3.  Polish the resulting full‑length vector with a fast
            hill‑climb.
        4.  Repeat until the time budget expires and return the best
            sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Use the length of a previous solution if available; otherwise 2000.
    n = 2000
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            n = min(4000, prev.size)          # keep it moderate
    n = int(n)

    fft_len_conv = next_pow2(2 * n - 1)        # for a*a
    fft_len_g = next_pow2(3 * n - 2)           # for gradient convolutions

    # initialise with a uniform distribution (guaranteed to be feasible)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # How many positions should we keep active?  Rough rule of thumb:
    #   s ≈ c·√(2·n) with 0.6 ≤ c ≤ 1.2 .  This gives enough freedom to
    #   flatten the convolution while keeping the number of ordered
    #   pairs per sum relatively low.
    # ------------------------------------------------------------------
    sqrt_2n = np.sqrt(2 * n)
    s_min = max(30, int(0.6 * sqrt_2n))
    s_max = max(s_min + 1, int(1.2 * sqrt_2n))
    support_sizes = list(range(s_min, s_max + 1, 5))
    rng.shuffle(support_sizes)

    iteration = 0
    while time.time() < deadline and iteration < 300:
        # --------------------------------------------------------------
        # 1) random support
        # --------------------------------------------------------------
        s = rng.choice(support_sizes)
        pos = rng.choice(n, size=s, replace=False)
        pos.sort()

        # --------------------------------------------------------------
        # 2) L‑BFGS on the restricted support
        # --------------------------------------------------------------
        y0 = rng.standard_normal(s)

        res = minimize(fun=obj_grad_support,
                       x0=y0,
                       args=(pos, n, fft_len_conv, fft_len_g, 0.08),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'gtol': 1e-12,
                                'disp': False})

        a_small_opt = softmax(res.x)                 # probabilities on the support
        a_full = np.zeros(n, dtype=np.float64)
        a_full[pos] = a_small_opt
        score = fast_score(a_full, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a_full.copy()
            print(f"[{iteration}] support={s} L‑BFGS score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3) short hill‑climb polishing (few hundred moves)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1000)

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing with a longer hill‑climb (uses the remaining time)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=4000)

    # Convert to plain Python list for the evaluator
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
