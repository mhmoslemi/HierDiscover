#!/usr/bin/env python
"""
Propose a low‑score step sequence.

The search combines three complementary ideas:

1. **Smooth‑max L‑BFGS** – minimise a differentiable log‑sum‑exp
   approximation of the max‑convolution.  Several random restarts and
   a continuation schedule for the smoothing parameter explore the
   landscape quickly.

2. **Exact linear‑programming refinement** – at a current point `a`
   let `M = max(a*a)`.  Solve the LP
       maximise   Σ g_j
       subject to a * g ≤ M   (pointwise)
                 g ≥ 0
   (only the constraints that are active at `M` are kept, i.e. the
   indices where a*a attains its maximum).  The LP gives a direction
   that can increase the total mass without raising the peak of the
   convolution.  After normalising `g` we move a little towards it.

3. **Aggressive hill‑climbing** – repeatedly transfer a random
   fraction (up to 10 %) of mass from one coordinate to another while
   preserving the sum.  Improvements are kept; otherwise the move is
   reverted.  This is cheap and can escape tiny local minima.

The algorithm works on the probability simplex (∑a = 1).  The
evaluation function `evaluate_sequence` is scale‑invariant, so the
final sequence may be returned directly after normalisation.

The whole procedure respects the 1000 s time budget, runs on a
single core, and makes no filesystem or network accesses.
"""
import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summed to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000,
               max_move: float = 0.10):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction (up to max_move) of the mass at i to j.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              M: float,
              fft_len_conv: int,
              n: int,
              deadline: float,
              blend: float = 0.20):
    """
    Perform one LP‑based refinement:
    maximise Σ g_j subject to a * g ≤ M (only active constraints).
    The resulting direction g is blended into a with weight `blend`.
    """
    # compute active indices (tight constraints)
    eps = max(M * 1e-12, 1e-12)
    conv = full_conv(a, fft_len_conv)
    active = np.where(conv >= M - eps)[0]

    # very small active set -> cheap LP
    if active.size == 0:
        return a

    rows = []
    for k in active:
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        row[j] = a[i]
        rows.append(row)

    A = np.stack(rows)                     # shape (m, n)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise Σ g_j

    # solve LP (highs is fast and has no external dependencies)
    lp_res = linprog(c, A_ub=A, b_ub=b,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'disp': False})

    if not lp_res.success:
        return a

    g0 = lp_res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return a

    # Normalise g to the simplex
    g = g0 / sum_g0

    # Blend a towards g
    new_a = (1.0 - blend) * a + blend * g
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a

        The best sequence found (non‑negative, sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Choose problem size and an initial distribution
    # ------------------------------------------------------------------
    max_n = 8000
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).ravel()
        n = min(max(seq0.size, 2), max_n)
        seq0 = seq0[:n]
        seq0[seq0 < 0] = 0.0
        if seq0.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = seq0 / seq0.sum()
    else:
        n = 6000                     # a reasonably large size
        a0 = np.full(n, 1.0 / n)

    # Ensure a valid simplex point
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helpers for scoring
    # ------------------------------------------------------------------
    def get_score(vec):
        """Return (objective, max_conv) for a probability vector."""
        conv = full_conv(vec, fft_len_conv)
        max_c = conv.max()
        return float(2.0 * n * max_c), max_c

    best_a = a0.copy()
    best_score, best_M = get_score(best_a)
    print(f"[init] n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
    mode_list = ['random', 'exp', 'power', 'edge', 'sin', 'double_exp', 'prev']
    iteration = 0

    while time.time() < deadline:
        lam = lam_schedule[iteration % len(lam_schedule)]

        # ------------------------------------------------------------------
        # Pick a starting point for L‑BFGS
        # ------------------------------------------------------------------
        mode = rng.choice(mode_list)
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        else:
            # generate a raw shape and convert to logits
            if mode == 'random':
                a_start = rng.random(n)
            elif mode == 'exp':
                rate = rng.uniform(0.0005, 0.02)
                a_start = np.exp(-rate * np.arange(n))
            elif mode == 'power':
                p = rng.uniform(0.2, 2.5)
                a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
            elif mode == 'edge':
                frac = rng.uniform(0.008, 0.04)
                m = max(1, int(n * frac))
                edge_val = np.sqrt(n / 2.0) / n
                centre_val = 1.0 / np.sqrt(2.0) / n
                a_start = np.full(n, centre_val)
                a_start[:m] = edge_val
                a_start[-m:] = edge_val
            elif mode == 'sin':
                a_start = np.sin(np.pi * (np.arange(n) + 0.5) / n) + 0.1
            elif mode == 'double_exp':
                rate = rng.uniform(0.0008, 0.015)
                left = np.exp(-rate * np.arange(n // 2))
                right = left[::-1]
                a_start = np.concatenate([left, right])
                if a_start.size < n:
                    a_start = np.pad(a_start, (0, n - a_start.size), constant_values=0.0)
            else:
                a_start = np.ones(n)

            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)

        # ------------------------------------------------------------------
        # L‑BFGS on the smoothed max
        # ------------------------------------------------------------------
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-9,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt, M_opt = get_score(a_opt)

        if score_opt < best_score:
            best_score, best_M = score_opt, M_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] L‑BFGS lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Short hill‑climb to polish L‑BFGS result
        # ------------------------------------------------------------------
        hc_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_dead,
                                        max_iter=1500,
                                        max_move=0.10)

        # ------------------------------------------------------------------
        # Occasionally run sub‑gradient refinement (few seconds)
        # ------------------------------------------------------------------
        if iteration % 15 == 0 and time.time() < deadline - 5.0:
            sg_dead = min(deadline, time.time() + 4.0)
            best_a, tmp_score = subgradient_optimize(best_a,
                                                     sg_dead,
                                                     fft_len_conv,
                                                     fft_len_g,
                                                     n)
            if tmp_score < best_score:
                best_score = tmp_score
                print(f"[{iteration}] sub‑gradient refined score={best_score:.6f}")

        # ------------------------------------------------------------------
        # LP‑based refinement – cheap and often reduces the peak
        # ------------------------------------------------------------------
        if iteration % 7 == 0 and time.time() < deadline - 2.0:
            # use the current max (best_M) as the bound
            best_a = lp_refine(best_a, best_M,
                               fft_len_conv, n,
                               deadline,
                               blend=0.18)
            best_score, best_M = get_score(best_a)

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing phase – longer hill‑climb
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000,
                                      max_move=0.12)

    final_seq = final_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
