"""
Propose a low‑score step function using a soft‑max‑smoothed sub‑gradient
descent on the probability simplex.  The objective to minimise is

    score = 2 * n * max_k (a * a)[k]          (with ∑ a_i = 1, a_i ≥ 0)

The max‑convolution is approximated by a smooth weighted max
(soft‑max) which yields a dense gradient that guides the update.
Updates are performed with a multiplicative (exponential) step
mirroring mirror‑descent w.r.t. KL‑divergence, which naturally keeps
the vector on the simplex.  Several random restarts, a slowly
increasing smoothness parameter (β), a diminishing step size and a
short polishing phase are used within the allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex (kept for possible use)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution a * a using real FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a with itself (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of 2 ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: gradient of the soft‑max‑smoothed peak of a*a
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_j a_j * p_{i+j} for i = 0,…,n‑1.
    Here p is a probability distribution over the 2n‑1 convolution indices.
    """
    n = a.shape[0]
    # reverse p for the convolution trick
    rev_p = p[::-1]
    size = 1 << ((3 * n - 2).bit_length())          # ≥ n + (2n‑1) - 1 = 3n‑2
    Fa = np.fft.rfft(a, size)
    Fp = np.fft.rfft(rev_p, size)
    conv_full = np.fft.irfft(Fa * Fp, size)          # length = size
    # extract the slice that gives Σ_j a_j * p_{i+j}
    start = n - 1
    end = 2 * n - 1
    sub = conv_full[start:end]                       # length n, corresponds to i=n‑1…0
    g = 2.0 * sub[::-1]                              # reverse to obtain i=0,…,n‑1
    # numerical noise may produce tiny negative values; clip to zero
    return np.maximum(g, 0.0)
