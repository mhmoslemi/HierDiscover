def smoothed_descent(a0, max_iters, step0, mu_start, mu_end, deadline):
    a = a0.copy()
    n = len(a)
    # Normalize sum to 1
    a = np.maximum(a, 0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s

    # Precompute fft sizes
    fft_size_conv = 1 << ((2*n - 1).bit_length())
    conv_len = 2*n - 1
    fft_size_grad = 1 << ((3*n - 2).bit_length())

    best_seq = a.copy()
    best_score = float('inf')
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # self convolution
        A = np.fft.rfft(a, n=fft_size_conv)
        b = np.fft.irfft(A * A, n=fft_size_conv)[:conv_len]
        max_b = b.max()
        score = 2.0 * n * max_b / (a.sum()**2)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[it {it}] new best approx score={score:.6f}")

        # compute mu for this iteration
        mu = mu_start + (mu_end - mu_start) * (it-1) / (max_iters-1)

        # compute softmax weights
        scaled = b / mu
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)  # avoid overflow
        sumexp = np.sum(exp_scaled)
        v = exp_scaled / sumexp

        # gradient
        a_rev = a[::-1]
        V = np.fft.rfft(v, n=fft_size_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_size_grad)
        d = np.fft.irfft(V * A_rev, n=fft_size_grad)[: (conv_len + n - 1)]
        grad = 2.0 * d[(n-1):(n-1)+n]

        # step size
        step = step0 / np.sqrt(it)

        # update & project
        a = project_simplex(a - step * grad)

    return best_seq, best_score

"""
Smoothed gradient descent on the simplex for minimizing the peak of the self‑convolution.
The algorithm uses a temperature µ to approximate the max by a soft‑max, computes the
gradient via FFT‑based convolution, and projects onto the probability simplex.
Multiple random restarts and perturbations of the best solution are performed
within the time budget.
"""

import time
import numpy as np

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive → return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def smoothed_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    mu_start: float,
    mu_end: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex using a soft‑max approximation of the max.
    Returns the best sequence (as an ndarray) and its corresponding score.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s

    n = a.shape[0]
    conv_len = 2 * n - 1
    # FFT sizes – chosen once because n never changes within a run.
    fft_size_conv = 1 << ((conv_len).bit_length())          # for a*a
    fft_size_grad = 1 << ((conv_len + n - 1).bit_length())  # for v * a_rev

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- self‑convolution and current score ----
        A = np.fft.rfft(a, n=fft_size_conv)
        b = np.fft.irfft(A * A, n=fft_size_conv)[:conv_len]
        max_b = float(b.max())
        # sum(a) is kept equal to 1, but keep formula for safety
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[it {it}] new best score={score:.6f}")

        # ---- soft‑max temperature schedule ----
        if max_iters == 1:
            mu = mu_end
        else:
            mu = mu_start + (mu_end - mu_start) * (it - 1) / (max_iters - 1)

        # ---- compute soft‑max weights ----
        scaled = b / mu
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)    # prevents overflow
        v = exp_scaled / np.sum(exp_scaled)        # shape (conv_len,)

        # ---- gradient via convolution (v * a_rev) ----
        a_rev = a[::-1]
        V = np.fft.rfft(v, n=fft_size_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_size_grad)
        d_full = np.fft.irfft(V * A_rev, n=fft_size_grad)[: conv_len + n - 1]
        # gradient = 2 * d[(n-1) : (n-1)+n]
        grad = 2.0 * d_full[(n - 1) : (n - 1) + n]

        # ---- diminishing step size (1/√t) ----
        step = step0 / np.sqrt(it)

        # ---- projected update ----
        a = project_simplex(a - step * grad)

    return best_seq, best_score

Smoothed gradient descent on the simplex for minimizing the peak of the
self‑convolution.

The algorithm approximates the max‑operator by a temperature‑controlled soft‑max,
computes the gradient of the smoothed objective by FFT‑based convolution and
projects the iterate onto the probability simplex.  Several random restarts,
including perturbations of the current best solution, are performed until the
time budget expires.  The best sequence found (as a plain Python list) is
returned.

# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}.

    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1

    if not np.any(cond):
        # all entries are non‑positive – the uniform distribution is the projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

    step0: float,
    mu_start: float,
    mu_end: float,

    mu_start, mu_end : float

        values over the iterations.  mu≈0 gives the true max; a larger mu
        yields a smoother objective.
    deadline : float

    best_score : float

    """

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s == 0.0:

        step = step0 / np.sqrt(it)
