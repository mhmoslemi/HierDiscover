assistantfinal**Explanation of the algorithm**

The function `propose_candidate` searches for a non‑negative step‑function  
`a = (a₀,…,a_{n‑1})` that minimises  

```
2·n·max(conv(a,a)) / ( Σ a_i )²            (1)
```

The score is invariant under scaling, therefore all candidates are kept
normalised to a unit sum.  

The search consists of three stages

1. **Systematic sweep of simple parametric families** –  
   power‑law and two‑sided power‑law sequences for many lengths
   (`4000 … 20000`) and many exponents (`p∈[0.40,0.70]`).  
   The cheap FFT‑based `fast_score` is used to keep only the
   `TOP_K` best constructions.

2. **Diverse random sampling** – a short budget is spent creating
   additional candidates (exponential, stretched‑power and random
   two‑sided power‑law shapes).  Any improvement updates the
   `TOP_K` list.

3. **Projected sub‑gradient refinement** – the best candidate is
   refined for the remaining time.  
   At each iteration the self‑convolution is computed with FFT,
   the *top‑K* largest convolution entries are taken as the active
   set and the corresponding sub‑gradient  

```
g_i = 2·Σ_{k∈active} a_{k‑i}     (with 0≤k‑i<n)
```

   is obtained in `O(K·n)`.  
   A diminishing step size `η = 0.5/√t` is used and the update
   `a ← Π_{simplex}(a – η·g)` projects back onto the probability
   simplex (non‑negative entries that sum to 1).  
   Every few hundred iterations the true (fast) score is recomputed
   and the best vector seen so far is stored.

When the time budget expires the best vector is evaluated with the
provided `evaluate_sequence` and returned.

The whole programme respects the 1000 s limit, prints progress
messages and contains no hidden closures or filesystem I/O.

```python
"""Propose a height sequence that minimises
   2 * n * max(conv(a,a)) / (sum(a)**2).

   The search consists of:
   1) systematic sweep of simple parametric families;
   2) a short random‑shape sampling phase;
   3) projected sub‑gradient refinement on the current best.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Cheap evaluation of (1) using an FFT convolution.
    The sequence is assumed to be non‑negative and already normalised
    to sum = 1.
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (parameterised families)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_stretched_power_law(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-lam·i)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p) * np.exp(-lam * (idx - 1))
    return seq


# ----------------------------------------------------------------------
# Small helper to maintain a list of the best candidates
# ----------------------------------------------------------------------


def add_to_top(seq: np.ndarray, score: float, top: list, max_size: int = 10):
    """Insert (seq,score) into a sorted list keeping only the best `max_size`."""
    if len(top) < max_size:
        top.append((score, seq.copy()))
        top.sort(key=lambda x: x[0])
    else:
        if score < top[-1][0]:
            top[-1] = (score, seq.copy())
            top.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement
# ----------------------------------------------------------------------


def refine_candidate(
    a: np.ndarray, end_time: float, active_k: int = 5, report_interval: int = 500
) -> tuple[np.ndarray, float]:
    """
    Refine `a` (already normalised) by projected sub‑gradient steps
    on the simplex, using the `active_k` largest convolution entries.
    Returns the best vector seen and its fast score.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    best_a = a.copy()
    best_sc = fast_score(a)
    it = 0

    while time.time() < end_time:
        it += 1
        # convolution (FFT)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        # indices of the `active_k` largest entries
        if active_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, active_k - 1)[:active_k]

        # sub‑gradient of the max (sum of contributions from active indices)
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            # each pair contributes 2*a_j to ∂conv_k/∂a_i
            g[i_range] += 2.0 * a[j_range]

        # diminishing step size
        eta = 0.5 / np.sqrt(it)

        # projected gradient step onto the simplex
        a_candidate = a - eta * g
        a = project_simplex(a_candidate)

        # occasional re‑evaluation
        if it % report_interval == 0:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[refine] iter {it:6d} score {cur_sc:.6f}")

    return best_a, best_sc

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence.
    Returns a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    # initialise RNGs
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin
    gen_deadline = start + 0.25 * budget_s     # 25 % of the budget for generation

    # ------------------------------------------------------------------
    # 0️⃣  starting point (previous best if available)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() == 0:
            init_seq = np.full(init_seq.shape[0], 1.0 / init_seq.shape[0])
        else:
            init_seq = init_seq / init_seq.sum()
    except Exception:
        # fallback – modest power law
        init_seq = generate_power_law(8000, 0.52)
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={init_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  systematic sweep of power‑law families
    # ------------------------------------------------------------------
    top_k = 10
    top_candidates: list[tuple[float, np.ndarray]] = []
    length_choices = [4000, 6000, 8000, 10000, 12000, 15000, 18000]
    # keep lengths ≤ 20000 (FFT stays fast)
    length_choices = [l for l in length_choices if l <= 20000]

    for n in length_choices:
        # power law sweep
        for p in np.arange(0.40, 0.71, 0.008):
            seq = generate_power_law(n, float(p))
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s < 1e-12:
                continue
            seq /= s
            sc = fast_score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[systematic] n={n} p={p:.3f} score={sc:.6f}")
            add_to_top(seq, sc, top_candidates, max_size=top_k)

        # two‑sided power‑law sweep (coarse grid)
        for p_l in np.arange(0.40, 0.71, 0.10):
            for p_r in np.arange(0.40, 0.71, 0.10):
                seq = generate_two_sided_power_law(
                    n, float(p_l), float(p_r), w_left=1.0, w_right=1.0
                )
                seq = np.maximum(seq, 0.0)
                s = seq.sum()
                if s < 1e-12:
                    continue
                seq /= s
                sc = fast_score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[systematic] n={n} twosided pL={p_l:.2f} pR={p_r:.2f} score={sc:.6f}")
                add_to_top(seq, sc, top_candidates, max_size=top_k)

    # ------------------------------------------------------------------
    # 2️⃣  diverse random search (short extra phase)
    # ------------------------------------------------------------------
    families = ["power", "exp", "twosided", "stretched"]
    while time.time() < gen_deadline - 5.0:   # keep a few seconds for safety
        n = random.choice(length_choices)

        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.38, 0.72)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            seq = generate_exponential(n, lam)

        elif family == "stretched":
            p = random.uniform(0.38, 0.70)
            lam = random.uniform(5e-4, 2.5e-2)
            seq = generate_stretched_power_law(n, p, lam)

        else:   # "twosided"
            p_l = random.uniform(0.38, 0.72)
            p_r = random.uniform(0.38, 0.72)
            w_l = random.uniform(0.3, 3.0)
            w_r = random.uniform(0.3, 3.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq /= s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[random] {family} n={n} score={sc:.6f}")

        add_to_top(seq, sc, top_candidates, max_size=top_k)

    # ------------------------------------------------------------------
    # 3️⃣  refinement of the current best (projected sub‑gradient)
    # ------------------------------------------------------------------
    print(f"[refine] starting refinement, best score so far {best_score:.6f}")
    refined_seq, refined_score = refine_candidate(
        best_seq.copy(), deadline, active_k=5, report_interval=500
    )
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined_seq.copy()
        print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  final verification with the true evaluation routine
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
```
