"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search consists of four stages:

1. **Choose a few problem sizes** – we try the size of the previously
   discovered construction (if available) and a few larger sizes.
   For each size we pre‑compute the FFT length and the phase factor needed
   for the reversed transform.

2. **Generate cheap candidate families** – simple parametric families
   (exponential, power‑law, two‑exponential mixtures) plus a handful of
   random Dirichlet draws are normalised to sum = 1 and evaluated by a
   single FFT.  The best ~5 candidates for the current size are kept.

3. **Adam optimisation on a smooth max** – for every kept candidate we
   run projected Adam on the probability simplex.  The objective ``max(conv)``
   is approximated by a soft‑max ``log‑sum‑exp`` with a smoothing parameter
   α that grows from 4 to a large value (≈ 500).  The gradient of the smooth
   max is obtained via a correlation computed with a second FFT.  The loop
   stops when the global time budget (≈ 1000 s) is exhausted.

4. **Local random hill‑climbing / simulated‑annealing** – starting from the
   best vector found so far we repeatedly move a small fraction of mass
   from a “source’’ index (chosen proportionally to its weight) to a random
   “target’’ index.  Improvements are kept; otherwise the move is accepted
   with a tiny temperature‑dependent probability, allowing occasional
   escapes from local minima.

All steps respect the time budget.  The final vector is scaled to the
reference normalisation `sum = sqrt(2·n)` (the evaluation function is
scale‑invariant, any positive scaling would work).  The routine returns a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max(conv(a, a)) using a real‑FFT of length L."""
    # rfft works on real input and returns the half‑spectrum
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # linear convolution lives in the first 2·len(a)‑1 entries
    return conv[: 2 * a.shape[0] - 1].max()


# ----------------------------------------------------------------------
def _smoothmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return the soft‑max weights (summing to 1) for a given α."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, L: int, alpha: float, phase_factor: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective w.r.t. a.
    The result is a vector of length len(a).
    """
    n = a.shape[0]
    # FFT of a
    fft_a = np.fft.rfft(a, n=L)
    # linear convolution a*a (only needed for weights)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]

    # soft‑max weights
    w = _smoothmax_weights(conv, alpha)

    # FFT of the weights
    fft_w = np.fft.rfft(w, n=L)

    # FFT of the reversed a (using the pre‑computed phase factor)
    fft_a_rev = np.conj(fft_a) * phase_factor

    # Correlation a ⋆ w = ifft(fft(w) * fft(reversed a))
    corr = np.fft.irfft(fft_w * fft_a_rev, n=L)

    # gradient = 2 * correlation, pick the slice that aligns with a
    return 2.0 * corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    L: int,
    phase_factor: np.ndarray,
    max_iters: int,
    deadline: float,
    start_alpha: float = 4.0,
    end_alpha: float = 500.0,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex to minimise max(conv(a,a)).
    Returns the best vector seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.08

    best_a = a.copy()
    best_obj = _conv_max(a, L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # smoothing parameter – linear schedule
        alpha = start_alpha + (end_alpha - start_alpha) * (it / max_iters)

        # gradient of the smooth‑max
        grad = _gradient_softmax(a, L, alpha, phase_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate
        obj_new = _conv_max(a_new, L)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            # occasional progress print
            print(f"    Adam iter {it:5d}: improvement → obj={obj_new:.8e}")

        a = a_new

    return best_a, best_obj


# ----------------------------------------------------------------------
def _local_random_hillclimb(
    a_start: np.ndarray,
    L: int,
    best_obj: float,
    deadline: float,
    move_frac: float = 0.02,
    temperature: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing moves: move a small random fraction of mass from a
    source index (chosen proportional to its weight) to a random target.
    Accept an improvement always; otherwise accept with probability
    exp(-(new‑old)/temperature) (simulated‑annealing style).
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_obj = best_obj

    while time.time() < deadline:
        i = np.random.choice(n, p=a)                     # source
        j = np.random.randint(0, n)                      # target
        if i == j:
            continue
        delta = np.random.rand() * move_frac * a[i]      # up to move_frac of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue

        obj_new = _conv_max(a_new, L)

        if obj_new < cur_obj:
            a, cur_obj = a_new, obj_new
            # print only occasional improvements
            print(f"    hill‑climb improved → obj={obj_new:.8e}")
        else:
            # occasional uphill move (temperature very low → rarely)
            if temperature > 0 and np.random.rand() < math.exp(-(obj_new - cur_obj) / temperature):
                a, cur_obj = a_new, obj_new

    return a, cur_obj
