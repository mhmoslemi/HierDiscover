Propose a candidate step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search proceeds in four phases while respecting a time budget:

1. **Broad parametric sweep** – a variety of simple families
   (power‑law, exponentials, U‑shapes with a baseline, Gaussian,
   Lorentzian) are generated, normalised to unit sum and the best
   one is kept.

2. **Projected sub‑gradient descent on the exact max** – starting from
   the best candidate of phase 1 we minimise the supremum norm of the
   convolution.  The sub‑gradient of `max(conv)` at the maximising
   index `k` is known analytically, so each iteration needs only a
   single FFT, a simple O(n) gradient construction, a step on the
   simplex (Euclidean projection) and an optional symmetrisation.
   A diminishing step‑size `α_t = α0 / sqrt(t)` is used and the best
   iterate is remembered.

3. **Fine‑grained random hill‑climbing** – after the sub‑gradient
   phase we perform cheap moves that transfer a tiny amount of mass
   from a large entry to a random entry.  Improvements are kept;
   this helps to escape the final plateaus of the deterministic
   descent.

4. **Final scaling** – the best unit‑sum vector is scaled to the
   normalisation used by the reference code (`sum = sqrt(2·n)`) and
   returned.

All heavy work is done with NumPy FFTs (real‑valued `rfft`/`irfft`);
the algorithm stays well below the 1000 s budget on a modern CPU.
"""

import time

# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} (keeps the sum unchanged)."""
    return 0.5 * (a + a[::-1])


def _conv_fft(a: np.ndarray, L: int, conv_len: int) -> np.ndarray:
    """Convolution a * a using real FFT; returns the first `conv_len` entries."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:conv_len]
    return conv


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at index k.
    The gradient is g_i = 2 * a_{k-i} when k-i is a valid index, else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # indices 0 … k contribute
        g[:k + 1] = a[:k + 1][::-1] * 2.0
    else:
        overflow = k - n + 1               # number of leading zeros
        seg_len = 2 * n - k - 1           # length of the overlapping segment
        if seg_len > 0:
            g[overflow:] = a[n - seg_len:][::-1] * 2.0
    return g


def _maybe_update(a: np.ndarray,
                  best_obj: float,
                  best_seq: np.ndarray,
                  L: int,
                  conv_len: int,
                  n: int):
    """Compute max‑convolution of a, update global best if improved."""
    conv = _conv_fft(a, L, conv_len)
    obj = conv.max()
    if obj < best_obj:
        best_obj = obj
        best_seq = a.copy()
        print(f"  new best | obj={obj:.8e} | score={2 * n * obj:.6f}")
    return best_obj, best_seq

        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # enforce simplex (the move already preserves sum, just need non‑negativity)
        if a_new[i] < 0:
            continue
        # evaluate cheaply
        conv_new = _conv_fft(a_new, L, conv_len)
        obj_new = conv_new.max()
        # accept if improvement or with Metropolis probability
        accept = False
        if obj_new < cur_obj:
            accept = True
        else:
            if T > 0 and np.random.rand() < math.exp(-(obj_new - cur_obj) / T):
                accept = True
        if accept:
            a_current = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_seq = a_new.copy()
                print(f"  move improved | obj={obj_new:.8e} | score={2*n*obj_new:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the reference normalisation and logging.
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)  # final check for sanity
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it} sub‑grad iters, {moves} random moves ---")
    return final_seq
