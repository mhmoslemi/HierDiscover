def propose_candidate(seed=0, budget_s=1000, n=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # initialization
    base = _half_order_series(n)
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Parameter sweep combos
    for delta in np.linspace(0.0, 1.5, 31):
        for gamma in np.linspace(0.0, 0.02, 11):
            tilt = (np.arange(1, n+1) ** (-delta)) * np.exp(-gamma * np.arange(n))
            cand = base * tilt
            cand /= cand.sum()
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[sweep] delta={delta:.3f} gamma={gamma:.4f} score={best_score:.6f}")
            if time.time() > deadline: break
        if time.time() > deadline: break

    # gradient descent on smooth surrogate
    cur_seq = best_seq.copy()
    # set initial learning rate
    lr = 0.5
    # schedule for alpha (smoothness)
    alphas = [5, 20, 80, 300, 1000]
    for alpha in alphas:
        if time.time() > deadline: break
        # run a number of inner iterations
        for it in range(2000):
            if time.time() > deadline: break
            # compute gradient via weighted sum of all conv entries
            conv = np.convolve(cur_seq, cur_seq)
            max_conv = conv.max()
            # compute weights
            # use v = exp(alpha * (conv - max_conv))
            v = np.exp(alpha * (conv - max_conv))
            v_sum = v.sum()
            # compute weighted gradient
            grad = np.zeros_like(cur_seq)
            # loop over k contributions
            for k, wk in enumerate(v):
                if wk == 0:
                    continue
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                rev = k - idx
                grad[idx] += 2.0 * wk * cur_seq[rev]
            grad /= v_sum
            # step
            cand = cur_seq - lr * grad
            cand = _project_simplex(cand)
            cand_score = evaluate_sequence(cand.tolist())
            # acceptance if improvement else maybe continue with reduced lr
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur_seq = cand
                # Increase lr slightly
                lr = min(lr * 1.05, 2.0)
                print(f"[grad] alpha={alpha} it={it} score={best_score:.6f}")
            else:
                # reduce lr
                lr = max(lr * 0.7, 1e-6)
                # occasionally try a random perturbation if lr tiny
                if lr < 1e-5 and np.random.rand() < 0.2:
                    # random jitter
                    jitter = np.random.rand(n)
                    cand2 = cur_seq * (1 + 0.01 * (jitter - 0.5))
                    cand2 = np.clip(cand2, 0, None)
                    cand2 /= cand2.sum()
                    cur_seq = cand2
            # break early if conv max has not changed for many iter maybe use tolerance
        # optional step-size reset
        lr = 0.5

    # Simulated annealing final refinement
    cur_seq = best_seq.copy()
    cur_score = best_score
    temp_start = 0.1
    temp_end = 1e-6
    total_iters = 200000
    for i in range(total_iters):
        if time.time() > deadline:
            break
        progress = i / total_iters
        temp = temp_start * (temp_end / temp_start) ** progress
        # propose move: move mass from i to j variable amount
        i1, i2 = np.random.choice(n, 2, replace=False)
        max_move = min(cur_seq[i1], 0.05)  # up to 5% of current mass
        delta = np.random.rand() * max_move
        cand = cur_seq.copy()
        cand[i1] -= delta
        cand[i2] += delta
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
            cur_seq = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[anneal] iter={i} score={best_score:.6f}")
        # Occasionally do bigger random reinitialization
        if i % 50000 == 0:
            # try mixing with uniform to escape plateau
            mix = np.random.rand()
            cand = (1 - mix) * cur_seq + mix * np.full(n, 1.0 / n)
            cand /= cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[anneal] mix iter={i} score={best_score:.6f}")

    print(f"Finished, best score = {best_score:.6f}")

"""
Hybrid optimisation for the self‑convolution bound.

Approach
--------
1) **Parametric sweep** – explore a two‑parameter family that tilts the
   classical half‑order (central binomial) series by a power law
   exponent δ and an exponential factor γ:
       a_i ∝ C(2i,i)/4**i · (i+1)^(-δ) · exp(-γ·i)

   The best candidate from this cheap exhaustive search (≈ few hundred
   evaluations) is taken as the starting point.

2) **Momentum projected sub‑gradient descent** – repeatedly identify the
   active set of convolution indices attaining the current maximum,
   build the sub‑gradient (average over the active set) and take a
   projected step on the probability simplex using an Adam‑style
   momentum term.  The step size is adapted on‑the‑fly; the loop runs
   until the time budget is exhausted.

3) **Simulated‑annealing refinement** – a Metropolis scheme that moves a
   small amount of mass between two random coordinates.  The temperature
   is exponentially cooled.  Occasionally a global jitter is performed
   to escape plateaux.

The best sequence discovered (the one giving the smallest value of
`2·n·max(conv(p,p))`, which is exactly the score returned by
`evaluate_sequence` for a normalised vector) is returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """
    First n terms of the half‑order (central binomial) series,
    normalised to sum to 1.
    a_i = C(2i,i) / 4**i.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_vector(p: np.ndarray) -> float:
    """Return the objective 2·n·max(conv(p,p)) for a probability vector."""
    n = p.shape[0]
    conv = np.convolve(p, p)
    return 2.0 * n * conv.max()

# Main search routine
# ----------------------------------------------------------------------

    """

        else:

    best_seq = cur_best.copy()
    best_score = _score_from_vector(best_seq)
    print(f"[init] starting score = {best_score:.6f}")

    # 1) PARAMETRIC SWEEP
    # ------------------------------------------------------------------
    base = _half_order_series(n)          # canonical half‑order series
    idx = np.arange(n, dtype=np.float64)  # 0‑based indices

    deltas = np.linspace(0.0, 2.0, 41)      # power‑law tilt
    gammas = np.linspace(0.0, 0.04, 11)    # exponential tilt

    # ------------------------------------------------------------------
    # Adam‑style parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(cur_seq)
    v = np.zeros_like(cur_seq)
    learning_rate = 0.3               # initial step size

    # stop condition is purely time‑driven
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # ---- compute active set of maximal convolution entries ----
        conv = np.convolve(cur_seq, cur_seq)
        max_conv = conv.max()
        # tolerance to capture all essentially maximal entries
        tol = 1e-12
        active = np.where(conv >= max_conv - tol)[0]

        # ---- sub‑gradient (average over active set) ----
        grad = np.zeros_like(cur_seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)
            rev = k - idx_range
            grad[idx_range] += 2.0 * cur_seq[rev]

        if active.size > 0:
            grad /= float(active.size)

        # ---- Adam update on the simplex ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        cand_seq = cur_seq - step
        cand_seq = _project_simplex(cand_seq)

        cand_score = _score_from_vector(cand_seq)

        # ---- accept if improvement, otherwise shrink learning rate ----
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
            # occasionally enlarge learning rate after a good move
            learning_rate = min(learning_rate * 1.02, 1.0)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[sgd {iteration:07d}] new best {best_score:.6f}")
        else:
            # no improvement – back‑off
            learning_rate = max(learning_rate * 0.7, 1e-5)
            # occasionally inject a small random jitter to leave plateaux
            if iteration % 2000 == 0:
                noise = np.random.randn(n) * 1e-3
                jitter = cur_seq * (1.0 + noise)
                jitter = np.clip(jitter, 0.0, None)
                jitter /= jitter.sum()
                cur_seq = _project_simplex(jitter)
                cur_score = _score_from_vector(cur_seq)

        # periodic progress report
        if iteration % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[sgd {iteration:07d}] time {elapsed:.1f}s best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) SIMULATED‑ANNEALING REFINEMENT
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    total_iters = 300_000
    init_temp = cur_score * 0.1   # start rather warm
    final_temp = 1e-6

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # exponential cooling
        progress = it / total_iters
        temp = init_temp * (final_temp / init_temp) ** progress

        # ---- propose a random mass transfer ----
        i, j = np.random.choice(n, 2, replace=False)
        max_move = min(cur_seq[i], 0.02)          # ≤2 % of the mass at i
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _score_from_vector(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[anneal {it:07d}] improved to {best_score:.6f}")

        # occasional global jitter to break out of deep plateaux
        if it % 50000 == 0 and it > 0:
            mix = np.random.rand()
            jitter = (1 - mix) * cur_seq + mix * np.full(n, 1.0 / n)
            jitter /= jitter.sum()
            jitter = _project_simplex(jitter)
            jitter_score = _score_from_vector(jitter)
            if jitter_score < cur_score:
                cur_seq = jitter
                cur_score = jitter_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[anneal {it:07d}] jitter improved to {best_score:.6f}")

    elapsed_total = time.time() - start_time
    print(f"Finished after {elapsed_total:.1f}s – best score = {best_score:.6f}")

    # Return as plain Python list of floats
    return best_seq.tolist()
