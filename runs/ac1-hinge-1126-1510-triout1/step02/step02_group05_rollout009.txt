def gradient_soft_max(a, fft_size, beta):
    # compute convolution b = a*a (full)
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[:2*n-1]
    # compute weights
    max_b = b.max()
    # shift to avoid overflow
    # w_i = exp(beta * (b_i - max_b))
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(b) / len(b)
    else:
        w /= w_sum
    # compute correlation w and a
    # Since a length n, w length 2n-1
    # compute linear cross-correlation
    corr_fft_size = 1 << ((len(w) + len(a) - 1).bit_length())
    Wa = np.fft.rfft(w, n=corr_fft_size)
    Aa = np.fft.rfft(a, n=corr_fft_size)
    corr = np.fft.irfft(Wa * np.conj(Aa), n=corr_fft_size)[:len(w)+len(a)-1]
    # correlation index shift: lag = -(len(a)-1) .. len(w)-1
    # We need values for q from 0..n-1: corr index = (len(a)-1) + q
    start = len(a)-1
    grad = 2.0 * corr[start:start+len(a)]
    return grad
```

"""
Improved search for a low‑peak self‑convolution step function.

The algorithm combines two ideas:

1. **Soft‑max gradient descent** – instead of moving only in the direction
   of the single active peak, we minimise a smooth approximation of the
   maximum (log‑sum‑exp).  The gradient is a weighted sum of the gradients of
   all convolution entries and can be computed efficiently using FFT‑based
   correlation.

2. **Refinement by projected sub‑gradient descent** – after the soft‑max
   phase we perform a few steps of the classic hard‑peak sub‑gradient method
   (the one used previously) to polish the solution.

Multiple random restarts are performed until the time budget runs out.
Each restart starts from a power‑law seed, a uniform vector, or a
perturbed copy of the current global best.  All iterates are projected onto
the probability simplex `{a_i ≥ 0, Σ a_i = 1}` using the O(n) algorithm of
Duchi et al.  The best sequence found is returned as a plain Python list.

The implementation works entirely with FFTs (O(n log n) per iteration) and
fits comfortably inside the 1000 s budget for n≈4 000–5 000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  O(n) algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full_fft(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Full linear (non‑circular) convolution of a with itself using FFT.
    Returns the length‑(2·len(a)‑1) array.
    """
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b


def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient method that directly reduces
    the active peak of the self‑convolution.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_size = next_pow2(2 * n - 1)

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and peak
        b = conv_full_fft(a, fft_size)
        max_b = b.max()
        score = 2.0 * n * max_b  # sum(a) == 1 after projection

        if score < best_score:
            best_score = float(score)
            best_seq = a.copy()
            # occasional reporting (can be silenced)
            print(f"[subgrad it {it}] new best score={best_score:.6f}")

        # --- sub‑gradient of the active peak ---------------------------------
        k = int(np.argmax(b))  # index of the active peak
        grad = np.zeros_like(a)

        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)

        # gradient_i = 2 * a_{k-i}
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]

        # diminishing step size
        step = step0 / np.sqrt(it)

        # gradient step + projection
        a = project_simplex(a - step * grad)

    return best_seq, best_score


def softmax_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    beta0: float,
    beta_inc: float,
    beta_max: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the peak:
    soft‑max (log‑sum‑exp) of the convolution entries.
    The gradient is a weighted sum of all entry‑gradients and is obtained
    via FFT‑based correlation.
    Returns the best sequence and its (hard‑peak) score.
    """
    a = a0.copy()
    n = a.shape[0]
    # normalise to the simplex – scaling does not affect the score
    a = a / a.sum()

    # FFT sizes (pre‑computed)
    conv_fft = next_pow2(2 * n - 1)          # for a*a
    corr_fft = next_pow2(3 * n - 2)          # for correlation w * a

    best_seq = a.copy()
    best_score = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute convolution and current hard‑peak score -----
        b = conv_full_fft(a, conv_fft)               # length 2n‑1
        max_b = b.max()
        score = 2.0 * n * max_b                       # sum(a)=1

        if score < best_score:
            best_score = float(score)
            best_seq = a.copy()
            print(f"[soft it {it}] new best score={best_score:.6f}")

        # ----- update smoothing parameter -----
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----- compute soft‑max weights -----
        # avoid division by zero (max_b>0 unless a is all zeros)
        if max_b <= 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            # w_i ∝ exp(beta * (b_i / max_b - 1))   (peak gets weight 1)
            w = np.exp(beta * (b / max_b - 1.0))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

        # ----- gradient of the smooth objective (weighted sum of gradients) -----
        # grad = 2 * correlation(a, w)   (see derivation in analysis)
        A_fft = np.fft.rfft(a, n=corr_fft)
        W_fft = np.fft.rfft(w, n=corr_fft)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), n=corr_fft)
        # length = n + (2n‑1) – 1 = 3n‑2
        L = 2 * n - 1
        grad = 2.0 * corr[L - 1 : L - 1 + n]   # slice lag‑0 … lag‑(n‑1)

        # ----- step size (standard diminishing) -----
        step = step0 / np.sqrt(it)

        # ----- projected gradient step -----
        a = project_simplex(a - step * grad)

    return best_seq, best_score


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.

    The routine runs several independent restarts; each restart consists of
    a soft‑max gradient phase followed by a short hard‑peak sub‑gradient
    polishing phase.  The best sequence encountered is returned as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.  If a previous construction is available we
    # reuse its length, otherwise we pick a moderately large size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]  # noqa: F821
        if n < 1:
            n = 4000
    except Exception:
        n = 4000

    # Ensure a decent size (larger n tends to lower the bound)
    if n < 4000:
        n = 4000

    # ------------------------------------------------------------------
    # Global best across all restarts
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # Parameters controlling the inner optimisation
    max_iters_per_restart = 8000          # generous budget per restart
    step0_soft = 0.6
    beta0 = 2.0
    beta_inc = 0.015                      # slowly sharpen the soft‑max
    beta_max = 30.0
    step0_sub = 0.3
    max_sub_iters = 2000                 # polishing phase

    restart = 0
    while time.time() < deadline:
        restart += 1

        # --------------------------------------------------------------
        # Initialise a fresh starting point
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.15:
            # Uniform start
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        elif r < 0.6:
            # Random power‑law
            p = np.random.uniform(0.45, 0.70)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a0 = idx ** (-p)
            a0 /= a0.sum()
        else:
            # Perturbed copy of the current global best (if any)
            if global_best_seq is not None:
                a0 = np.array(global_best_seq, copy=True)
                eps = 0.02 * np.random.randn(n)
                a0 = np.maximum(a0 + eps, 0.0)
                s = a0.sum()
                if s == 0:
                    a0 = np.full(n, 1.0 / n)
                else:
                    a0 /= s
            else:
                # fallback power‑law
                p = 0.55
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()

        # --------------------------------------------------------------
        # Determine how many iterations we can afford for this restart
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        # conservative estimate: ~0.003 s per iteration for the soft phase
        it_budget = int(min(max_iters_per_restart, time_left / 0.003))
        if it_budget < 200:
            break

        # --------------------------------------------------------------
        # Phase 1 – soft‑max gradient descent
        # --------------------------------------------------------------
        seq, score = softmax_descent(
            a0,
            max_iters=it_budget,
            step0=step0_soft,
            beta0=beta0,
            beta_inc=beta_inc,
            beta_max=beta_max,
            deadline=deadline,
        )

        # --------------------------------------------------------------
        # Phase 2 – hard‑peak sub‑gradient polishing (optional)
        # --------------------------------------------------------------
        if time.time() < deadline:
            time_left = deadline - time.time()
            sub_iters = int(min(max_sub_iters, time_left / 0.003))
            if sub_iters > 0:
                seq, score = subgradient_descent(
                    seq, max_iters=sub_iters, step0=step0_sub, deadline=deadline
                )

        # --------------------------------------------------------------
        # Update global best
        # --------------------------------------------------------------
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new global best score={global_best_score:.6f} "
                f"(total it≈{it_budget})"
            )

        # Occasionally shrink the step sizes if progress stalls
        if restart % 5 == 0:
            step0_soft = max(0.2, step0_soft * 0.9)
            step0_sub = max(0.1, step0_sub * 0.9)

    # ------------------------------------------------------------------
    # Safety net – if nothing succeeded, fall back to a simple power‑law.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()

    # Convert to plain Python list as required by the external harness.
    return global_best_seq.tolist()
