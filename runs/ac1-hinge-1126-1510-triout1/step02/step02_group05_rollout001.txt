"""
Exponentiated‑gradient optimisation of the self‑convolution peak.

The objective (for a sequence a with Σa = 1) is
    F(a) = 2·n·max(conv(a,a))

which is convex on the simplex.  We minimise a smooth approximation
    f_α(a) = (1/α)·log Σ exp(α·conv_j(a))

using exponentiated‑gradient (mirror‑descent w.r.t. KL) and a
projection onto the simplex (renormalisation).  The smoothing parameter
α is increased gradually (α≈5 → 200) so that the early phase spreads the
mass and the final phase concentrates on the true maximum.  Multiple
random restarts and cheap perturbations of the current best are used
until the 1000 s budget expires.

The function `propose_candidate` returns the best sequence found as a
plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return b


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def gradient_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_j w_j·conv_j(a,a) w.r.t. a.
    Result = 2 * (w ⨂ a_rev)[n-1 : n-1+n]  (full convolution).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = len(w)
    size = 1 << ((m + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    AR = np.fft.rfft(a_rev, n=size)
    c = np.fft.irfft(W * AR, n=size)
    start = n - 1
    g = 2.0 * c[start : start + n]
    return g


def evaluate_score(a: np.ndarray, n: int) -> float:
    """Quick evaluation: 2·n·max(conv(a,a)) (a already normalised)."""
    conv = conv_self(a)
    return 2.0 * n * np.max(conv)
