"""
Search for a non‑negative step‑function minimizing
    2 * n * max(conv(a,a)) / (sum(a)^2)

The algorithm combines:
* a small library of deterministic windows (uniform, Hann, Blackman);
* massive random sampling of simple parametric families (power‑law,
  exponential, Gaussian, double‑Gaussian);
* a refinement stage that first applies an exponentiated‑gradient
  descent with line‑search on the step size, then a projected sub‑gradient
  descent (the same method that produced the previous state‑of‑the‑art).
All candidates are kept on the probability simplex, because the objective
is scale‑invariant.
The routine respects the 1000 s time budget and returns the best sequence
found.
"""

import time
import numpy as np

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """Full linear convolution a*a (non‑circular) and its maximal entry."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


def score_sequence(a: np.ndarray, fft_size: int) -> float:
    """Return the evaluation function value (sum(a) is assumed to be 1)."""
    _, max_b = compute_conv_max(a, fft_size)
    return 2.0 * a.shape[0] * max_b


def refine_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply an exponentiated‑gradient phase followed by a projected
    sub‑gradient phase to improve a candidate sequence.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_size = 1 << ((2 * n - 2).bit_length())
    best_seq = a.copy()
    best_score = score_sequence(a, fft_size)

    # ------------- Exponentiated‑gradient stage -----------------
    max_eg_iters = 4000
    for it in range(1, max_eg_iters + 1):
        if time.time() >= deadline:
            break
        # convolution and peak index
        b, max_b = compute_conv_max(a, fft_size)
        k = int(np.argmax(b))

        # gradient of the active peak (zero outside the active slice)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]

        # try three step‑size candidates around a base value
        base_eta = 0.5 / np.sqrt(it)
        best_local_seq = None
        best_local_score = best_score
        for factor in (0.5, 1.0, 2.0):
            eta = base_eta * factor
            a_candidate = a * np.exp(-eta * grad)
            s = a_candidate.sum()
            if s == 0.0:
                continue
            a_candidate = a_candidate / s
            sc = score_sequence(a_candidate, fft_size)
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = a_candidate

        if best_local_seq is not None:
            a = best_local_seq
            if best_local_score < best_score:
                best_score = best_local_score
                best_seq = a.copy()

    # ------------- Projected sub‑gradient stage ---------------
    step0 = 0.8
    max_sg_iters = 4000
    for it in range(1, max_sg_iters + 1):
        if time.time() >= deadline:
            break
        b, max_b = compute_conv_max(a, fft_size)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best score={cur_score:.6f}")

        # gradient of the active peak
        k = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score

    refines each of them with a two‑phase gradient method.  The best sequence
    found is returned as a plain Python list.
    """

    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------
    # Choose a working length.  Re‑use a previously discovered length
    # if available, otherwise fall back to a medium‑size instance.
    # ------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n < 2000:
            n = 4000
    except Exception:
        n = 4000

Search for a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm mixes three ideas:

1.  A small library of deterministic windows (uniform, Hann, Blackman);
    they are cheap to compute and already give scores ⩽ 2.

2.  Massive random sampling of simple parametric families
    (power‑law, exponential decay, single‑Gaussian, double‑Gaussian).
    The ten best candidates are kept.

3.  Each promising candidate is refined by a two‑phase gradient scheme:
    – an exponentiated‑gradient descent with a tiny line‑search on the step
      size; – a projected sub‑gradient descent (the method that produced the
      previous best result).  After the main refinements a few stochastic
      perturbations of the current champion are tried, again followed by the
      same two‑phase optimiser.

All vectors are kept on the probability simplex (they are normalised to sum
to 1), because the objective is scale‑invariant.  The routine respects the
1000 s wall‑clock budget and returns the best sequence it has discovered as a
plain Python list.
"""

import time
import numpy as np


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """Full linear convolution a*a (non‑circular) and its maximal entry."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


def score_sequence(a: np.ndarray, fft_size: int) -> float:
    """Evaluation function value, assuming Σ a = 1."""
    _, max_b = compute_conv_max(a, fft_size)
    return 2.0 * a.shape[0] * max_b


def refine_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Two‑phase refinement:
    * exponentiated‑gradient descent with a tiny line‑search,
    * projected sub‑gradient descent.
    Returns the best sequence found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_size = 1 << ((2 * n - 2).bit_length())
    best_seq = a.copy()
    best_score = score_sequence(a, fft_size)

    # ----- exponentiated‑gradient phase -----
    for it in range(1, 4001):
        if time.time() >= deadline:
            break
        b, _ = compute_conv_max(a, fft_size)
        k = int(np.argmax(b))

        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]

        base_eta = 0.5 / np.sqrt(it)
        best_local_seq = None
        best_local_score = best_score
        for factor in (0.5, 1.0, 2.0):
            eta = base_eta * factor
            a_candidate = a * np.exp(-eta * grad)
            s = a_candidate.sum()
            if s == 0.0:
                continue
            a_candidate = a_candidate / s
            sc = score_sequence(a_candidate, fft_size)
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = a_candidate

        if best_local_seq is not None:
            a = best_local_seq
            if best_local_score < best_score:
                best_score = best_local_score
                best_seq = a.copy()

    # ----- projected sub‑gradient phase -----
    step0 = 0.8
    for it in range(1, 4001):
        if time.time() >= deadline:

        b, _ = compute_conv_max(a, fft_size)
        cur_score = 2.0 * n * np.max(b)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best score={cur_score:.6f}")

        k = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))

        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
