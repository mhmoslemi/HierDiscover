    import time, numpy as np, math, random

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    # Determine length n
    try:
        base_seq = np.asarray(height_sequence_1, dtype=float)
        n0 = len(base_seq)
    except Exception:
        base_seq = None
        n0 = None

    n = n0 if n0 is not None else 8000
    # Optionally increase n with limited time
    # Actually we can try larger if time remains after initial optimization.

    target_sum = math.sqrt(2.0 * n)
    print(f"Optimizing sequence length {n}, target sum {target_sum:.3f}")

    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    def conv_full(a):
        fft_a = np.fft.fft(a, n=L_conv)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        return conv

    def correlation_full(w, a_rev):
        fft_w = np.fft.fft(w, n=L_corr)
        fft_rev = np.fft.fft(a_rev, n=L_corr)
        corr = np.real(np.fft.ifft(fft_w * fft_rev))
        return corr

    def objective(a):
        # a assumed sum = 1
        return 2.0 * n * conv_full(a).max()

    # Build initial vectors
    start_vectors = []

    # 1) previous best scaled to simplex if available
    if base_seq is not None:
        a0 = base_seq / base_seq.sum()
        start_vectors.append(a0)

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) random Dirichlet draws
    for _ in range(5):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # 4) power-law sequences
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        a = w / w.sum()
        start_vectors.append(a)

    # 5) exponential shapes
    for lam in (0.0005, 0.001, 0.002, 0.005):
        w = np.exp(lam * np.arange(n))
        a = w / w.sum()
        start_vectors.append(a)

    best_a = None
    best_score = float('inf')

    # Adam hyperparameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.05
    max_iters_per_restart = 20000
    beta_start = 0.5
    beta_end = 200.0

    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break
        eps = 1e-12
        v = np.log(a_init + eps)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        for it in range(1, max_iters_per_restart + 1):
            # Time check
            if time.time() > deadline:
                break

            a = _softmax(v)  # sum=1
            # evaluate
            conv = conv_full(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[restart {restart_id}] iter {it:5d} new best {best_score:.6f}")

            # Update beta
            beta = beta_start + (beta_end - beta_start) * it / max_iters_per_restart
            # compute smooth weights
            # to avoid overflow, shift by max
            exps = np.exp(beta * (conv - cur_max))
            w = exps / exps.sum()

            # compute gradient wrt a: 2 * correlation of w and a_rev
            a_rev = a[::-1]
            corr_full = correlation_full(w, a_rev)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]  # length n

            # gradient wrt v (softmax parameters)
            dot = np.dot(grad_a, a)

"""
Propose a step‐function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimisation consists of three stages

1.  **Multiple random restarts** – initialise a on the probability simplex
    (sum = 1) with several heuristics (previous best, uniform, Dirichlet,
    power‑law and exponential shapes).

2.  **Adam optimisation on a smooth approximation of the max** – we replace
    max(conv) by a log‑sum‑exp with a temperature β that is increased from a
    small to a large value.  The parameters are the unconstrained logits v
    (a = softmax(v)), guaranteeing non‑negativity and the simplex constraint.
    Adam with bias‑correction updates v.  The gradient of the smooth max is
    2·corr(w, a) where w are the soft‑max weights over the convolution entries.

3.  **Simple local refinement** – starting from the best vector found,
    repeatedly move a small amount of mass from one random index to another
    (keeping the total‑mass constraint).  Accept the move if the true
    objective (with the hard max) improves.  This stochastic hill‑climbing
    helps to escape tiny plateaus that the smooth‑max optimiser may miss.

All heavy work (convolution, correlation) is performed with FFTs (O(n log n))
and the whole routine respects the 1000 s time budget.  The final sequence
is rescaled to the normalisation Σa = √(2 n) used by the reference code and
clipped to the domain accepted by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures – pure top‑level utilities)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_max = np.max(v)
    e = np.exp(v - v_max)
    return e / np.sum(e)


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Full correlation w ⋆ a_rev (used for the gradient).
    Length of the result is len(w) + len(a_rev) – 1.
    """
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


def _objective_scaled(a: np.ndarray, n: int) -> float:
    """
    Return the objective after scaling the simplex vector to the
    reference sum √(2 n):   2·n·max(conv(a,a)).
    The input a must already satisfy Σa = 1.
    """
    conv = _conv_full(a)
    return 2.0 * n * np.max(conv)

    # -------------------------------------------------------------

    transfer_fraction = 0.05          # up to 5 % of the source mass per move
