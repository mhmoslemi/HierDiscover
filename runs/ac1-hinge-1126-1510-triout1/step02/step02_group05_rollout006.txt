def optimize(a0, n, max_iters, deadline):
    a = a0.copy()
    a = a / a.sum()
    fft_size = 1 << ((2*n-1).bit_length())
    momentum = np.zeros_like(a)
    beta = 0.9
    best_a = a.copy(); best_score = compute_score(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        b, max_b = compute_conv_max(a, fft_size)
        # calculate score
        score = 2.0*n*max_b/(a.sum()**2)  # sum should be 1
        if score < best_score:
            best_score = score; best_a = a.copy()
            print("[opt] new best score", score)
        # find active peak idx k
        k = int(np.argmax(b))
        # compute subgradient
        grad = np.zeros_like(a)
        i_low = max(0, k-(n-1))
        i_high = min(k, n-1)
        # vectorized: for i in i_low..i_high:
        # j = k-i
        # grad[i] = 2 * a[j]
        # we can compute via slice
        # Use array indices
        idx_i = np.arange(i_low, i_high+1)
        idx_j = k - idx_i
        grad[idx_i] = 2*a[idx_j]
        # optionally, add small regularization
        # momentum update
        momentum = beta*momentum + (1-beta)*grad
        # step size
        step = step0 / np.sqrt(it)
        # update: choose method
        if use_exp_update:
            # exponentiated gradient
            a = a * np.exp(-step * momentum)  # elementwise
        else:
            # Euclidean projection
            a = a - step * momentum
        # project onto simplex
        a = project_simplex(a)
    return best_a, best_score

def compute_score(a, n, fft_size):
    b, max_b = compute_conv_max(a, fft_size)

    return float(2*n*max_b/(sum_a*sum_a))

def gradient_update_sub(a, n, fft_size, step, momentum, beta):
    # compute conv & gradient
    b, _ = compute_conv_max(a, fft_size)
    k = int(np.argmax(b))
    grad = np.zeros_like(a)
    i_low = max(0, k-(n-1))
    i_high = min(k, n-1)
    idx_i = np.arange(i_low, i_high+1)
    idx_j = k - idx_i
    grad[idx_i] = 2*a[idx_j]
    # momentum
    momentum = beta * momentum + (1-beta) * grad
    a_new = a - step * momentum
    a_new = project_simplex(a_new)
    return a_new, momentum

def gradient_update_exp(a, n, fft_size, step, momentum, beta):
    # similar but exponentiate
    b, _ = compute_conv_max(a, fft_size)

"""
Proposed solution for the self‑convolution peak minimisation problem.

The algorithm works in two phases:

1. **Exploration** – for a handful of candidate lengths (derived from the
   existing `height_sequence_1` variable or sensible defaults) we generate
   a few simple families of sequences (uniform, power‑law, exponential)
   and evaluate their scores using a fast FFT‑based convolution.  The best
   length and the best raw sequence from this cheap search are kept as a
   starting point.

2. **Refinement** – starting from the best raw sequence we run a projected
   sub‑gradient optimisation on the simplex.  The sub‑gradient is the
   gradient of the active convolution peak (the entry of the self‑convolution
   that attains the maximum).  The update uses an O(n) projection onto the
   probability simplex (Duchi et al., 2008) and a simple momentum term.
   After a number of iterations we switch to an exponentiated‑gradient
   (multiplicative) update, which is well‑suited for non‑negative variables.
   The refinement runs for the remainder of the allocated time budget.

All operations are performed with NumPy and FFTs, so even sequences of
several ten‑thousands of elements are handled comfortably within the 1 000 s
budget.  The final list of floats is returned; the harness will clip and
scale it as required before calling `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σ x = 1} (O(n) algorithm)."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """Full linear convolution a*a (via FFT) and its maximal entry."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


def compute_score(a: np.ndarray, n: int, fft_size: int) -> float:
    """Exact score 2·n·max(conv(a,a))/sum(a)² (sum(a) should be 1)."""
    _, max_b = compute_conv_max(a, fft_size)
    s = a.sum()
    if s < 1e-12:               # safeguard (should never happen)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Optimisation core (projected sub‑gradient / exponentiated‑gradient)
# ----------------------------------------------------------------------


def subgrad_update(a: np.ndarray, n: int, fft_size: int,
                  step: float, momentum: np.ndarray,
                  beta: float) -> tuple[np.ndarray, np.ndarray]:
    """Euclidean projected sub‑gradient step."""
    b, _ = compute_conv_max(a, fft_size)
    k = int(np.argmax(b))
    grad = np.zeros_like(a)
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    idx_i = np.arange(i_low, i_high + 1)
    idx_j = k - idx_i
    grad[idx_i] = 2.0 * a[idx_j]
    momentum = beta * momentum + (1.0 - beta) * grad
    a_new = a - step * momentum
    a_new = project_simplex(a_new)
    return a_new, momentum


def expgrad_update(a: np.ndarray, n: int, fft_size: int,
                  step: float, momentum: np.ndarray,
                  beta: float) -> tuple[np.ndarray, np.ndarray]:
    """Exponentiated‑gradient (multiplicative) update with simplex projection."""
    b, _ = compute_conv_max(a, fft_size)
    k = int(np.argmax(b))
    grad = np.zeros_like(a)
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    idx_i = np.arange(i_low, i_high + 1)
    idx_j = k - idx_i
    grad[idx_i] = 2.0 * a[idx_j]
    momentum = beta * momentum + (1.0 - beta) * grad
    a_new = a * np.exp(-step * momentum)    # element‑wise
    a_new = project_simplex(a_new)
    return a_new, momentum


def refine_sequence(a0: np.ndarray, n: int, runtime: float) -> tuple[np.ndarray, float]:
    """
    Run a long‑running refinement (sub‑gradient → exponentiated‑gradient)
    on the simplex for at most `runtime` seconds.
    Returns the best sequence seen and its score.
    """
    deadline = time.time() + runtime
    fft_size = 1 << ((2 * n - 1).bit_length())
    a = a0.copy()
    a = project_simplex(a)          # guarantee sum = 1
    best_a = a.copy()
    best_score = compute_score(a, n, fft_size)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.6                     # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        step = step0 / np.sqrt(it)

        # switch to multiplicative updates after a while
        if it < 20000:
            a, momentum = subgrad_update(a, n, fft_size, step, momentum, beta)
        else:
            a, momentum = expgrad_update(a, n, fft_size, step, momentum, beta)

        # evaluate
        cur_score = compute_score(a, n, fft_size)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[refine] it {it:,} new best score = {best_score:.6f}")

    return best_a, best_score

        if time.time() >= scan_deadline:
            break

        fft_size = 1 << ((2 * n - 1).bit_length())
        candidates = []

        # Uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Random power‑law seeds
        for _ in range(4):
            p = np.random.uniform(0.45, 0.78)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = idx ** (-p)
            a /= a.sum()
            candidates.append(a)

        # Random exponential seeds
        for _ in range(4):
            lam = np.random.uniform(0.0, 0.0025)
            idx = np.arange(0, n, dtype=np.float64)
            a = np.exp(-lam * idx)
            a /= a.sum()
            candidates.append(a)

        # Evaluate all candidates for this length
        for a0 in candidates:
            score = compute_score(a0, n, fft_size)
            if score < global_best_score:
                global_best_score = score
                global_best_seq = a0.copy()
                global_best_n = n
                print(f"[scan] length {n:,} new best score = {score:.6f}")

    # --------------------------------------------------------------
    # If the scan failed to produce anything (should not happen), fall back.
    # --------------------------------------------------------------
    if global_best_seq is None:
        n = base_n
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()
        global_best_n = n

    # --------------------------------------------------------------
    # Refinement phase – spend the remaining time polishing the best candidate.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:                     # need at least a few seconds
        refined_seq, refined_score = refine_sequence(
            global_best_seq, global_best_n, time_left
        )
        if refined_score < global_best_score:
            global_best_seq = refined_seq
            global_best_score = refined_score
            print(f"[final] refined best score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # Return as plain Python list.
    # --------------------------------------------------------------
    return global_best_seq.tolist()
