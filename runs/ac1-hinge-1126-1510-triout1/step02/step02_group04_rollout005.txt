* extra fine‑tuning after all restarts, starting from the best found
  solution and using a higher‑β, lower‑lr regime;
* a final stochastic perturbation phase to escape possible shallow
  local minima.

All work is performed on the simplex ∑a = 1 (the objective is scale‑invariant)
and the final sequence is scaled to the normalisation √(2 n) required by
`evaluate_sequence`.  The implementation respects the 1000 s wall‑time
budget, printing progress and returning the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Full correlation (w ⋆ a_rev) used for the gradient."""
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


def _project_simplex(y: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of y onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if y.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = y.shape[0]
    # sort descending
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(y, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    x = np.maximum(y - theta, 0.0)
    return x


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the allotted CPU time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # -------------------------------------------------------------
    # Determine length n: use the length of the previous best if available
    try:
        n = len(height_sequence_1)                 # type: ignore
    except Exception:
        n = 6000                                    # reasonable default

    target_sum = math.sqrt(2.0 * n)                 # required normalisation

    print(f"--- Optimising sequence of length n={n} (target sum={target_sum:.3f}) ---")

    # Pre‑compute FFT lengths (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())       # for a⋆a
    L_corr = 1 << ((3 * n - 2).bit_length())       # for correlation

    # -------------------------------------------------------------
    # Helper to compute the (non‑smoothed) objective value for a on the simplex
    def _objective_score(a: np.ndarray) -> float:
        conv = _conv_full(a, L_conv)
        return 2.0 * n * conv.max()               # same as evaluate_sequence after scaling

    # -------------------------------------------------------------
    # Build a diverse set of starting points on the simplex
    start_vectors = []

    # 0) previous best (if present)
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        a0 = seq0 / seq0.sum()
        start_vectors.append(a0)
    except Exception:
        pass

    # 1) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 2) Dirichlet draws with a range of concentration parameters
    for alpha in (0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
        start_vectors.append(np.random.dirichlet(np.full(n, alpha)))

    # 3) power‑law shapes
    for p in (0.2, 0.5, 1.0, 1.5, 2.0, 3.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        a = w / w.sum()
        start_vectors.append(a)

    # 4) two‑block shape (mass at both ends, empty middle)
    block_len = max(1, n // 12)
    blk = np.concatenate([np.full(block_len, 0.5 / block_len),
                          np.zeros(n - 2 * block_len),
                          np.full(block_len, 0.5 / block_len)])
    if blk.sum() > 0:
        start_vectors.append(blk / blk.sum())

    # 5) alternating pattern (high‑low)
    alt = np.zeros(n, dtype=float)
    alt[::2] = 1.0
    if alt.sum() > 0:
        start_vectors.append(alt / alt.sum())

    # -------------------------------------------------------------
    # Adam hyper‑parameters (shared across restarts)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Temperature (β) schedule parameters
    beta_start = 5.0
    beta_end = 200.0

    # Learning‑rate schedule (base and minimum)
    lr_base = 0.07
    lr_min = 0.004

    max_iters_per_restart = 15000   # enough iterations for each run

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------------------
    # Multi‑restart optimisation
    # -------------------------------------------------------------
    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        # Initialise soft‑max parameters v (unconstrained)
        eps = 1e-12
        v = np.log(a_init + eps)
        m = np.zeros_like(v)          # first moment estimate
        v2 = np.zeros_like(v)         # second moment estimate

        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                break

            # ----- map to simplex -----
            a = _softmax(v)                        # a ≥ 0, Σa = 1

            # ----- evaluate current objective -----
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            cur_score = 2.0 * n * conv_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a * target_sum           # store scaled version
                print(f"[restart {restart_id}] iter {it:5d}  new best = {best_score:.6f}")

            # ----- smooth‑max weight (soft‑max over convolution entries) -----
            # Linear temperature schedule, but we bias a bit toward the end
            beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_restart) ** 1.2
            # Numeric stability: shift by max before exponentiation
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()

            # ----- gradient w.r.t. a (2·corr(w, a)) -----
            corr_full = _correlation_full(w, a[::-1], L_corr)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]   # central n entries

            # ----- gradient w.r.t. soft‑max parameters v -----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                 # ∂obj/∂v

            # ----- Adam update with a simple learning‑rate decay -----
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # Linear decay of lr from base to minimum
            lr = lr_min + (lr_base - lr_min) * (1.0 - it / max_iters_per_restart)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # If the multi‑restart phase found nothing, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # -------------------------------------------------------------
    # Fine‑tuning phase starting from the best sequence discovered
    # -------------------------------------------------------------
    a_start = best_seq / target_sum                 # back to simplex
    v = np.log(a_start + 1e-12)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    fine_iters = 25000
    for it in range(fine_iters):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_full(a, L_conv)
        conv_max = conv.max()
        cur_score = 2.0 * n * conv_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a * target_sum
            print(f"[fine] iter {it:5d}  new best = {best_score:.6f}")

        # Temperature pushed even higher for a near‑exact max
        beta = 250.0 + 50.0 * (it / fine_iters)          # start high, grow a little
        w_exp = np.exp(beta * (conv - conv_max))
        w = w_exp / w_exp.sum()

        corr_full = _correlation_full(w, a[::-1], L_corr)
        grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        # Very small learning‑rate in fine phase
        lr = 0.002 + 0.001 * (1.0 - it / fine_iters)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # Optional stochastic perturbation / local search
    # -------------------------------------------------------------
    # Perform a few short random walks from the current best to try escaping
    # shallow minima.  Each walk is a handful of Adam steps with a tiny
    # random perturbation added to the simplex variables.
    perturb_walks = 30
    steps_per_walk = 200
    for walk in range(perturb_walks):
        if time.time() > deadline:
            break
        # initialise from current best
        a = best_seq / target_sum
        v = np.log(a + 1e-12)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        # add a tiny Gaussian perturbation to v
        v += 0.01 * np.random.randn(*v.shape)

        for it in range(steps_per_walk):
            if time.time() > deadline:
                break
            a = _softmax(v)
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            cur_score = 2.0 * n * conv_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a * target_sum
                print(f"[perturb {walk}] iter {it:3d}  new best = {best_score:.6f}")

            # use a moderate temperature for the short walk
            beta = 100.0
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()
            corr_full = _correlation_full(w, a[::-1], L_corr)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            lr = 0.001
            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # Final safety clipping (required by evaluate_sequence) and return
    # -------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"--- Finished optimisation: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
