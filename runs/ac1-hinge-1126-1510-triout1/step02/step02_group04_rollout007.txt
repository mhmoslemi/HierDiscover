"""
Algorithm: LP‑refinement + Adam fine‑tuning

1.  Initialise a non‑negative step sequence `f` (length `n`) from the
    global variable ``height_sequence_1`` if it exists, otherwise use a
    uniform distribution.  The sequence is scaled to the conventional
    normalisation Σf = √(2 n).

2.  Iteratively improve the shape by solving a linear programme:

        maximise   Σ g_j
        subject to  g_j ≥ 0
                    (f * g)[k] ≤ M   for all k

    where `M = max(f*f)` is the current worst convolution coefficient.
    The constraints are linear in the unknown vector `g`.  The LP returns
    a feasible `g` with a larger L¹‑norm while keeping the convolution
    bound unchanged.  By moving a fraction `t` towards the scaled `g`
    the bound `M` can be reduced.

3.  The LP step is repeated a few times (while time permits).  After each
    step the best sequence seen so far (smallest `max(conv)`) is stored.

4.  Finally a short Adam optimisation (smooth‑max surrogate of the
    max‑operator) is run starting from the best sequence to polish the
    solution.

All numbers are kept non‑negative and clipped to the range required by
``evaluate_sequence``.  The function respects the 1000 s time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all defined at module level – no closures)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _fft_len(L: int) -> int:
    """Next power of two ≥ L (good for FFT)."""
    return 1 << ((L - 1).bit_length())


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT, result length = 2*len(a)-1."""
    fa = np.fft.fft(a, n=L)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * a.size - 1]


def _corr_full_fft(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Full correlation w ⋆ a_rev (used for the gradient)."""
    fw = np.fft.fft(w, n=L)
    fa_rev = np.fft.fft(a_rev, n=L)
    corr = np.fft.ifft(fw * fa_rev).real
    return corr


def _lp_step(f: np.ndarray, M: float, target_sum: float, t: float,
             max_time: float) -> tuple[np.ndarray, bool]:
    """
    Perform a single LP improvement step.
    Returns the new sequence and a boolean indicating success.
    """
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog

    n = f.size
    rows = 2 * n - 1

    # Build the (rows × n) constraint matrix for (f * g)[k] ≤ M.
    # Row k contains the coefficients f_{k-j} for all j such that 0≤k-j<n.
    row_idx = []
    col_idx = []
    data = []

    for j in range(n):
        fj = f[j]
        start = j
        end = j + n
        if end > rows:
            end = rows
        length = end - start
        if length <= 0:
            continue
        row_idx.extend(range(start, end))
        col_idx.extend([j] * length)
        data.extend([fj] * length)

    A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    b_ub = np.full(rows, M, dtype=float)

    # Linear programme: maximise Σ g_j  <=> minimise -Σ g_j
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve – we let SciPy stop if the global time budget is exhausted.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'disp': False})

    if not res.success:
        return f, False

    g = res.x
    if g.sum() == 0.0:
        return f, False

    # Scale g to the target L¹‑norm.
    g = g * (target_sum / g.sum())

    # Convex update.
    f_new = (1.0 - t) * f + t * g
    f_new = np.clip(f_new, 0.0, 1000.0)          # safety clipping

    # Quick time‑check – the caller will enforce the real deadline.
    if time.time() > max_time:
        return f_new, False

    return f_new, True


def _adam_refine(a0: np.ndarray, target_sum: float,
                 deadline: float) -> np.ndarray:
    """
    Adam optimisation on the simplex (a = soft‑max(v)) using a smooth‑max
    surrogate for the max‑convolution.  Stops when the time budget runs out.
    """
    n = a0.size
    L_conv = _fft_len(2 * n - 1)
    L_corr = _fft_len(3 * n - 2)

    # initialise soft‑max parameters
    eps = 1e-12
    v = np.log(a0 + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    beta_start, beta_end = 5.0, 120.0
    max_iters = 6000

    best_a = a0.copy()
    best_score = 2.0 * n * _conv_full_fft(a0, L_conv).max()   # objective for sum=1

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # a ≥0, Σa = 1

        # ---- convolution and current (scaled) objective ----
        conv = _conv_full_fft(a, L_conv)
        conv_max = conv.max()
        cur_score = 2.0 * n * conv_max     # same as evaluate_sequence for Σa=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {it:5d}  new best = {best_score:.6f}")

        # ---- smooth‑max weights (soft‑max over convolution entries) ----
        beta = beta_start + (beta_end - beta_start) * it / max_iters
        w_exp = np.exp(beta * (conv - conv_max))      # shift for stability
        w = w_exp / w_exp.sum()                       # shape (2n‑1,)

        # ---- gradient w.r.t. a (2·corr(w , a)) ----
        corr = _corr_full_fft(w, a[::-1], L_corr)
        grad_a = 2.0 * corr[n - 1: n - 1 + n]         # segment of length n

        # ---- gradient w.r.t. soft‑max parameters v ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # Scale the best found a to the required L¹‑norm.
    best_a = best_a * (target_sum / best_a.sum())
    return best_a

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # Length of the candidate (reuse existing construction if possible)
    try:

    except Exception:

    target_sum = math.sqrt(2.0 * n)                # normalisation used by the reference

    # --------------------------------------------------------------

    else:

    f = np.clip(f, 0.0, None)                     # ensure non‑negative
    if f.sum() == 0.0:

    f = f * (target_sum / f.sum())                # scale to the normalisation
    best_f = f.copy()
    best_score = np.convolve(f, f).max()
    print(f"Initial max(conv) = {best_score:.6f}")

    # --------------------------------------------------------------
    # LP refinement phase
    max_lp_iters = 6                               # few costly LP solves
    step_fraction = 0.3                            # convex combination weight

    for lp_it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        # current worst coefficient
        M = np.convolve(f, f).max()

        # LP update
        f, ok = _lp_step(f, M, target_sum, step_fraction, deadline)

    print(f"--- Finished: best max(conv) = {best_score:.6f} (n={n}) ---")
    return best_f.tolist()
