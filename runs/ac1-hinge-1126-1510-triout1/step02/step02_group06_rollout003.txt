def refine_sequence(a, max_iters, deadline):
    n = len(a)
    fft_len = next_pow2(2*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute conv
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        idx = int(np.argmax(conv))
        # gradient
        g = np.zeros_like(a)
        i_start = max(0, idx-(n-1))
        i_end = min(n-1, idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end+1)
            j_range = idx - i_range
            g[i_range] = 2 * a[j_range]
        # update with Adam
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # step size
        eta = 0.05/np.sqrt(t)  # adjust
        # exponentiated gradient:
        a = a * np.exp(-eta * m_hat/(np.sqrt(v_hat)+epsilon))
        a = project_simplex(a)
        # periodically evaluate
        if t % 200 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

"""
Propose a step function by a hybrid random‑search + interleaved placement and
mirror‑descent (exponentiated‑gradient) on the simplex.

The algorithm proceeds in three stages within the 1000 s budget:

1️⃣  **Broad random exploration** – generate many candidates of different
    families (power‑law, exponential, Beta‑like, two‑sided power‑law,
    combined power‑law × exponential).  For each shape we optionally
    *interleave* the sorted values using a coprime stride, which spreads
    large coefficients evenly across the index range.  Every candidate is
    normalised to sum = 1 and scored quickly with an FFT‑based convolution.

2️⃣  **Local refinement** – the current best sequence (and a few perturbed
    copies) are improved by an Adam‑style exponentiated‑gradient update
    on the probability simplex.  The gradient is the sub‑gradient of the
    currently maximal convolution entry (all active indices are used).  After
    each step we project back onto the simplex (Euclidean projection).  This
    directly reduces the peak of the self‑convolution and therefore the
    evaluation score.

3️⃣  **Verification** – the best sequence found is finally passed to the
    supplied `evaluate_sequence` function and returned as a plain Python list.

All helper functions are defined at top‑level, no closures or filesystem I/O
are used, and the routine respects the time limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using FFT convolution.
    The sequence must be non‑negative and already normalised (sum = 1).
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    # zero‑pad to a power of two for a fast FFT
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    # score = 2 * n * max_conv / (sum_a ** 2), but sum_a = 1 after normalisation
    return float(2.0 * n * max_conv)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero → uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_combined(n: int, p: float, lam: float) -> np.ndarray:
    """a_i = (i+1)^{-p} * exp(-lam * i)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p) * np.exp(-lam * (idx - 1))


def generate_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ x^{α-1} (1-x)^{β-1} where x = i/(n+1).  This yields a smooth
    Beta‑like shape on a discrete grid.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    x = i / (n + 1.0)
    # avoid exact zeros at the boundaries for α<1 or β<1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    return np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)


# ----------------------------------------------------------------------
# Helper: spread large values uniformly (interleaving)
# ----------------------------------------------------------------------


def interleave(seq: np.ndarray) -> np.ndarray:
    """
    Take a non‑negative sequence, sort it descending and place the values
    on the index line with a random stride that is coprime to the length.
    This scatters large coefficients as evenly as possible.
    """
    n = seq.shape[0]
    sorted_seq = np.sort(seq)[::-1]  # descending
    # pick a random stride coprime with n
    step = random.randint(1, n - 1)
    while math.gcd(step, n) != 1:
        step = random.randint(1, n - 1)
    positions = (np.arange(n) * step) % n
    out = np.empty_like(seq)
    out[positions] = sorted_seq
    return out

        if random.random() < 0.35:
            seq = interleave(seq)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new best ({family}) n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refinement – Adam‑style exponentiated gradient on the simplex
    # ------------------------------------------------------------------
    # Build a small pool (best + a few perturbed variants)
    pool = [best_seq]
    for _ in range(3):
        a = best_seq.copy()
        n = a.shape[0]
        # random small mass transfers between random positions
        for __ in range(6):
            i = random.randrange(n)
            j = random.randrange(n)
            if i == j:
                continue
            delta = a[i] * random.uniform(0.01, 0.07)
            a[i] -= delta
            a[j] += delta
        a = project_simplex(a)
        pool.append(a)

    refined_best_seq = best_seq.copy()
    refined_best_score = best_score

    for pool_idx, candidate in enumerate(pool):
        if time.time() > deadline - 5.0:
            break
        a = candidate.copy()
        n = a.shape[0]
        fft_len = next_pow2(2 * n)
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-12
        it = 0
        while time.time() < deadline - 5.0:
            it += 1
            # convolution via FFT
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
            max_val = conv.max()
            # identify *all* indices attaining the max (within tolerance)
            tol = max_val * 1e-12
            active = np.where(conv >= max_val - tol)[0]

            # sub‑gradient of the peak (sum over all active locations)
            g = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                if i_start <= i_end:
                    i_range = np.arange(i_start, i_end + 1)
                    j_range = k - i_range
                    g[i_range] += 2.0 * a[j_range]

            # Adam updates (first‑order moments)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size decreasing with sqrt(iter)
            eta = 0.15 / (np.sqrt(it) + 1.0)

            # exponentiated‑gradient step (ensures positivity)
            a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
            a = project_simplex(a)  # back onto simplex

            # periodic full evaluation
            if it % 400 == 0:
                cur_score = fast_score(a)
                if cur_score < refined_best_score:
                    refined_best_score = cur_score
                    refined_best_seq = a.copy()
                    print(
                        f"[{time.time() - start:.1f}s] pool {pool_idx} it {it} improved score={cur_score:.6f}"
                    )
                # early stop if we already beat the target comfortably
                if refined_best_score < 1.48:
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = refined_best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Final score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
