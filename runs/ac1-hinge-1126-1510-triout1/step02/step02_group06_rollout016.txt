assistantfinal**Solution Overview**

The task is to minimise  

\[
F(a)=\frac{2\,n\,\max\limits_k (a\ast a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `a∗a` is the *full* convolution (the antidiagonal sums of the outer‐product of `a`).  
The ratio is invariant under scaling, therefore we can always normalise the vector to
`sum(a)=1` and simply minimise `2·n·max(conv)`.

The previous implementation performed a random search followed by a projected
sub‑gradient update that only used the *single* active pair of the maximal
convolution entry.  That approach converges, but very slowly when the optimum is
close to the bound.

Below we give a new algorithm that improves the search in two ways

1. **Richer random families** – besides power‑law and exponential shapes we also
   generate two‑sided, bimodal (two separated Gaussian bumps) and logarithmic
   families.  A short random phase quickly finds a handful of promising candidates.

2. **Smoothed optimisation** – instead of the non‑smooth `max` we minimise the
   *p‑norm* of the convolution  

   \[
   \Phi_p(a)=\bigl(\sum_k (a\ast a)_k^{\,p}\bigr)^{1/p},
   \qquad p>1,
   \]

   whose gradient can be written in closed form:

   \[
   \nabla\Phi_p(a)_i = 2\sum_{j=0}^{n-1} a_j\,
   \frac{(a\ast a)_{i+j}^{\,p-1}}{\bigl(\sum_k (a\ast a)_k^{\,p}\bigr)^{\frac{p-1}{p}}}.
   \]

   The gradient is computed by two FFT‑based convolutions,
   i.e. in `O(n log n)` time.  
   A diminishing step size `η = 0.5/√t` together with Euclidean projection onto the
   probability simplex (`project_simplex`) yields a simple yet effective
   projected (mirror) gradient descent.  We start with a moderate smoothing
   (`p=4`) and switch to a stronger one (`p=8`) after a few thousand iterations
   – this “annealing” helps to escape shallow plateaus.

The whole procedure respects the 1000 s budget, logs progress, and finally
returns the best sequence found (as a plain Python list).  All helper functions
are top‑level, no closures are used, and the code relies only on `numpy` and the
provided `evaluate_sequence`.

python
"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of two phases:

1. Random exploration of several parametric families (power‑law,
   exponential, two‑sided, bimodal Gaussians, logarithmic).  The best
   few candidates are kept.

2. Refined optimisation of each kept candidate by projected gradient descent
   on the smooth p‑norm surrogate of the max‑convolution.  The gradient is
   evaluated via FFT, a diminishing step size η = 0.5/√t and Euclidean projection
   onto the probability simplex.  The smoothing exponent p is increased
   during the run (p=4 → p=8) to obtain faster convergence.

The routine respects the 1000 s time limit, prints intermediate scores,
and finally returns the best sequence as a plain list of floats.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (must stay at top level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two no smaller than x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using FFT.
    The sequence is assumed to be non‑negative and already normalised
    (sum == 1).  The returned value equals 2*n*max(conv(seq,seq)).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # zero‑pad to a power of two for fast FFT
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv.real) / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families for the random phase
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """Left and right power‑law tails added together."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_bimodal(n: int, mu1: float, sigma1: float, mu2: float, sigma2: float,
                     w1: float = 0.5) -> np.ndarray:
    """Two Gaussian bumps (weights w1 and 1‑w1)."""
    x = np.arange(n, dtype=np.float64)
    g1 = w1 * np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
    g2 = (1.0 - w1) * np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
    return g1 + g2


def generate_logarithmic(n: int, shift: float = 1.0) -> np.ndarray:
    """a_i ∝ 1 / (log(i+2) + shift)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return 1.0 / (np.log(idx + 1) + shift)


# ----------------------------------------------------------------------
# Gradient descent on the smooth surrogate
# ----------------------------------------------------------------------


def smooth_gradient_step(a: np.ndarray, p: float, fft_len1: int, fft_len2: int,
                         L: int) -> np.ndarray:
    """
    One projected gradient step for the p‑norm surrogate.

    Returns the new iterate (already projected onto the simplex).
    """
    # ----- convolution a*a (full) ------------------------------------
    fa = np.fft.rfft(a, n=fft_len1)
    conv = np.fft.irfft(fa * fa, n=fft_len1)[:L]               # length 2n‑1

    # ----- compute the weight w_k = conv_k^{p‑1} / Σ conv^p^{(p‑1)/p} ----
    conv_pow_p = conv ** p
    sum_pow = np.sum(conv_pow_p)
    # numerical safety – sum_pow cannot be zero for a non‑zero vector
    if sum_pow == 0.0:
        return a.copy()
    denom = sum_pow ** ((p - 1) / p)
    w = conv ** (p - 1) / denom                               # length 2n‑1

    # ----- gradient = 2 * (a * w_rev) evaluated at the right indices ---
    w_rev = w[::-1]                                            # reversed
    fa2 = np.fft.rfft(a, n=fft_len2)
    fw2 = np.fft.rfft(w_rev, n=fft_len2)
    conv2_full = np.fft.irfft(fa2 * fw2, n=fft_len2)          # length n+L‑1
    # required slice: g_i = 2 * conv2_full[L‑1‑i]
    g = 2.0 * conv2_full[L - 1 - np.arange(a.size)]

    # ----- simple diminishing step size -------------------------------
    # iteration number is hidden inside the step size; we will multiply by
    # a factor that is passed externally.
    return g  # the caller will apply step‑size and projection


def refine_sequence(a0: np.ndarray, deadline: float, start: float) -> tuple:
    """
    Refine a single candidate with projected gradient descent on the smooth
    surrogate.  The function returns (best_seq, best_score) found before the
    deadline.
    """
    a = a0.copy()
    n = a.size
    L = 2 * n - 1
    fft_len1 = next_pow2(L)               # length for a*a
    fft_len2 = next_pow2(n + L - 1)       # length for a * w_rev

    best_seq = a.copy()
    best_score = fast_score(a)

    iteration = 0
    while time.time() < deadline - 0.5:   # small safety margin
        iteration += 1

        # smoothing exponent: start with p=4, after 5000 iters switch to p=8
        p = 4.0 if iteration < 5000 else 8.0

        # ----- gradient (without step size) -------------------------
        g = smooth_gradient_step(a, p, fft_len1, fft_len2, L)

        # ----- step size = 0.5 / sqrt(iter) ------------------------
        eta = 0.5 / np.sqrt(iteration)

        # ----- projected gradient update ----------------------------
        a_candidate = a - eta * g
        a = project_simplex(a_candidate)

        # ----- occasional exact evaluation ---------------------------
        if iteration % 500 == 0:
            cur = fast_score(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] refinement it={iteration:5d} "
                      f"new best {cur:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative height sequence.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # keep half a second for the final check

    # --------------------------------------------------------------
    # 0️⃣  Initialise – use the previous best if it exists
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() == 0:
            raise ValueError
        init_seq = init_seq / init_seq.sum()                     # normalise to sum=1
    except Exception:
        # fallback: a modest power‑law that works for any n
        init_len = 3000
        init_seq = generate_power_law(init_len, 0.55)
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Randomised family search (≈30 % of the budget)
    # --------------------------------------------------------------
    # allowed lengths – we keep them ≤ 15000 to stay fast
    length_choices = [best_seq.size,
                      best_seq.size + 500,
                      best_seq.size + 1000,
                      4000, 6000, 8000, 10000, 12000, 15000]
    length_choices = [int(l) for l in length_choices if 0 < l <= 15000]

    families = ["power", "exp", "twosided", "bimodal", "log"]
    random_deadline = deadline - 250.0   # reserve about 250 s for refinement

    while time.time() < random_deadline:
        n = random.choice(length_choices)

        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.20, 1.20)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.2, 2.5)
            wr = random.uniform(0.2, 2.5)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "bimodal":
            # two Gaussian bumps placed roughly at 1/3 and 2/3 of the interval
            mu1 = random.uniform(0.1, 0.3) * n
            mu2 = random.uniform(0.6, 0.9) * n
            sigma1 = random.uniform(0.02, 0.07) * n
            sigma2 = random.uniform(0.02, 0.07) * n
            w1 = random.uniform(0.3, 0.7)
            seq = generate_bimodal(n, mu1, sigma1, mu2, sigma2, w1)

        else:  # logarithmic
            shift = random.uniform(0.1, 2.0)
            seq = generate_logarithmic(n, shift)

        # normalise and guard against degenerate cases
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new random best ({family}) "
                  f"n={n} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Keep a few top candidates for deeper refinement
    # --------------------------------------------------------------
    # we keep the current best plus a couple of random runners‑up
    top_candidates = [(best_score, best_seq.copy())]

    # add a few more variants (if time still allows) – they are cheap
    for _ in range(4):
        n = random.choice(length_choices)
        p = random.uniform(0.4, 0.8)
        cand = generate_power_law(n, p)
        cand = cand / cand.sum()
        sc = fast_score(cand)
        top_candidates.append((sc, cand))

    # sort and retain the best three distinct candidates
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:3]

    # --------------------------------------------------------------
    # 3️⃣  Refine each top candidate with the smoothed gradient method
    # --------------------------------------------------------------
    for idx, (score, cand) in enumerate(top_candidates, 1):
        print(f"[{time.time() - start:.1f}s] start refinement #{idx} "
              f"length={cand.size}, start_score={score:.6f}")
        refined_seq, refined_score = refine_sequence(cand, deadline, start)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start:.1f}s] ** new overall best "
                  f"{best_score:.6f} (refinement #{idx})")

    # ------------------------------------------------------------------
    # 4️⃣  Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]  # noqa: F821
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
