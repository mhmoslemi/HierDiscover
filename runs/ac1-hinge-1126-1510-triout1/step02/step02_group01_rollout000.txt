def objective_and_smooth(y, n, fft_len_conv, fft_len_g, lam):
   a = softmax(y)
   # convolution
   conv = conv_fft(a, fft_len_conv, n)  
   # compute smooth max as before.
   max_c = conv.max()
   # smoothing by lam = lam_factor * max_c? Actually lam passed directly.
   # we set lam as absolute smoothing param rather than factor.
   shifted = conv - max_c
   exp_shift = np.exp(shifted / lam)
   Z = exp_shift.sum()
   smooth_max = max_c + lam * np.log(Z)
   f = 2.0 * n * smooth_max
   # gradient
   w = exp_shift / Z  # shape (2n-1)
   # gradient conv wrt a: g_i = sum_{j} a_j w_{i+j} for i from 0..n-1 (same as autocorrelation)
   # can compute via FFT: g_rev = ifft(fft(a_rev) * fft(w))
   # pre-compute reversed a
   a_rev = a[::-1]
   fa_rev = np.fft.fft(a_rev, fft_len_g)
   fw = np.fft.fft(w, fft_len_g)
   conv_rev = np.real(np.fft.ifft(fa_rev * fw))
   g_vec = conv_rev[(n-1):(n-1+n)]   # shape n
   mean_g = np.dot(a, g_vec)
   grad_y = a * (g_vec - mean_g)
   return f, grad_y

    a = a_initial.copy()
    conv = conv_fft(a, fft_len, n)
    best_a = a.copy()
    best_score = 2*n*conv.max()
    best_conv = conv.copy()
    current_score = best_score
    rng = np.random.default_rng()
    T0 = 0.01 * best_score  maybe.
    T = T0
    iteration = 0
    start = time.time()
    while time.time() < time_limit:
        # schedule temperature decreasing
        tau = 0.0005  # what? adapt based on remaining.
        # generate random move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta_max = min(a[i], 0.1)   # move up to 10% of a[i] but not exceed.
        delta = rng.random() * delta_max
        if delta <= 0:
            continue
        # propose
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # recompute conv
        conv_new = conv_fft(a_new, fft_len, n)
        score_new = 2*n*conv_new.max()
        # acceptance criteria
        if score_new < current_score or rng.random() < np.exp(-(score_new-current_score)/T):
            a = a_new
            conv = conv_new
            current_score = score_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        # reduce temperature gradually
        iteration += 1
        T = T0 * np.exp(-iteration / 5000)   # choose schedule length

                                  lam_schedule,

            best_a = a_opt.copy()
            best_y = y_opt.copy()

        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline, max_iter=3000)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

                                             anneal_deadline, rng)

    for lam in lam_schedule[-3:]:  # last few small smoothings
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=best_y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 200, "gtol": 1e-12, "disp": False})
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final LBFGS] lam={lam:.4f} fast score {best_score:.6f}")

    # final hill climb
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        final_deadline, max_iter=5000)
        best_y = np.log(best_a + 1e-12)

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Propose a low‑evaluation step‑function sequence.

The optimizer works on the probability simplex (soft‑max parametrisation)
and alternates three phases:

1. **Exploratory L‑BFGS runs** – many random starts and a continuation
   schedule that slowly reduces the smooth‑max smoothing parameter.
   Each run is followed by a quick greedy hill‑climb.

2. **Simulated‑annealing refinement** – a stochastic mass‑transfer
   search that accepts occasional uphill moves, allowing the algorithm
   to escape small local minima that the gradient‑based phase may miss.

3. **Final polishing** – a few L‑BFGS passes with a very small smoothing
   factor and a final hill‑climb.

All sequences are kept on the simplex (sum = 1) – the official
`evaluate_sequence` is scale‑invariant, so the score is simply
`2·n·max(conv(a))`.  The code tries to minimise the maximal entry of the
autoconvolution while respecting the non‑negativity constraints.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using an FFT of length ``fft_len``.
    ``a`` must have length ``n``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the official score for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained vector y.
    The smoothing width is lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                    # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max (w.r.t. a) -----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1  (convolution a_rev * w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 120,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random fraction of mass
    from i to j (keeping the sum = 1).  Keep a move only if the fast_score
    strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score + 1e-12 < best_score:      # strict improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer search with a simple temperature schedule.
    Accepts uphill moves with probability exp(-(Δscore)/T).
    """
    a_cur = a_start.copy()
    score_cur = start_score
    best_a = a_cur.copy()
    best_score = start_score

    start_time = time.time()
    total_time = max(1e-6, deadline - start_time)
    T0 = 0.01 * start_score  # initial temperature ≈ 1 % of the score

    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start_time
        fraction = elapsed / total_time
        T = T0 * max(0.0, 1.0 - fraction)

        # choose a random transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # allow larger moves while the temperature is high
        max_delta = min(a_cur[i], 0.20)          # at most 20 % of source mass
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # propose move
        a_cur[i] -= delta
        a_cur[j] += delta

        conv_new = conv_fft(a_cur, fft_len, n)
        score_new = float(2.0 * n * conv_new.max())
        accept = False
        if score_new < score_cur:
            accept = True
        else:
            # uphill acceptance
            prob = np.exp(-(score_new - score_cur) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            score_cur = score_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_cur.copy()
        else:
            # revert the move
            a_cur[i] += delta
            a_cur[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin for final evaluation

    # --------------------------------------------------------------
    # problem size – keep it moderate for speed, but allow reuse of a
    # longer previously discovered construction if available.
    # --------------------------------------------------------------
    n = 5000
    try:
        # reuse a previously found sequence if it exists
        previous = globals()["height_sequence_1"]
        if isinstance(previous, (list, tuple, np.ndarray)):
            prev_arr = np.asarray(previous, dtype=float)
            if prev_arr.ndim == 1 and prev_arr.size >= 1000:
                n = max(n, int(prev_arr.size))
    except Exception:
        pass

    # Clip n to a comfortable upper bound (memory + speed)
    n = min(n, 8000)

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise with a (normalized) existing sequence if possible,
    # otherwise fall back to uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size == n:
            a0 = init_seq.copy()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # Normalise to the probability simplex
    a0_sum = a0.sum()
    if a0_sum <= 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0_sum

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.30, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001]
    maxiter_stage = 120  # per smoothing level
    iteration = 0

    # --------------------------------------------------------------
    # Phase 1 – random restarts with L‑BFGS + short hill‑climb
    # Run until roughly half of the time budget is consumed.
    # --------------------------------------------------------------
    while time.time() < deadline - 200.0:   # reserve ~200 s for refinement phases
        mode = rng.choice(["random", "exp", "power", "edge", "mix", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n, dtype=float))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "mix":
            # convex combination of exponential and power‑law shapes
            w = rng.random()
            rate = rng.uniform(0.001, 0.01)
            p = rng.uniform(0.5, 1.5)
            exp_part = np.exp(-rate * np.arange(n, dtype=float))
            pow_part = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = w * exp_part + (1.0 - w) * pow_part
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # "best" – perturb current best
            y0 = best_y + rng.normal(scale=0.03, size=n)

        # run the continuation L‑BFGS schedule
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt + 1e-12 < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # a brief greedy hill‑climb to polish the new candidate
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement on the current best.
    # --------------------------------------------------------------
    anneal_deadline = deadline - 30.0   # keep ~30 s for final polishing
    if time.time() < anneal_deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
            rng,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[anneal] best fast score after annealing = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final gradient‑based polishing (few passes with tiny smoothing)
    # --------------------------------------------------------------
    for lam in lam_schedule[-4:]:          # the four smallest smoothing factors
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 250, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final LBFGS] lam={lam:.4f} fast score = {best_score:.6f}")

    # final greedy polish (use whatever time remains)
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
