        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2 * n * m

        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2 * n * m

    cur_score = best_score
    
    # optimization parameters

    step = 0.2

    iteration = 0

        conv = np.convolve(cur_seq, cur_seq)
        m = conv.max()

        cur_score = 2 * n * cur_max
        if cur_score < best_score:

            best_max = cur_max
            best_seq = cur_seq.copy()

        mh = np.max(p * conv)

            cur_seq /= cur_seq.sum()
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()

def _half_order_series(n):
    """Return first n terms of a_i = C(2i,i)/4**i, normalized to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a

def _project_simplex(v):
    """Project vector v onto probability simplex (non‑negative, sum=1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0

    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)

                delta = np.random.rand() * max_delta
                cand_seq = cur_seq.copy()
                cand_seq[i] -= delta
                cand_seq[j] += delta
                cand_conv = np.convolve(cand_seq, cand_seq)
                cand_max = cand_conv.max()
                accept = False
                if cand_max < cur_max:
                    accept = True
                else:
                    # very small Metropolis acceptance to allow occasional uphill moves
                    temp = temp_initial * ((temp_final / temp_initial) ** (iteration / 50000.0))
                    prob = math.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if np.random.rand() < prob:
                        accept = True
                if accept:
                    cur_seq = cand_seq
                    cur_max = cand_max
                    cur_score = 2.0 * n * cur_max
        else:
            # -------- gradient descent on smoothed max --------------------
            # smooth max via log‑sum‑exp (parameter p controls sharpness)
            weighted = p * conv
            w = np.exp(weighted - np.max(weighted))
            w /= w.sum()                     # w is a probability vector over the 2n‑1 conv entries

            # gradient = 2 * Σ_i w_i * a_{i‑j}
            w_rev = w[::-1]                  # reversed for convolution
            conv_rev = np.convolve(cur_seq, w_rev)           # length 3n‑2
            # slice indices n‑1 … 2n‑2 (inclusive) and reverse to obtain gradient[j]
            grad = 2.0 * conv_rev[(n - 1):(2 * n - 1)][::-1]

            # simple back‑tracking line search on the simplex
            improved = False
            trial_step = step
            for _ in range(5):
                cand_seq = cur_seq - trial_step * grad
                cand_seq = _project_simplex(cand_seq)
                cand_conv = np.convolve(cand_seq, cand_seq)
                cand_max = cand_conv.max()
                cand_score = 2.0 * n * cand_max
                if cand_score < cur_score:
                    cur_seq = cand_seq
                    cur_max = cand_max
                    cur_score = cand_score
                    step = min(step * 1.3, step_max)
                    improved = True
                    break
                else:
                    trial_step *= 0.5
            if not improved:
                step = max(step * 0.7, step_min)
                # occasional jitter to escape flat plateaux
                if iteration % 800 == 0:
                    sigma = 0.001
                    noise = np.random.randn(n) * sigma
                    cand_seq = cur_seq * (1.0 + noise)
                    cand_seq = np.clip(cand_seq, 0.0, None)
                    cand_seq /= cand_seq.sum()
                    cur_seq = cand_seq

        # --------------------------------------------------------------
        # increase smoothing parameter (sharpen the approximation of max)
        # --------------------------------------------------------------
        if p < p_max:
            p = min(p_max, p + p_inc)

        # --------------------------------------------------------------
        # occasional pull towards the uniform distribution (helps to stay
        # away from extremely spiky corners of the simplex)
        # --------------------------------------------------------------
        if iteration % 5000 == 0:
            alpha = 0.005
            cur_seq = (1.0 - alpha) * cur_seq + alpha * uniform
            cur_seq /= cur_seq.sum()

        # --------------------------------------------------------------
        # progress report
        # --------------------------------------------------------------
        if iteration % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"[{iteration:6d}] elapsed {elapsed:.1f}s – best {best_score:.6f}")

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Return as a plain Python list – the scorer expects a list.
    return best_seq.tolist()
