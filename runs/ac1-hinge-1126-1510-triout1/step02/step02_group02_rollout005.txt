import time, math, numpy as np

def _project_simplex(v): ... (as above)

def _conv_max(a, L):
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max()

def _smoothmax_weights(conv, alpha):
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()
    return w

def _grad_from_weights(a, w, phase, L):
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    n = a.shape[0]
    return 2.0 * corr[n-1:n-1+n]

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2).

    Strategy:
    1. Load any previously known solution.
    2. Exhaustive sweep over several parametric families
       (power‑law, exponential, beta‑distribution, sinusoid, …).
       Keep the best candidate.
    3. Refine the best candidate with Adam on a smooth‑max
       approximation of the supremum norm of the convolution.
    4. Finetune with a simulated‑annealing hill‑climber that
       moves small amounts of mass between indices.
    The whole process respects the supplied time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Determine length n
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        n = len(init)
        if init.sum() > 0:
            init = init / init.sum()
        else:
            init = None
    except Exception:
        init = None
        n = 6000   # fallback length

    target_sum = math.sqrt(2.0 * n)

    # FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 2).bit_length())
    # Phase factor for reversed FFT (used in gradient)
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    best_seq = None
    best_obj = float('inf')
    # ------------------------------------------------------------------
    # 0. Use previous global candidate if present
    if init is not None:
        obj = _conv_max(init, L)
        best_seq = init.copy()
        best_obj = obj
        print(f"Init candidate from global: max={obj:.8e}, score={2*n*obj:.6f}")

    # ------------------------------------------------------------------
    # Helper for updating best
    def update_if_better(a_raw):
        nonlocal best_seq, best_obj
        if a_raw.sum() == 0:
            return
        a = a_raw.astype(np.float64)
        a = a / a.sum()
        obj = _conv_max(a, L)
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"  new best obj={obj:.8e} score={2*n*obj:.6f}")

    # ------------------------------------------------------------------
    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    # 1. Power‑law (including decreasing)
    exps = np.linspace(-0.5, 2.0, 31)
    idx = np.arange(1, n+1, dtype=np.float64)
    for p in exps:
        if time.time() > deadline:
            break
        shape_inc = idx ** p
        update_if_better(shape_inc)
        update_if_better(shape_inc[::-1])

    # 2. Exponential decay and its mirror
    lambdas = np.linspace(0.0, 6.0, 31)
    for lam in lambdas:
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
        update_if_better(exp_shape)
        update_if_better(exp_shape[::-1])

    # 3. Double exponential (U‑shaped) with independent left/right rates
    lams = np.linspace(0.0, 6.0, 11)
    for lam_left in lams:
        if time.time() > deadline:
            break
        left = np.exp(-lam_left * np.arange(n, dtype=np.float64))
        for lam_right in lams[::3]:   # coarser on right side
            if time.time() > deadline:
                break
            right = np.exp(-lam_right * np.arange(n-1, -1, -1, dtype=np.float64))
            u_shape = left + right
            update_if_better(u_shape)

    # 4. Symmetric Beta family a_i ∝ (i+1)^{α-1} (n-i)^{α-1}
    alphas = np.linspace(0.1, 2.5, 25)
    i0 = np.arange(1, n+1, dtype=np.float64)
    for a in alphas:
        if time.time() > deadline:
            break
        beta_shape = (i0 ** (a - 1)) * ((n - i0 + 1) ** (a - 1))
        update_if_better(beta_shape)

    # 5. Asymmetric Beta family (α,β) grid (coarse)
    alphas_grid = np.linspace(0.2, 2.0, 9)
    betas_grid = np.linspace(0.2, 2.0, 9)
    for a in alphas_grid:
        if time.time() > deadline:
            break
        for b in betas_grid:
            if time.time() > deadline:
                break
            beta_shape = (i0 ** (a - 1)) * ((n - i0 + 1) ** (b - 1))
            update_if_better(beta_shape)

    # 6. Sine‑based shapes a_i = sin(π (i+0.5)/n)^γ
    gammas = np.linspace(0.2, 3.0, 15)
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    sin_vals = np.sin(theta)
    for g in gammas:
        if time.time() > deadline:
            break
        shape = sin_vals ** g
        update_if_better(shape)

    # 7. Cosine‑based (U‑shaped) shape a_i = (1 - cos(2π i/(n-1)))^γ
    # This creates high at ends, low middle.
    cos_vals = 1 - np.cos(2 * np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    for g in gammas:
        if time.time() > deadline:
            break
        shape = cos_vals ** g
        update_if_better(shape)

    # ------------------------------------------------------------------
    # 2. Adam descent on smooth max
    if best_seq is None:
        # Fallback: simple exponential
        best_seq = np.exp(-0.5 * np.arange(n, dtype=np.float64))
        best_seq = best_seq / best_seq.sum()
        best_obj = _conv_max(best_seq, L)

    a = best_seq.copy()
    # Re‑evaluate objective for safety
    best_obj = _conv_max(a, L)
    print(f"--- Phase 2: Adam descent (starting obj={best_obj:.8e}) ---")

    # Adam hyper‑parameters
    max_iters = 25000  # will be cut by deadline
    lr0 = 0.1
    beta1, beta2 = 0.9, 0.999
    eps = 1e-10
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    alpha_start, alpha_end = 5.0, 180.0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # convolution
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv.max()
        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _smoothmax_weights(conv, alpha)

        # gradient
        grad = _grad_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # keep on simplex
        a_new = _project_simplex(a_new)

        # evaluate new point
        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"Iter {it:5d}: improved obj={obj_new:.8e} score={2*n*obj_new:.6f}")
        else:
            # keep a even if not improvement (to continue search)
            a = a_new

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing local moves
    print("--- Phase 3: simulated‑annealing hill‑climb ---")
    a_curr = best_seq.copy()
    obj_curr = best_obj
    # initial temperature as a fraction of current objective
    T0 = obj_curr * 0.02
    Tmin = obj_curr * 1e-6
    # exponential cooling
    decay = 0.9995
    T = T0

    while time.time() < deadline:
        # pick source with probability proportional to mass
        i = np.random.choice(n, p=a_curr)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer a random fraction up to 5% of source mass
        delta = (np.random.rand() - 0.5) * 0.04 * a_curr[i]  # can be negative -> move opposite direction
        # enforce non‑negativity
        if a_curr[i] - delta < 0:
            continue
        a_new = a_curr.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project onto simplex is not needed because we kept sum
        obj_new = _conv_max(a_new, L)
        # acceptance
        if obj_new < obj_curr or np.random.rand() < math.exp(-(obj_new - obj_curr) / max(T, 1e-12)):
            a_curr = a_new
            obj_curr = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_seq = a_new.copy()
                print(f"  SA improved obj={obj_new:.8e} score={2*n*obj_new:.6f}")
        # cooling
        T = max(T * decay, Tmin)

    # ------------------------------------------------------------------
    # Final scaling to required sum
    final_seq = (best_seq * target_sum).tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it} Adam iters ---")
    except Exception:
        # evaluate_sequence may not be defined in this isolated environment
        print(f"--- Finished: best raw obj = {best_obj:.8e}, score approx = {2*n*best_obj:.6f} (n={n}) ---")
    return final_seq

"""
Propose a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search proceeds in three stages while respecting a 1000 s time budget.

1. **Parametric sweep** – many simple families (power‑law,
   exponential, beta‑distribution, sinusoidal, U‑shaped, …) are generated,
   normalised to unit sum and the best (smallest max‑convolution) is kept.

2. **Adam refinement** – the current best vector is optimised with Adam
   on a smooth approximation of the supremum norm of the convolution.
   The soft‑max temperature α grows from a low to a high value,
   allowing the algorithm to first see a smooth landscape and then focus
   on the true max.

3. **Simulated‑annealing hill‑climb** – cheap random mass‑transfer moves
   are performed, accepting improvements unconditionally and
   occasional worsening moves according to an exponential cooling schedule.

All calculations use FFT‑based convolution for speed, the iterate is
projected onto the probability simplex after each update, and the final
vector is scaled to the normalisation ``sum = sqrt(2·n)`` required by
the reference code.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max(conv(a, a)) using an FFT of length L (≥2·len(a)-1)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _smoothmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution entries.  Returns a probability vector."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray,
                       phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective.
    a – current probability vector (length n)
    w – soft‑max weights (length 2n‑1)
    phase – pre‑computed phase factor for the reversed FFT
    """
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    n = a.shape[0]
    return 2.0 * corr[n - 1 : n - 1 + n]
