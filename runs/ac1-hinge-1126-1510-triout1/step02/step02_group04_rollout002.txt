"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the sum is normalised to ‖a‖₁ = √(2 n), the evaluation reduces to
minimising the maximum entry of the (linear) self‑convolution.
The algorithm works in two stages:

1. **Extended Adam optimisation** – many initial shapes (uniform, Dirichlet,
   several classic window functions, power‑law families and the previous best)
   are projected onto the simplex via a soft‑max parametrisation.
   A smooth log‑sum‑exp approximation of the max‑operator is used, with a
   linearly increasing temperature β and the Adam optimiser on the unconstrained
   variables.  The loop is run for many iterations and restarts, keeping the
   best sequence found (still on the unit simplex).

2. **Simulated‑annealing refinement** – starting from the best simplex vector
   the algorithm performs random “mass‑transfer’’ moves (δ taken from one index
   and added to another) while monitoring the true max‑convolution.  A simple
   temperature schedule accepts occasional uphill moves, allowing the solution
   to escape shallow local minima.  The best sequence encountered during this
   phase is returned, scaled to the required sum √(2 n) and clipped to the
   range required by ``evaluate_sequence``.

Both stages use FFT‑based convolution (real‑valued rfft/irfft) for speed.
The whole search respects a 1000‑second budget (a small safety margin is
kept before the hard timeout).  Progress and intermediate bounds are printed.
"""

# Helper functions (top‑level, no closures)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_max = np.max(v)
    e = np.exp(v - v_max)
    return e / np.sum(e)


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, L_conv: int) -> float:
    """Return max of the linear convolution a*a (full length 2·len(a)-1)."""
    # real FFT version – a is real, length L_conv ≥ 2*len(a)-1
    fa = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(fa * fa, n=L_conv)[: 2 * a.size - 1]
    return float(np.max(conv))


def _full_correlation(w: np.ndarray, a_rev: np.ndarray, L_corr: int) -> np.ndarray:
    """Full correlation of w with a_rev (both real)."""
    fw = np.fft.rfft(w, n=L_corr)
    fa = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(fw * fa, n=L_corr)
    return corr


def _gradient_a(w: np.ndarray, a: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max approximation w.r.t. a:
    grad_a = 2 * corr(w , a).
    """
    # correlation with reversed a returns length = len(w)+len(a)-1
    corr_full = _full_correlation(w, a[::-1], L_corr)
    n = a.shape[0]
    # the valid segment of length n starts at index n-1
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _generate_windows(n: int) -> list[np.ndarray]:
    """Return a list of deterministic window shapes (already normalised to sum 1)."""
    seqs = []

    # Hann (sin²) – also known as Hanning
    i = np.arange(n, dtype=float)
    hann = np.sin(np.pi * i / (n - 1)) ** 2
    seqs.append(hann / np.sum(hann))

    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    seqs.append(hamming / np.sum(hamming))

    # Blackman
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    seqs.append(blackman / np.sum(blackman))

    # Tukey (alpha = 0.5)
    alpha = 0.5
    tukey = np.ones_like(i)
    half = int(alpha * (n - 1) / 2)
    # rising cosine
    tukey[:half + 1] = 0.5 * (1 + np.cos(np.pi * (-(i[:half + 1]) / (alpha * (n - 1) / 2) + 1)))
    # falling cosine
    tukey[-half:] = 0.5 * (1 + np.cos(np.pi * (i[-half:] / (alpha * (n - 1) / 2) - 1)))
    seqs.append(tukey / np.sum(tukey))

    # Gaussian windows with a few sigma values
    centre = (n - 1) / 2.0
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        gauss = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        seqs.append(gauss / np.sum(gauss))

    # Simple exponential decay (various rates)
    for r in (0.999, 0.998, 0.995):
        exp_decay = r ** i
        seqs.append(exp_decay / np.sum(exp_decay))

    return seqs


def _random_dirichlet_vectors(n: int, count: int) -> list[np.ndarray]:
    """Return a list of `count` random Dirichlet draws on the simplex."""
    vectors = []
    for _ in range(count):
        vec = np.random.dirichlet(np.ones(n))
        vectors.append(vec)
    return vectors

    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------------------
    # Determine length of the candidate (use previous best if present)
    try:
        n = len(height_sequence_1)  # type: ignore
    except Exception:
        n = 5600  # a sane default – close to the size that gave the previous best
    target_sum = math.sqrt(2.0 * n)   # the normalisation used by the reference code
    print(f"--- Optimising sequence of length n={n} (target sum={target_sum:.3f}) ---")

    # pre‑compute FFT lengths
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for correlation (w length 2n-1, a length n)

    # -------------------------------------------------------------
    # Helper for the original (non‑smoothed) objective on the simplex
    def _objective_on_simplex(a: np.ndarray) -> float:
        """Return max(conv(a,a)) – a already sums to 1."""
        return _conv_max(a, L_conv)

    # Keep track of best solution (always stored on the unit simplex)
    best_a = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # Build a reasonably diverse set of initial points on the simplex
    start_vectors = []

    # 1) previous best if available
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        a0 = seq0 / np.sum(seq0)
        start_vectors.append(a0)
    except Exception:
        pass

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) deterministic window shapes
    start_vectors.extend(_generate_windows(n))

    # 4) random Dirichlet draws (small number – good diversity)
    start_vectors.extend(_random_dirichlet_vectors(n, 3))

    # 5) simple power‑law shapes (same as previous code, but with more exponents)
    for p in (0.2, 0.5, 1.0, 1.5, 2.0, 3.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        start_vectors.append(w / np.sum(w))

    # -------------------------------------------------------------
    # Adam hyper‑parameters (shared across restarts)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                 # base learning rate for v (soft‑max parameters)

    beta_start = 5.0
    beta_end = 200.0          # more aggressive final smoothing
    max_iters_per_restart = 15000   # long enough for decent convergence

    # Main Adam loop over all restarts
    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        # initialise unconstrained variable v (soft‑max parameters)
        eps = 1e-12
        v = np.log(a_init + eps)
        m = np.zeros_like(v)      # first moment estimate
        v2 = np.zeros_like(v)     # second moment estimate (Adam's v)

        # per‑restart loop
        for it in range(max_iters_per_restart):
            now = time.time()
            if now > deadline:
                break

            # ------- map to simplex -------
            a = _softmax(v)                      # a ≥ 0, Σa = 1

            # ------- evaluate current objective -------
            conv_max = _conv_max(a, L_conv)      # max of full convolution
            cur_val = conv_max                    # because sum=1

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[restart {restart_id}] iter {it:5d}  new best = {best_val:.6f}")

            # ------- smooth‑max (log‑sum‑exp) weights -------
            beta = beta_start + (beta_end - beta_start) * it / max_iters_per_restart
            # shift by max for numerical stability
            w_exp = np.exp(beta * (np.fft.irfft(np.fft.rfft(a, n=L_conv)**2, n=L_conv)[:2 * n - 1] - conv_max))
            w = w_exp / np.sum(w_exp)

            # ------- gradient w.r.t. a -------
            grad_a = _gradient_a(w, a, L_corr)

            # ------- gradient w.r.t. soft‑max variables v -------
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ------- Adam update -------
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # If Adam produced no candidate (should not happen), fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective_on_simplex(best_a)

    # -------------------------------------------------------------
    # Simulated‑annealing refinement (local random mass‑transfer moves)
    # Work again on the simplex (sum = 1).  The best a from Adam is the start.
    a_cur = best_a.copy()
    cur_val = best_val
    best_local_a = a_cur.copy()
    best_local_val = cur_val

    # annealing schedule parameters
    T_start = best_local_val * 0.05 + 0.001   # small positive temperature
    T_end = 1e-6
    decay = 0.99995   # exponential decay per iteration

    temperature = T_start

    # step size control – maximum amount of mass moved in one proposal
    max_step = 0.02   # 2 % of the total mass per move (adjustable)

    # approximate max number of local iterations given remaining time
    remaining = deadline - time.time()
    # rough cost per iteration ~ 0.001 s (FFT + bookkeeping), be conservative
    max_local_iters = int(remaining / 0.0015)

    i_iter = 0
    while time.time() < deadline and i_iter < max_local_iters:
        i_iter += 1

        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # amount to move from i to j (must be ≤ a_cur[i] and ≤ max_step)
        if a_cur[i] <= 0.0:
            continue
        delta = np.random.rand() * min(a_cur[i], max_step)

        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        new_val = _conv_max(a_new, L_conv)

        # accept if better, otherwise accept with probability exp(-(Δ)/T)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move acceptance
            prob = math.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a_cur = a_new
            cur_val = new_val
            # keep track of global best
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_a = a_new.copy()
                print(f"[SA] iter {i_iter:7d}  new best = {best_local_val:.6f}")

        # temperature schedule
        temperature = max(T_end, temperature * decay)

    # -------------------------------------------------------------
    # Final best sequence (scaled to the appropriate sum)
    best_on_simplex = best_local_a if best_local_val < best_val else best_a
    best_sequence = (best_on_simplex * target_sum).astype(float)

    # Clip to the range enforced by evaluate_sequence (just to be safe)
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)

    print(f"--- Finished: best max‑conv = {np.max(np.convolve(best_sequence, best_sequence)):.6f} "
          f"(equivalently objective = {best_local_val if best_local_val < best_val else best_val:.6f}) "
          f"with n={n} ---")
    return best_sequence.tolist()
