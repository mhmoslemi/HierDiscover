def proj_simplex(v):
    """Projects onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0
    # define n, can be extended.
    default_n = 5000
    # try to reuse existing sequence
    n = default_n
    if 'height_sequence_1' in globals():
        init_seq = globals()['height_sequence_1']
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            arr = np.array(init_seq, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                if arr.size > n:
                    n = arr.size
    # compute FFT lengths
    fft_len = next_pow2(2 * n - 1)
    # initial best candidate: uniform
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # schedule for L-BFGS smoothing
    lam_schedule = [0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003]

    iteration = 0
    while time.time() < deadline:
        # allocate a slice for this iteration (approx). Use small slice ~3 secs
        slice_deadline = min(deadline, time.time() + 3.0)
        # pick mode for initial a
        mode = rng.choice(["random", "exp", "power", "edge", "sine", "best"])
        if mode == "random":
            a0 = rng.random(n)
            a0 /= a0.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n+1, dtype=float) ** (-p))
            a0 /= a0.sum()
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
        elif mode == "sine":
            x = np.arange(n) + 0.5
            a0 = np.sin(np.pi * x / n)
            a0[a0 < 0] = 0
            a0 /= a0.sum()
        else:  # best
            a0 = best_a.copy()
            # add small perturbation
            a0 += rng.normal(scale=0.01, size=n)
            a0 = np.clip(a0, 0, None)
            if a0.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()
        # now subgradient descent
        cand_a, cand_max = subgradient_optimize(a0, fft_len, n, slice_deadline)
        cand_score = 2.0 * n * cand_max
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand_a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] mode={mode} subgrad improved score = {best_score:.6f}")

        # L-BFGS refine
        y_opt = schedule_optimize(best_y, n, fft_len, fft_len, lam_schedule, maxiter_per_stage=60)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

def subgradient_optimize(a_start, fft_len, n, deadline):
    a = a_start.copy()
    # ensure sum=1
    a = np.clip(a, 0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    best_a = a.copy()
    # compute initial max
    conv = conv_fft(a, fft_len, n)
    best_max = np.max(conv)
    iter = 0
    while time.time() < deadline:
        conv = conv_fft(a, fft_len, n)
        max_k = int(np.argmax(conv))
        # compute gradient
        indices = np.arange(n)
        j = max_k - indices
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        # gradient = 2*a[j] for mask
        grad[mask] = 2.0 * a[j[mask]]
        # step size schedule
        step = 0.5 / np.sqrt(iter + 1)
        a_new = a - step * grad
        # projection
        a_new = proj_simplex(a_new)
        # evaluate new max
        conv_new = conv_fft(a_new, fft_len, n)
        max_new = np.max(conv_new)
        # update best if improvement
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        # set a for next iteration
        a = a_new
        iter += 1
        # occasional random perturbation to escape stagnation
        if iter % 200 == 0:
            # random jitter
            jitter = np.random.default_rng().normal(scale=0.01, size=n)
            a = np.clip(a + jitter, 0, None)
            if a.sum() == 0:
                a = np.full(n, 1.0/n)
            else:
                a = a / a.sum()
    return best_a, best_max

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30):
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z
    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n-1):(n-1+n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

"""
A multi‑stage optimizer for the step‑function problem.

The algorithm combines three complementary phases:

1. **Randomised initialisation** – a variety of shapes (random,
   exponential, power‑law, sine, edge‑heavy) are sampled on the
   probability simplex.

2. **Sub‑gradient descent on the exact max‑convolution** – the objective
   `max(conv(a,a))` is nondifferentiable but its sub‑gradient can be
   written explicitly:
   `∂b_k/∂a_i = 2·a_{k-i}` where `k = argmax(conv)`.  After a gradient
   step the iterate is projected onto the simplex.  This quickly drives
   the peak of the convolution down while preserving the sum‑to‑one
   constraint.

3. **Smooth‑max L‑BFGS refinement** – the L‑BFGS continuation described
   in the reference code is applied to the soft‑max parametrisation
   `a = softmax(y)`.  A decreasing smoothing schedule turns the smooth
   max into the true max, polishing the solution found by the
   sub‑gradient phase.

A short greedy hill‑climb (mass‑transfer moves) is performed after each
outer iteration to capture any remaining improvement.  The best
sequence encountered is finally evaluated with the provided
`evaluate_sequence` function and returned.

The whole search respects the 1000 s time budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the official objective: 2·n·max(a*a)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x = 1}.
    """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser for the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(a_start: np.ndarray,
                         fft_len: int,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) by sub‑gradient descent.
    Returns the best vector found and its maximal convolution entry.
    """
    a = np.clip(a_start, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    best_a = a.copy()
    conv = conv_fft(a, fft_len, n)
    best_max = np.max(conv)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of b_k = a * a at index k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step‑size
        step = 0.5 / math.sqrt(it + 1.0)
        a_new = a - step * grad
        a_new = proj_simplex(a_new)

        # evaluate
        conv_new = conv_fft(a_new, fft_len, n)
        max_new = np.max(conv_new)

        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        # occasional random jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            a = proj_simplex(np.clip(a + jitter, 0.0, None))

    return best_a, best_max


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Keeps a move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move ≤5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS refinement (same as the reference implementation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y."""
    a = softmax(y)

    # --------------- convolution a*a -----------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --------------- gradient of smooth max -------------
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_schedule: list[float],
                      maxiter_per_stage: int = 120) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0    # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    default_n = 5000
    n = default_n
    # optionally reuse a previously discovered construction
    if "height_sequence_1" in globals():
        seq = globals()["height_sequence_1"]
        try:
            arr = np.asarray(seq, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                if arr.size > n:
                    n = int(arr.size)
        except Exception:
            pass

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial best: uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # use the previous construction if available
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                tmp = np.clip(tmp, 0.0, None)
                if tmp.sum() > 0.0:
                    tmp = tmp / tmp.sum()
                    if tmp.size == n:
                        best_a = tmp.copy()
                    else:
                        # pad or truncate to the chosen n
                        if tmp.size < n:
                            pad = np.zeros(n - tmp.size)
                            best_a = np.concatenate([tmp, pad])
                        else:
                            best_a = tmp[:n]
                    best_score = fast_score(best_a, fft_len, n)
                    best_y = np.log(best_a + 1e-12)
                    print(f"[init] reused height_sequence_1 with score = {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003]

    iteration = 0

    while time.time() < deadline:
        # each outer iteration gets a short time slice for the sub‑gradient phase
        slice_deadline = min(deadline, time.time() + 4.0)

        # --------------------------------------------------------------
        # Random initialisation of a (on the simplex)
        # --------------------------------------------------------------
        mode = rng.choice(["random", "exp", "power", "edge", "sine", "best"])
        if mode == "random":
            a0 = rng.random(n)
            a0 /= a0.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
        elif mode == "sine":
            x = np.arange(n) + 0.5
            a0 = np.sin(np.pi * x / n)
            a0[a0 < 0] = 0.0
            a0 /= a0.sum()
        else:          # mode == "best"
            a0 = best_a.copy()
            a0 += rng.normal(scale=0.02, size=n)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()

        # --------------------------------------------------------------
        # Sub‑gradient descent on exact max‑convolution
        # --------------------------------------------------------------
        cand_a, cand_max = subgradient_optimize(a0, fft_len, n, slice_deadline)
        cand_score = 2.0 * n * cand_max
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand_a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] mode={mode} sub‑grad improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Smooth‑max L‑BFGS refinement (continuation)
        # --------------------------------------------------------------
        y_opt = schedule_optimize(best_y, n, fft_len, fft_len_g,
                                  lam_schedule, maxiter_per_stage=60)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] L‑BFGS refined score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Aggressive local hill‑climb (mass transfer)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        hc_deadline, max_iter=3000)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
