def proj_simplex(v):
    # Projects onto simplex {x>=0, sum=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑evaluation step sequence.

The algorithm first builds a few sensible “seed’’ sequences (uniform,
Hann‑window, exponential, beta‑pdf …) and keeps the best one.
Starting from the current best it performs a continuation L‑BFGS
optimisation on a smooth approximation of the max‑convolution entry.
After each L‑BFGS run a cheap projected sub‑gradient descent on the
exact (non‑smooth) objective is applied, followed by a greedy hill‑climb
that moves small amounts of mass between indices.  The whole loop is
repeated with many random restarts until the time budget runs out.
All operations are vectorised and use FFT‑based convolutions, so the
method scales to several thousand coefficients within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, sum(x)=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    The smoothing width is lam = max(conv) * lam_factor.
    a = softmax(y) is the probability vector (sum = 1).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a small random fraction of mass from i to j.
    Keep the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    max_iter: int = 10000,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective
    max_k (a * a)[k].  The step size follows η_t = init_step / sqrt(t).
    Each iteration needs one FFT‑based convolution to locate the active index.
    """
    a = a.copy()
    t = 1
    while time.time() < deadline and t <= max_iter:
        # compute convolution and the index of the maximal entry
        conv = conv_fft(a, fft_len, n)
        k = int(conv.argmax())

        # gradient of the active anti‑diagonal (sub‑gradient)
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k :: -1]
        else:
            size = 2 * n - 1 - k
            start = k - (n - 1)
            grad[start:] = 2.0 * a[:size][::-1]

        # step size schedule
        eta = init_step / np.sqrt(t)
        a = proj_simplex(a - eta * grad)
        t += 1
    return a


def generate_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a handful of deterministic starting sequences.
    All are normalised to sum = 1.
    """
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Hann window (zero at ends, peak = 1)
    i = np.arange(n)
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    candidates.append(hann / hann.sum())

    # Cosine “raised‑cosine’’ window (similar to Hann but a bit narrower)
    cosine = np.cos(np.pi * (i - (n - 1) / 2) / (n - 1))
    cosine = np.maximum(cosine, 0.0)          # clip negative lobes
    candidates.append(cosine / cosine.sum())

    # Exponential decay
    rate = 0.0015
    exp_decay = np.exp(-rate * i)
    candidates.append(exp_decay / exp_decay.sum())

    # Power‑law decay
    p = 1.2
    power = (i + 1) ** (-p)
    candidates.append(power / power.sum())

    # Deterministic Beta‑pdf (α,β in (0.5,3))
    α, β = 0.8, 2.2
    x = i / (n - 1)
    beta_pdf = (x ** (α - 1)) * ((1 - x) ** (β - 1))
    beta_pdf[beta_pdf == 0] = 1e-12
    candidates.append(beta_pdf / beta_pdf.sum())

    # Random Dirichlet draws (added later inside the main loop)

    return candidates

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build a set of seed vectors and keep the best according to fast_score
    # ------------------------------------------------------------------
    seed_vectors = generate_initial_candidates(n)

        prev_a = np.asarray(prev, dtype=float).ravel()
        if prev_a.shape[0] == n:

            seed_vectors.append(prev_a)
    except Exception:

    if best_a is None:
        # fallback – uniform (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len_conv, n)

    best_y = np.log(best_a + 1e-12)   # soft‑max parameters corresponding to best_a

    # ------------------------------------------------------------------

    lam_schedule = [

        elif mode == "power":

        elif mode == "beta":
