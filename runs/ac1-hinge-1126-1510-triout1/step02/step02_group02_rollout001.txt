Propose a candidate step‑function that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm improves over the reference implementation in three
directions:

1. **Richer parametric sweep** – families not only power‑law and pure
   exponentials but also two‑sided mixtures, Gaussian bumps and
   cosine‑shaped windows.  All shapes are normalised to unit sum and the
   best one (in terms of `max(conv)`) is kept.

2. **Two‑stage optimisation** –  
   *Stage A*: Adam on a smooth approximation of the max (soft‑max) with a
   much longer schedule (up to ≈ 1 × 10⁵ iterations) and a slower rise of
   the smoothing parameter.  
   *Stage B*: “hard” sub‑gradient descent on the true max using the
   active set of binding lags.  The update is an exponentiated‑gradient
   (mirror descent on the simplex), which respects non‑negativity
   automatically and often yields a finer reduction of the supremum.

3. **Final annealed hill‑climbing** – cheap random mass transfers with a
   simple Metropolis acceptance rule to escape tiny plateaus.

All steps respect a user‑supplied time budget (default 1000 s).  The
result is returned as a plain Python list scaled so that
`sum(sequence) == sqrt(2·n)`, which is the normalisation expected by
`evaluate_sequence`.  The code uses only NumPy, math and time and does
not rely on any file system or network access.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_autocorr(a: np.ndarray, L: int, n: int) -> np.ndarray:
    """Linear (non‑circular) autocorrelation a * a of length 2·n‑1."""
    fft_a = np.fft.fft(a, n=L)
    prod = fft_a * fft_a
    conv = np.real(np.fft.ifft(prod))[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_via_weights(a: np.ndarray, w: np.ndarray,
                          fft_a_rev: np.ndarray,
                          L: int, n: int) -> np.ndarray:
    """
    Compute 2 * ∂/∂a  (w·conv(a,a)) where w is a weight vector of length 2·n‑1.
    The gradient is obtained by a convolution of the reversed a with w.
    """
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            print(f"Sub‑iter {sub_it:6d}: improved | obj={obj_new:.8e} | score={2 * n * obj_new:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 4. Random local moves with a simple Metropolis acceptance ---------------
    print("--- Phase 4: annealed random local moves ---")
    a_current = best_seq.copy()
    cur_obj = best_obj
    while time.time() < deadline:
        # pick a source index with probability proportional to its mass
        src = np.random.choice(n, p=a_current)
        dst = np.random.randint(0, n)
        if src == dst:
            continue
        # transfer a random fraction (up to 5 %) of the source mass
        delta = np.random.rand() * 0.05 * a_current[src]
        if delta <= 0.0:
            continue
        a_new = a_current.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        if a_new[src] < 0.0:
            continue

        # evaluate
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        # simple Metropolis rule (temperature decays with time)
        elapsed = time.time() - start_time
        remaining = max(deadline - time.time(), 1e-9)
        temperature = 0.01 * (remaining / (budget_s - 2.0))   # small temperature
        accept = obj_new < cur_obj or np.random.rand() < math.exp(-(obj_new - cur_obj) / max(temperature, 1e-12))
        if accept:
            a_current = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj, best_seq = obj_new, a_new.copy()
                print(f"  move improved | obj={obj_new:.8e} | score={2 * n * obj_new:.6f}")

    # ------------------------------------------------------------------
    # 5. Final scaling to the reference normalisation -------------------------
    final_seq = (best_seq * target_sum).tolist()
    # Optional: one final call to the provided evaluator for logging purposes.
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it + sub_it} iterations ---")
    except Exception:
        # In the unlikely event `evaluate_sequence` is unavailable.
        print(f"--- Finished: best raw obj = {best_obj:.6e} (n={n}) after {it + sub_it} iterations ---")
    return final_seq
