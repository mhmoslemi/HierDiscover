    n = len(f)
    # Build A_ub matrix shape (2n-1, n)

    # Actually building via for-loop: A = np.zeros((2*n-1, n))

    if res.success:

    return g0

"""
Hybrid optimisation for the self‑convolution bound.

* Start from a good deterministic seed (half‑order series or the
  previously found `height_sequence_1`).
* Occasionally improve the current sequence by solving a linear program:
  maximise the total mass of a new sequence g under the same maximal
  self‑convolution bound as the current f.
* The main loop alternates cheap random pair‑exchanges (simulated
  annealing) with a smoothed sub‑gradient step:
    – a soft‑max over the convolution entries provides a smooth
      approximation of the “max” operator,
    – the resulting gradient is obtained efficiently by an FFT‑based
      correlation,
    – an exponentiated‑gradient / projected‑gradient update keeps the
      vector on the probability simplex.
* Step‑sizes and temperatures are gradually reduced; occasional jitter
  helps escape plateaus.
* The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i, normalised to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """Matrix A such that (A @ g) = f * g (linear convolution, length 2·n‑1)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        A[k, idx] = f[k - idx]
    return A


def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max 1ᵀg   subject to  f * g ≤ M  (entry‑wise) and g ≥ 0.
    Returns the optimal g (may be all‑zero if the LP fails).
    """
    A = _build_convolution_matrix(f)          # shape (2n‑1, n)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones_like(f)                      # maximise sum(g)  → minimise -sum(g)
    # use HiGHS via scipy's 'highs' method – fast and reliable
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    # fallback – uniform distribution
    return np.full_like(f, 1.0 / f.shape[0])

            a_pad = np.zeros(N, dtype=np.float64)
            w_pad[:L] = w
            a_pad[:n] = cur_seq
            W = np.fft.fft(w_pad)
            A = np.fft.fft(a_pad)
            corr = np.fft.ifft(W * np.conj(A)).real
            grad = 2.0 * corr[:n]                         # gradient of the smoothed max

            # projected (or exponentiated) gradient step
            trial_step = step_g
            improved = False
            for _ in range(5):                           # back‑tracking line search
                cand = cur_seq - trial_step * grad
                cand = _project_simplex(cand)
                cand_conv = np.convolve(cand, cand)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    cur_seq = cand
                    cur_score = cand_score
                    cur_max = cand_conv.max()
                    improved = True
                    break
                trial_step *= 0.5

            if improved:
                step_g = min(step_g * 1.1, step_g_max)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[{iteration:6d}] Gradient step improved best to {best_score:.6f}")
            else:
                step_g = max(step_g * 0.7, step_g_min)
                if iteration % 1000 == 0:                # occasional jitter
                    sigma = 0.001
                    noise = np.random.randn(n) * sigma
                    jitter = cur_seq * (1.0 + noise)
                    jitter = np.clip(jitter, 0.0, None)
                    jitter /= jitter.sum()
                    cur_seq = jitter
                    cur_score = 2.0 * n * np.convolve(cur_seq, cur_seq).max()
                    cur_max = np.convolve(cur_seq, cur_seq).max()

        # --------------------------------------------------------------
        # c) periodic LP refinement (global improvement)
        # --------------------------------------------------------------
        if iteration % lp_interval == 0:
            try:
                g0 = _lp_optimize(cur_seq, cur_max)
                if g0.sum() > 0:
                    g0 = g0 / g0.sum()
                    cand = (1.0 - lp_mix) * cur_seq + lp_mix * g0
                    cand = cand / cand.sum()
                    cand_conv = np.convolve(cand, cand)
                    cand_score = 2.0 * n * cand_conv.max()
                    if cand_score < cur_score:
                        cur_seq = cand
                        cur_score = cand_score
                        cur_max = cand_conv.max()
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = cur_seq.copy()
                            print(f"[{iteration:6d}] LP refinement improved best to {best_score:.6f}")
            except Exception:
                pass  # LP occasional numerical issues are ignored

        # --------------------------------------------------------------
        # progress report
        # --------------------------------------------------------------
        if iteration % 2000 == 0:
            print(f"[{iteration:6d}] elapsed {elapsed:.1f}s best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
