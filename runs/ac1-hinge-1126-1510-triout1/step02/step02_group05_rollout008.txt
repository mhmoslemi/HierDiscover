"""
Propose a low‑score height sequence.

Algorithm
---------
* Work on the simplex {a_i ≥ 0, Σ a_i = 1}.  The evaluation score is
  2·n·max(conv(a,a)), so we only need to minimise the peak of the
  self‑convolution.
* For each restart we pick a length `n` (default 4000) and initialise a
  random power‑law distribution.
* Optimisation is a **projected accelerated sub‑gradient method**:
    – The active set of indices where the convolution attains its maximum
      is identified (typically a single shift).
    – The sub‑gradient is the (equal) sum of the gradients of all active
      shifts:  ∂conv[k]/∂a_i = 2·a_{k‑i} (zero when the index is out of range).
    – A momentum term (`v`) implements Nesterov‑style acceleration.
    – After each step we project back onto the probability simplex with the
      O(n) algorithm of Duchi et al.
* Stagnation is broken by small random jitter and the momentum buffer is
  reset.
* Many independent restarts are performed until the time budget expires;
  the best sequence seen overall is returned.

The method explores many initial points, uses a more aggressive update
(adding momentum) and occasional perturbations, which typically pushes the
score below the previous best (≈1.527) and can reach the target ≈1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σ x = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """Full linear convolution a*a (via FFT) and its maximal entry."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())

# Optimisation core
# ----------------------------------------------------------------------


def active_subgradient_optimize(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    mu: float,
    deadline: float,

    Projected accelerated sub‑gradient descent for min max(conv(a,a)).

    Parameters
    ----------
    a0 : np.ndarray
        Starting point on the simplex (will be normalised internally).
    max_iters : int
        Upper bound on iteration count for this run.
    step0 : float
        Initial step size (scaled by 1/√iter).
    mu : float
        Momentum coefficient (0 ≤ mu < 1).
    deadline : float
        Unix timestamp after which the routine must stop.

    Returns

    best_seq : np.ndarray
        Best sequence encountered.
    best_score : float
        Corresponding evaluation score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]

    # Normalise to the simplex (sum = 1)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_size = 1 << ((2 * n - 1).bit_length())  # next power of two

    best_seq = a.copy()

    v = np.zeros_like(a)  # momentum buffer
    last_improve = 0

    # tolerances for active‑set detection
    atol = 1e-14
    reltol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and current score ----
        conv, max_b = compute_conv_max(a, fft_size)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            last_improve = it
            if it % 200 == 0:
                print(f"[it {it}] new best score = {score:.6f}")

        # ---- active set (indices where conv is (almost) maximal) ----
        active = np.where(conv >= max_b - max(atol, max_b * reltol))[0]

        # ---- sub‑gradient (sum of gradients of all active shifts) ----
        grad = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i_vals = np.arange(i_low, i_high + 1)
            j_vals = k - i_vals
            grad[i_vals] += 2.0 * a[j_vals]

        if active.size > 1:
            grad /= float(active.size)

        # ---- momentum step ----
        step = step0 / np.sqrt(it)
        v = mu * v - step * grad
        a_candidate = a + v
        a = project_simplex(a_candidate)

        # ---- occasional jitter to escape stagnation ----
        if it - last_improve > 2000:
            jitter = 0.008 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            v.fill(0.0)          # reset momentum
            last_improve = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.

    The routine performs many independent restarts, each running the
    accelerated sub‑gradient optimiser described above.  The best sequence
    seen across all restarts (any length) is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Candidate lengths – we try a few values, favouring the previously
    # successful size (4000) but also exploring neighbours.
    # ------------------------------------------------------------------
    candidate_lengths = [3000, 3500, 4000, 4500, 5000]

    # ------------------------------------------------------------------
    # Global best across all restarts
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Parameters for the inner optimiser
    # ------------------------------------------------------------------
    max_iters_per_restart = 3000           # generous but fast per run
    step0_initial = 1.0
    momentum_mu = 0.9

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ---- pick a length for this restart ----
        n = int(np.random.choice(candidate_lengths))

        # ---- initialise a fresh starting point ----
        # random power‑law with a small random offset
        p = np.random.uniform(0.35, 0.75)
        idx = np.arange(1, n + 1, dtype=np.float64)
        a0 = idx ** (-p)
        if np.random.rand() < 0.2:
            a0 += 0.02 * np.random.randn(n)      # small jitter
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0_sum

        # Occasionally seed from a previously found good sequence (if any)
        if global_best_seq is not None and np.random.rand() < 0.15:
            # resize previous best to the new length by interpolation
            old = np.array(global_best_seq, dtype=np.float64)
            if len(old) != n:
                x_old = np.linspace(0, 1, len(old))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
                a0 = np.maximum(a0, 0.0)
                a0_sum = a0.sum()
                if a0_sum > 0:
                    a0 /= a0_sum
                else:
                    a0 = np.full(n, 1.0 / n)

        # ---- decide how many iterations we can afford ----
        time_left = deadline - time.time()
        # empirical cost ≈ 0.003 s per iteration (convolution dominates)
        it_budget = min(max_iters_per_restart, int(time_left / 0.0035))
        if it_budget <= 0:
            break

        # ---- run the optimiser ----
        seq, score = active_subgradient_optimize(
            a0,
            max_iters=it_budget,
            step0=step0_initial,
            mu=momentum_mu,
            deadline=deadline,
        )

        # ---- keep the best overall ----
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new global best = {score:.6f} "
                f"(n={n}, iters={it_budget})"
            )

        # ---- slowly decay the step size if progress stalls ----
        if restart % 7 == 0:
            step0_initial = max(0.2, step0_initial * 0.85)

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()

    # Return as a plain list as required by the external harness.
    return global_best_seq.tolist()
