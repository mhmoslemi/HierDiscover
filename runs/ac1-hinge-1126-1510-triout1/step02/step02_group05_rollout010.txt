"""
Projected accelerated gradient descent with smooth max‐approximation.

The score to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the objective is homogeneous of degree 0 we can work on the
probability simplex (Σa_i = 1, a_i ≥ 0).  The non‑smooth “max’’ is
replaced by a soft‑max with a temperature λ that is slowly increased.
The gradient of the smoothed objective can be written in closed form
as

    grad = a + (a_rev (*) p)[n‑1 : n‑1+n]

where a_rev is a reversed copy of a, p is the soft‑max probability
vector over the full self‑convolution and (*) denotes a linear (FFT‑
based) convolution.  After each gradient step a projection onto the
simplex (O(n) Duchi et al. algorithm) restores feasibility.
A simple momentum term (heavy‑ball) accelerates convergence.
The routine runs until the allotted time budget expires, keeps the
best sequence seen and finally performs a short stochastic refinement
pass.  The returned list can be fed directly into `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n) algorithm of Duchi et al.)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, exact linear convolution)
# ----------------------------------------------------------------------


def full_convolve_fft(x: np.ndarray, y: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the linear convolution of x and y using an FFT of length size_fft."""
    X = np.fft.rfft(x, n=size_fft)
    Y = np.fft.rfft(y, n=size_fft)
    conv = np.fft.irfft(X * Y, n=size_fft)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def optimise_sequence(
    a0: np.ndarray,
    deadline: float,
    lambda_start: float = 0.5,
    lambda_end: float = 20.0,
    lambda_ramp: int = 5000,
    step0: float = 0.8,
    momentum: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution objective on the simplex
    using accelerated projected gradient descent.

    Returns the best sequence (as a numpy array) encountered and its
    (exact) evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT size sufficient for both a*a (length 2n‑1) and a_rev*p (length 3n‑2)
    size_fft = 1 << ((3 * n).bit_length())

    # momentum vector
    v = np.zeros_like(a)

    best_seq = a.copy()
    # exact score (no smoothing) – compute once
    A = np.fft.rfft(a, n=size_fft)
    b_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    best_score = 2.0 * n * b_full.max() / (a.sum() ** 2)

    it = 0
    while time.time() < deadline:
        it += 1

        # ------- current temperature (smoothness) ----------
        if it <= lambda_ramp:
            lam = lambda_start + (lambda_end - lambda_start) * (it / lambda_ramp)
        else:
            lam = lambda_end

        # ------- full self‑convolution --------------------
        A = np.fft.rfft(a, n=size_fft)
        b_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
        max_b = b_full.max()

        # ------- exact score (for tracking) ---------------
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[iter {it:>6}] new best score = {score:.6f}")

        # ------- soft‑max over convolution entries ----------
        # stability trick: subtract the max before exp
        shift = max_b
        exp_vals = np.exp(lam * (b_full - shift))
        p = exp_vals / exp_vals.sum()   # p is a probability vector, Σp = 1

        # ------- gradient of the smoothed objective -------
        a_rev = a[::-1]
        # convolution a_rev (*) p
        conv_rev = full_convolve_fft(a_rev, p, size_fft)
        term = conv_rev[n - 1 : n - 1 + n]   # length n slice
        grad = term + a                      # gradient w.r.t a

        # ------- step‑size (standard diminishing) -------
        step = step0 / np.sqrt(it)

        # ------- heavy‑ball momentum update -------------
        v = momentum * v - step * grad
        a_candidate = a + v

        # ------- projection back onto the simplex --------
        a = project_simplex(a_candidate)

    # final exact score for the best sequence found
    A = np.fft.rfft(best_seq, n=size_fft)
    b_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    best_score = 2.0 * n * b_full.max() / (best_seq.sum() ** 2)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic refinement (small random perturbations)
# ----------------------------------------------------------------------


def stochastic_refinement(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Apply a cheap random‑walk search around `seq` until time runs out."""
    n = seq.shape[0]
    size_fft = 1 << ((3 * n).bit_length())

    best_seq = seq.copy()
    A = np.fft.rfft(best_seq, n=size_fft)
    b_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    best_score = 2.0 * n * b_full.max() / (best_seq.sum() ** 2)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # Gaussian perturbation, keep it small
        eps = 0.002 * rng.standard_normal(n)
        cand = best_seq + eps
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        A = np.fft.rfft(cand, n=size_fft)
        b_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
        score = 2.0 * n * b_full.max() / (cand.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[refine {it}] improved score = {score:.6f}")

    return best_seq, best_score
