"""
Propose a candidate step‑function (non‑negative heights) that tries to
minimise 2 n·max(conv(a,a)) / (sum(a)**2).  

The search has two phases:

1. **Broad random exploration** – a few thousand candidates are drawn from
   many families (power‑law, exponential, two‑sided power‑law, arcsine‑like,
   cosine and Gaussian windows) with a variety of lengths.  The best
   candidate (lowest fast_score) is kept.

2. **Multiplicative sub‑gradient descent** – starting from the best
   candidate, we iteratively reduce the peak of the self‑convolution.
   For the peak index `m` the sub‑gradient is
       g_i = 2·a_{m‑i}  (where defined, otherwise 0).
   A multiplicative update
       a_i ← a_i·exp(−η·g_i)
   keeps the variables non‑negative; a final normalisation projects onto
   the probability simplex.  The step size η slowly decays, occasional
   jitter is added to escape plateaus and the current best solution is
   stored whenever it improves.  The whole process respects the 1000 s
   budget.

The returned list is ready for the provided `evaluate_sequence` function.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using FFT based convolution.  The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  The unnormalised sequence behaves like
    1/√i and gives a near‑flat self‑convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            i = np.arange(n, dtype=np.float64)
            cand = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))  # Hann window

        elif family == "gaussian":
            sigma = random.uniform(0.07, 0.30) * n
            mid = (n - 1) / 2.0
            i = np.arange(n, dtype=np.float64)
            cand = np.exp(-0.5 * ((i - mid) / sigma) ** 2)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new random best ({family}) n={n} score={sc:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣  Multiplicative sub‑gradient refinement
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    fft_len = next_pow2(2 * n)

    iteration = 0
    eta0 = 8.0               # initial step‑size factor
    tau = 5000.0             # controls decay of η
    eval_interval = 2000    # how often to re‑evaluate the fast score
    jitter_interval = 5000  # how often to add a tiny random perturbation
    jitter_scale = 0.01      # relative magnitude of the perturbation

    while time.time() < deadline - 1.0:
        iteration += 1

        # ----- convolution & peak index -----
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv_full))

        # ----- sub‑gradient of the peak (active set) -----
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1)
        j_range = max_idx - i_range
        g[i_range] = 2.0 * a[j_range]

        # ----- multiplicative update with decaying step size -----
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # ----- projection onto the simplex (simple normalisation) -----
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- occasional jitter to avoid stagnation -----
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum <= 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

        # ----- periodic evaluation and bookkeeping -----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] iter {iteration}, new best score={cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
