"""
Propose a step‐function sequence with a low evaluation score.

The algorithm combines several complementary heuristics:

1.  **Soft‑max parametrisation** – a vector `y` lives in ℝⁿ and
    `a = softmax(y)` guarantees a ≥ 0 and Σa = 1, eliminating the
    explicit simplex projection in most places.

2.  **Continuation on a smooth max** – we minimise
    `2·n·smooth_max(a*a)` where the smooth max is a log‑sum‑exp
    approximation of `max(a*a)`.  The smoothing factor λ is decreased
    gradually (λ‑schedule) so the optimiser first sees a smooth landscape
    and then refines the true max.

3.  **L‑BFGS** – SciPy’s bounded L‑BFGS solver is used on the smooth
    objective (gradient supplied analytically via FFT).  Random restarts
    with diverse initial shapes (uniform, exponential, power‑law,
    edge‑heavy, previous best) are performed until the time budget is
    exhausted.

4.  **Projected sub‑gradient** – after a few L‑BFGS iterations a
    sub‑gradient step on the genuine max‑objective is applied.  The step
    size follows a 1/√t schedule and each iterate is projected back onto
    the simplex.

5.  **Greedy / simulated‑annealing hill‑climb** – a fast mass‑transfer
    routine repeatedly moves a small fraction of mass from one index to
    another; it accepts any improving move and, with a low temperature,
    occasionally accepts worsening moves to escape local minima.

6.  **Final polish** – a longer hill‑climb runs until the deadline.

All heavy convolutions are performed with FFT (O(n log n)).  The search
works with `n = 6000` (≈ 6 k entries) – large enough to give the
algorithm flexibility while still being fast.  The best sequence found
is returned as a plain list of floats.

The function `propose_candidate` is the entry point; it must be
importable and callable by the evaluation harness.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already on the simplex."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------
def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution of a_rev with w gives ∂/∂a_i of the smoothed max
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Directly minimise 2·n·max(a*a) using a sub‑gradient method."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set mask
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient = 2 * (mask * a_rev) restricted to n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy / annealed hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len_conv: int,
               n: int,
               deadline: float,
               max_iter: int = 3000,
               temperature: float = 0.0):
    """
    Mass‑transfer hill‑climb.
    If temperature > 0 a simulated‑annealing acceptance rule is used.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            it += 1
            continue
        # move a random fraction of the mass at i to j
        delta = best[i] * rng.random() * 0.07  # up to 7 % transfer
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len_conv, n)

        accept = False
        if new_score < best_score - 1e-12:
            accept = True
        elif temperature > 0.0:
            # Metropolis acceptance
            prob = math.exp(-(new_score - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            best_score = new_score
            # keep move
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Initial‑shape generators (logits y)
# ----------------------------------------------------------------------
def init_logits(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting logit vector `y` for the soft‑max parametrisation.
    Supported modes:
        - 'prev'   : use the previous best sequence if available
        - 'uniform': all equal (logits zero)
        - 'random' : standard normal
        - 'exp'    : exponential decay
        - 'power'  : power‑law decay
        - 'edge'   : heavy mass at both ends
    """
    if mode == 'prev' and 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=float)
        if seq.size < n:
            # pad with tiny uniform mass
            pad = np.full(n - seq.size, 1e-12)
            seq = np.concatenate([seq, pad])
        elif seq.size > n:
            seq = seq[:n]
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            seq = np.full(n, 1.0 / n)
        else:
            seq = seq / seq.sum()
        return np.log(seq + 1e-12)

    if mode == 'uniform':
        return np.zeros(n)

    if mode == 'random':
        return rng.standard_normal(n)

    if mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
        a /= a.sum()
        return np.log(a + 1e-12)

    if mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        return np.log(a + 1e-12)

    # edge‑heavy: a small bump at the two borders, flatter middle
    if mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
        a /= a.sum()
        return np.log(a + 1e-12)

    # fallback to uniform
    return np.zeros(n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises

        E(a) = 2·n·max(a*a) / (sum a)^2

    The routine uses a mixture of smooth‑max L‑BFGS, sub‑gradient descent,
    and stochastic hill‑climbing, all within the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderately large n gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 6000
    a0 = np.full(n, 1.0 / n)                 # start from the uniform simplex point

    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient‑related convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # continuation schedule for the smooth max (large → small smoothing)
    lam_schedule = [0.6, 0.4, 0.25, 0.15, 0.08, 0.05, 0.02, 0.01]

    iteration = 0
    modes = ['prev', 'uniform', 'random', 'exp', 'power', 'edge']

    while time.time() < deadline:
        # pick a starting shape
        mode = rng.choice(modes)
        y0 = init_logits(mode, n, rng)

        # run the L‑BFGS continuation
        y_cur = y0.copy()
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            res = minimize(fun=objective_and_grad,
                           x0=y_cur,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 180,
                                    'gtol': 1e-9,
                                    'disp': False})
            y_cur = res.x
            a_cur = softmax(y_cur)
            cur_score = fast_score(a_cur, fft_len_conv, n)

            # short stochastic hill‑climb after each smoothing level
            climb_deadline = min(deadline, time.time() + 0.35)
            a_cur, cur_score = hill_climb(a_cur, cur_score,
                                          fft_len_conv, n,
                                          climb_deadline,
                                          max_iter=800,
                                          temperature=0.0)
            # use the refined a as the new start for the next λ
            y_cur = np.log(a_cur + 1e-12)

        # after the continuation chain evaluate the candidate
        final_score = fast_score(a_cur, fft_len_conv, n)
        if final_score < best_score:
            best_score = final_score
            best_a = a_cur.copy()
            print(f"[{iteration}] new best (mode={mode}) score={best_score:.6f}")

        # every few iterations launch a modest sub‑gradient refinement
        if iteration % 15 == 0 and time.time() < deadline - 6.0:
            sg_deadline = min(deadline, time.time() + 5.0)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[{iteration}] sub‑gradient refined score={best_score:.6f}")

        # occasional annealed hill‑climb to escape local minima
        if iteration % 7 == 0 and time.time() < deadline - 4.0:
            anneal_deadline = min(deadline, time.time() + 3.0)
            best_a, best_score = hill_climb(best_a, best_score,
                                            fft_len_conv, n,
                                            anneal_deadline,
                                            max_iter=1500,
                                            temperature=0.005)
            print(f"[{iteration}] annealed climb score={best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing – a longer hill‑climb with a tiny temperature
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=5000,
                                    temperature=0.001)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
