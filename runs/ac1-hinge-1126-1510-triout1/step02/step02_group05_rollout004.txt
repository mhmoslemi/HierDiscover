def build_convolution_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # For each j (col), the column is f shifted down by j (i.e., A[k,j] = f[k-j] if 0<=k-j <n else 0)
    for j in range(n):
        # start row where f index = 0: k = j
        # length = n
        # so rows from j to j+n-1 inclusive get f values
        A[j:j+n, j] = f
    return A

    best_score = float('inf')
    it = 0

        score = M_cur  # since sum=s_target

        # build A matrix

        # Fill by columns
        for j in range(n):

        b_ub = M_cur * np.ones(2*n-1, dtype=np.float64)

"""LP‑based optimizer for the self‑convolution peak.

This implementation follows the iterative linear‑program scheme
described in the problem statement.  Starting from a power‑law
distribution we repeatedly:
  1. Compute the current peak M = max(conv(f,f)).
  2. Solve a linear program that maximises the ℓ₁‑norm of a
     candidate g subject to the linear constraints (f * g) ≤ M.
  3. Rescale g to have the same ℓ₁‑norm as f and mix it with f.
The mixing step size decays with the iteration count.
The algorithm runs until the allocated time budget expires and
returns the best sequence observed.
"""

import time
import math
import numpy as np

# SciPy is allowed and provides a fast LP solver.
from scipy.optimize import linprog


def compute_conv_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return the full linear convolution a*a and its maximum entry."""
    n = a.shape[0]
    # Use next power‑of‑two for efficient FFT.
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv, float(conv.max())


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix A such that A @ g = conv(f, g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # Each column j is the vector f shifted down by j positions.
    for j in range(n):
        A[j : j + n, j] = f
    return A


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   maximise Σ g_j   s.t.   conv(f, g) ≤ M , g ≥ 0.

    Returns the optimal g (unscaled) or None on failure.
    """
    n = f.shape[0]
    A = build_convolution_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimise -sum(g)  → maximise sum(g)
    # bounds for each variable: g_j ≥ 0, no explicit upper bound.
    bounds = [(0.0, None)] * n
    # Use HiGHS – the default for recent SciPy builds – which is fast for dense problems.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x
