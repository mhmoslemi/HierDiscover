"""
Hybrid of linear-programming alternating projection and local refinement.
The algorithm alternates:
1. Solve a linear program max sum(b) s.t. f*b <= M (self-convolution bound).
   Scale the solution to sum 1 and (if it improves the max‑convolution) replace f.
2. After the bound stops improving, perform a projected sub‑gradient descent
   on the L∞‑convolution norm together with occasional random pair exchanges.
The whole search stops after the allotted time (default 1000 s) and returns
the best sequence found (list of non‑negative floats, sum≈1).
"""
import time
import math
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n matrix T such that T @ b = f * b (linear convolution)."""
    n = f.shape[0]
    # each diagonal offset d (0 ≤ d < n) contains the vector f.
    offsets = np.arange(n, dtype=int)
    # SciPy copies the data; using a list of references is fine.
    diags = [f] * n
    T = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return T

def _max_convolution(seq: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of seq."""
    conv = np.convolve(seq, seq)
    return conv.max()

def _gradient_of_active(seq: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """Return the sub‑gradient of the max‐convolution (averaged over the active set)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    M = conv.max()
    active = np.where(conv >= M - tol)[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev = k - idx
        grad[idx] += 2.0 * seq[rev]
    if len(active) > 0:
        grad /= float(len(active))
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence minimizing ``evaluate_sequence``.
    The routine uses an alternating LP + local refinement strategy.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1) PARAMETERS & INITIAL SEQUENCE
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    # try to use the previously discovered construction if present
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=float)
        if len(init) >= n:
            cur_seq = init[:n].copy()
        else:
            cur_seq = np.concatenate([init, np.zeros(n - len(init))])
    else:
        cur_seq = _half_order_series(n)
    # Normalise to sum = 1 (evaluate_sequence is scale‑invariant)
    cur_seq = np.maximum(cur_seq, 0.0)
    cur_seq /= cur_seq.sum()

    best_seq = cur_seq.copy()
    best_score = 2.0 * n * _max_convolution(best_seq)   # same as evaluate_sequence for sum=1
    print(f"[init] start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) PHASE I – ALTERNATING LINEAR PROGRAMMING
    # ------------------------------------------------------------------
    lp_iter = 0

"""
Hybrid LP + local refinement for the self‑convolution bound.

The algorithm consists of two stages.

1) **Alternating linear program** – starting from a feasible sequence f
   (norm‑1) we compute M = max(f * f).  Holding f fixed we solve
   a linear programme

        maximise   Σ b_j
        subject to  T_f @ b ≤ M ,   b ≥ 0 ,

   where T_f is the (2n‑1)×n matrix that implements the linear convolution
   (T_f @ b = f * b).  The optimum b is scaled to Σ b = 1, giving a candidate
   g.  If max(g * g) < M we replace f by g; otherwise we back‑track a
   convex combination (1‑α)f + αg until an improvement is observed.
   The process repeats until the bound stops decreasing or the time budget
   for this phase is exhausted.

2) **Projected sub‑gradient / random pair exchange** – with the sequence
   obtained from stage 1 we perform cheap hill‑climbing moves:
   * move a small amount of mass between two random indices,
   * take a projected sub‑gradient step on the active set of maximal
     convolution entries (the sub‑gradient of the L∞‑norm of the
     self‑convolution),
   * occasional jitter to escape plateaus.

The whole search stops after the allotted budget (default 1000 s) and the
best sequence found (as a plain list of non‑negative floats, summed to 1)
is returned.
"""

import time
import math
import numpy as np
import cvxpy as cp
import scipy.sparse as sp


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n matrix T such that T @ b = f * b (linear convolution).
    Each diagonal offset d (0 ≤ d < n) contains the vector f.
    """
    n = f.shape[0]
    offsets = np.arange(n, dtype=int)
    diags = [f] * n                      # repeat the same data on every diagonal
    T = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return T


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``seq``."""
    return np.convolve(seq, seq).max()


def _gradient_of_active(seq: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the L∞‑norm of the self‑convolution, averaged over all
    active (maximal) indices.  For an active index k the gradient contribution
    at position i is 2·seq[k‑i] whenever i participates in the k‑th convolution.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    M = conv.max()
    active = np.where(conv >= M - tol)[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev = k - idx
        grad[idx] += 2.0 * seq[rev]
    if len(active) > 0:
        grad /= float(len(active))
    return grad
