def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    # pick n
    n = 5000
    # try to reuse previous
    try:
        init_seq = globals().get('height_sequence_1')
        if isinstance(init_seq, (list, np.ndarray)):
            init_seq = np.asarray(init_seq, dtype=float)
            if init_seq.ndim == 1 and init_seq.size >= 1000:
                n = max(n, init_seq.size)
    except Exception:
        pass

    # Choose block count
    B = max(100, min(800, n//5))
    # Ensure at least 2 orders of magnitude lower dimension than n for speed.
    block_len = n // B
    remainder = n % B
    block_lengths = np.full(B, block_len, dtype=int)
    block_lengths[:remainder] += 1
    block_start = np.concatenate(([0], np.cumsum(block_lengths)[:-1]))

    fft_len = next_pow2(2*n - 1)

    # function to expand y to a
    def expand_y(y):
        raw = np.exp(y)
        total = np.dot(raw, block_lengths)
        a_val = raw / total
        # repeat each value according to its block length
        return np.repeat(a_val, block_lengths)

    # fast_score from y
    def fast_score_y(y):
        a = expand_y(y)
        conv = conv_fft(a, fft_len, n)
        return float(2.0 * n * conv.max())

    # initialize y
    if 'height_sequence_1' in globals():
        seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        seq = seq[:n]  # truncate if longer
        # compute block averages
        block_means = np.add.reduceat(seq, block_start)
        block_means = block_means / block_lengths
        # ensure positive
        eps = 1e-12
        y0 = np.log(block_means + eps)
        best_y = y0.copy()
        best_a = expand_y(best_y)
        best_score = fast_score_y(best_y)
        print(f"[init] using height_sequence_1 block init, fast score = {best_score:.6f}")
    else:
        # uniform init
        best_y = np.zeros(B)  # raw = 1 => uniform a
        best_a = expand_y(best_y)
        best_score = fast_score_y(best_y)
        print(f"[init] uniform block init, fast score = {best_score:.6f}")

    # hyperparameters
    lam_schedule = [0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001]

    iteration = 0

    # main loop with restarts
    while time.time() < deadline:
        # choose start mode
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'best'])
        if mode == 'random':
            y_start = rng.standard_normal(B)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            # create exponentially decreasing block values
            idx = np.arange(B)
            vals = np.exp(-rate * idx)
            y_start = np.log(vals + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            idx = np.arange(1, B+1)
            vals = idx.astype(float) ** (-p)
            y_start = np.log(vals + 1e-12)
        elif mode == 'edge':
            # heavier on both ends
            m = max(1, int(B * rng.uniform(0.02, 0.08)))
            high = np.full(m, 5.0)  # exponent as log of large value
            low = np.full(B - 2*m, 0.0)
            vals = np.concatenate([high, low, high[::-1]])
            y_start = vals + rng.normal(scale=0.5, size=B)
        else:  # best
            y_start = best_y + rng.normal(scale=0.1, size=B)

        # run continuation L-BFGS
        y_opt = schedule_optimize_blocks(
            y_start,
            n,
            block_lengths,
            block_start,
            fft_len,
            lam_schedule,
            maxiter_per_stage=250,
        )
        a_opt = expand_y(y_opt)
        score_opt = fast_score_y(y_opt)

        if score_opt < best_score:
            best_score = score_opt
            best_y = y_opt.copy()
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # gradient-aware hill climb on blocks
        best_y, best_score = block_hill_climb(
            best_y,
            best_score,
            block_lengths,
            block_start,
            fft_len,
            n,
            deadline,
            max_iter=2000,
        )
        iteration += 1

    # final refinement: few more L-BFGS passes on best_y
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad_blocks,
            x0=best_y,
            args=(n, block_lengths, block_start, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 400, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_score = fast_score_y(best_y)
        print(f"[final refine] lam={lam:.4f} fast score = {best_score:.6f}")

    # final per-index greedy polishing (optional)
    if time.time() < deadline:
        best_a_full = expand_y(best_y)  # full vector
        best_a_full, best_score = hill_climb_full(
            best_a_full,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )
        final_seq = best_a_full.tolist()
    else:
        final_seq = expand_y(best_y).tolist()

    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

def block_hill_climb(y, cur_score, block_lengths, block_start, fft_len, n, deadline, max_iter=2000):
    best_y = y.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute current a and conv
        a = expand_y(best_y)  # need expand_y defined
        conv = conv_fft(a, fft_len, n)
        p = int(np.argmax(conv))
        # compute gradient for actual max index
        # compute g_per_index
        indices = np.arange(n)
        partner = p - indices
        mask = (partner >= 0) & (partner < n)
        g = np.zeros(n, dtype=float)
        g[mask] = 2.0 * a[partner[mask]]
        # sum per block
        G = np.add.reduceat(g, block_start)
        i_max = int(np.argmax(G))
        i_min = int(np.argmin(G))
        raw = np.exp(best_y)
        if raw[i_max] <= 0:
            continue
        delta_raw = raw[i_max] * rng.random() * 0.1  # up to 10%
        # apply move
        raw_i_max_new = raw[i_max] - delta_raw
        raw_i_min_new = raw[i_min] + delta_raw
        if raw_i_max_new < 0:
            continue
        raw_new = raw.copy()
        raw_new[i_max] = raw_i_max_new
        raw_new[i_min] = raw_i_min_new
        # new y
        y_new = np.log(raw_new + 1e-300)  # avoid log(0)
        # evaluate
        new_score = fast_score_y(y_new)
        if new_score < best_score:
            best_y = y_new
            best_score = new_score
        # else discard, try next iteration
    return best_y, best_score

"""
Step‑function optimizer (block‑wise, smoothed max).

The algorithm works on a probability simplex using a log‑softmax
parameterisation.  The vector is divided into contiguous blocks
(e.g. 5‑10 indices each) and only the block heights are optimised.
A smoothed log‑sum‑exp approximation of the maximum convolution entry
is minimised by L‑BFGS‑B with a continuation schedule on the smoothing
parameter.  After each L‑BFGS run a cheap gradient‑aware hill‑climb
moves mass from blocks that contribute most to the current peak
convolution to blocks that contribute least.  Finally a short
per‑index greedy hill‑climb refines the result.  The whole search runs
for the allotted time (default 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    `a` must have length n, `fft_len` is a power of two ≥ 2·n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


# ----------------------------------------------------------------------
# Block representation helpers
# ----------------------------------------------------------------------


def expand_blocks(y: np.ndarray, block_lengths: np.ndarray) -> np.ndarray:
    """
    Map block log‑variables y (size B) to a full probability vector a (size n).
    Each block k receives the same value.
    """
    raw = np.exp(y)                      # unnormalised block weights
    total = np.dot(raw, block_lengths)   # normalising constant Σ raw·len
    a_val = raw / total                  # value assigned to every element in a block
    return np.repeat(a_val, block_lengths)


def fast_score_from_y(y: np.ndarray,
                     block_lengths: np.ndarray,
                     fft_len: int,
                     n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a block‑parameterised vector."""
    a = expand_blocks(y, block_lengths)
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. block variables
# ----------------------------------------------------------------------


def objective_and_grad_blocks(y: np.ndarray,
                              n: int,
                              block_lengths: np.ndarray,
                              block_start: np.ndarray,
                              fft_len: int,
                              lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed max objective 2·n·smooth_max(a*a) and gradient w.r.t. block
    log‑variables y.
    """
    # ----- expand to full probability vector -----
    raw = np.exp(y)
    total = np.dot(raw, block_lengths)           # Σ raw_k·L_k
    a = np.repeat(raw / total, block_lengths)   # length n, sum = 1

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max (log‑sum‑exp) -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max = 0 now
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max                     # objective value

    # ----- gradient w.r.t. a (full vector) -----
    w = exp_shift / Z                           # weights for each convolution entry
    # g = ∂ smooth_max / ∂ a  (size n)
    fa_rev = np.fft.fft(a[::-1], fft_len)
    fw = np.fft.fft(w, fft_len)
    grad_full = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_full[(n - 1) : (n - 1 + n)]

    # ----- aggregate gradient to blocks -----
    G = np.add.reduceat(g_vec, block_start)     # shape (B,)

    # ----- chain rule through the soft‑max mapping -----
    # raw_k = exp(y_k), a_i = raw_{block(i)} / total
    S = np.dot(G, raw)                          # Σ G_k * raw_k
    grad_y = raw * (G * total - block_lengths * S) / (total ** 2)

    return float(f), grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (block version)
# ----------------------------------------------------------------------


def schedule_optimize_blocks(y_start: np.ndarray,
                             n: int,
                             block_lengths: np.ndarray,
                             block_start: np.ndarray,
                             fft_len: int,
                             lam_schedule: list[float],
                             maxiter_per_stage: int = 30) -> np.ndarray:
    """
    Run a sequence of L‑BFGS‑B optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad_blocks,
            x0=y,
            args=(n, block_lengths, block_start, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Gradient‑aware hill climbing on blocks
# ----------------------------------------------------------------------


def block_hill_climb(y: np.ndarray,
                     cur_score: float,
                     block_lengths: np.ndarray,
                     block_start: np.ndarray,
                     fft_len: int,
                     n: int,
                     deadline: float,
                     max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb that uses the exact max‑convolution
    gradient to decide where to move probability mass between blocks.
    """
    best_y = y.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # pre‑compute index array for speed
    all_idx = np.arange(n)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current full vector and convolution
        a = expand_blocks(best_y, block_lengths)
        conv = conv_fft(a, fft_len, n)
        p = int(np.argmax(conv))

        # gradient of the *actual* max entry (no smoothing)
        partner = p - all_idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros(n, dtype=float)
        g[mask] = 2.0 * a[partner[mask]]

        # block‑wise aggregation
        G = np.add.reduceat(g, block_start)

        i_max = int(np.argmax(G))
        i_min = int(np.argmin(G))
        if i_max == i_min:
            continue

        # raw block weights
        raw = np.exp(best_y)
        if raw[i_max] <= 0:
            continue

        # move a random fraction (up to 10 %) of the source block mass
        delta_raw = raw[i_max] * rng.random() * 0.10
        if delta_raw <= 0:
            continue

        raw[i_max] -= delta_raw
        raw[i_min] += delta_raw

        # re‑parameterise
        y_new = np.log(raw + 1e-300)  # safeguard against log(0)

        # evaluate
        new_score = fast_score_from_y(y_new, block_lengths, fft_len, n)
        if new_score < best_score:
            best_y = y_new
            best_score = new_score
        else:
            # revert the move
            raw[i_max] += delta_raw
            raw[i_min] -= delta_raw

    return best_y, best_score


# ----------------------------------------------------------------------
# Simple per‑index greedy hill climb (used for final polishing)
# ----------------------------------------------------------------------


def fast_score_full(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact 2·n·max(a*a) for a full vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def hill_climb_full(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    max_iter: int = 8000) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the full (non‑blocked) vector.
    Moves a random fraction (≤5 %) of mass from i to j and keeps the
    change only if the score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score_full(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score
