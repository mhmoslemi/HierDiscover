    start = time.time()
    deadline = start + budget_s - 5.0
    # Determine n:
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = len(init_seq)
        init_seq = init_seq / init_seq.sum()
    except Exception:
        n = 8000
        init_seq = None

    # Choose length: maybe increase n if more time left? For simplicity use n.

    # Precompute FFT length L

    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # Keep best sequence and best max conv value
    best_seq = None
    best_max = float('inf')

    def maybe_update(seq):

        # ensure sum=1
        seq_norm = seq / seq.sum()
        conv = compute_conv(seq_norm, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = seq_norm.copy()

    # Phase 0: if init_seq present
    if init_seq is not None:
        maybe_update(init_seq)
        print(f"init global candidate max={best_max:.8e}")

    # Phase 1: parametric sweep
    print(f"--- Phase 1 sweep, n={n}")
    # Families:
    # uniform, power, exponential, double exp, beta, arcsine approx
    xs = np.arange(n, dtype=np.float64)
    # Exponents
    for p in np.linspace(0.0, 2.0, 41):
        if time.time()>deadline: break
        base = (xs+1)**p
        maybe_update(base)
        maybe_update(base[::-1])
    # exponentials
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time()>deadline: break
        exp_shape = np.exp(-lam * xs)
        maybe_update(exp_shape)
        maybe_update(exp_shape[::-1])
    # double exponential left+right
    for lam in np.linspace(0.0,5.0,21):

        left = np.exp(-lam * xs)
        right = left[::-1]

    for a in alphas:
        for b in betas:
            if time.time()>deadline: break
            # Beta density on [0,1] discretized: x^{a-1} * (1-x)^{b-1}
            x = (xs+0.5)/n

            maybe_update(shape)
            maybe_update(shape[::-1])

"""
Propose a step‑function that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

1. **Initialisation** – use a previously found construction if available,
   otherwise fall back to a uniform vector.  The vector is always
   normalised to have unit L1‑norm (the final scaling to √(2 n) is done
   at the very end).

2. **Parametric sweep** – generate a large pool of simple families
   (power‑law, exponential, double‑exponential, Beta‑densities and an
   arcsine‑like shape).  Each candidate is projected on the simplex and
   the best one (smallest max‑convolution) is kept.

3. **Adam on a smoothed max** – replace the hard max by a soft‑max
   `log‑sum‑exp(α·conv)/α`.  The gradient of this smooth objective is
   `2·correlation(softmax_weights, a)`.  Adam updates are performed on
   the simplex and the smoothing parameter α is increased from a small
   to a large value (4 → 150).  This stage brings the solution close to
   a good basin.

4. **Sub‑gradient descent on the true max** – once α is large we switch
   to a genuine sub‑gradient of `max(conv)`.  The active set consists of
   all lags whose convolution value is within numerical tolerance of
   the maximum.  The sub‑gradient is computed by the same FFT trick as
   for the smoothed case, and a diminishing step size η/√t is used.

5. **Cheap random hill‑climbing** – finally, many tiny mass transfers
   between random coordinates are tried; any improvement is kept.
   This helps to escape tiny plateaus that Adam/sub‑gradient may miss.

All heavy work is performed with FFTs (complexity O(L log L) per
iteration).  The whole routine respects the supplied time budget and
returns the best sequence found, scaled to the normalisation used by
`evaluate_sequence`.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # very unlikely
        w[conv.argmax()] = 1.0
    return w

    Search for a non‑negative step function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats.
    """

    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ---- Determine the length ------------------------------------------------

        n = 8000

    phase = np.exp(2j * np.pi * np.arange(L) / L)

    best_seq = None

        cur_max = conv.max()
        best_seq = init_seq.copy()

    # ---- 1.  Parametric sweep ------------------------------------------------
    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    xs = np.arange(n, dtype=np.float64)

    # power‑law family
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1) ** p
        for cand in (base, base[::-1]):
            cand = cand / cand.sum()
            conv = _compute_convolution(cand, L)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"  new best (power p={p:.2f})  max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # pure exponential family
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):
            cand = cand / cand.sum()
            conv = _compute_convolution(cand, L)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"  new best (exp λ={lam:.2f})   max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # double‑exponential (U‑shaped) family
    for lam in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        cand = u_shape / u_shape.sum()
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  new best (U‑shape λ={lam:.2f}) max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # Beta‑density family (discrete Beta(a,b) shape)
    alphas = [0.5, 1.0, 2.0, 5.0]
    betas = [0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta_shape > 0):
                for cand in (beta_shape, beta_shape[::-1]):
                    cand = cand / cand.sum()
                    conv = _compute_convolution(cand, L)
                    cur_max = conv.max()
                    if cur_max < best_max:
                        best_max, best_seq = cur_max, cand.copy()
                        print(f"  new best (Beta a={a} b={b}) max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # arcsine‑like shape (1/√[x(1‑x)]) – a classic “U‑profile”
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    for cand in (arcsine, arcsine[::-1]):
        cand = cand / cand.sum()
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  new best (arcsine)        max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # If sweep never produced a candidate (should not happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _compute_convolution(best_seq, L).max()

    # ---- 2. Adam descent on smoothed max ------------------------------------
    print("--- Phase 2: Adam descent with smooth‑max ---")
    a = best_seq.copy()                       # always stays on the simplex
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.10                                 # base learning rate
    max_adam_iters = 50000
    it = 0
    alpha_start, alpha_end = 4.0, 150.0

    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smoothed objective
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate the new point
        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"  Adam it {it:5d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # ---- 3. Sub‑gradient descent on the true max ----------------------------
    print("--- Phase 3: sub‑gradient descent on true max ---")
    a = best_seq.copy()
    sub_max_iters = 60000
    sub_it = 0
    while time.time() < deadline and sub_it < sub_max_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        w = _submax_weights(conv)                     # uniform on active set
        grad = _gradient_from_weights(a, w, phase, L)

        step_size = 0.5 / np.sqrt(sub_it)            # diminishing step
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"  Sub‑grad it {sub_it:5d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        a = a_new

    # ---- 4. Cheap random local moves (hill‑climbing) ------------------------
    print("--- Phase 4: random local moves ---")
    a_current = best_seq.copy()
    while time.time() < deadline:
        # choose a source proportionally to its mass (large entries more likely)
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.02 * a_current[i]   # up to 2 % transfer
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            a_current = a_new
            print(f"  move improved max={max_new:.8e}  score={2*n*max_new:.6f}")

    # ---- Final scaling -------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)  # just for logging
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
          f"after {it} Adam iters, {sub_it} sub‑grad iters ---")
    return final_seq
