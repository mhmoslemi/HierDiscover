def random_fourier_candidate(n, rng):
    # Generate random phases for half-spectrum
    F = np.zeros(n, dtype=complex)
    # For k=0, real only, can set F[0]=0
    # For k=1..n/2-1, generate random phase phi
    # Need Hermitian symmetry: F[n-k] = conj(F[k])
    # We can set magnitude = 1 or random.
    magnitude = 1.0
    # We'll set random amplitude perhaps uniform [0,1] to reduce negative values? We'll try magnitude=1.
    # We'll generate random angles phi_k uniform [0, 2π).
    # Build F.
    half = n // 2
    phi = rng.random(half-1) * 2 * np.pi
    F[0] = 0.0 + 0.0j
    for k in range(1, half):
        F[k] = magnitude * np.exp(1j * phi[k-1])
        F[n-k] = np.conj(F[k])
    if n % 2 == 0:
        F[half] = magnitude * np.exp(1j * rng.random() * 2*np.pi)  # Nyquist freq
    a = np.real(np.fft.ifft(F))
    # shift to be nonnegative
    a -= a.min()
    # If all zeros (unlikely), add small epsilon.
    if a.max() == 0:
        a = np.ones_like(a)
    a /= a.sum()
    return a

"""
Hybrid optimizer for the step‑function “max‑convolution” problem.

The algorithm combines several cheap global heuristics (random exponentials,
power laws, piece‑wise constant patterns and a random Fourier‑phase
construction) with a smooth‑max L‑BFGS refinement and aggressive
greedy hill‑climbing.  The search runs until the allotted time budget
expires, constantly keeping the best sequence found.

Key ideas
---------
* Work on the probability simplex (sum = 1) – the evaluation is
  scale‑invariant, so we can safely renormalise after each move.
* Use a smooth approximation of the max‑convolution (log‑sum‑exp) and
  L‑BFGS‑B with a continuation schedule that drives the smoothing parameter
  to zero.
* After each gradient‑based run, perform a fast greedy hill‑climb that
  transfers a small random fraction of mass between two positions.
* Occasionally generate fresh candidates with simple parametric families
  (exponential, power‑law, piece‑wise constant) and a random‑phase Fourier
  construction; keep any improvement.
* If a previous solution is available via the global name
  ``height_sequence_1`` it is used as a warm start.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a with itself (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of   2·n·max(a*a)   for a probability vector (sum = 1).
    The convolution is performed with a single FFT.
    """
    n = a.shape[0]
    conv = conv_fft(a, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    ``lam_factor`` controls the smoothing width:
        lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}   for i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 150,
) -> np.ndarray:
    """
    Run a series of L‑BFGS‑B optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Repeatedly move a random fraction of mass from i to j if the move
    strictly improves the (fast) score.  Stops when time expires or the
    iteration budget is exhausted.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximum fraction of source mass we are willing to move
        max_frac = 0.10
        delta = best[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def random_fourier_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative probability vector via a random Fourier‑phase
    construction.  The raw inverse FFT may contain negative values; we
    shift it to be non‑negative and renormalise.
    """
    # Build Hermitian‑symmetric spectrum with unit magnitude.
    F = np.zeros(n, dtype=complex)
    half = n // 2
    # DC component
    F[0] = 0.0
    # Positive frequencies
    phis = rng.random(half - 1) * 2.0 * np.pi
    F[1:half] = np.exp(1j * phis)
    F[-half + 1 :] = np.conj(F[1:half])
    if n % 2 == 0:
        # Nyquist term (real)
        F[half] = rng.random() * 2.0 - 1.0

    a_raw = np.real(np.fft.ifft(F))
    a_shift = a_raw - a_raw.min()          # make all entries ≥ 0
    if a_shift.max() == 0:
        a_shift = np.ones_like(a_shift)
    a = a_shift / a_shift.sum()
    return a

        if a0.sum() > 0.0:
            a0 /= a0.sum()
            sc = fast_score(a0, fft_len_conv)
            if sc < best_score:
                best_a = a0
                best_score = sc
                best_y = np.log(best_a + 1e-12)
                print(f"[init] previous best fast score = {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop – mix of random generators and local refinement
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6,
        0.45,
        0.30,
        0.20,
        0.15,
        0.10,
        0.07,
        0.04,
        0.02,
        0.01,
        0.005,
        0.0025,
        0.00125,
        0.0006,
    ]

    iteration = 0
    while time.time() < deadline:
        # ----- choose a generation strategy -----
        mode = rng.choice(
            ["random", "exp", "power", "edge", "segment", "fourier", "best"]
        )
        if mode == "random":
            a0 = rng.random(n)
            a0 /= a0.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
        elif mode == "power":
            p = rng.uniform(0.3, 3.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
        elif mode == "edge":
            # heavy mass near both ends (1‑4 % of the indices)
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
        elif mode == "segment":
            # random piece‑wise constant shape
            m = rng.integers(5, 30)                # number of segments
            seg_vals = rng.random(m)
            seg_vals /= seg_vals.sum()
            seg_len = max(1, n // m)
            a0 = np.repeat(seg_vals, seg_len)[:n]
            if a0.size < n:
                a0 = np.concatenate([a0, np.full(n - a0.size, seg_vals[-1])])
            a0 /= a0.sum()
        elif mode == "fourier":
            a0 = random_fourier_candidate(n, rng)
        else:  # perturb current best
            delta = rng.normal(scale=0.02, size=n)
            a0 = best_a + delta
            a0 = np.maximum(0.0, a0)
            if a0.sum() == 0.0:
                a0 = best_a.copy()
            a0 /= a0.sum()

        # ----- evaluate candidate (fast score) -----
        cand_score = fast_score(a0, fft_len_conv)
        if cand_score < best_score:
            best_a = a0
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- local improvement via hill‑climb -----
        # a short time slice (≈0.25 s) for a deeper greedy search
        hc_deadline = min(deadline, time.time() + 0.25)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement using the smooth‑max L‑BFGS continuation schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_y = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=250,
        )
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv)
        print(f"[final L‑BFGS] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final hill] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
