"""
Propose a candidate sequence that minimizes the self‑convolution supremum.

Algorithm
---------
1. Initialise a probability vector `a` on the simplex.
   * If a global `height_sequence_1` is available, use it.
   * Otherwise start from the “half‑order” distribution (C(2i,i)/4**i).
   * The vector is represented as soft‑max parameters `w` (a = softmax(w))
     to keep non‑negativity automatically.

2. Optimise a *smooth* surrogate of the objective:
       M_ε(a) = ε·log∑_k exp(conv(a)[k]/ε)   (smooth approximation of max)
   using Adam on the parameters `w`.  The gradient of M_ε w.r.t. `w`
   can be written in closed form and is computed by a few FFT‑based
   convolutions ( O(n log n) per iteration).

3. Anneal the smoothing parameter ε from a moderate start (≈ half the
   initial maximal convolution value) down to ~1e‑7.  When ε is large
   the landscape is smooth; as ε → 0 the gradient approaches the true
   sub‑gradient of the max‑convolution.

4. Keep track of the best (un‑smoothed) score `2·n·max(conv(a))`
   observed during the run and return that sequence.

The method combines a deterministic gradient descent (Adam) with an
annealing schedule, which is very different from the random pair‑exchange
and sub‑gradient scheme used previously.  It typically finds a better
distribution within the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum == 1)."""
    m = np.max(w)
    e = np.exp(w - m)
    return e / e.sum()


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (fast for n≈1000). Returns length len(x)+len(y)-1."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())   # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:size]
    return conv


def _half_order_series(n: int) -> np.ndarray:
    """Half‑order (central binomial) coefficients, normalised to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:

    Return a list of non‑negative heights (a probability vector) that
    (hopefully) yields a lower score than previous attempts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so that the caller can also stop
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # 1) problem size and initialisation
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    total_runtime = deadline - start_time

    # Try to start from a previously discovered good sequence
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        if init_seq.shape[0] != n:
            # resize / truncate if needed
            if init_seq.shape[0] > n:
                init_seq = init_seq[:n]
            else:
                # pad with tiny uniform mass
                pad = np.full(n - init_seq.shape[0], 1e-12, dtype=np.float64)
                init_seq = np.concatenate([init_seq, pad])
        init_seq = np.maximum(init_seq, 1e-15)      # avoid exact zeros
        init_seq /= init_seq.sum()
        w = np.log(init_seq)                       # soft‑max parameters
        print("[init] Using global height_sequence_1 as start.")
    except Exception:
        # fallback: half‑order series
        base = _half_order_series(n)
        # a small convex blend with uniform helps escaping shallow local minima
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        a0 = 0.7 * base + 0.3 * uniform
        a0 /= a0.sum()
        w = np.log(a0)
        print("[init] Started from blended half‑order + uniform.")

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    # Best solution observed so far
    best_seq = None
    best_score = float("inf")
    best_iter = -1

    # ------------------------------------------------------------------
    # 2) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start_time
        progress = elapsed / total_runtime

        # ------- Annealing schedule for the smooth max -------
        # Start with epsilon ~ ½ of the initial max‑convolution,
        # finish at a very small value.
        if best_seq is None:

            conv_tmp = _fft_convolve(a_tmp, a_tmp)

        else:

        eps_final = 1e-7
        epsilon = eps0 * ((eps_final / eps0) ** progress)

        # ------- Current point on the simplex -------
        a = _softmax(w)                     # shape (n,)
        conv = _fft_convolve(a, a)          # length 2n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # keep best exact score
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            best_iter = t
            print(f"[{t:6d}] New best score {best_score:.6f}")

        # ------- Gradient of the smooth maximum -------
        # p_k ∝ exp((conv_k - max_conv)/epsilon)  (stable softmax)
        shifted = conv - max_conv
        # avoid overflow/underflow
        p = np.exp(shifted / epsilon)
        p_sum = p.sum()
        if p_sum == 0.0:
            # numerical safety – fall back to a hard max indicator
            p = np.zeros_like(p)
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum                      # now ∑ p_k = 1

        # gradient w.r.t. a:  g_i = 2 * Σ_k p_k * a_{k-i}
        a_rev = a[::-1]
        grad_full = _fft_convolve(p, a_rev)           # length 3n-2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice of length n

        # convert to gradient w.r.t. w (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)                   # shape (n,)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate scaled with epsilon (larger epsilon → larger steps)
        base_lr = 0.2
        lr = base_lr * (epsilon / eps0 + 1e-3)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * 0.001
            w += jitter

        # occasional status report
        if t % 2000 == 0:
            print(f"[{t:6d}] time {elapsed:.1f}s, score {cur_score:.6f}, ε {epsilon:.2e}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence seen
    # ------------------------------------------------------------------
    if best_seq is None:
        # This should never happen, but fall back to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("No valid sequence found – returning uniform.")
    else:
        print(f"Finished after {t} iterations (best at iter {best_iter}), best score = {best_score:.6f}")

    # Ensure plain Python list of floats
    return best_seq.tolist()
