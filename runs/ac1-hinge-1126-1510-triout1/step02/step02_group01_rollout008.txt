def adam_optimize(y0, n, fft_len_conv, deadline, max_iters=2000, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_y, best_score = y.copy(), np.inf
    t = 0
    while time.time() < deadline and t < max_iters:
        a = softmax(y)
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()
        # compute subgrad for max
        g = subgrad_max_conv(a, conv)
        # compute grad_y
        mean = np.dot(a, g)
        grad_y = a * (g - mean)
        t += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= alpha * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate fast score occasionally
        if t % 10 == 0:
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_y = y.copy()
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    return best_a, best_score

def subgrad_max_conv(a, conv):
    max_val = conv.max()
    idxs = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(a)
    a_rev = a[::-1]
    n = a.shape[0]
    for k in idxs:
        shift = n - 1 - k
        if shift >= 0:
            # g[i] += 2 * a_rev[i + shift] for i where i+shift < n
            slice_len = n - shift
            g[:slice_len] += 2 * a_rev[shift:]
        else:
            # shift negative: i + shift starts negative for small i, so start from -shift
            s = -shift
            slice_len = n - s
            g[s:] += 2 * a_rev[:slice_len]
    if len(idxs) > 0:
        g /= len(idxs)
    return g
```

"""
Propose a low‑evaluation step sequence.

The algorithm combines three ingredients:

1.  **Random restarts** – generate diverse initial distributions
    (uniform, exponential, power‑law, edge‑biased or a perturbed version
    of the current best).

2.  **Exponentiated‑gradient descent on the simplex** – each iteration
    computes the exact max‑convolution entry, builds the sub‑gradient
    (∂ max (a∗a) / ∂ a_i = 2·a_{k‑i} for the maximal index k) and updates
    the probabilities by a multiplicative step
    `a_i ← a_i·exp(−η·g_i)`.  The step size η follows a √t schedule.
    The iterate is renormalised after every step, which automatically
    respects the non‑negativity and sum‑to‑one constraints.

3.  **Greedy hill‑climbing** – after the gradient phase a short
    mass‑transfer hill climb moves a random fraction of the mass between
    two positions whenever the fast‑score improves.

The loop repeats while time permits, keeping the best sequence seen.
A final longer gradient run and hill‑climb polish the current best
before the official `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the real part of length 2*n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast version of the evaluation function for a probability vector:
    score = 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (with sum(a)=1, a≥0).

    For a given offset k the gradient component is
        g_i = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0).

    If several offsets attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to collect all (numerically) maximal entries
    tol = max_val - 1e-12
    idxs = np.where(conv >= tol)[0]
    if idxs.size == 0:
        return np.zeros_like(a)

    a_rev = a[::-1]
    g = np.zeros_like(a)

    for k in idxs:
        shift = n - 1 - k  # shift needed on the reversed vector
        if shift >= 0:
            # contributes to the first n‑shift entries
            length = n - shift
            g[:length] += 2.0 * a_rev[shift:]
        else:
            # negative shift – contributes to the last n+shift entries
            s = -shift
            length = n - s
            g[s:] += 2.0 * a_rev[:length]

    g /= idxs.size
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the simplex
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 600,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the probability
    simplex.  Returns the best vector and its fast_score found
    during the run.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = conv_fft(a, fft_len, n)
        g = subgrad_max_conv(a, conv)

        # step size decreasing like 1/√t
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update (avoid overflow/underflow)
        mult = np.exp(-eta * g)
        a = a * mult

        s = a.sum()
        if s == 0.0:
            # numerically collapsed – re‑initialise uniformly
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (identical to the previous code)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill climb: repeatedly move a random fraction
    of the probability mass from i to j (preserving the sum).  Keep the
    move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the current mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # problem size – we use a length of a few thousand elements
    # ------------------------------------------------------------------
    n = 5000
    try:
        # reuse a previously constructed sequence if it is long enough
        height_sequence_1  # noqa: F821
        init_seq = globals()["height_sequence_1"]
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            arr = np.asarray(init_seq, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, arr.size)
    except Exception:
        pass

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise best solution (fallback: uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    # if a previous construction exists, seed from it
    try:
        height_sequence_1  # noqa: F821
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size >= n:
            best_a = seq[:n].copy()
            best_a = np.maximum(best_a, 1e-12)       # avoid exact zeros
            best_a = best_a / best_a.sum()
            best_score = fast_score(best_a, fft_len, n)
    except Exception:
        pass

    print(f"[init] n={n} best fast score = {best_score:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- pick a random start distribution -----
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            a0 = rng.gamma(shape=1.0, scale=1.0, size=n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "edge":
            # a few heavy edges at both ends
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a0 = np.full(n, 1e-6)
            a0[:m_edge] = 0.5 / m_edge
            a0[-m_edge:] = 0.5 / m_edge
        else:  # perturb the current best
            noise = rng.normal(scale=0.03, size=n)
            a0 = best_a * np.exp(noise)

        a0 = np.maximum(a0, 1e-12)
        a0 = a0 / a0.sum()

        # ----- exponentiated‑gradient optimisation -----
        # allocate a short time slice for this restart
        sub_deadline = min(deadline, time.time() + 1.0)
        a_opt, score_opt = exp_gradient_optimize(
            a_start=a0,
            n=n,
            fft_len=fft_len,
            deadline=sub_deadline,
            max_iters=400,
            eta0=0.6,
        )

        # ----- local greedy hill‑climb (short) -----
        if time.time() < deadline:
            a_opt, score_opt = hill_climb(
                a_opt,
                score_opt,
                fft_len,
                n,
                deadline,
                max_iter=2500,
            )

        # ----- keep the best -----
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )
        iteration += 1

    # ------------------------------------------------------------------
    # final polishing phase on the current best
    # ------------------------------------------------------------------
    # a longer exponentiated‑gradient run
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            a_start=best_a,
            n=n,
            fft_len=fft_len,
            deadline=deadline,
            max_iters=2000,
            eta0=0.4,
        )
        print(f"[final‑grad] fast score = {best_score:.6f}")

    # final greedy hill‑climb (as much time as left)
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final‑hill] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
