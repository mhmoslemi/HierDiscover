the allotted time budget (≈1000 s) and finally returns the best
non‑negative sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x  (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max yielding a probability vector (sum=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2*n‑1) using FFT.
    a is length n, fft_len must be a power of two ≥ 2*n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1).
    This is exactly the value returned by the official evaluator up to
    the scaling factor (the scaling cancels out because we keep sum=1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max objective for a probability vector a = softmax(y) and its
    gradient w.r.t. the unrestricted vector y.

    The smoothing parameter is λ = lam_factor * max(conv) + 1e-12.
    """
    # map to simplex
    a = softmax(y)

    # ----------- convolution a*a (full) -------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----------- gradient of smooth max -------------
    w = exp_shift / Z                       # ∂ smooth_max / ∂ conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1      (cross‑correlation)
    # compute via FFT: reverse a, convolve with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    move_fraction: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb (mass‑transfer moves).
    Randomly pick i → j and transfer up to move_fraction of a[i].
    Keep the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move from i to j
        delta = best[i] * rng.random() * move_fraction
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 120,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations decreasing the smoothing
    factor λ (given indirectly by lam_schedule).  The result of each
    stage seeds the next.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0      # small safety margin

    # ---------------------------------------------------------------
    # Problem size – start with a medium size, enlarge if a previous
    # construction is available.
    # ---------------------------------------------------------------
    n = 8000                                      # default size
    try:
        init_seq = globals().get("height_sequence_1")
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            init_arr = np.asarray(init_seq, dtype=float)
            if init_arr.ndim == 1 and init_arr.size >= 2000:
                n = max(n, init_arr.size)
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)           # for a*a
    fft_len_g = next_pow2(3 * n - 2)              # for gradient step

    # ---------------------------------------------------------------
    # Initial candidate – uniform distribution (known baseline)
    # ---------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)               # softmax argument
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ---------------------------------------------------------------
    # A finer smoothing schedule than the reference implementation.
    lam_schedule = [
        0.7, 0.45, 0.30, 0.18, 0.10,
        0.06, 0.035, 0.020, 0.012, 0.007,
        0.004, 0.002, 0.001, 0.0005
    ]

    iteration = 0
    no_improve_iters = 0
    max_no_improve = 25    # stop early if stagnation (time check will still stop)

    while time.time() < deadline:
        # -----------------------------------------------------------
        # Choose a diverse start point
        # -----------------------------------------------------------
        mode = rng.choice(["random", "exp", "power", "edge", "gauss", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.015)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # heavy mass near the two ends (1‑3 % of the points)
            m_edge = max(1, int(n * rng.uniform(0.008, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "gauss":
            # mixture of a few Gaussian bumps
            K = rng.integers(2, 6)                     # number of components
            a0 = np.zeros(n, dtype=float)
            total_w = 0.0
            for _ in range(K):
                centre = rng.integers(0, n)
                sigma = rng.uniform(0.02, 0.12) * n   # relative width
                weight = rng.random()
                total_w += weight
                xs = np.arange(n)
                a0 += weight * np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # "best"
            y0 = best_y + rng.normal(scale=0.04, size=n)

        # -----------------------------------------------------------
        # Continuation L‑BFGS on the smooth‑max objective
        # -----------------------------------------------------------
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=120,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode:<6} improved fast score = {best_score:.6f}")
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # -----------------------------------------------------------
        # Aggressive local hill‑climb (short time slice)
        # -----------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3500,
            move_fraction=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

        # early‑stop if we have stalled for many iterations
        if no_improve_iters >= max_no_improve:
            break

    # ------------------------------------------------------------------
    # Final refinement – one more continuation pass with tighter λ
    # ------------------------------------------------------------------
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 250, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5f} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
            move_fraction=0.06,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
