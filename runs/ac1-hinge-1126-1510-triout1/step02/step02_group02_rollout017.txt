assistantfinal**Solution Overview**

The evaluation function  

```
E(a)=2·n·max(conv(a,a)) / (∑a)²
```  

is *scale invariant* – we may restrict ourselves to probability vectors  
`a ≥ 0 , ∑a = 1`.  
The goal is therefore to minimise `max(conv(a,a))`.  

The algorithm consists of three, completely independent, stages:

1. **Large‑scale parametric sweep** – a cheap exploration of many simple
   families (power‑law, exponential, beta‑distribution, sine‑shapes,
   mixtures of exponentials …).  
   Every candidate is normalised to unit sum, its convolution is obtained
   by a single FFT and the best candidate is remembered.

2. **Adam optimisation on a smoothed max** – the non‑smooth `max`
   is replaced by a log‑sum‑exp soft‑max  
   `Sα(c)=log∑exp(α·c)/α`.  For a given `α` the gradient is a weighted
   correlation which can be computed with two FFTs.  Adam updates are
   performed on the probability simplex (Euclidean projection after each
   step).  `α` is increased from `4` to `300` during the run, turning the
   smooth objective gradually into the true max.  The loop stops when the
   global time budget is exhausted.

3. **Intensive local hill‑climbing** – after the gradient phase a cheap
   stochastic move (transfer a small random amount of mass from one index
   to another) is attempted many thousands of times.  Any improving move
   is kept.  This stage removes the tiny plateau that often remains after
   the Adam phase.

All three stages respect the total time limit (`budget_s`).  
The best sequence found is finally scaled to the normalisation used by the
reference code (`∑a = √(2·n)`) – the evaluation score is unchanged.

The implementation uses only `numpy`, `math`, `time` and the supplied
`evaluate_sequence`.  No closures or file I/O are used; every helper
function is defined at the module level.

```python
"""
propose_candidate – search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm:
  1. parametric sweep over a large collection of simple families
  2. Adam optimisation on a soft‑max approximation of the max‑norm
  3. cheap stochastic hill‑climbing
All steps respect a global 1000 s budget and return the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """
    Return max(conv(a,a)) for a vector a (assumed to sum to 1).
    Convolution is performed via FFT of length L (L ≥ 2·len(a)-1).
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for a vector `conv` (length 2·n‑1) and smoothing
    parameter `alpha`.  Returns a vector of the same length that sums to 1.
    """
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()
    return w


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   L: int,
                   deadline: float,
                   alpha_start: float = 4.0,
                   alpha_end: float = 300.0,
                   lr0: float = 0.08,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max of the convolution.
    Returns the best vector seen (unit‑sum) and its max‑convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max(a, L)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # phase factor needed for the reversed FFT (a_rev = a[::-1])
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution of the current iterate -----
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # ----- soft‑max weighting -----
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # ----- gradient of the smooth max (2·correlation) -----
        # reversed FFT of a (no extra FFT)
        fft_a_rev = np.conj(fft_a) * phase
        fft_w = np.fft.fft(w, n=L)
        corr_full = np.real(np.fft.ifft(fft_w * fft_a_rev))
        # extract the part that corresponds to the gradient (length n)
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # ----- Adam update on the probability simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- evaluate the new point -----
        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"Adam iter {it:6d}: improved obj={obj_new:.8e} "
                  f"score={2 * n * obj_new:.6f}")

        a = a_new

    return best_a, best_obj


# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  L: int,
                  current_best_obj: float,
                  deadline: float,
                  max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly transfer a small random amount
    of mass from one index to another.  Any improvement is kept.
    """
    n = a0.shape[0]
    a_cur = a0.copy()
    best_obj = current_best_obj

    while time.time() < deadline:
        # choose source proportionally to its current mass
        i = np.random.choice(n, p=a_cur)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * max_move_frac * a_cur[i]
        if delta <= 0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:          # safety (should not happen)
            continue

        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            a_cur = a_new
            print(f"  local move improved obj={obj_new:.8e} "
                  f"score={2 * n * obj_new:.6f}")

    return a_cur, best_obj

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list whose elements are scaled to
    sum = sqrt(2·n) (the scaling does not affect the score).
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # a small safety margin

    # ------------------------------------------------------------------
    # Decide the length of the sequence.
    # Prefer the length of a previously found construction if available.
    default_n = 2000

    except Exception:
        n = default_n

    target_sum = math.sqrt(2.0 * n)

    # FFT length (next power of two ≥ 2·n‑1)
    L = 1 << ((2 * n - 2).bit_length())

    best_seq = None
    best_obj = float("inf")   # stores min max(conv)

    # ------------------------------------------------------------------
    # Helper to evaluate a raw (non‑normalised) candidate and update the global best.
    def _maybe_update(raw_vec: np.ndarray):
        nonlocal best_seq, best_obj
        if raw_vec.sum() == 0.0:
            return
        a = raw_vec / raw_vec.sum()
        obj = _conv_max(a, L)
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"  new best (parametric) obj={obj:.8e} "
                  f"score={2 * n * obj:.6f}")

    # ------------------------------------------------------------------
    # 0.  Insert a possible global candidate (if it exists)
    try:
        if init_global.sum() > 0:
            a0 = init_global / init_global.sum()
            obj0 = _conv_max(a0, L)
            best_seq, best_obj = a0.copy(), obj0
            print(f"Initial global candidate score = {2 * n * obj0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Parametric sweep over many families
    print(f"--- Phase 1: parametric sweep (n={n}) ---")

    # --- power‑law ----------------------------------------------------
    exps = np.linspace(0.0, 3.0, 31)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in exps:
        if time.time() > deadline:
            break
        base = idx ** p
        _maybe_update(base)                 # increasing
        _maybe_update(base[::-1])           # decreasing

    # --- exponential --------------------------------------------------
    lambdas = np.linspace(0.0, 5.0, 31)
    for lam in lambdas:
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
        _maybe_update(exp_shape)           # decreasing
        _maybe_update(exp_shape[::-1])     # increasing

    # --- U‑shaped (sum of two exponentials) -------------------------
    for lam in lambdas[::4]:
        if time.time() > deadline:
            break
        left = np.exp(-lam * np.arange(n, dtype=np.float64))
        u_shape = left + left[::-1]
        _maybe_update(u_shape)

    # --- beta family (a_i ∝ i^{α‑1} (n‑i)^{β‑1}) --------------------
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    i = np.arange(1, n + 1, dtype=np.float64)
    for a in alphas:
        if time.time() > deadline:
            break
        for b in betas:
            beta_vec = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
            _maybe_update(beta_vec)

    # --- sine‑shaped families -----------------------------------------
    powers = np.linspace(0.5, 4.0, 15)
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    sin_base = np.sin(theta)
    for p in powers:
        if time.time() > deadline:
            break
        a = sin_base ** p
        _maybe_update(a)

    # --- triangular‑type (i·(n+1−i)) with an exponent ----------------
    tri_base = (idx * (n + 1 - idx))
    for p in powers:
        if time.time() > deadline:
            break
        a = tri_base ** p
        _maybe_update(a)

    # --- mixture of two exponentials ----------------------------------
    ws = [0.2, 0.5, 0.8]
    lam_grid = np.linspace(0.0, 5.0, 11)
    for w in ws:
        for lam1 in lam_grid:
            if time.time() > deadline:
                break
            for lam2 in lam_grid:
                a = w * np.exp(-lam1 * np.arange(n, dtype=np.float64)) \
                    + (1.0 - w) * np.exp(-lam2 * np.arange(n, dtype=np.float64))
                _maybe_update(a)

    # ------------------------------------------------------------------
    # 2.  Adam optimisation (smooth max)
    if best_seq is None:
        # fall back to uniform if everything failed
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print("--- Phase 2: Adam optimisation (smooth max) ---")
    a_opt, obj_opt = _adam_optimize(best_seq, L, deadline)
    best_seq, best_obj = a_opt, obj_opt

    # ------------------------------------------------------------------
    # 3.  Local random moves (hill climbing)
    print("--- Phase 3: stochastic local moves ---")
    a_final, obj_final = _local_search(best_seq, L, best_obj, deadline)
    best_seq, best_obj = a_final, obj_final

    # ------------------------------------------------------------------
    # Scale to the normalisation used by the reference code and return
    final_seq = (best_seq * target_sum).tolist()
    # one final evaluation for logging purposes (the function is already imported)
    final_score = evaluate_sequence(final_seq)
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
          f"after {time.time() - start_time:.1f}s ---")
    return final_seq
```
