"""
Propose a low‑score step‑function using a larger problem size and a
projected sub‑gradient method with an adaptive line‑search.  The algorithm
runs many short restarts, each starting from a different random shape
(uniform, power‑law or a perturbation of the current best) and performs
a few thousand iterations of a sub‑gradient step that targets the active
peak of the self‑convolution.  For every iteration a few candidate step
sizes are tried and the one giving the smallest score is accepted.
The vector is always projected onto the probability simplex
(a_i ≥ 0, Σ a_i = 1), so the objective 2·n·max(conv)/sum² is scale‑free.
A modest increase of the length (≈ 6000–8000) compared with the previous
solution gives extra “room’’ for the mass and typically pushes the score
below the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Simplex projection (O(n) Duchi et al.)
# --------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------
def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """
    Compute the full linear convolution a*a and return the array together
    with its maximal entry.
    """
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


# --------------------------------------------------------------
# One descent run (sub‑gradient with line‑search)
# --------------------------------------------------------------
def descent_run(a0: np.ndarray,
                max_iters: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a projected sub‑gradient descent on the peak of a*a.
    Returns the best sequence and its corresponding score.
    """
    a = a0.copy()
    n = a.shape[0]

    # enforce simplex – the exact scale does not affect the objective
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_size = 1 << ((2 * n - 1).bit_length())   # next power of two

    best_seq = a.copy()
    best_score = float("inf")

    # constants for step‑size policy
    base_step0 = 0.9   # initial magnitude before 1/√t scaling
    step_factors = [1.0, 0.5, 0.2, 0.1]   # candidates tried each iteration

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------------ evaluate current point -----------------------
        conv_full, max_b = compute_conv_max(a, fft_size)
        sum_a = a.sum()                     # should be ≈1
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[iter {it:5d}] new best score = {score:.6f}")

        # ------------ compute sub‑gradient of the active peak ------
        k = int(np.argmax(conv_full))       # index where the peak occurs
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)

        grad = np.zeros_like(a)
        # grad_i = 2 * a_{k-i} on the overlapping interval
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k - idx]

        # ------------ line‑search on step size --------------------
        step_base = base_step0 / np.sqrt(it)
        best_local_a = a
        best_local_score = score

        for fac in step_factors:
            step = step_base * fac
            cand = project_simplex(a - step * grad)
            conv_cand, max_b_cand = compute_conv_max(cand, fft_size)
            sum_cand = cand.sum()
            cand_score = 2.0 * n * max_b_cand / (sum_cand * sum_cand)

            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand

        a = best_local_a

    return best_seq, best_score

        global_best_seq /= global_best_seq.sum()

    return global_best_seq.tolist()
