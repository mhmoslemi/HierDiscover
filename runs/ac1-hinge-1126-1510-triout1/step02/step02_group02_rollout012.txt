"""
Propose a step‑function (non‑negative sequence) that minimises
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three independent stages that together give a
better chance to escape the local minima observed in the reference code.

1️⃣  **Diverse initialisations** – we keep a copy of the previous global
    construction (if any), add the uniform distribution and several
    random Dirichlet samples.  Each is evaluated with a cheap FFT‑based
    convolution and the best is kept.

2️⃣  **Adam optimisation on a smooth approximation of the max** – the
    true objective `max(conv)` is replaced by a log‑sum‑exp (soft‑max)
    with a temperature parameter `α`.  The gradient of the smooth max
    can be computed analytically via a single extra FFT (correlation
    with a reversed copy of the current vector).  Adam updates are
    performed on the probability simplex (Euclidean projection after each
    step).  The temperature α is increased slowly (4 → 200) so that the
    optimisation first sees a smooth landscape and then concentrates on
    the true supremum.  The routine is run from several random starts;
    the best result across all restarts is retained.

3️⃣  **Temperature‑controlled hill‑climbing** – a simple simulated‑annealing
    move transfers a small random fraction of mass from a source index
    (chosen proportionally to its weight) to a uniformly chosen target.
    Moves that improve the exact `max(conv)` are always accepted; otherwise
    they are accepted with probability `exp(-Δ / T)`, where the temperature
    `T` decays linearly with the remaining time.  This phase explores
    neighbourhoods that Adam cannot reach because the objective is
    non‑smooth.

All phases respect a global time budget (`budget_s`, default 1000 s).  The
final vector is scaled to the normalisation required by `evaluate_sequence`
(`sum = sqrt(2·n)`) and returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2*len(a)-1) using FFT length L."""
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    conv = np.fft.ifft(fft_a * fft_a).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, L: int, alpha: float, phase: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max
        g(alpha) = logsumexp(alpha * conv(a,a)) / alpha
    with respect to a.  The gradient equals 2 * correlation(w, a_rev),
    where w are the soft‑max weights over the convolution entries.
    """
    n = a.shape[0]

    # convolution a*a
    fft_a = np.fft.fft(a, n=L)
    conv = np.fft.ifft(fft_a * fft_a).real[: 2 * n - 1]

    # soft‑max weights (stable version)
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()                         # length 2n‑1, sum = 1

    # correlation with reversed a (gradient of each conv entry is a_rev)
    fft_a_rev = np.conj(fft_a) * phase                         # FFT of a reversed and zero‑padded
    fft_w = np.fft.fft(w, n=L)
    corr = np.fft.ifft(fft_w * fft_a_rev).real                # length L

    # extract the part that corresponds to ∂/∂a_i
    grad = 2.0 * corr[n - 1 : n - 1 + n]                      # shape (n,)
    return grad

            if improve:
                cur_val = new_val
                if cur_val < best_obj:
                    best_obj = cur_val
                    best_a = a_cur.copy()
                    print(f"[Hill] improved conv_max={best_obj:.8e} (T={T:.2e})")

    # --------------------------------------------------------------
    # final scaling to the normalisation used by `evaluate_sequence`
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_a * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)  # just for logging
    print(f"[+] Finished: best score = {final_score:.6f} (n={n})")
    return final_seq
