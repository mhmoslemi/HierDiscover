import numpy as np
import time
import math

def next_pow2(x):
    return 1 << ((x-1).bit_length())

def full_conv(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))
    return conv[: 2*a.shape[0] - 1]

def fast_score(a, fft_len, n):
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

        else:

    # ------------------------------------------------------------------

            best_a = a.copy()

            if rng.random() < prob:

                best_score = cand_score
        it += 1

    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

"""
Search for a non‑negative step sequence with a low evaluation score.

The algorithm works in two stages:

1. **Parametric sweep** – we generate families of symmetric
   exponential or power‑law step sequences
   a_i = exp(-α·i) + exp(-α·(n‑1‑i))   or
   a_i = (i+1)^-p + (n‑i)^-p,
   normalise them to sum‑to‑one and keep the best one.
   These analytically give a much smaller max of the auto‑convolution
   than the uniform distribution.

2. **Stochastic hill‑climb** – starting from the best candidate,
   we repeatedly move a small fraction of mass from a random index
   i to another index j (keeping the total sum unchanged).  Improvements
   are always accepted; occasional uphill moves are accepted with a
   decreasing temperature (simulated annealing) to escape local minima.

The final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based convolution, fast evaluation, etc.)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised (sum(a)=1).  The value returned equals the score that
    ``evaluate_sequence`` would give for the same vector.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())
