"""
Propose a candidate height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ideas:

1️⃣  **Diverse random sampling** – a quick sweep over several parametric
    families (power‑law, exponential, two‑sided power‑law, mixture).
    The best sampled sequence is kept as a warm‑start.

2️⃣  **Soft‑max gradient descent on the simplex** – the non‑smooth
    “max” in the objective is replaced by a smooth soft‑max
      M(α) = Σ_k exp(α·c_k)·c_k / Σ_k exp(α·c_k)
    where c = a*a is the self‑convolution.
    For a fixed α the gradient of M is
        ∇M = 2 · conv( w , a_rev )
    with w the soft‑max weights and a_rev the reversed a.
    An Adam optimiser (with Euclidean projection onto the probability
    simplex) is used to update a.  The temperature α is increased
    gradually so that the optimisation focuses more on the true peak.

3️⃣  **Brief hill‑climbing** – after the Adam phase a cheap
    random‑move local search is performed to polish the result.

The whole search respects the 1000 s budget, prints progress and finally
returns the best sequence found as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using an FFT‑based convolution.
    The sequence is assumed to be non‑negative and already normalised
    (sum(seq) = 1).  The returned value equals the value of
    evaluate_sequence for the same normalised sequence.
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score height sequence.
    The function respects the given time budget and returns a plain
    Python list of non‑negative floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1️⃣  Initialise from previously discovered sequence (if any)
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # fallback: modest power‑law
        init_seq = generate_power_law(8000, 0.55)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full(init_seq.shape, 1.0 / init_seq.shape[0], dtype=np.float64)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Randomised family search (quick diversity)
    # --------------------------------------------------------------
    length_choices = [6000, 8000, 10000, 12000]

            pr = random.uniform(0.38, 0.72)
            wl = random.uniform(0.3, 3.0)

        else:   # "mix" – elementwise maximum of two opposite power‑laws
            p1 = random.uniform(0.38, 0.72)
            p2 = random.uniform(0.38, 0.72)
            seq1 = generate_power_law(n, p1)
            seq2 = generate_power_law(n, p2)[::-1]
            seq = np.maximum(seq1, seq2)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:

        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:

    # --------------------------------------------------------------
    # 3️⃣  Soft‑max Adam refinement on the best candidate
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    L_grad = next_pow2(3 * n)          # for w * a_rev

    # Adam hyper‑parameters

    beta2 = 0.999
    eps = 1e-8

    # we aim to increase α roughly linearly over the main phase
    incr_alpha = (alpha_max - alpha) / 15000.0

    iteration = 0
    while time.time() < deadline - 80.0:   # keep ~80 s for polishing
        iteration += 1
        t += 1

        # ---------- self‑convolution a*a ----------
        fa = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(fa * fa, n=L_self)[: 2 * n - 1]

        # ---------- soft‑max weights ----------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + incr_alpha)
        scaled = conv * alpha
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)
        w = exp_vals / np.sum(exp_vals)

        # ---------- gradient of the soft‑max (M) ----------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=L_grad)
        fr = np.fft.rfft(a_rev, n=L_grad)
        conv_wr = np.fft.irfft(fw * fr, n=L_grad)
        # d_i = conv_wr[i + n - 1]  (i = 0 … n‑1)
        d = conv_wr[n - 1 : n - 1 + n]
        grad_M = 2.0 * d          # ∇M

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_M
        v = beta2 * v + (1.0 - beta2) * (grad_M * grad_M)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = project_simplex(a_candidate)

        # ---------- occasional true evaluation ----------
        if iteration % 500 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] iter {iteration} improved to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Tiny hill‑climbing polishing (random local moves)
    # --------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        # move a small fraction of mass from i to j
        delta = a[i] * random.uniform(0.0, 0.03)
        if delta <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # projection is not needed – sum unchanged and both stays ≥0
        cur_score = fast_score(a_candidate)
        if cur_score < best_score:

            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start:.1f}s] hill‑climb improved to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]  # noqa: F821
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
