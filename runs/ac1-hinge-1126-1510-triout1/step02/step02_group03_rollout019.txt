def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # global best
    best_score = float('inf')
    best_seq = None

    # try multiple lengths
    lengths = [800, 1200, 1600, 2000, 2600, 3200]  # maybe up to 4000
    # ensure not exceed memory/time.

    # For each length, do some restarts
    for n in lengths:
        # Precompute some base shapes for mixture
        base_shapes = []
        i = np.arange(n, dtype=float)
        # uniform
        base_shapes.append(np.ones_like(i))
        # sqrt difference shape
        base_shapes.append(np.sqrt(i+1) - np.sqrt(i))
        # power law exponent random later
        # exponential
        base_shapes.append(np.exp(-0.001 * i))
        # More shapes maybe
        while time.time() < deadline:
            # random mixture of base shapes
            coeffs = np.random.rand(len(base_shapes))
            a = np.zeros(n)
            for c, shape in zip(coeffs, base_shapes):
                a += c * shape
            # add random noise
            a += 0.01 * np.random.rand(n)
            a = np.clip(a, 0, None)
            if a.sum() == 0:
                a = np.ones(n)
            a = a / a.sum()  # simplex

            # Adam optimizer
            # we will keep x_i = log(a_i) but easiest: keep a directly but enforce projection onto simplex after update
            # We maintain momentum m, v per coordinate.

            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            step_size = 1.0  # base learning_rate for Adam (per iteration)
            t = 0

            while time.time() < deadline and t < 2000:
                # compute convolution using fft
                conv = _conv_fft(a)
                M = conv.max()
                # Identify top indices (> M - 1e-8 maybe)
                tol = max(1e-12, 1e-6 * M)
                active = np.where(conv >= M - tol)[0]
                # compute subgradient
                g = _subgrad_active(a, active)   # shape (n,)
                # Adam update on a (we treat a as variables but need to ensure positivity)
                t += 1
                lr_t = step_size * np.sqrt(1 - beta2**t) / (1 - beta1**t)
                m = beta1 * m + (1 - beta1) * g
                v = beta2 * v + (1 - beta2) * (g * g)
                a_new = a - lr_t * m / (np.sqrt(v) + eps)
                # projection onto simplex
                a_new = _project_to_simplex(a_new)
                # evaluation
                score = evaluate_sequence(a_new.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a_new.copy()
                    print(f"[n={n}] New best {best_score:.6f} at t={t} (elapsed={time.time()-start:.1f}s)")
                a = a_new

            # End inner optimizer, possibly early exit
            if time.time() >= deadline:
                break

    # fallback
    if best_seq is None:
        # uniform seq
        n = 1542
        a = np.ones(n) / n
        best_seq = a

    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning best score {best_score:.6f} after {time.time()-start:.1f}s")
    return final_seq.tolist()

"""
Propose a step‑function sequence with a low evaluation score.

The search proceeds in three stages:
  1.  Build a set of diverse initial candidates (sqrt‑diff, power‑law,
      exponential, random and a sweep over the “difference‑of‑powers”
      family).  The best of these seeds is kept.
  2.  Refine the current best seed with an Adam‑accelerated projected
      sub‑gradient method on the maximum auto‑correlation entry.
      Multiple restarts are performed while time permits.
  3.  A short simulated‑annealing / “mass‑swap” polishing phase
      (randomly move a small amount of mass between two positions) is
      applied to the best sequence found so far.

All updates are projected onto the probability simplex
{a_i ≥ 0 , Σ a_i = 1}; scaling the final vector to [0,1000] does not
affect the score because the evaluation function is homogeneous.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution a*a using FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the maximal autocorrelation entry
# ----------------------------------------------------------------------
def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the autocorrelation peak over active indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_idx.size == 0:
        return g
    for k in active_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g /= float(active_idx.size)
    return g
