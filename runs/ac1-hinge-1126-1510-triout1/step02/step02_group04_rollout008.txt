def _project_to_simplex(v):
    # v can be any array; returns projection onto simplex of sum=1, nonnegative.
    # Implementation: sort v decreasing, compute cumulative sum, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv/np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):

    if v.ndim != 1:

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two phases:

1) **Sub‑gradient mirror descent** (exponentiated gradient) on the true
   non‑smooth objective.  At each iteration the convolution is computed by
   FFT, the active indices of the maximal autocorrelation are identified,
   an (averaged) sub‑gradient is obtained via a correlation FFT and the
   simplex variables are updated in log‑space with a decaying stepsize.
   Random jitter is injected occasionally to avoid stagnation.

2) **Smooth‑max Adam refinement** – essentially the method used in the
   reference code – starting from the best point found in phase 1.
   The smooth approximation of the max is gradually sharpened (β is
   increased) while Adam optimisation on the soft‑max parametrisation is
   applied.

Both phases respect a 1000 s wall‑clock budget and keep track of the best
sequence found.  The final sequence is scaled to the normalisation
Σa = √(2 n) required by the reference implementation and clipped to the
range enforced by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    vmax = v.max()
    e = np.exp(v - vmax)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of
    Duchi et al. (2008) and works for any 1‑D NumPy array.
    """
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_length(min_len: int) -> int:
    """
    Return the next power of two >= min_len (suitable for np.fft).
    """
    return 1 << ((min_len - 1).bit_length())


def _conv_and_rev(a: np.ndarray, L: int, n: int, phase: np.ndarray):
    """
    Compute the full linear convolution a*a (length 2·n‑1) via FFT of length L
    and also return the FFT of the reversed a needed for correlation.
    Returns (conv, A_hat, A_rev_hat) where:
        conv      – real array of length 2·n‑1,
        A_hat     – FFT of a (length L),
        A_rev_hat – FFT of a reversed (obtained by conjugation+phase).
    """
    A_hat = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(A_hat * A_hat))
    conv = conv_full[: 2 * n - 1]
    A_rev_hat = np.conj(A_hat) * phase
    return conv, A_hat, A_rev_hat


def _correlation_from_fft(W_hat: np.ndarray, A_rev_hat: np.ndarray,
                         n: int) -> np.ndarray:
    """
    Given the FFT of a weight vector w (length L) and the FFT of
    a reversed a, compute 2·(w ⋆ a_rev) on the segment that corresponds to
    the gradient w.r.t. a (exactly n entries).
    """
    corr_full = np.real(np.fft.ifft(W_hat * A_rev_hat))
    # the relevant slice is indices n‑1 … n‑1+n‑1
    return 2.0 * corr_full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation function.  The method combines a sub‑gradient
    mirror‑descent phase with a smooth‑max Adam refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # ----- Determine length of the sequence --------------------------------
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 6000  # fallback length
    target_sum = math.sqrt(2.0 * n)  # normalisation used by reference code

    print(f"--- Optimising sequence of length n={n}, target sum={target_sum:.3f} ---")

    # ----- FFT parameters ----------------------------------------------------
    # For convolution (a*a) we need at least 2·n‑1 points.
    # For the correlation w⋆a_rev we need at least 3·n‑2 points.
    # Use a single length covering both requirements.
    L = _fft_length(3 * n - 2)

    # Pre‑compute the phase factor that converts FFT(a) → FFT(a[::-1]).
    k = np.arange(L)
    phase = np.exp(-2j * np.pi * k * (L - 1) / L)

    # ----- Initial sequence (simplex, Σa = 1) -------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a = seq0 / seq0.sum()
    except Exception:
        a = np.full(n, 1.0 / n)

    # ----- Book‑keeping -------------------------------------------------------
    best_a = a.copy()
    # initial objective value (with Σa = 1)
    conv, _, _ = _conv_and_rev(a, L, n, phase)
    best_score = 2.0 * n * conv.max()
    print(f"[init]   score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 1 – Sub‑gradient mirror descent (exponentiated gradient)
    # ----------------------------------------------------------------------
    eta0 = 0.6    # initial stepsize in log‑space
    max_iters_phase1 = 20000  # hard cap – actual count limited by time
    eps_log = 1e-30            # avoid log(0)
    iter1 = 0

    deadline_phase1 = start_time + (budget_s - 5.0) * 0.55  # ~55 % of the budget

    while time.time() < deadline_phase1 and iter1 < max_iters_phase1:
        # --- convolution and max -------------------------------------------------
        conv, A_hat, A_rev_hat = _conv_and_rev(a, L, n, phase)
        max_val = conv.max()

        # --- active set (indices attaining the max) ----------------------------
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # --- build uniform weight vector on the active set --------------------
        w = np.zeros(2 * n - 1, dtype=float)
        w[active] = 1.0 / len(active)

        # --- gradient w.r.t. a (sub‑gradient of the max) -----------------------
        W_hat = np.fft.fft(w, n=L)
        grad = _correlation_from_fft(W_hat, A_rev_hat, n)

        # --- exponentiated‑gradient update (log‑space) -------------------------
        eta = eta0 / math.sqrt(iter1 + 1.0)
        log_a = np.log(a + eps_log)
        log_a -= eta * grad
        # numerical stabilisation – remove common offset
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()               # projection onto the simplex

        # --- keep best primitive ------------------------------------------------
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[EG]    iter {iter1:5d}  new best = {best_score:.6f}")

        # --- occasional random jitter to escape plateaus -----------------------
        if (iter1 + 1) % 4000 == 0:
            noise = np.random.normal(scale=0.01, size=n)
            a = a + noise
            a = _project_simplex(a)

        iter1 += 1

    # ----------------------------------------------------------------------
    # Phase 2 – Smooth‑max Adam refinement (as in the reference implementation)
    # ----------------------------------------------------------------------
    # reuse the best point from phase 1 as a start, together with a few
    # alternative seeds to increase robustness.
    start_vectors = [best_a, np.full(n, 1.0 / n)]
    for _ in range(2):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    beta_start = 5.0
    beta_end = 120.0
    max_iters_per_restart = 8000

    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break
        # initialise soft‑max parameters
        v = np.log(a_init + eps_log)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                break

            a = _softmax(v)

            # convolution (a * a)
            conv, A_hat, A_rev_hat = _conv_and_rev(a, L, n, phase)
            conv_max = conv.max()
            cur_score = 2.0 * n * conv_max

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam]  restart {restart_id} iter {it:5d}  new best = {best_score:.6f}")

            # smooth‑max weights (soft‑max over convolution entries)
            beta = beta_start + (beta_end - beta_start) * it / max_iters_per_restart
            exp_term = np.exp(beta * (conv - conv_max))
            w = exp_term / exp_term.sum()

            # gradient w.r.t. a (smooth version)
            W_hat = np.fft.fft(w, n=L)
            grad_a = _correlation_from_fft(W_hat, A_rev_hat, n)

            # gradient w.r.t. soft‑max parameters v (see derivation in the paper)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on v
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # ----------------------------------------------------------------------
    # Final scaling, clipping and return
    # ----------------------------------------------------------------------
    best_seq = best_a * target_sum                 # enforce Σa = √(2 n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)      # safety for evaluate_sequence

    # Optional: report the value reported by the provided evaluator
    try:
        eval_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        eval_score = float('nan')
    print(f"--- Finished: best raw score = {best_score:.6f}, "
          f"evaluator = {eval_score:.6f} (n={n}) ---")

    return best_seq.tolist()
