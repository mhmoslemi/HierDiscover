"""
Propose a candidate step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three phases:

1. **Initialisation** – start from the best known sequence (if provided)
   or from a simple power‑law shape. The vector is normalised to sum to 1.

2. **Adam‑based smoothed max‑gradient descent** – at each iteration we
   compute the self‑convolution of the current vector, turn the
   convolution into a smooth approximation of the L∞‑norm using a
   soft‑max weighting, and take a projected Adam step on the simplex
   (non‑negative entries that sum to 1).  The smoothing factor (soft‑max
   temperature) is increased slowly, turning the objective from a
   smooth average into the true max.

3. **Periodic evaluation & random restarts** – every few thousand
   iterations we evaluate the cheap FFT based score and keep the best
   vector seen so far.  Occasionally a small Gaussian perturbation of
   the current best is applied and the Adam moments are reset to help
   escape shallow local minima.

The whole routine respects the supplied time budget (default 1000 s)
and finally returns the best sequence as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using an FFT convolution.
    The sequence is assumed to be non‑negative and already normalised
    to sum to 1 (the scale does not matter for the ratio).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # length of linear convolution
    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (i = 0 … n‑1) – not normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) – not normalised."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^(-p_left) + w_right * (n-i)^(-p_right)
    – not normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def random_initial(n: int) -> np.ndarray:
    """
    Produce a random-ish initial vector by mixing a few families and
    adding a tiny amount of noise.  The result is normalised to sum to 1.
    """
    family = random.choice(["power", "exp", "twosided", "uniform"])
    if family == "power":
        p = random.uniform(0.4, 0.7)
        seq = generate_power_law(n, p)
    elif family == "exp":
        lam = random.uniform(5e-4, 2.5e-2)
        seq = generate_exponential(n, lam)
    elif family == "twosided":
        p_l = random.uniform(0.4, 0.7)
        p_r = random.uniform(0.4, 0.7)
        w_l = random.uniform(0.3, 3.0)
        w_r = random.uniform(0.3, 3.0)
        seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)
    else:  # uniform
        seq = np.ones(n, dtype=np.float64)

    # add a tiny jitter to avoid exact symmetry
    jitter = np.random.rand(n) * 1e-6
    seq = seq + jitter
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total < 1e-12:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within the provided time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for final eval

    # ------------------------------------------------------------------
    # 1️⃣  Choose problem size and an initial vector
    # ------------------------------------------------------------------
    # Prefer to reuse the best known construction if present
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
        n = init_seq.size
        print(f"[init] Using supplied height_sequence_1 of length {n}.")
    except Exception:
        # no external sequence – fall back to a modest power‑law
        n = 10000
        init_seq = generate_power_law(n, 0.55)
        print(f"[init] No external sequence; using power‑law length {n}.")

    # Normalise – the objective is scale‑invariant
    init_seq = np.maximum(init_seq, 0.0)
    total = init_seq.sum()
    if total < 1e-12:
        init_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init_seq = init_seq / total

    # Keep the current best
    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] Initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑based smoothed max‑gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters (tuned empirically)
    base_lr = 0.5                # base learning rate for Adam
    alpha = 10.0                 # initial soft‑max temperature
    alpha_inc = 1.03             # multiplicative increase every epoch
    epoch_len = 5000             # iterations after which we increase alpha
    eval_interval = 2000         # evaluate fast_score this often
    restart_interval = 20000     # add noise and reset moments

    # pre‑computed FFT lengths
    conv_len = 2 * n - 1
    fft_len_a = next_pow2(conv_len)                # for a*a
    fft_len_wa = next_pow2(conv_len + n - 1)       # for w*a_rev

    iteration = 0
    while time.time() < deadline - 2.0:
        iteration += 1

        # ---- compute self‑convolution a*a ----
        fa = np.fft.rfft(a, n=fft_len_a)
        conv = np.fft.irfft(fa * fa, n=fft_len_a)[:conv_len]

        max_conv = conv.max()

        # ---- soft‑max weighting of convolution entries ----
        # shift by max to avoid overflow; diff ≤ 0
        diff = conv - max_conv
        # scaling of temperature – larger alpha → sharper approximation
        w_raw = np.exp(alpha * diff)
        w_sum = w_raw.sum()
        if w_sum == 0:
            # fallback (should not happen)
            w = np.full_like(w_raw, 1.0 / conv_len)
        else:
            w = w_raw / w_sum

        # ---- gradient of the smoothed max ----
        # g_i = 2 * Σ_k w_k * a_{k-i}   (linear correlation)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_wa)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_wa)
        conv_wa = np.fft.irfft(fw * fa_rev, n=fft_len_wa)[: (conv_len + n - 1)]
        # slice corresponds to indices i = 0..n-1 in the correlation result
        grad = 2.0 * conv_wa[n - 1 : n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        # learning‑rate schedule (optional decay)
        lr = base_lr / (np.sqrt(iteration) + 1.0)

        a_candidate = a - lr * (m_hat / (np.sqrt(v_hat) + eps))

        # ---- simplex projection ----
        a = project_simplex(a_candidate)

        # ---- periodic evaluation ----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.1f}s] iter={iteration:,} "
                    f"new best fast_score = {cur_score:.6f}"
                )

        # ---- increase smoothing temperature slowly ----
        if iteration % epoch_len == 0:
            alpha *= alpha_inc

        # ---- occasional random restart around the current best ----
        if iteration % restart_interval == 0:
            # perturbed copy of best_seq + projection
            perturb = np.random.randn(n) * 0.01
            a = best_seq + perturb
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            print(
                f"[{time.time() - start_time:.1f}s] iter={iteration:,} restart with perturbation"
            )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification via the provided evaluate_sequence
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]  # noqa: F821
    print(
        f"Finished. Best final score reported by evaluate_sequence = {final_score:.6f}"
    )
    return final_seq
