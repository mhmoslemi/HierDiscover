def subgradient_descent(a, fft_len, n, deadline=None, max_iter=200, step0=0.1, decay=0.99, epsilon=1e-12):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = step0
    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        conv = conv_fft(best, fft_len, n)
        max_c = conv.max()
        # active indices
        active = np.where(conv >= max_c - epsilon)[0]
        # compute gradient
        grad = np.zeros_like(best)
        # for each active
        # vectorized: use loops
        for i in active:
            idx = np.arange(n)
            mask = (i - idx >= 0) & (i - idx < n)
            grad[mask] += best[i - idx[mask]]
        if np.all(grad == 0):
            break
        # update away from gradient
        new = best - step * grad
        # enforce non-neg and renorm
        new = np.maximum(new, 0.0)
        if new.sum() == 0.0:
            # fallback: add small uniform
            new = np.full(n, 1.0 / n)
        else:
            new = new / new.sum()
        new_score = fast_score(new, fft_len, n)
        if new_score < best_score:
            best = new
            best_score = new_score
            step = step * 1.05  # increase step if improvement
        else:
            step = step * decay
        # optional early exit if step small
        if step < 1e-12:
            break
    return best, best_score

def subgradient_descent(a, fft_len, n, deadline=None, max_iter=200, step0=0.1, decay=0.99, epsilon_rel=1e-12):

   best = a.copy()
   best_score = fast_score(best, fft_len, n)
   step = step0
   for it in range(max_iter):
       curr_time = time.time()
       if deadline is not None and curr_time >= deadline:
           break
       conv = conv_fft(best, fft_len, n)
       max_c = conv.max()
       # active indices with values close to max
       eps = epsilon_rel * max_c if max_c > 0 else epsilon_rel
       active = np.where(conv >= max_c - eps)[0]
       # compute gradient
       grad = np.zeros_like(best)

"""
Search for a low‑evaluation step‑function sequence.

The algorithm combines several complementary optimisation phases:

1.  Multi‑restart L‑BFGS on a smooth approximation of the maximal
    convolution entry.  A continuation schedule gradually reduces the
    smoothing (lam_factor) so that the solution approaches the true
    max‑convolution objective.

2.  Sub‑gradient descent on the exact max‑convolution (the non‑smooth
    objective).  For the currently active convolution index the
    gradient is the reversed slice of the sequence; the update moves
    mass away from the indices that contribute most to the peak.

3.  A greedy “hill‑climb” that repeatedly transfers a small random
    fraction of mass between two positions while preserving the
    simplex constraint.

4.  A final refinement that repeats the smooth L‑BFGS schedule with
    more iterations and a short polishing of sub‑gradient descent
    followed by hill‑climbing.

All moves keep the coefficients non‑negative and normalised (the
objective is scale‑invariant, so a sum of 1 is used for convenience).
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (non‑negative, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast computation of 2·n·max(a*a) for a probability vector (sum=1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the convolution together with its gradient w.r.t. y.
    lam = max(conv) * lam_factor acts as the smoothing width.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = max_c + lam·log(∑exp((conv‑max_c)/lam))
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # shape (2·n‑1,)

    # gradient w.r.t. a (convolution of a with w restricted to n entries)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]  # length n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 80,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations, each with a different
    smoothing factor (lam_factor).  The result of one stage seeds the next.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Sub‑gradient descent for the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float | None = None,
    max_iter: int = 200,
    step0: float = 0.1,
    decay: float = 0.97,
    eps_rel: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient descent on 2·n·max(conv) (the exact non‑smooth
    objective).  The gradient for a given active convolution index i
    is the reversed slice of a that contributes to that entry.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = step0

    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break

        conv = conv_fft(best, fft_len, n)
        max_c = conv.max()
        eps = eps_rel * max_c if max_c > 0 else eps_rel
        active = np.where(conv >= max_c - eps)[0]

        # accumulate gradient from all active indices
        grad = np.zeros_like(best)
        for idx in active:
            # indices j with 0 ≤ idx‑j < n
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            grad[j] += best[idx - j]

        if not np.any(grad):
            break

        new = best - step * grad
        new = np.maximum(new, 0.0)

        s = new.sum()
        if s == 0.0:
            new = np.full(n, 1.0 / n)
        else:
            new = new / s

        new_score = fast_score(new, fft_len, n)

        if new_score < best_score:
            best = new
            best_score = new_score
            step *= 1.02  # tiny boost when we improve
        else:
            step *= decay

        if step < 1e-14:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (mass transfers)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from one index to another.
    Keep the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5 % of source mass
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score
