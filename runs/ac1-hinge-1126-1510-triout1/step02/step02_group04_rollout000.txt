def propose_candidate(seed=..., budget_s=1000, **kwargs):
    import time, math, numpy as np

    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # choose n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # compute FFT length for conv and correlation
    L = 1 << ((2 * n - 1).bit_length())

    # define helper functions

    def softmax(v):
        vmax = v.max()
        e = np.exp(v - vmax)
        return e / e.sum()

    def conv_full(a):
        # compute convolution using FFT; output length 2*n -1
        # a padded to L
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        return conv

    def correlation_full(w, a):
        # Note: correlation w * a_rev approximated as w convolve a_rev
        # in frequency domain: fft(w)*fft(a_rev)
        # but a_rev is reversed a, i.e., a[::-1]
        rev = a[::-1]
        fw = np.fft.rfft(w, n=L)
        fr = np.fft.rfft(rev, n=L)
        corr = np.fft.irfft(fw * fr, n=L)
        # central part corresponding to n..2*n-2 (offset indices)
        start = n-1
        return corr[start:start+n]

    def objective_maxconv(a):
        # a sum = 1; return max conv
        conv = conv_full(a)
        return conv.max()

    # Set best tracking
    best_seq = None
    best_val = float('inf')

    # Phase 0: param search (exponential)
    def exponential_shape(lmbda):
        # lmbda>0 
        i = np.arange(n, dtype=float)
        a = np.exp(-lmbda * i)
        a /= a.sum()
        return a

    # try a few λ values log spaced
    for lmbda in np.logspace(-3, 0.5, num=30):
        if time.time() > deadline:
            break
        a = exponential_shape(lmbda)
        val = objective_maxconv(a)
        if val < best_val:
            best_val = val
            best_seq = a * target_sum
            print(f"Exp λ={lmbda:.4f} new best {best_val:.6f}")

    # Phase 1: Adam optimization from multiple initializations
    # define parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.06
    beta_start = 5.0
    beta_end = 200.0
    max_iters = 15000

    # assemble initial vectors >= includes best found from phase0, uniform, random Dirichlet
    init_vectors = []

    if best_seq is not None:
        init_vectors.append(best_seq / target_sum)

    # uniform
    init_vectors.append(np.full(n, 1.0 / n))

    # previous best if exists (height_sequence_1)
    try:
        seq_prev = np.asarray(height_sequence_1, dtype=float)
        init_vectors.append(seq_prev / seq_prev.sum())
    except Exception:
        pass

    # add some random Dirichlet with small concentration for sparsity
    for conc in [0.3, 0.5, 0.8]:
        for _ in range(2):
            init_vectors.append(np.random.dirichlet(np.full(n, conc)))

    # add power-law shapes
    for p in [0.3, 0.6, 1.2, 1.8]:
        w = (np.arange(1, n+1, dtype=float) ** p)
        init_vectors.append(w / w.sum())

    # Adam loops
    for restart_id, a0 in enumerate(init_vectors, 1):
        if time.time() > deadline:
            break
        v = np.log(a0 + 1e-12)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)
        for it in range(max_iters):
            if time.time() > deadline:
                break
            a = softmax(v)
            conv = conv_full(a)
            conv_max = conv.max()
            cur_val = conv_max  # because sum normalized to 1
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a * target_sum
                print(f"[restart {restart_id:02d}] iter {it:5d} new best {best_val:.6f}")

            # smooth max weight
            beta = beta_start + (beta_end - beta_start) * it / max_iters
            # using max for numerical stability
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()
            # compute gradient via correlation
            grad_a = 2.0 * correlation_full(w, a)
            # gradient wrt v (softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            t = it + 1
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)
            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # Phase 2: simple local random perturbation (hill climbing)
    # we keep a copy of best normalized a
    if best_seq is not None:
        a = best_seq / target_sum
    else:
        a = np.full(n, 1.0 / n)

    # hill climbing for remaining budget
    temp = 0.1
    while time.time() < deadline:
        # propose a perturbation
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # amount to move
        delta = (np.random.rand() * 0.02) * a[i]  # move up to 2% of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegativity (should be)
        # evaluate
        val_new = objective_maxconv(a_new)
        if val_new < best_val or np.random.rand() < np.exp(-(val_new - best_val) / temp):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a * target_sum
                print(f"Hill climb new best {best_val:.6f}")
        # decay temperature
        temp *= 0.99995
        if temp < 1e-6:
            temp = 1e-6

    # final clip
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"--- Final best score={best_val:.6f} (n={n}) ---")
    return best_seq.tolist()

"""
Propose a step‑function sequence minimising
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1.  **Parametric sweep** – try exponential (and optionally Gaussian) shapes
    a_i ∝ exp(-λ·i).  Small λ gives a long tail, large λ a very spiky
    distribution.  For each λ we evaluate the true max‑convolution
    (sum normalised to 1) and keep the best.

2.  **Adam optimiser on the smooth‑max surrogate** – initialise several
    restarts (including the best shape from phase 1, uniform, sparse
    Dirichlet draws and power‑law profiles).  Variables are the
    unconstrained log‑soft‑max parameters `v`; the objective is a
    log‑sum‑exp approximation of the max with a temperature β that grows
    from 5 to 200.  Adam updates are performed until the time budget is
    exhausted.  The best sequence found (scaled to Σa=√(2n)) is stored.

3.  **Simple hill‑climbing (simulated‑annealing style)** – starting from
    the best sequence, repeatedly move a small amount of mass from a
    random index i to another index j.  Moves that improve the true
    max‑convolution are accepted; otherwise they are accepted with a
    probability decreasing with a temperature schedule.

All operations respect the non‑negativity and sum constraints, and the
final sequence is clipped to the range `[0,1000]` required by
`evaluate_sequence`.  The routine adapts the length `n` from the global
`height_sequence_1` if it exists, otherwise defaults to 8000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a*a using FFT.
    Output length = 2*len(a)-1 (truncated to that length).
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear correlation of w with a_rev (a reversed).
    Returned length = len(w) + len(a_rev) - 1.
    """
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


def _objective_maxconv(a: np.ndarray, L_conv: int) -> float:
    """Return max(conv(a,a)) for a that already satisfies Σa = 1."""
    conv = _conv_full(a, L_conv)
    return conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises the evaluation function.  The routine respects the given
    time budget and returns the best sequence found (as a plain Python
    list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------------------
    # Determine problem size (length of the sequence)
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 8000                     # reasonable default if not provided

    target_sum = math.sqrt(2.0 * n)   # scaling used by the reference code
    print(f"--- Optimising sequence of length n={n} (target sum={target_sum:.3f}) ---")

    # FFT lengths (next power of two for speed)
    L_conv = 1 << ((2 * n - 1).bit_length())   # for a*a
    L_corr = 1 << ((3 * n - 2).bit_length())   # for correlation with weights

    # -------------------------------------------------------------
    # Keep track of the best solution seen so far
    best_seq = None          # will store the *scaled* sequence
    best_val = float("inf")  # stores the objective value (max conv)

    # -------------------------------------------------------------
    # Phase 0 – parametric sweep over exponential shapes
    def exponential_shape(lmbda: float) -> np.ndarray:
        """Return a normalised exponential shape a_i ∝ exp(-λ·i)."""
        i = np.arange(n, dtype=float)
        a = np.exp(-lmbda * i)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        return a

    # Sweep λ on a log‑scale (including very flat and very steep distributions)
    for lmbda in np.logspace(-3, 0.5, num=30):
        if time.time() > deadline:
            break
        a = exponential_shape(lmbda)
        val = _objective_maxconv(a, L_conv)
        if val < best_val:
            best_val = val
            best_seq = a * target_sum
            print(f"[exp sweep] λ={lmbda: .4f}  new best {best_val:.6f}")

    # -------------------------------------------------------------
    # Phase 1 – Adam optimisation on smooth‑max surrogate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.06
    beta_start = 5.0
    beta_end = 200.0
    max_iters = 15000

    # Build a collection of starting vectors on the unit simplex
    start_vectors = []

    # (a) Best from phase 0 (if any)
    if best_seq is not None:
        start_vectors.append(best_seq / target_sum)

    # (b) Uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # (c) Previously discovered best (global variable)
    try:
        seq_prev = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if seq_prev.size == n:
            start_vectors.append(seq_prev / seq_prev.sum())
    except Exception:
        pass

    # (d) Sparse Dirichlet draws (concentration < 1 encourages sparsity)
    for conc in (0.3, 0.5, 0.8):
        for _ in range(2):
            start_vectors.append(np.random.dirichlet(np.full(n, conc)))

    # (e) Simple power‑law profiles
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        start_vectors.append(w / w.sum())

    # Adam loops over all initialisations
    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        # Initialise soft‑max parameters v
        v = np.log(a_init + 1e-12)
        m = np.zeros_like(v)   # 1st moment
        v2 = np.zeros_like(v)  # 2nd moment

        for it in range(max_iters):
            if time.time() > deadline:
                break

            # Current point on the simplex
            a = _softmax(v)                    # Σa = 1, a ≥ 0

            # True objective (max convolution) for monitoring
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            cur_val = conv_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a * target_sum
                print(f"[restart {restart_id:02d}] iter {it:5d}  new best {best_val:.6f}")

            # ---- smooth‑max (log‑sum‑exp) approximation ----
            beta = beta_start + (beta_end - beta_start) * it / max_iters
            # soft‑max over convolution entries (numerically stable)
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()            # weights sum to 1

            # ---- gradient w.r.t. a ----
            # 2 * correlation(w, a_rev)  (see derivation in comments)
            a_rev = a[::-1]
            corr_full = _correlation_full(w, a_rev, L_corr)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

            # ---- gradient w.r.t. soft‑max parameters v ----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ---- Adam update ----
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # Phase 2 – hill‑climbing (local random perturbations)
    # Work on the normalised vector a (Σa = 1)
    if best_seq is not None:
        a = best_seq / target_sum
    else:
        a = np.full(n, 1.0 / n)

    temp = 0.1  # initial temperature for simulated‑annealing acceptance

    while time.time() < deadline:
        # Propose a small transfer of mass from i to j
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # Move up to 2 % of the source weight
        delta = (np.random.rand() * 0.02) * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Evaluate the true max‑convolution
        val_new = _objective_maxconv(a_new, L_conv)

        # Acceptance rule
        if val_new < best_val or np.random.rand() < math.exp(-(val_new - best_val) / temp):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a * target_sum
                print(f"[hill‑climb] new best {best_val:.6f}")

        # Cool down
        temp = max(temp * 0.99995, 1e-6)

    # -------------------------------------------------------------
    # Finalise – ensure the sequence respects the clipping limits
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Re‑scale exactly to the required L1‑norm (guard against tiny drift)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq * (target_sum / total)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished: best score = {best_val:.6f} (n={n}) ---")
    return best_seq.tolist()
