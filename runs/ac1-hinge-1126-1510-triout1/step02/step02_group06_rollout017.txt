"""
Propose a step‑function sequence minimizing
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

1.  Warm‑start from the global ``height_sequence_1`` if it exists,
    otherwise from a Beta‑distribution shape (flexible two‑sided power‑law).

2.  **Diversified random search** (≈25 % of the time budget):
    generate candidates from several parametric families
    (power‑law, exponential, two‑sided power‑law, Beta‑distribution),
    normalise them to sum = 1 and keep the best according to a fast
    FFT‑based evaluation.

3.  **Refinement** (the remaining time):
    projected (simplex) gradient descent on the true objective.
    The sub‑gradient is taken from *all* convolution entries that are
    within a shrinking tolerance of the current maximum, i.e. a
    multi‑active set.  Momentum (Nesterov‑style) and an adaptive
    step‑size  η = 0.5/√t are used.  Occasionally a small random
    jitter is tried to escape shallow local minima.

4.  Return the best sequence found (list of floats).  The final
    sequence is also evaluated with the provided ``evaluate_sequence``.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective using FFT."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) """
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) """
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right) """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on {1,…,n}:
        a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    return seq

# Main search routine
# ----------------------------------------------------------------------
