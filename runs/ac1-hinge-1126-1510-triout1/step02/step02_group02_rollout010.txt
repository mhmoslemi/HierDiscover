"""
Search for a non‑negative step function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is built around a strict time budget (default 1000 s) and
consists of four phases:

1. **Parametric sweep** – generates a few classic families (power–law,
   exponential, U‑shaped) and keeps the best normalized candidate.

2. **Multi‑start Adam optimisation** – a smooth approximation of the
   max (soft‑max) is minimised using Adam on the probability simplex.
   Several random Dirichlet starts are tried; the best result of all
   runs is kept.

3. **Sub‑gradient refinement** – after the Adam phase we switch to the
   true L∞‑objective.  At each iteration we locate the worst convolution
   index, compute its sub‑gradient (a simple reversed‑shift of the
   current vector) and perform a decreasing‑step projected sub‑gradient
   descent.

4. **Biased random‑move annealing** – a cheap hill‑climbing / simulated
   annealing phase that repeatedly moves a small fraction of mass from
   entries that heavily contribute to the current worst convolution to
   entries that contribute little.  The temperature decays linearly with
   the remaining time.

All phases respect the provided budget, the sequence is always kept
on the simplex (non‑negative and sum = 1) and finally scaled to the
reference normalisation `sum = sqrt(2·n)` before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """ Return max(conv(a,a)) using FFT of size L.  a is assumed to sum to 1. """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """ Full convolution (real) of a with itself, length 2n‑1. """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """ Smooth max (log‑sum‑exp) weights for a given α. """
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()
    return w


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    L: int,
    max_iters: int,
    deadline: float,
    phase: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max surrogate.
    Returns the best vector (sum=1) found and its max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max(a, L)

    # Adam hyper‑parameters
    lr0 = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha_start, alpha_end = 4.0, 200.0  # a wider range than the reference code

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and its soft‑max weights
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv.max()
        if cur_max < best_obj:
            best_obj, best_a = cur_max, a.copy()

        # smooth‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth‑max (2 · correlation)
        fft_w = np.fft.fft(w, n=L)
        fft_a_rev = np.conj(fft_a) * phase
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update on the simplex (Euclidean projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        a = a_new

    # final evaluation
    final_obj = _conv_max(best_a, L)
    return best_a, final_obj


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    L: int,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true max‑convolution.
    Learning rate decays like 1/sqrt(t).  Returns best vector and its max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max(a, L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, L)
        cur_max = conv.max()
        if cur_max < best_obj:
            best_obj, best_a = cur_max, a.copy()

        k_star = int(np.argmax(conv))

        # sub‑gradient: g_i = 2 * a_{k*‑i} for valid indices
        idx = np.arange(n)
        mask = (k_star - idx >= 0) & (k_star - idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[k_star - idx[mask]]

        # step size (simple schedule)
        lr = 0.5 / np.sqrt(it)
        a_new = a - lr * g
        a_new = _project_simplex(a_new)
        a = a_new

    final_obj = _conv_max(best_a, L)
    return best_a, final_obj


# ----------------------------------------------------------------------
def _biased_random_moves(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Heuristic hill‑climbing with a temperature that decays linearly.
    Mass is moved from indices that heavily contribute to the current
    worst convolution to indices that contribute little.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max(a, L)

    start_time = time.time()
    while time.time() < deadline:
        # recompute the current worst convolution index
        conv = _conv_full(a, L)
        k_star = int(np.argmax(conv))

        # contribution of each coordinate to that worst index
        idx = np.arange(n)
        mask = (k_star - idx >= 0) & (k_star - idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[k_star - idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            src_probs = a  # fallback
        else:
            src_probs = contrib / total_contrib

        # source index (biased towards high contribution)
        i = np.random.choice(n, p=src_probs)
        # target index (biased towards low mass)
        inv_mass = 1.0 / (a + 1e-12)
        tgt_probs = inv_mass / inv_mass.sum()
        j = np.random.choice(n, p=tgt_probs)
        if i == j:
            continue

        # move a random fraction of a[i]; allow relatively large jumps
        max_frac = 0.25
        delta = np.random.rand() * max_frac * a[i]
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        obj_new = _conv_max(a_new, L)

        # Simple simulated‑annealing acceptance
        if obj_new < best_obj:
            best_obj, best_a, a = obj_new, a_new.copy(), a_new
        else:
            # temperature proportional to remaining time
            elapsed = time.time() - start_time
            frac = max(0.0, 1.0 - elapsed / (deadline - start_time))
            T = 0.02 * frac  # tiny temperature – just to escape tiny plateaus
            if np.random.rand() < math.exp(-(obj_new - best_obj) / (T + 1e-12)):
                a = a_new

    final_obj = _conv_max(best_a, L)
    return best_a, final_obj
