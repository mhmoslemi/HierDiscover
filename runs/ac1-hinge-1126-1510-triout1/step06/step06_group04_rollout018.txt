```
def exp_subgradient_descent(a_start, fft_len, n, deadline, eta0=0.5):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute gradient (subgradient) g_i = 2*a_{m-i}
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >=0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # step size
        eta = eta0 / np.sqrt(it)
        # exponentiated update
        a_new = a * np.exp(-eta * g)
        # avoid all zeros
        sum_a_new = np.sum(a_new)
        if sum_a_new <= 0:
            # fallback to small random perturbation
            a_new = a + 1e-12
            sum_a_new = np.sum(a_new)
        a_new = a_new / sum_a_new
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Method
------
* Work on the probability simplex (∑a = 1) – the evaluation is scale‑invariant.
* Choose a moderate length (≈ 5000–8000).  If a previous construction
  ``height_sequence_1`` exists we reuse its length.
* Multi‑restart continuation L‑BFGS on a smooth log‑sum‑exp approximation
  of the max‑convolution (geometric schedule of the smoothing factor).
* Aggressive projected sub‑gradient descent (standard linear sub‑gradient)
  for a large part of the remaining budget.
* A short “exponential‑gradient” sub‑gradient phase (mirror‑descent) to
  escape shallow plateaux.
* Fine L‑BFGS refinement with a very low smoothing factor.
* Simulated‑annealing hill‑climb followed by a final greedy hill‑climb.
* The best sequence found is returned; the harness will call
  ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast approximation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    # shifted max to 0 before exponentiation
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: each stage uses a smaller smoothing factor.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random, exp, power, edge, sidon, binary, bump, harmonic, best.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                sums.add(2 * candidate)
            if len(chosen) >= int(np.sqrt(2 * n)):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (equivalently minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))  # index of maximal entry
        # Sub‑gradient of that entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def exp_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.50,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) variant of sub‑gradient descent.
    Keeps iterates strictly positive and projects by normalisation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a * np.exp(-eta * g)
        # renormalise onto simplex
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    initial_temp: float = 0.10,
    final_temp: float = 1e-6,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing hill‑climb.
    Moves are the same as in ``hill_climb`` but also accept worsening moves
    with a temperature‑dependent probability.
    """
    current = a.copy()
    current_score = cur_score
    best = a.copy()
    best_score = cur_score
    start = time.time()

    while time.time() < deadline:
        # exponential temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        t = elapsed / total if total > 0 else 1.0
        temp = initial_temp * (final_temp / initial_temp) ** t

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = current[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        trial = current.copy()
        trial[i] -= delta
        trial[j] += delta
        trial_score = fast_score(trial, fft_len, n)

        # acceptance rule (minimisation)
        if trial_score < current_score or rng.random() < np.exp((current_score - trial_score) / (temp + 1e-12)):
            current = trial
            current_score = trial_score
            if trial_score < best_score:
                best = trial.copy()
                best_score = trial_score

    return best, best_score

    # Problem size – use a fairly large n but not too large.
    # --------------------------------------------------------------
    default_n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 9000)

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate on the simplex (sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            if tmp.size > n:
                best_a = tmp[:n]
                best_a = best_a / best_a.sum()
            else:
                best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Multi‑restart L‑BFGS (≈ 30 % of the time budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=45))
    phase1_deadline = start_time + 0.30 * budget_s
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "best",
    ]
    restart = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart {restart:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
            max_move=0.15,
        )
        best_y = np.log(best_a + 1e-12)
        restart += 1

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈ 45 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * budget_s
    print("[phase2] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        phase2_deadline,
        eta0=0.5,
        max_iters=2_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[phase2] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2b – Exponential‑gradient sub‑gradient (if time left)
    # --------------------------------------------------------------
    if time.time() < phase2_deadline - 5.0:
        print("[phase2b] exponential‑gradient descent")
        best_a, best_score = exp_subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            phase2_deadline,
            eta0=0.5,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[phase2b] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Fine L‑BFGS refinement (≈ 20 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[phase3] fine L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=30))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=800,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[phase3] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[phase4] simulated‑annealing hill‑climb")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
            initial_temp=0.10,
            final_temp=1e-6,
            max_move=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 5 – Aggressive final hill‑climb (any leftover time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # the harness supplies this function
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
