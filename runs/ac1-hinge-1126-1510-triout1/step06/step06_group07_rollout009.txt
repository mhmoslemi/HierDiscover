"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
------------------
* Work with a probability vector a (∑a = 1).  All updates keep it on the
  simplex via Euclidean projection.
* Global exploration:
    - Multiple random restarts of a continuation L‑BFGS on the
      log‑soft‑max parameters.  The objective is a smooth‑max
      approximation of the exact quantity 2·n·max(conv).
    - A richer set of initial “modes’’ (random, exponential,
      power‑law, edge‑heavy, sine‑shaped, triangular and a
      spectral‑sqrt seed) is used to increase diversity.
* Local refinement:
    - Targeted hill‑climb that reduces the currently maximal convolution
      entry.
    - Random hill‑climb for escaping tiny plateaus.
    - Adaptive projected sub‑gradient descent on the exact max with a
      back‑tracking line‑search.
* The whole search runs inside the allotted time budget (default 1000 s).
  The best sequence found is stored in the global name
  ``height_sequence_1`` and finally evaluated with the provided
  ``evaluate_sequence`` routine.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for ``evaluate_sequence`` when ∑a = 1.

    Returns 2·n·max(a*a) computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that explicitly lowers the current maximal
    convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a destination j with a tiny partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        # sample a few candidates and take the best among them
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Adaptive projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgrad_descent_adaptive(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    init_step: float = 0.5,
    min_step: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max of the auto‑convolution.
    Uses a back‑tracking line‑search to take as large a step as possible.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        # current convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_c = conv.max()
        active = np.where(conv >= max_c - 1e-12)[0]

        # gradient of max (convex hull of gradients of active shifts)
        # compute via FFT: gradient = 2 * (best *conv_active) where conv_active is
        # the convolution of best with the indicator of active shifts.
        ind = np.zeros(2 * n - 1, dtype=np.float64)
        ind[active] = 1.0
        fi = np.fft.fft(ind, fft_len)
        grad_full = np.real(np.fft.ifft(fa * fi))
        grad = 2.0 * grad_full[:n]   # shape (n,)

        # backtracking line search
        step = init_step / np.sqrt(it)
        if step < min_step:
            step = min_step
        improved = False
        while step >= min_step:
            cand = proj_simplex(best - step * grad)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score - 1e-12:
                best = cand
                best_score = cand_score
                improved = True
                break
            step *= 0.5

        if not improved:
            # small random jog to possibly escape a flat region
            if rng.random() < 0.001:
                delta = rng.normal(scale=0.001, size=n)
                cand = proj_simplex(best + delta)
                cand_score = fast_score(cand, fft_len, n)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score

    return best, best_score


# ----------------------------------------------------------------------
# Initialisation helpers (different “modes’’)
# ----------------------------------------------------------------------


def init_random(n: int) -> np.ndarray:
    return np.random.standard_normal(n)


def init_exponential(n: int) -> np.ndarray:
    rate = np.random.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def init_powerlaw(n: int) -> np.ndarray:
    p = np.random.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def init_edge_heavy(n: int) -> np.ndarray:
    m_edge = max(1, int(n * np.random.uniform(0.01, 0.04)))
    edge_val = np.sqrt(n / 2.0) / n
    centre_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, centre_val)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return a


def init_sine(n: int) -> np.ndarray:
    # half‑sine, always positive
    a = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def init_triangular(n: int) -> np.ndarray:
    # symmetric triangle (tent) shape
    mid = n // 2
    left = np.arange(1, mid + 1, dtype=float)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    a = np.concatenate([left, right])
    a /= a.sum()
    return a


def init_spectral_sqrt(n: int, fft_len: int) -> np.ndarray:
    """
    Rough spectral‑factor initialisation: take sqrt of the DFT of a uniform
    auto‑convolution and invert.  Afterwards clip negatives and renormalise.
    """
    u = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    U = np.fft.fft(u, fft_len)
    A = np.sqrt(U)                     # complex principal square root
    a_complex = np.fft.ifft(A).real
    a = a_complex[:n]
    a[a < 0.0] = 0.0
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # Reserve a safety margin for the final evaluation call
    deadline = time.time() + budget_s - 4.0

    # --------------------------------------------------------------
    # Choose problem size – reuse previous construction if convenient
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to the probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT sizes (kept constant)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Global search – continuation L‑BFGS with many diverse restarts
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-12, num=30)   # finer near zero
    maxiter_per_stage = 800                         # enough for progress

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Targeted hill‑climb (quick, aggressive)
        # ------------------------------------------------------------------
        t_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ------------------------------------------------------------------
        # Random hill‑climb (helps escape tiny plateaus)
        # ------------------------------------------------------------------

            a=best_a,

            n=n,
            deadline=r_dead,
            max_iter=8000,
            max_move_frac=0.20,

        iteration += 1

    # --------------------------------------------------------------
    # Adaptive projected sub‑gradient descent on the *exact* max
    # --------------------------------------------------------------
    subgrad_dead = deadline - 2.0
    best_a, best_score = subgrad_descent_adaptive(
        a=best_a,
        cur_score=best_score,
        n=n,
        fft_len=fft_len_conv,
        deadline=subgrad_dead,
        max_iter=5000,
        init_step=0.5,
        min_step=1e-6,
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final aggressive random hill‑climb with any remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # Call the provided evaluation routine (the harness will have imported it)
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
