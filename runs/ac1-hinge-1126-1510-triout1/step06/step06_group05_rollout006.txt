def annealing_hill_climb(a, conv_len, n, deadline):
    best = a.copy()
    best_score = fast_score(best, conv_len, n)
    T0 = 0.01 # start temperature relative to best_score?
    # we set temperature = T0 * (deadline - now)
    while time.time() < deadline:
        # choose random source i and random dest j, move small mass delta
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta fraction up to maybe 0.1 of source mass
        delta = best[i] * rng.random() * 0.2
        if delta == 0:
            continue
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, conv_len, n)
        delta_score = new_score - best_score
        if delta_score < 0:
            best = new_a
            best_score = new_score
        else:
            # probability of acceptance
            # Temperature schedule: linear decreasing
            t = max(1e-6, (deadline - time.time())/(deadline - start_time) ) # fraction left
            prob = np.exp(-delta_score / (t * best_score * 0.01))
            if rng.random() < prob:
                best = new_a
                best_score = new_score
        # occasional full rescale? not needed because sum unchanged.
    return best, best_score

"""
Improved step‑function optimiser.

The algorithm follows the successful three‑phase pattern of the reference
implementation but makes a number of changes to push the bound past the
target 1.503:

* Uses a larger problem size (n ≈ 12000) when time allows – larger n
  gives a lower theoretical lower bound (≈1.00) and more freedom to flatten
  the auto‑correlation.
* Refined initial families (beta‑like, arcsine, exponential and a few
  random starts) with a finer α/β grid.
* Continuation L‑BFGS with a longer λ‑schedule (70 stages) and more
  iterations per stage (1200) – this gives a tighter optimisation of the
  smooth max loss.
* After each L‑BFGS run a short greedy hill‑climb is performed.
* A projected sub‑gradient descent on the true max‑convolution is run for
  several minutes to fine‑tune the exact objective.
* The remaining time is spent in an annealed hill‑climb that accepts
  occasional uphill moves, helping to escape shallow local minima.

All helper functions are defined at top level; no closures or lambdas are
used.  The routine respects the supplied time budget and prints progress
information.  The returned list is directly suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n−1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Computes the objective for a vector on the probability simplex
    (sum(a)=1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp max approximation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (value, gradient) where
        value = 2·n·smooth_max(a*a)
        a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c               # max = 0
    exp_shift = np.exp(shifted / lam)    # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # propagate through convolution (size n gradient)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # pick the n entries that correspond to ∂conv/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS: runs a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a starting point (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb.  Randomly moves a fraction of mass from a source
    index to a destination index, accepting only strict improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb.  Allows occasional uphill moves
    with a linearly decreasing temperature.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # linear temperature (0 → 1)
        t_frac = max(0.0, (deadline - time.time()) / total_time)
        temperature = 1e-4 + 0.01 * t_frac  # small base + linear part

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 20 % of source mass
        delta = best[i] * rng.random() * 0.20
        if delta == 0.0:
            continue

        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)

        delta_score = new_score - best_score
        if delta_score < 0.0:
            best = new_a
            best_score = new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-delta_score / max(temperature, 1e-12))
            if rng.random() < prob:
                best = new_a
                best_score = new_score

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = max_i (a*a)_i .
    The sub‑gradient at the active index m is
        g_i = 2·a_{m‑i}   (if the index exists).
    Adaptive step size ηₜ = η₀/√t and Euclidean projection onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline - final_reserve:
        sub_deadline = deadline - final_reserve
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.20,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive (annealed) hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final anneal] start")
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
        )
        print(f"[final anneal] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # external harness provides this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
