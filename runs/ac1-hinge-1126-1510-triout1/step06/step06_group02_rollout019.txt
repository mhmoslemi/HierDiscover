def simple_proj_subgradient(a, k_max, eta):
    g = np.zeros_like(a)
    # Compute contributions:
    # For i where 0 <= k_max - i < n
    low = max(0, k_max - (n-1))
    high = min(n-1, k_max)
    # i runs from low..high
    idx = np.arange(low, high+1)
    g[idx] = 2 * a[k_max - idx]
    a_new = a - eta * g
    # clip
    a_new = np.maximum(a_new, 0.0)
    # renorm
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        # fallback to uniform
        a_new = np.full_like(a, 1.0 / len(a))
    return a_new
```

def gradient_descent_optimize(a_init, n, fft_len, deadline):
    a = a_init.copy()
    best = a.copy()
    best_score = fast_score(a)
    iter = 0
    eta = 0.5
    while time.time() < deadline:
        conv = conv_fft(a, fft_len, n)
        k_max = int(np.argmax(conv))
        # compute gradient slice
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2 * a[k_max - idx]
        # line search
        step = eta / np.sqrt(iter+1)
        # try a few step scaling factors
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            a_new = a - step * factor * g
            # clip negatives
            a_new = np.maximum(a_new, 0.0)
            total = a_new.sum()
            if total == 0:
                continue
            a_new /= total
            new_score = fast_score(a_new, fft_len, n)
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # maybe random perturb
            if iter % 1000 == 0:
                # add small gaussian noise
                noise = np.random.default_rng().normal(scale=0.01, size=n)
                a_new = a + noise
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                a = a_new
        iter += 1
    return best, best_score
```

"""
Propose a step‐function (non‑negative list of floats) with a low
evaluate_sequence score.

Approach
--------
1. Initialise a probability vector a of length n (≈ 5000‑8000).  If a
   previous construction is present (`height_sequence_1`) we start from it,
   otherwise we use a collection of heuristics (uniform, exponential,
   edge‑heavy, inverse‑sqrt, Dirichlet, …).

2. Optimise the *smooth* surrogate
        φ_λ(a)=log‑sum‑exp((a*a)/λ)
   using continuation L‑BFGS on the logits y=logit(a).  λ is driven from
   0.5 down to 1e‑10 in many stages, each stage performing several
   L‑BFGS iterations.

3. Refine the result with a large budget of *projected sub‑gradient*
   descent on the true objective 2·n·max(a*a).  The sub‑gradient of the
   active convolution entry is computed analytically (O(n) time) and an
   exponentiated‑gradient update with a decaying stepsize is applied.
   The iterate is always re‑normalised to stay on the simplex.

4. Finish with an aggressive hill‑climb that repeatedly moves a random
   fraction of mass from one coordinate to another, accepting only
   improving moves.

All heavy operations (FFT‑based convolution, gradient) are O(n log n)
or O(n); the whole search respects the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the max) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = log‑sum‑exp
    shifted = conv - max_c                 # max becomes 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weight for each convolution entry
    w = exp_shift / Z

    # gradient of smooth max w.r.t. a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initialisers for the probability vector (logits)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a start point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "inverse_sqrt":
        # u‑shaped distribution: 1/√i + 1/√(n‑i)
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        # sparse Dirichlet (α = 0.1)
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on 2·n·max(a*a).
    Uses the sub‑gradient given by the *active* convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry:
        # g_i = 2 * a_{k_max-i}  (if index inside the array)
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        # decaying step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            # slowly increase step size after a success
            eta0 = min(eta0 * 1.02, 2.0)
        else:
            # keep the old point, but still decay step size
            a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (random mass moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 10_000,
    max_move: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score
