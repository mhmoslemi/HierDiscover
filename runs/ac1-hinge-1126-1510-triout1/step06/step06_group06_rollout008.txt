2. Repeatedly:
   * Compute the current maximal auto‑convolution value and its
     sub‑gradient (the gradient of the entry attaining the maximum).
   * Update the vector by a *multiplicative* exponential‑gradient step
     ``a_i ← a_i·exp(-η·g_i)`` where ``g_i`` is the sub‑gradient.
   * Renormalise to sum 1.  The learning rate η decays geometrically.
   * Keep the best vector seen so far.
3. After the time budget is almost exhausted, perform a short
   random‑hill‑climb to polish the solution.
4. Return the best sequence found and store it globally for the next
   run.

The multiplicative update respects the non‑negativity constraint and
often moves mass away from the pair of indices that are responsible for
the current peak, allowing the optimiser to escape the plateaus that
stymied the previous L‑BFGS + sub‑gradient approach.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a, a)) – the objective when ``sum(a)=1``."""
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_convolution_subgrad(a: np.ndarray, n: int, fft_len: int):
    """
    Return (max_conv, subgradient) of the maximal entry of the
    linear auto‑convolution of ``a``.
    ``subgradient`` has shape (n,) and satisfies
    ``max_conv + <subgradient, d>`` ≤ ``max_conv(a+d)`` for small ``d``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map shift to the first half (0 ≤ k ≤ n‑1)
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        # derivative of Σ a_i²
        grad = 2.0 * a
    else:
        # forward contribution  a_{i+k}
        grad[: n - k] += a[k:]
        # backward contribution a_{i‑k}
        grad[k:] += a[: n - k]

    return float(max_c), grad.astype(np.float64)


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Simple hill‑climb (used for final polishing)
# -------------------------------------------------------------

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex (strictly improving only)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# -------------------------------------------------------------
# Core optimisation routine – multiplicative exponential‑gradient
# -------------------------------------------------------------

def expo_gradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_eta: float = 2000.0,
    decay: float = 0.9995,
    min_eta: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Run multiplicative exponential‑gradient descent on the simplex.
    Returns the best vector and its associated objective value.
    """
    a = a0.copy()
    eta = init_eta
    best_a = a.copy()
    best_score = fast_score(best_a, n)

    iter_cnt = 0
    while time.time() < deadline:
        max_c, grad = max_convolution_subgrad(a, n, fft_len)
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[EG] iter={iter_cnt} score={best_score:.6f}")

        # exponential update (multiplicative)
        factor = np.exp(-eta * grad)
        a = a * factor
        # keep positivity and re‑normalise
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # decay learning rate
        eta = max(min_eta, eta * decay)
        iter_cnt += 1

    return best_a, best_score


# -------------------------------------------------------------
# Main entry point required by the evaluation harness
# -------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops a couple of seconds
        before this limit to stay safe.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size and initial vector
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to simplex (objective is scale‑invariant)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    # ------------------------------------------------------------------
    # Prepare FFT length used throughout the search
    # ------------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)

    # Current best score (fast surrogate)
    best_score = fast_score(best_a, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Multi‑restart exponential‑gradient descent
    # ------------------------------------------------------------------
    start_vectors = [
        best_a,                                      # previous best (if any)
        np.full(n, 1.0 / n),                         # uniform
        rng.dirichlet(np.ones(n) * 0.5),             # moderately sparse
        rng.dirichlet(np.ones(n) * 0.1),             # very sparse
    ]

    # some extra random starts if time permits
    extra_starts = 2
    for _ in range(extra_starts):
        start_vectors.append(rng.dirichlet(np.ones(n) * rng.uniform(0.05, 0.5)))

    for idx, a_start in enumerate(start_vectors):
        if time.time() >= deadline - 5.0:
            break
        print(f"[EG] start #{idx} – launching exponential‑gradient descent")
        a_opt, score_opt = expo_gradient_descent(
            a0=a_start,
            n=n,
            fft_len=fft_len,
            deadline=deadline - 4.0,   # keep a few seconds for polishing
            init_eta=2000.0,
            decay=0.9995,
            min_eta=1e-3,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[EG] new global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Final polishing with a cheap random hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[polish] final hill‑climb")
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            max_iter=30000,
            max_move_frac=0.25,
        )

    # 3. Official evaluation and return
    # ------------------------------------------------------------------

    return final_seq
