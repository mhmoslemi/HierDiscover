            # also need i+i not equal existing sum but sums already include duplicates? include i+i in sums.

            # Actually we can maintain set of sums already existing; need ensure none of new_sums already in sums.
            if any(val in sums for val in new_sums):
                conflict = True
        if not conflict:
            # add i to set
            for val in new_sums:
                sums.add(val)
            S.append(i)
            if m_target and len(S) >= m_target:
                break
        i += 1
    return S

def subgrad_descent_line_search(a, cur_score, n, fft_len, deadline, max_iter=10000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    lower_bound = 2 * n / (2 * n - 1) # approx minimal possible fast_score
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # compute conv
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k = int(np.argmax(conv))
        # gradient of conv_k
        grad = np.zeros_like(best)
        # For each i: j = k - i; if 0 <= j < n => grad_i = 2 * best[j]
        # This is vectorized by computing shift.
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * best[j[mask]]
        # direction = -grad
        # compute step size via Polyak using lower bound
        norm2 = np.dot(grad, grad)
        if norm2 == 0:
            break
        step = (best_score - lower_bound) / norm2
        step = max(step * 0.9, 1e-6)  # safety
        # line search: try decreasing step if not improvement
        for _ in range(5):
            cand = proj_simplex(best - step * grad)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best, best_score = cand, cand_score
                break
            else:
                step *= 0.5
        # occasional random perturb if no improvement
        if step < 1e-8:
            # random perturb
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = best[i] * rng.random() * 0.1
                best[i] -= delta
                best[j] += delta
                best = proj_simplex(best)
                best_score = fast_score(best, fft_len, n)
    return best, best_score

    # Build constraint matrix: For each k (0..2n-2), compute coefficients of b_j.
    # We'll build matrix rows as list of (row_index, col_index, value)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # contributions: for i from max(0, k-n+1) .. min(k, n-1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)

"""
Optimise step-function heights to minimise the evaluation function.
The algorithm combines:
* a global quasi‑Newton search on a smoothed max using L‑BFGS,
* a Polyak step‑size projected sub‑gradient descent on the exact max,
* short random hill‑climbs for escaping plateaus,
* warm‑starts from the previously best sequence (height_sequence_1).

The sub‑gradient step uses the theoretical lower bound 2 n/(2 n‑1) as
target and a line‑search to guarantee a strict decrease.
All operations run on the probability simplex (∑a=1) via the soft‑max
parameterisation and Euclidean projection.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Surrogate of evaluate_sequence for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def smooth_obj_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                        fft_len_g: int, lam: float):
    """Smoothed max of the auto‑convolution and its gradient w.r.t. y."""
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    # smooth max
    shifted = conv - max_c   # max at zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z                 # d(smooth_max)/d conv
    # back‑propagation to a
    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]
    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)
    return float(f), grad_y.astype(np.float64)


def schedule_optimize(y_start: np.ndarray, n: int,
                      fft_len_conv: int, fft_len_g: int,
                      lam_factors: list[float],
                      maxiter_per_stage: int = 800) -> np.ndarray:
    """Continuation L‑BFGS on a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=smooth_obj_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y


def polyak_subgrad(a: np.ndarray, cur_score: float,
                   n: int, fft_len: int,
                   lower_bound: float,
                   deadline: float,
                   max_iter: int = 5000,
                   factor: float = 0.9) -> tuple[np.ndarray,float]:
    """
    Projected sub‑gradient descent with Polyak step‑size and back‑tracking.
    """
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        # full convolution via FFT
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))
        # gradient of fast_score w.r.t. a (including factor 2*n)
        grad = np.zeros_like(best)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 4.0 * n * best[j[mask]]   # 4 n a_{k-i}
        gnorm2 = np.dot(grad, grad)
        if gnorm2 == 0.0:
            break
        # Polyak step based on the theoretical lower bound
        step = factor * (best_sc - lower_bound) / gnorm2
        if step <= 0.0:
            break
        # simple back‑tracking line search – at most 5 reductions
        for _ in range(5):
            cand = proj_simplex(best - step * grad)
            cand_sc = fast_score(cand, fft_len, n)
            if cand_sc < best_sc - 1e-12:
                best, best_sc = cand, cand_sc
                break
            step *= 0.5
        # occasional random perturbation to avoid stagnation
        if step * np.linalg.norm(grad) < 1e-10:
            # random transfer of a tiny amount of mass
            i, j = np.random.choice(n, size=2, replace=False)
            delta = best[i] * 0.02
            best[i] -= delta
            best[j] += delta
            best = proj_simplex(best)
            best_sc = fast_score(best, fft_len, n)
    return best, best_sc


def hill_climb(a: np.ndarray, cur_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 5000,
               max_move_frac: float = 0.20) -> tuple[np.ndarray,float]:
    """
    Simple random mass‑transfer hill‑climb. Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)

        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Optimise a non‑negative step‑function (probability vector) to minimise

    2·n·max(a*a) / (sum(a)²)

which is exactly the value returned by ``evaluate_sequence`` when the vector
is normalised to sum = 1.  The method mixes three complementary phases:

1. **Continuation L‑BFGS** on a smooth approximation of the “max”
   (the smooth‑max uses a geometric λ‑schedule that approaches the true
   max).  Each run starts from a diverse seed (random, exponential,
   power‑law, edge‑heavy, sinc‑shaped or a perturbation of the current best).

2. **Polyak‑step projected sub‑gradient descent** on the exact objective.
   Using the theoretical lower bound 2 n/(2 n‑1) the step size
   ``(f‑lower)/‖∇f‖²`` guarantees a strict decrease; a short back‑track
   ensures acceptance.

3. **Random hill‑climb** (mass transfer on the simplex) to escape possible
   plateaus and to perform a final fine‑tuning.

The best vector seen is stored in the global name ``height_sequence_1`` so
that later invocations can start from it.  All operations respect the
non‑negativity and simplex constraints, and the routine stops before the
time budget expires, returning the sequence that achieved the lowest score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (must stay at top level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(y: np.ndarray, n: int,
                        fft_len_conv: int, fft_len_g: int,
                        lam: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- a*a (full convolution) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smooth approximation of max
    shifted = conv - max_c                     # peak at zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                         # d(smooth_max)/d conv_i

    # back‑propagation to a (g_i = Σ_j a_j w_{i+j})
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(y_start: np.ndarray, n: int,
                      fft_len_conv: int, fft_len_g: int,
                      lam_factors: list[float],
                      maxiter_per_stage: int = 800) -> np.ndarray:
    """Continuation L‑BFGS over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=smooth_obj_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000,
               max_move_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb – only strictly improving moves kept."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Polyak‑step projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def polyak_subgrad(a: np.ndarray,
                   cur_score: float,
                   n: int,
                   fft_len: int,
                   lower_bound: float,
                   deadline: float,
                   max_iter: int = 8000,
                   factor: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Polyak step size and a short
    back‑tracking line search.  ``lower_bound`` is the theoretical minimum
    (2·n/(2·n‑1)).
    """
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        # ----- current convolution and the worst shift -----
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # ----- sub‑gradient of the fast_score (2·n·max) -----
        grad = np.zeros_like(best)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        # d(2·n·max)/d a_i = 4·n·a_{k‑i}
        grad[mask] = 4.0 * n * best[j[mask]]

        gnorm2 = np.dot(grad, grad)
        if gnorm2 == 0.0:
            break

        # Polyak step (scaled by ``factor`` for safety)
        step = factor * (best_sc - lower_bound) / gnorm2
        if step <= 0.0:
            break

        # ----- back‑tracking line search (max 5 reductions) -----
        for _ in range(5):
            cand = proj_simplex(best - step * grad)
            cand_sc = fast_score(cand, fft_len, n)
            if cand_sc < best_sc - 1e-12:
                best, best_sc = cand, cand_sc
                break
            step *= 0.5

        # occasional tiny random perturbation to keep the search alive
        if step * np.sqrt(gnorm2) < 1e-10:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = best[i] * 0.02
            best[i] -= delta
            best[j] += delta
            best = proj_simplex(best)
            best_sc = fast_score(best, fft_len, n)

    return best, best_sc

        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "sinc":
            # |sinc|‑shaped mass, centred
            alpha = rng.uniform(2.0, 8.0)
            x = np.arange(n) - n / 2.0
            a0 = np.abs(np.sinc(alpha * x / n)) + 1e-9
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturbation of the current best
            y0 = best_y + rng.normal(scale=0.08, size=n)

        # ------- continuation L‑BFGS ---------------------------------------
        y_opt = schedule_optimize(y0, n,
                                  fft_len_conv, fft_len_g,
                                  lam_factors, maxiter_per_stage)
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score, best_a, best_y = sc_opt, a_opt, y_opt
            print(f"[{iter_cnt}] mode={mode} → new best fast score = {best_score:.6f}")

        # brief random hill‑climb to shake the search space
        t_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        t_dead, max_iter=1200,
                                        max_move_frac=0.25)
        best_y = np.log(best_a + 1e-12)

        iter_cnt += 1

    # --------------------------------------------------------------
    # Aggressive Polyak‑step sub‑gradient descent (local refinement)
    # --------------------------------------------------------------
    sub_dead = deadline - 5.0
    best_a, best_score = polyak_subgrad(best_a, best_score,
                                        n, fft_len_conv,
                                        lower_fast,
                                        sub_dead,
                                        max_iter=15000,
                                        factor=0.9)
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final random hill‑climb with the remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        deadline,
                                        max_iter=25000,
                                        max_move_frac=0.25)
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # ``evaluate_sequence`` is provided by the harness
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
