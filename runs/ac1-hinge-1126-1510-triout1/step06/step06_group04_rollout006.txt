"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm
---------
1. Choose a problem size `n`.  If a previous construction (`height_sequence_1`)
   is available we keep its length (up to a maximum of 12000) otherwise we use
   `n = 10000`.  The objective is scale‑invariant, therefore all vectors are
   normalised to sum to 1 (probability simplex).

2. Multi‑restart smoothed optimisation.
   * Several random initialisations (`random`, `exp`, `power`, `edge`,
     `sidon`, `binary`, `bump`, `harmonic`) are generated.
   * Each start is optimised with a continuation L‑BFGS on the
     log‑sum‑exp smoothed max‑convolution.  The smoothing parameter is
     decreased geometrically (35 → 45 stages) and each stage receives
     a moderate number of L‑BFGS iterations.
   * Whenever a run improves the fast score (`2·n·max(conv)`) the best
     solution is stored.

3. A long simulated‑annealing hill‑climb.
   * Starting from the best solution found, random mass transfers
     `a_i → a_j` are performed.
   * Moves are accepted if they improve the fast score or with a
     Metropolis probability `exp(-(Δ)/T)` where the temperature `T`
     decays linearly to zero.
   * The best iterate over the whole annealing run is kept.

4. Final fine‑grained L‑BFGS refinement (very low smoothing) and a
   short greedy hill‑climb to polish the result.

5. The sequence is returned as a plain list; the harness evaluates it
   with `evaluate_sequence`.

The whole procedure respects the time budget (`budget_s` seconds) and
periodically checks the deadline.

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and gradient.
    The smoothing parameter is `lam = max(conv) * lam_factor`.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                    # ∂smooth_max / ∂conv_i

    # gradient of conv w.r.t a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray, n: int,
                      fft_len_conv: int, fft_len_g: int,
                      lam_schedule, maxiter_per_stage: int = 500):
    """Continuation L‑BFGS with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:

        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break

                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb_greedy(a: np.ndarray, cur_score: float,
                      fft_len: int, n: int,
                      deadline: float,
                      max_iter: int = 5000,
                      max_move: float = 0.20):
    """Simple greedy hill‑climb (only accept improving moves)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(a_start: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        init_temp: float = 0.015,
                        final_temp: float = 0.0001,
                        max_iter: int = 60000,
                        max_move: float = 0.25):
    """
    Simulated‑annealing style hill‑climb.
    Moves mass from a random i to a random j.
    An uphill move is accepted with probability exp(-Δ/T).
    """
    best = a_start.copy()
    best_score = cur_score
    cur = a_start.copy()
    cur_score_local = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # temperature linearly decreasing
        T = init_temp + (final_temp - init_temp) * (it / max_iter)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        cand_score = fast_score(cur, fft_len, n)

        accept = False
        if cand_score < cur_score_local:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(cand_score - cur_score_local) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score_local = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    default_n = 10000                     # a reasonably large size
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 12000)

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)       # parameters for soft‑max
    print(f"[init]  n={n}  start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Multi‑restart L‑BFGS (≈ 55 % of the time budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(1.0, 5e-8, num=45))
    iteration = 0
    modes = [
        "random", "exp", "power", "edge",
        "sidon", "binary", "bump", "harmonic", "best"
    ]
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(y0,
                                  n,
                                  fft_len_conv,
                                  fft_len_g,
                                  lam_schedule,
                                  maxiter_per_stage=600)

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode:<8} improved fast score = {best_score:.6f}")

        # brief greedy polishing
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb_greedy(best_a,
                                                best_score,
                                                fft_len_conv,
                                                n,
                                                hc_deadline,
                                                max_iter=2500)
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing hill‑climb (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[anneal] starting simulated‑annealing hill‑climb")
    best_a, best_score = simulated_annealing(best_a,
                                             best_score,
                                             fft_len_conv,
                                             n,
                                             anneal_deadline,
                                             init_temp=0.015,
                                             final_temp=0.0002,
                                             max_iter=40000,
                                             max_move=0.30)
    best_y = np.log(best_a + 1e-12)
    print(f"[anneal] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final fine L‑BFGS refinement (very low smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] starting final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=30))
        y_fine = schedule_optimize(best_y,
                                   n,
                                   fft_len_conv,
                                   fft_len_g,
                                   fine_lam_schedule,
                                   maxiter_per_stage=800)
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Aggressive greedy hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive greedy hill‑climb")
        best_a, best_score = hill_climb_greedy(best_a,
                                               best_score,
                                               fft_len_conv,
                                               n,
                                               deadline,
                                               max_iter=30000,
                                               max_move=0.35)
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
