"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm Overview
------------------
1. Work on the probability simplex (∑a = 1); the evaluation is scale‑invariant.
2. Use a moderately large length (default n≈8000, up to 9000 if a seed
   sequence is supplied).
3. Multi‑restart continuation L‑BFGS on a smooth log‑sum‑exp approximation
   of the max‑convolution.  A geometric schedule of smoothing factors is
   employed; each stage runs a short L‑BFGS.
4. Projected sub‑gradient descent on the true (non‑smooth) objective
   2·n·max(conv) helps drive the solution toward the lower bound.
5. A *targeted* hill‑climb exploits the structure of the active
   convolution index: mass is moved away from the pair (i,m‑i) that
   contributes most to the current maximal entry and towards a low‑mass
   location.
6. A short random hill‑climb finishes the remaining time.
7. The best sequence found is returned; the harness evaluates it with
   ``evaluate_sequence``.

The combination of smooth optimisation, sub‑gradient descent and a
structure‑aware local search usually pushes the score well below the
previous 1.5079 threshold within the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random, exp, power, edge, sidon, binary, bump, harmonic.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # simple greedy construction of a Sidon‑like set
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (equivalently minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))                       # index of maximal entry
        # Sub‑gradient of that entry (ordered pairs)
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / math.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that explicitly targets the active convolution index.
    At each step we move a small fraction of mass away from the pair
    (i, m‑i) that contributes most to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contributions of each index to conv[m]
        i_vals = np.arange(n)
        counterpart = m - i_vals
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[i_vals[mask]] * best[counterpart[mask]]

        # source index (largest contribution)
        src = int(np.argmax(contrib))
        if contrib[src] == 0:
            # nothing to improve – break
            break

        # destination index – choose a low‑mass position (randomised among the
        # bottom 20 % of the distribution)
        low_thresh = np.percentile(best, 20)
        low_candidates = np.where(best <= low_thresh)[0]
        if low_candidates.size == 0:
            dst = rng.integers(0, n)
        else:
            dst = int(rng.choice(low_candidates))
        if src == dst:
            continue

        # move a fraction of mass from src to dst
        delta = best[src] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[dst] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[src] += delta
            best[dst] -= delta

    return best, best_score

    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 9000)

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Multi‑restart continuation L‑BFGS (≈ 45 % of the budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=40))
    iteration = 0
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "best",
    ]
    phase1_deadline = start_time + 0.45 * budget_s

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # brief random hill‑climb polishing
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈ 30 % of the budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.80 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=1_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Targeted hill‑climb (≈ 15 % of the budget)
    # --------------------------------------------------------------
    target_deadline = min(deadline - 5.0, start_time + 0.95 * budget_s)
    print("[target] starting targeted hill‑climb")
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        target_deadline,
        max_iter=8000,
        max_move=0.25,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[target] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Fine L‑BFGS refinement (very low smoothing) if time permits
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] starting final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=30))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=600,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Aggressive random hill‑climb with remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_hc_deadline = deadline
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
