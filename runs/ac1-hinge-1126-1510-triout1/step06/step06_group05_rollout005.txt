"""
propose_candidate
=================

Search for a non‑negative step function (list of floats) that yields a
low ``evaluate_sequence`` score.

The algorithm is structured in five phases, each respecting the overall
time budget (default 1000 s):

1. **Parametric grid** – quick scan of several Beta‑like families.
2. **Sidon‑set exploration** – generate many sparse candidates on
   (approximately) Sidon sets; keep the best.
3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the true max‑convolution.  The smoothing factor is
   decreased geometrically; each run starts from a random heuristic or
   a perturbed version of the current best.
4. **Adam‑style sub‑gradient descent** – project the true sub‑gradient
   (active‑index) onto the probability simplex, using Adam’s adaptive
   step‑size to refine the solution.
5. **Aggressive hill‑climb** – greedy mass transfers between coordinates
   until time runs out.

All helper functions are defined at module level; no closures or lambdas
are used.  The routine works with any initial global variable
``height_sequence_1`` (if present) and scales the final sequence only
by the harness, so the sum of the coefficients is irrelevant.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    # softmax with numerical stability
    m = np.max(y)
    exp_y = np.exp(y - m)
    a = exp_y / exp_y.sum()

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                           # ∂smooth_max/∂conv

    # ---- propagate gradient through convolution ---------------------------
    # Compute g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy random Sidon set of target size ~1.6*sqrt(n)
        target = int(1.6 * np.sqrt(n))
        chosen = []
        sums = set()
        attempts = 0
        while len(chosen) < target and attempts < 5 * target:
            cand = rng.integers(0, n)
            if cand in chosen:
                attempts += 1
                continue
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            attempts += 1
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / len(chosen)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j, accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a 1/√t step‑size schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def adam_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.10,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the true max‑convolution
    objective.  The sub‑gradient is taken at the active index (the one
    achieving the maximum).
    """
    a = a_start.copy()
    m_est = np.zeros_like(a)
    v_est = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        idx_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        idx = np.arange(n)
        counterpart = idx_max - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam moments
        m_est = beta1 * m_est + (1.0 - beta1) * g
        v_est = beta2 * v_est + (1.0 - beta2) * (g * g)

        m_hat = m_est / (1.0 - beta1 ** t)
        v_hat = v_est / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_onto_simplex(a)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose problem size (keep it moderate but not too small)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 12000)  # allow a bit larger if available
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric grid search
    # ------------------------------------------------------------------
    alphas = np.linspace(-0.8, 0.5, 7)
    betas = np.linspace(-0.8, 0.5, 7)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Sidon‑set and random sparse exploration
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.25 * budget_s
    while time.time() < phase2_deadline:
        mode = rng.choice(["sidon", "random"])
        y0 = generate_initial_y(mode, n, rng)
        a0 = np.exp(y0)  # this yields a positive vector that sums to >0
        a0 = a0 / a0.sum()
        sc = fast_score(a0, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
            best_y = y0.copy()
            print(f"[sidon] mode={mode} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Continuation L‑BFGS (many restarts, coarse schedule)
    # ------------------------------------------------------------------
    lam_schedule_coarse = list(np.geomspace(0.5, 1e-6, num=20))
    lam_schedule_fine = list(np.geomspace(1e-6, 1e-12, num=30))
    lam_schedule = lam_schedule_coarse + lam_schedule_fine

    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]
    iteration = 0
    phase3_deadline = start_time + 0.70 * budget_s
    while time.time() < phase3_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,  # relatively cheap per stage
        )
        a_opt = np.exp(y_opt)
        a_opt = a_opt / a_opt.sum()
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 4 – Adam‑style projected sub‑gradient descent (refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        adam_deadline = deadline - 40.0
        print("[adam] start")
        best_a, best_score = adam_descent(
            best_a,
            fft_len_conv,
            n,
            adam_deadline,
            lr=0.12,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # The harness supplies evaluate_sequence; call it only for reporting.
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[final] could not evaluate: {e}")

    return final_seq
