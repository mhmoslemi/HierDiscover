# Helpers – pure, no closures
# ------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Wang‑Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` (2·n·max(a*a))."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max (log‑sum‑exp)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth‑max w.r.t. conv
    w = exp_shift / Z

    # propagate gradient back to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ∂f/∂a
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max: diag(a) – a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focused hill‑climb that reduces the current maximal convolution entry."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient
    of the true (non‑smoothed) max‑operator.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to appropriate shift 0 ≤ k ≤ n‑1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    A number of heuristics are mixed to keep diversity across restarts.
    """
    mode = rng.choice(
        ["random", "exp", "power", "inc_power", "edge", "left_exp", "sparse", "best"]
    )
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "inc_power":
        p = rng.uniform(0.5, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** p)
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "left_exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "sparse":
        # a few random spikes (2‑5 % density)
        density = rng.uniform(0.02, 0.05)
        k = max(1, int(density * n))
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a0 = np.zeros(n)
        a0[idx] = vals
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb current best
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len: int,
    lam_factor: float,
    max_steps: int,
    deadline: float,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    Works directly in the log‑soft‑max space, so the simplex is
    automatically respected.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    step_size = 0.12  # learning‑rate for the log‑space
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len, n)

    t = 0
    while t < max_steps and time.time() < deadline:
        t += 1
        loss, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)

        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        y -= (step_size * m_hat) / (np.sqrt(v_hat) + eps)

        if t % 250 == 0:
            cur = fast_score(softmax(y), fft_len, n)
            if cur < best_score:
                best_score = cur
                best_y = y.copy()
    return best_y


def shift_and_score(a: np.ndarray, fft_len: int, n: int, shift: int) -> float:
    """Return the fast_score of a zero‑padded shift of ``a`` by ``shift`` positions."""
    if shift == 0:
        return fast_score(a, fft_len, n)
    if shift > 0:
        b = np.concatenate((np.zeros(shift), a[:-shift]))
    else:
        s = -shift
        b = np.concatenate((a[s:], np.zeros(s)))
    return fast_score(b, fft_len, n)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget (seconds). The function stops before this limit.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------
    # Problem size and initial point
    # --------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=35)   # slightly finer λ‑grid
    maxiter_per_stage = 1200                         # more work per stage
    iteration = 0

    # --------------------------------------------------------
    # Main multi‑restart loop
    # --------------------------------------------------------
    while time.time() < deadline - 8.0:
        # ---- initialise -------------------------------------------------
        y0 = init_y(rng, n, best_a, iteration)

        # ---- L‑BFGS global search --------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=next_pow2(3 * n - 2),
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # ---- Adam fine‑tuning ------------------------------------------
        adam_deadline = min(deadline, time.time() + 1.2)
        y_fine = adam_optimize(
            y_start=y_opt,
            n=n,
            fft_len=fft_len_conv,
            lam_factor=0.025,          # relatively low smoothing
            max_steps=6000,
            deadline=adam_deadline,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            best_score = score_fine
            print(f"[{iteration}] Adam fine‑tune improved fast score = {best_score:.6f}")

        # ---- Targeted hill‑climb ---------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- Random hill‑climb -----------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------
    # Projected sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0
    eps = 1e-12

    while time.time() < deadline - 2.0:
        subgrad_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        step = step0 / np.sqrt(subgrad_iter)

        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92

        if subgrad_iter % 200 == 0:
            print(f"[subgrad] iter={subgrad_iter} best fast score={best_score:.6f}")

    # --------------------------------------------------------
    # Zero‑padded rotation exploration (cheap cheap moves)
    # --------------------------------------------------------
    rot_deadline = deadline
    while time.time() < rot_deadline:
        # try a handful of random shifts
        for _ in range(5):
            shift = rng.integers(-int(0.02 * n), int(0.02 * n) + 1)  # small shifts
            cur = shift_and_score(best_a, fft_len_conv, n, shift)
            if cur < best_score:
                best_score = cur
                if shift > 0:
                    best_a = np.concatenate((np.zeros(shift), best_a[:-shift]))
                elif shift < 0:
                    s = -shift
                    best_a = np.concatenate((best_a[s:], np.zeros(s)))
                best_y = np.log(best_a + 1e-12)
                print(f"[rotate] shift={shift} improved fast score = {best_score:.6f}")

    # --------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # --------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
