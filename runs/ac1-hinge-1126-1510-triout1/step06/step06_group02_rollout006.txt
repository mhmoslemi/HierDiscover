def adam_optimize(y_start, n, fft_len_conv, fft_len_g, lam, max_iters, deadline, lr=0.1):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    while time.time() < deadline and t_iter < max_iters:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        # maybe rescale to avoid large values? Could optionally keep from saturating.
    return y
```

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation function

    2·n·max(a*a) / (sum(a)^2)

which, after normalising the sum to 1, is simply

    2·n·max(a*a).

The algorithm consists of three phases:

1. **Exploratory L‑BFGS continuation** – many random starts are
   optimised on a smooth log‑sum‑exp surrogate of the max‑convolution.
2. **Active‑set linear‑program refinement** – from the best candidate
   we solve a small LP that maximises the total mass of a new distribution
   while keeping the current maximal convolution value as a bound.
   Mixing the new distribution lowers the true max.
3. **Simulated‑annealing hill climbing** – a final greedy/annealed walk
   moves tiny amounts of mass between coordinates, accepting occasional
   worsening moves to escape shallow plateaus.

All operations use FFT‑based convolutions for speed.  The routine is
designed to run for up to 1000 s; the time budget is split roughly as
60 % L‑BFGS, 20 % LP refinement and 20 % annealed hill climbing.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # max = 0 now
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                       # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the *average* of the sub‑gradients of all (almost) active entries,
    and a back‑tracking line search to guarantee monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                # because d (a_i a_j) / d a_i = 2 a_j

        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def lp_refine(
    f: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 12,
    mix: float = 0.30,
    rel_tol: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Active‑set linear‑program refinement.
    For the current distribution `f` we find a new distribution `g`
    maximising its total mass while respecting the bound
        (f * g)_k ≤ M   for all (active) convolution indices k,
    where `M` is the current maximal convolution value.
    Mixing `g` into `f` (with weight `mix`) can only lower the true max.
    """
    best = f.copy()
    best_score = fast_score(best, fft_len, n)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        conv = conv_fft(best, fft_len, n)
        M = conv.max()
        # active set – indices where the convolution is essentially maximal
        tol = max(1e-12, M * rel_tol)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        # Build the (small) constraint matrix A·g ≤ M
        rows = []
        for k in active:
            row = np.zeros(n)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                i_vals = k - idx
                row[idx] = best[i_vals]
            rows.append(row)
        A = np.vstack(rows)
        b = np.full(A.shape[0], M)

        # Linear programme: maximise sum(g)  →  minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            break
        g = res.x
        if g.sum() <= 1e-12:
            break
        g_norm = g / g.sum()

        # Mix and project back onto the simplex
        new_f = (1 - mix) * best + mix * g_norm
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()
        new_score = fast_score(new_f, fft_len, n)

        if new_score < best_score - 1e-12:
            best = new_f
            best_score = new_score
        else:
            # No improvement – stop early
            break

    return best, best_score


def annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_move: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing.
    Randomly moves a fraction of mass from i to j, accepting moves that
    improve the fast objective and occasionally accepting worsening moves
    with a temperature that decays exponentially.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    temperature = cur_score * 0.02    # initial temperature (relative)
    min_temp = 1e-12
    decay = 0.9995

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)

        if new_score < best_score:
            best = new_a
            best_score = new_score
            temperature = max(min_temp, temperature * 0.995)
        else:
            # probabilistic acceptance
            if temperature > 0 and rng.random() < np.exp(-(new_score - best_score) / temperature):
                best = new_a
                best_score = new_score
        temperature = max(min_temp, temperature * decay)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp
        else:
            # Pad/truncate to the required length
            if tmp.size < n:
                best_a = np.concatenate([tmp, np.zeros(n - tmp.size)])
            else:
                best_a = tmp[:n]
    else:
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS from many starts.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-10, num=70))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",
        "best",
    ]
    iteration = 0

    # Reserve ~250 s for LP + final hill climb.
    while time.time() < deadline - 250:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        a_opt /= a_opt.sum()
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – active‑set LP refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        lp_deadline = deadline - 70
        print("[lp_refine] start")
        best_a, best_score = lp_refine(
            best_a, n, fft_len_conv, lp_deadline, max_iters=12, mix=0.30
        )
        print(f"[lp_refine] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing hill climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final annealing HC] start")
        best_a, best_score = annealing_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_move=0.08
        )
        print(f"[final annealing HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
