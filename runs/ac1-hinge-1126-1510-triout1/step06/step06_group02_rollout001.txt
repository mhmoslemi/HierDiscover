def simulated_annealing(a_initial, cur_score, fft_len, n, deadline, max_iter=None):
    best = a_initial.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # Set initial temperature to maybe 0.05*cur_score; gradually decay.
    T_start = cur_score * 0.02 # some fraction (maybe 0.02 of score)
    T_end = cur_score * 1e-6
    total_time = deadline - time.time()
    # compute total iterations allowed approx due to time: each iteration conv O(n log n). So limit.
    # We'll while time < deadline:
    while time.time() < deadline:
        # proportion of elapsed
        t_elapsed = time.time() - start
        # compute temperature
        t_frac = t_elapsed / total_time
        T = T_start * (1 - t_frac) + T_end * t_frac  # linear decay
        # propose move: random i1, i2, amount delta (maybe up to fraction of mass at i1)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # Choose delta fraction of source
        max_delta = best[i] * 0.2 # up to 20% of mass
        delta = max_delta * rng.random()
        if delta == 0: continue
        # apply move
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)
        delta_score = new_score - best_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            best = new_a
            best_score = new_score
        # else reject
    return best, best_score

"""
Propose a non‑negative step function with a low evaluate_sequence score.

Algorithm
---------
1. Choose a problem size `n` (up to 10 000).  If a previous construction exists,
   reuse its length, otherwise set `n = 8000`.
2. Initialise a pool of starting points on the probability simplex using many
   heuristics (random Gaussian logits, exponential, power‑law, edge‑heavy,
   arcsine, Sidon‑set, Dirichlet, Beta‑shape, cosine‑shape, symmetric power,
   plus perturbations of the best candidate found so far).
3. For each start run a *continuation* L‑BFGS optimisation on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The smoothing factor λ
   (controlled by `lam_factor`) is decreased geometrically over 70 stages.
   The variables are the unconstrained logits `y`; the soft‑max maps them onto
   the simplex.
4. After each L‑BFGS run perform a short projected hill‑climb that moves a
   random fraction of mass between two coordinates and accepts only improving
   moves.
5. Refine the best vector with a *multi‑active* projected sub‑gradient method
   that averages the sub‑gradients of all (almost) active convolution entries
   and uses a back‑tracking line‑search.
6. Apply a *simulated‑annealing* stage that accepts occasional worsening
   moves with a temperature schedule – this helps escape shallow local minima.
7. Finish with an aggressive hill‑climb to polish the result.
8. Return the best probability vector (sum = 1).  The evaluation function is
   scale‑invariant, so this satisfies the harness.

The whole search respects the 1000 s budget by checking the wall‑clock time
continuously.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        if k == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.zeros(n)
            a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        # sparse Dirichlet draw (α = 0.1) – many tiny entries
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # a_i ∝ (i+1)^α * (n-i)^β  (discrete Beta‑like shape)
        alpha = rng.uniform(-0.4, 2.5)
        beta = rng.uniform(-0.4, 2.5)
        i = np.arange(n, dtype=float) + 1.0
        a0 = (i ** alpha) * ((n + 1 - i) ** beta)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        y0 = np.log(a0 + 1e-12)

    elif mode == "cosine":
        # a_i ∝ (1 - cos(π i/(n-1)))^p  – raised‑cosine shape
        p = rng.uniform(0.3, 5.0)
        base = 0.5 - 0.5 * np.cos(np.pi * np.arange(n) / (n - 1))
        a0 = base ** p
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        y0 = np.log(a0 + 1e-12)

    elif mode == "pow_sym":
        # a_i ∝ (i·(n+1-i))^{-p}  – symmetric power‑law concentrated at edges
        p = rng.uniform(0.1, 2.5)
        i = np.arange(n, dtype=float) + 1.0
        a0 = (i * (n + 1 - i)) ** (-p)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Simple hill‑climb (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Multi‑active projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The sub‑gradient is the average of the gradients of all (almost) active
    convolution entries.  A simple back‑tracking line‑search guarantees
    monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec  # because d (a_i a_j) / d a_i = 2 a_j

        # step size (decreasing) with simple back‑tracking
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even the reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score

def simulated_annealing(

    fft_len: int,
    n: int,
    deadline: float,
    max_move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing that moves a random fraction of mass between two
    coordinates.  Moves that increase the fast objective are occasionally
    accepted with a temperature that decays linearly with elapsed time.
    """
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start, 1e-6)

    # temperature scale – a few percent of the current score
    T_start = best_score * 0.04
    T_end = best_score * 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start
        tfrac = elapsed / total
        T = T_start * (1.0 - tfrac) + T_end * tfrac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = fast_score(new_a, fft_len, n)
        Δ = new_score - best_score

        if Δ < 0.0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            best = new_a
            best_score = new_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats (probability vector).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            if best_a.sum() == 0.0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the continuation L‑BFGS stage.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=70))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",
        "beta",
        "cosine",
        "pow_sym",
        "best",
    ]

    # Reserve roughly 300 s for the L‑BFGS / short hill‑climb phase.
    lbgfs_deadline = deadline - 530.0
    iteration = 0

    while time.time() < lbgfs_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # brief hill‑climb after each L‑BFGS run (≈ 0.8 s)
        hc_deadline = min(lbgfs_deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Refine with the multi‑active projected sub‑gradient method.
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 200.0
    if time.time() < subgrad_deadline:
        print("[multi‑subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.40,
            max_iters=250_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[multi‑subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing to escape shallow local minima.
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 70.0
    if time.time() < anneal_deadline:
        print("[annealing] start")
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, anneal_deadline
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[annealing] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
