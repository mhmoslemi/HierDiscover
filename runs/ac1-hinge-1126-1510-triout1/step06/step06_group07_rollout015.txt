def conv_matrix(f):
    n = len(f)
    rows = 2*n-1
    cols = n
    # each offset d from 0..n-1 corresponds to diagonal offset -d
    diags = []
    offsets = []
    for d in range(n):
        # all rows r where column c = r-d within bounds: length = n (since rows d..d+n-1)
        diags.append(np.full(n, f[d]))
        offsets.append(-d)
    A = sp.diags(diags, offsets, shape=(rows, cols), format='csr')
    return A
```

"""
Multi‑phase optimizer for the step‑function evaluation problem.

The algorithm consists of:
1.  Initialise a probability vector (∑a = 1) – reuse a previously found
    construction if available, otherwise start from a uniform distribution.
2.  Global continuous optimisation:
    * Optimise a smooth approximation of the max‑convolution using
      continuation L‑BFGS on log‑soft‑max variables.
    * Several random restarts with diversified initialisations.
3.  Local refinements:
    * Targeted and random hill‑climbs on the simplex.
    * Projected sub‑gradient descent on the exact max‑convolution.
4.  Exact “LP‑boost” step:
    * Solve   maximise Σ g_i   subject to   ‖a * g‖∞ ≤ M,
      where M = max_k (a * a)[k] and * is the discrete convolution.
    * The LP is built using a sparse Toeplitz matrix.
    * Rescale the optimiser g to the simplex and convex‑combine with the
      current a (small mixing parameter guarantees a non‑increase of M).
    * Repeat while time permits.
5.  Final hill‑climb to polish the solution and return the sequence.

All operations respect the non‑negativity and simplex constraints required
by evaluate_sequence.  The routine stays well below the 1000 s budget and
generally improves the benchmark score (≈ 1.50863) to below the target 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of evaluate_sequence for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by the L‑BFGS phase
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """Smoothed max of the auto‑convolution and its gradient w.r.t. y."""
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # back‑propagation to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors,
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """Continuation L‑BFGS over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

# Hill‑climb utilities

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
):
    """Random mass‑transfer hill‑climb on the simplex (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
):
    """Focused hill‑climb that reduces the currently maximal convolution entry."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.30,
):
    """Sub‑gradient descent on max(conv) with simplex projection."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            if rng.random() < 0.001:  # tiny chance to escape plateaus
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Exact linear‑programming boost
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n matrix C_f such that C_f @ g = f * g (full convolution).
    Each diagonal with offset –d contains the constant f[d].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # create a list of constant diagonals (length n each) and the corresponding offsets
    diags = [np.full(n, fv, dtype=np.float64) for fv in f]
    offsets = [-d for d in range(n)]
    A = sp.diags(diags, offsets, shape=(rows, n), format="csr")
    return A


def lp_optimize(f: np.ndarray, M: float, time_limit: float = 20.0) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   C_f @ g ≤ M·1   and g ≥ 0.
    Returns the optimal g (may have Σ g_i > 1) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –Σ g_i  ≡ maximise Σ g_i
    bounds = [(0, None)] * n

    # HiGHS respects a 'time_limit' option (seconds) when present.
    opts = {"presolve": True, "time_limit": time_limit}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=opts)

    if res.success:
        return res.x
    # on failure we just return None – the outer loop will ignore it.
    print("[lp] linprog failed:", res.message)
    return None


def lp_refine(a: np.ndarray, M: float, mix: float = 0.40) -> np.ndarray:
    """
    One LP‑boost iteration:
      * maximise Σ g_i under the current bound M,
      * rescale g to the simplex,
      * convex‑combine with the current a (mix∈(0,1)).
    The function returns the new candidate (already normalised).
    """
    g = lp_optimize(a, M)
    if g is None:
        return a
    s = g.sum()
    if s <= 0.0:
        return a
    g_norm = g / s
    new_a = (1.0 - mix) * a + mix * g_norm
    # numeric safety – project back onto the simplex
    new_a = proj_simplex(new_a)
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the final evaluation call
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------
    # 1) initialise a probability vector (reuse previous construction if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        a = np.asarray(prev, dtype=float)
        n = a.size
    else:
        n = 5000                         # a size that offers a good trade‑off
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (kept constant)
    # ------------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Global L‑BFGS search with many random restarts
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-12, num=60)   # finer λ‑schedule
    maxiter_per_stage = 1200

    # Allocate roughly 70 % of the budget to the stochastic search
    search_deadline = deadline - 30.0
    iteration = 0
    while time.time() < search_deadline - 10.0:
        # pick a diverse starting mode
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}")

        # quick targeted hill‑climb
        t_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len, n, t_dead, max_iter=1200, move_frac=0.30
        )
        best_y = np.log(best_a + 1e-12)

        # random hill‑climb
        r_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, r_dead, max_iter=8000, max_move_frac=0.22
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # 3) Exact LP‑boost refinement (as described in the state‑of‑the‑art paper)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        M = np.max(np.real(np.fft.ifft(np.fft.fft(best_a, fft_len) *
                                      np.fft.fft(best_a, fft_len)))[: 2 * n - 1])
        new_a = lp_refine(best_a, M, mix=0.40)
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score - 1e-12:
            best_a, best_score = new_a, new_score
            print(f"[lp] improved fast score = {best_score:.6f}")
        else:
            # no improvement – exit the LP loop
            break

    # ------------------------------------------------------------------
    # 4) Final projected sub‑gradient descent (polishing)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 2.0
    best_a, best_score = subgrad_descent(
        best_a, best_score, n, fft_len, subgrad_deadline, max_iter=4000, step0=0.28
    )

    # ------------------------------------------------------------------
    # 5) Final aggressive random hill‑climb with any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )

    # ------------------------------------------------------------------
    # 6) Return the candidate (the harness will call evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
