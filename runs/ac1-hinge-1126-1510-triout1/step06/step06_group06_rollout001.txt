"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------

The search proceeds in several complementary phases that together
explore the space more aggressively than the baseline code.

1. **Initial point** – Use the best sequence found so far (`height_sequence_1`)
   if it exists, otherwise start from a uniform distribution.

2. **Diverse random seeds** – The function `init_y_extended` creates a
   large variety of starting points: random, exponential, power‑law,
   edge‑biased, edge‑power, two‑block, and sparse random supports.
   This greatly widens the basin of attraction for the subsequent
   L‑BFGS optimiser.

3. **Continuation L‑BFGS** – A geometric λ‑schedule (smooth‑max
   temperature) is applied.  The optimiser is called repeatedly with
   decreasing smoothing factors, giving a smooth continuation from a
   coarse to a very sharp approximation of the true maximum‑convolution.
   Each restart keeps the best candidate.

4. **Parameter sweep of simple shapes** – Before the heavy optimisation
   a cheap random‑search over the above shape families is performed.
   Each candidate is evaluated with a fast FFT based surrogate
   (`fast_score`).  The best of these seeds becomes the starting point
   for L‑BFGS.

5. **Local refinements** – Targeted hill‑climb (moves mass away from
   the worst contributing pair) and a classic random hill‑climb are
   applied while time permits.

6. **Projected sub‑gradient descent** – Optimises the true
   `max(conv(a,a))` objective on the simplex using sub‑gradients.

7. **Simulated annealing** – A final stochastic phase explores
   uphill moves with a temperature schedule, helping escape
   remaining local minima.

8. **Final polishing** – One last random hill‑climb.

All internal evaluations use `fast_score`, which computes
`2·n·max(conv(a,a))` via an FFT and assumes `sum(a)=1`.
The returned sequence is stored in the global variable
`height_sequence_1` for reuse and finally evaluated with the
provided `evaluate_sequence`.

The whole procedure respects the given time budget (default 1000 s)
and stops a few seconds early to avoid a forced termination.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a probability vector:
    2·n·max(conv(a,a)) where the convolution is computed via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    The smoothing parameter is `lam = lam_factor * max(conv)`.
    """
    a = softmax(y)

    # --- convolution a*a (full) ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max ---
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. conv ---
    w = exp_shift / Z

    # --- propagate gradient back to a ---
    # note: w is defined on the full convolution support (2n-1)
    # We need the gradient w.r.t. a, which is the convolution of w
    # with the reversed a.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that corresponds to the original n entries
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    In each step the shift giving the maximal overlap is identified,
    the most contributing index pair is found, and a fraction of mass
    is moved away from the worst contributor to a location whose partner
    under the same shift is (almost) empty.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the symmetric half so that 0 <= k <= n-1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


# ----------------------------------------------------------------------
# Rich initialisation (extended)
# ----------------------------------------------------------------------


def init_y_extended(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    Several diverse heuristics are mixed to keep the restarts diverse.
    """
    mode = rng.choice(
        ["random", "exp", "power", "edge", "edge_power", "two_block", "sparse", "best"]
    )
    if mode == "random":
        return rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    elif mode == "edge":
        # simple edge bump (original version)
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    elif mode == "edge_power":
        # power‑law decay on both ends
        m_edge = max(1, int(n * rng.uniform(0.02, 0.20)))
        p = rng.uniform(0.2, 2.5)
        weights = (np.arange(1, m_edge + 1, dtype=float) ** (-p))
        a0 = np.zeros(n, dtype=float)
        a0[:m_edge] = weights
        a0[-m_edge:] = weights[::-1]
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    elif mode == "two_block":
        # uniform mass on two symmetric blocks at the ends
        block_len = max(1, int(n * rng.uniform(0.01, 0.10)))
        a0 = np.zeros(n, dtype=float)
        a0[:block_len] = 0.5 / block_len
        a0[-block_len:] = 0.5 / block_len
        # the rest stays zero, sum already 1
        return np.log(a0 + 1e-12)

    elif mode == "sparse":
        # random sparse support with uniform weights
        support_sz = max(1, int(n * rng.uniform(0.001, 0.03)))
        idx = rng.choice(n, size=support_sz, replace=False)
        a0 = np.zeros(n, dtype=float)
        a0[idx] = 1.0 / support_sz
        return np.log(a0 + 1e-12)

    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


# ----------------------------------------------------------------------
# Simulated annealing (stochastic large‑scale refinement)
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_T: float = 0.05,
    final_T: float = 1e-6,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random mass transfer.
    Accepted uphill moves decrease over time, enabling escape from
    shallow local minima.
    """
    rng = np.random.default_rng()
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    start = time.time()
    total = max(deadline - start, 1e-9)

    while time.time() < deadline:
        # linear in log‑time temperature schedule
        elapsed = time.time() - start
        frac = elapsed / total
        T = init_T * (final_T / init_T) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative heights (probability simplex, sum≈1).
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin (2 s) for the final evaluation call
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Load previous solution if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to probability simplex (scale‑invariant objective)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap random shape exploration (no gradients)
    # ------------------------------------------------------------------
    shape_deadline = min(deadline, time.time() + 30.0)  # spend ≤30 s here
    while time.time() < shape_deadline:
        mode = rng.choice(["edge_power", "two_block", "sparse"])
        if mode == "edge_power":
            m_edge = max(1, int(n * rng.uniform(0.02, 0.15)))
            p = rng.uniform(0.2, 2.5)
            w = (np.arange(1, m_edge + 1, dtype=float) ** (-p))
            a0 = np.zeros(n, dtype=float)
            a0[:m_edge] = w
            a0[-m_edge:] = w[::-1]
            a0 /= a0.sum()
        elif mode == "two_block":
            blk = max(1, int(n * rng.uniform(0.01, 0.08)))
            a0 = np.zeros(n, dtype=float)
            a0[:blk] = 0.5 / blk
            a0[-blk:] = 0.5 / blk
        else:  # sparse
            sz = max(1, int(n * rng.uniform(0.001, 0.02)))
            idx = rng.choice(n, size=sz, replace=False)
            a0 = np.zeros(n, dtype=float)
            a0[idx] = 1.0 / sz
        # normalise (already 1) and evaluate
        score = fast_score(a0, fft_len_conv, n)
        if score < best_score:
            best_a = a0.copy()
            best_y = np.log(best_a + 1e-12)
            best_score = score
            print(f"[shape] new best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – global optimisation (multiple L‑BFGS restarts)
    # ------------------------------------------------------------------
    # a geometric λ‑schedule ranging from a relatively smooth objective
    # down to an almost exact max‑operator.
    lam_factors = np.geomspace(0.5, 1e-7, num=45)
    maxiter_per_stage = 600
    iteration = 0

    while time.time() < deadline - 5.0:
        y0 = init_y_extended(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Targeted hill‑climb
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ------------------------------------------------------------------
        # Random hill‑climb
        # ------------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max‑conv objective
    # ------------------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0
    eps = 1e-12
    while time.time() < deadline - 5.0:
        subgrad_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        step = step0 / np.sqrt(subgrad_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92

        if subgrad_iter % 200 == 0:
            print(f"[subgrad] iter={subgrad_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (global stochastic refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        anneal_deadline = deadline - 2.0
        best_a, best_score = simulated_annealing(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=anneal_deadline,
            init_T=0.04,
            final_T=1e-7,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[anneal] best fast score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
