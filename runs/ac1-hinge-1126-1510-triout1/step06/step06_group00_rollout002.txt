"""
Propose a step‑function sequence with a very low evaluation score.

The algorithm is a multi‑stage local search that combines:
  * many random‑restart L‑BFGS optimisations of a log‑sum‑exp
    smooth approximation of the max‑convolution,
  * a long exponentiated‑gradient descent on the exact (hard) max,
  * a refined L‑BFGS “fine‑tune” with an even smaller smoothing
    parameter,
  * a short simulated‑annealing phase and a final greedy hill‑climb.

All stages respect a global 1000 s time budget (configurable via the
`budget_s` argument).  The best probability vector (sum = 1) found
before the deadline is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule – decreasing like 1/sqrt(t)
        eta = 0.8 / np.sqrt(it + 1.0)

        # exponentiated gradient update (clipping to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'triangle', 'gauss'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'triangle':
        # linear ramp up then down
        half = n // 2
        up = np.linspace(1, n, half, endpoint=False)
        down = up[::-1]
        a = np.concatenate([up, down]) if n % 2 == 0 else np.concatenate([up, [n//2+1], down])
        a = a.astype(float)
    else:  # gauss
        centre = n / 2.0
        sigma = rng.uniform(0.1 * n, 0.3 * n)
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Short simulated annealing that swaps a random fraction of mass."""
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """Greedy hill‑climb: accept only strictly improving pairwise mass moves."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    phase4_end = deadline

    # ------------------------------------------------------------------
    # Stage 1 – many random‑restart L‑BFGS continuations
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002,
                    0.0001, 5e-05, 2e-05, 1e-05]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑restart] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long exponentiated‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g)
        print(f"[exp‑grad] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – refined L‑BFGS with a tighter smoothing schedule
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # a tighter schedule focused on the small‑λ region
        fine_lam_schedule = [0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06]
        fine_maxiter = 2500
        best_a, best_score = lbfgs_continuation(best_a,
                                                fine_lam_schedule,
                                                fine_maxiter,
                                                phase3_end,
                                                fft_len_conv,
                                                fft_len_g,
                                                n)
        print(f"[refine‑L‑BFGS] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short simulated annealing + final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        anneal_deadline = phase4_end - 0.30
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] best fast_score={best_score:.6f}")

        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=12000)
        print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence as a plain list.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)  # pylint: disable=undefined-variable
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] could not evaluate sequence:", e)
    return final_seq
