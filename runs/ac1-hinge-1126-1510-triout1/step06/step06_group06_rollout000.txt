"""
Hybrid optimizer for the autocorrelation sup‑norm problem.

The algorithm consists of four stages:

1. **Parameterized shape search** – a cheap grid over two simple
   families (inverse‑product power and symmetric exponential) mixed
   with the uniform distribution.  The best shape is kept.

2. **Smooth‑max L‑BFGS** – a continuation of L‑BFGS optimisations
   over a geometric λ‑schedule (the smoothed max temperature).  The
   current best solution supplies the starting point for many short
   restarts.

3. **Local hill‑climbing** – a fast random mass‑transfer climb
   alternating with a targeted climb that moves mass away from the
   indices that contribute most to the current maximal convolution
   entry.

4. **Projected sub‑gradient descent** – direct optimisation of the
   true max‑convolution objective on the probability simplex,
   followed by a final polishing hill‑climb.

All operations work on the probability simplex, i.e. the sequence
is always normalised to sum = 1, which makes the objective
scale‑invariant.  The final (clamped) sequence is stored in the
global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a)), where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max objective and L‑BFGS wrapper
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max
    shifted = conv - max_c          # peak at zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # propagate gradient back to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation of L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage,
                     "gtol": 1e-14,
                     "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local hill‑climb procedures
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are retained.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))  # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weights for that shift (used to choose a safe target)
        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient
    of the (non‑smoothed) max‑operator.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the non‑negative half‑shift
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


# ----------------------------------------------------------------------
# Diverse initialisation for L‑BFGS
# ----------------------------------------------------------------------


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    A variety of heuristics are mixed to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The method combines a cheap parametric grid search, several L‑BFGS
    restarts with a smooth‑max surrogate, targeted/random hill‑climbs,
    and a projected sub‑gradient descent on the true objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size and initial point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
        best_a = np.maximum(best_a, 0.0)
        s = best_a.sum()
        if s == 0.0:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a = best_a / s
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Parameterised shape grid search
    # ------------------------------------------------------------------
    # two families: (i+1)*(n-i) power, and symmetric exponentials
    p_vals = np.linspace(0.02, 0.48, 20)                 # exponent for power‑law family
    mix_vals = np.linspace(0.0, 1.0, 11)                # mixing weight with uniform
    idx = np.arange(n, dtype=float) + 1.0
    prod = idx * (n - idx + 1.0)                         # (i+1)*(n-i)

    for p in p_vals:
        # family A: inverse‑product power
        shape_a = prod ** (-p)
        shape_a /= shape_a.sum()
        # family B: symmetric exponentials
        shape_b = np.exp(-p * (idx - 1)) + np.exp(-p * (n - idx))
        shape_b /= shape_b.sum()

        for shape in (shape_a, shape_b):
            for mix in mix_vals:
                if time.time() >= deadline:
                    break
                cand = (1.0 - mix) * (1.0 / n) + mix * shape
                cand /= cand.sum()
                sc = fast_score(cand, fft_len, n)
                if sc < best_score - 1e-12:
                    best_a = cand.copy()
                    best_score = sc
                    print(f"[grid] p={p:.3f} mix={mix:.2f} new best = {best_score:.6f}")

    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 2. Smooth‑max L‑BFGS optimisation (multiple restarts)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-6, num=30)   # temperature schedule
    maxiter_per_stage = 500
    restarts = 30
    fft_len_g = next_pow2(3 * n - 2)

    for it in range(restarts):
        if time.time() >= deadline - 5.0:
            break
        y0 = init_y(rng, n, best_a, it)
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len, n)
        if sc_opt < best_score - 1e-12:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = sc_opt
            print(f"[L‑BFGS {it}] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local hill‑climbing phase
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # targeted hill‑climb (short burst)
        t_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=t_dead,
            max_iter=2000,
            move_frac=0.35,
        )
        # random hill‑climb (short burst)
        t_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=t_dead,
            max_iter=4000,
            max_move_frac=0.25,
        )

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    step0 = 0.5
    it_sub = 0
    while time.time() < deadline - 2.0:
        it_sub += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len)
        step = step0 / np.sqrt(it_sub)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92
        if it_sub % 200 == 0:
            print(f"[subgrad] iter={it_sub} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.30,
        )

    # ------------------------------------------------------------------
    # 5. Official evaluation and result reporting
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
