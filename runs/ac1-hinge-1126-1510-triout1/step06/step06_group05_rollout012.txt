            arr = np.concatenate([arr, np.zeros(n - arr.size)])
        elif arr.size > n:
            arr = arr[:n]
        best_a = arr / np.sum(arr)
    else:
        best_a = np.full(n, 1.0/n)

    fft_len_conv = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)

    best_y = np.log(best_a + 1e-12)

"""
propose_candidate
=================

Search for a non‑negative step‑function (list of floats) that yields a low
``evaluate_sequence`` score.  The routine improves on the previous
implementation by

* using a larger problem size (≈ 8000 points) for a finer discretisation,
* performing a much finer β‑like grid‑search,
* running a continuation L‑BFGS optimiser with more stages and more
  iterations per stage,
* adding a few additional parametric initialisers,
* using an Adam‑style projected sub‑gradient descent for the true
  max‑convolution objective, and
* ending with an aggressive hill‑climb that runs for the remaining time.

All helper functions are defined at module level – no closures or lambdas
– and only ``numpy``, ``scipy`` and the supplied ``evaluate_sequence`` are
used.  The total running time respects the provided budget (default
1000 s).  The final list of heights is returned; its sum is irrelevant
because the evaluation function normalises it internally.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) via a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1)…(2n‑2) correspond to i = 0…n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)        # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "linear":
        # a triangular “tent” shape
        a0 = np.minimum(np.arange(n) + 1, n - np.arange(n))
        a0 = a0.astype(float)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "double_exp":
        # symmetric double‑exponential (Laplace) centred at the middle
        rate = rng.uniform(0.001, 0.02)
        centre = (n - 1) / 2.0
        a0 = np.exp(-rate * np.abs(np.arange(n) - centre))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local optimisation utilities

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j, accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_adam(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the true objective
        f(a) = max_i (a*a)_i .
    The sub‑gradient at the active index m is
        g_i = 2·a_{m‑i}   (if the index exists).
    Adaptive step size ηₜ = η₀/√t and exponential moving average of the
    sub‑gradient.  Euclidean projection onto the simplex is performed after
    each update.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    v = np.zeros_like(a)  # momentum buffer
    it = 0

    while time.time() < deadline and it < 500_000:
        it += 1

        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        v = beta * v + (1.0 - beta) * g
        a_new = a - eta * v
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

        best_score = sc
        best_a = prob.copy()

    iteration = 0
    modes = [

        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "linear",
        "double_exp",
        "best",
    ]

    # reserve ~300 s for the final polishing stages

    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode:12s} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to tidy up each L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Refine with Adam‑style projected sub‑gradient descent on the true
    # max‑convolution objective.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[subgrad‑adam] start")
        best_a, best_score = subgradient_adam(
            best_a, fft_len_conv, n, sub_deadline, eta0=0.20, beta=0.9
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad‑adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final‑HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=50000,
        )
        print(f"[final‑HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # the harness provides this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
