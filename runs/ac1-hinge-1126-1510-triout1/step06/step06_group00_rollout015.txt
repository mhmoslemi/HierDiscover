def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30):
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam, deadline, fft_len_conv, fft_len_g, n):

def exp_gradient_optimize(a_start, deadline, n, fft_len_conv, fft_len_g, rng):
    a = a_start.copy()

def hill_climb(a, current_score, fft_len, n, deadline, max_iter=4000):

            continue

        best[j] += delta

        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

def lp_refine(a, best_score, deadline, n, fft_len):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute conv
        conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)))[:2*n-1]
        M = conv.max()
        # select top indices (active set)
        # Use a tolerance: include any index within 0.9995 * M
        # But limit to max_k indices.
        max_candidates = 30
        # Use argpartition to get top values
        if max_candidates >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -max_candidates)[-max_candidates:]
        # Build constraint matrix rows (len(idxs) x n)
        m = len(idxs)
        A = np.zeros((m, n))
        for row_i, idx in enumerate(idxs):
            # compute range of j
            start = max(0, idx - (n - 1))
            end = min(idx, n - 1)
            # j vector
            j = np.arange(start, end + 1, dtype=int)
            A[row_i, j] = a[idx - j]
        # Solve LP maximize sum(b) s.t. A @ b <= M, b >= 0
        c = -np.ones(n)  # minimize negative sum => maximize sum
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            # If LP fails, break or try random perturbation
            # random perturbation
            sigma = rng.uniform(0.001, 0.01)
            cand = a + rng.normal(0, sigma, size=n)
            cand = np.maximum(cand, 0)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                a = cand
                best_score = cand_score
                print("[LP noise improve] score", best_score)
            continue
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        # try mixing with different t
        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            a_new = (1 - t) * a + t * g
            a_new /= a_new.sum()
            new_score = fast_score(a_new, fft_len, n)

            if g_score < best_score:
                a = g
                best_score = g_score
                print("[LP replace] score %.6f" % best_score)

"""
Search for a non‑negative step‑function sequence with a low evaluation score.

The pipeline (within a 1000 s budget) is:
1. Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
2. Exponentiated–gradient descent on the exact max‑convolution.
3. Active‑set linear‑program refinement: maximise the total mass of a companion
   distribution b under the linear constraints (a * b) ≤ M, where M is the current
   maximal self‑convolution.  Mixing a with the normalised b (several mixing
   ratios are tried) strictly decreases the objective.
4. Final greedy hill‑climb that transfers a small random fraction of mass
   between two coordinates and keeps only improving moves.

All stages respect the global deadline and the returned list of floats is
passed to ``evaluate_sequence`` for the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1.
    """
    conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) *
                               np.fft.fft(a, fft_len)))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) automatically stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule – decreasing like 1/√t
        eta = 0.6 / np.sqrt(it + 1.0)

        # exponentiated gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def lp_refine(a: np.ndarray,
              best_score: float,
              deadline: float,
              n: int,
              fft_len: int):
    """
    Active‑set linear‑program refinement.
    For the current sequence a, solve
        maximise   sum(b)
        subject to a * b ≤ M   (M = max(a*a)), b ≥ 0.
    The optimum b is normalised to sum 1 and mixed back into a.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        # current convolution and bound M
        conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) *
                                   np.fft.fft(a, fft_len)))[: 2 * n - 1]
        M = conv.max()

        # pick a small active set of indices where the convolution is large
        max_candidates = min(50, conv.size)
        idxs = np.argpartition(conv, -max_candidates)[-max_candidates:]

        m = len(idxs)
        A = np.zeros((m, n))
        for row_i, idx in enumerate(idxs):
            start = max(0, idx - (n - 1))
            end = min(idx, n - 1)
            j = np.arange(start, end + 1, dtype=int)
            A[row_i, j] = a[idx - j]

        # linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds,
                      method='highs')
        if not res.success:
            # fall back to a tiny random perturbation
            sigma = rng.uniform(0.001, 0.01)
            cand = a + rng.normal(0.0, sigma, size=n)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                a = cand
                best_score = cand_score
                print("[LP‑noise] improved to %.6f" % best_score)
            continue

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            break
        g = b_opt / sum_b                     # normalise to sum 1

        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            a_new = (1.0 - t) * a + t * g
            a_new /= a_new.sum()
            new_score = fast_score(a_new, fft_len, n)
            if new_score < best_score - 1e-12:
                a = a_new
                best_score = new_score
                improved = True
                print("[LP‑mix t=%.2f] score %.6f" % (t, best_score))
                break
        if not improved:
            g_score = fast_score(g, fft_len, n)
            if g_score < best_score - 1e-12:
                a = g
                best_score = g_score
                print("[LP‑replace] score %.6f" % best_score)

    return a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)        # for gradient steps

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print("[init] n=%d, fast_score=%.6f" % (n, best_score))

    # ------------------------------------------------------------------
    # Global time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS (≈45 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002]
    maxiter_per_lam = 800
    restarts = 0
    while time.time() < phase1_end and restarts < 20:
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print("[L‑BFGS] new best score=%.6f" % best_score)
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (≈35 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   rng)
        print("[exp‑grad] score=%.6f" % best_score)

    # ------------------------------------------------------------------
    # Phase 3 – active‑set LP refinement (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_end = deadline - 0.30
    if time.time() < phase3_end:
        best_a, best_score = lp_refine(best_a,
                                       best_score,
                                       phase3_end,
                                       n,
                                       fft_len_conv)
        print("[LP‑refine] score=%.6f" % best_score)

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    deadline,
                                    max_iter=12000)
    print("[final hill‑climb] score=%.6f" % best_score)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score = %.6f" % final_eval)

    return final_seq
