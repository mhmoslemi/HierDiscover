"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑ a_i = 1).  The objective
  `evaluate_sequence` is homogeneous, so the scale cancels out.
* Choose a moderately large problem size (n ≈ 9 000–10 000) – enough
  degrees of freedom without making each iteration too costly.
* Initialise many diverse candidates (arcsine‑Beta, semicircle,
  uniform, sparse binary mask, random Gaussian bump and mixtures).
* For each start run a continuation L‑BFGS on the smooth
  log‑sum‑exp approximation of ``max(conv)``.  The smoothing factor
  λ is decreased geometrically (0.5 → 1e‑8).  The L‑BFGS runs are cheap
  because the convolution and its gradient are computed via FFT.
* Keep the best candidate (lowest fast score = 2·n·max(conv)).
* Do a dedicated, long L‑BFGS run from that best start with a finer
  schedule and more inner iterations – this drives the solution close
  to a local optimum of the true objective.
* Polish the result with three complementary local‑search phases:
      – Projected sub‑gradient descent on the true max‑convolution.
      – Simulated‑annealing style random mass transfers.
      – Targeted hill‑climb that moves mass away from indices that
        actively contribute to the current maximal convolution entry.
* Finally evaluate the sequence with the provided ``evaluate_sequence``
  and return it.

The whole procedure respects the 1000 s budget, prints progress
information and returns a plain Python list.

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sum = 1, all non‑negative)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a normalized probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp approximation of 2·n·max(conv) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parameters).

    The smoothing parameter λ is taken as lam_factor * max(conv) + eps.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    # conv = a * a, so ∂conv/∂a = 2 * (a convolved with delta) -> use FFT trick
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract central n entries
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def continuation_lbfgs(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: np.ndarray,
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Perform L‑BFGS on the smoothed objective while slowly decreasing the
    smoothing factor (continuation method).  Returns the final parameter
    vector y (log‑softmax).
    """
    y = y0.copy()
    for lam in lam_schedule:
        # L‑BFGS on the current smoothed problem
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    """β(α,α) density on (0,1) – α≈0.5 is the classical arcsine."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # guarantee positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mix_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of arcsine and uniform (weight α∈[0,1])."""
    alpha = rng.random()
    a1 = arcsine_init(n, alpha=0.70)
    a2 = uniform_init(n)
    a = alpha * a1 + (1 - alpha) * a2
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return log‑softmax parameters y for a given mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    elif mode == "mix":
        a0 = mix_init(n, rng)
    else:
        a0 = uniform_init(n)

    return np.log(a0 + 1e-12)

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random hill‑climb that only accepts moves improving the fast score."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from active contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # indices i that appear in the maximal convolution entry
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)

        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective 2·n·max(conv).
    The step size decays as η₀ / sqrt(t).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # Sub‑gradient of the active entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    init_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_temp: float = 1e-3,
    cooling: float = 0.9995,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Randomised annealing that moves a fraction of mass between two positions.
    Accepts worsening moves with probability exp(-Δ/ T).
    """
    a = a_start.copy()
    best = a.copy()
    best_score = init_score
    cur_score = init_score
    T = init_temp
    it = 0

    while time.time() < deadline:
        it += 1
        # pick source with some mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a, fft_len, n)
        diff = new_score - cur_score

        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        T *= cooling
        # slowly shrink move magnitude as temperature goes down
        max_move = max(0.02, max_move * cooling)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Prefer a larger n if we have enough time; cap at 12000 to keep FFT cheap.
    # If a previous construction is provided, respect its length if it is already
    # reasonably large; otherwise use a fresh size.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        base_n = max(5000, tmp.size)
    else:
        base_n = 8000
    # Target n ~ 9500 but make it a power‑of‑two‑friendly size for FFT
    n = min(12000, max(7000, base_n))
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise best candidate on the simplex (sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp.copy()
            best_a = best_a / best_a.sum()
        else:
            best_a = uniform_init(n)
    else:
        best_a = uniform_init(n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑start continuation L‑BFGS (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 1e-8, num=30)
    modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump", "mix"]
    mode_idx = 0
    multi_start_deadline = start_time + 0.30 * budget_s

    while time.time() < multi_start_deadline:
        mode = modes[mode_idx % len(modes)]
        mode_idx += 1
        y0 = generate_initial_y(mode, n, rng)

        y_opt = continuation_lbfgs(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[multi‑start] mode={mode:<9} improved fast = {best_score:.6f}")

        # occasional perturbation of the current best to escape shallow minima
        if rng.random() < 0.15:
            y0 = best_y + rng.normal(scale=0.03, size=n)
            y_opt = continuation_lbfgs(
                y0,
                n,
                fft_len_conv,
                fft_len_g,
                lam_schedule,
                maxiter_per_stage=300,
            )
            a_opt = softmax(y_opt)
            score_opt = fast_score(a_opt, fft_len_conv, n)
            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                best_y = y_opt.copy()
                print(f"[perturb] improved fast = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Focused long L‑BFGS run on the current best (≈ 35 % of the budget)
    # ------------------------------------------------------------------
    refined_deadline = start_time + 0.70 * budget_s
    lam_schedule_refined = np.geomspace(0.5, 1e-10, num=45)
    y_opt = continuation_lbfgs(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule_refined,
        maxiter_per_stage=600,
    )
    best_a = softmax(y_opt)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[refine] after focused L‑BFGS fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.85 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.35,
        max_iters=500_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing & targeted hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 2.0
        print("[anneal] simulated annealing")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
            rng,
            init_temp=best_score * 0.001,
            cooling=0.9997,
            max_move=0.20,
        )
        print(f"[anneal] fast score = {best_score:.6f}")

    if time.time() < deadline:
        final_hc_deadline = deadline
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=8000,
            max_move=0.18,
        )
        print(f"[final] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # provided by the harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[final] evaluation failed: {e}")
        final_val = float("inf")

    return final_seq
