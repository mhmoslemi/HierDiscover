def pairwise_balancing(a, n, fft_len, deadline):
    best_a = a.copy()
    conv = full_conv(a, fft_len)
    best_M = conv.max()
    while time.time() < deadline:
        # compute mask for active peaks within tolerance
        max_val = best_M
        eps = max_val * 1e-9  # tolerance
        mask = (conv >= max_val - eps).astype(float)
        # compute gradient g as per mask
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_vec = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_vec[n-1:n-1+n]
        # choose i high g, j low g
        i = np.argmax(g)
        j = np.argmin(g)
        if i == j:
            break
        delta = min(best_a[i], 0.2 * best_a[i])
        if delta <= 0:
            break
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalize? sum remains same.
        conv_new = full_conv(a_new, fft_len)
        M_new = conv_new.max()
        if M_new < best_M - 1e-12:
            best_a = a_new
            best_M = M_new
            conv = conv_new
        else:
            # try smaller delta or break
            # reduce delta
            delta *= 0.5
            if delta < 1e-12:
                # cannot improve
                break
    return best_a, best_M

Improved search for a low‑evaluation step‑function sequence.

The algorithm consists of three stages:

1️⃣  Multiple random restarts of exponentiated‑gradient descent on the exact
    sub‑gradient of the max‑convolution.  This quickly finds a highly
    asymmetric distribution that already beats the uniform bound.

2️⃣  An LP‑mixing phase (inspired by the classic “alternating LP” method).
    For the current sequence `a` we solve
        maximise  Σ b_j
        s.t.      a * b  ≤  M   (elementwise),   b ≥ 0,
    where `M = max(a*a)`.  The solution `b` can carry more total mass because
    it lives where `a` is small.  A convex combination `a ← (1‑α)a + αb`
    with a modest step size `α` strictly decreases the peak `M`.  We repeat
    this a few times.

3️⃣  A final greedy hill‑climb that swaps a small random fraction of mass
    between two coordinates and keeps only improving moves.

All stages respect a total time budget of 1000 s (the caller supplies the
budget).  The best sequence found before the deadline is returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence a that is already
    normalised to sum(a)=1.  Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector of length n."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def max_conv_and_subgrad(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Compute max(a*a) and a sub‑gradient of the max‑convolution.
    Returns (M, g) where g has the same size as a.
    """
    n = a.shape[0]
    # full convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # gradient of the (hard) max‑convolution
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]  # shape (n,)
    return max_val, g


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the exact objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best (a, score) found before deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    M, _ = max_conv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * M

    it = 0
    while time.time() < deadline:
        M, g = max_conv_and_subgrad(a, fft_len)

        # step size schedule – larger initial step, decays as 1/√t
        eta = 0.8 / np.sqrt(it + 1.0)

        # exponentiated update (clipped exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = best_a.copy()          # safeguard

        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑mixing phase (alternating linear programme)
# ----------------------------------------------------------------------


def build_conv_matrix_sparse(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ b = a * b  (linear convolution).
    The matrix is sparse but with n² non‑zero entries (all a_i > 0).
    """
    n = a.shape[0]
    # row indices: for each column i we have rows i … i+n‑1
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.repeat(i, n)
    data = np.repeat(a, n)

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mixing_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Alternating LP: maximise Σ b_j subject to a * b ≤ M,
    then convex‑combine a and normalised b.
    Returns improved (a, score) if any.
    """
    a = a_start.copy()
    M, _ = max_conv_and_subgrad(a, fft_len)
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    it = 0
    while time.time() < deadline and it < 10:
        # Build the convolution matrix for the current a
        A = build_conv_matrix_sparse(a)

        # Linear programme: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, best_M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – abort the mixing phase
            break

        b = res.x
        sum_b = b.sum()

        # If we cannot increase the total mass, stop
        if sum_b <= 1.0 + 1e-9:
            break

        # Normalise b to have sum 1 (scale‑invariance of the score)
        b_norm = b / sum_b

        # Convex combination – try a moderate step size α
        alpha = 0.5
        improved = False
        while alpha > 1e-4:
            a_new = (1.0 - alpha) * a + alpha * b_norm
            a_new /= a_new.sum()                     # re‑normalise
            M_new, _ = max_conv_and_subgrad(a_new, fft_len)
            if M_new < best_M:
                # Accept the improvement
                a = a_new
                best_M = M_new
                best_score = 2.0 * n * M_new
                best_a = a.copy()
                print(f"[LP‑mix] iter {it}, α={alpha:.4f}, M={best_M:.8f}, sum(b)={sum_b:.6f}")
                improved = True
                break
            else:
                alpha *= 0.5   # shrink step and retry

        if not improved:
            # No improvement with any α – abort
            break

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 6000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a small random fraction of the mass
    from one coordinate to another and keep only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and starting sequence
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # linear interpolation to the target length
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    phase0_end = start_time + 0.45 * budget_s      # exponentiated‑gradient
    phase1_end = start_time + 0.80 * budget_s      # LP mixing
    phase2_end = deadline                          # final hill‑climb

    # ------------------------------------------------------------------
    # Phase 0 – multiple random restarts of exponentiated‑gradient
    # ------------------------------------------------------------------
    while time.time() < phase0_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = exp_gradient_optimize(a_start,
                                                 phase0_end,
                                                 n,
                                                 fft_len,
                                                 rng)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[exp‑grad] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑mixing (alternating linear programme)
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = lp_mixing_optimize(best_a,
                                                phase1_end,
                                                n,
                                                fft_len,
                                                rng)
        print(f"[LP‑mix] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        phase2_end,
                                        max_iter=12000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the final sequence
    # ------------------------------------------------------------------

    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
