"""
Propose a step‑function minimising `evaluate_sequence`.

The algorithm is a multi‑phase heuristic:

1. **Problem size** – work on a moderately large n (≈ 10 000).  Larger
   n gives more flexibility for the convolution to flatten.
2. **Initialisation** – start from the best known construction
   (`height_sequence_1`) if present, up‑sampled to the chosen n,
   otherwise from several analytic shapes (arcsine, semicircle, uniform,
   random Gaussian, sparse binary mask, Gaussian bump).
3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution using a geometric schedule of
   smoothing parameters (λ from 0.5 down to 1e‑9).  Several random
   restarts (soft‑max parameters perturbed with Gaussian noise) are
   performed; the best result is kept.
4. **Exponential‑gradient descent** – a projected mirror‑descent step
   on the true objective.  The sub‑gradient of the active convolution
   entry is used to multiply the current probabilities by
   `exp(‑η·g)`, followed by normalisation.  The step‑size η decays as
   1/√t.
5. **Aggressive hill‑climbing** – random and targeted greedy moves
   (mass transfers) that only accept improvements of the fast
   (FFT‑based) score.
6. **Final evaluation** – the obtained sequence is handed to
   `evaluate_sequence` and returned.

All convolution work is performed with real FFTs for speed and the
search respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Helper utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the convolution of length 2*n‑1 (real part only).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a probability vector (∑a=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_init(n: int, alpha: float = 0.70) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)

    y0 = np.log(a0 + 1e-12)
    return y0


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution objective and its gradient w.r.t. the
    unconstrained soft‑max parameters `y`.

    The smoothing constant is λ = max(conv)·lam_factor  (plus a tiny ε).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv (vector w, same length as conv)
    w = exp_shift / Z

    # Gradient of conv = a * a  w.r.t. a via reverse convolution.
    # We need d(conv)/da = 2 * (a convolved with delta shifted)
    # The following trick computes grad_y = a * (g - <a,g>) where g is
    # the back‑propagated gradient.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a short L‑BFGS optimisation for each
    λ in `lam_schedule`, feeding the result of one stage as the start of
    the next.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb (mass transfer) that never worsens the fast score."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves mass away from
    indices that actively contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)

        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def exp_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.50,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true objective using the exponential map.
    The update is a_i ← a_i·exp(‑η·g_i) with subsequent normalisation,
    where g_i is the sub‑gradient of the active convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the max‑entry:
        g = np.zeros_like(a)
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        # exponential update (avoid overflow)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = uniform_init(n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The method combines several random restarts of a
    continuation L‑BFGS optimiser, an exponential‑gradient phase, and
    aggressive hill‑climbing.  It respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (n).  We aim for around 10 000 entries.
    # ------------------------------------------------------------------
    # If a previous construction exists, up‑sample it; otherwise start
    # from uniform.
    init_seq = globals().get("height_sequence_1", None)
    desired_n = 10000
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size > 0:
            # Upsample / downsample by linear interpolation to desired_n
            old_x = np.linspace(0.0, 1.0, tmp.size)
            new_x = np.linspace(0.0, 1.0, desired_n)
            a0 = np.interp(new_x, old_x, tmp)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                best_a = uniform_init(desired_n)
            else:
                best_a = a0 / a0.sum()
            n = desired_n
        else:
            n = desired_n
            best_a = uniform_init(n)
    else:
        n = desired_n
        best_a = uniform_init(n)

    # FFT length (next power of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with random restarts
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=30))
    maxiter_per_stage = 300
    modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump"]
    mode_cycle = 0
    iteration = 0
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline:
        # every few iterations we jitter the current best point
        if iteration % 6 == 0 and iteration > 0:
            y0 = best_y + rng.normal(scale=0.04, size=n)
            mode = "jitter"
        else:
            mode = modes[mode_cycle % len(modes)]
            mode_cycle += 1
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # a brief random hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponential‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    exp_deadline = min(deadline - 20.0, start_time + 0.85 * budget_s)
    print("[exp] starting exponential‑gradient descent")
    best_a, best_score = exp_gradient_descent(
        best_a, fft_len_conv, n, exp_deadline, eta0=0.50, max_iters=600_000
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[exp] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive final hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=20_000,
            max_move=0.30,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=12_000,
                max_move=0.22,
            )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will provide evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
    except Exception as e:  # pragma: no cover
        print("[error] evaluate_sequence raised:", e)
        final_val = float("inf")
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
