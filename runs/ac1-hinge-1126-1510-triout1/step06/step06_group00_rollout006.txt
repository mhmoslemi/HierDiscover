    elif shape_type == 'power':

        idx = np.arange(1, n+1)
        a = idx ** (-p) + (n - idx + 1) ** (-p)
    else: # mix
        lam = 10**rng.uniform(np.log10(1e-5), np.log10(0.5))
        p = rng.uniform(0.2, 3.0)
        alpha = rng.uniform(0, 1)
        a_exp = np.exp(-lam * np.arange(n)) + np.exp(-lam * np.arange(n-1, -1, -1))
        idx = np.arange(1, n+1)
        a_pow = idx**(-p) + (n - idx + 1) ** (-p)
        a = alpha * a_exp + (1-alpha) * a_pow
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0:
        continue
    a = a / s
    score = fast_score(a, fft_len_conv, n)
    if score < best_score:
        best_score = score
        best_a = a.copy()
        print(f"[param search] new best {best_score:.6f} shape {shape_type}")

def exp_gradient_optimize(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64) # shape (2n-1)
        # Compute subgradient g
        a_rev = a[::-1]
        # Convolution of mask with a_rev:
        # (mask * a_rev) conv:
        # We'll compute via FFT length for convolution: need length >= (2n-1)+(n)-1 = 3n-2
        fft_len_g = next_pow2(3*n-2)
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]  # size n
        # step size
        eta = 0.6 / np.sqrt(it + 1.0)
        # exponentiated gradient update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a = a / s
        # evaluate
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score
```

def hill_climb(a, current_score, fft_len, n, deadline, max_iter=20000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # choose delta as a fraction of a[i]
        delta = best[i] * rng.random() * 0.2  # up to 20%
        if delta <= 0:
            it += 1
            continue
        # propose new
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute new score
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score
```

Search for a non‑negative step‑function (distribution) that minimizes
    2·n·max(a*a) / (sum(a)^2)
where n = length of the sequence.  The algorithm consists of
    1.  Random sampling of several analytic families (exponential,
        power‑law, mixed, sparse support and arcsine‑type),
        keeping the best candidate.
    2.  A long run of exponentiated‑gradient descent on the exact
        max‑convolution (sub‑gradient of the ∞‑norm).  The update
        a ← a·exp(−η·g) automatically stays on the probability simplex.
    3.  A final greedy hill‑climb that moves a random fraction of mass
        between two coordinates and accepts only improving moves.
All stages respect a global 1000 s budget and the best sequence found
before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a distribution a (sum(a)=1):
        2·n·max_{k}(a*a)[k]
    using FFT‑based convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the exact max‑convolution.
    The sub‑gradient is computed from the mask of indices where the
    convolution achieves its maximum.
    Returns the best vector found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # shape (2n‑1)

        # gradient of the (hard) max‑convolution:
        #  g_i = 2 * sum_{j} a_j * mask_{i+j}
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n - 1: n - 1 + n]               # size n

        # step size schedule
        eta = 0.6 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # evaluate
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly transfers a random fraction of the
    mass from one coordinate to another, accepting only strictly improving
    moves.
    """
    best = a.copy()
    best_sc = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a random fraction (up to 20 %) of the mass at i to j
        delta = best[i] * rng.random() * 0.20
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low evaluation
    score.  The function respects a 1000‑second time budget and returns the
    best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)       # for any legacy code using np.random

    # --------------------------------------------------------------
    # Problem size & initial sequence
    # --------------------------------------------------------------
    n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # Linear interpolation to target length n
            x_old = np.linspace(0.0, 1.0, init_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len, n)
    print(f"[init] fast_score={best_sc:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – random analytic families
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.35 * budget_s
    families = ['exp', 'power', 'mix', 'sparse', 'arcsine']
    i_arr = np.arange(n)
    idx1 = np.arange(1, n + 1, dtype=float)

    while time.time() < phase1_deadline:
        fam = rng.choice(families)

        if fam == 'exp':
            # lam ∈ [1e-8, 1]  (log‑uniform)
            lam = 10 ** rng.uniform(-8, -0.0)
            a = np.exp(-lam * i_arr) + np.exp(-lam * (n - 1 - i_arr))

        elif fam == 'power':
            p = rng.uniform(0.2, 4.0)
            a = idx1 ** (-p) + (n - idx1 + 1) ** (-p)

        elif fam == 'mix':
            lam = 10 ** rng.uniform(-8, -0.0)
            p = rng.uniform(0.2, 4.0)
            alpha = rng.random()
            a_exp = np.exp(-lam * i_arr) + np.exp(-lam * (n - 1 - i_arr))
            a_pow = idx1 ** (-p) + (n - idx1 + 1) ** (-p)
            a = alpha * a_exp + (1.0 - alpha) * a_pow

        elif fam == 'sparse':
            # random support size between 50 and min(2000, n)
            s = rng.integers(50, min(2000, n))
            positions = rng.choice(n, size=s, replace=False)
            a = np.zeros(n, dtype=float)
            a[positions] = 1.0 / s

        else:  # 'arcsine' – 1/sqrt(i·(n‑i))
            a = 1.0 / np.sqrt((i_arr + 1) * (n - i_arr))

        # Normalise
        s = a.sum()
        if s <= 0.0:
            continue
        a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[family:{fam}] new best fast_score={best_sc:.6f}")

        # Occasionally perturb the current best to escape tiny basins
        if rng.random() < 0.008:
            sigma = rng.uniform(0.001, 0.015)
            noisy = best_a + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            ss = noisy.sum()
            if ss > 0.0:
                noisy = noisy / ss
                sc = fast_score(noisy, fft_len, n)
                if sc < best_sc:
                    best_sc = sc
                    best_a = noisy.copy()
                    print(f"[noise] new best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = deadline - 0.2  # keep ~0.2 s for the final hill‑climb
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                phase2_deadline,
                                                n,
                                                fft_len,
                                                fft_len_g)
        print(f"[exp‑grad] refined fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     deadline,
                                     max_iter=50000)
        print(f"[hill‑climb] final fast_score={best_sc:.6f}")

    return final_seq
