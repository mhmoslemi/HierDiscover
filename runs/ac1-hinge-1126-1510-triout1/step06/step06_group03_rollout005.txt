        conv = conv_fft(best, fft_len, n)
        m = conv.argmax()
        # compute contributions list: pairs (i,j) where i+j=m, i in [0,n), j in [0,n)
        # for each i compute product = best[i] * best[j]
        # we want to move mass from larger product pair's larger component.
        # choose i among these pairs weighted by product maybe.
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        pair_i = idx[mask]
        pair_j = j[mask]
        # compute contributions
        contrib = best[pair_i] * best[pair_j]
        if contrib.sum() == 0:
            # fallback random
            i = rng.integers(0, n)
        else:
            # choose a pair with probability proportional to contribution
            probs = contrib / contrib.sum()
            chosen_pair_idx = rng.choice(len(pair_i), p=probs)
            # Now choose which side to reduce: we take the larger of the two
            i = pair_i[chosen_pair_idx] if best[pair_i[chosen_pair_idx]] >= best[pair_j[chosen_pair_idx]] else pair_j[chosen_pair_idx]
        # Pick destination jdest far: we prefer index with low mass
        # Choose random index, but maybe weighted by (1 - a) to favor low components.
        dest = rng.integers(0, n)
        # Ensure not same
        if dest == i:
            continue
        delta = best[i] * rng.random() * move_frac
        if delta <= 0:
            continue
        best[i] -= delta
        best[dest] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[dest] -= delta

    return best, best_score

"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1).  The objective
  `evaluate_sequence` is homogeneous, so the scale does not matter.
* Use a relatively large problem size (n≈10 000) – this gives many degrees of
  freedom while keeping FFT‑based convolution cheap.
* Initialise many analytically motivated shapes:
      – symmetric β‑distribution a_i ∝ x^{α‑1}(1‑x)^{α‑1} (α∈{0.55,…,0.85})
      – discrete Wigner semicircle
      – uniform, random Gaussian, sparse binary mask, Gaussian bump
* Optimise the smoothed objective
      f_λ(a)=2·n·smooth_max(a*a)   (smooth_max ≈ log‑sum‑exp)
  with continuation L‑BFGS on the unconstrained log‑softmax parameters.
  A fine geometric λ‑schedule (0.5 → 1e‑9) with several hundred L‑BFGS
  iterations per step drives the solution toward the true max‑convolution.
* After each L‑BFGS run perform a quick random hill‑climb to polish the
  candidate.
* Run a long projected sub‑gradient descent on the true objective
  (max of the linear convolution) with a diminishing step size.
* Finish with aggressive local searches:
      – random hill‑climb,
      – targeted hill‑climb that moves mass away from the current maximal
        convolution entry,
      – pairwise‑imbalance move that deliberately unbalances the two indices
        that contribute most to the current peak.
* The best candidate is finally evaluated with the provided `evaluate_sequence`
  and returned.

The whole routine respects the 1000 s wall‑time budget; intermediate results
are printed for monitoring.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns probabilities that sum to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a of length 2*n-1 using FFT.
    a is length n, fft_len ≥ 2*n-1.
    Returns real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when sum(a)=1:
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initial shapes
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    """Symmetric beta(α,α) density on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)                      # force positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a log‑softmax starting vector for a given mode.
    Supported modes:
        arcsine, arcsine_<alpha>, semicircle, uniform,
        random, binary, bump
    """
    if mode.startswith("arcsine_"):
        try:
            alpha = float(mode.split("_")[1])
        except Exception:
            alpha = 0.75
        a0 = arcsine_init(n, alpha=alpha)
    elif mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.75)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        # fallback to random Gaussian
        a0 = random_gaussian_init(n, rng)

    # Convert to log‑softmax parameters
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for continuation L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(a*a) and its gradient w.r.t. the
    unconstrained vector y (log‑softmax parameters).

    lam = max(conv) * lam_factor  (smoothing constant)
    """
    a = softmax(y)

    # linear convolution a*a  (via FFT)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv - max_c                     # centre at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z       # shape (2n-1,)

    # gradient of conv w.r.t. a (reverse convolution)
    # (conv = a * a) => grad = 2 * (a reverse‑convolved with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂smooth_max/∂a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax (y -> a)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing parameter λ.
    Each stage starts from the result of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12,
                     "disp": False, "maxcor": 10},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random greedy hill‑climb that transfers a fraction of mass
    from one index to another if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from the indices
    that actively contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)

        # pick a destination that is (heuristically) far from the peak region
        jdest = rng.integers(0, n)
        if i == jdest:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[jdest] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[jdest] -= delta
    return best, best_score


def pairwise_imbalance_move(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Specialized local move that deliberately unbalances the pair (i,j)
    contributing most to the current peak of the auto‑convolution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # indices (i,j) with i+j=m and 0≤i,j<n
        i_arr = np.arange(n)
        j_arr = m - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        i_valid = i_arr[mask]
        j_valid = j_arr[mask]

        if i_valid.size == 0:
            # fallback to random move
            i = rng.integers(0, n)
        else:
            # contribution of each pair = a_i * a_j
            contrib = best[i_valid] * best[j_valid]
            if contrib.sum() == 0.0:
                i = rng.choice(i_valid)
            else:
                probs = contrib / contrib.sum()
                idx = rng.choice(len(i_valid), p=probs)
                # take the larger member of the pair to reduce its value
                if best[i_valid[idx]] >= best[j_valid[idx]]:
                    i = i_valid[idx]
                else:
                    i = j_valid[idx]

        # destination far away from the current peak (heuristic):
        jdest = rng.integers(0, n)
        if jdest == i:
            continue

        delta = best[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[jdest] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[jdest] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a)=2·n·max(a*a)  (equivalently minimise max(a*a) with sum=1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin so the harness does not kill us
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more expressive power)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            # honour a reasonably large existing sequence but cap at 12k
            n = min(max(tmp.size, 8000), 12000)
        else:
            n = 10000
    else:
        n = 10000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = uniform_init(n)
    else:
        best_a = uniform_init(n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (≈ 55 % of time budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=60))
    maxiter_per_stage = 600

    # Modes, plus a few arcsine variants with different α
    modes = [
        "arcsine",
        "arcsine_0.55",
        "arcsine_0.65",
        "arcsine_0.75",
        "arcsine_0.85",
        "semicircle",
        "uniform",
        "random",
        "binary",
        "bump",
    ]

    iteration = 0
    phase_deadline = start_time + 0.55 * budget_s

    while time.time() < phase_deadline:
        # every few rounds restart from a noisy version of the current best
        if iteration % 9 == 0 and iteration > 0:
            mode = "best"
            y0 = best_y + rng.normal(scale=0.025, size=n)
        else:
            mode = modes[iteration % len(modes)]
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} new best fast={best_score:.6f}")

        # a short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.90)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
            max_move=0.25,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (≈ 30 % of time budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.45,
        max_iters=600_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
            max_move=0.30,
        )
    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
            max_move=0.25,
        )
    if time.time() < deadline:
        print("[final] pairwise imbalance move")
        best_a, best_score = pairwise_imbalance_move(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
            move_frac=0.25,
        )

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # the harness provides this
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
