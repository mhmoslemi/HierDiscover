Propose a step‑function with a lower ``evaluate_sequence`` score.

This implementation builds on the previous L‑BFGS + hill‑climb approach
but adds several improvements:

* Uses a larger problem size (≈10 k) when possible – extra degrees of
  freedom give a lower theoretical bound.
* A longer, smoother continuation schedule (80 λ‑values, 1200 L‑BFGS
  iterations per stage) to approach the non‑smooth max‑convolution
  more faithfully.
* More diverse initialisations (random Gaussian, double‑exponential,
  power‑law, and a perturbed copy of the current best).
* A simple Adam‑style projected sub‑gradient phase with an adaptive
  learning‑rate schedule that often beats vanilla sub‑gradient.
* An aggressive final hill‑climb that runs as long as the overall
  time‑budget permits.

All operations are performed on the probability simplex (∑a = 1), so
the fast surrogate score ``2·n·max(conv)`` equals the true objective
when no clipping occurs.  The routine respects the supplied budget
(``budget_s``) and returns the best sequence found.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the true objective (∑a = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv
    w = exp_shift / Z

    # propagate through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that transfers a random fraction of mass from a
    source index to a destination index.  Only improvements of the fast
    score are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient method on the true max‑convolution
    objective.  Uses the sub‑gradient at the active index.
    """
    a = a_start.copy()
    m_exp = np.zeros_like(a)   # first moment
    v_exp = np.zeros_like(a)   # second moment
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        idx = np.arange(n)
        counterpart = k - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update on the *negative* sub‑gradient (we minimise)
        m_exp = beta1 * m_exp + (1 - beta1) * g
        v_exp = beta2 * v_exp + (1 - beta2) * (g * g)

        m_hat = m_exp / (1 - beta1 ** it)
        v_hat = v_exp / (1 - beta2 ** it)

        step = eta0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            # reuse length but allow a modest increase
            n = min(max(tmp.size, 8000), 12000)
        else:
            n = 10000
    else:
        n = 10000

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric grid search (Beta‑like families)
    # ------------------------------------------------------------------
    alphas = np.linspace(-0.7, 0.4, 7)
    betas = np.linspace(-0.7, 0.4, 7)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=80))
    iteration = 0
    modes = ["random", "gauss", "double_exp", "power", "edge", "best"]

    # Reserve about 250 s for the Adam‑style sub‑gradient phase and final HC
    while time.time() < deadline - 250.0:
        mode = rng.choice(modes)
        if mode == "best":
            # perturb current best
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to polish L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[adam subgrad] start")
        best_a, best_score = adam_subgradient(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[adam subgrad] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # harness supplies this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
