"""
propose_candidate
=================
Search for a non‑negative step function (list of floats) with a low
``evaluate_sequence`` score.

The strategy is a multi‑stage local search:

1. **Problem size** – use the length of the previously‑found
   construction (if at least 4000) otherwise 8000.

2. **Cheap parametric grid** – test a family a_i ∝ (i+1)^α (n‑i)^β
   on a coarse grid and keep the best vector (probability simplex).

3. **Continuation L‑BFGS** – minimise a log‑sum‑exp smooth
   approximation of the true ‑∞‑norm of the convolution.
   The smoothing factor λ is reduced geometrically (≈60 stages) and
   each stage runs a moderate L‑BFGS (≈400 iterations).  Many random
   restarts (different initial families, plus perturbations of the
   current best) are performed until most of the time budget has been
   spent.

4. **Adam‑style optimisation** – after L‑BFGS, continue minimising the
   same smooth objective with an Adam optimiser on the unconstrained
   logits y (soft‑max gives the probability vector).  The smoothing
   factor is kept very small (λ≈10⁻⁶·max) so the optimisation focuses
   on the true max‑convolution.

5. **Aggressive hill‑climb** – a greedy “move mass from i to j’’
   procedure that evaluates the exact objective after each move.
   It runs until the deadline is reached.

All vectors are kept on the probability simplex (sum = 1) so that the
final ``evaluate_sequence`` call returns exactly the fast score
(2·n·max(conv)).  The routine respects the supplied time budget and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact value of the evaluation function for a probability vector
    (sum = 1):   2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(a*a) ,  a = softmax(y),

    and grad_y is the gradient w.r.t. y.  The smoothing factor is
    λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max (log‑sum‑exp)
    shifted = conv - max_c          # max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # entries that correspond to gradient wrt a_j
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)   # (diag(a)-aaᵀ)·g_vec
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a log‑probability vector y for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # simple quadratic Sidon set (i² mod p) approximated by squares
        # use a prime p close to n
        p = int(np.floor(n ** 0.5)) ** 2 + 1
        idx = (np.arange(p) ** 2 % p)
        idx = idx[idx < n]
        k = len(idx)
        a0 = np.zeros(n)
        a0[idx] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "random_sparse":
        # pick a random support of size between 2%–20% of n
        support_frac = rng.uniform(0.02, 0.20)
        k = max(2, int(n * support_frac))
        indices = rng.choice(n, size=k, replace=False)
        vals = rng.random(k) + 0.1  # avoid exact zeros
        a0 = np.zeros(n)
        a0[indices] = vals
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    keeping the sum constant.  Accept only improvements of the exact
    fast_score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15   # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_optimize_until(
    y_start: np.ndarray,
    a_start: np.ndarray,
    init_score: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    lam_factor: float = 1e-6,
    lr: float = 0.05,
    max_check_interval: int = 500,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed objective.
    Runs until the supplied deadline.  Periodically evaluates the exact
    fast_score and keeps the best probability vector found.
    """
    y = y_start.copy()
    a = a_start.copy()
    best_a = a.copy()
    best_score = init_score

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % max_check_interval == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return y, best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the outer harness (a few seconds)
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose problem size (use a previously‑found long sequence if
    # available, otherwise 8000).
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric grid search (Beta‑like families)
    # ------------------------------------------------------------------
    alphas = np.linspace(-1.2, 0.8, 11)
    betas = np.linspace(-1.2, 0.8, 11)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-10, num=60))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "random_sparse",
        "best",
    ]
    iteration = 0

    # Reserve ~300 s for Adam + final hill‑climb
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline - 250.0, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style fine tuning (very small smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        adam_deadline = deadline - 120.0
        print("[adam] start fine‑tuning")
        best_y, best_a, best_score = adam_optimize_until(
            best_y,
            best_a,
            best_score,
            n,
            fft_len_conv,
            fft_len_g,
            adam_deadline,
            lam_factor=1e-6,
            lr=0.05,
            max_check_interval=500,
        )
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # external harness supplies this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
