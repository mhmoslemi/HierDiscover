def hill_climb_sa(a, cur_score, fft_len, n, deadline, max_iter=20000, init_temp=1e-3, final_temp=1e-6):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    temp = init_temp
    T_decay = (final_temp / init_temp) ** (1/(max_iter-1))
    for iter in range(max_iter):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # propose moving delta in [0, max_move * a[i]]
        delta = best[i] * rng.random() * 0.25
        if delta == 0: continue
        # apply move
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        # acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / temp):
            # accept
            if new_score < best_score:
                best_score = new_score
            # else keep best_score unchanged
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        temp *= T_decay
    return best, best_score

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using a real FFT.
    a must be of length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective assuming sum(a)=1:
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective and its gradient (for Adam)
# ----------------------------------------------------------------------
def smooth_obj_and_grad(y: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_g: int,
                        lam_factor: float) -> tuple[float, np.ndarray, float]:
    """
    Compute a smooth approximation of 2·n·max(conv) and its gradient
    w.r.t. the unconstrained logits y (soft‑max parametrisation).

    Parameters
    ----------
    y : np.ndarray
        Logits of length n (pre‑softmax).
    n : int
        Length of the step function.
    fft_len_conv, fft_len_g : int
        FFT lengths for the forward and reverse convolutions.
    lam_factor : float
        Multiplicative factor for the smoothing parameter:
        λ = max(conv) * lam_factor.

    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : np.ndarray
        Gradient w.r.t. y (same shape as y).
    max_c : float
        The actual maximal convolution entry (unsmoothed).
    """
    # a = softmax(y)
    a = softmax(y)

    # forward convolution a*a
    conv = conv_fft(a, fft_len_conv, n)
    max_c = conv.max()

    # smoothing parameter λ
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp approximation of max
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the gradient (∂smooth_max/∂conv)
    w = exp_shift / Z

    # reverse convolution (cross‑correlation) a_rev * w
    # a_rev = a[::-1]
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # gradient components correspond to the middle n entries
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y, max_c


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  max_iters: int,
                  deadline: float,
                  lam_start: float = 0.30,
                  lam_end: float = 1e-6) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Stochastic Adam optimiser with geometric smoothing decay.

    Returns the best logits, the corresponding probability vector and the best
    fast_score observed during the run.
    """
    y = y0.copy().astype(float)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # geometric schedule for the smoothing factor
    lam_factor = lam_start
    lam_decay = (lam_end / lam_start) ** (1.0 / max_iters)

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = y.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # evaluate smoothed objective and gradient
        f, grad_y, max_c = smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # current (unsmoothed) score and best‑track update
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = softmax(y)
            best_y = y.copy()

        # Adam step (learning rate slowly decays with sqrt(it))
        lr = 0.05 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # decay the smoothing factor
        lam_factor *= lam_decay

    return best_y, best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (works directly on the probability vector)
# ----------------------------------------------------------------------
def hill_climb_sa(a_start: np.ndarray,
                  score_start: float,
                  fft_len: int,
                  n: int,
                  deadline: float,
                  max_iter: int = 30_000,
                  init_temp: float = 1e-3,
                  final_temp: float = 1e-9,
                  max_move: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb with a simple temperature schedule.
    Occasionally accepts worsening moves to escape local minima.
    """
    best = a_start.copy()
    best_score = score_start
    rng = np.random.default_rng()

    # geometric temperature decay
    if max_iter > 1:
        temp_factor = (final_temp / init_temp) ** (1.0 / (max_iter - 1))
    else:
        temp_factor = 1.0
    temp = init_temp

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move from i to j
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        # propose move
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)

        # accept / reject
        if new_score < best_score or rng.random() < math.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # accept
            if new_score < best_score:
                best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        # cooling
        temp *= temp_factor

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain list of non‑negative floats (the sum may be arbitrary
    because ``evaluate_sequence`` rescales internally).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin inside the allotted budget
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – use a near‑power‑of‑two length for fast FFTs.
    # --------------------------------------------------------------
    default_n = 8192
    # allow the previous best construction to dictate a larger size
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 12000)

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the best candidate on the simplex (sum = 1).
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)               # logits for Adam
    print(f"[init] n={n} | start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Multiple short Adam runs (≈ 55 % of total budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s
    run_counter = 0
    while time.time() < phase1_deadline:
        run_counter += 1
        # choose a starting point: either a fresh random one or a noisy perturbation of the best
        if rng.random() < 0.4:
            y0 = rng.standard_normal(n)                      # pure random
        else:
            y0 = best_y + rng.normal(scale=0.08, size=n)     # perturb current best

        # Adam parameters – a few ten‑thousands of iterations are cheap with rFFT
        max_inner_iters = 25_000
        y_opt, a_opt, score_opt = adam_optimize(
            y0, n, fft_len_conv, fft_len_g,
            max_iters=max_inner_iters,
            deadline=phase1_deadline,
            lam_start=0.30,
            lam_end=1e-6,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[run {run_counter:04d}] improved fast score = {best_score:.6f}")

        # brief polishing by a fast greedy hill‑climb (no temperature)
        best_a, best_score = hill_climb_sa(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline=min(deadline, time.time() + 0.6),   # a short burst
            max_iter=2_000,
            init_temp=0.0,        # pure greedy
            final_temp=0.0,
            max_move=0.20,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 2 – Long simulated‑annealing hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"[final hill‑climb] remaining time ≈ {remaining:.1f}s")
        best_a, best_score = hill_climb_sa(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline=deadline,
            max_iter=200_000,
            init_temp=1e-3,
            final_temp=1e-9,
            max_move=0.30,
        )
        print(f"[final hill‑climb] final fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation – the harness will also call evaluate_sequence,
    # but we do it here for a final printed bound.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
```

This implementation follows the plan described above:
* it works on a probability simplex via a soft‑max parametrisation,
* performs many short Adam runs with a geometric smoothing schedule,
* continually improves the best candidate,
* finishes with an aggressive simulated‑annealing hill‑climb,
* and finally returns the sequence to be evaluated.
