"""
Optimisation of a non‑negative step function to minimise
    2·n·max(a*a) / (sum(a)^2)

The algorithm combines several complementary strategies:

1. **Adam‑style sub‑gradient optimisation** – many random‑restart runs on the
   true max‑convolution (the sub‑gradient is obtained from the active
   indices where the convolution attains its maximum).  Adam’s adaptive
   learning rate and bias‑correction accelerate convergence.

2. **Heavy‑L‑BFGS continuation** – a short L‑BFGS phase on a smoothed
   log‑sum‑exp surrogate, with a schedule of decreasing smoothness.
   This quickly polishes the solution obtained by Adam.

3. **Simulated annealing** – stochastic transfers of mass between random
   coordinates to escape any remaining local minima.

4. **Greedy pairwise hill‑climb** – deterministic moves that transfer a
   fraction of mass from the coordinate with the largest sub‑gradient to the
   one with the smallest, accepting only improving moves.  This finalises
   the construction.

The code respects the 1000 s time budget, prints progress, and finally
returns the best list of floats found before the deadline.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    2·n·max(a*a) when a already lies on the simplex (∑a = 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random probability vectors."""
    # Dirichlet(1) == normalized Gamma(1)
    a = rng.gamma(1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def max_conv_subgrad(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """
    Returns (max_val, sub‑gradient w.r.t. a) of the max‑convolution.
    The sub‑gradient corresponds to the sum of gradients over all active
    (max‑tight) shifts.
    """
    n = a.shape[0]
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)  # active set indicator

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Adam‑style optimisation on the true objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    max_iter: int = 10_000,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam on the non‑smooth objective 2·n·max(a*a).  The iterate is always
    projected onto the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        max_val, g = max_conv_subgrad(a, fft_len_conv, fft_len_g)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr / (np.sqrt(v_hat) + 1e-8)
        a_candidate = simplex_projection(a - step * m_hat)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS continuation on a smoothed log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).  Returns (objective, gradient w.r.t. y).
    """
    # softmax ensures a lies on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def random_anneal(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simple SA that moves a random fraction of mass between two coordinates."""
    cur_a = a.copy()
    cur_best = cur_score
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    T = 0.12
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.08
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.84
            T = max(T, 3e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise hill‑climb using the sub‑gradient
# ----------------------------------------------------------------------


def pairwise_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the index with the largest sub‑gradient
    to the one with the smallest, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # obtain the sub‑gradient of the max‑convolution
        _, g = max_conv_subgrad(best, fft_len_conv, fft_len_conv)   # reuse length
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max == i_min or best[i_max] < 1e-14:
            # random perturbation to avoid stalling
            best = add_noise(best, sigma=0.001, rng=rng)
            it += 1
            continue
        # move up to 12 % of the mass from i_max to i_min
        delta = best[i_max] * rng.random() * 0.12
        cand = best.copy()
        cand[i_max] -= delta
        cand[i_min] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best = cand
        else:
            # revert
            # (nothing to do – cand is discarded)
            pass
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    max_n = 8000                     # a larger n gives a better theoretical limit
    n = max_n

    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            old_n = init_seq.size
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    phase1_end = start_time + 0.40 * budget_s   # many Adam runs
    phase2_end = start_time + 0.65 * budget_s   # L‑BFGS refinement
    phase3_end = start_time + 0.85 * budget_s   # simulated annealing
    # remainder for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many Adam restarts
    # ------------------------------------------------------------------
    while time.time() < phase1_end:
        # random start: either a noisy perturbation of the current best
        # or a fresh Dirichlet sample
        if rng.random() < 0.35:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = adam_optimize(
            a_start,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
            max_iter=2500,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short L‑BFGS continuation with very small smoothing
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
        best_a, best_score = lbfgs_continuation(
            best_a,
            lam_schedule,
            maxiter_per_lam=300,
            deadline=phase2_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        print(f"[L‑BFGS‑fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_conv, n, phase3_end
        )
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final deterministic pairwise hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = pairwise_hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=20000
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final exact evaluation (the external function does the clipping etc.)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
