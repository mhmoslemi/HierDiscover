"""
propose_candidate
=================

Search for a non‑negative step function (list of floats) that yields a
low ``evaluate_sequence`` score.

The algorithm consists of five stages:

1. **Problem size selection** – reuse the previously found sequence if
   it is long enough, otherwise fall back to n≈5000 (a good trade‑off).

2. **Cheap parametric sweep** – try a grid of Beta‑like families
   (a_i ∝ (i+1)^α (n‑i)^β) and keep the best as a starting point.

3. **Continuation L‑BFGS** – minimise a log‑sum‑exp approximation
   of the max convolution.  The smoothing parameter λ is reduced
   geometrically over 80 stages, each stage performs up to 600 L‑BFGS
   iterations.

4. **Projected gradient descent on the true max‑convolution** – a
   sub‑gradient step is taken along the exact gradient of the current
   maximal convolution entry, with a back‑tracking line‑search and a
   projection onto the probability simplex.  This stage is the main
   source of improvement over the previous baseline.

5. **Final aggressive hill‑climb** – a simple greedy move of mass
   between two coordinates, accepting only improvements, to polish
   the result.

All stages respect the time budget (default 1000 s).  The routine
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (sum = 1):   2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective
        f = 2·n·smooth_max(a*a)   where a = softmax(y).
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                # max = 0 now
    exp_shift = np.exp(shifted / lam)     # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # propagate through convolution (a ↔ conv)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # extract gradient for each a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # abort early if we run out of time (checked outside)
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Initial distribution generators (used for random restarts)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # simple constructive Sidon‑like set
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Greedy hill‑climb (used as a polishing step)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j, accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Projected gradient descent on the true max‐convolution
# ----------------------------------------------------------------------


def maxconv_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    init_step: float = 0.8,
    min_step: float = 1e-5,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective.
    At each iteration a sub‑gradient (the derivative of the active
    convolution entry) is computed, a back‑tracking line‑search is
    performed and the result is projected back onto the simplex.
    Returns the best vector seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    step = init_step
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- compute current max‑convolution entry ---------------------------------
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # --- sub‑gradient of max(conv) ------------------------------------------------
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]   # derivative of (a*a)_m

        # --- back‑tracking line search ------------------------------------------------
        improved = False
        cur_score = fast_score(a, fft_len, n)

        # direction is negative sub‑gradient
        d = -g

        # Try a few step reductions
        trial_step = step
        while trial_step >= min_step:
            a_trial = a + trial_step * d
            a_trial = project_onto_simplex(a_trial)
            trial_score = fast_score(a_trial, fft_len, n)
            if trial_score < cur_score - 1e-12:
                improved = True
                break
            trial_step *= 0.5

        if not improved:
            # No improvement – shrink step size for future attempts
            step = max(step * 0.5, min_step)
            continue

        # accept the trial point
        a = a_trial
        cur_score = trial_score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # slowly increase step size if we succeed (helps escaping plateaus)
        step = min(step * 1.02, 1.0)

        # occasional early exit if we are within a tight tolerance
        if best_score < 1.503:
            # a score already beats the target – keep it and continue polishing later
            break

    return best_a, best_score
