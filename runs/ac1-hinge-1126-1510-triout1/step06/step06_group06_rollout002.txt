def mirror_descent(a_init, n, fft_len, deadline):
    a = a_init.copy()
    best_local = a.copy()
    best_local_score = fast_score(a, fft_len, n)
    iter = 0
    eta0 = 0.5
    g2 = np.zeros_like(a)

    while time.time() < deadline:
        iter += 1
        max_c, grad = max_convolution_subgrad(a, n, fft_len)  # grad for max conv
        # gradient for objective: 2*n*grad, but we just need direction; scale not important for mirror descent.
        grad_obj = 2.0 * n * grad
        g2 += grad_obj**2
        eta = eta0 / (np.sqrt(g2) + 1e-12)  # elementwise adaptation
        # compute update: multiply each coordinate by exp(-eta_i * grad_i)
        factor = np.exp(-eta * grad_obj)
        a = a * factor
        # renormalize onto simplex (divide by sum)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # evaluate fast score
        score = fast_score(a, fft_len, n)
        if score < best_local_score - 1e-12:
            best_local_score = score
            best_local = a.copy()
        # occasional random perturb
        if iter % 200 == 0:
            # renormalize with small noise
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
    return best_local, best_local_score

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------

1. Initialise from a stored solution (``height_sequence_1``) or a uniform
   distribution on the simplex.

2. Global optimisation:
   * Several random restarts of an L‑BFGS optimiser applied to a
     smooth‑max surrogate of the true objective.
   * After each L‑BFGS run a short targeted hill‑climb and a random
     hill‑climb are performed.

3. Refinement via a simple Adam optimiser (still on the smooth‑max)
   to escape possible plateaus left by L‑BFGS.

4. Projected sub‑gradient descent directly on the true objective
   (max auto‑convolution).  The step size decays with √iteration.

5. If time permits, up‑sample the current best sequence (linear
   interpolation, factor = 2) and run a few sub‑gradient steps on the
   larger problem – a larger dimension often yields a lower bound.

6. Final polishing hill‑climb for any remaining budget.

All moves respect the non‑negativity simplex by either a soft‑max
parameterisation or an Euclidean projection onto the simplex.
The best sequence found is stored in the global variable
``height_sequence_1`` and finally evaluated with the provided
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a)), where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Objective (smooth max) and its gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # full auto‑convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max (log‑sum‑exp trick)
    shifted = conv - max_c          # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # back‑prop to a (convolution‑type operation)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient w.r.t. logits y (soft‑max Jacobian)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Adam optimiser (still on the smooth‑max surrogate)
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_lam: int = 150,
) -> np.ndarray:
    """
    Adam on the smooth‑max objective.  The logits are updated in the
    unrestricted space; after optimisation they are turned into a
    probability vector by the soft‑max.
    """
    y = y_start.copy()
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    for lam in lam_factors:
        for _ in range(steps_per_lam):
            t += 1
            _, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor=lam
            )
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            lr = 0.02 * (np.sqrt(1 - b2**t) / (1 - b1**t))
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


# ----------------------------------------------------------------------
# Simple heuristics for generating diverse start points
# ----------------------------------------------------------------------


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point ``y`` (log‑soft‑max parameters) for L‑BFGS.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    if mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
    else:   # perturb the current best solution
        a0 = best_a
        return np.log(a0 + 1e-12) + rng.normal(scale=0.07, size=n)

    a0 /= a0.sum()
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map peak index to a shift k in the range 0…n‑1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a                     # derivative of sum_i a_i^2
    else:
        # forward contribution
        grad[: n - k] += a[k:]
        # backward contribution
        grad[k:] += a[: n - k]
    return max_c, grad


# ----------------------------------------------------------------------
# Hill‑climbing helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are retained.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the currently maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weights under the same shift
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        # pick a target index with the smallest partner weight
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best list of non‑negative heights found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Load previous construction or start from uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}  initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)   # decreasing λ‑schedule
    maxiter_per_stage = 600
    iteration = 0

    while time.time() < deadline - 200.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=next_pow2(3 * n - 2),
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_a = a_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # short hill‑climbs after each restart
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, tc_deadline, max_iter=800, move_frac=0.30
        )
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000, max_move_frac=0.20
        )

        iteration += 1

    # ------------------------------------------------------------------
    # Stage 2 – Adam on the smooth‑max surrogate (helps escape stagnation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        y_start = np.log(best_a + 1e-12)
        y_adam = adam_optimize(
            y_start, n, fft_len_conv, next_pow2(3 * n - 2), lam_factors, steps_per_lam=120
        )
        a_adam = softmax(y_adam)
        score_adam = fast_score(a_adam, fft_len_conv, n)
        if score_adam < best_score:
            best_a = a_adam
            best_score = score_adam
            print(f"[adam] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Projected sub‑gradient descent on the true max‑conv objective
    # ------------------------------------------------------------------
    subgrad_step0 = 0.6
    sub_iter = 0
    while time.time() < deadline - 200.0:
        sub_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)
        grad_obj = 2.0 * n * grad                # d/d a of 2·n·max(conv)
        step = subgrad_step0 / np.sqrt(sub_iter)
        cand = proj_simplex(best_a - step * grad_obj)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            subgrad_step0 = min(subgrad_step0 * 1.01, 1.0)
        else:
            subgrad_step0 *= 0.92

        if sub_iter % 200 == 0:
            print(f"[subgrad] iter={sub_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Upsampling (factor 2) and a few extra sub‑gradient steps
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        # linear up‑sampling by factor 2
        factor = 2
        n2 = n * factor
        x = np.arange(n)
        x_new = np.linspace(0, n - 1, n2)
        a_up = np.interp(x_new, x, best_a)
        a_up = a_up / a_up.sum()

        fft_len_conv2 = next_pow2(2 * n2 - 1)

        # a handful of sub‑gradient steps on the larger problem
        for _ in range(30):
            max_c2, grad2 = max_convolution_subgrad(a_up, n2, fft_len_conv2)
            grad_obj2 = 2.0 * n2 * grad2
            step2 = 0.5 / np.sqrt(_ + 1)
            a_up = proj_simplex(a_up - step2 * grad_obj2)

        up_score = fast_score(a_up, fft_len_conv2, n2)
        if up_score < best_score:
            best_a = a_up
            best_score = up_score
            n = n2
            fft_len_conv = fft_len_conv2
            print(f"[upsample] improved fast score = {best_score:.6f}  n={n}")
        else:
            break   # no improvement → exit up‑sampling loop

    # ------------------------------------------------------------------
    # Stage 5 – Final random hill‑climb (use any remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
