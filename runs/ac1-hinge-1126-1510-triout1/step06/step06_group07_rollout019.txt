def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # Choose n: using previous candidate length if available.
    prev = globals().get("height_sequence_1")
    if prev is not None:
        n_prev = len(prev)
        # Use same length if large enough else default 5000
        n = max(4000, n_prev)  # ensure at least 4000
    else:
        n = 5000

    # Precompute FFT length
    fft_len = 1 << ((2 * n - 2).bit_length())  # next pow2

    # Search sigma values
    mu = (n - 1) / 2.0
    # Sigma in [n/6, n/2]
    sigma_low = n / 6.0
    sigma_high = n / 2.0
    # Use 30 points grid
    sigmas = np.linspace(sigma_low, sigma_high, 30)

    best_score = float("inf")
    best_a = None

    for sigma in sigmas:
        if time.time() >= deadline:
            break
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        # compute fast_score
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a

    # Optional local refinement: simple random hill climb for few seconds
    # Use same algorithm as earlier but with limited iterations
    if best_a is not None and time.time() < deadline:
        max_iter = 20000
        rng = np.random.default_rng(seed)
        while time.time() < deadline and max_iter > 0:
            max_iter -= 1
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = best_a[i] * rng.random() * 0.1
            if delta <= 0:
                continue
            best_a[i] -= delta
            best_a[j] += delta
            # compute fast_score quickly via FFT
            fa = np.fft.fft(best_a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            new_score = float(2.0 * n * conv.max())
            if new_score < best_score - 1e-12:
                best_score = new_score
                # keep move
            else:
                best_a[i] += delta
                best_a[j] -= delta

    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Simple Gaussian‑shape construction for the step‑function.

The algorithm:
* Choose a length n (reuse a previously found sequence if available,
  otherwise use 5000).
* Scan a small set of standard deviations σ for a centred Gaussian
  a_i ∝ exp(‑½((i‑μ)/σ)²) (µ = (n‑1)/2), normalising to sum = 1.
* For each σ evaluate the cheap surrogate 2·n·max(conv) using an FFT.
* Keep the σ that gives the lowest surrogate.
* Perform a very cheap random mass‑transfer hill‑climb for the remaining
  time budget to polish the solution.
* Return the final sequence; the global name ``height_sequence_1`` is
  updated for possible reuse.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of evaluate_sequence (2·n·max(convolution))."""
    fa = np.fft.fft(a, fft_len)

Simple Gaussian‑shape construction for the step‑function.

The algorithm:
* Choose a length n (reuse a previously found sequence if available,
  otherwise use 5000).
* Scan a small set of standard deviations σ for a centred Gaussian
  a_i ∝ exp(‑½((i‑μ)/σ)²) (µ = (n‑1)/2), normalising to sum = 1.
* For each σ evaluate the cheap surrogate 2·n·max(conv) using an FFT.
* Keep the σ that gives the lowest surrogate.
* Perform a very cheap random mass‑transfer hill‑climb for the remaining
  time budget to polish the solution.
* Return the final sequence; the global name ``height_sequence_1`` is
  updated for possible reuse.

import time

# ----------------------------------------------------------------------

    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())

    fa = np.fft.fft(a, fft_len)
