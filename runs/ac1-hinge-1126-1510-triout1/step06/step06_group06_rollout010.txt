"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm (≈ 1000 s budget)
--------------------------------
1.  Initialise a probability vector `a` (sum = 1).  Several diverse
    patterns are tried (uniform, exponential, cosine, random Dirichlet).
2.  **Adam on a smoothed max** – minimise a log‑sum‑exp surrogate of
    `max(conv(a,a))` in the log‑soft‑max space `y = log(a)`.  The
    temperature is tied to the current peak via a factor `lam`,
    producing a smooth, differentiable objective.
3.  **Targeted mass‑transfer hill‑climb** – repeatedly identify the
    shift `k` that attains the maximal convolution entry, pick the
    index pair contributing most to it and move a fraction of its mass
    to a low‑partner location.
4.  **Simulated annealing** – a classic Metropolis‐Hastings walk on
    the simplex that can escape shallow local minima.
5.  **Projected sub‑gradient descent** on the true (non‑smoothed)
    `max(conv(a,a))`.  The sub‑gradient is obtained from the active
    shift and the step size follows a simple 1/√t schedule.
6.  The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s time limit by checking the deadline
before each major loop.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact objective without the sum‑square normalisation)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact 2·n·max(conv(a,a)) computed with FFT (a already sums to 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. log‑soft‑max params
# ----------------------------------------------------------------------


def smooth_max_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (smooth objective value, gradient in the `y` space).

    The surrogate is 2·n·smooth_max(conv) where smooth_max is a
    log‑sum‑exp with temperature `lam = lam_factor * max(conv)`.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = float(2.0 * n * smooth_max)

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z

    # gradient of conv wrt a (cross‑correlation with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (∂a/∂y = diag(a) - a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return f_val, grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    steps: int,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Simple Adam optimiser on the smooth surrogate.
    Returns the final `y` vector.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    for t in range(1, steps + 1):
        _, grad = smooth_max_and_grad(
            y, n, fft_len_conv, fft_len_g, lam_factor=lam_factor
        )
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Targeted hill‑climb (as in the reference implementation, a bit tweaked)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """Move mass away from the index pair that contributes most to the peak."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))  # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to pick a good target)
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Random mass‑transfer hill‑climb (simple)
# ----------------------------------------------------------------------


def random_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Purely stochastic moves that are kept only if they improve the score."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.01,
    cooling: float = 0.9995,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Metropolis walk that accepts uphill moves with a temperature schedule."""
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline and T > 1e-12:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        accept = (dE < 0) or (rng.random() < np.exp(-dE / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            cur[i] += delta
            cur[j] -= delta

        T *= cooling

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def max_convolution_subgrad(
    a: np.ndarray, n: int, fft_len: int
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient.
    Returns (max_value, gradient) where `gradient` has the same shape as `a`.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the half‑interval 0 ≤ k ≤ n-1 (symmetry)
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        # derivative of sum_i a_i^2
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


def subgradient_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.45,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with a 1/√t step‑size schedule."""
    cur = a.copy()
    cur_score = fast_score(cur, fft_len, n)
    step0 = init_step
    it = 0
    eps = 1e-12

    while time.time() < deadline:
        it += 1
        max_c, grad = max_convolution_subgrad(cur, n, fft_len)

        step = step0 / np.sqrt(it)
        cand = proj_simplex(cur - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < cur_score - eps:
            cur = cand
            cur_score = cand_score
            step0 = min(step0 * 1.02, 1.0)  # slowly increase when progress is made
        else:
            step0 *= 0.92  # shrink step if we are stuck

    return cur, cur_score


# ----------------------------------------------------------------------
# Starting‑point generators
# ----------------------------------------------------------------------


def make_initial_y(rng: np.random.Generator, n: int, mode: str) -> np.ndarray:
    """Return log‑soft‑max parameters for a given heuristic."""
    if mode == "uniform":
        a0 = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
    elif mode == "cos":
        x = np.linspace(0, np.pi, n)
        a0 = (np.cos(x) + 1) / 2.0
        a0 /= a0.sum()
    elif mode == "rand":
        a0 = rng.random(n)
        a0 /= a0.sum()
    else:  # fallback
        a0 = np.full(n, 1.0 / n)
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops a few seconds before
        the hard limit to avoid termination.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size & initial vector
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = np.maximum(arr, 0.0)
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n}  initial fast score = {best_score:.6f}")

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Main optimisation loop (alternating phases)
    # ------------------------------------------------------------------
    iteration = 0
    lam_schedule = np.geomspace(0.5, 1e-7, num=38)   # smoother than before

    while time.time() < deadline - 5.0:
        # ----- pick a random start pattern for this outer iteration -----
        mode = rng.choice(["uniform", "exp", "cos", "rand"])
        y0 = make_initial_y(rng, n, mode)

        # ----- Adam on smoothed max -----
        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factor=0.25,
            steps=1800,
            lr=0.03,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            print(f"[{iteration}] Adam ({mode}) improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb (short burst) -----
        tc_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, tc_dead, max_iter=1200, move_frac=0.35
        )
        print(f"[{iteration}] after targeted hill‑climb score = {best_score:.6f}")

        # ----- random hill‑climb (escape plateaus) -----
        rc_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = random_hill_climb(
            best_a, best_score, fft_len_conv, n, rc_dead, max_iter=4000, max_move_frac=0.22
        )
        print(f"[{iteration}] after random hill‑climb score = {best_score:.6f}")

        # ----- simulated annealing (exploration) -----
        sa_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, sa_dead, init_temp=0.015, cooling=0.9993
        )
        print(f"[{iteration}] after SA score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the true objective (fine tuning)
    # ------------------------------------------------------------------
    subgrad_dead = deadline - 1.5
    if time.time() < subgrad_dead:
        best_a, best_score = subgradient_descent(
            best_a, n, fft_len_conv, subgrad_dead, init_step=0.45
        )
        print(f"[subgrad] final sub‑gradient score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a longer random hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
            max_move_frac=0.28,
        )
        print(f"[final hill‑climb] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation & return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
