"""
propose_candidate
=================

A multi‑stage optimiser for the “minimise max(conv) / (sum)²’’ problem.
The pipeline is:

1. **Choose a problem size** – reuse a previously discovered sequence if
   available, otherwise start from 5 000 items and upscale (≤12 000) to
   obtain a better theoretical lower bound.

2. **Fast parametric sweep** – a dense β‑type family (powers of the index)
   plus a few smooth analytical shapes (sin², Gaussian).  The best
   candidate becomes the starting point for the heavier optimisation.

3. **Continuation L‑BFGS** – minimise a log‑sum‑exp (smooth‑max)
   approximation of the true objective.  The smoothing factor λ is reduced
   geometrically (≈30 stages).  Each stage performs a modest number of
   L‑BFGS iterations (maxiter≈300).  A handful of random restarts
   (including perturbations of the current best) explore the landscape.

4. **Adam‑style projected sub‑gradient descent** – optimisation of the
   exact non‑smooth objective (max‑convolution) on the probability simplex.
   The sub‑gradient is computed analytically in O(n) and the update is
   performed on the pre‑soft‑max vector y with Adam momentum.

5. **Final greedy hill‑climb** – a fast local search that repeatedly moves
   a random fraction of mass from one index to another, accepting only
   improvements of the true fast score.

All helper routines are top‑level, use only ``numpy``/``scipy`` and respect
the time budget (default 1000 s).  The returned list is a non‑negative step
function; its scale does not affect the score because the evaluation
function is scale‑invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, yields a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) – the objective when a is a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ---- propagate gradient through the convolution (a*a) ----------------
    #   w.r.t a: derivative of a*a is the convolution of a_rev with w.
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "hanning":
        a0 = np.sin(np.pi * np.arange(n) / (n - 1)) ** 2
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "gaussian":
        sigma = rng.uniform(0.1, 0.4) * n
        x = np.arange(n)
        mu = (n - 1) / 2.0
        a0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i → j,
    accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_subgradient_refine(
    y_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    lr: float = 0.20,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the exact max‑convolution objective.
    Operates on the pre‑soft‑max vector y; the simplex constraint is
    enforced automatically by the soft‑max transform.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and active index
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient w.r.t a (ordered pairs) – O(n)
        g_a = np.zeros(n)
        low = max(0, m_idx - n + 1)
        high = min(m_idx, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g_a[idx] = 2.0 * a[m_idx - idx]

        # gradient w.r.t y via soft‑max Jacobian
        mean_g = np.dot(a, g_a)
        grad_y = a * (g_a - mean_g)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the current best
        cur_a = softmax(y)
        cur_score = fast_score(cur_a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

    return best_a, best_score

    n = target_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            # upsample / interpolate to the new size
            old_x = np.linspace(0, 1, arr.size)
            new_x = np.linspace(0, 1, n)
            best_a = np.interp(new_x, old_x, arr)
            best_a = np.maximum(best_a, 0.0)
            if best_a.sum() == 0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a /= best_a.sum()
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Expanded parametric grid search (Beta‑like families + smooth shapes)
    # ------------------------------------------------------------------
    alphas = np.linspace(-0.9, 0.6, 11)
    betas = np.linspace(-0.9, 0.6, 11)

    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            s = prob.sum()
            if s <= 0:
                continue
            prob = prob / s
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # sine‑squared family
    for pwr in [1, 2, 3]:
        a0 = np.sin(np.pi * np.arange(n) / (n - 1)) ** pwr
        a0 = a0 / a0.sum()
        sc = fast_score(a0, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[sine] power={pwr} → fast_score = {sc:.6f}")

    # Gaussian‑shaped families
    for sigma_factor in [0.15, 0.25, 0.35]:
        x = np.arange(n)
        mu = (n - 1) / 2.0
        sigma = sigma_factor * n
        a0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        a0 = a0 / a0.sum()
        sc = fast_score(a0, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[gauss] σ_factor={sigma_factor:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (random restarts + perturbations)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=30))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "best",
        "hanning",
        "gaussian",
    ]

    iteration = 0
    # reserve ~250 s for the final Adam + hill‑climb phases
    reserve_refine = 250.0

    while time.time() < deadline - reserve_refine:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short greedy hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline - reserve_refine, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style sub‑gradient refinement on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[adam] start")
        best_a, best_score = adam_subgradient_refine(
            best_y,
            n,
            fft_len_conv,
            sub_deadline,
            lr=0.20,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=10000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
