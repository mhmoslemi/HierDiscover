"""
Propose a low‑score non‑negative step‑function sequence.

The algorithm is a parallel‑friendly, multi‑restart optimiser:
* Work on the log‑soft‑max parameters `y` (so the probabilities `a`
  automatically stay on the simplex).
* Global optimisation:
    - Continuation L‑BFGS on a smooth approximation of the max‑convolution.
    - A finer λ‑schedule and slightly more iterations per continuation stage
      than the reference implementation.
* Local refinement:
    - Targeted hill‑climb that moves mass away from the shift where the
      convolution attains its maximum.
    - Random hill‑climb for escaping plateaus.
    - Projected sub‑gradient descent on the true (unsmoothed) max.
* The search runs two independent “workers” sequentially (no OS processes,
  safe on all platforms) and keeps the best result.
* Finally the best vector is evaluated with the provided `evaluate_sequence`
  routine and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective value when `a` is already normalised (sum = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Return a smoothed approximation of 2·n·max(conv(a,a))
    together with the gradient w.r.t. the log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # full auto‑convolution (linear, not circular)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # λ is a tiny fraction of the current maximum; the +1e‑12 term prevents
    # division by zero when max_c is extremely small.
    lam = max_c * lam_factor + 1e-12

    # smooth‑max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth‑max w.r.t. conv entries
    w = exp_shift / Z  # shape (2n‑1,)

    # back‑propagation through convolution:
    # g_i = Σ_j a_j * w_{i+j}   (i = 0..n‑1)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (Jacobian = diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 1000,
) -> np.ndarray:
    """
    Perform a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        # L‑BFGS with exact gradient
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Accepts only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that attacks the shift attaining the current maximum.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a destination with smallest partner weight under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        # sample a handful of candidates to keep the move cheap
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding sub‑gradient (2·a shifted) and takes a projected step.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        iter_no += 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t a
        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        step = step0 / np.sqrt(iter_no)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance to escape flat regions
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Upscaling helper (optional, used when a previous construction exists)
# ----------------------------------------------------------------------


def upscale_distribution(old_a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = old_a.shape[0]
    if new_n == old_n:
        return old_a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, old_a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a = np.full(new_n, 1.0 / new_n)
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Allowed runtime in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------
    # Choose problem size – reuse previous construction if possible
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            # upscale to a slightly larger problem to gain more flexibility
            target_n = min(8000, max(5000, int(arr.size * 1.5)))
            a = upscale_distribution(arr, target_n)
            n = target_n
        else:
            n = 8000
            a = np.full(n, 1.0 / n)
    else:
        n = 8000
        a = np.full(n, 1.0 / n)

    # Normalise onto the simplex
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_a = a
    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT size (constant throughout the run)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # schedule for the smoothing parameter λ
    lam_factors = np.geomspace(0.5, 1e-12, num=60)   # finer schedule than before
    maxiter_per_stage = 1000                         # a bit more work per stage

    iteration = 0
    deadline = time.time() + budget_s - 4.0  # reserve a few seconds for final eval

    # ------------------------------------------------------------------
    # Main optimisation loop – continue until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # pick an initialisation strategy
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.03)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.06, size=n)

        # ----- global optimisation via continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}")

        # ----- local refinements -----
        # Targeted hill‑climb (aggressive)
        t_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1500,
            move_frac=0.35,
        )
        best_y = np.log(best_a + 1e-12)

        # Random hill‑climb (helps escape plateaus)
        r_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=10000,
            max_move_frac=0.23,
        )
        best_y = np.log(best_a + 1e-12)

        # Short sub‑gradient descent (exact max)
        s_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=s_dead,
            max_iter=2000,
            step0=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final refinement phases (use up remaining budget)
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 2.0
    best_a, best_score = subgrad_descent(
        a=best_a,
        cur_score=best_score,
        n=n,
        fft_len=fft_len_conv,
        deadline=subgrad_deadline,
        max_iter=8000,
        step0=0.25,
    )
    best_y = np.log(best_a + 1e-12)

    # Final aggressive random hill‑climb using any leftover time
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
