def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=20000):
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute convolution
        conv = full_conv(best_a, fft_len)
        # find index of max
        k_max = int(conv.argmax())
        # compute partner index for each i
        # contributions: contrib_i = a[i] * a[k_max - i] if in range, else 0
        # Find i with largest contribution
        # use vectorized operations
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(best_a)
        contrib[mask] = best_a[i[mask]] * best_a[j[mask]]
        i_max = int(np.argmax(contrib))
        j_max = int(k_max - i_max)
        # choose a target index p far from both i_max and j_max (e.g., random)
        p = rng.integers(0, n)
        while p == i_max or p == j_max:
            p = rng.integers(0, n)
        # transfer small fraction from i_max to p
        delta = best_a[i_max] * rng.random() * 0.05  # 0-5% of mass at i_max
        if delta <= 0.0:
            it += 1
            continue
        cand = best_a.copy()
        cand[i_max] -= delta
        cand[p] += delta
        # optionally also move some from j_max if distinct and delta2>0
        if j_max != i_max:
            delta2 = best_a[j_max] * rng.random() * 0.05
            cand[j_max] -= delta2
            cand[p] += delta2
        # ensure nonnegative and normalize (but sum unchanged)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            it += 1
            continue
        cand = cand / s
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = cand
        it += 1
    return best_a, best_score

"""
Optimise a non‑negative step sequence so that the value
    2·n·max(conv(a,a)) / (∑a)²
is as small as possible.  The algorithm is a
pipeline of many cheap restarts followed by
refinements:

1.  Generate a diverse pool of seed distributions (uniform,
    exponential, power‑law, arcsine‑like, sinusoidal and the
    previous best if available).

2.  For each seed run a continuation of L‑BFGS on a
    smooth‐log‑sum‑exp approximation of the max‑convolution.
    The smoothing parameter λ is decreased in stages.

3.  Refine the current best with a projected sub‑gradient method
    (momentum) and an exponentiated‑gradient (mirror descent)
    on the simplex.

4.  Perform a very fine L‑BFGS run with tiny λ.

5.  Targeted hill‑climb: repeatedly locate the index k where the
    convolution attains its maximum, pick the pair (i,j) that
    contributes most to this value and move a small fraction of
    probability mass away from i (and j) towards a random location.
    Only strictly improving moves are kept.

6.  Return the best sequence found before the deadline.
   The returned list is evaluated with the provided
   ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1).  Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (ensures positivity and sum‑to‑one)
    a = np.exp(y - np.max(y))
    a = a / a.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12               # smoothing temperature
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                               # d smooth_max / d conv_i

    # back‑propagation through convolution (a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a (only the central n entries are needed)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.  Returns the best a
    found and its fast_score.
    """
    y = np.log(a_start + 1e-12)                       # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover – extremely rare
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum (non‑smooth max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # active set

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step‑size 0.7 / √it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror‑descent on the simplex)
    on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape plateaus
        if (it + 1) % 5000 == 0:
            if rng.random() < 0.3:
                sigma = rng.uniform(0.001, 0.008)
                a = a + rng.normal(0.0, sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine (Beta(0.5,0.5)) shape – strong mass near the ends."""
    i = np.arange(n)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-12)
    a /= a.sum()
    return a


def sinusoidal_initial(n: int) -> np.ndarray:
    """Half‑wave sinusoid – larger in the centre, zero at the ends."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def sin_squared_initial(n: int) -> np.ndarray:
    """Sin² shape – still positive, but flatter than the pure sine."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Targeted hill‑climb that looks at the active convolution index
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Greedy improvement that always moves probability mass away from the
    pair of coordinates that contributes most to the current maximal
    convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution
        conv = full_conv(best, fft_len)
        k_max = int(conv.argmax())

        # contributions to the maximal entry
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[i[mask]] * best[j[mask]]

        i_max = int(np.argmax(contrib))
        j_max = int(k_max - i_max)

        # pick a random target far from i_max/j_max
        p = rng.integers(0, n)
        while p == i_max or p == j_max:
            p = rng.integers(0, n)

        # move a small fraction of mass from i_max (and possibly j_max) to p
        delta_i = best[i_max] * rng.random() * 0.06  # ≤6 % of that coordinate
        if delta_i <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i_max] -= delta_i
        cand[p] += delta_i

        if j_max != i_max:
            delta_j = best[j_max] * rng.random() * 0.06
            cand[j_max] -= delta_j
            cand[p] += delta_j

        # re‑project onto the simplex (renormalise)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            it += 1
            continue
        cand /= s

        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
            best = cand
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000                       # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # start from a uniform distribution (baseline)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Assemble a pool of promising starts
    # ------------------------------------------------------------------
    init_pool = []

    # a few handcrafted shapes
    init_pool.append(arcsine_initial(n))
    init_pool.append(sinusoidal_initial(n))
    init_pool.append(sin_squared_initial(n))

    # several exponential / power‑law variants
    for rate in [0.001, 0.003, 0.008]:
        exp_shape = np.exp(-rate * np.arange(n))
        init_pool.append(exp_shape / exp_shape.sum())
    for p in [0.5, 1.0, 1.5, 2.0]:
        power = (np.arange(1, n + 1, dtype=float) ** (-p))
        init_pool.append(power / power.sum())

    # a handful of completely random starts
    for _ in range(7):
        init_pool.append(random_initial(n, rng))

    # if a previous construction is available, add it (re‑normalised)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                init_pool.append(prev / s)

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS continuation runs with decreasing λ
    # ------------------------------------------------------------------
    lam_schedule = [
        0.70, 0.50, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 600

    for a_start in init_pool:
        if time.time() >= deadline:
            break
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[LBFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑grad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑grad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – very fine L‑BFGS run (tiny λ)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam = [0.00008, 0.00004, 0.00002, 0.00001]
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam,
            300,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = best_a.copy()
            print(f"[fine‑LBFGS] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – targeted hill‑climb exploiting the active convolution index
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline
        )
        print(f"[targeted‑HC] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
