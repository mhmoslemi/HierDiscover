def annealing_hill_climb(a, cur_score, fft_len, n, deadline):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    temp0 = 0.1  # initial temperature relative to cur_score
    while time.time() < deadline:
        t = (deadline - time.time()) / (deadline - start)  # proportion left
        temp = temp0 * t
        i,j = rng.integers(0,n,2)
        if i==j: continue
        delta = best[i] * rng.random() * 0.1
        if delta <= 0: continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score or np.exp(-(new_score-best_score)/temp) > rng.random():
            if new_score < best_score:
                best_score = new_score
            # accept - keep new config
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best,best_score

propose_candidate
=================

Search for a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.  The algorithm combines several ideas that
worked well in earlier attempts and adds a few refinements:

1. **Extended parametric grid search**  
   Try many Beta‑like families a_i ∝ (i+1)^α (n‑i)^β with a
   much wider range of (α,β) than before.  The best result becomes the
   start point for the subsequent optimisation.

2. **Continuation L‑BFGS on a log‑sum‑exp smoothed max**  
   A schedule of decreasing smoothing factors (λ) is used; each stage
   runs a fairly long L‑BFGS optimisation.  Random restarts from a few
   heuristic seeds (exponential, edge‑heavy, arcsine, Sidon‑set,
   plus perturbations of the current best) are performed while the
   time budget permits.

3. **Projected sub‑gradient descent with momentum**  
   After the L‑BFGS phase a sub‑gradient method on the true max‑conv
   objective (the exact evaluation used in the harness) is run with an
   adaptive step‑size and Nesterov‑style momentum to polish the result.

4. **Aggressive hill‑climb + final random‑perturbation sweep**  
   A greedy hill‑climb moves a random fraction of the mass between two
   coordinates and keeps improvements.  When time is almost exhausted
   a short random‑perturbation search is performed to escape possible
   tiny local minima.

All helpers are defined at top‑level (no closures or lambdas) and only
``numpy``/``scipy`` are used.  The routine respects the supplied time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater-or-equal to *x* (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(a*a)   (a = softmax(y))

    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    # Compute g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)        # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return an initial pre‑soft‑max vector for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # simple greedy Sidon set construction
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = max(1, len(chosen))
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "best":
        # placeholder, caller will replace with a perturbation of the current best
        raise RuntimeError("mode 'best' should be handled by the caller")

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j,
    accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    momentum: float = 0.9,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = max_i (a*a)_i .
    Uses Nesterov‑style momentum to accelerate convergence.
    """
    a = a_start.copy()
    vel = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max‑conv at the active index m
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Nesterov momentum update
        eta = eta0 / np.sqrt(it)
        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def random_perturbation_search(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    attempts: int = 2000,
    sigma: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Tiny random perturbations around *a*; keep any improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and attempts > 0:
        attempts -= 1
        cand = best + rng.normal(scale=sigma, size=n)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluate_sequence score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # safety margin for the outer harness
    deadline = time.time() + budget_s - 1.5

    # ------------------------------------------------------------------
    # Choose problem size.  Prefer a longer vector when a previous one
    # exists, otherwise use 8000 as a reasonable compromise.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    desired_n = 8000
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            desired_n = min(max(tmp.size, 5000), 10000)

    n = desired_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            # if the previous sequence has a different length, just start uniform
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Extended parametric grid search (Beta‑like families)
    # ------------------------------------------------------------------
    alphas = np.linspace(-2.0, 1.0, 31)   # step ≈0.1
    betas = np.linspace(-2.0, 1.0, 31)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            # avoid NaNs from negative powers on zero; all entries >=0
            prob = np.maximum(prob, 0.0)
            prob_sum = prob.sum()
            if prob_sum == 0.0:
                continue
            prob /= prob_sum
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # also test the explicit arcsine shape if it wasn't covered
    i = np.arange(n, dtype=float)
    a_arc = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    a_arc = a_arc / a_arc.sum()
    sc_arc = fast_score(a_arc, fft_len_conv, n)
    if sc_arc < best_score:
        best_score = sc_arc
        best_a = a_arc.copy()
        best_y = np.log(best_a + 1e-12)
        print(f"[grid] arcsine → fast_score = {sc_arc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    # Reserve ~250 s for the sub‑gradient / final polishing stages
    while time.time() < deadline - 250.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=900,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode:<7} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Refine with projected sub‑gradient descent (with momentum)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.20,
            momentum=0.9,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb + random perturbation sweep
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        # final random tweaks
        best_a, best_score = random_perturbation_search(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            attempts=3000,
            sigma=0.002,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # harness supplies this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
