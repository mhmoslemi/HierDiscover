"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
1.  Initialise a probability vector (∑a = 1) – reuse the previously
    found best sequence if it exists, otherwise start from a uniform
    or a simple parametric shape.
2.  **Random parametric search** – generate many candidates from a few
    families (exponential, power‑law, edge‑plateau, mixed) and keep the
    best according to a fast surrogate
    ``fast_score = 2·n·max(conv)`` (the exact metric when ∑a = 1).
3.  **Continuation L‑BFGS** on the unconstrained log‑soft‑max variables.
    The smooth‑max surrogate of ``max(conv)`` is used; the smoothing
    parameter λ is decreased geometrically (50 stages).  Each stage
    performs a few hundred L‑BFGS iterations.
4.  **Local refinements**
    * targeted hill‑climb – move mass from indices that contribute most
      to the maximal convolution entry to indices that contribute the
      least;
    * random hill‑climb – cheap mass‑transfer moves with acceptance
      only if the surrogate score improves;
    * sub‑gradient descent – exact max‑convolution gradient with Euclidean
      projection onto the probability simplex.
5.  Use the remaining time for an aggressive random hill‑climb.
6.  Return the best sequence (as a plain ``list[float]``) and store it
    globally as ``height_sequence_1`` for possible reuse by later runs.
   The final sequence is evaluated with the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:                     # numerical fallback
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a probability vector.
    Returns 2·n·max(convolution(a,a)).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12  # smoothing width

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: repeatedly minimise the smooth‑max objective
    while decreasing the smoothing factor λ.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14,
                     "disp": False, "maxcor": 10},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # full convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose a destination with a tiny partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding gradient, takes a step opposite to it and projects back onto
    the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k: g_i = 2 * a_{k-i} (if the partner index exists)
        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        # diminishing step size
        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (∑ = 1).
    """
    rng = np.random.default_rng(seed)
    # Reserve a tiny safety margin for final evaluation
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    #  Initialise problem size and starting vector
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000                     # default size
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    #  Pre‑compute FFT lengths used throughout the run
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    #  Phase 1 – random parametric search (cheap, wide exploration)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        mode = rng.choice(["exp", "power", "edge", "mix"])
        if mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            dist = np.arange(n)
            a0 = np.exp(-rate * np.minimum(dist, n - 1 - dist))
        elif mode == "power":
            p = rng.uniform(0.2, 3.0)
            dist = np.arange(n)
            a0 = (np.minimum(dist, n - 1 - dist) + 1.0) ** (-p)
        elif mode == "edge":
            # edge‑plateau shape, guaranteed non‑negative
            w_frac = rng.uniform(0.01, 0.07)
            w = max(1, int(w_frac * n))
            factor = rng.uniform(2.0, 6.0)               # edge / interior ratio
            interior = 1.0 / (2 * w * factor + (n - 2 * w))
            edge = factor * interior
            a0 = np.full(n, interior)
            a0[:w] = edge
            a0[-w:] = edge
        else:  # "mix"
            rate = rng.uniform(0.001, 0.02)
            exp_shape = np.exp(-rate * np.minimum(np.arange(n),
                                                   n - 1 - np.arange(n)))
            w_frac = rng.uniform(0.01, 0.04)
            w = max(1, int(w_frac * n))
            boost = rng.uniform(1.5, 3.0)
            a0 = exp_shape.copy()
            a0[:w] *= boost
            a0[-w:] *= boost

        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        score = fast_score(a0, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a0
            best_y = np.log(best_a + 1e-12)
            print(f"[rand] new best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    #  Phase 2 – continuation L‑BFGS with occasional local refinements
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-12, num=50)   # fine λ‑schedule
    maxiter_per_stage = 1200                         # more work per stage

    iteration = 0
    while time.time() < deadline - 15.0:
        # pick a diverse starting point
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.minimum(np.arange(n),
                                            n - 1 - np.arange(n)))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            w_frac = rng.uniform(0.01, 0.04)
            w = max(1, int(w_frac * n))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:w] = edge_val
            a0[-w:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.08, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} → new best fast_score = {best_score:.6f}")

        # ----- targeted hill‑climb (quick, aggressive) -----
        t_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- random hill‑climb (helps escape tiny plateaus) -----
        r_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=8000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    #  Phase 3 – exact sub‑gradient descent (fine polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        sub_dead = deadline - 2.0
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=sub_dead,
            max_iter=8000,
            step0=0.35,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] post‑subgrad fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    #  Phase 4 – final aggressive random hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=30000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    #  Return the sequence and evaluate it with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
