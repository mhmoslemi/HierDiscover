"""
Propose a step function minimizing 2*n*max(conv) / sum^2.
Algorithm:
- Choose a moderate problem size (~8000).
- Initialise from several analytic shapes.
- Run a continuation L‑BFGS on a smoothed max‑convolution objective.
  Use a finer λ‑schedule and more L‑BFGS iterations (500 per stage). 
- Perform several random restarts (perturb best solution and re‑optimise).
- Refine with a simple simulated‑annealing routine that moves small
  amounts of mass between indices while occasionally accepting
  uphill moves.
- Finish with fast greedy hill‑climbs (random and targeted).
All steps respect a 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a

def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a

def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a

def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)
    y0 = np.log(a0 + 1e-12)
    return y0

def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                 # derivative of smooth_max w.r.t conv entries

    # gradient of weighted sum of convolutions w.r.t a
    # reverse a for the cross‑correlation
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]      # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(y_start: np.ndarray,
                     n: int,
                     fft_len_conv: int,
                     fft_len_g: int,
                     lam_schedule: list[float],
                     maxiter_per_stage: int = 500):
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000,
               max_move: float = 0.20):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 3000,
                        max_move: float = 0.25):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]
        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def simulated_annealing(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_move: float = 0.20):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    # temperature schedule – exponential decay
    T_start = best_score * 0.01
    T_end = best_score * 1e-6
    T = T_start
    decay = 0.998
    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        d = new_score - cur_score
        if d < 0.0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        # temperature decay
        T = max(T * decay, T_end)
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score 2·n·max(conv) / (∑a)^2.
    The algorithm combines a continuation L‑BFGS phase, random restarts,
    simulated annealing and final greedy hill‑climbs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size.  Using a larger n gives more freedom.
    # ------------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise best candidate – use uniform if no external hint.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)

            best_y = y_opt.copy()

                  f"improved fast score = {best_score:.6f}")

        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500,
                                        max_move=0.25)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # Simulated annealing (≈ 45 % of the budget)
    # ------------------------------------------------------------------

    print("[anneal] starting simulated annealing")

                                             max_move=0.25)
    best_y = np.log(best_a + 1e-12)
    print(f"[anneal] finished with fast score = {best_score:.6f}")

        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=20000,
                                        max_move=0.30)
        if time.time() < deadline:

                                                     best_score,

                                                     n,
                                                     deadline,
                                                     max_iter=10000,
                                                     max_move=0.20)

    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness

"""
Propose a step‑function (list of non‑negative floats) that minimises

    2·n·max(conv) / (∑a)²

where ``conv`` is the linear self‑convolution of the sequence.
The algorithm combines:

* a continuation L‑BFGS phase on a log‑sum‑exp smooth approximation of the
  max‑convolution,
* several random restarts and perturbations of the current best,
* a simulated‑annealing stage that moves small amounts of mass between
  indices while occasionally accepting uphill moves,
* final greedy hill‑climbs (random and targeted) to polish the result.

All steps respect the 1000 s budget.  The returned list is ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    a is length n, fft_len ≥ 2·n‑1.
    Returns the (2·n‑1)‑long real convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a normalised probability vector
    (∑a = 1): 2·n·max(conv).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, ∑x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initial shapes
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a log‑softmax starting point for the chosen analytic shape.
    """
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)

    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max‑convolution objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained log‑softmax parameters y.
    """
    a = softmax(y)

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- log‑sum‑exp smoothing -------------------------------------------
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth_max w.r.t. conv entries -----------------------
    w = exp_shift / Z                          # ∂smooth_max/∂conv

    # --- gradient of Σ w_k·conv_k w.r.t. a (reverse convolution) ---------
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]    # size n

    # --- chain rule through softmax --------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                     n: int,
                     fft_len_conv: int,
                     fft_len_g: int,
                     lam_schedule: list[float],
                     maxiter_per_stage: int = 500):
    """
    Continuation L‑BFGS: run a short L‑BFGS optimisation for each λ
    in the schedule, starting each stage from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000,
               max_move: float = 0.20):
    """
    Random greedy hill‑climb: move a random fraction of mass from i to j.
    Keep the move only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 3000,
                        max_move: float = 0.25):
    """
    Hill‑climb that preferentially moves mass away from indices that
    actively contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def simulated_annealing(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_move: float = 0.20):
    """
    Simple simulated‑annealing on the true objective.
    Small amounts of mass are moved between two sites; uphill moves are
    occasionally accepted according to an exponential cooling schedule.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score

    rng = np.random.default_rng()
    T_start = best_score * 0.01
    T_end = best_score * 1e-6
    T = T_start
    decay = 0.998

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        d = new_score - cur_score
        if d < 0.0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        T = max(T * decay, T_end)

    return best, best_score

                                  fft_len_conv,
                                  fft_len_g,
                                  lam_schedule,
                                  maxiter_per_stage=maxiter_per_stage)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode_name} improved fast score = {best_score:.6f}")

        # Short random hill‑climb to polish the L‑BFGS output.
        hc_deadline = min(deadline, time.time() + 0.80)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500,
                                        max_move=0.25)
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Simulated annealing phase (≈ 45 % of the budget)
    # --------------------------------------------------------------
    anneal_deadline = min(deadline - 5.0, start_time + 0.80 * budget_s)
    print("[anneal] starting simulated annealing")
    best_a, best_score = simulated_annealing(best_a,
                                             best_score,
                                             fft_len_conv,
                                             n,
                                             anneal_deadline,
                                             max_move=0.25)
    best_y = np.log(best_a + 1e-12)
    print(f"[anneal] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final greedy polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=20000,
                                        max_move=0.30)
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(best_a,
                                                     best_score,
                                                     fft_len_conv,
                                                     n,
                                                     deadline,
                                                     max_iter=10000,
                                                     max_move=0.20)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
