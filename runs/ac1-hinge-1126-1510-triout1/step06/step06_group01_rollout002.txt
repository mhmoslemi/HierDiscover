"""
Propose a low‑evaluation step sequence.

Algorithm outline
-----------------
1. Choose a set of problem sizes (n = 512, 1024, 2048, 4000).  Shorter
   sequences are much cheaper to optimise, so we can afford many
   random restarts on them and later keep the best shape found for any
   n.

2. For each n
   * start from a portfolio of hand‑crafted deterministic shapes
     (uniform, triangular, quadratic, a few random Beta‑like profiles)
     and keep the best one according to the fast objective
       fast_score = 2·n·max(conv(a,a)).
   * Run a *continuation* L‑BFGS on a log‑softmax parametrisation.
     The continuation schedule slowly reduces the smoothing factor
     (lam_factor in the log‑sum‑exp approximation of the max) from
     0.6 down to 2·10⁻⁴, solving many stages with many iterations.
     Random restarts are performed while time permits.
   * Refine the result with a long projected sub‑gradient descent
     (with momentum) directly on the true max‑convolution.
   * Further improve with exponentiated‑gradient descent on the simplex.
   * Apply a short “smooth‑and‑re‑project’’ filter (3‑point kernel) a few
     times – this often reduces the convolution peak without harming the
     sum‑normalisation.
   * Finish with a greedy hill‑climb that moves a tiny fraction of mass
     between two coordinates only when it strictly improves the fast
     score.

3. Keep the globally best sequence across all n and return it.
   The final sequence is evaluated with the provided ``evaluate_sequence``
   to make sure it is admissible.

All operations are performed with FFTs (O(n log n)) and the vector is
kept on the probability simplex (non‑negative, sum = 1), which makes the
objective scale‑invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a already lies on the simplex.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each smoothing factor (lam_factor) in lam_schedule.
    Each stage starts from the result of the previous one.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Very rare; skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum directly on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.22 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape plateaus
        if (it + 1) % 4000 == 0:
            if rng.random() < 0.35:
                sigma = rng.uniform(0.001, 0.007)
                a = a + rng.normal(0.0, sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 10000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: transfer a small random fraction of mass between two
    coordinates, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.09   # up to 9 % of the mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def deterministic_shapes(n: int) -> list[np.ndarray]:
    """Return a list of a few hand‑crafted shapes for a given length."""
    shapes = []

    # uniform
    shapes.append(np.full(n, 1.0 / n))

    # triangular (peak in the centre)
    t = np.arange(1, n // 2 + 1)
    if n % 2 == 0:
        tri = np.concatenate([t, t[::-1]])
    else:
        tri = np.concatenate([t, [n // 2 + 1], t[::-1]])
    tri = tri.astype(float)
    tri /= tri.sum()
    shapes.append(tri)

    # quadratic (parabolic)
    q = np.arange(1, n + 1) * np.arange(n, 0, -1)
    q = q.astype(float)
    q /= q.sum()
    shapes.append(q)

    # sin‑shaped (half‑sine)
    x = np.arange(n) + 0.5
    s = np.sin(np.pi * x / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    shapes.append(s)

    # a few random Beta‑like profiles
    rng = np.random.default_rng()
    for _ in range(4):
        alpha = rng.uniform(0.2, 3.0)
        beta = rng.uniform(0.2, 3.0)
        i = np.arange(1, n + 1)
        b = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
        b = b.astype(float)
        b /= b.sum()
        shapes.append(b)

    return shapes

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # initialise RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes we will explore (shorter sizes are cheaper)
    # ------------------------------------------------------------------
    size_candidates = [512, 1024, 2048, 4000]

    best_seq = None
    best_score = float("inf")

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    for n in size_candidates:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising for n = {n} ===")
        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_g = next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Phase 0 – deterministic seeds
        # ------------------------------------------------------------------
        local_best_a = None
        local_best_score = float("inf")
        for shape in deterministic_shapes(n):
            sc = fast_score(shape, fft_len_conv, n)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = shape.copy()
        print(f"[deterministic] initial best fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – random‑restart L‑BFGS continuation
        # ------------------------------------------------------------------
        lam_schedule = [
            0.6, 0.4, 0.25, 0.15, 0.08, 0.04,
            0.02, 0.01, 0.005, 0.002, 0.001,
            0.0005, 0.0002,
        ]
        maxiter_per_lam = 1200
        phase1_deadline = time.time() + (deadline - time.time()) * 0.35

        while time.time() < phase1_deadline:
            # Occasionally start from a noisy version of the current best
            if local_best_a is not None and rng.random() < 0.22:
                sigma = rng.uniform(0.001, 0.018)
                a_start = add_noise(local_best_a, sigma, rng)
            else:
                a_start = random_initial(n, rng)

            a_opt, sc_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase1_deadline,
                fft_len_conv,
                fft_len_g,
                n,
            )
            if sc_opt < local_best_score:
                local_best_score = sc_opt
                local_best_a = a_opt.copy()
                print(f"[L‑BFGS] new best fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – projected sub‑gradient descent with momentum
        # ------------------------------------------------------------------
        phase2_deadline = time.time() + (deadline - time.time()) * 0.25
        local_best_a, local_best_score = subgradient_optimize_momentum(
            local_best_a,
            phase2_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑grad] best fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – exponentiated‑gradient descent
        # ------------------------------------------------------------------
        phase3_deadline = time.time() + (deadline - time.time()) * 0.20
        local_best_a, local_best_score = exp_gradient_optimize(
            local_best_a,
            phase3_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[exp‑grad] best fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – tiny smoothing filter + re‑projection (optional)
        # ------------------------------------------------------------------
        kernel = np.array([0.25, 0.5, 0.25])
        smooth_iters = 5
        for _ in range(smooth_iters):
            smoothed = np.convolve(local_best_a, kernel, mode="same")
            smoothed = np.maximum(smoothed, 0.0)
            smoothed /= smoothed.sum()
            sc = fast_score(smoothed, fft_len_conv, n)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = smoothed
                print(f"[smooth] improved fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 5 – greedy hill‑climb
        # ------------------------------------------------------------------
        final_deadline = deadline
        local_best_a, local_best_score = hill_climb(
            local_best_a,
            local_best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=18000,
        )
        print(f"[hill‑climb] final fast_score = {local_best_score:.6f}")

        # ------------------------------------------------------------------
        # Keep the globally best solution
        # ------------------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.tolist()
            print(f"[global] new overall best fast_score = {best_score:.6f}")

        # Perform a final (exact) evaluation with the provided function
        # before moving on to another n (this also safeguards against
        # numerical quirks).
        exact = evaluate_sequence(best_seq)
        print(f"[global] exact evaluated score = {exact:.6f}")

    # If for any reason we never produced a sequence, fall back to uniform.
    if best_seq is None:
        best_seq = [1.0 / 4000] * 4000
        best_score = evaluate_sequence(best_seq)
        print("[fallback] uniform sequence used, score =", best_score)

    return best_seq
