def simulated_annealing_hc(a, cur_score, fft_len, n, deadline, rng, max_iter=50000):
    best = a.copy()
    best_score = cur_score
    temp0 = 0.02
    temp_end = 0.0001
    total = max_iter
    for it in range(total):
        if time.time() >= deadline:
            break
        # temperature schedule linear decay
        T = temp0 + (temp_end - temp0) * it / total
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j: continue
        # source is i if a[i]>0 else j
        if best[i] < 1e-12:
            continue
        delta = best[i] * rng.random() * 0.05  # move up to 5%
        if delta <= 0:
            continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        new_score = fast_score(new, fft_len, n)
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            best = new
            if new_score < best_score:
                best_score = new_score
    return best, best_score

"""
Propose a non‑negative step function with a low evaluate_sequence score.

Algorithm
---------
1. Choose a problem size `n`.  If a previous construction (`height_sequence_1`)
   exists, reuse its length (capped at 10000); otherwise use `n = 8000`.
2. Initialise a pool of starting points on the probability simplex:
   - random Gaussian logits,
   - exponential, symmetric exponential,
   - power‑law, arcsine, *product‑arcsine* (new),
   - Dirichlet (α=0.1),
   - a small perturbation of the current best.
3. **Coarse** continuation L‑BFGS: 30 smoothing stages (λ from 0.5 to 1e‑4),
   500 L‑BFGS iterations per stage.  Keep the best vector.
4. Short hill‑climb after each coarse run (fast `max‑convolution` score).
5. **Fine** continuation L‑BFGS from the best point: 80 stages (λ down to 1e‑12),
   1200 iterations per stage – a deeper local optimisation.
6. **Projected multi‑active sub‑gradient descent** for a long period
   (up to ~80 % of the remaining time).  Uses the exact `max` rather than a smooth surrogate.
7. **Simulated‑annealing hill‑climb**: many random mass transfers accepted
   with a Metropolis rule; temperature decays linearly.
8. Return the best sequence found (it already satisfies non‑negativity; the harness
   evaluates the exact score).

The extra “product‑arcsine’’ initialisation and the longer fine‑tuned L‑BFGS
plus a simulated‑annealing hill‑climb give a better chance of escaping local
minima and approaching the theoretical lower bound (score ≈ 1.0).
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a must have length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast (approximate) evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. y.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_schedule: list[float],
                      maxiter_per_stage: int = 800) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arc_prod":  # new mode: product‑arcsine shape
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb: move a random fraction of mass from i to j, accept only improvements."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(a_start: np.ndarray,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int,
                         deadline: float,
                         eta0: float = 0.30,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the *average* of the sub‑gradients of all (almost) active entries,
    and a simple back‑tracking line‑search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient = 2 * (a_rev * w) convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        # step size decreasing like 1/sqrt(t)
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # back‑track up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def simulated_annealing_hc(a: np.ndarray,
                           cur_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_iter: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a Metropolis acceptance rule.
    Temperature decays linearly from T0 to Tf.
    """
    best = a.copy()
    best_score = cur_score

    T0 = 0.02
    Tf = 0.0002

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        T = T0 + (Tf - T0) * (it / max_iter)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or best[i] < 1e-12:
            continue

        # move up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue

        new = best.copy()
        new[i] -= delta
        new[j] += delta

        new_score = fast_score(new, fft_len, n)

        if new_score < best_score:
            best = new
            best_score = new_score
            continue

        # Metropolis acceptance
        prob = np.exp(-(new_score - best_score) / max(T, 1e-12))
        if rng.random() < prob:
            best = new

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1: coarse continuation L‑BFGS + short hill‑climb
    # ------------------------------------------------------------------
    modes = ["random", "exp", "exp_sym", "power", "edge",
             "arcsine", "arc_prod", "sidon", "dirichlet", "best"]
    lam_sched_coarse = np.geomspace(0.5, 1e-4, num=30)
    maxiter_coarse = 500

    while time.time() < deadline - 260.0:   # leave ~260 s for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_sched_coarse,
            maxiter_per_stage=maxiter_coarse,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[coarse] mode={mode} → improved fast score = {best_score:.6f}")

        # very short hill‑climb to polish each run
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase‑2: refined continuation L‑BFGS from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        lam_sched_fine = np.geomspace(0.5, 1e-12, num=80)
        maxiter_fine = 1200
        print("[refine] start fine L‑BFGS")
        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_sched_fine,
            maxiter_per_stage=maxiter_fine,
        )
        best_a = softmax(y_opt)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = y_opt
        print(f"[refine] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3: long multi‑active sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        sub_deadline = deadline - 30.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑4: aggressive simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[annealing] start")
        best_a, best_score = simulated_annealing_hc(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
            max_iter=120_000,
        )
        print(f"[annealing] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
