"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Initialisation** – use the previously best sequence (if present) or a
   carefully shaped “U‑distribution” (arcsine density) and also keep a
   uniform baseline.
2. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp approximation
   to the max‑convolution with a decreasing smoothing schedule.  Many
   random restarts (with small Gaussian perturbations of the current best)
   are run while time permits.
3. **Exact exponentiated‑gradient descent** – minimise the true max‑convolution
   using the sub‑gradient (mask of maximal entries) and a multiplicative
   update that stays on the simplex.
4. **Targeted pairwise mass transfer** – repeatedly move a small fraction
   of mass from the coordinate with the largest sub‑gradient to the one
   with the smallest.  The move is accepted only if the max‑convolution
   decreases (a simple greedy but highly directed hill‑climb).
5. **Short simulated annealing** – a few temperature‑controlled random
   swaps to escape possible local minima.
6. **Final greedy hill‑climb** – fine‑tune with tiny random pairwise moves.

All phases respect a 1000 s wall‑clock budget; the best sequence found
before the deadline is returned.  The objective that is minimised is

    score = 2 * n * max( a * a )      with   sum(a) = 1.

The code is fully self‑contained and uses only NumPy, SciPy and the
provided ``evaluate_sequence`` for the final exact score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when ∑a = 1."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract part corresponding to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very rare – just skip this lambda.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'arcsine'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # arcsine
        # arcsine density on (0,1): f(x) ∝ 1/√(x(1-x))
        x = (np.arange(n) + 0.5) / n
        a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-16)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise to a and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size decreasing like 1/√t, but with a slightly larger initial factor
        eta = 0.8 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_pairwise_optimize(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  rng: np.random.Generator):
    """
    Greedy directed pairwise mass transfer using the exact sub‑gradient.
    Moves mass from the coordinate with largest sub‑gradient to the one with
    smallest, accepting only improving moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # step fraction controls how much mass is moved each iteration
    step_frac = 0.05
    it = 0
    while time.time() < deadline:
        # compute convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # indices with extreme gradient values
        i = np.argmax(g)          # mass should leave here
        j = np.argmin(g)          # mass should go here
        if i == j:
            # fallback to random pair if gradient is flat
            i, j = rng.integers(0, n, size=2)
            if i == j:
                it += 1
                continue

        # amount of mass to relocate
        delta = a[i] * step_frac
        if delta <= 0.0:
            it += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        else:
            # occasionally accept a worse move to escape plateaus
            if rng.random() < 0.001:
                a = cand

        it += 1
        # slowly reduce the step size as we get closer to a local optimum
        if it % 2000 == 0:
            step_frac = max(step_frac * 0.7, 0.005)

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple simulated annealing that transfers a random fraction of mass
    between two positions.  Worsening moves are accepted with a decaying
    temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.10
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Conventional greedy hill‑climb that moves a small random fraction of the
    mass from one coordinate to another, accepting only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # ------------------------------------------------------------------

            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # also keep a strong “U‑shaped” arcsine start for diversity
    arcsine_start = random_initial(n, rng)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Time budget split (with safety margin)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep 0.5 s safety

    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.80 * budget_s   # exp‑gradient
    phase3_end = start_time + 0.93 * budget_s   # pairwise directed moves
    # remaining time for annealing + final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            # Randomly choose between uniform, arcsine or a fresh random init
            a_start = rng.choice([a0, arcsine_start, random_initial(n, rng)])
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed pairwise mass transfers (sub‑gradient guided)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_pairwise_optimize(best_a,
                                                           phase3_end,
                                                           n,
                                                           fft_len_conv,
                                                           fft_len_g,
                                                           rng)
        print(f"[pairwise‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.30
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    # Convert to plain list for the final evaluation
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
