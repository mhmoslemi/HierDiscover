"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Main ideas
----------
1. **Parametric sweep** – explore simple families (Beta‑densities,
   sinusoidal powers, polynomial “raised‑cosine”) on the probability simplex.
   These give a very good starting point very cheaply (a few hundred FFTs).

2. **Continuation L‑BFGS** – smooth the max‑convolution with a
   log‑sum‑exp approximation and gradually tighten the smoothing.
   The schedule uses many stages with a larger number of L‑BFGS
   iterations for the later (harder) stages.

3. **Random restarts** – perturb the current best solution and run the
   continuation L‑BFGS again; keep any improvement.

4. **Simulated‑annealing hill‑climb** – a fast greedy move (transfer a
   fraction of mass from one index to another) is accepted either when
   it improves the fast score or with a temperature‑controlled
   probability.  The temperature is exponentially annealed to zero,
   allowing occasional uphill moves early on.

All phases respect the global time budget (default 1000 s).  The final
sequence is returned as a plain list; `evaluate_sequence` is called only
once at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient wrt conv (weights) ------------------------------------
    w = exp_shift / Z                 # ∂smooth_max/∂conv

    # ---- chain rule: conv = a * a ---------------------------------------
    # reverse‑convolution trick (see original source)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through softmax --------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple parametric families (used in the initial sweep)
# ----------------------------------------------------------------------


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def sin_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n) + 0.5) / n
    a = np.sin(angles) ** p
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def raised_cosine_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ (1 - ((i-(n‑1)/2)/( (n‑1)/2 ))² )^c, zero outside the interval."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.clip(1.0 - dist * dist, 0.0, None) ** c
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30000,
    max_move: float = 0.25,
    T_start: float = 2e-3,
    T_end: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Random hill‑climb with Metropolis acceptance.
    Moves a random fraction of mass from i → j.
    The temperature decays exponentially from T_start to T_end.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # exponential temperature schedule
        tau = it / max_iter
        T = T_start * (T_end / T_start) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(new_score - best_score) / T):
                accept = True

        if accept:
            best_score = new_score
        else:
            # revert move
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size n (use existing construction if possible)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        n = max(4000, min(tmp.size, 12000))   # keep it in a reasonable range
    else:
        n = 8000                               # default size

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep for a good starting point
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # uniform (baseline)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[param] uniform baseline fast score = {best_score:.6f}")

    # ----- Beta family -------------------------------------------------
    alphas = np.arange(0.4, 2.01, 0.2)
    betas = np.arange(0.4, 2.01, 0.2)
    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                break
            a = beta_family(n, alpha, beta)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a
                print(f"[param] beta({alpha:.2f},{beta:.2f}) improved = {sc:.6f}")

    # ----- Sinusoidal power family ------------------------------------
    for p in np.arange(0.2, 4.2, 0.2):
        if time.time() > deadline:
            break
        a = sin_power_family(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a
            print(f"[param] sin^{p:.2f} improved = {sc:.6f}")

    # ----- Raised‑cosine (polynomial) family -------------------------
    for c in np.arange(0.5, 5.0, 0.5):
        if time.time() > deadline:
            break
        a = raised_cosine_family(n, c)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a
            print(f"[param] raised_cosine^{c:.1f} improved = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS from the best parametric shape
    # ------------------------------------------------------------------
    y = np.log(best_a + 1e-12)          # log‑softmax parameters

    lam_schedule = list(np.geomspace(0.3, 1e-12, num=30))
    # allocate a relatively generous number of iterations per stage
    maxiter_per_stage = 400

    print("[L‑BFGS] start continuation")
    y_opt = schedule_optimize(
        y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule,
        maxiter_per_stage=maxiter_per_stage,
    )
    a_opt = softmax(y_opt)
    score_opt = fast_score(a_opt, fft_len_conv, n)

    if score_opt < best_score:
        best_score, best_a = score_opt, a_opt
        print(f"[L‑BFGS] improved fast score = {score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – a few random restarts of the continuation L‑BFGS
    # ------------------------------------------------------------------
    restarts_deadline = min(deadline, start_time + 0.55 * budget_s)
    restart_cnt = 0
    while time.time() < restarts_deadline and restart_cnt < 5:
        restart_cnt += 1
        # perturb current best log‑parameters
        y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.03, size=n)
        y_restart = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_restart = softmax(y_restart)
        sc_restart = fast_score(a_restart, fft_len_conv, n)
        if sc_restart < best_score:
            best_score, best_a = sc_restart, a_restart
            print(f"[restart {restart_cnt}] improved fast score = {sc_restart:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb on the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[SA] start simulated‑annealing hill‑climb")
        best_a, best_score = simulated_annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
            max_move=0.30,
            T_start=2e-3,
            T_end=1e-7,
        )
        print(f"[SA] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – supplied by harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
```

This implementation adds a quick parametric sweep over several expressive families, a more thorough continuation‑L‑BFGS (with many stages and larger iteration counts), random restarts to escape possible basins, and finally a simulated‑annealing hill‑climb that can accept occasional uphill moves. All phases are time‑aware and respect the 1000 s budget, aiming to push the evaluation score below the target 1.503.
