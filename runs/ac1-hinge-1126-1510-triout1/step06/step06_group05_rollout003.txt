"""
propose_candidate
=================

Search for a non‑negative step function (list of floats) that minimizes the
``evaluate_sequence`` score.  The algorithm consists of the following stages:

1. **Problem size selection** – reuse an existing long sequence if available,
   otherwise pick a size between 8000 and 12000 (a good trade‑off between
   expressivity and speed).

2. **Parametric grid search** – enumerate several well‑known families
   (beta‑type, cosine windows, exponential, edge‑spikes, Gaussian).  The best
   candidate from this cheap sweep is used as the starting point for the
   subsequent optimisation.

3. **Continuation L‑BFGS on a smoothed max‑convolution** – minimise a
   log‑sum‑exp (soft‑max) approximation of the true objective.
   The smoothing factor (λ) is decreased geometrically over 70 stages.
   Many random restarts (including noisy perturbations of the current best)
   are performed while time permits.  After each optimisation a short greedy
   hill‑climb refines the solution.

4. **Projected sub‑gradient descent** – optimise the true objective
   ``max(conv)`` using an adaptive step‑size (ηₜ = η₀/√t) and projection
   onto the probability simplex.

5. **Final aggressive hill‑climb** – a greedy local search until the time
   budget is exhausted.

All helper functions are defined at top‑level, no closures or lambdas are used.
The code respects the global 1000 s time budget (leaving a small safety margin)
and prints progress information.  The returned list is the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z

    # propagate gradient through convolution (a is real, length n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^α (n-i)^β."""
    i = np.arange(n, dtype=float)
    prob = (i + 1) ** alpha * (n - i) ** beta
    prob = np.maximum(prob, 0.0)
    if prob.sum() == 0:
        prob = np.full(n, 1.0 / n)
    else:
        prob = prob / prob.sum()
    return prob


def generate_cosine_window(n: int, power: float) -> np.ndarray:
    """a_i ∝ (1 - cos(π i/(n-1)))^power (Hann‑like)."""
    i = np.arange(n, dtype=float)
    base = 0.5 - 0.5 * np.cos(np.pi * i / (n - 1))
    prob = base ** power
    prob = np.maximum(prob, 0.0)
    prob /= prob.sum()
    return prob


def generate_exponential(n: int, rate: float, symmetric: bool = True) -> np.ndarray:
    """Exponential decay from the left (and optionally mirrored)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    if symmetric:
        a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_edge_spike(n: int, edge_frac: float, weight_factor: float) -> np.ndarray:
    """Heavy equal mass on the first/last edge_frac proportion."""
    m = max(1, int(round(edge_frac * n)))
    uniform = 1.0 / n
    w_edge = weight_factor * uniform
    a = np.full(n, uniform)
    a[:m] = w_edge
    a[-m:] = w_edge
    # renormalise
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random_sidons(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon set (no repeated sums) and equal
    weight on the selected indices.
    """
    chosen = []
    sums = set()
    for cand in rng.permutation(n):
        conflict = False
        for c in chosen:
            if (cand + c) in sums or (2 * cand) in sums:
                conflict = True
                break
        if not conflict:
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
        if len(chosen) >= int(np.sqrt(n) * 1.5):
            break
    k = len(chosen)
    a = np.zeros(n)
    a[chosen] = 1.0 / k
    a /= a.sum()
    return a

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to
    j, accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = max_i (a*a)_i .
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  Respects the given time budget and returns

    """

            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = np.maximum(arr, 0.0)
            if best_a.sum() == 0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a /= best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric grid search
    # ------------------------------------------------------------------
    # 1. beta‑type families (wide range)
    alphas = np.linspace(-2.0, 2.0, 17)
    betas = np.linspace(-2.0, 2.0, 17)
    for alpha in alphas:
        for beta in betas:
            prob = generate_beta_family(n, alpha, beta)
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[beta] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # 2. cosine windows (Hann‑like)
    powers = np.arange(0.5, 3.1, 0.3)
    for p in powers:
        prob = generate_cosine_window(n, p)
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[cos] p={p:.2f} → fast_score = {sc:.6f}")

    # 3. symmetric exponential families
    rates = np.geomspace(0.001, 0.05, 12)
    for r in rates:
        prob = generate_exponential(n, r, symmetric=True)
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[exp] rate={r:.5f} → fast_score = {sc:.6f}")

    # 4. edge‑spike families
    edge_fracs = np.geomspace(0.001, 0.08, 9)
    weight_facs = np.geomspace(1.0, 30.0, 7)
    for ef in edge_fracs:
        for wf in weight_facs:
            prob = generate_edge_spike(n, ef, wf)
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[edge] frac={ef:.5f} wf={wf:.2f} → fast_score = {sc:.6f}")

    # 5. Gaussian families (centered, but still useful as baseline)
    sigma_fracs = np.linspace(0.05, 0.30, 6)
    for sf in sigma_fracs:
        prob = generate_gaussian(n, sf)
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[gauss] sigma_frac={sf:.2f} → fast_score = {sc:.6f}")

    # 6. Random Sidon‑type patterns
    for _ in range(7):
        prob = generate_random_sidons(n, rng)
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[sidon] → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-11, num=70))
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    # Reserve ~260 s for the sub‑gradient + final polishing stages
    while time.time() < deadline - 260.0:
        mode = rng.choice(modes)
        if mode == "best":

                prob = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)

            elif mode == "sidon":

            else:
                prob = np.full(n, 1.0 / n)

            y0 = np.log(prob + 1e-12)

        y_opt = schedule_optimize(
            y_start=y0,

            fft_len_g=fft_len_g,

        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Refine with projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 130.0:

        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.22,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # the harness supplies this function
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
