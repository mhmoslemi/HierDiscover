def subgrad_optimize_exp(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    while time.time() < deadline:
        # compute conv via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        M = conv.max()
        avg = conv.mean()
        # mask for max entries within tolerance
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(np.float64)
        # gradient of max (subgradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n-1):(n-1+n)]
        # polyak step size
        norm_sq = np.dot(g, g)
        if norm_sq == 0:
            break
        step = (M - avg) / norm_sq
        # clamp to reasonable range
        step = max(min(step, 0.6), 0)  # ensure non-negative? Actually step positive sign.
        # update using exponentiated gradient? We'll use projection onto simplex with step
        a_candidate = a - step * g
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s <= 0:
            a_candidate = best_a.copy()

        # Evaluate

"""
Find a non‑negative step function (list of floats) that minimises
    2·n·max(a*a) / (sum a)^2
using a multi‑phase optimisation:
* many random‑restart L‑BFGS on a log‑sum‑exp smooth approximation
* a Polyak‑step sub‑gradient descent on the exact max‑convolution
* a short Adam refinement on the smoothed objective
* aggressive hill‑climb moves.
The routine respects the 1000 s budget and returns the best sequence found.
"""
import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # extremely rare
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

    elif mode == 'edge':

        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)

        M = conv.max()
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))

        eta = 0.6 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

def subgrad_polyak_optimize(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len_conv: int,
                            fft_len_g: int):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        avg = conv.mean()
        eps = M * 1e-12 + 1e-12

        if norm_sq == 0.0:

        step = max(min(step, 0.5), 0.0)
        a_candidate = a - step * g

        s = a_candidate.sum()

        else:
            a_candidate = a_candidate / s

        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.12
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

"""
Multi‑phase optimisation for the step‑function benchmark.

The routine builds a non‑negative sequence a of length 4000 (or a
user‑provided length) that minimises

    2·n·max(a*a) / (sum a)²

where “*” denotes linear convolution.
The search consists of:

1.  Many random‑restart L‑BFGS runs on a log‑sum‑exp smooth
    approximation of the max‑convolution.  A decreasing λ schedule
    gradually removes the smoothing.
2.  A Polyak‑step sub‑gradient descent on the *exact* max‑convolution.
    The step length (M‑avg)/‖g‖² drives the value toward the average
    of the convolution entries, which is a known lower bound.
3.  Exponentiated‑gradient descent (the original “exp‑grad” stage) – a
    fast way to stay on the simplex while following the sub‑gradient.
4.  An aggressive hill‑climb that transfers a small random fraction of
    mass between two coordinates and keeps only improving moves.

The whole procedure respects the 1000 s time limit; the best sequence
found before the deadline is returned as a plain list of floats.
"""

import numpy as np
from scipy.optimize import minimize

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → prob. simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast proxy for the evaluation when sum(a)=1: 2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Log‑sum‑exp smooth approximation of max(a*a) with gradient.
    Variables are logits y, where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial sequence
    # --------------------------------------------------------------
    max_n = 4000
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        n = max_n
        if init_seq.ndim == 1 and init_seq.size > 0:
            # linear interpolation to the target length
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS (≈45 % of the budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
    ]
    maxiter_per_lam = 300  # modest per‑lambda budget

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.15:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Polyak sub‑gradient descent (≈30 % of the budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgrad_polyak_optimize(
            best_a, phase2_end, n, fft_len_conv, fft_len_g
        )
        print(f"[subgrad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (≈15 % of the budget)
    # --------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len_conv, fft_len_g, rng
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[hill‑climb] final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the answer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
