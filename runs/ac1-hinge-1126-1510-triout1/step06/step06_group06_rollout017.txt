"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. **Initialisation** – use a stored solution if available, otherwise a uniform
   distribution of length `n = 5000`.

2. **Parametric sweep** – evaluate a handful of simple families
   (exponential, power‑law, edge‑boosted, linear, quadratic) on a grid of
   parameters.  The best candidate from this cheap search becomes the starting
   point for the next phase.

3. **Smooth‑max L‑BFGS refinement** – re‑optimise the log‑soft‑max variables
   `y = log(a)` with a continuation on the temperature `λ`.  The surrogate
   objective is a smooth approximation of `max(conv(a,a))`.  Several random
   restarts (perturbations of the current best) are performed until the time
   budget is almost exhausted.

4. **Local polishing** – a targeted hill‑climb that moves mass away from the
   pair of indices that currently contributes most to the maximal convolution
   entry, followed by a random mass‑transfer hill‑climb.

5. **Projected sub‑gradient descent** – directly minimises the true
   non‑smoothed objective `2·n·max(conv(a,a))` on the simplex with a
   diminishing step‑size schedule.

6. **Return** – the best non‑negative sequence found (scaled automatically when
   `evaluate_sequence` is called).  The result is stored in the global variable
   ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a)) where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smooth‑max of the auto‑convolution and its gradient w.r.t. `y`."""
    a = softmax(y)

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth‑max w.r.t. conv
    w = exp_shift / Z

    # back‑propagate through convolution
    # note: for the gradient we need the convolution of a with w reversed
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # pick the slice that corresponds to the gradient w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors,
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """Continuation of L‑BFGS over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 400,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """Move mass away from the worst contributing pair for the current max shift."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))  # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a target with a small partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def max_convolution_subgrad(a: np.ndarray, n: int, fft_len: int):
    """Return (max_conv, sub‑gradient) for the true max‑convolution."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    k_max = int(np.argmax(conv))

    grad = np.zeros_like(a)
    # gradient of a_i * a_{k-i} w.r.t a_i
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] += a[j]
        # symmetric term (i appears as the second index)
        j2 = i - k_max
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return max_c, grad


# ----------------------------------------------------------------------
# Parametric families for the cheap exploratory sweep
# ----------------------------------------------------------------------


def gen_exponential(theta: float, n: int) -> np.ndarray:
    a = np.exp(-theta * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def gen_power(p: float, n: int) -> np.ndarray:
    a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def gen_linear(alpha: float, n: int) -> np.ndarray:
    a = np.maximum(0.0, 1.0 - alpha * np.arange(n, dtype=float) / n)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def gen_quadratic(beta: float, n: int) -> np.ndarray:
    a = np.maximum(0.0, 1.0 - beta * (np.arange(n, dtype=float) / n) ** 2)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def gen_edge(frac: float, ratio: float, n: int) -> np.ndarray:
    """‘Edge’ pattern – high values on the two borders."""
    m = max(1, int(frac * n))
    low = 1.0
    high = ratio * low
    # solve low * (n - 2m) + high * 2m = 1   →  low = 1 / (n - 2m + 2m * ratio)
    low = 1.0 / (n - 2 * m + 2 * m * ratio)
    high = ratio * low
    a = np.full(n, low)
    a[:m] = high
    a[-m:] = high
    return a / a.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size and initial solution
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Fast parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = min(deadline, time.time() + 80.0)  # reserve about a minute

    # --- exponential ---
    for theta in np.geomspace(5e-5, 0.02, num=28):
        if time.time() >= sweep_deadline:
            break
        a = gen_exponential(theta, n)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] exp theta={theta:.5g} → score={s:.6f}")

    # --- power law ---
    for p in np.linspace(0.1, 5.0, 28):
        if time.time() >= sweep_deadline:
            break
        a = gen_power(p, n)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] power p={p:.3f} → score={s:.6f}")

    # --- linear ---
    for alpha in np.linspace(0.0, 2.0, 26):
        if time.time() >= sweep_deadline:
            break
        a = gen_linear(alpha, n)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] linear α={alpha:.3f} → score={s:.6f}")

    # --- quadratic ---
    for beta in np.linspace(0.0, 4.0, 22):
        if time.time() >= sweep_deadline:
            break
        a = gen_quadratic(beta, n)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] quadratic β={beta:.3f} → score={s:.6f}")

    # --- edge pattern ---
    for frac in np.linspace(0.01, 0.07, 13):
        for ratio in np.linspace(2.0, 30.0, 13):
            if time.time() >= sweep_deadline:
                break
            a = gen_edge(frac, ratio, n)
            s = fast_score(a, fft_len_conv, n)
            if s < best_score:
                best_a, best_score = a, s
                print(f"[sweep] edge f={frac:.3f} r={ratio:.1f} → score={s:.6f}")

    # --------------------------------------------------------------
    # 2. L‑BFGS refinement with several random restarts
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)
    maxiter_per_stage = 600
    iteration = 0

    while time.time() < deadline - 5.0:
        iteration += 1
        # start point: log‑soft‑max of best + Gaussian perturbation
        y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.04, size=n)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=next_pow2(3 * n - 2),
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        s_opt = fast_score(a_opt, fft_len_conv, n)

        if s_opt < best_score:
            best_a, best_score = a_opt, s_opt
            print(f"[{iteration}] L‑BFGS improved score={best_score:.6f}")

        # ----------------------------------------------------------
        # Targeted hill‑climb (quick, cheap)
        # ----------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=800,
            move_frac=0.30,
        )

        # ----------------------------------------------------------
        # Random hill‑climb (adds stochastic escape)
        # ----------------------------------------------------------
        rc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=3000,
            max_move_frac=0.20,
        )

        # small early‑stop if we run out of time
        if time.time() > deadline - 2.0:
            break

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the true objective
    # --------------------------------------------------------------
    step0 = 0.45
    sub_iter = 0
    eps = 1e-12

    while time.time() < deadline - 2.0:
        sub_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)
        step = step0 / np.sqrt(sub_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a, best_score = cand, cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92

        if sub_iter % 200 == 0:
            print(f"[subgrad] iter={sub_iter} best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final polishing with a longer random hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.25,
        )
        print(f"[final‑hill] polished score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the solution (the official evaluation will re‑scale)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq  # keep for possible reuse
    final_val = evaluate_sequence(final_seq)  # the provided evaluator
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
