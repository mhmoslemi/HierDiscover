def max_conv_and_gradient(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))[:2*len(a)-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(float)  # shape (2n-1,)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    g_full = np.real(np.fft.ifft(fa_rev*fm))
    g = 2 * g_full[len(a)-1:len(a)-1+len(a)] # extract central part.
    return max_val, g
```

def subgradient_descent(a0, deadline, n, fft_len):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        max_val, g = max_conv_and_gradient(a, fft_len)
        # step size schedule
        eta = 0.7 / np.sqrt(it+1)
        # update
        a_candidate = a - eta * g
        a_candidate = simplex_projection(a_candidate)
        # keep sum=1
        # guarantee non-negativity via projection.
        score = fast_score(a_candidate, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
        a = a_candidate
        # occasional random perturbation
        if it % 2000 == 0 and it > 0:
            sigma = 0.001 * (1 + it // 2000)
            a = add_noise(a, sigma, rng)
        it += 1
    return best_a, best_score
```

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a population of diverse probability vectors (size ~30) on the
   simplex of length n = 4000 (or from the supplied `height_sequence_1`).
2. Run a *projected sub‑gradient descent* on the exact max‑convolution
   (hard L∞ objective).  The descent uses an additive step (negative
   gradient) with Euclidean projection onto the simplex and a simple
   step‑size schedule η = η0 / √t.  Several step‑size multipliers are tried
   each iteration and the best candidate is kept.
3. Evolutionary loop: repeatedly create children by averaging two parents
   (chosen from the current best half), adding a little Gaussian noise,
   projecting onto the simplex and polishing the child with a short
   sub‑gradient run.  The child replaces the worst individual if it is
   better.  This explores the non‑convex landscape more globally.
4. After most of the time budget is spent, a greedy hill‑climb moves a
   random fraction of mass between two coordinates and accepts only true
   improvements.  This fine‑tunes the best solution found.
5. Finally the solution is evaluated with the provided `evaluate_sequence`
   and returned as a plain list of floats.

All heavy work is done with FFT‑based convolutions, which makes the
search fast enough to spend the full 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector using several families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution‑max and sub‑gradient
# ----------------------------------------------------------------------


def conv_max_and_grad(a: np.ndarray, fft_len: int):
    """
    Compute the maximum of the self‑convolution and a sub‑gradient
    (w.r.t. a) of that maximum.
    """
    n = a.shape[0]
    # full convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # gradient of the (hard) max‑convolution
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_full = np.real(np.fft.ifft(fa_rev * fm))
    # extract the central n entries
    grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]
    return max_val, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (additive)
# ----------------------------------------------------------------------


def subgrad_descent(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator,
                    max_iter: int = None):
    """
    Projected sub‑gradient descent on the exact L∞ objective.
    Returns the best vector and its fast_score seen during the run.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    base_eta = 0.5

    while time.time() < deadline and (max_iter is None or it < max_iter):
        _, grad = conv_max_and_grad(a, fft_len)
        eta = base_eta / np.sqrt(it + 1.0)

        # try a few step‑size multipliers and keep the best candidate
        candidates = []
        for mul in (0.5, 1.0, 2.0):
            cand = a - eta * mul * grad
            cand = simplex_projection(cand)
            candidates.append(cand)

        scores = [fast_score(c, fft_len, n) for c in candidates]
        best_idx = int(np.argmin(scores))
        a = candidates[best_idx]
        cur_score = scores[best_idx]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional stochastic perturbation to escape plateaus
        if it % 2000 == 0 and it > 0:
            a = add_noise(a, sigma=0.001 * (1 + it // 5000), rng=rng)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Evolutionary population search
# ----------------------------------------------------------------------


def evolve_population(pop_size: int,
                      n: int,
                      fft_len: int,
                      deadline: float,
                      rng: np.random.Generator,
                      init_vec: np.ndarray = None):
    """
    Simple evolutionary algorithm.
    Maintains a population of probability vectors, creates children by
    averaging two parents plus Gaussian noise, polishes each child with a
    short sub‑gradient run, and replaces the worst individual when the
    child is better.
    """
    # Initialise population
    population = [random_initial(n, rng) for _ in range(pop_size)]
    scores = [fast_score(p, fft_len, n) for p in population]

    # If an explicit initial vector is supplied, insert it (replace worst)
    if init_vec is not None:
        worst = int(np.argmax(scores))
        population[worst] = init_vec.copy()
        scores[worst] = fast_score(init_vec, fft_len, n)

    best_idx = int(np.argmin(scores))
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    gen = 0
    # Main evolutionary loop
    while time.time() < deadline:
        # Choose two parents from the current elite half
        elite_size = max(2, pop_size // 2)
        elite_inds = np.argsort(scores)[:elite_size]
        i, j = rng.choice(elite_inds, size=2, replace=False)

        child = 0.5 * (population[i] + population[j])
        child = add_noise(child, sigma=0.001, rng=rng)
        child = simplex_projection(child)

        # Short polishing with sub‑gradient (time‑bounded)
        child_deadline = min(deadline, time.time() + 0.25)  # ≤ 0.25 s per child
        child, child_score = subgrad_descent(child,
                                             child_deadline,
                                             n,
                                             fft_len,
                                             rng,
                                             max_iter=2000)

        # Replace the worst individual if improvement
        worst = int(np.argmax(scores))
        if child_score < scores[worst]:
            population[worst] = child
            scores[worst] = child_score
            if child_score < best_score:
                best_score = child_score
                best_a = child.copy()

        gen += 1
        # occasional status report
        if gen % 500 == 0:
            print(f"[evo] gen={gen}, best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               start_score: float,
               n: int,
               fft_len: int,
               deadline: float):
    """
    Greedy pairwise mass transfers that accept only strictly improving moves.
    """
    a = a_start.copy()
    cur_score = start_score
    rng = np.random.default_rng(0)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return a, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial vector
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # Try to use the previously discovered construction, if present
    if 'height_sequence_1' in globals():
        init_arr = np.array(globals()['height_sequence_1'], dtype=float)
        if init_arr.size > 0:
            x_old = np.linspace(0.0, 1.0, init_arr.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, init_arr)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    phase1_end = start_time + 0.70 * budget_s   # evolutionary phase
    phase2_end = start_time + 0.92 * budget_s   # final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – evolutionary search with sub‑gradient polishing
    # --------------------------------------------------------------
    pop_size = 30
    best_a, best_score = evolve_population(pop_size,
                                            n,
                                            fft_len,
                                            phase1_end,
                                            rng,
                                            init_vec=a0)
    print(f"[phase1] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb (strict improvements)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        n,
                                        fft_len,
                                        phase2_end)
        print(f"[phase2 hill‑climb] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
