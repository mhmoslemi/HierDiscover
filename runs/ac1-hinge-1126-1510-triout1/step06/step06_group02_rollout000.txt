"""
propose_candidate – Heuristic search for a low‑score non‑negative step function.

The algorithm consists of three phases:

1. **Continuation L‑BFGS** – A smoothed log‑sum‑exp surrogate of the max‑convolution
   is minimised using a schedule of decreasing smoothing parameters.
   Several random starts (including perturbations of the current best) are tried.
2. **Projected multi‑active sub‑gradient descent** – Optimises the exact,
   non‑smooth objective using the averaged sub‑gradient of all (almost) active
   convolution entries.  The step size decays as 1/√t and a back‑tracking line‑search
   guarantees monotone improvement.
3. **Final aggressive hill‑climb** – Repeatedly transfers a random fraction of mass
   between two coordinates, accepting only moves that improve the exact score.
   This runs until the overall time budget (≈ 1000 s) is exhausted.

The sequence length is chosen adaptively (up to 10 000).  The code respects the
security checks of `evaluate_sequence` and returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using an FFT of length `fft_len`.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of the objective (sum_a = 1) using FFT.
    Returns 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x | x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max re‑parameterisation
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y)),
    and grad_y = ∂f/∂y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                     # max = 0
    exp_shift = np.exp(shifted / lam)         # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of L‑BFGS optimisations with decreasing
    smoothing factor.  Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point (pre‑soft‑max) according to the requested heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local optimisation helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly transfer a random fraction of mass from i to j.
    Accept only improving moves.  Stops on deadline or after `max_iter` attempts.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.9,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true (non‑smooth) objective.
    Uses the uniform average over all (almost) active convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # Relative tolerance to capture a stable active set
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        # Build weight vector w over convolution indices
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # Gradient of the (averaged) max entry w.r.t a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec  # derivative of a_i * a_j term

        # Step size with simple back‑tracking
        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # Even after reductions no improvement – take a reduced step anyway.
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # leave a tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1:
            # aim for a length in [8000, 10000] but keep existing data if already large
            if tmp.size >= 8000:
                n = min(tmp.size, 10000)
            else:
                n = max(8000, tmp.size)
        else:
            n = 8000
    else:
        n = 8000

    # Pad or truncate the initial sequence to length n and normalise.
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.size < n:
            # pad with zeros
            a0 = np.concatenate([tmp, np.zeros(n - tmp.size, dtype=float)])
        else:
            a0 = tmp[:n]
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} → fast score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths.
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Continuation L‑BFGS phase (≈ 300 s).
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=40))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",
        "best",
    ]
    iteration = 0
    # Reserve roughly 300 s for this phase
    while time.time() < deadline - 350.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.9f}"
            )

        # Brief hill‑climb after each L‑BFGS run (≈ 1 s)
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
            max_move=0.12,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Projected multi‑active sub‑gradient descent (≈ 200 s).
    # --------------------------------------------------------------
    if time.time() < deadline - 200.0:
        sub_deadline = deadline - 80.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.9,
            max_iters=600_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final aggressive hill‑climb using the remaining budget.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.20,
        )
        print(f"[final HC] finished → fast score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Official evaluation.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.9f}")

    return final_seq
