"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. **Problem size** – Use a long vector (default n = 5000) and keep all
   entries non‑negative and summing to 1 (the objective is scale invariant).

2. **Warm‑start** – Several deterministic “shapes’’ are tried:
   * uniform,
   * a decreasing power law,
   * the *arcsine* shape   a_i ∝ 1/√(i+1)  (its auto‑convolution is almost flat).

   The best of these is taken as the initial point.

3. **Local optimisation** – A simulated‑annealing (SA) loop works on the
   probability simplex by repeatedly moving a random fraction of mass from
   one index to another.  The move is accepted if it improves the fast
   surrogate 2·n·max(conv) or with a Metropolis probability that decays
   with a linear temperature schedule.  The SA loop runs until the global
   time budget is almost exhausted.

4. **Final polishing** – A pure hill‑climb (only improving moves) is
   performed for a short remaining period to clean up the last few
   fluctuations.

The whole search stays well inside the 1000 s budget, constantly prints
progress, and finally returns the best sequence found (stored in the
global ``height_sequence_1`` for reuse).  The evaluation uses the provided
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (no closures, all top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence``: 2·n·max(conv(a,a))."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def init_sequence(rng: np.random.Generator, n: int, mode: str) -> np.ndarray:
    """
    Produce a deterministic or random starting probability vector.
    Modes:
      * uniform    – a_i = 1/n
      * power      – a_i ∝ (i+1)^(-p)  with random p∈[0.3,2]
      * arcsine    – a_i ∝ 1/√(i+1)
      * exponential– a_i ∝ exp(-λ i)   with random λ
      * random     – Dirichlet(α=1)
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
    elif mode == "arcsine":
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
        a /= a.sum()
    elif mode == "exponential":
        lam = rng.uniform(0.001, 0.02)
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
    elif mode == "random":
        a = rng.dirichlet(np.ones(n))
    else:
        raise ValueError(f"unknown mode {mode}")
    return a


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.12,
    final_temp: float = 1e-5,
    max_move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Annealing on the probability simplex.
    Transfer a random fraction of mass from i → j.
    Non‑improving moves are accepted with probability exp(-Δ/τ),
    where τ is the current temperature (linear schedule).
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    step = 0
    while time.time() < deadline and total_time > 0:
        # linear temperature decay
        elapsed = time.time() - (deadline - total_time)
        tau = init_temp + (final_temp - init_temp) * (elapsed / total_time)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        accept = False
        if dE < -1e-15:
            accept = True
        else:
            # Metropolis criterion
            if np.exp(-dE / max(tau, 1e-12)) > rng.random():
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best = cur.copy()
                best_score = new_score
        else:
            # revert the move
            cur[i] += delta
            cur[j] -= delta

        step += 1
        if step % 5000 == 0:
            print(
                f"[SA] step={step} τ={tau:.3e} cur={cur_score:.6f} best={best_score:.6f}"
            )
    return best, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Pure improving hill‑climb on the simplex.
    Only strictly improving mass‑transfer moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    ``evaluate_sequence`` score.  The routine uses several deterministic
    initialisations, a simulated‑annealing phase for global exploration
    and a final hill‑climb for polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size (reuse previous result if available)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # guarantee a valid probability vector
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Try a few deterministic starting shapes and keep the best
    # ------------------------------------------------------------------
    modes = ["uniform", "power", "exponential", "arcsine"]
    for mode in modes:
        a0 = init_sequence(rng, n, mode)
        score0 = fast_score(a0, fft_len, n)
        print(f"[init] mode={mode} fast score = {score0:.6f}")
        if score0 < best_score:
            best_a = a0
            best_score = score0
            print(f"[init] best updated to mode={mode} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (SA + hill‑climb) until time runs out
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        # SA phase (uses about 15‑20 % of remaining budget)
        sa_deadline = min(deadline, time.time() + 0.20 * (deadline - time.time()))
        best_a, best_score = simulated_annealing(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=sa_deadline,
            init_temp=0.12,
            final_temp=1e-5,
            max_move_frac=0.28,
        )
        print(f"[{iteration}] after SA  fast score = {best_score:.6f}")

        # short hill‑climb to lock‑in any improvements
        hc_deadline = min(deadline, time.time() + 0.10 * (deadline - time.time()))
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.22,
        )
        print(f"[{iteration}] after HC  fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a longer pure hill‑climb (any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] polishing with extended hill‑climb")
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            max_iter=12000,
            max_move_frac=0.25,
        )
        print(f"[final] polished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
