Pseudo:

```
def simulated_annealing(a, cur_score, fft_len, n, deadline):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        t = max(1e-6, (total_time - elapsed) / total_time)  # linear cooling (1 to 0)
        T = t * 0.01  # scale temperature based on typical score differences
        # propose move
        i = rng.integers(0, n)
        if cur[i] == 0: continue
        j = rng.integers(0, n)
        if i == j: continue
        delta = cur[i] * rng.random() * 0.3
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
1. Initialise from the previously best known sequence (if any) or from a
   uniform distribution of length ``n = 5000``.
2. GLOBAL OPTIMISATION – several random restarts of a smooth‑max
   L‑BFGS optimiser (log‑soft‑max parametrisation).  A fine λ‑schedule
   (geometric from 0.8 to 1e‑7) is used; each stage performs a bounded
   number of L‑BFGS iterations.
3. SIMULATED ANNEALING – a temperature controlled random mass‑transfer
   walk on the simplex that is allowed to accept worsening moves.
   The walk explores valleys that L‑BFGS cannot leave.
4. LOCAL REFINEMENT – a deterministic hill‑climb that only keeps
   strictly improving moves (as in the reference code).
5. Return the best sequence found (saved in ``height_sequence_1``).

The objective is
    2·n·max(conv(a,a)) / (sum a)²   (scale‑invariant)
so we keep the vector on the probability simplex (sum = 1).
All convolutions are performed by FFT for speed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``.
    Returns 2·n·max( a * a ) where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # full convolution a*a (linear, not circular)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = max_c + lam * log( sum exp((conv-max_c)/lam) )
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # back‑propagation: gradient w.r.t. a
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # indices that correspond to the gradient for a (length n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point ``y`` (log‑soft‑max parameters) for L‑BFGS.
    A mixture of heuristics is used to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        centre_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, centre_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb that only accepts strictly improving moves."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Worsening moves are accepted with probability exp(-Δ/T).
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start, 1e-6)

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling from T0 to 0
        T = 0.01 * max(0.0, 1.0 - elapsed / total)

        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a fraction of the i‑th mass to j
        delta = cur[i] * rng.random() * max_move_frac
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score

        if dE <= 0.0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta

    return best, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops short of this limit.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size and starting point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # normalise to a probability simplex (objective is scale‑invariant)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # GLOBAL OPTIMISATION – L‑BFGS with many restarts
    # ------------------------------------------------------------------
    # a generous λ‑schedule (starting fairly smooth and ending very sharp)
    lam_factors = np.geomspace(0.8, 1e-7, num=55)
    maxiter_per_stage = 600
    iteration = 0

    # allocate roughly 70 % of the time budget to the L‑BFGS phase
    lbfgs_deadline = time.time() + (budget_s * 0.70)

    while time.time() < lbfgs_deadline - 3.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=next_pow2(3 * n - 2),
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # SIMULATED ANNEALING – escape possible local minima
    # ------------------------------------------------------------------
    sa_deadline = deadline - 8.0  # keep a little buffer for final polishing
    best_a, best_score = simulated_annealing(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=sa_deadline,
        max_move_frac=0.35,
    )
    print(f"[sa] best fast score after annealing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # FINAL LOCAL REFINEMENT – deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        print(f"[final hill‑climb] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated true score = {final_val:.6f}")
    return final_seq
