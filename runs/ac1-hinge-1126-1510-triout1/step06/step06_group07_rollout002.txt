def simulated_annealing(a, init_score, fft_len, n, deadline, start_temp=0.01, end_temp=1e-5, move_frac_max=0.2):
    best = a.copy()
    best_score = init_score
    cur = a.copy()
    cur_score = init_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        t = elapsed / total_time
        T = start_temp * (1 - t) + end_temp * t  # linear cooling
        # pick i, j distinct
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j+=1
        # step size based on T possibly
        max_delta = cur[i] * move_frac_max
        if max_delta <= 1e-12:
            continue
        # dynamic scale: delta = max_delta * rng.random()
        # maybe scale proportional to T? but we want bigger moves early?
        delta = max_delta * rng.random()
        # Enforce non-negative
        if delta <= 0: continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        diff = new_score - cur_score
        if diff <= 0 or rng.random() < np.exp(-diff / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject move, revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score
```

def simulated_annealing(a, cur_score, fft_len, n, deadline,
                        start_temp=0.02, end_temp=1e-6,
                        move_frac_max=0.25):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total_time)
        # compute temperature linearly or geometric
        t_frac = elapsed / total_time
        T = start_temp * (1 - t_frac) + end_temp * t_frac
        # pick i
        i = rng.integers(0, n)
        if cur[i] <= 0:
            continue
        # pick j not equal i
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # choose delta relative to cur[i]
        max_delta = cur[i] * move_frac_max
        if max_delta < 1e-12:
            continue
        delta = max_delta * rng.random()
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        diff = new_score - cur_score
        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject
            cur[i] += delta
            cur[j] -= delta
        iteration += 1
        # occasional recompute T using updated start time
    return best, best_score

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
------------------
* Initialise a probability vector a (∑a = 1).  Re‑use the best
  sequence from a previous run if available, otherwise start from
  a uniform distribution of length n≈5000.
* Global optimisation:
    - Multi‑restart continuation L‑BFGS on the smooth‑max surrogate of
      the auto‑convolution.  A finer λ‑schedule (50 → 70 stages) and a
      larger per‑stage budget (1200 → 1500 iterations) are used to
      obtain higher‑quality local minima.
* Local refinements:
    - Targeted hill‑climb (reduces the current worst convolution entry).
    - Random hill‑climb (mass‑transfer between random coordinates).
* Exact max optimisation:
    - Projected sub‑gradient descent on the true max of the convolution.
* Final polishing:
    - Simulated‑annealing (mass‑transfer moves accepted with a temperature
      schedule) followed by a short random hill‑climb.
* The best sequence seen is stored globally as ``height_sequence_1`` and
  finally evaluated with the supplied ``evaluate_sequence`` function.

The search runs for the whole time budget (default 1000 s) and prints
intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (quadratic) and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # back‑propagation to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing λ.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (via FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose a destination with a tiny partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max of the auto‑convolution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        iter_no += 1
        # compute convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k
        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        # decreasing step size
        step = step0 / np.sqrt(iter_no)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional random acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 0.02,
    end_temp: float = 1e-6,
    move_frac_max: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex:
    move a random fraction of mass from i to j.
    Accept uphill moves with probability exp(-Δ/ T).
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-12)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (elapsed / total_time)

        # pick distinct indices
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur[i] * move_frac_max
        if max_delta <= 1e-14:
            continue
        delta = max_delta * rng.random()
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        diff = new_score - cur_score
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The function runs for approximately ``budget_s`` seconds and returns
    the best non‑negative sequence discovered.
    """
    rng = np.random.default_rng(seed)
    # keep a safety margin for the final evaluation call
    deadline = time.time() + budget_s - 4.0

    # --------------------------------------------------------------
    # Problem size – reuse previous large construction if possible
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = np.maximum(arr, 0.0)
            if best_a.sum() == 0.0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a /= best_a.sum()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Ensure we are on the probability simplex
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Global L‑BFGS search (many restarts)
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-13, num=70)   # finer λ schedule
    maxiter_per_stage = 1500                        # more work per stage

    iter_no = 0
    while time.time() < deadline - 20.0:
        # diversify start point
        mode = rng.choice(["random", "exp", "power", "edge", "best", "perturb"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.03)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.05)))
            edge_val = 0.7 / n
            centre_val = (1.0 - 2 * m_edge * edge_val) / (n - 2 * m_edge)
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "best":
            y0 = best_y + rng.normal(scale=0.05, size=n)
        else:  # perturb
            y0 = best_y + rng.normal(scale=0.08, size=n)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iter_no}] mode={mode} → new best fast score = {best_score:.6f}")

        # quick targeted hill‑climb
        t_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1500,
            move_frac=0.35,
        )
        best_y = np.log(best_a + 1e-12)

        # quick random hill‑climb
        r_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=5000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

        iter_no += 1

    # --------------------------------------------------------------
    # Exact max optimisation via sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = deadline - 5.0
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=sub_dead,
            max_iter=5000,
            step0=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Simulated annealing for fine‑grained improvement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            start_temp=0.02,
            end_temp=1e-6,
            move_frac_max=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final short random hill‑climb (polish)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=5000,
            max_move_frac=0.15,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # The harness will have imported ``evaluate_sequence``.
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
