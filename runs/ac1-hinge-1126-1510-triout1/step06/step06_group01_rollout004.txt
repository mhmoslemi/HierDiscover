def lp_improvement_step(f, deadline):
    # compute M
    n = f.shape[0]
    fft_len_conv = next_pow2(2*n - 1)
    conv = full_conv(f, fft_len_conv)
    M = conv.max()
    # build sparse constraint matrix
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.arange(n).repeat(n)
    data = np.tile(f, n)  # f repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # solve LP
    b_ub = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # maximize sum(g0)
    res = linprog(c=c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return f, float('inf')
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return f, float('inf')
    g = g0 / sum_g0
    # mixing
    t = 0.2
    f_new = (1 - t) * f + t * g
    # renormalize to sum 1 (should already be)
    f_new = f_new / f_new.sum()
    # compute new score
    score = fast_score(f_new, fft_len_conv, n)
    return f_new, score

"""Improved search for a step‑function with a low evaluation score.

The algorithm consists of three main stages (all run inside the
time‑budget that is given to ``propose_candidate``):

1. **Continuation L‑BFGS** – many random restarts on a smoothed
   log‑sum‑exp (LSE) surrogate of the maximum auto‑convolution.
   The smoothing factor is decreased progressively.

2. **Linear‑programming mixing** – given the current distribution *f*,
   solve the LP
       maximise  Σ g_j   subject to   f * g ≤ M ,   g ≥ 0
   where *M* is the present maximum of ``f*f``.
   The optimal *g* (after normalisation) is mixed with *f*.
   Repeating this step strictly reduces the peak of the auto‑convolution.

3. **Final refinement** – a short projected sub‑gradient descent with
   momentum followed by a deterministic greedy hill‑climb.

All convolutions are performed with FFTs (O(n log n)).  The search works
with a moderate sequence length (default n≈1500) which balances
solution quality and the ability to use dense linear programs during
the LP‑mixing phase.  The returned list of floats is guaranteed to be
non‑negative; the external ``evaluate_sequence`` function is called only
once at the end.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a via FFT (length 2·len(a)‑1).
    Result is the first 2·len(a)-1 entries of the circular convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a)  for a distribution that already sums to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    fvalue = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return fvalue, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Heavy L‑BFGS continuation on the smoothed LSE surrogate.
    Returns the best distribution found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # map to logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best {best_score:.6f}")
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_mixing_step(f: np.ndarray, deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    One LP‑based improvement:
      maximise Σ g_j   s.t.   f * g ≤ M ,   g ≥ 0
    where M is the present maximum of (f*f).
    The solution g is normalised to sum‑1 and mixed with f.
    Returns the new distribution and its fast_score.
    """
    if time.time() >= deadline:
        return f, fast_score(f, next_pow2(2 * n - 1), n)

    fft_len = next_pow2(2 * n - 1)
    conv_f = full_conv(f, fft_len)
    M = conv_f.max()

    # Build the Toeplitz‐like constraint matrix A (shape (2n‑1) × n)
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.arange(n).repeat(n)
    data = np.tile(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Solve the LP: maximise Σ g_j   <=> minimise -Σ g_j
    c = -np.ones(n)                     # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    res = linprog(c=c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # If LP fails, fallback to the current f.
        return f, fast_score(f, fft_len, n)

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return f, fast_score(f, fft_len, n)

    # Normalise to the probability simplex
    g = g0 / sum_g0

    # Mixing step (use a modest step size to guarantee stability)
    t = 0.2
    f_new = (1.0 - t) * f + t * g
    f_new = f_new / f_new.sum()

    new_score = fast_score(f_new, fft_len, n)
    return f_new, new_score
