def exponentiated_gradient_descent(a_start, fft_len, n, deadline, eta0=0.2):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time()<deadline and it < 500000:
        it+=1
        conv = conv_fft(a, fft_len, n)
        m = np.argmax(conv)
        # gradient simplified: g_i = 2*a_{m-i} if 0<=m-i<n else 0
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >=0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # learning rate decays
        eta = eta0 / np.sqrt(it)
        # multiplicative update
        a = a * np.exp(-eta * g)
        # renorm
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
            continue
        a = a / sum_a
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def simulated_annealing(a_start, fft_len, n, deadline, init_temp=0.01, final_temp=1e-5):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    T0 = init_temp
    Tf = final_temp
    total_steps = 0
    # Determine total steps based on time remaining (approx)
    while time.time() < deadline:
        elapsed = time.time() - start
        # maybe compute temperature decreasing linearly with time
        frac = (deadline - time.time())/(deadline - start)
        T = max(Tf, T0 * frac)  # linear cooling
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.5
        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a, fft_len, n)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        total_steps += 1
    return best_a, best_score

"""
Propose a low‑score step‑function.

Algorithm Overview
------------------
1. **Problem size** – Use a moderately large `n` (≈ 8000) unless a longer
   previous sequence is supplied.  All work is performed on the probability
   simplex (∑a = 1) because the evaluation is scale‑invariant.

2. **Fast utilities** – Convolution is computed by FFT, and the objective
   `2·n·max(conv)` (denominator is 1) is evaluated by `fast_score`.

3. **Phase 0 – Simple pattern sweep** – Try a handful of hand‑crafted families
   (power‑law, exponential, Gaussian bump) and keep the best.

4. **Phase 1 – Multi‑restart L‑BFGS** – For many random initialisations a
   continuation L‑BFGS is run on a log‑sum‑exp smooth approximation of the
   max‑convolution.  The schedule gradually reduces the smoothing factor.
   After each run a brief greedy hill‑climb is applied.

5. **Phase 2 – Sub‑gradient / Exponentiated‑gradient descent** –  
   *Projected sub‑gradient descent* on the true max‑convolution reduces the
   active peak.  Then an *exponentiated‑gradient* (multiplicative‑weights)
   update is performed on the same objective.

6. **Phase 3 – Simulated annealing** – A Metropolis walk that moves a random
   fraction of mass from one index to another, accepting worsening moves
   according to a slowly cooling temperature schedule.

7. **Phase 4 – Fine L‑BFGS refinement** – Very low smoothing (λ≈10⁻¹²) with
   generous iteration limits.

8. **Phase 5 – Aggressive hill‑climb** – Greedy mass‑transfer moves with a
   larger step size finish the search.

9. **Return** – Convert the best probability vector to a plain list of floats
   and hand it to `evaluate_sequence`.  All numbers are non‑negative and
   automatically satisfy the evaluation’s clamping constraints.

The procedure respects the 1000 s budget, adapts its time allocation,
and makes no assumptions about the external `evaluate_sequence` implementation.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
    2·n·max(conv)  (the denominator ∑a² = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector `y` (soft‑max parametrisation).

    The smoothing parameter is `lam = max(conv) * lam_factor`.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv w.r.t a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation.
    Each stage uses a different smoothing factor from `lam_schedule`.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point `y` (pre‑softmax) for a given mode.
    Modes: random, exp, power, edge, sidon, binary, bump, harmonic.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = any((candidate + c) in sums or (2 * candidate) in sums for c in chosen)
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a) = 2·n·max(conv) .
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))  # index of maximal entry

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def exponentiated_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative‑weights (exponentiated) gradient descent on the true max‑conv.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)  # multiplicative update
        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_temp: float = 0.005,
    final_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style annealing: random mass transfers with a decreasing temperature.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline:
        # linear cooling based on remaining fraction of time
        frac = max(0.0, (deadline - time.time()) / (deadline - start))
        T = final_temp + (init_temp - final_temp) * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = a[i] * rng.random() * 0.5
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # brief greedy hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.70 * budget_s)
    print("[phase2] projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=2_000_000,
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 2b – Exponentiated‑gradient descent (multiplicative updates)
    # ------------------------------------------------------------------
    expgrad_deadline = min(deadline - 15.0, time.time() + (deadline - time.time()) * 0.1)
    print("[phase2b] exponentiated‑gradient descent")
    best_a, best_score = exponentiated_gradient_descent(
        best_a,
        fft_len_conv,
        n,
        expgrad_deadline,
        eta0=0.30,
        max_iters=500_000,
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("[phase3] simulated annealing")
        best_a, best_score = simulated_annealing(
            best_a,
            fft_len_conv,
            n,
            deadline - 10.0,
            rng,
            init_temp=0.005,
            final_temp=1e-6,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 4 – Fine L‑BFGS refinement (very low smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=30))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=800,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Aggressive hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.35,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
