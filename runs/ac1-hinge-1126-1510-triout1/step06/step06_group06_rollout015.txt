    a = softmax(y)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

    w = conv**(p-1)

"""
Improved step‑function optimisation.

Main ideas
-----------
* Use the exact sub‑gradient of the *max* autocorrelation (the same
  gradient that the original code used for a projected sub‑gradient step)
  but apply it with a *multiplicative* (exponentiated‑gradient) update
  on the simplex.  This very often moves mass faster away from the
  indices that dominate the current maximal convolution entry.
* Run the multiplicative descent until the time budget is almost spent,
  then finish with the targeted‑ and random‑hill‑climbs that were already
  proven useful.
* Initialise from the best known construction (global variable
  ``height_sequence_1``) but also blend in a 1/√i power‑law seed – this
  gives the optimiser a much richer starting point.
* All heavy work uses FFT‑based convolutions (O(n log n)) and therefore
  easily fits into the 1000 s budget even for n≈5000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Small utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv(a,a)) – the actual evaluation when sum(a)=1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal entry of a*a and a sub‑gradient of that max.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = float(conv.max())
    k_max = int(np.argmax(conv))

    # map to the symmetric half 0 ≤ k ≤ n‑1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Classic random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 1500,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that moves mass away from the pair of indices
    that contributes most to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def exp_gradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the simplex.
    Uses the exact sub‑gradient of the *max* autocorrelation.
    """
    a = a0.copy()
    # current score = 2·n·max_c (because sum(a)=1)
    max_c, grad = max_convolution_subgrad(a, n, fft_len)
    best_score = 2.0 * n * max_c
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # step size that decays like 1/√t
        eta = eta0 / np.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)          # avoid exact zeros
        a = a / a.sum()                    # simplex projection (automatic)

        # evaluate new point
        max_c, grad = max_convolution_subgrad(a, n, fft_len)
        cur_score = 2.0 * n * max_c

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # occasionally print progress
            if it % 500 == 0:
                print(f"[exp‑gd] iter={it} score={best_score:.6f}")

    return best_a, best_score
