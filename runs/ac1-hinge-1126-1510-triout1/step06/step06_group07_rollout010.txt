def swap_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000, move_frac=0.2):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # pick two distinct sources and two distinct destinations
        i1, i2, j1, j2 = rng.choice(n, size=4, replace=False)
        # compute amount to move from each source
        delta1 = best[i1] * move_frac * rng.random()
        delta2 = best[i2] * move_frac * rng.random()
        if delta1 <= 0 or delta2 <=0:
            continue
        # apply moves
        best[i1] -= delta1
        best[i2] -= delta2
        best[j1] += delta1
        best[j2] += delta2
        # evaluate
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i1] += delta1
            best[i2] += delta2
            best[j1] -= delta1
            best[j2] -= delta2
    return best, best_score

"""
Hybrid multi‑restart optimization for the step‑function min‑max convolution problem.

The algorithm mixes:
  * Continuation L‑BFGS on a smooth Log‑Sum‑Exp approximation of the max,
    starting from a variety of parameterised initial shapes.
  * Aggressive targeted and random hill‑climbing on the simplex.
  * A swap‑based hill‑climb that moves mass between two sources and two
    destinations in one step.
  * Projected sub‑gradient descent on the exact max convolution.
All phases run until the 1000 s budget expires.  The best sequence found is
returned and stored as ``height_sequence_2`` for possible reuse.
"""
import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smoothed max of the autocorrelation and its gradient w.r.t. log‑soft‑max."""
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak at zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # back‑propagation to a
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)

def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """Continuation L‑BFGS over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------
def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focused hill‑climb that attacks the currently worst shift."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose a destination whose partner under shift k is light
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=8)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def swap_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """Two‑source / two‑dest mass swap hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # pick four distinct indices
        i1, i2, j1, j2 = rng.choice(n, size=4, replace=False)
        # move a portion from i1,i2 to j1,j2
        d1 = best[i1] * move_frac * rng.random()
        d2 = best[i2] * move_frac * rng.random()
        if d1 <= 0.0 or d2 <= 0.0:
            continue
        best[i1] -= d1
        best[i2] -= d2
        best[j1] += d1
        best[j2] += d2
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i1] += d1
            best[i2] += d2
            best[j1] -= d1
            best[j2] -= d2
    return best, best_score

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent w.r.t. the exact max of the auto‑convolution."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k: (gradient_i = 2 * a_{k-i})
        grad = np.zeros_like(best)
        # fill only where partner exists
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        # i runs from low..high, partner = k-i
        grad[low:high + 1] = 2.0 * best[k - np.arange(low, high + 1)]

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score

    else:

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)

            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focused hill‑climb that attacks the currently worst shift."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=8)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def swap_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """Two‑source / two‑dest mass swap hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i1, i2, j1, j2 = rng.choice(n, size=4, replace=False)

        d1 = best[i1] * move_frac * rng.random()
        d2 = best[i2] * move_frac * rng.random()
        if d1 <= 0.0 or d2 <= 0.0:
            continue

        best[i1] -= d1
        best[i2] -= d2
        best[j1] += d1
        best[j2] += d2

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i1] += d1
            best[i2] += d2
            best[j1] -= d1
            best[j2] -= d2
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent w.r.t. the exact max of the auto‑convolution."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k: (gradient_i = 2 * a_{k-i}) where defined
        grad = np.zeros_like(best)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        grad[low : high + 1] = 2.0 * best[k - np.arange(low, high + 1)]

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            if rng.random() < 0.001:   # tiny stochastic acceptance
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The routine follows a multi‑restart continuation‑L‑BFGS phase,
    aggressive hill‑climbing, and a final projected sub‑gradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin for final call

    # --------------------------------------------------------------
    # Choose problem size – reuse a previously found construction if possible
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT sizes (constant throughout the run)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-12, num=60)   # smoother cooling schedule
    maxiter_per_stage = 1200                         # more work per L‑BFGS stage

    iteration = 0
    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 15.0:
        # ---- pick a diverse starting point ----------------------
        mode = rng.choice(
            ["random", "exp", "power", "edge", "inv_sqrt", "beta", "best"]
        )
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = rng.uniform(0.01, 0.04)
            centre_val = 1.0 / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "inv_sqrt":
            i = np.arange(n, dtype=float)
            eps = 1e-6
            a0 = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "beta":
            alpha = rng.uniform(0.3, 2.0)
            beta = rng.uniform(0.3, 2.0)
            x = (np.arange(1, n + 1) - 0.5) / n
            a0 = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
            a0 += 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0)
        else:   # mode == "best"
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ---- continuation L‑BFGS ---------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}")

        # ---- targeted hill‑climb --------------------------------
        t_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=800,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- random hill‑climb ----------------------------------
        r_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=6000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- swap hill‑climb ------------------------------------
        s_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = swap_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=s_dead,
            max_iter=1200,
            move_frac=0.18,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- sub‑gradient descent -------------------------------
        sg_dead = min(deadline, time.time() + 1.0)
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=sg_dead,
            max_iter=2500,
            step0=0.27,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final aggressive random hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_2"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
