def propose_candidate(seed=42, budget_s=1000):
   start_time = time.time()
   deadline = start_time + budget_s - 2.0

   n = 12000
   fft_len = next_pow2(2*n - 1)
   fft_len_g = next_pow2(3*n - 2)   # for gradient conv; could be same as fft_len

   # initialization: best_a uniform
   best_a = uniform_init(n)
   best_score = fast_score(best_a, fft_len, n)

   # define lambda schedule slightly broader: [0.5, 0.2, 0.1, 0.05, 0.02,0.01,... 1e-9]
   lam_schedule = list(np.geomspace(0.5, 1e-9, num=60))

   # configure L-BFGS maxiter per stage = 800 for more thorough optimize.
   maxiter_per_stage = 800

   # Search loop:
   modes = ["arcsine","semicircle","uniform","random","binary","bump"]
   rng = np.random.default_rng(seed)

   while time.time() < deadline * 0.6:   # use 60% for L-BFGS
        mode = rng.choice(modes)
        y0 = generate_initial_y(mode, n, rng)
        # add small noise to best_y also sometimes
        if rng.random() < 0.2:
            y0 = best_y + rng.normal(scale=0.01, size=n)

        y_opt = schedule_optimize(y0, n, fft_len, fft_len_g, lam_schedule, maxiter_per_stage)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[L-BFGS] mode={mode}, score={best_score:.6f}")

        # quick hill climb
        limit = min(deadline - time.time(), 20)   # run hill climb <20 secs
        if limit > 1:
            hc_deadline = time.time() + limit
            best_a, best_score = hill_climb(best_a, best_score, fft_len, n, hc_deadline, max_iter=2000)
            best_y = np.log(best_a + 1e-12)

   # Then continue with Adam subgradient descent until time contains remaining.

   # Adam implementation
   a = best_a.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1, beta2, eps = 0.9, 0.999, 1e-8
   t = 0
   while time.time() < deadline * 0.9:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        # compute subgradient g_i = 2*a_{m_idx - i}, if valid.
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # update m,v
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)   # adapt step per coordinate
        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[Adam] iter {t}, score {best_score:.6f}")
        a = a_new

   # final hill climb
   if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a, best_score, fft_len, n, final_deadline, max_iter=10000, max_move=0.3)
        best_a, best_score = targeted_hill_climb(best_a, best_score, fft_len, n, final_deadline, max_iter=5000, max_move=0.2)

   final_seq = best_a.tolist()
   final_val = evaluate_sequence(final_seq)
   print(f"[final] evaluated score = {final_val:.6f}")
   return final_seq

def adam_subgradient(
    a_start, fft_len, n, deadline,
    alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
    max_iters=10_000_000,
):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional logging
            if t % 10000 == 0:
                print(f"[adam] iter {t}, best score {best_score:.6f}")
        a = a_new
    return best_a, best_score
```

"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑ a_i = 1).  The objective
  `evaluate_sequence` is homogeneous, so the scale cancels out.
* Use a moderately large problem size (n = 12000).  Larger n gives the optimizer
  more freedom while the FFT‑based convolution stays cheap.
* Initialise several analytically motivated shapes (arcsine, semicircle,
  uniform, random Gaussian, sparse binary, Gaussian bump) and also the current
  best sequence (if any).  Each shape is turned into log‑softmax parameters
  ``y`` (so the simplex constraint is automatically satisfied).
* Perform a *continuation L‑BFGS* on a smooth approximation of the max‑convolution
  (log‑sum‑exp).  The smoothing factor λ is decreased geometrically from 0.5 to
  1e‑9 (60 stages).  Each stage runs up to 800 LBFGS iterations.
* After each L‑BFGS run a short random hill‑climb (mass transfer between two
  indices) is applied to guarantee monotone improvement.
* When the bulk of the time budget is exhausted, switch to an *Adam‑style*
  projected sub‑gradient descent on the true objective
  `f(a)=2·n·max(conv(a,a))`.  The gradient of the active convolution entry is
  cheap to compute and the iterate is projected onto the simplex after each
  update.
* Finish with a longer random hill‑climb and a targeted hill‑climb that moves
  mass away from the indices that contribute most to the current maximal
  convolution entry.
* The final sequence is evaluated with the supplied ``evaluate_sequence`` and
  returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector with sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:  # fallback – should never happen
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum = 1): 2·n·max(conv)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic initial shapes

def arcsine_init(n: int, alpha: float = 0.70) -> np.ndarray:
    """β(α,α) density on (0,1) – arcsine‑like for 0<α<2."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution on {0,…,n‑1}."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a log‑softmax vector y for a given mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) – used in L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient w.r.t. y.
    ``lam_factor`` is multiplied by the current maximal convolution value to obtain
    the smoothing constant λ.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- log‑sum‑exp smoothing -------------------------------------------
    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient wrt convolution ----------------------------------------
    w = exp_shift / Z                     # d smooth_max / d conv

    # ---- gradient of convolution wrt a (reverse convolution) ----------
    # conv = a * a  -->  d conv / d a = 2 * (a reversed convolved with unit)
    # Here we need ∂f/∂a = (w  * 2*a)  (convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to each a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through softmax --------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage re‑optimises the smoothed objective for a smaller λ.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2500,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random hill‑climb that transfers a fraction of mass between two indices."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,

    n: int,

    contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j2 = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j2 = rng.integers(0, n)

        if i == j2:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j2] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j2] -= delta
    return best, best_score


def adam_subgradient(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a)).
    """
    a = a_start.copy()
    m_vec = np.zeros_like(a)  # first moment
    v_vec = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry:
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update (on the sub‑gradient)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * (g * g)

        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)

        step = alpha / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            if t % 5000 == 0:
                print(f"[adam] iter {t:6d}, best score {best_score:.6f}")

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin
    main_deadline = start_time + 0.55 * budget_s    # about half the budget for LBFGS

    # ------------------------------------------------------------------
    # Problem size – we deliberately use a fairly large n.
    # ------------------------------------------------------------------
    n = 12000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initial best candidate (uniform probability vector)
    # ------------------------------------------------------------------
    best_a = uniform_init(n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=60))  # finer continuation
    maxiter_per_stage = 800                               # more inner work

    modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump"]
    mode_cycle = 0
    iteration = 0

    while time.time() < main_deadline:
        # occasionally inject a noisy copy of the current best solution
        if iteration % 8 == 0 and iteration > 0:
            mode = "best"
            y0 = best_y + rng.normal(scale=0.015, size=n)
        else:
            mode = modes[mode_cycle % len(modes)]
            mode_cycle += 1
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- short random hill‑climb to polish the LBFGS result -----
        hc_deadline = min(deadline, time.time() + 0.90)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style projected sub‑gradient descent (remaining budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting Adam‑style projected sub‑gradient descent")
    best_a, best_score = adam_subgradient(
        a_start=best_a,
        fft_len=fft_len_conv,
        n=n,
        deadline=subgrad_deadline,
        alpha=0.05,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
        max_iters=5_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=12_000,
            max_move=0.30,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=8_000,
                max_move=0.20,
            )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
