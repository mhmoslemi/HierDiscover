"""
Step‑function optimizer (≈ 1 k s budget)

The goal is to minimise 2·n·max(conv(a,a)) / (sum a)²
for a non‑negative vector a (scale‑invariant, we keep sum(a)=1).

Algorithm
---------
1. **Initial pool** – deterministic windows (Hann, Blackman, Kaiser,
   Chebyshev, DPSS) + several random Dirichlet‑like draws + the previous
   best sequence (if any).  Keep the best by the exact FFT‑based score.

2. **Phase 1 – L‑BFGS with log‑sum‑exp smoothing** – minimize a smooth
   surrogate of the max‑convolution.  A geometric schedule of the
   smoothing parameter λ from 5·10⁻¹ down to 5·10⁻⁸ is used, and a few
   random restarts are performed.  The search works in the soft‑max
   (logits) space so that the simplex constraint is automatic.

3. **Phase 2 – Linear‑program refinement** – given the current a,
   solve a linear programme  
        maximise   1ᵀb
        subject to   T(a)·b ≤ M   ,   b ≥ 0
   where T(a) is the Toeplitz matrix of a and M = max(conv(a,a)).
   If the optimiser returns b with sum(b)>1, we rescale b to unit sum;
   this strictly reduces the max‑convolution and therefore the
   objective.  The LP is solved with SciPy’s HiGHS solver (sparse
   matrix, O(n²) non‑zeros).  We repeat the LP a few times (or until
   no improvement).

4. **Phase 3 – Fine‑grained L‑BFGS** – run a short L‑BFGS refinement
   starting from the LP‑improved vector.

5. **Phase 4 – Greedy hill‑climb** – tiny pairwise mass transfers are
   accepted only if they improve the exact score.  This polishes the
   solution to a local optimum.

All heavy work uses FFT‑based convolution (`fast_score`) which evaluates
the exact objective for sum(a)=1.  The routine respects the 1000 s time
budget and finally returns the best list of floats.  The result is
checked with the provided `evaluate_sequence` function.
"""

import time
import numpy as np
import scipy.optimize
import scipy.signal
import scipy.sparse

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------
def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a

def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = scipy.signal.windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = scipy.signal.windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = scipy.signal.windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    x → a = softmax(y)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, starting each stage from the
    previous result.  Returns the best vector found and its exact fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = scipy.optimize.minimize(fun=objective_and_grad,
                                          x0=y,
                                          args=(n, fft_len_conv, fft_len_g, lam),
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': maxiter_per_lam,
                                                   'gtol': 1e-12,
                                                   'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc

# ----------------------------------------------------------------------
# Linear‑program refinement
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              n: int,
              fft_len: int,
              deadline,
              max_iter: int = 10,
              verbose: bool = True):
    """
    Perform LP refinement: maximise sum(b) subject to Toeplitz(a)·b ≤ M,
    b ≥ 0.  The resulting b is rescaled to unit sum and accepted if it
    improves the exact fast_score.  The process is repeated a few times.
    """
    # Pre‑compute constant index arrays for the Toeplitz matrix
    rows = np.add.outer(np.arange(n, dtype=np.int32),
                        np.arange(n, dtype=np.int32)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int32), n)

    shape = (2 * n - 1, n)

    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # current bound M
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build sparse Toeplitz matrix with current a
        data = np.repeat(best_a, n)   # length n*n
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=shape)

        # Solve LP: maximise 1ᵀb  ⇔ minimise -1ᵀb
        res = scipy.optimize.linprog(c=-np.ones(n),
                                     A_ub=A,
                                     b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                                     bounds=[(0, None)] * n,
                                     method='highs')
        if not res.success:
            if verbose:
                print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            if verbose:
                print("[LP] iteration {}: sum(b) non‑positive, abort".format(it))
            break

        # Rescale to unit sum (preserves feasibility and reduces max‑conv)
        b_scaled = b / sum_b

        # Clip to avoid >1000 after scaling (very unlikely)
        if b_scaled.max() > 1000.0:
            b_scaled = np.clip(b_scaled, 0.0, 1000.0)
            b_scaled /= b_scaled.sum()

        # Symmetrise (optional – often beneficial)
        b_scaled = (b_scaled + b_scaled[::-1]) * 0.5

        sc = fast_score(b_scaled, fft_len, n)
        if sc + 1e-12 < best_sc:          # strict improvement
            best_sc = sc
            best_a = b_scaled.copy()
            if verbose:
                print(f"[LP] iter={it}, improved fast_score={best_sc:.8f}")
        else:
            # No improvement – stop early
            if verbose:
                print(f"[LP] iter={it}, no improvement (score={sc:.8f})")
            break
        it += 1
    return best_a, best_sc

# ----------------------------------------------------------------------
# Simple greedy hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.03):
    """
    Deterministic improvement by tiny pairwise mass transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 4096 works well within the time budget
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build initial pool
    # ------------------------------------------------------------------
    candidates = []

    # previous solution (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Interpolate to current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, prev)
            a_old = np.clip(a_old, 0.0, None)
            if a_old.sum() > 0:
                a_old = a_old / a_old.sum()
            else:
                a_old = np.full(n, 1.0 / n)
            candidates.append(a_old)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(chebwin_init(n, attenuation=rng.uniform(40.0, 80.0)))
    candidates.append(dpss_init(n, half_bw=rng.uniform(1.5, 5.0)))

    # random families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # beta‑shaped families
    for _ in range(8):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # ------------------------------------------------------------------
    # Keep the best deterministic start (exact score)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS (multiple random restarts)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = np.geomspace(5e-1, 5e-8, num=50)   # finer schedule
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # random blend of current best and a fresh random candidate
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑program refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.20 * budget_s
        print("[LP] start refinement")
        best_a, best_sc = lp_refine(best_a, n, fft_len, lp_deadline,
                                    max_iter=12, verbose=True)
        print(f"[LP] best fast_score after refinement={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – short L‑BFGS fine‑tuning (starting from LP result)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        ft_deadline = time.time() + 0.15 * budget_s
        print("[L‑BFGS‑fine] start")
        # using a tighter lam schedule
        lam_fine = np.geomspace(5e-2, 5e-9, num=40)
        a_fine, sc_fine = lbfgs_continuation(best_a,
                                             lam_fine,
                                             maxiter_per_lam=1500,
                                             deadline=ft_deadline,
                                             fft_len_conv=fft_len,
                                             fft_len_g=fft_len,
                                             n=n)
        if sc_fine < best_sc:
            best_sc = sc_fine
            best_a = a_fine.copy()
            print(f"[L‑BFGS‑fine] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill‑climb] start polishing")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.02)
        print(f"[Hill‑climb] final fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final sanity‑check & return
    # --------------------------------------------------------------
    final_seq = np.clip(best_a, 0.0, None)
    total = final_seq.sum()
    if total <= 0.0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / total

    final_seq_list = final_seq.tolist()
    # official evaluation (only once)
    final_eval = evaluate_sequence(final_seq_list)
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
