def propose_candidate(seed=0,budget_s=1000):
    rng = np.random.default_rng(seed)
    n=4096
    fft_len = next_pow2(2*n-1)

    # init candidates
    candidates = []
    # if prev exists load
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim==1 and prev.size>0:
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # deterministic families
    candidates += [spectral_init(n), hann_init(n), blackman_init(n),
                   gaussian_init(n), kaiser_init(n, beta=rng.uniform(6,12))]
    # random many
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    # beta
    for _ in range(4):
        candidates.append(beta_init(n, rng.uniform(0.2,2.5), rng.uniform(0.2,2.5)))

    # keep best
    best_a = None
    best_sc = float('inf')
    for a in candidates:
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    print("[init] best fast_score", best_sc)

    # Phase 1 L-BFGS extended
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    phase1_end = start_time + 0.4 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-7, num=50)
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # random start near best
        base = random_initial(n, rng)
        alpha = rng.random() * 0.5
        a0 = (1-alpha)*best_a + alpha*base
        a0 = a0 / a0.sum()
        a_opt, sc_opt = lbfgs_continuation(a0, lam_schedule, maxiter_per_lam, phase1_end, fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print("[L-BFGS-global] new best", best_sc)

    # Phase 2 EG with smaller top fraction
    if time.time() < deadline:
        eg_deadline = time.time() + 0.15 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a, eg_deadline, n, fft_len,
                                                top_fraction=0.012)
        print("[EG] best", best_sc)

    # Phase 3 subgradient longer
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * budget_s
        print("[SubGrad] start")
        best_a, best_sc = subgradient_optimize(best_a, sub_deadline, n, fft_len, rng)
        print("[SubGrad] best", best_sc)

    # Phase 4 Adam longer
    if time.time() < deadline:
        print("[Adam] start")
        best_a, best_sc = adam_optimize(best_a, deadline, n, fft_len, rng)
        print("[Adam] best", best_sc)

    # Phase5 LP refine with more active constraints
    if time.time() < deadline:
        lp_deadline = time.time() + min(40.0, deadline - time.time())
        print("[LP] start")
        best_a, best_sc = lp_refine(best_a, lp_deadline, n, fft_len,
                                    max_active=60, init_step=0.15)
        print("[LP] best", best_sc)

    # Phase6 SA longer
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a, sa_deadline, n, fft_len, rng,
                                            init_temp=1e-3, final_temp=1e-8,
                                            max_moves=150000)
        print("[SA] best", best_sc)

    # Phase7 Hillclimb final
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill-climb] start")
        best_a, best_sc = hill_climb(best_a, best_sc, fft_len, n, hill_deadline, rng,
                                     max_iter=50000, delta_frac=0.03)
        print("[Hill-climb] best", best_sc)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score", final_eval)
    return final_seq

"""
Enhanced step‑function search.

The algorithm builds a diverse pool of seeds (deterministic windows,
random Dirichlet draws, beta‑shapes, etc.), then performs a multi‑phase
refinement:

1️⃣  Extensive L‑BFGS on a log‑sum‑exp smoothed “max‑convolution”.
    A finer λ‑schedule and more iterations per stage give a stronger
    local optimum.

2️⃣  Exponentiated‑gradient (EG) descent that attacks only the largest
    convolution entries (tiny top‑fraction).

3️⃣  Projected‑gradient descent on the true non‑smooth objective (exact
    max‐convolution).  After each step we project onto the probability
    simplex, automatically respecting non‑negativity and mass
    conservation.

4️⃣  Adam optimisation on the smoothed objective (log‑sum‑exp).

5️⃣  Linear‑programming refinement with a larger active set of binding
    constraints.  The tiny LP maximises the total mass of a candidate
    under the active convolution constraints, then the design is
    convex‑combined with a slowly decaying step.

6️⃣  Simulated‑annealing (pairwise mass transfers) for a longer schedule.

7️⃣  Final greedy hill‑climb.

All phases respect a global 1000 s deadline.  The best sequence found
is returned and finally evaluated with the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f   = 2·n·smooth_max(conv)
       grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # linear convolution in frequency domain
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # <= 0
    exp_shift = np.exp(shifted / lam)   # (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root of uniform spectrum) init."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on a uniform grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in the schedule, chaining the result.
    Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.01):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)

        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def proj_gradient_optimize(a_start: np.ndarray,
                           deadline: float,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator):
    """
    Projected gradient descent on the exact max‑convolution.
    After each step the iterate is projected onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    eta = 1.0           # initial step size
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # candidate step
        cand = a - eta * g
        cand = simplex_projection(cand)

        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_sc:
            best_sc = sc_cand
            best_a = cand.copy()
            a = cand
            eta = min(1.5, eta * 1.05)   # accelerate when successful
        else:
            eta *= 0.5                     # back‑track otherwise

        # occasional random shake to escape tiny plateaux
        if rng.random() < 0.001:
            a = random_initial(n, rng)
            a /= a.sum()
        it += 1
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """Adam optimiser on the smoothed max‑convolution objective."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 120000):
    """Cheap SA with pairwise mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb (pairwise transfers) with strict improvement."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              max_active: int = 30,
              init_step: float = 0.2):
    """
    Repeatedly solve a tiny LP that maximises Σb under the active
    convolution constraints f * b ≤ M, where M is the current peak.
    The solution b is mixed into f with a small convex step.
    """
    f = a_start.copy()
    best_f = f.copy()
    best_sc = fast_score(best_f, fft_len, n)
    t = init_step
    iteration = 0

    while time.time() < deadline:
        fa = np.fft.rfft(f, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active constraints: up to max_active largest values
        if max_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            topk = max_active
            active_idx = np.argpartition(-conv, topk - 1)[:topk]

        rows = []
        cols = []
        data = []
        f_rev = f[::-1]

        for r, k in enumerate(active_idx):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                b_idx = k - j
                coeff = f_rev[j]
                rows.append(r)
                cols.append(b_idx)
                data.append(coeff)

        A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M)

        c = -np.ones(n)                     # maximise Σb  ⇔ minimise -Σb
        bounds = [(0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("[LP] linprog failed – terminating LP phase")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b                     # normalise to simplex

        f = (1.0 - t) * f + t * g
        sc = fast_score(f, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[LP] iter={iteration}, new best fast_score={best_sc:.8f}")

        t = max(0.01, t * 0.995)              # decay step size
        iteration += 1

    return best_f, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # Use the globally stored best sequence, if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # Randomised families
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families (including U‑shapes)
    for _ in range(8):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # Keep the best of the pool
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS (more aggressive)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s   # 40 % of time
    lam_schedule = np.geomspace(0.5, 5e-7, num=50)   # finer schedule
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # blend the current best with a fresh random point
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.12 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.012)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected‑gradient on exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        pg_deadline = time.time() + 0.12 * budget_s
        print("[ProjGrad] start")
        best_a, best_sc = proj_gradient_optimize(best_a,
                                                 pg_deadline,
                                                 n,
                                                 fft_len,
                                                 rng)
        print(f"[ProjGrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation on the smooth objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Linear‑programming refinement (bigger active set)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(35.0, deadline - time.time())
        if lp_deadline > time.time():
            print("[LP] start")
            best_a, best_sc = lp_refine(best_a,
                                        lp_deadline,
                                        n,
                                        fft_len,
                                        max_active=60,   # more bindings
                                        init_step=0.15)
            print(f"[LP] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Simulated annealing (longer schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=1e-3,
                                              final_temp=1e-8,
                                              max_moves=150000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=50000,
                                     delta_frac=0.03)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluator rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # the true competition score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
