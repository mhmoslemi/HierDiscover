"""
Propose a low‑score step function.

Algorithm
---------
1. **Small problem size** – n=1024 (fast to optimise, the theoretical
   lower bound ≈ 1 is the same for any n).
2. **Linear‑programming improvement** – for a current distribution `f`
   solve the LP  

       maximise   Σ g_j
       subject to f * g ≤ M   (M = max(f*f))
                  g ≥ 0

   Then replace `f` by the normalised optimiser `g`.  Each LP step
   provably never increases the objective `2·n·max(f*f)`.
3. **L‑BFGS refinement** – smooth approximation of the max‑convolution
   (log‑sum‑exp) with a geometric schedule of smoothing parameters.
4. **Greedy hill‑climb** – cheap pairwise mass transfers to polish the
   solution.
5. **Final sanity‑check** – clip, re‑normalise and evaluate with the
   provided `evaluate_sequence`.

The whole search respects a 1000 s budget; intermediate scores are printed.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
from scipy.signal import windows

# ----------------------------------------------------------------------
# Minor utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective for a normalised vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    x → a = softmax(y)
    """
    a = softmax(y)

    # convolution a*a (linear, zero‑padded)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Random initialisers (simple families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random point on the simplex.
    """
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple smoothing levels)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`.
    Returns best vector found (still on the simplex) and its fast_score.
    """
    y = np.log(a_start + 1e-12)   # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (pairwise mass transfers)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------


def lp_improve(a_start: np.ndarray, deadline: float):
    """
    Repeatedly solve the LP
        maximise Σ g_j   subject to   f*g ≤ M,   g ≥ 0
    and replace f with the normalised optimiser.
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    # pre‑compute the index structure of the Toeplitz matrix (independent of a)
    I = np.arange(n).reshape(-1, 1)          # shape (n,1)
    J = np.arange(n).reshape(1, -1)          # shape (1,n)
    row_ind = (I + J).ravel()                # length n*n
    col_ind = np.repeat(np.arange(n), n)     # length n*n

    while time.time() < deadline:
        M = np.max(np.convolve(a, a))

        # Build sparse constraint matrix A_{ub} * g ≤ M
        data = np.tile(a, n)                 # a repeated for each column
        A = sp.csr_matrix((data, (row_ind, col_ind)),
                          shape=(2 * n - 1, n))

        b = np.full(2 * n - 1, M)
        c = -np.ones(n)                      # maximise Σg  → minimise −Σg

        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            # LP failed – quit the LP phase
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                       # keep on the simplex

        # Evaluate the new candidate
        sc = fast_score(g, fft_len, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = g.copy()
            a = g.copy()
            print(f"[LP] new best fast_score={best_sc:.8f}")
        else:
            # No improvement – stop the LP loop
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps the LP tractable but still
    # allows a score close to the theoretical lower bound ≈ 1.
    # ------------------------------------------------------------------
    n = 1024
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n)
    best_sc = fast_score(a, fft_len, n)
    print(f"[init] n={n} uniform fast_score={best_sc:.8f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – Linear‑programming improvement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.25 * budget_s
        print("[LP] start")
        a, best_sc = lp_improve(a, lp_deadline)
        print(f"[LP] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement (smooth surrogate of the max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase1_end = time.time() + 0.35 * budget_s
        lam_schedule = np.geomspace(5e-1, 5e-6, num=30)
        maxiter_per_lam = 500

        while time.time() < phase1_end:
            # random restart blended with current best
            a_start = random_initial(n, rng)
            a_start = 0.5 * a + 0.5 * a_start
            a_start = a_start / a_start.sum()
            a_opt, sc_opt = lbfgs_continuation(a_start,
                                               lam_schedule,
                                               maxiter_per_lam,
                                               phase1_end,
                                               fft_len, fft_len, n)
            if sc_opt < best_sc - 1e-12:
                best_sc = sc_opt
                a = a_opt.copy()
                print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print("[Hill‑climb] start")
        a, best_sc = hill_climb(a, best_sc, fft_len, n,
                                hill_deadline, rng,
                                max_iter=200000, delta_frac=0.02)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Final sanity‑check & return
    # ------------------------------------------------------------------
    final_seq = np.clip(a, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official scorer
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
