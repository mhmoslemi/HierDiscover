```
import time, numpy as np, sys
from scipy.optimize import minimize, differential_evolution

def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def fast_score(a, fft_len):
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*len(a)-1]
    return float(2.0 * len(a) * conv.max())
```

Define shape generators:

```
def exponential_shape(n, lam):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0/n)
    a = a / a_sum
    return a

def powerlaw_shape(n, p):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum==0:
        return np.full(n, 1.0/n)
    a = a / a_sum
    return a

def double_power_shape(n, p, q):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** (-p) * (n - i + 1) ** (-q)
    a_sum = a.sum()
    if a_sum==0:
        return np.full(n, 1.0/n)
    a = a / a_sum
    return a
```

```
def lbfgs_refine_simple(a_start, lam_factor_list, maxiter_per, deadline):
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    for lam_factor in lam_factor_list:
        if time.time() > deadline:
            break
        res = minimize(..., args=(n, fft_len, fft_len, lam_factor), method='L-BFGS-B', jac=True, options={'maxiter':maxiter_per,'gtol':1e-12,'disp':False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score
```

```
def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 5000
    fft_len = next_pow2(2*n - 1)

    candidates = []

    # initial families (some from previous code)
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(triangular_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # exponential grid search
    for lam in np.linspace(0.001, 0.02, 30):
        candidates.append(exponential_shape(n, lam))

    # power law grid
    for p in np.linspace(0.1, 2.0, 30):
        candidates.append(powerlaw_shape(n, p))

    # double power grid (coarse)
    for p,q in [(0.5,1.5), (1.0,0.5), (0.8,0.8), (0.3,1.7), (1.7,0.3)]:
        candidates.append(double_power_shape(n, p, q))

    # random families
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    # pick best by fast_score
    best_a = candidates[0]
    best_score = fast_score(best_a, fft_len)
    for a in candidates[1:]:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print("[init] best fast_score =", best_score)

    # L-BFGS refinement
    lam_factors = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter = 4000
    phase1_end = start + 0.5 * budget_s   # half time for refine
    best_a, best_score = lbfgs_refine_simple(best_a, lam_factors, maxiter, phase1_end)
    print("[L-BFGS] best fast_score =", best_score)

    # simple hill-climb mass transfer
    # allocate remaining time
    hill_end = deadline
    best_a, best_score = hill_climb(best_a, best_score, fft_len, n, hill_end, rng, max_iter=800000, delta_frac=0.08)
    print("[Hill] best fast_score =", best_score)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] eval =", final_eval)
    return final_seq
```

"""
Propose a low‑score step sequence.

The algorithm builds a diverse pool of candidate windows (Hann, Blackman,
Kaiser, Gaussian, triangular, inverse‑sqrt, random draws, …) and
augments it with a few simple parametric families (exponential,
power‑law, Kaiser beta, Gaussian sigma).  Each family is optimised with a
cheap differential‑evolution run; the best candidate is taken as a start
point for a longer L‑BFGS refinement on a smoothed max‑convolution
objective, followed by an Adam phase and a final greedy hill‑climb.
All work respects a 1000 s budget and the returned list is fed to the
provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return float(2.0 * len(a) * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def double_power_shape(n: int, p: float, q: float) -> np.ndarray:
    """Two‑sided power law a_i ∝ (i+1)^(-p)·(n-i)^(‑q)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * ((n - i + 1) ** (-q))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    L‑BFGS refinement on a smoothed max‑convolution for a whole λ schedule.
    """
    y = np.log(a_start + 1e-12)  # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    lam_factor: float = 0.001,
    lr: float = 0.08,
    max_iter: int = 40000,
):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 300000,
    delta_frac: float = 0.09,
):
    """
    Strictly improving greedy hill‑climb on the exact fast_score.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)  # legacy safety

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Problem size – 5000 has been a good compromise historically.
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial candidate pool (deterministic + random families)
    # ------------------------------------------------------------------
    candidates = []

    # Deterministic families
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(triangular_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Parameterised families – cheap DE searches
    # ------------------------------------------------------------------

    # Kaiser window with free beta
    def kaiser_obj(x):
        beta = x[0]
        a = kaiser_init(n, beta)
        return fast_score(a, fft_len)

    if time.time() < deadline:
        res = differential_evolution(
            kaiser_obj,
            bounds=[(0.0, 20.0)],
            maxiter=200,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        candidates.append(kaiser_init(n, res.x[0]))
        print("[DE] kaiser beta:", res.x[0], "score:", res.fun)

    # Gaussian window with free sigma factor
    def gauss_obj(x):
        sigma = x[0]
        a = gaussian_init(n, sigma_factor=sigma)
        return fast_score(a, fft_len)

    if time.time() < deadline:
        res = differential_evolution(
            gauss_obj,
            bounds=[(0.05, 0.8)],
            maxiter=200,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        candidates.append(gaussian_init(n, sigma_factor=res.x[0]))
        print("[DE] gaussian sigma:", res.x[0], "score:", res.fun)

    # Simple exponential decay
    def exp_obj(x):
        lam = x[0]
        a = exponential_shape(n, lam)
        return fast_score(a, fft_len)

    if time.time() < deadline:
        res = differential_evolution(
            exp_obj,
            bounds=[(5e-4, 0.02)],
            maxiter=200,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        candidates.append(exponential_shape(n, res.x[0]))
        print("[DE] exponential lam:", res.x[0], "score:", res.fun)

    # Power‑law decay
    def pow_obj(x):
        p = x[0]
        a = powerlaw_shape(n, p)
        return fast_score(a, fft_len)

    if time.time() < deadline:
        res = differential_evolution(
            pow_obj,
            bounds=[(0.1, 3.0)],
            maxiter=200,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        candidates.append(powerlaw_shape(n, res.x[0]))
        print("[DE] power‑law p:", res.x[0], "score:", res.fun)

    # Double‑power (asymmetric) family (tiny grid – cheap)
    for p, q in [(0.7, 1.5), (1.2, 0.8), (0.9, 0.9)]:
        candidates.append(double_power_shape(n, p, q))

    # ------------------------------------------------------------------
    # Pick the best start according to the *exact* fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈45 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 5000

    best_a, best_score = lbfgs_refine(
        best_a,
        lam_schedule,
        maxiter_per_lam,
        phase1_end,
        fft_len,
        n,
    )
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # Random restarts for L‑BFGS (few more chances)
    restarts = 3
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(
            a_start,
            lam_schedule,
            maxiter_per_lam // 2,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            lam_factor=0.001,
            lr=0.05,
            max_iter=80000,
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=800000,
            delta_frac=0.10,
        )
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
