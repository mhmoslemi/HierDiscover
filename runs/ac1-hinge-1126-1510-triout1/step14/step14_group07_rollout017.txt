"""
Propose a step sequence that minimizes the official evaluation function.

The algorithm consists of three stages:

1.  **Broad candidate generation** – deterministic windows, many random
    Dirichlet draws and a few parametric families (exponential decay,
    skewed power‑law and logistic).  Every candidate is evaluated with a fast
    FFT based exact objective (2·n·max(conv)).  The best ~20 candidates are
    retained.

2.  **Intensive polishing** – each retained candidate is refined by a
    multi‑stage L‑BFGS optimisation on a smooth approximation of the
    max‑convolution, followed by a few random restarts.  This phase consumes
    roughly one third of the time budget.

3.  **Fine‑grained descent** – the current best sequence is further improved
    by Adam on the smoothed max, a focused exponentiated‑gradient step,
    exact sub‑gradient descent on the true max, a short simulated‑annealing
    run and finally a strictly improving greedy hill‑climb.

All steps respect the simplex constraints (non‑negative entries, sum = 1)
and the total runtime is limited by the supplied *budget_s* (default 1000 s).
The returned list is ready for the official ``evaluate_sequence`` call.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast objective (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with sum = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def exponential_init(n: int, c: float) -> np.ndarray:
    """Decaying exponential a_i ∝ exp(-c·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def skewed_power_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^α · (n-i)^β.  α,β may be negative.
    """
    i = np.arange(n, dtype=np.float64)
    # use logs to avoid overflow
    log_a = alpha * np.log(i + 1.0) + beta * np.log(np.maximum(n - i, 1.0))
    log_a -= np.max(log_a)        # shift for numerical stability
    a = np.exp(log_a)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def logistic_init(n: int, centre: float, scale: float) -> np.ndarray:
    """Logistic decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    # clip arguments to avoid overflow of exp
    z = scale * (i - centre)
    z = np.clip(z, -700, 700)
    a = 1.0 / (1.0 + np.exp(z))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns the best vector seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.05,
                  max_iter: int = 200000):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    focusing on the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = sub
        if idx % 2 == 0:
            i_mid = idx // 2
            if 0 <= i_mid < n:
                g[i_mid] += a[i_mid]
        g *= 2.0

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 800000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.25
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 400000,
               delta_frac: float = 0.10):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

        c = rng.uniform(0.0005, 4.0)
        candidate_pool.append(exponential_init(n, c))

    # skewed power‑law (α,β) – allow negative exponents for decreasing shapes
    for _ in range(120):
        alpha = rng.uniform(-5.0, 5.0)
        beta = rng.uniform(-5.0, 5.0)
        candidate_pool.append(skewed_power_init(n, alpha, beta))

    # logistic shapes
    for _ in range(40):
        centre = rng.integers(0, n)
        scale = rng.uniform(0.001, 0.6)
        candidate_pool.append(logistic_init(n, centre, scale))

    # optional previous solution (if any)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidate_pool.append(a0)

    # ------------------------------------------------------------------
    # Keep the best K candidates according to the exact fast_score
    # ------------------------------------------------------------------
    K = 20
    best_candidates = []      # list of tuples (score, vector)

    for idx, cand in enumerate(candidate_pool):
        sc = fast_score(cand, fft_len, n)
        if len(best_candidates) < K:
            best_candidates.append((sc, cand.copy()))
            best_candidates.sort(key=lambda x: x[0])   # keep sorted
        else:
            # worst (largest) score among stored
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, cand.copy())
                best_candidates.sort(key=lambda x: x[0])

        if (idx + 1) % 100 == 0:
            print(f"[init] evaluated {idx + 1}/{len(candidate_pool)} candidates, "
                  f"current best fast_score={best_candidates[0][0]:.8f}")

    best_a, best_score = best_candidates[0][1].copy(), best_candidates[0][0]
    print(f"[init] selected best initial fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 2) Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 2.1 Phase – L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s

    # lambda schedule – we go to extremely small smoothing values
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11,
        5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 4000

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # Additional random restarts within phase‑1 budget
    restarts = 6
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase1_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # 2.2 Phase – Adam on smoothed max (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.50 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.0005,
                                           lr=0.04,
                                           max_iter=120000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 2.3 Phase – Exp‑gradient focusing on a tiny top fraction (≈10 % budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.60 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.01)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 2.4 Phase – Exact‑max sub‑gradient descent (≈10 % budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.70 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 2.5 Phase – Simulated annealing (≈15 % budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.90 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-9,
                                                 max_moves=900000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 2.6 Phase – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
