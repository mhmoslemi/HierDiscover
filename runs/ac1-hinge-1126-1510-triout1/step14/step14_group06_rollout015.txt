"""
Improved step‑function search.

The algorithm follows three main ideas:

1. **Rich initial pool** – as before we generate many classic windows,
   random draws and a few Beta families.

2. **Low‑dimensional linear‑programming refinement** – we down‑sample the
   current best vector to a modest size (≈1500).  For the small problem we
   solve the LP described in the statement:
        maximise   Σ g_j
        subject to f * g ≤ M   (convolution with the fixed f)
                   g ≥ 0 .
   The LP is solved with SciPy’s high‑performance “highs” interior‑point
   method.  The resulting g is mixed with f (convex combination) and the
   process is repeated a few times.  Upsampling the refined small vector
   back to 5000 entries provides a new, stronger start.

3. **Full‑size polishing** – a long L‑BFGS run on the smooth log‑sum‑exp
   surrogate of the max‑convolution, followed by a very aggressive
   greedy hill‑climb that only accepts strictly improving mass‑transfer
   moves.  The hill‑climb uses a decreasing step‑size schedule so that
   fine‑grained adjustments are possible at the end of the budget.

All steps respect the 1000 s budget and print progress information.
The returned list is evaluated with the official ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog
from scipy.linalg import toeplitz
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Low‑dimensional Beta families
# ----------------------------------------------------------------------


def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Beta density on the grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_mixture_shape(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    a1 = beta_shape(n, p1, p1)
    a2 = beta_shape(n, p2, p2)
    a = w * a1 + (1.0 - w) * a2
    return a

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once, keeping the best vector.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 500000,
               start_delta_frac: float = 0.10,
               end_delta_frac: float = 0.02):
    """
    Greedy hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    # schedule for delta fraction
    while time.time() < deadline and it < max_iter:
        # linearly decay delta between start and end
        prog = it / max_iter
        delta_frac = start_delta_frac * (1 - prog) + end_delta_frac * prog
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Low‑dimensional linear‑programming refinement
# ----------------------------------------------------------------------


def lp_step_small(a: np.ndarray, M: float):
    """
    One LP step for a small problem:
        maximise Σ g_j  subject to  a * g ≤ M ,  g ≥ 0.
    Returns the normalised optimal g, or None on failure.
    """
    n = a.shape[0]
    # Build Toeplitz matrix for convolution constraints.
    # First column: a followed by (n‑1) zeros, length 2n‑1.
    col = np.concatenate([a, np.zeros(n - 1)])
    # First row: a0 then (n‑1) zeros.
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    T_dense = toeplitz(col, row)               # (2n‑1) × n
    T = csr_matrix(T_dense)                    # sparse for Linprog
    b = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)                         # maximise Σ g  ⇔ minimise −Σ g
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception as e:   # pragma: no cover
        print("[LP] linprog exception:", e)
        return None
    if not res.success:
        # print("[LP] not successful:", res.message)
        return None
    g = res.x
    sm = g.sum()
    if sm <= 0:
        return None
    return g / sm


def lp_refine_small(a: np.ndarray,
                    max_iters: int,
                    deadline: float,
                    mix_factor: float = 0.20):
    """
    Repeated LP refinement on a down‑sampled vector.
    Returns a refined vector (still sum = 1).
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute current M
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        g = lp_step_small(a, M)
        if g is None:
            break
        # convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = a / a.sum()
    return a


def downsample(a_big: np.ndarray, n_small: int) -> np.ndarray:
    """Linear interpolation from big to small size, preserving non‑negativity."""
    x_big = np.linspace(0.0, 1.0, a_big.size)
    x_small = np.linspace(0.0, 1.0, n_small)
    a_small = np.interp(x_small, x_big, a_big)
    a_small = np.maximum(a_small, 0.0)
    a_small = a_small / a_small.sum()
    return a_small


def upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from small to big size, then renormalise."""
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big = a_big / a_big.sum()
    return a_big

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – fixed at 5000 (as in the reference implementation)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            candidates.append(prev / prev.sum())

    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    for _ in range(7):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(4):
        candidates.append(kaiser_init(n, beta=rng.uniform(5.0, 12.0)))

    # ------------------------------------------------------------------
    # Low‑dimensional Beta search (single & mixture)
    # ------------------------------------------------------------------
    def beta_obj(params):
        p, q = params
        a = beta_shape(n, p, q)
        return fast_score(a, fft_len, n)

    bounds = [(0.1, 2.5), (0.1, 2.5)]
    if time.time() < deadline:
        result_beta = differential_evolution(
            beta_obj,
            bounds=bounds,
            maxiter=150,
            popsize=13,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating='deferred')
        a_beta = beta_shape(n, result_beta.x[0], result_beta.x[1])
        candidates.append(a_beta)

    def beta_mix_obj(params):
        p1, p2, w = params
        a = beta_mixture_shape(n, p1, p2, w)
        return fast_score(a, fft_len, n)

    bounds_mix = [(0.1, 2.5), (0.1, 2.5), (0.0, 1.0)]
    if time.time() < deadline:
        result_mix = differential_evolution(
            beta_mix_obj,
            bounds=bounds_mix,
            maxiter=150,
            popsize=13,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating='deferred')
        a_mix = beta_mixture_shape(n, result_mix.x[0], result_mix.x[1], result_mix.x[2])
        candidates.append(a_mix)

    # ------------------------------------------------------------------
    # Pick the best initial vector (exact fast_score)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase A – low‑dimensional LP refinement (≈20 % of budget)
    # ------------------------------------------------------------------
    phaseA_end = start_time + 0.20 * budget_s
    if time.time() < phaseA_end:
        n_small = 1500                     # a comfortable size for LP
        a_small = downsample(best_a, n_small)
        a_small = lp_refine_small(a_small,
                                  max_iters=12,
                                  deadline=phaseA_end,
                                  mix_factor=0.18)
        a_up = upsample(a_small, n)
        sc_up = fast_score(a_up, fft_len, n)
        if sc_up < best_score:
            best_score = sc_up
            best_a = a_up
            print(f"[LP‑refine] improved fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase B – intensive L‑BFGS polishing (≈40 % of budget)
    # ------------------------------------------------------------------
    phaseB_end = start_time + 0.60 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 5000

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phaseB_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # occasional random restarts during the L‑BFGS window
    restarts = 3
    while time.time() < phaseB_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phaseB_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            print(f"[L‑BFGS‑restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase C – final aggressive hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur = fast_score(best_a, fft_len, n)
        best_a, best_score = hill_climb(best_a,
                                        cur,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        start_delta_frac=0.10,
                                        end_delta_frac=0.01)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
