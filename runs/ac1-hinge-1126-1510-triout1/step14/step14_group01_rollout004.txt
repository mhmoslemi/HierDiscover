"""
Propose a low‑score step function.

The search runs a multi‑phase optimisation on a probability vector
(a_i ≥ 0, Σ a_i = 1).  Phases (within the 1000 s budget):

0.  Build a diverse pool of candidate windows (Hann, Blackman, Kaiser,
    Gaussian, Beta‑type, random simplexes) and keep the best 10.
1.  Multi‑restart L‑BFGS on a log‑softmax parametrisation with a
    decreasing log‑sum‑exp smoothing schedule.
2.  Adam optimiser that flattens the top‑k entries of the self‑convolution.
3.  Exponentiated‑gradient descent on the exact max‑convolution.
4.  Large‑scale hill‑climb (tiny mass‑transfer moves) with occasional
    jitter and a simple simulated‑annealing acceptance rule.
5.  Final projected sub‑gradient polishing on the true max entry.

All phases respect the global deadline; the best vector seen so far is
always retained.  The returned list is evaluated by ``evaluate_sequence``,
which internally rescales it, so the optimisation is performed on the
probability simplex (scale‑invariant objective).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast, uses FFT)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum≈1).
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Initialisation families (deterministic & random)
# ----------------------------------------------------------------------


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on the discrete grid (mid‑point rule)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS driver (continuation over λ schedule)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS for each λ in lam_schedule, starting each stage from prev."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

# ----------------------------------------------------------------------


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.02):
    """Adam optimiser flattening the top‑k convolution entries."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient w.r.t. a of masked sum
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # store best
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def eg_optimize(a_start: np.ndarray,
                deadline: float,
                n: int,
                fft_len: int,
                rng: np.random.Generator,
                init_lr: float = 1.0,
                top_start: float = 0.12,
                top_end: float = 0.02):
    """Exponentiated‑gradient (mirror descent) on the exact max entry."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    start_time = time.time()
    while time.time() < deadline:
        t += 1
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # linearly decay the top‑k fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        top_frac = max(top_end, top_start * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        idxs = np.argpartition(conv, -k)[-k:]
        mask = np.zeros_like(conv)
        mask[idxs] = 1.0

        # gradient of masked sum
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        # EG step
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with occasional jitter (simulated annealing flavour)
# ----------------------------------------------------------------------


def hill_climb_sa(a_start: np.ndarray,
                  cur_score: float,
                  fft_len: int,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  delta_frac: float = 0.02):
    """
    Hill‑climb that accepts improving moves and, with a slowly
    decreasing temperature, occasionally accepts worsening moves.
    """
    best = a_start.copy()
    best_score = cur_score
    curr = a_start.copy()
    curr_score = cur_score

    # temperature schedule
    T0 = best_score * 0.02
    start_time = time.time()
    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T0 * max(0.0, 1.0 - elapsed / total)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # sized delta depends on source mass
        delta = curr[i] * rng.random() * delta_frac
        if delta <= 0:
            continue

        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        dE = sc - curr_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            # accept (or keep old)
            curr = cand
            curr_score = sc
            if sc < best_score:
                best_score = sc
                best = cand.copy()

        # every few thousand steps add a small jitter to the whole vector
        if rng.integers(0, 4000) == 0:
            jitter = rng.normal(scale=1e-5, size=n)
            cand2 = np.maximum(curr + jitter, 0.0)
            s = cand2.sum()
            if s > 0:
                cand2 /= s
            else:
                cand2 = np.full_like(cand2, 1.0 / n)
            sc2 = fast_score(cand2, fft_len, n)
            if sc2 < best_score:
                best_score = sc2
                best = cand2.copy()
            # accept jitter if it improves or with tiny probability
            if sc2 < curr_score or rng.random() < np.exp(-(sc2 - curr_score) / max(T, 1e-12)):
                curr = cand2
                curr_score = sc2

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = 0.5 / np.sqrt(t)
        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Pool handling utilities
# ----------------------------------------------------------------------


def insert_into_pool(pool, vec, score, max_size=10):
    """Maintain a sorted pool (ascending score) of at most max_size entries."""
    if len(pool) < max_size:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    else:
        if score < pool[-1][0]:
            pool[-1] = (score, vec.copy())
            pool.sort(key=lambda x: x[0])
    return pool

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (power‑of‑two for fast FFTs)
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial diverse pool of candidates
    # ------------------------------------------------------------------
    pool = []

    # If a previous solution exists, upsample it
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            pool = insert_into_pool(pool, a0, fast_score(a0, fft_len, n))

    # Deterministic families
    deterministic_fns = [
        triangular_init,
        hann_init,
        blackman_init,
        lambda N: kaiser_init(N, 8.0),
        lambda N: kaiser_init(N, 12.0),
        lambda N: gaussian_init(N, sigma_factor=0.22),
        lambda N: gaussian_init(N, sigma_factor=0.28),
    ]

    for fn in deterministic_fns:
        cand = fn(n)
        pool = insert_into_pool(pool, cand, fast_score(cand, fft_len, n))

    # Symmetric Beta sweep (fine)
    for alpha in np.linspace(0.5, 12.0, 12):
        cand = beta_init(n, alpha, alpha)
        pool = insert_into_pool(pool, cand, fast_score(cand, fft_len, n))

    # Additional random draws
    for _ in range(20):
        cand = random_initial(n, rng)
        pool = insert_into_pool(pool, cand, fast_score(cand, fft_len, n))

    best_score, best_a = pool[0][0], pool[0][1].copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Multi‑restart L‑BFGS (≈15 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # create a random convex combination of the current best and a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3   # up‑weight randomness
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            pool = insert_into_pool(pool, best_a, best_score)
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈30 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.15,
            end_top=0.02
        )
        pool = insert_into_pool(pool, best_a, best_score)
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = min(deadline, time.time() + 0.25 * budget_s)
        print("[EG] start")
        best_a, best_score = eg_optimize(
            best_a,
            eg_deadline,
            n,
            fft_len,
            rng,
            init_lr=1.0,
            top_start=0.12,
            top_end=0.02
        )
        pool = insert_into_pool(pool, best_a, best_score)
        print(f"[EG] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Hill‑climb with simulated‑annealing flavour (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = min(deadline, time.time() + 0.20 * budget_s)
        print("[Hill‑climb‑SA] start")
        best_a, best_score = hill_climb_sa(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            delta_frac=0.02
        )
        pool = insert_into_pool(pool, best_a, best_score)
        print(f"[Hill‑climb‑SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final sub‑gradient polishing (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_deadline = min(deadline, time.time() + 0.10 * budget_s)
        print("[Fine‑tune] start")
        best_a, best_score = fine_tune_subgradient(
            best_a,
            fine_deadline,
            n,
            fft_len
        )
        print(f"[Fine‑tune] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the best candidate (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
