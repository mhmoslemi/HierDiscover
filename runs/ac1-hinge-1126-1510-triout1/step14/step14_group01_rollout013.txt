def adam_exact_optimize(a_start, deadline, n, fft_len, rng, t0=0.5):
    y = np.log(a_start + 1e-12)
    m1, m2 = np.zeros_like(y), np.zeros_like(y)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    while time.time() < deadline:
        t += 1
        a = softmax(y)
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_idx = int(np.argmax(conv))
        # compute gradient wrt a (exact)
        rev_idx = max_idx
        # compute a_rev = reverse a for index mapping
        # we want for each i: a[max_idx - i] if within range else 0
        # Equivalent to reversed a shifted?
        grad_a = np.zeros_like(a)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad_a[i] = 4.0 * n * a[j]  # factor includes 4n
        # chain to logits
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)
        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * grad_y
        m2 = beta2 * m2 + (1 - beta2) * (grad_y**2)
        m1_hat = m1 / (1 - beta1**t)
        m2_hat = m2 / (1 - beta2**t)
        lr = 0.1 / np.sqrt(t)   # maybe smaller stable
        y -= lr * m1_hat / (np.sqrt(m2_hat) + eps)
        # projection via softmax ensures positivity and sum=1
        # keep best
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

The search consists of several stages, each respecting the global 1000 s
budget:

0️⃣  Build a diverse pool of initial candidates (classical windows,
    Beta‑families, random draws, and an up‑sample of the previous best).

1️⃣  Smoothed L‑BFGS on a log‑sum‑exp approximation of the max‑convolution.
    The smoothing parameter λ is driven down to 1e‑12, with several restarts.

2️⃣  Adam optimisation that flattens the *k* largest convolution entries
    (top‑k mask).  The schedule for *k* fades from 12 % to 2 % of the shifts.

3️⃣  Adam on the *exact* max‑convolution (no smoothing).  The gradient of the
    max entry is computed analytically (g_i = 4 n · a_{m−i}); the update is
    performed in the soft‑max (logits) space and projected onto the simplex
    via the soft‑max.

4️⃣  Greedy hill‑climb with guided mass‑transfer moves: mass is moved from
    the index with the largest gradient to the one with the smallest,
    only accepting strictly improving steps.

5️⃣  Final fine‑tuning by projected sub‑gradient descent on the exact
    max‑convolution.

The best vector seen at any stage is retained; the loop terminates early
if the score drops below the target 1.503.
"""

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective used for bookkeeping (no sum scaling – we keep sum=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum≈1). Returns 2·n·max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def full_convolution(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full self‑convolution of a (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (various flavours)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous result.
    Returns best a and its fast_score found before deadline.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            # progress printed by caller
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.02):
    """
    Adam optimiser that uses a top‑k mask (instead of the single maximal entry).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        # mask of largest entries
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked max (linear combination of entries)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score


def adam_exact_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       init_lr: float = 0.2):
    """
    Adam optimisation using the *exact* max‑convolution gradient.
    Gradient w.r.t. a_i is 4·n·a_{m−i} where m = argmax conv.
    Updates are performed in logits (softmax) space.
    """
    y = np.log(a_start + 1e-12)
    m1 = np.zeros_like(y)
    m2 = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and max index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        m_idx = int(np.argmax(conv))

        # exact gradient of the max entry
        idx = np.arange(n)
        j = m_idx - idx
        mask = (j >= 0) & (j < n)
        grad_a = np.zeros_like(a)
        # factor 4·n comes from derivative of 2·n·max(conv) (since sum=1)
        grad_a[mask] = 4.0 * n * a[j[mask]]

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * grad_y
        m2 = beta2 * m2 + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m1_hat = m1 / (1.0 - beta1_pow)
        m2_hat = m2 / (1.0 - beta2_pow)

        lr = init_lr / np.sqrt(t)
        y -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # optional small jitter to escape plateaus
        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        # keep best
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """
    Guided hill‑climb: move mass from the index with the largest exact gradient
    to the index with the smallest, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute exact gradient once per few iterations for cheapness
        if it % 1000 == 0:
            conv = full_convolution(best, fft_len, n)
            m_idx = int(np.argmax(conv))
            idx = np.arange(n)
            j = m_idx - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(best)
            grad[mask] = best[j[mask]]
        # pick donor and receiver based on gradient
        i = int(np.argmax(grad))   # largest gradient → donor
        j = int(np.argmin(grad))   # smallest gradient → receiver
        if i == j:
            # fallback to random pair
            i, j = rng.integers(0, n, size=2)
        if best[i] <= 0:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
            # refresh gradient after improvement
            conv = full_convolution(best, fft_len, n)
            m_idx = int(np.argmax(conv))
            idx = np.arange(n)
            jdx = m_idx - idx
            mask = (jdx >= 0) & (jdx < n)
            grad = np.zeros_like(best)
            grad[mask] = best[jdx[mask]]
        it += 1
    return best, best_score


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """
    Projected sub‑gradient descent on the exact max‑convolution
    (simple decreasing step size).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a, fft_len, n)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = 0.5 / np.sqrt(t)
        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4096 offers a good trade‑off between precision and speed
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # up‑sample previous best if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.22))
    candidates.append(gaussian_init(n, sigma_factor=0.28))
    for beta in (6.0, 9.0, 12.0):
        candidates.append(kaiser_init(n, beta))

    # symmetric Beta sweep (more fine‑grained than before)
    for alpha in np.linspace(0.7, 5.0, 11):
        candidates.append(beta_init(n, alpha, alpha))

    # asymmetric Beta sweep (adds extra flexibility)
    for a_par in (0.8, 2.0, 4.0):
        for b_par in (0.8, 2.0, 4.0):
            if a_par == b_par:
                continue
            candidates.append(beta_init(n, a_par, b_par))

    # random draws
    for _ in range(15):
        candidates.append(random_initial(n, rng))

    # evaluate everything with the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    TARGET = 1.503  # early‑stop threshold

    # ------------------------------------------------------------------
    # Phase 1 – Smoothed L‑BFGS (≈25 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    # extended smoothing schedule (down to 1e‑12)
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7, 5e-8, 1e-8, 5e-9, 1e-9,
        5e-10, 1e-10, 5e-11, 1e-11, 5e-12, 1e-12
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")
            if best_score < TARGET:
                break

    if best_score < TARGET:
        print("[early] reached target after L‑BFGS")
        final_seq = best_a.tolist()
        print(f"[final] evaluated score = {evaluate_sequence(final_seq):.8f}")
        return final_seq

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.10,
            end_top=0.02
        )
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")
        if best_score < TARGET:
            print("[early] reached target after Adam‑topk")
            final_seq = best_a.tolist()
            print(f"[final] evaluated score = {evaluate_sequence(final_seq):.8f}")
            return final_seq

    # ------------------------------------------------------------------
    # Phase 3 – Adam on exact max (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exact_deadline = time.time() + 0.20 * budget_s
        print("[Adam‑exact] start")
        best_a, best_score = adam_exact_optimize(
            best_a,
            exact_deadline,
            n,
            fft_len,
            rng,
            init_lr=0.2
        )
        print(f"[Adam‑exact] best fast_score={best_score:.8f}")
        if best_score < TARGET:
            print("[early] reached target after Adam‑exact")
            final_seq = best_a.tolist()
            print(f"[final] evaluated score = {evaluate_sequence(final_seq):.8f}")
            return final_seq

    # ------------------------------------------------------------------
    # Phase 4 – Guided hill‑climb (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.20 * budget_s)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=200000,
            delta_frac=0.05
        )
        print(f"[hill‑climb] best fast_score={best_score:.8f}")
        if best_score < TARGET:
            print("[early] reached target after hill‑climb")
            final_seq = best_a.tolist()
            print(f"[final] evaluated score = {evaluate_sequence(final_seq):.8f}")
            return final_seq

    # ------------------------------------------------------------------
    # Phase 5 – Final fine‑tuning (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        fine_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_score = fine_tune_subgradient(
            best_a,
            fine_deadline,
            n,
            fft_len
        )
        print(f"[fine‑tune] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
