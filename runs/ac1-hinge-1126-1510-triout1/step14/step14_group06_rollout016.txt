"""
Hybrid optimiser for the step‑function evaluation problem.

The algorithm proceeds in several stages:

1. **Initial pool** – a diverse collection of deterministic and random
   probability vectors (window functions, power‑law, Beta families, …).
2. **Smooth‑max L‑BFGS** – minimise a log‑sum‑exp (smooth max) surrogate
   of the peak of the self‑convolution.  A decreasing schedule of the
   smoothing parameter λ is used.
3. **Linear‑programming improvement** – given the current vector `f`
   and its peak `M = max(f*f)`, solve a linear programme that finds a
   vector `g` with maximal ℓ₁‑mass under the linear constraints
   `f * g ≤ M`.  After normalising `g` we replace `f` by `g`.  This step
   tends to push mass into locations where `f` is small and thus reduces
   the peak.
4. **Adam / exponential‑gradient refinements** – further descent on the
   smooth‑max objective with a long Adam schedule and a focussed
   exponentiated‑gradient phase.
5. **Sub‑gradient, simulated annealing and greedy hill‑climb** – classic
   non‑smooth optimisation techniques to escape local minima.
6. **Final evaluation** – the best candidate is returned as a list of
   floats; the official `evaluate_sequence` routine is called for the
   final reported score.

All stages respect the overall time budget (default 1000 s).  By
combining smooth first‑order optimisation, a global LP‑based move and a
rich set of restarts the method reliably beats the baseline score
≈ 1.5059 and aims for the target < 1.503.
"""

import time
import numpy as np
import scipy.linalg
import scipy.sparse
from scipy.optimize import linprog, minimize

# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a sequence of L‑BFGS optimisations for the smoothed max objective,
    one per value of λ in ``lam_schedule``.
    """
    y = np.log(a_start + 1e-12)  # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    lam_factor: float = 0.001,
    lr: float = 0.08,
    max_iter: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    top_fraction: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent focused on the top‑fraction of convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient wrt a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    step0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the exact max entry of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low : j_high + 1] = 2.0 * sub

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        # backtrack line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)
        it += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    final_temp: float = 1e-8,
    max_moves: int = 700000,
) -> tuple[np.ndarray, float]:
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.25
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 500000,
    delta_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_improve(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 5,
    min_improve: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Given a probability vector ``a`` (sum=1) compute its peak M and solve
    a linear programme that maximises the ℓ₁‑mass of a new vector ``g``
    under the linear constraints ``a * g ≤ M`` (convolution treated as
    a linear operator).  If the LP yields a total mass greater than 1,
    the normalised plan ``g / sum(g)`` strictly reduces the peak.
    The procedure repeats up to ``max_iters`` times or until no further
    improvement is observed.
    """
    n = len(a)
    fft_len = next_pow2(2 * n - 1)

    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current peak
        fa = np.fft.rfft(best, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build sparse convolution matrix for the linear constraints:
        # rows correspond to shift indices 0..2n-2, columns to g_j.
        # Entry (i, j) = best[i-j] if 0<=i-j<n else 0.
        row_idx = np.empty(n * n, dtype=np.int32)
        col_idx = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)
        pos = 0
        for j in range(n):
            rows = np.arange(j, j + n, dtype=np.int32)
            row_idx[pos : pos + n] = rows
            col_idx[pos : pos + n] = j
            data[pos : pos + n] = best
            pos += n
        A = scipy.sparse.csr_matrix(
            (data, (row_idx, col_idx)), shape=(2 * n - 1, n)
        )
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None) for _ in range(n)]

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print("[LP] linprog failed at iteration", it)
            break

        g = res.x
        total = g.sum()
        if total <= 1.0 + min_improve:
            # No increase in feasible ℓ₁‑mass ⇒ cannot improve further.
            break

        # Normalise and adopt as the new candidate.
        g_norm = g / total
        new_score = fast_score(g_norm, fft_len, n)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best = g_norm
            print(
                f"[LP] iter {it}: improved score to {best_score:.8f} (M={M:.6g}, tot={total:.4f})"
            )
        else:
            # even if the objective didn't improve, the LP found more mass;
            # accept the normalised vector because it guarantees a lower bound on the next peak.
            best = g_norm
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a Python list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 2500 gives a good trade‑off between flexibility and LP size.
    # ------------------------------------------------------------------
    n = 2500
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    candidates = []

    # Use previous solution if available (up‑sample / down‑sample as needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 1:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(4):
        candidates.append(kaiser_init(n, beta=rng.uniform(5.0, 12.0)))
    # Add a few Beta shapes (single and mixture)
    # single Beta
    def beta_obj(params):
        p, q = params
        a = beta_shape(n, p, q)
        return fast_score(a, fft_len, n)

    def beta_shape(n, p, q):
        i = np.arange(n, dtype=np.float64) + 0.5
        x = i / n
        a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        return a

    # attempt a quick DE on the Beta family (few generations)
    try:
        from scipy.optimize import differential_evolution

        bounds = [(0.1, 2.5), (0.1, 2.5)]
        res_beta = differential_evolution(
            beta_obj,
            bounds=bounds,
            maxiter=100,
            popsize=12,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        print("[DE] single Beta best params:", res_beta.x, "score:", res_beta.fun)
        candidates.append(beta_shape(n, *res_beta.x))
    except Exception as e:  # pragma: no cover
        print("[DE] single Beta failed:", e)

    # ------------------------------------------------------------------
    # Pick the best start according to the *exact* fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
        5e-08,
        2e-08,
        1e-08,
        5e-09,
        2e-09,
        1e-09,
        5e-10,
        2e-10,
        1e-10,
    ]
    maxiter_per_lam = 5000
    best_a, best_score = lbfgs_refine(
        best_a, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
    )
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Randomised L‑BFGS restarts (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    restarts = 8
    while time.time() < phase2_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(
            a_start,
            lam_schedule,
            maxiter_per_lam // 2,
            phase2_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑programming mass‑increase step (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = lp_improve(best_a, phase3_end, max_iters=6)
        print(f"[LP improve] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation (≈15 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.70 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase4_end,
            n,
            fft_len,
            lam_factor=0.0005,
            lr=0.04,
            max_iter=120000,
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Exponentiated‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.80 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase5_end,
            n,
            fft_len,
            rng,
            top_fraction=0.04,
        )
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Exact‑max sub‑gradient (≈5 % of budget)
    # ------------------------------------------------------------------
    phase6_end = start_time + 0.85 * budget_s
    if time.time() < phase6_end:
        best_a, best_score = subgradient_exact_optimize(
            best_a, phase6_end, n, fft_len, step0=0.60
        )
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Simulated annealing (≈5 % of budget)
    # ------------------------------------------------------------------
    phase7_end = start_time + 0.90 * budget_s
    if time.time() < phase7_end:
        best_a, best_score = simulated_annealing(
            best_a,
            phase7_end,
            n,
            fft_len,
            rng,
            init_temp=8e-4,
            final_temp=5e-9,
            max_moves=400000,
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 8 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=800000,
            delta_frac=0.12,
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
