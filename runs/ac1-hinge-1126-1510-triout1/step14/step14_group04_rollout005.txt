def propose_candidate(seed=..., budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 4096  # maybe bigger if time.
    fft_len = next_pow2(2*n - 1)

    best_seq = None
    best_score = np.inf

    for density in [0.3, 0.5, 0.6]:
        m = int(round(n * density))
        # initial set
        idx = rng.choice(n, size=m, replace=False)
        b = np.zeros(n, dtype=np.float64)
        b[idx] = 1.0
        # compute conv
        conv = np.fft.irfft(np.fft.rfft(b, fft_len) * np.fft.rfft(b, fft_len), fft_len)[:2*n-1]
        max_count = conv.max()
        score = 2.0 * n * max_count / (m**2)
        # local search
        # Simulated annealing loop
        # We'll maintain best for this density.
        T0 = 0.1 * max_count  # some initial temperature
        T_final = 1e-3
        max_moves = int(1e5)  # adjust based on time
        for move in range(max_moves):
            if time.time() >= deadline:
                break
            # temperature schedule
            t = move / max_moves
            T = T0 * (T_final / T0) ** t
            # propose swap
            i = rng.integers(0, n)
            if b[i] == 0:
                continue
            j = rng.integers(0, n)
            if b[j] == 1:
                continue
            # perform swap and compute new max
            b[i] = 0.0
            b[j] = 1.0
            conv_new = np.fft.irfft(np.fft.rfft(b, fft_len) * np.fft.rfft(b, fft_len), fft_len)[:2*n-1]
            max_new = conv_new.max()
            delta = max_new - max_count
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                # accept
                max_count = max_new
                # maybe update best
                if max_new < max_count: ... (should reflect)
                # keep b as new
                if max_new < (best_score * (m**2) / (2*n)): # compute better?
                    # recompute sequence weights
                    a = b / m
                    sc = 2*n*max_new/(m**2)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        print(f"[dens {density}] new best score {best_score:.8f}")
                # else keep b
            else:
                # revert
                b[i] = 1.0
                b[j] = 0.0

        # after loop maybe compute final best for this density.

    # After search, if no best_seq found, fallback to previous (like uniform)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Final check evaluate_sequence
    final_seq = best_seq.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score", final_eval)
    return final_seq

"""
Propose a low‑score step function.

Algorithm
---------
1. Fix a problem size n (4096).  The score is invariant under scaling,
   so we keep the total sum equal to 1.
2. Choose a density d (≈ 0.5) and create a binary indicator b of length n
   with exactly m = ⌊d·n⌋ positions set to 1.  The actual step‑function is
   a = b / m  (all selected positions have equal height, the rest 0).
3. The objective for a binary vector is
        score = 2·n·max(b*b) / m²                (max of the integer convolution)
   where “*” denotes linear convolution.
   For a given m this is equivalent to minimising the maximum entry of
   the integer convolution of b with itself.
4. Starting from a random b we run a short simulated‑annealing / hill‑climb
   that repeatedly swaps a 1‑position i with a 0‑position j.
   The full convolution of the updated binary vector is recomputed by FFT
   (O(n log n)) and the move is accepted if it lowers the peak or with a
   probability exp(‑Δ/T) where the temperature T decays geometrically.
5. Several densities are tried (0.48, 0.5, 0.55) and the best sequence seen
   during the whole run is kept.
6. The final sequence (a = b / m) is returned; a single call to the official
   ``evaluate_sequence`` is performed for the final sanity check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two FFT length works best.
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Densities to try (fraction of positions that are non‑zero)
    # --------------------------------------------------------------
    densities = [0.48, 0.50, 0.55]

    # --------------------------------------------------------------
    # Main optimisation loop – repeated random restarts with SA.
    # --------------------------------------------------------------
    while time.time() < deadline:
        for density in densities:
            if time.time() >= deadline:
                break

            m = max(2, int(round(n * density)))    # number of 1‑entries
            # ----- initialise a random binary vector -----------------
            b = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=m, replace=False)
            b[idx] = 1.0

            # pre‑compute current convolution (integer counts)
            fb = np.fft.rfft(b, fft_len)
            conv = np.fft.irfft(fb * fb, fft_len)[: 2 * n - 1]
            conv_int = np.rint(conv).astype(np.int64)
            max_c = int(conv_int.max())
            cur_score = 2.0 * n * max_c / (m * m)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = (b / m).copy()
                print(f"[init] density={density:.2f}, score={best_score:.8f}")

            # ----- list of positions that are 1 (for O(1) sampling) --
            ones = np.array(idx, dtype=np.int64)

            # ----- annealing schedule --------------------------------
            total_moves = 25000                     # per restart
            T_start = max_c * 0.02 + 1.0
            T_final = 0.001

            for move in range(total_moves):
                if time.time() >= deadline:
                    break

                # geometric temperature decay
                t_frac = move / total_moves
                T = T_start * (T_final / T_start) ** t_frac

                # propose a swap: i ∈ ones, j ∉ S
                pos_i = rng.integers(0, len(ones))
                i = ones[pos_i]

                j = rng.integers(0, n)
                while b[j] == 1.0:
                    j = rng.integers(0, n)

                # perform the swap
                b[i] = 0.0
                b[j] = 1.0

                # recompute convolution of the modified binary vector
                fb = np.fft.rfft(b, fft_len)
                conv = np.fft.irfft(fb * fb, fft_len)[: 2 * n - 1]
                conv_int = np.rint(conv).astype(np.int64)
                max_new = int(conv_int.max())
                delta = max_new - max_c

                # acceptance test
                accept = False
                if delta <= 0:
                    accept = True
                elif T > 0.0:
                    if rng.random() < np.exp(-delta / T):
                        accept = True

                if accept:
                    max_c = max_new
                    ones[pos_i] = j                     # update the set of 1‑indices
                    cur_score = 2.0 * n * max_c / (m * m)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = (b / m).copy()
                        print(f"[improve] density={density:.2f}, score={best_score:.8f}")
                else:
                    # revert the move
                    b[i] = 1.0
                    b[j] = 0.0

        # If we exhausted the densities in the current pass we simply loop
        # and start a new random restart until the time budget ends.

    # --------------------------------------------------------------
    # Fallback (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Final sanity‑check and official evaluation
    # --------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
