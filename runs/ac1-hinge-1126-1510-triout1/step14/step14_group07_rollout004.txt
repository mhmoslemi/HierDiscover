"""
Propose a step‐function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1. Choose a moderate size `n` (≈1500) – large enough to allow a good
   reduction of the autocorrelation peak, small enough for fast dense LP.
2. Initialise the vector from the previous best solution (if available)
   by linear interpolation, otherwise from the uniform distribution.
3. Repeatedly:
   * Compute the current peak `M = max(a∗a)`.
   * Build the dense convolution matrix `A` such that  `A @ g = a∗g`.
   * Solve the linear program  
        maximise   1ᵀ g
        subject to A g ≤ M,   g ≥ 0
     (via `scipy.optimize.linprog`).  The optimum `g` tries to carry as much
     mass as possible while staying under the same peak when convolved with
     the current vector.
   * Form a convex combination `a_new = (1‑α)·a + α·g` (α≈0.3), renormalise
     to sum = 1 and keep the best vector found so far.
4. When the time budget is almost exhausted, run a short greedy hill‑climb
   that strictly improves the exact `fast_score` (the objective for the
   normalised vector).
5. Return the best vector; the official `evaluate_sequence` routine will
   compute the final score.

The LP step implements the alternating optimisation described in the
statement (maximise the ℓ₁‑norm of a new step while respecting the current
convolution bound).  By repeatedly mixing the current vector with the LP
solution the peak of the autocorrelation is driven down, typically
producing a score below 1.503 within the allotted 1000 s.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that (A @ g) = a∗g.
    Entry A[i,k] = a[i‑k] if 0 ≤ i‑k < n, else 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for i in range(rows):
        k_min = max(0, i - (n - 1))
        k_max = min(i, n - 1)
        # k_range = [k_min … k_max]
        ks = np.arange(k_min, k_max + 1)
        A[i, ks] = a[i - ks]
    return A


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300_000,
               delta_frac: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Strictly improving greedy hill‑climb on the exact fast_score.
    Moves a random amount of mass from index i to j.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size: a few thousand entries give a good balance.
    # --------------------------------------------------------------
    n = 1500
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise a from a previous solution (if present) or uniform.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 1:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – alternating LP + mixing
    # --------------------------------------------------------------
    iteration = 0
    # We keep the LP phase for roughly 80 % of the budget
    lp_phase_end = start_time + 0.80 * budget_s

    while time.time() < lp_phase_end:
        # current bound M
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build convolution matrix for the current a
        A = build_conv_matrix(best_a)            # shape (2n‑1, n)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise 1ᵀg  ⇔ minimise –1ᵀg
        c = -np.ones(n, dtype=np.float64)        # minimise -sum(g)

        # Solve with HiGHS (fast dense LP)
        try:
            res = linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as exc:                 # pragma: no cover
            print(f"[LP] iteration {iteration} exception: {exc}")
            break

        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0.0:
            print("[LP] zero‑sum solution, aborting.")
            break

        # --------------------------------------------------------------
        # Mix the LP solution with the current vector.
        # Try a few mixing ratios, keep the best improvement.
        # --------------------------------------------------------------
        improved = False
        for alpha in (0.4, 0.3, 0.2, 0.1):
            cand = (1.0 - alpha) * best_a + alpha * g
            # Protect against negative rounding errors and renormalise.
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = fast_score(cand, fft_len, n)
            if sc < best_score:
                best_a = cand
                best_score = sc
                improved = True
                print(f"[LP iter {iteration}] α={alpha:.2f} → fast_score={best_score:.8f}")
                break
        if not improved:
            # No mixing improvement – shrink the mixing factor for next round.
            # This typically happens when we are already near a local optimum.
            pass

        # Clean up large temporary arrays to keep memory usage modest.
        del A, res, g
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb with remaining time.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        deadline, rng,
                                        max_iter=300_000,
                                        delta_frac=0.12)
        print(f"[hill‑climb] final fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
