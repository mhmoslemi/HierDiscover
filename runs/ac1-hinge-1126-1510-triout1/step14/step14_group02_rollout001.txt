"""
Propose a low‑score step‑function.

The algorithm works in several stages:

1. **Problem size** – use a larger power‑of‑two (n=16384) to give the
   optimizer more degrees of freedom.  The scoring function is scale‑invariant,
   so we keep the vector on the simplex (sum = 1).

2. **Initial pool** – a collection of deterministic (spectral, Hann,
   Gaussian, …) and diversified random vectors.  If a previous solution is
   available we up‑sample it to the new size.

3. **L‑BFGS continuation** – minimise a smoothed version of the max‑autocorrelation
   using a schedule of smoothing parameters λ.  Several random restarts are
   performed while time permits.

4. **Adam with top‑k mask** – a fast stochastic optimiser that only focuses on
   the currently largest convolution entries.

5. **Adam on a log‑sum‑exp smooth max** – a second Adam phase with a slowly
   decreasing smoothing λ.

6. **Simulated annealing** – cheap stochastic moves on the simplex.

7. **Sub‑gradient descent** – a deterministic descent on the true max‑convolution
   (sub‑gradient of the max) with projection onto the simplex.

8. **Final greedy hill‑climb** – tiny mass‑transfer moves to finish the search.

All phases keep the best vector found (according to the cheap “fast_score”
which equals the evaluation score for sum = 1).  At the end we call the
official `evaluate_sequence` once and return the resulting list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast score (exact max convolution for a probability vector)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a probability vector a (sum = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of a⋆a and its gradient w.r.t. logits y (a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # self‑convolution (full length)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Construct a spectral guess via sqrt of uniform spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'double_edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # double_edge – heavy mass on both extreme ends with a gap
        gap_frac = rng.uniform(0.02, 0.1)
        gap = int(gap_frac * n)
        edge_len = max(1, (n - gap) // 2)
        a = np.full(n, 1e-12)   # tiny background to avoid zeros in log
        heavy = (0.5) / edge_len
        a[:edge_len] = heavy
        a[-edge_len:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.30) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits (stable start)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.001):
    """
    Adam optimiser that focuses on a shrinking top‑k mask of the convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (full length)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # gradient w.r.t a
        grad_y = a * (g_a - np.dot(a, g_a))

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # decreasing step size
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc

def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 600000):
    """
    Simple SA on the simplex with small mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        step0: float = 0.4,
                        min_step: float = 1e-5):
    """
    Deterministic sub‑gradient descent on the true max‑convolution.
    """
    a = a_start.copy()
    best = a.copy()
    best_sc = fast_score(a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        # compute convolution and mask of maximal entries
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)

        # sub‑gradient of max (2·a_rev⋆mask, central slice)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fmask = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fmask, fft_len)
        grad = 2.0 * grad_rev[n - 1:n - 1 + n]

        # step size schedule
        step = max(step0 / np.sqrt(t), min_step)

        # gradient descent step with projection onto simplex
        a = simplex_projection(a - step * grad)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    return best, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.03):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an old vector to a larger dimension."""
    x_old = np.linspace(0, 1, seq.size)
    x_new = np.linspace(0, 1, new_n)
    a = np.interp(x_new, x_old, seq)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            a0 = upsample_sequence(prev, n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.30))
    candidates.append(kaiser_init(n, beta=8.0))

    # randomised families
    for _ in range(14):
        candidates.append(random_initial(n, rng))

    # pick the best based on the cheap fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (core optimisation)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # 45 % of budget
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8
    ]
    maxiter_per_lam = 1800   # a bit tighter to keep within time

    while time.time() < phase1_end:
        # Mix current best with a fresh random vector to escape shallow minima
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = time.time() + 0.20 * budget_s
        print("[Adam‑topk] start")
        best_a, best_sc = adam_topk_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            start_top=0.15,
            end_top=0.001
        )
        print(f"[Adam‑topk] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on log‑sum‑exp smooth max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_end = time.time() + 0.15 * budget_s
        print("[Adam‑smooth] start")
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            lam_start=5e-2,
            lam_end=1e-9
        )
        print(f"[Adam‑smooth] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.10 * budget_s)
        print("[SA] start")
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=500000
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – Sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + min(remaining, 0.08 * budget_s)
        print("[Sub‑gradient] start")
        best_a, best_sc = subgradient_descent(
            best_a,
            subgrad_deadline,
            n,
            fft_len,
            rng,
            step0=0.35,
            min_step=5e-6
        )
        print(f"[Sub‑gradient] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.12 * budget_s)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=400000,
            delta_frac=0.025
        )
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official scoring function
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
