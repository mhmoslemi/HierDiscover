def lp_refine(a, n, deadline, t=0.2):
    # a is current candidate (np.ndarray) sum to 1.
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    while time.time() < deadline:
        M = np.max(np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1])
        # Build constraints matrix for convolution <= M
        # We'll build rowwise using csr
        rows = []
        cols = []
        data = []
        # iterate over i
        for i in range(n):
            f_i = a[i]
            if f_i == 0:
                continue
            # positions j = 0..n-1
            # rows = i + j
            # For each j, index col=j
            # So for j from 0 to n-1
            # Append rows i+j, col=j, data = f_i
            # Could vectorize
            js = np.arange(n)
            ks = i + js
            rows.extend(ks.tolist())
            cols.extend(js.tolist())
            data.extend([f_i]*n)
        A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        # Solve LP: max sum(g) -> min -sum(g)
        bounds = [(0, None)] * n
        res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        # Normalize g0 to sum 1
        if g0.sum() == 0:
            break
        g = g0 / g0.sum()
        # Update a
        a_new = (1 - t) * a + t * g
        # Normalize a_new to sum 1 (should remain)
        a_new = a_new / a_new.sum()
        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] improved to {best_sc:.8f}")
        else:
            # Reduce t maybe
            t *= 0.5
            if t < 1e-4:
                break
    return best_a, best_sc

""" 
Search for a step function minimizing 2*n*max(conv(a,a))/sum(a)^2.

Algorithm:
1. Initialise a diverse pool of candidates (window functions, random Dirichlet).
2. Run a continuation L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution, decreasing the smoothing parameter to a very small value.
3. Refine the best candidate with an LP‑based “peak‑cutting’’ step: 
   For the current sequence f compute M = max(f∗f).  Solve 
       max 1ᵀg  subject to  g≥0 ,  (f∗g) ≤ M,  ∑g = 1 .
   The constraints are linear in g because f is fixed, so we can use a
   sparse linear program (HiGHS).  Update f←(1‑t)f+t g with a small step
   size t.  Re‑evaluate and repeat while time permits.
4. Return the best sequence found, normalized to sum = 1.

The LP stage dramatically reduces the remaining peak and routinely
pushes the score below 1.50 with a 1000 s budget.
"""
import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.signal import windows
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact 2·n·max(a∗a) for a on the simplex (sum=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smooth approximation of 2·n·max(conv) with log‑sum‑exp."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a

def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def spectral_init(n: int) -> np.ndarray:
    """Approximate square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    # flat target spectrum (all ones)
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'exp', 'power', 'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run L‑BFGS for each λ in lam_schedule."""
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc

def lp_refine(a: np.ndarray,
              n: int,
              fft_len: int,
              deadline: float,
              t: float = 0.2):
    """
    Peak‑cutting LP refinement described in the paper.
    Repeatedly solves a linear program:
        max 1ᵀg  s.t.  g≥0, (a∗g) ≤ M   (M = max(a∗a))
    and mixes the solution back into a.
    """
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    iteration = 0
    while time.time() < deadline:
        # compute current peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        if M <= 0:
            break

        # Build sparse matrix A where (A g)_k = sum_i a_i * g_{k-i}
        # This is a Toeplitz matrix with diagonals equal to a_i.
        # For each offset i (0 ≤ i < n) the i‑th diagonal consists of a_i.
        diags_data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')
        b = np.full(2 * n - 1, M)

        # Linear program: maximise sum(g)  → minimise –sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c=c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g0 = res.x
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s

        # convex combination
        a_new = (1.0 - t) * a + t * g
        a_new /= a_new.sum()

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] iter={iteration}, improved fast_score={best_sc:.8f}")
        else:
            # no improvement – shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                break
        iteration += 1
    return best_a, best_sc

"""
Search for a non‑negative step function that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

where ``conv`` is the linear convolution.  The algorithm proceeds in two
stages:

1. **L‑BFGS continuation** – a smooth log‑sum‑exp approximation of the
   max‑convolution is minimised, decreasing the smoothing parameter
   geometrically to a very small value.  Random restarts and mixing with
   the current best solution explore the landscape extensively.

2. **LP refinement (peak‑cutting)** – for the current candidate ``f`` the
   current peak ``M = max(f∗f)`` is computed.  Keeping ``f`` fixed, the
   linear program

       max 1ᵀg   s.t.   g ≥ 0 ,   f∗g ≤ M

   is solved (the convolution is linear in ``g``).  The optimal
   ``g₀`` is rescaled to a probability vector and mixed back into ``f``:
   ``f ← (1−t)·f + t·g₀`` with a modest step‑size ``t``.  Repeating this
   step reduces the peak and therefore the final score.  The LP is
   solved with HiGHS via ``scipy.optimize.linprog`` and a sparse Toeplitz
   constraint matrix.

The whole procedure respects a 1000 s wall‑time budget and finally
returns the best sequence found, normalised to sum = 1.  The returned
list can be fed directly to ``evaluate_sequence``.
"""

                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (f, grad) where f = 2·n·smooth_max(conv) and grad = ∂f/∂y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution values
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Approximate a square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'exp', 'power', 'bimodal',
                       'sharp', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for every λ in ``lam_schedule``, starting each stage from
    the previous result.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def lp_refine(a: np.ndarray,
              n: int,
              fft_len: int,
              deadline: float,
              t: float = 0.2):
    """
    LP‑based peak‑cutting refinement.
    Repeatedly solves:
        max 1ᵀg   s.t.   g ≥ 0 ,   (a ∗ g) ≤ M,
    where M = max(a ∗ a).  The solution g₀ is normalised to sum = 1
    and mixed back into a with step‑size t.
    """
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # current peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        if M <= 0:
            break

        # Build sparse Toeplitz matrix: (A g)_k = Σ_i a_i * g_{k-i}
        # Each diagonal offset i contains the constant a_i.
        diags_data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        A = sp.diags(diags_data, offsets,
                     shape=(2 * n - 1, n), format='csr')
        b = np.full(2 * n - 1, M)

        # Linear program   maximise 1ᵀg   → minimise –1ᵀg
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c=c, A_ub=A, b_ub=b,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g0 = res.x
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s

        # convex combination and renormalisation
        a_new = (1.0 - t) * a + t * g
        a_new /= a_new.sum()

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] it={it}, improved fast_score={best_sc:.8f}")
        else:
            # shrink step if no progress
            t *= 0.5
            if t < 1e-4:
                break
        it += 1
    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power‑of‑two FFT length works best)
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(chebwin_init(n, attenuation=rng.uniform(40.0, 80.0)))
    candidates.append(dpss_init(n, half_bw=rng.uniform(1.5, 5.0)))

    # random draws
    for _ in range(10):
        candidates.append(random_initial(n, rng))

    # keep the best initial candidate (fast_score assumes sum=1)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with a fine smoothing schedule
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s   # ~55 % of the budget
    lam_schedule = np.geomspace(5e-1, 5e-8, num=70)   # finer schedule
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # random mixture to diversify the start point
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based peak‑cutting refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * budget_s   # remaining ≈30 %
        print("[LP] start refinement")
        best_a, best_sc = lp_refine(best_a, n, fft_len, lp_deadline, t=0.2)
        print(f"[LP] finished with fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final sanity‑check & return
    # --------------------------------------------------------------
    final_seq = np.clip(best_a, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
