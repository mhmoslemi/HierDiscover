    start = time.time()
    budget = budget_s
    deadline = start + budget - 0.5

    # Choose n
    n = 8192  # maybe adjust down if near deadline?

    fft_len = next_pow2(2*n - 1)

    # Build candidate list
    candidates = []

    # Interpolate previous best if exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        # upsample
        x_old = np.linspace(0, 1, prev.size)
        x_new = np.linspace(0, 1, n)
        a0 = np.interp(x_new, x_old, prev)
        a0 = np.clip(a0, 0, None)
        a0 = a0 / a0.sum()
        candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # Chebyshev windows
    try:
        from scipy.signal.windows import chebwin
        for ripple in [30, 60, 80]:
            w = chebwin(n, ripple)
            w = np.maximum(w, 0)
            w /= w.sum()
            candidates.append(w)
    except Exception:
        pass

    # Random draws
    for _ in range(15):
        candidates.append(random_initial(n, rng))

    # Choose best
    best_a = candidates[0]
    best_score = fast_score(best_a, fft_len, n)
    for a in candidates[1:]:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_a = a
            best_score = sc
    print(...)

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n-1):(n-1+n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def eg_exact_max(a_start, deadline, n, fft_len, init_lr=0.5):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        # gradient: g_j = 2 * a_{k_max - j}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def hill_climb(a, cur_score, fft_len, n, deadline, rng, max_iter=50000, delta_frac=0.04):
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # Ensure non-negative after move (i may have become slightly negative due to floating rounding)
        # but delta <= best[i] so cand[i] >= 0.
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

"""
Improved search for a low‑score step function.

The algorithm works in three stages for a given size `n`:
1. **Diverse initial pool** – deterministic windows (Hann,
   Blackman, Kaiser, DPSS, Chebyshev …) together with many random
   Dirichlet draws.  The best of these is kept as a starting point.
2. **Smooth L‑BFGS refinement** – we minimise a log‑sum‑exp
   approximation of the L∞ norm of the auto‑convolution.
   A short λ‑schedule (down to 1e‑9 · max) is used and many restarts
   are performed while respecting a global time budget.
3. **Exact exponentiated‑gradient + hill‑climb** – a sub‑gradient
   descent on the true max‑convolution objective followed by a greedy
   mass‑transfer hill‑climb (coarse then fine moves).

The whole search is first attempted with `n = 8192`; if time permits a
shorter run with `n = 4096` is performed and the best sequence overall
is returned.  The public `evaluate_sequence` function is called only
once, at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient
    w.r.t. the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic windows + random draws)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def chebwin_init(n: int, ripple: float = 60.0) -> np.ndarray:
    """Chebyshev window (fallback to Hann if unavailable)."""
    try:
        from scipy.signal.windows import chebwin
        w = chebwin(n, at=ripple)  # 'at' is the attenuation in dB
    except Exception:          # pragma: no cover
        w = hann_init(n)        # safe fallback
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a variety of simple distributions on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # dirichlet
        concentration = rng.uniform(0.2, 5.0)
        a = rng.dirichlet(np.full(n, concentration))
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT (fast)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/sqrt(t))
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.04):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.  Only accepts
    strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Core search for a given problem size
# ----------------------------------------------------------------------


def run_search_for_n(n: int,
                     rng: np.random.Generator,
                     overall_deadline: float):
    """
    Perform the three‑phase optimisation for a specific length `n`.
    Returns the best probability vector and its fast_score.
    """
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse initial pool
    # ------------------------------------------------------------------
    candidates = []

    # initialise from a previously found sequence, if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
                candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # a few Chebyshev variants
    for rip in (30, 60, 80):
        candidates.append(chebwin_init(n, ripple=rip))

    # random draws
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time allocation (fractions of the remaining budget)
    # ------------------------------------------------------------------
    total_time = overall_deadline - time.time()
    if total_time <= 0:
        return best_a, best_score

    phase1_frac, phase2_frac, phase3_frac = 0.30, 0.55, 0.15
    phase1_end = time.time() + phase1_frac * total_time

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS (≈30 % of remaining budget)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7, 5e-8, 1e-8, 5e-9, 1e-9
    ]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient on max‑conv (≈55 % of budget)
    # ------------------------------------------------------------------
    if time.time() < overall_deadline - 5.0:
        eg_deadline = overall_deadline - 5.0  # leave a little cushion
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < overall_deadline:
        hill_deadline = overall_deadline
        # coarse moves
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=80000,
            delta_frac=0.04
        )
        # fine moves
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=40000,
            delta_frac=0.01
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin – the caller will abort if we exceed budget_s
    overall_deadline = start_time + budget_s - 0.5

    best_sequence = None
    best_overall_score = float('inf')

    # First try a larger problem size; if time permits we also try a
    # smaller size and keep the best overall result.
    for n in (8192, 4096):
        if time.time() >= overall_deadline:
            break
        # allocate a proportion of the remaining time to this size
        remaining = overall_deadline - time.time()
        if n == 8192:
            # give the large run about 60 % of the remaining budget
            sub_deadline = time.time() + 0.60 * remaining
        else:
            # the remaining time goes to the smaller run
            sub_deadline = overall_deadline

        a, sc = run_search_for_n(n, rng, sub_deadline)
        if sc < best_overall_score:
            best_overall_score = sc
            best_sequence = a.tolist()
            print(f"[overall] new best (n={n}) fast_score={sc:.8f}")

    # Final official evaluation (the function rescales internally)
    final_eval = evaluate_sequence(best_sequence)   # pylint: disable=undefined-variable
    print(f"[final] evaluated score = {final_eval:.8f}")

    return best_sequence
