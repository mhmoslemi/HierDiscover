    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    while time.time() < deadline:
        # find gradient approximations

        max_transfer = best[i]
        if max_transfer < 1e-10:
            i,j = rng.integers(0,n, size=2)
            continue
        # try transfer fractions
        for fraction in [0.5, 0.2, 0.1, 0.05, 0.01]:
            delta = max_transfer * fraction
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best = cand
                best_sc = sc
                break

"""
Searcher for a low‑score step function.

The algorithm consists of three stages:

1. **Broad deterministic / parametric sweep** – generate many candidates
   (window families, exponential / power laws, beta densities,
   and the same windows multiplied by a linear ramp).  Keep the best
   according to the exact fast objective
   `2·n·max(conv(a,a))`.

2. **Continuous refinement** – run L‑BFGS on a smooth
   log‑sum‑exp approximation of the max‑convolution.
   Several random restarts are performed; the best result is kept.

3. **Precise greedy polishing** – pairwise mass‑transfer moves with a
   simple line‑search that accept only strictly improving steps.
   This is a cheap deterministic hill‑climb that often pushes the
   score a little further.

All work respects the global budget (≈ 1000 s).  The final sequence
is returned as a plain Python list; it is subsequently evaluated with
the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective (independent of scaling)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial families (all return a vector that sums to 1)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def exp_init(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        return uniform_init(n)
    return a / a_sum


def power_init(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return uniform_init(n)
    return a / s


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return uniform_init(n)
    return a / s


def linear_ramp(base: np.ndarray, slope: float) -> np.ndarray:
    """Multiply a base window by a linear ramp.
    slope>0 makes the right side larger, slope<0 makes the left side larger.
    """
    n = base.size
    ramp = np.linspace(1.0 - slope, 1.0 + slope, n)
    ramp = np.clip(ramp, 0.0, None)
    a = base * ramp
    s = a.sum()
    if s == 0:
        return base.copy()
    return a / s


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    Transformation: x → a = softmax(y)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage
    from the previous result.  Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Precise greedy hill‑climb (pairwise transfer with line search)
# ----------------------------------------------------------------------


def hill_climb_precise(a: np.ndarray,
                       fft_len: int,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_iters: int = 200000):
    """
    Deterministic hill‑climb that moves mass from the currently
    “worst” index to the “best” one.  A simple back‑tracking line
    search ensures only strictly improving updates are kept.
    """
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # compute sub‑gradient of the exact max‑convolution
        fa = np.fft.rfft(best, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)

        rev = best[::-1]
        rev_f = np.fft.rfft(rev, fft_len)
        mask_f = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(rev_f * mask_f, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        i = np.argmax(g)   # index we would like to decrease
        j = np.argmin(g)   # index we would like to increase
        if i == j or best[i] < 1e-12:
            # fallback to a random pair
            i, j = rng.integers(0, n, size=2)
            if i == j or best[i] < 1e-12:
                it += 1
                continue

        # maximal admissible transfer
        max_transfer = best[i]

        # try a few candidate step sizes (largest first)
        improved = False
        for frac in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            delta = max_transfer * frac
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best = cand
                best_sc = sc
                improved = True
                break
        if not improved:
            # as a last resort try a tiny random move
            delta = max_transfer * rng.random() * 0.001
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best = cand
                best_sc = sc
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse initial pool (deterministic + parametric sweep)
    # ------------------------------------------------------------------
    candidates = []

    # 1) classic windows (including some random parameterisations)
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(chebwin_init(n, attenuation=rng.uniform(40.0, 80.0)))
    candidates.append(dpss_init(n, half_bw=rng.uniform(1.5, 5.0)))

    # 2) exponential families
    for rate in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
        candidates.append(exp_init(n, rate))

    # 3) power‑law families
    for p in (0.2, 0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        candidates.append(power_init(n, p))

    # 4) beta‑shaped families (grid over α,β)
    alphas = (0.1, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0)
    betas   = (0.1, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0)
    for a in alphas:
        for b in betas:
            candidates.append(beta_init(n, a, b))

    # 5) skewed versions of a few windows
    slopes = (-0.8, -0.6, -0.4, -0.2, 0.2, 0.4, 0.6, 0.8)
    for slope in slopes:
        candidates.append(linear_ramp(hann_init(n), slope))
        candidates.append(linear_ramp(kaiser_init(n, beta=rng.uniform(6.0, 12.0)), slope))

    # 6) random Dirichlet‑like draws (more than before)
    for _ in range(12):
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
        a /= a.sum()
        candidates.append(a)

    # Keep the best candidate from the pool
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] best fast_score after sweep = {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements (multi‑restart)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = np.geomspace(5e-1, 5e-7, num=48)   # finer schedule
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # random convex combination of the current best and a fresh random vector
        rand_vec = np.random.dirichlet(np.ones(n), size=1)[0]
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new global best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the exact objective (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        # Adam working directly on the simplex (softmax of logits)
        y = np.log(best_a + 1e-12)
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        beta1_pow = 1.0
        beta2_pow = 1.0

        best_sc_adam = best_sc
        best_a_adam = best_a.copy()

        t = 0
        while time.time() < deadline:
            t += 1
            a = softmax(y)

            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
            max_c = conv.max()
            mask = (conv >= max_c - 1e-14).astype(np.float64)

            rev = a[::-1]
            rev_f = np.fft.rfft(rev, fft_len)
            mask_f = np.fft.rfft(mask, fft_len)
            rev_grad = np.fft.irfft(rev_f * mask_f, fft_len)
            g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

            # gradient w.r.t. logits
            grad_y = a * (2.0 * n * g_a - np.dot(a, 2.0 * n * g_a))

            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = 0.3 / np.sqrt(t)          # a bit more aggressive than before
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional jitter to escape tiny plateaus
            if t % 4000 == 0:
                y += rng.normal(scale=1e-4, size=y.shape)

            a_curr = softmax(y)
            sc = fast_score(a_curr, fft_len, n)
            if sc < best_sc_adam:
                best_sc_adam = sc
                best_a_adam = a_curr.copy()
                print(f"[Adam] iter={t}, new best fast_score={best_sc_adam:.8f}")

        # adopt the best Adam result
        best_a = best_a_adam
        best_sc = best_sc_adam
        print(f"[Adam] finished with best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Precise greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print("[Hill‑climb‑precise] start")
        best_a, best_sc = hill_climb_precise(best_a,
                                             fft_len,
                                             n,
                                             hill_deadline,
                                             rng)
        print(f"[Hill‑climb‑precise] final fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Final sanity‑check & return
    # ------------------------------------------------------------------
    final_seq = np.clip(best_a, 0.0, None)
    if final_seq.sum() == 0.0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
