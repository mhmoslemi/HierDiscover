   draws to keep diversity.

2. **Exact exponentiated‑gradient descent** – after the smooth phase we
   switch to a true sub‑gradient descent on the non‑smooth objective.
   The exact gradient at the maximal convolution index is
   `g_j = 2·a_{k_max‑j}` (where the index is in range).  An EG update
   stays on the simplex and a decaying step‑size `η = 0.8 / √t` is used.

3. **Fine‑grained hill‑climb** – a greedy random mass‑transfer move
   (move a tiny fraction of mass from one index to another) is applied
   until the time budget expires.  Only strictly improving moves are
   accepted.

All phases respect a total wall‑clock budget of ~1000 s (the value of
`budget_s`), and the best sequence found is returned.  The sequence is
normalised to sum = 1; the official `evaluate_sequence` is scale‑invariant,
so this does not affect the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss
# -------------------------------------------------------------
# Utility functions
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# -------------------------------------------------------------
# Fast objective (exact) – used for bookkeeping
# -------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) and its gradient – for L‑BFGS
# -------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient
    w.r.t. the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -------------------------------------------------------------
# Initialisation helpers
# -------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a vector on the simplex from various simple distributions."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:                        # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# -------------------------------------------------------------
# Optimisation primitives
# -------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)                      # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.8):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/√t)
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.05):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.  Only accepts
    strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative sequence with a low evaluate_sequence score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Problem size
    # ---------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ---------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ---------------------------------------------------------
    candidates = []

    # reuse previously found construction if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # random draws
    for _ in range(16):
        candidates.append(random_initial(n, rng))

    # pick the best by the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Time budget handling
    # ---------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.4  # small safety margin

    # ---------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS (≈35 % of budget)
    # ---------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.8, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03,
        0.02, 0.01, 5e-3, 2e-3, 1e-3,
        5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 5000

    while time.time() < phase1_end:
        # blend current best with a fresh random guess
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.45        # up to 45 % random
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient (≈50 % of budget)
    # ---------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0   # keep a few seconds for polishing
        print("[EG‑max] start exponentiated‑gradient")
        best_a, best_score = eg_exact_max(best_a,
                                          eg_deadline,
                                          n,
                                          fft_len,
                                          init_lr=0.8)
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Phase 3 – fine‑grained hill‑climb (≈15 % of budget)
    # ---------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=150000,
                                        delta_frac=0.05)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Final official evaluation (scale‑invariant, but we call it for the log)
    # ---------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
