def build_convolution_matrix(a):
    n = a.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)

    j_arr = np.arange(n, dtype=np.int32)

        cols[pos:pos+n] = j_arr
        data[pos:pos+n] = a[i]
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def hill_climb_refine(a, n, fft_len, deadline):
    best_a = a
    best_M = conv_max(a, fft_len)
    best_score = 2*n*best_M
    rng = np.random.default_rng()
    max_iter = int(5e5)
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it+=1; continue
        delta = best_a[i] * rng.random() * 0.05
        if delta <= 0:
            it+=1; continue
        # move mass
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Ensure non-negativity maybe automatically; cand[i] >=0
        if cand[i] < 0:
            it+=1; continue
        M_cand = conv_max(cand, fft_len)
        if M_cand < best_M:
            best_a = cand
            best_M = M_cand
            best_score = 2*n*best_M
        it+=1

"""
Simple Frank‑Wolfe / LP based optimizer for the max‑self‑convolution problem.

The method starts from a good known solution (or the uniform vector) and
repeatedly:
  1. computes M = max(a * a);
  2. solves a linear program  max∑b_j  s.t. a * b ≤ M, b≥0;
  3. rescales the LP solution to a probability vector;
  4. convex‑combines the current a with the LP solution (line‑search over a few
     mixing weights) and keeps the best mixture.
The LP is explicit because a * b is linear in b.  After a few LP steps we
polish the result with a cheap greedy hill‑climb.

The routine respects the 1000 s wall‑time, works with n≈2500 (≈100 MiB sparse
matrix) and usually reaches a score ≤ 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the (linear) self‑convolution a*a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2n‑1)×n) such that A @ b = a * b.
    Row k corresponds to convolution index k (0‑based).
    """
    n = a.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    # reuse a column index vector
    j = np.arange(n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + j          # row index = i + j
        cols[pos:pos + n] = j              # column index = j
        data[pos:pos + n] = a[i]           # coefficient = a_i
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max∑b_j  s.t. a*b ≤ M, b≥0.
    Returns the raw solution (not normalised) or None on failure.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)               # (2n‑1)×n matrix
    # Scale constraints to RHS = 1 (i.e. (a*b)/M ≤ 1)
    scale = 1.0 / M
    A_scaled = A.multiply(scale)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)      # minimise -∑b_j  ⇒ maximise ∑b_j
    bounds = [(0.0, None)] * n

    # HiGHS is fast and accepts sparse matrices
    res = linprog(c, A_ub=A_scaled, b_ub=b_ub,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x


def mix_with_line_search(a: np.ndarray, b_raw: np.ndarray,
                         fft_len: int, cur_M: float) -> tuple[np.ndarray, float]:
    """
    Rescale the LP solution to a probability vector and try a few
    convex combinations with the current a. Returns the best new a and its
    convolution maximum.
    """
    sum_b = b_raw.sum()
    if sum_b <= 0.0:
        return a, cur_M
    b = b_raw / sum_b                      # now ∑b_j = 1

    best_a = a
    best_M = cur_M

    # try a handful of mixing coefficients (including pure b)
    for t in (0.05, 0.10, 0.20, 0.30, 0.50, 0.70, 1.0):
        a_mix = (1.0 - t) * a + t * b
        M_mix = conv_max(a_mix, fft_len)
        if M_mix < best_M:
            best_M = M_mix
            best_a = a_mix
    return best_a, best_M


def lp_iterative_optimize(a: np.ndarray, fft_len: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP + line‑search steps (Frank‑Wolfe style). Stops when no
    improvement is possible or time runs out.
    Returns the refined a and its fast‑score (2 n · max conv).
    """
    n = a.shape[0]
    cur_M = conv_max(a, fft_len)
    best_score = 2.0 * n * cur_M
    it = 0

    while time.time() < deadline:
        it += 1
        b_raw = solve_lp(a, cur_M)
        if b_raw is None:
            print("[LP] solver failed – terminating LP loop")
            break
        sum_b = b_raw.sum()
        # If we cannot increase the sum beyond the current normalisation,
        # the current point is (approximately) optimal.
        if sum_b <= 1.0 + 1e-9:
            print(f"[LP] iteration {it}: sum(b)≈{sum_b:.6f} ≤ 1 – stop")
            break

        a_new, M_new = mix_with_line_search(a, b_raw, fft_len, cur_M)

        if M_new < cur_M - 1e-15:          # genuine improvement
            a = a_new
            cur_M = M_new
            best_score = 2.0 * n * cur_M
            print(f"[LP] it {it:02d} – max(conv)={cur_M:.9f}  score={best_score:.8f}")
        else:
            print(f"[LP] iteration {it}: no improvement – stop")
            break
    return a, best_score


def hill_climb(a: np.ndarray, fft_len: int,
               deadline: float, max_iter: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a tiny amount of probability mass from
    one index to another if it strictly reduces the max‑convolution.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_M = conv_max(best_a, fft_len)
    best_score = 2.0 * n * best_M

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a small fraction of the mass at i to j
        delta = best_a[i] * rng.random() * 0.05   # up to 5 % of a[i]
        if delta <= 0.0:
            it += 1
            continue
        cand = best_a.copy()
        cand[i] -= delta
        if cand[i] < 0.0:               # numerical guard
            it += 1
            continue
        cand[j] += delta
        M_cand = conv_max(cand, fft_len)
        if M_cand < best_M:
            best_a = cand
            best_M = M_cand
            best_score = 2.0 * n * best_M
        it += 1
    return best_a, best_score

# Main entry point as required by the framework
# ----------------------------------------------------------------------

    """

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 2500 gives a favorable memory / time trade‑off.
    # ------------------------------------------------------------------
    n = 2500
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the best known solution if available,
    # otherwise fall back to the uniform vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        if a0.ndim != 1 or a0.size == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            # linear interpolation to the new length
            x_old = np.linspace(0.0, 1.0, a0.size)
            x_new = np.linspace(0.0, 1.0, n)

    else:

    a0 = np.clip(a0, 0.0, None)
