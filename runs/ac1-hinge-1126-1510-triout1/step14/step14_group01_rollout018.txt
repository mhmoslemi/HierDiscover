"""
improved_step_function_search.py

Search for a non‑negative step function (vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear convolution.  The objective is convex;
the implementation mixes several optimisation ideas:

*  diverse deterministic + random initialisations (window functions,
  Beta, Dirichlet, etc.);
*  smoothed log‑sum‑exp max minimisation with L‑BFGS (many λ‑passes);
*  Adam optimiser on a top‑k mask of the largest convolution entries;
*  Exponentiated‑gradient descent (same top‑k schedule);
*  aggressive hill‑climb (tiny mass transfers);
*  simulated‑annealing on the soft‑max logits to escape local minima;
*  final projected sub‑gradient polishing.

The whole search respects the 1000 s budget and always keeps the
best sequence seen so far.  The returned list may be arbitrarily
scaled – the official `evaluate_sequence` rescales internally.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise (tiny numerical errors)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum≈1).  Returns 2·n·max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))

# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------

                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def dirichlet_initial(n: int, rng: np.random.Generator, alpha: float = 0.3) -> np.ndarray:
    """Sparse Dirichlet draw – useful for low‑peak autocorrelation."""
    a = rng.gamma(alpha, 1.0, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.01):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked max (linear combination of entries)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]  # d max/ d a (ignoring 2·n)

        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # a slightly larger initial step
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def eg_optimize(a_start: np.ndarray,
                deadline: float,
                n: int,
                fft_len: int,
                rng: np.random.Generator,
                init_lr: float = 1.0,
                top_start: float = 0.15,
                top_end: float = 0.005):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    start_time = time.time()
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        top_frac = max(top_end, top_start * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        idxs = np.argpartition(conv, -k)[-k:]
        mask = np.zeros_like(conv)
        mask[idxs] = 1.0

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 500000,
               delta_frac: float = 0.07):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.
    Only strictly improving moves are kept.  Very many iterations
    (bounded by time) are attempted.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a small fraction of mass from i to j
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 0.12,
                        final_temp: float = 0.001):
    """
    Simple SA on the soft‑max logits.  Proposes Gaussian perturbations,
    accepts according to the Metropolis rule, and keeps the best ever.
    """
    current = a_start.copy()
    cur_score = fast_score(current, fft_len, n)
    best = current.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        T = max(final_temp, init_temp * (1.0 - elapsed / total))

        # perturb logits
        y = np.log(current + 1e-12)
        sigma = T * 0.6
        y_prop = y + rng.normal(scale=sigma, size=n)
        cand = softmax(y_prop)

        sc = fast_score(cand, fft_len, n)

        if sc < best_score:
            best_score = sc
            best = cand.copy()

        delta = sc - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            current = cand
            cur_score = sc

    return best, best_score


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses the true arg‑max to build a sub‑gradient and projects onto
    the simplex after each (small) step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = 0.5 / np.sqrt(t)
        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Uses a multi‑phase scheme (L‑BFGS → Adam → EG → hill‑climb →
    simulated‑annealing → sub‑gradient polishing).  The best sequence
    is returned before the 1000 s deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – larger n gives a finer representation but costs
    # a bit more time.  8192 is still very fast on a modern CPU.
    # --------------------------------------------------------------
    n = 8192
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two length for FFTs

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # include previous solution if available (up‑sample if needed)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.22))
    candidates.append(gaussian_init(n, sigma_factor=0.28))
    for beta in (6.0, 9.0, 12.0):
        candidates.append(kaiser_init(n, beta))

    # Beta‑sweep (symmetric)
    for alpha in np.linspace(0.8, 4.0, 7):
        candidates.append(beta_init(n, alpha, alpha))

    # Dirichlet draws (sparse)
    for _ in range(8):
        candidates.append(dirichlet_initial(n, rng, alpha=0.3))

    # Randomised families
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # Pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time allocation (soft)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – Smoothed L‑BFGS (≈15 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule = [
        0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07,
        0.05, 0.03, 0.02, 0.01, 5e-3, 2e-3, 1e-3,
        5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 1e-7
    ]
    maxiter_per_lam = 1500

    # a few random restarts inside the phase
    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.45
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.15,
            end_top=0.01
        )
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent (≈25 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 2.0   # keep a few seconds for polishing
        print("[EG] start")
        best_a, best_score = eg_optimize(
            best_a,
            eg_deadline,
            n,
            fft_len,
            rng,
            init_lr=1.0,
            top_start=0.15,
            top_end=0.005
        )
        print(f"[EG] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – Greedy hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.20 * budget_s)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=800000,
            delta_frac=0.07
        )
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈5 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(30.0, deadline - time.time())
        print("[SA] start")
        best_a, best_score = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=0.12,
            final_temp=0.001
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 6 – Final sub‑gradient polishing (≈5 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        fine_deadline = deadline
        best_a, best_score = fine_tune_subgradient(
            best_a,
            fine_deadline,
            n,
            fft_len
        )
        print(f"[fine‑tune] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
