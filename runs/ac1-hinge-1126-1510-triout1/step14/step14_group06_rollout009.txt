"""
Propose a low‑score step sequence.

Algorithm Overview
------------------
1. **Initial pool** – generate a diverse set of candidates:
   deterministic windows, random Dirichlet draws with many
   concentration parameters, random spectral vectors with random phase
   and a few parametric mixtures (Gaussian, Beta, etc.).
2. **Fast screening** – evaluate all pool members with the exact
   `fast_score` (2·n·max(conv)) and keep the top few.
3. **L‑BFGS polishing** – starting from each elite candidate run a
   quasi‑Newton optimisation on a smoothed max‑convolution
   (log‑sum‑exp) with a very fine λ‑schedule (λ≈max·factor).  The
   optimisation works on logits (soft‑max) so the simplex constraint is
   satisfied automatically.
4. **Adam & Gradient‑descent refinements** – continue optimisation with
   Adam on the same smooth objective, decreasing the smoothing factor.
5. **Directed mass‑transfer** – repeatedly locate the current maximal
   convolution index, compute its exact sub‑gradient and perform a
   small transfer of probability mass from high‑gradient components to
   low‑gradient ones, projecting back onto the simplex.  Accept only
   strictly improving moves (greedy).
6. **Final hill‑climb** – a best‑of‑k greedy local search that tries
   several random source/target pairs per iteration and keeps the
   best improvement.
7. Return the best vector found (as a plain list).  The final
   evaluation with the official `evaluate_sequence` routine is printed
   for reference.

The whole routine respects the user‑supplied time budget (default
1000 s) – the deadline is checked frequently and the algorithm adapts
its work‑load accordingly.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def convolve_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution (full length 2·n‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a probability vector: 2·n·max(conv)."""
    n = a.shape[0]
    conv = convolve_self(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_max_and_grad(
    y: np.ndarray, n: int, fft_len: int, lam_factor: float
) -> tuple[float, np.ndarray]:
    """
    Returns (smooth objective, gradient) w.r.t. logits y.
    The underlying probability vector is a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_val = max_c + lam * np.log(Z)
    obj = 2.0 * n * smooth_val

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    back = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]

    # extract gradient w.r.t. a (size n) centred at positions n‑1 … n‑1+n‑1
    grad_a = back[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)

    return obj, grad_y

def spectral_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random‑phase spectral guess (projected onto the simplex)."""
    L = next_pow2(2 * n - 1)
    # random magnitudes (uniform) and random phases
    mags = np.sqrt(rng.random(L // 2 + 1))
    phases = rng.uniform(0, 2 * np.pi, size=mags.shape)
    # construct Hermitian symmetric spectrum
    spec = np.zeros(L, dtype=np.complex128)
    spec[: mags.shape[0]] = mags * np.exp(1j * phases)
    spec[-(mags.shape[0] - 1) :] = np.conj(spec[1 : mags.shape[0] - 1][::-1])
    a = np.real(np.fft.ifft(spec))[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def random_dirichlet_init(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x /= x.sum()
    return x


def gaussian_mixture_init(
    n: int, rng: np.random.Generator, k: int = 2
) -> np.ndarray:
    """Mixture of `k` Gaussians with random means, sigmas and weights."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    weights = rng.random(k)
    weights /= weights.sum()
    for w in weights:
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.05 * n, 0.3 * n)
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a

def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule: list[float],
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smoothed max objective for each lambda in
    ``lam_schedule``.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_max_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    lam_init: float = 0.001,
    lr: float = 0.07,
    max_iter: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.  λ is kept constant
    (lam_init) – this works well once we are already near a good local
    minimum.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = smooth_max_and_grad(y, n, fft_len, lam_init)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def exact_subgradient_step(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute a single feasible descent direction for the exact max‑convolution.
    Returns the updated probability vector (projected onto the simplex).
    """
    n = a.shape[0]
    conv = convolve_self(a, fft_len)
    i_max = int(conv.argmax())

    # gradient of conv[i_max] w.r.t. a:
    g = np.zeros_like(a)
    # j such that 0 <= j < n and 0 <= i_max - j < n
    j_low = max(0, i_max - (n - 1))
    j_high = min(i_max, n - 1)
    # for valid j, derivative is 2 * a[i_max - j]
    rev_idx = i_max - np.arange(j_low, j_high + 1)
    g[j_low : j_high + 1] = 2.0 * a[rev_idx]

    # move opposite to the gradient, keeping sum=1
    # simple projected gradient step with diminishing stepsize
    step = 0.2 / np.sqrt(np.sum(g**2) + 1e-12)  # normalise step size
    d = -g
    a_new = a + step * d
    a_proj = simplex_projection(a_new)
    return a_proj


def directed_mass_transfer(
    a: np.ndarray,
    fft_len: int,
    max_moves: int,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Greedy mass‑transfer moves guided by the exact sub‑gradient of the
    current maximal convolution entry.  Only strictly improving moves
    are kept.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)
    moves = 0

    while time.time() < deadline and moves < max_moves:
        # sub‑gradient step (projected)
        a_candidate = exact_subgradient_step(best_a, fft_len)

        sc = fast_score(a_candidate, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate
        moves += 1
    return best_a


def greedy_hill_climb(
    a: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 200000,
    delta_frac: float = 0.08,
) -> np.ndarray:
    """
    Strictly improving greedy hill‑climb.
    Each iteration tries a few random source/target pairs and keeps the
    best improvement.
    """
    n = a.shape[0]
    best = a.copy()
    best_sc = fast_score(best, fft_len)
    it = 0
    while time.time() < deadline and it < max_iter:
        # try K random moves, keep the best one
        K = 5
        best_local = None
        best_local_sc = best_sc
        for _ in range(K):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = best[i] * rng.random() * delta_frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = cand
        if best_local is not None:
            best = best_local
            best_sc = best_local_sc
        it += 1
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 5000 entries works well for the benchmark.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a large and diverse pool of initial candidates
    # ------------------------------------------------------------------
    pool = []

    # deterministic windows
    pool.append(hann_init(n))
    pool.append(blackman_init(n))
    pool.append(kaiser_init(n, beta=9.0))
    pool.append(triangular_init(n))

    # random spectral vectors
    for _ in range(4):
        pool.append(spectral_init(n, rng))

    # Dirichlet draws with many concentration parameters
    alphas = [0.3, 0.6, 0.9, 1.2, 2.0, 5.0]
    for alpha in alphas:
        for _ in range(6):
            pool.append(random_dirichlet_init(n, rng, alpha))

    # Gaussian mixtures (2 components)
    for _ in range(5):
        pool.append(gaussian_mixture_init(n, rng, k=2))

    # If a previous solution exists, up‑sample it (helps escape local minima)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            pool.append(a0)

    # ------------------------------------------------------------------
    # Fast screening – keep the top candidates
    # ------------------------------------------------------------------
    scores = [(fast_score(c, fft_len), c) for c in pool]
    scores.sort(key=lambda x: x[0])
    elite = [c for _, c in scores[:8]]
    best_a = elite[0].copy()
    best_sc = fast_score(best_a, fft_len)
    print(f"[init] n={n}, best fast_score from pool = {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------

    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
        5e-08,
        2e-08,
        1e-08,
        5e-09,
        2e-09,
        1e-09,
        5e-10,
        2e-10,
        1e-10,
    ]

    maxiter_per_lam = 8000  # a bit more thorough than the reference

        if time.time() >= phase1_end:
            break
        a_opt, sc_opt = lbfgs_refine(
            start_vec, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")

    restarts = 3
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet_init(n, rng, alpha=1.0)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(
            a_start, lam_schedule, maxiter_per_lam // 2, phase1_end, fft_len, n

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑restart] improved best fast_score={best_sc:.8f}")
        restarts -= 1

            lr=0.06,

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a = greedy_hill_climb(
            best_a,
            fft_len,
            deadline,
            rng,
            max_iter=300000,
            delta_frac=0.10,
        )
        best_sc = fast_score(best_a, fft_len)
        print(f"[Final hill‑climb] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)  # official routine
    print(f"[final] evaluated score = {final_score:.8f}")

    # expose the best sequence for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
