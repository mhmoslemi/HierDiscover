def edge_shape(n, edge_frac, edge_weight):
    m = max(1, int(n * edge_frac))
    # allocate edge_weight mass to edges (both sides total)
    edge_mass = min(edge_weight, 1.0)
    interior_mass = 1.0 - edge_mass
    a = np.zeros(n)
    if 2*m >= n:
        # edges overlap, just fill uniformly
        a[:] = 1.0 / n
    else:
        # edges halves
        a[:m] = edge_mass / (2*m)  # half of edge weight each side? Actually allocate half to each side.
        a[-m:] = edge_mass / (2*m)
        interior_len = n - 2*m
        if interior_len > 0:
            a[m:-m] = interior_mass / interior_len
    # ensure sum 1
    a /= a.sum()
    return a
```

"""
Propose a low‑score step sequence.

Algorithm (high‑level)
-----------------------
1. Build a rich pool of initial probability vectors:
   • classic windows (hann, blackman, kaiser, gaussian, …)
   • random draws, power‑law, inverse‑sqrt, arcsine
   • new families emphasising the edges (edge‑shape),
     a two‑sided exponential (double‑exp) and alternating sign.
   • parametric families (Beta, mixed Beta, edge‑shape, alternating)
     explored with a cheap differential‑evolution optimiser.

2. Pick the best candidate according to the exact objective
   `fast_score = 2·n·max(self‑convolution)`.

3. Intensive L‑BFGS polishing.
   The smoothed‑max objective uses a *real* λ schedule (previous code
   used a λ‑factor → the schedule was ineffective).  Starting with a
   large λ we gradually reduce it, letting the optimiser first
   “see” a smooth landscape and then converge to the true nonsmooth
   maximum.  Several random restarts are performed until the time
   budget for this phase expires.

4. Follow‑up local optimisation phases:
   – Adam on the same smoothed objective,
   – Exponentiated‑gradient on the exact max,
   – Sub‑gradient descent on the exact max,
   – Simulated annealing (mass‑transfer moves),
   – A final greedy hill‑climb that only accepts strictly
     improving moves.

5. Return the best vector found; it is finally evaluated with the
   official `evaluate_sequence` routine.

All phases respect the global 1000 s budget.  The implementation is
pure NumPy/SciPy and works on a single process (the heavy work is
FFT‑based and therefore fast enough for n≈5000).
"""

import time

from scipy.optimize import minimize, differential_evolution

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Exact score (fast) – assumes a is a probability vector (sum=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Smoothed maximum of the convolution (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_max_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
):
    """
    Smoothed max of self‑convolution with a *fixed* smoothing parameter λ.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    shifted = conv - max_c
    # avoid overflow / underflow
    exp_shift = np.exp(np.clip(shifted / lam, -50, 50))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z  # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:

    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()

def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:

    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

def edge_shape(n: int, edge_frac: float, edge_weight: float) -> np.ndarray:
    """
    U‑shaped distribution: a fraction `edge_frac` of indices at each side
    receives `edge_weight` of the total mass (split equally between both sides).
    The remaining mass is spread uniformly over the interior.
    """
    edge_frac = float(edge_frac)
    edge_weight = float(edge_weight)
    m = max(1, int(n * edge_frac))
    a = np.zeros(n, dtype=np.float64)

    # if the edge blocks overlap, fall back to uniform
    if 2 * m >= n:
        a[:] = 1.0 / n
    else:
        # split edge mass equally left/right
        left_mass = right_mass = edge_weight / 2.0
        a[:m] = left_mass / m
        a[-m:] = right_mass / m
        interior_len = n - 2 * m
        if interior_len > 0:
            a[m:-m] = (1.0 - edge_weight) / interior_len
    a /= a.sum()
    return a


def double_exp_shape(n: int, decay: float) -> np.ndarray:
    """
    Two‑sided exponential that is largest at the edges and decays towards
    the centre.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)  # distance to nearest edge
    a = np.exp(-decay * d)
    a /= a.sum()
    return a


def alternating_shape(n: int, amp: float) -> np.ndarray:
    """
    High‑low alternating pattern.  amp∈[0,1) controls the amplitude.
    """
    base = 1.0
    a = base + amp * ((-1) ** np.arange(n))
    # shift to make all entries non‑negative
    if a.min() < 0:
        a -= a.min()
    a /= a.sum()
    return a

def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_mixture_shape(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    a1 = beta_shape(n, p1, p1)
    a2 = beta_shape(n, p2, p2)
    a = w * a1 + (1.0 - w) * a2
    return a

def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run L‑BFGS over a decreasing λ schedule.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # protect against too‑small λ (numerical blow‑up)
        lam = max(lam, 1e-12)
        try:
            res = minimize(
                fun=smooth_max_obj_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    lam: float = 1e-4,
    lr: float = 0.05,
    max_iter: int = 40000,
):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = smooth_max_obj_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    top_fraction: float = 0.02,
):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def subgradient_exact_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    step0: float = 0.30,
):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of the active entry
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low : j_high + 1] = 2.0 * sub

        # project onto the simplex (zero‑sum direction)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    final_temp: float = 1e-8,
    max_moves: int = 600000,
):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.25
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_sc:
                best_sc = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_sc


def hill_climb(
    a: np.ndarray,
    cur_sc: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 300000,
    delta_frac: float = 0.09,
):
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)  # legacy safety

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed, as in the reference implementation)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Assemble a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) previous solution, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates.append(a0)

    # 2) deterministic spectral / window families
    candidates.extend(
        [
            spectral_init(n),
            triangular_init(n),
            hann_init(n),
            blackman_init(n),
            gaussian_init(n),
            kaiser_init(n, beta=9.0),
            inv_sqrt_init(n),
            arcsine_init(n),
        ]
    )

    # 3) random families
    for _ in range(9):
        candidates.append(random_initial(n, rng))
    for _ in range(6):
        candidates.append(pow_decay_init(n, rng))

    # 4) new edge‑heavy families
    for ef in (0.01, 0.025, 0.05):
        for ew in (0.5, 0.7, 0.9):
            candidates.append(edge_shape(n, ef, ew))
    for decay in (0.001, 0.003, 0.01):
        candidates.append(double_exp_shape(n, decay))
    for amp in (0.3, 0.6, 0.9):
        candidates.append(alternating_shape(n, amp))

    # ------------------------------------------------------------------
    # Low‑dimensional parametric search (Beta, mixed Beta, edge, alternating)
    # ------------------------------------------------------------------

    # single Beta
    def beta_obj(params):
        p, q = params
        a = beta_shape(n, p, q)
        return fast_score(a, fft_len, n)

    if time.time() < deadline:
        bounds = [(0.1, 2.5), (0.1, 2.5)]
        result_beta = differential_evolution(
            beta_obj,
            bounds=bounds,
            maxiter=250,
            popsize=20,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        print("[DE] single Beta best params:", result_beta.x, "score:", result_beta.fun)
        candidates.append(beta_shape(n, result_beta.x[0], result_beta.x[1]))

    # mixture of two Betas
    def beta_mix_obj(params):
        p1, p2, w = params
        a = beta_mixture_shape(n, p1, p2, w)
        return fast_score(a, fft_len, n)

    if time.time() < deadline:
        bounds_mix = [(0.1, 2.5), (0.1, 2.5), (0.0, 1.0)]
        result_mix = differential_evolution(
            beta_mix_obj,
            bounds=bounds_mix,
            maxiter=250,
            popsize=20,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        print("[DE] Beta mixture best params:", result_mix.x, "score:", result_mix.fun)
        candidates.append(
            beta_mixture_shape(n, result_mix.x[0], result_mix.x[1], result_mix.x[2])
        )

    # edge‑shape DE
    def edge_obj(params):
        ef, ew = params
        # force reasonable ranges
        ef = np.clip(ef, 1e-4, 0.45)
        ew = np.clip(ew, 0.0, 1.0)
        a = edge_shape(n, ef, ew)
        return fast_score(a, fft_len, n)

    if time.time() < deadline:
        bounds_edge = [(0.0001, 0.45), (0.0, 1.0)]
        result_edge = differential_evolution(
            edge_obj,
            bounds=bounds_edge,
            maxiter=150,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        print("[DE] Edge shape best params:", result_edge.x, "score:", result_edge.fun)
        candidates.append(edge_shape(n, result_edge.x[0], result_edge.x[1]))

    # alternating DE
    def alt_obj(params):
        amp = np.clip(params[0], 0.0, 0.99)
        a = alternating_shape(n, amp)
        return fast_score(a, fft_len, n)

    if time.time() < deadline:
        bounds_alt = [(0.0, 0.99)]
        result_alt = differential_evolution(
            alt_obj,
            bounds=bounds_alt,
            maxiter=150,
            popsize=15,
            seed=int(rng.integers(1e9)),
            polish=False,
            updating="deferred",
        )
        print("[DE] Alternating shape best amp:", result_alt.x, "score:", result_alt.fun)
        candidates.append(alternating_shape(n, result_alt.x[0]))

    # ------------------------------------------------------------------
    # Select the best start (exact fast_score)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    # λ schedule – absolute values (starting relatively large)
    lam_schedule = [
        1e-3,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
        5e-7,
        2e-7,
        1e-7,
    ]
    maxiter_per_lam = 7000

    best_a, best_sc = lbfgs_refine(
        best_a, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
    )
    print(f"[L‑BFGS] best fast_score={best_sc:.8f}")

    # Random restarts within Phase 1
    restarts = 4
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(
            a_start,
            lam_schedule,
            maxiter_per_lam // 2,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_sc:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.50 * budget_s
    if time.time() < phase2_end:
        best_a, best_sc = adam_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            lam=lam_schedule[-1],  # final, small λ
            lr=0.04,
            max_iter=40000,
        )
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.60 * budget_s
    if time.time() < phase3_end:
        best_a, best_sc = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, top_fraction=0.02
        )
        print(f"[Exp‑grad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact sub‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.70 * budget_s
    if time.time() < phase4_end:
        best_a, best_sc = subgradient_exact_optimize(
            best_a, phase4_end, n, fft_len, step0=0.30
        )
        print(f"[Sub‑grad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.85 * budget_s
    if time.time() < phase5_end:
        best_a, best_sc = simulated_annealing(
            best_a,
            phase5_end,
            n,
            fft_len,
            rng,
            init_temp=6e-4,
            final_temp=1e-8,
            max_moves=800000,
        )
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=600000,
            delta_frac=0.11,
        )
        print(f"[final hill‑climb] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official scoring routine
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
