"""
Propose a step sequence minimizing the evaluation score.

The algorithm builds a diverse pool of deterministic windows (triangular,
Hann, Blackman, Kaiser, Gaussian, arcsine, sine‑power, Beta‑distribution
shapes), samples random Dirichlet vectors and convex mixtures, picks the
best among them (according to the exact fast score), then refines the
solution with several optimisation phases:

1. L‑BFGS on a smoothed max‑convolution objective (log‑sum‑exp) with a
   fine λ‑schedule.
2. Projected sub‑gradient descent on the true max‑convolution using a
   simple line‑search.
3. A final greedy hill‑climb that transfers small amounts of mass
   between coordinates while staying on the probability simplex.

All phases respect the given time budget (default 1000 s). The final
sequence (a list of non‑negative floats) is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Window families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def sin_power_init(n: int, p: float = 2.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p, symmetric, non‑negative."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def beta_window_init(
    n: int, alpha: float = 0.5, beta: float = 0.5, eps: float = 1e-12
) -> np.ndarray:
    """
    Symmetric Beta‑distribution shape on {0,…,n‑1}.
    a_i ∝ x^{α‑1} (1‑x)^{β‑1} with x = (i+0.5)/n.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Work in log‑space to avoid overflow/underflow.
    log_a = (alpha - 1.0) * np.log(np.clip(x, eps, 1.0 - eps)) + (
        beta - 1.0
    ) * np.log(np.clip(1.0 - x, eps, 1.0 - eps))
    a = np.exp(log_a)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a

def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run L‑BFGS on the smoothed max objective (log‑sum‑exp) for the whole
    λ schedule.  Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits for softmax
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-13, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    step0: float = 0.35,
):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses a simple back‑tracking line‑search to accept moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # compute convolution and locate maximum
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a : g_i = 2 * a_{idx-i} (if in range)
        g = np.zeros_like(a)
        # valid i are where j = idx - i lies in [0, n-1]
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        # slice of a that contributes
        g[low : high + 1] = 2.0 * a[idx - np.arange(low, high + 1)]

        # zero‑sum projection: keep sum(a) = 1
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # step size schedule
        alpha = step0 / np.sqrt(it + 1.0)

        # simple back‑tracking line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # fallback to projection without improvement
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 500000,
    delta_frac: float = 0.10,
):
    """
    Strictly improving greedy hill‑climb.
    Moves a random fraction of mass from one coordinate to another.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 5000 items give a good trade‑off between quality
    # and runtime.
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an extensive pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # Deterministic classical windows
    candidates.append(uniform_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.20))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(gaussian_init(n, sigma_factor=0.30))
    candidates.append(gaussian_init(n, sigma_factor=0.35))
    candidates.append(gaussian_init(n, sigma_factor=0.40))
    candidates.append(kaiser_init(n, beta=4.0))
    candidates.append(kaiser_init(n, beta=6.0))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(kaiser_init(n, beta=10.0))
    candidates.append(arcsine_init(n))

    # Sine‑power windows (p from 0.5 to 4.0)
    for p in np.linspace(0.5, 4.0, 8):
        candidates.append(sin_power_init(n, p=p))

    # Symmetric Beta windows (α = β)
    for a in np.linspace(0.2, 5.0, 12):
        candidates.append(beta_window_init(n, alpha=a, beta=a))

    # Random Dirichlet draws
    for _ in range(30):
        candidates.append(random_dirichlet(n, rng))

    # Convex mixtures of uniform with each deterministic window
    uniform = uniform_init(n)
    for base in candidates[2:]:  # skip the already‑uniform one
        for w in np.linspace(0.0, 1.0, 11):
            a = (1.0 - w) * uniform + w * base
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
                candidates.append(a)

    # Optionally interpolate a previously exported solution if present.
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
                candidates.append(a0)

    # --------------------------------------------------------------
    # Pick the best start according to the exact fast_score
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈45 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
        5e-08,
        2e-08,
        1e-08,
        5e-09,
        2e-09,
        1e-09,
        5e-10,
        2e-10,
        1e-10,
    ]
    maxiter_per_lam = 8000  # more iterations per λ

    best_a, best_score = lbfgs_refine(
        best_a, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
    )
    print(f"[L‑BFGS] best fast_score = {best_score:.8f}")

    # Additional random restarts (still within Phase 1)
    restarts = 6
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(
            a_start,
            lam_schedule,
            maxiter_per_lam // 2,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score = {best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – Exact‑max sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_exact_optimize(
            best_a, phase2_end, n, fft_len, step0=0.45
        )
        print(f"[Sub‑grad] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=800000,
            delta_frac=0.12,
        )
        print(f"[final hill‑climb] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
