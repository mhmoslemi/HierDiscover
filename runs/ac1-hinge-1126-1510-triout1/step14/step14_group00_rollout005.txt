def local_max_reduce(a, n, fft_len, max_iters, rng):

    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    for it in range(max_iters):
        # compute conv and find max index
        conv_full = np.fft.irfft(np.fft.rfft(best, fft_len) * np.fft.rfft(best, fft_len), fft_len)[:2*n-1]
        max_idxs = np.argwhere(conv_full >= conv_full.max() - 1e-12).flatten()
        # Choose a random max index
        k = rng.choice(max_idxs)
        # Determine i candidates that contribute to this max: i such that a_i > 0 and a_{k-i} > 0.
        # but we can instead choose a random i in [0,n-1] and j = k - i
        i = rng.integers(0, n)
        j = k - i
        if 0 <= j < n and best[i] > 1e-12 and best[j] > 1e-12:
            # Transfer a small fraction from i to neighbor or j?
            # Actually we want reduce a_i * a_j term (since it contributes to max). Decrease either i or j and move mass elsewhere.
            # Choose donor = i or j whichever larger, move delta to random other index l with low a.
            donor = i if best[i] > best[j] else j
            other = i if donor == j else j
            # pick recipient l far from donor maybe.
            l = rng.integers(0, n)
            if l == donor: continue
            delta = best[donor] * rng.random() * 0.05
            # ensure not reduce product too much causing other values condition.
            cand = best.copy()
            cand[donor] -= delta
            cand[l] += delta
            # No negative check
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best = cand
                best_sc = sc
    return best, best_sc

Define function refine_lp(a, M, n, solver='ECOS'):

    b = cp.Variable(n, nonneg=True)
    # constraints:
    # For each k, (a * b)[k] <= M
    constraints = []
    # Precompute for each k the indices and coefficients of a.
    for k in range(2*n - 1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # coefficients of b[j] where j = k-i
        # So we can generate row vector of shape n with a[i] at position j = k-i.
        # Use cp.sum(cp.multiply(a_slice, b_slice))
        # For each i in i_start..i_end:
        #   term = a[i] * b[k-i]
        # So list of indexes for b: j = k - i
        # we can make expression sum_{i} a[i] * b[j].
        # So we can index b with these indices.
        # Since cp.Variable can't be indexed with list? It can via b[j].
        # We'll construct expression via cp.sum(cp.multiply(a[i_start:i_end+1], b[k-i_start:k-i_start+(i_end-i_start+1)]))
        a_slice = a[i_start:i_end+1]
        b_slice = b[k - i_start:k - i_start + (i_end - i_start +1)]
        constraints.append(cp.sum(cp.multiply(a_slice, b_slice)) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    prob.solve(solver=solver, warm_start=True, max_iters=2000)  # maybe set options.

"""
Cascading multi‑scale optimizer for the non‑negative step‑function problem.

The search proceeds in two stages.
1.  A coarse problem (n≈1024) is solved with a rich set of
    deterministic/random initialisations, L‑BFGS continuation,
    Adam (top‑k & smooth‑max) and a single linear‑program refinement
    that maximises the total mass while keeping the self‑convolution
    below the current maximum.
2.  The coarse solution is interpolated to the full size (n=8192) and
    refined again by Adam (top‑k), Adam (smooth‑max), simulated annealing
    and a greedy hill‑climb.

All updates respect the probability simplex (non‑negative,
sum = 1) via a soft‑max parametrisation, and the global 1000 s budget
is split dynamically between the phases.  The final sequence is
returned as a plain Python list; ``evaluate_sequence`` will rescale it
internally.
"""

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (equals the official score when the sequence
# sums to 1).
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30,
                       edge_weight: float = 0.0) -> tuple[float, np.ndarray]:
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    if edge_weight > 0.0:
        pen_grad = np.zeros_like(a)
        pen_grad[0] = 2.0 * edge_weight * a[0]
        pen_grad[-1] = 2.0 * edge_weight * a[-1]
        g_vec = g_vec + pen_grad

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike
        a = np.full(n, 1.0 / n)
        idx = rng.integers(0, n)
        a[idx] = rng.uniform(2.0, 5.0) / n
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 9.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, att: float = 80.0) -> np.ndarray:
    if chebwin is None:
        return kaiser_init(n, beta=8.0)
    a = chebwin(n, at=att)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, halfband: float = 0.2) -> np.ndarray:
    if dpss is None:
        return kaiser_init(n, beta=8.0)
    a = dpss(n, halfband, Kmax=1)[0]
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Optimisation primitives (all operate on probability vectors)

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam, 0.0),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 5e-2,
                         lam_end: float = 1e-7):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(
            y, n, fft_len, fft_len,
            lam_factor=lam / (a @ a) if (a @ a) > 0 else lam,
            edge_weight=0.0)

        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# Linear‑program refinement (optional, requires cvxpy)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, max_conv: float) -> np.ndarray:
    """
    Solve  max Σ b_i   s.t.  b_i ≥ 0  and  (a * b)[k] ≤ max_conv  ∀k.
    Returns a normalised vector (sum=1). If solving fails we return the
    input unchanged.
    """
    try:
        import cvxpy as cp
    except Exception:
        return a

    n = a.shape[0]
    if toeplitz is None:
        return a

    # Build Toeplitz matrix of the convolution operator
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)   # shape (2n‑1, n)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= max_conv]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=500, warm_start=True)
    except Exception:
        return a

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return a

    b_val = np.maximum(b_var.value, 0.0)
    total = b_val.sum()
    if total < 1e-12:
        return a
    return b_val / total

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Multi‑scale search for a low‑autocorrelation step function.
    Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Coarse problem (n≈1024) – heavy optimisation
    # ------------------------------------------------------------------
    n_coarse = 1024
    fft_len_coarse = next_pow2(2 * n_coarse - 1)

    # Initialise a diverse pool
    init_pool = [
        spectral_init(n_coarse),
        triangular_init(n_coarse),
        hann_init(n_coarse),
        blackman_init(n_coarse),
        gaussian_init(n_coarse),
        kaiser_init(n_coarse),
    ]
    for _ in range(8):
        init_pool.append(random_initial(n_coarse, rng))

    best_a = init_pool[0].copy()
    best_sc = fast_score(best_a, fft_len_coarse, n_coarse)
    for cand in init_pool[1:]:
        sc = fast_score(cand, fft_len_coarse, n_coarse)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init coarse] n={n_coarse}, best fast_score={best_sc:.9f}")

    # --------- L‑BFGS continuation (short but multiple restarts) ----------
    phase_end = time.time() + 0.25 * (deadline - time.time())
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7
    ]
    while time.time() < phase_end:
        rand_vec = random_initial(n_coarse, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam=1500,
            deadline=phase_end,
            fft_len_conv=fft_len_coarse,
            fft_len_g=fft_len_coarse,
            n=n_coarse
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS coarse] new best fast_score={best_sc:.9f}")

    # --------- Adam with top‑k mask ----------
    if time.time() < deadline:
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n_coarse,
            fft_len_coarse,
            rng,
            start_top=0.20,
            end_top=0.0005
        )
        print(f"[Adam‑topk coarse] best fast_score={best_sc:.9f}")

    # --------- Adam on smooth‑max ----------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.20 * (deadline - time.time())
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n_coarse,
            fft_len_coarse,
            rng,
            lam_start=5e-2,
            lam_end=1e-7
        )
        print(f"[Adam‑smooth coarse] best fast_score={best_sc:.9f}")

    # --------- Linear‑program refinement (optional) ----------
    if time.time() < deadline:
        M = np.convolve(best_a, best_a).max()
        refined = lp_refine(best_a, M)
        sc_ref = fast_score(refined, fft_len_coarse, n_coarse)
        if sc_ref < best_sc:
            best_sc = sc_ref
            best_a = refined.copy()
            print(f"[LP refine] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Upscale to the full size (n=8192) and final polishing
    # ------------------------------------------------------------------
    n_big = 8192
    fft_len_big = next_pow2(2 * n_big - 1)

    # Linear interpolation to the larger grid
    x_coarse = np.linspace(0, 1, n_coarse)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_coarse, best_a)
    a_big = np.clip(a_big, 0.0, None)
    a_big /= a_big.sum()
    best_a = a_big.copy()
    best_sc = fast_score(best_a, fft_len_big, n_big)
    print(f"[upscale] n={n_big}, fast_score={best_sc:.9f}")

    # --------- Adam top‑k on full size ----------
    if time.time() < deadline:
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n_big,
            fft_len_big,
            rng,
            start_top=0.12,
            end_top=0.0005
        )
        print(f"[Adam‑topk big] best fast_score={best_sc:.9f}")

    # --------- Adam smooth‑max on full size ----------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.15 * (deadline - time.time())
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n_big,
            fft_len_big,
            rng,
            lam_start=2e-2,
            lam_end=1e-8
        )
        print(f"[Adam‑smooth big] best fast_score={best_sc:.9f}")

    # --------- Simulated annealing ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n_big,
            fft_len_big,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=250000
        )
        print(f"[SA big] best fast_score={best_sc:.9f}")

    # --------- Greedy hill‑climb ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 12.0)
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len_big,
            n_big,
            hill_deadline,
            rng,
            max_iter=180000,
            delta_frac=0.04
        )
        print(f"[hill‑climb big] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.9f}")
    return final_seq
