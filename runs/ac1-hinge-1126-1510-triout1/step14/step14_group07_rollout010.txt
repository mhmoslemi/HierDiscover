"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
* Choose a moderately large problem size (n≈10 000) – a larger n gives
  more degrees of freedom for flattening the self‑convolution, while
  the FFT‑based fast_score remains cheap.
* Build a diverse pool of starting points:
    – uniform, linear, exponential, cosine‑type windows,
      random Dirichlet samples,
      the previously‑found solution (upsampled if needed).
* Pick the best starter by the exact fast_score.
* Phase 1 – L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
  A long λ‑schedule (high smoothing → fine smoothing) and many iterations
  per λ give a high‑quality local optimum.
* Phase 2 – Several random L‑BFGS restarts from perturbed versions of the
  current best point (helps escape shallow basins).
* Phase 3 – Adam optimisation on the same smooth objective (tiny smoothing).
* Phase 4 – Exponentiated‑gradient descent that concentrates on the top‑fraction
  of convolution entries.
* Phase 5 – Exact‑max sub‑gradient descent with a decreasing step size.
* Phase 6 – Simulated annealing that mixes two move types:
    – *mass transfer* (move a small amount from one index to another),
    – *swap* (exchange the values of two positions).
  Temperature is geometrically annealed.
* Phase 7 – A final greedy hill‑climb that only accepts strict improvements.
* Return the best vector found (as a plain list of floats).  The official
  ``evaluate_sequence`` is called once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    # full self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def linear_decrease_init(n: int) -> np.ndarray:
    """Linear ramp from 2/(n+1) down to 0 (normalised)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    return a


def exponential_init(n: int, rng: np.random.Generator, rate: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(-rate * i)  (i = 0…n‑1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a = a / a.sum()
    return a


def cosine_window_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def upsample_previous(prev: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a previous solution to a new length."""
    if prev.ndim != 1 or prev.size == 0:
        return uniform_init(n)
    x_old = np.linspace(0, 1, prev.size)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, prev)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = uniform_init(n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.04,
                  max_iter: int = 300000):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.015):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    focusing on the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # contribution from a_i * a_{idx-i}
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = sub
        # add symmetric term when i == idx-i
        if idx % 2 == 0:
            i_mid = idx // 2
            if 0 <= i_mid < n:
                g[i_mid] += a[i_mid]
        g *= 2.0

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """
    Simulated Annealing with two move types:
        – mass transfer (move a fraction of one entry to another),
        – swap (exchange two entries).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        # geometric temperature schedule
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        # choose move type
        if rng.random() < 0.7:
            # mass‑transfer move (as in the original code)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                move += 1
                continue
            delta = a[i] * rng.random() * 0.25
            if delta <= 0.0:
                move += 1
                continue
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
        else:
            # swap move – exchange a[i] and a[j]
            i, j = rng.integers(0, n, size=2)
            if i == j:
                move += 1
                continue
            a_candidate = a.copy()
            a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 600000,
               delta_frac: float = 0.12):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    final_eval = evaluate_sequence(final_seq)   # official score

    # expose the sequence for possible reuse in later runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
