def simulated_annealing(a_start, deadline, n, fft_len, rng):
    best = a_start.copy()
    best_score = fast_score(best, fft_len, n)
    cur = best.copy()
    cur_score = best_score
    # Temperature schedule: start T0 = 0.01, end close to 0.
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        T = max(1e-5, (0.02) * (1 - fraction))  # linear decay

        # pick i, j distinct
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose delta as fraction of a[i], maybe up to 0.1 * a[i]
        delta_max = cur[i] * 0.1
        if delta_max <= 0:
            continue
        delta = rng.random() * delta_max
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

"""
Hybrid optimizer for the step‑function attack.

The algorithm consists of four stages, each respecting a part of the
1000 s wall‑clock budget:

1. **Heavy L‑BFGS smoothing** – we minimise a log‑sum‑exp
   approximation of the maximal self‑convolution.  Several restarts
   from a mixture of the current best vector and a fresh random draw
   are performed.

2. **Exact exponentiated‑gradient descent** – a sub‑gradient method on
   the true non‑smooth objective (the maximum entry of a*a).  This
   quickly pushes mass away from the current worst convolution entry.

3. **Simulated‑annealing hill‑climb** – a stochastic local search that
   repeatedly transfers a small amount of probability mass from one
   index to another, accepting worsening moves with a temperature that
   slowly cools down.  The candidate is kept symmetric (a[i]=a[n‑1‑i])
   because the convolution is symmetric and this reduces the search
   space.

4. **Final short L‑BFGS polish** – a brief continuation of the smoothed
   optimisation starting from the best solution found so far.

All stages use fast FFT‑based convolution (length the next power of two)
and keep the vector on the probability simplex (non‑negative, sum = 1).
The returned list is handed to ``evaluate_sequence`` which rescales it
internally; the optimisation works directly on the normalised vector.

The implementation is fully self‑contained, requires only NumPy and
SciPy, and respects the 1000 s time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = np.sum(e)
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector ``a`` (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient
    w.r.t. the softmax logits ``y`` (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:                     # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto the simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered and
    its exact score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:                     # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT (fast)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a:
        #   g_j = 2 * a_{k_max - j}  (if the index is in range)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/sqrt(t))
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Simple simulated‑annealing hill‑climb.
    Moves consist of transferring a random fraction of the probability
    mass from one index to another; the candidate is kept symmetric
    (a[i] = a[n-1-i]).  Accepts worsening moves with a linearly cooling
    temperature.
    """
    # start from a symmetric copy
    cur = (a_start + a_start[::-1]) * 0.5
    cur_score = fast_score(cur, fft_len, n)

    best = cur.copy()
    best_score = cur_score

    start = time.time()
    start_T = 2e-2
    end_T = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / max(1e-12, (deadline - start))
        T = start_T * (1.0 - frac) + end_T * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move: up to 30 % of the source mass
        max_delta = cur[i] * 0.30
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce symmetry (preserves total mass)
        cand = (cand + cand[::-1]) * 0.5

        # tiny numerical safety – clip negatives and renormalise
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0 or not np.isfinite(s):
            continue
        cand = cand / s

        cand_score = fast_score(cand, fft_len, n)
        dscore = cand_score - cur_score

        if dscore < 0 or rng.random() < np.exp(-dscore / (T + 1e-16)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional tiny perturbation of the best solution to escape
        if rng.random() < 0.001:
            perturb = random_initial(n, rng) * 0.02
            cur = best * 0.98 + perturb
            cur = cur / cur.sum()
            cur_score = fast_score(cur, fft_len, n)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    candidates = []

    # reuse previously discovered construction if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # a few random draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    # Phase cut‑offs (roughly 30 % / 30 % / 30 % / 10 %)
    phase1_end = start_time + 0.30 * budget_s
    phase2_end = start_time + 0.60 * budget_s
    phase3_end = start_time + 0.90 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS smoothing
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05,
        0.02, 0.01, 5e-3, 2e-3, 1e-3,
        5e-4, 2e-4, 1e-4, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 1e-7
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # blend current best with a fresh random draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        eg_deadline = phase2_end - 5.0   # keep a tiny cushion
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        print("[SA] start simulated‑annealing hill‑climb")
        best_a, best_score = simulated_annealing(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final short L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_lam = lam_schedule[-5:]   # a short fine‑tuning schedule
        final_maxiter = 1500
        print("[final L‑BFGS] polishing")
        a_opt, sc_opt = lbfgs_continuation(
            best_a,
            final_lam,
            final_maxiter,
            deadline,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
