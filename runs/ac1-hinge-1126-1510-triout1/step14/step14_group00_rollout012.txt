def subgradient_descent(a_start, deadline, n, fft_len, rng, max_iter=20000):
    a = a_start.copy()
    best = a.copy()
    best_sc = fast_score(a, fft_len, n)

    it = 0
    # step schedule: initially large, then decaying
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]

        max_val = conv.max()
        max_idx = np.argmax(conv)

        # compute gradient of max conv wrt a
        grad = np.zeros(n)
        # vectorized: compute slice for overlapping region
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        # indices where product defined
        idx_range = np.arange(i_start, i_end + 1)
        j = max_idx - idx_range
        # contributions: gradient at idx_range = 2 * a[j]
        grad[idx_range] = 2 * a[j]

        # step size
        # adapt: eta = 0.5 / (1 + it) maybe
        eta = 0.5 / (it ** 0.5)  # decaying
        a_new = a - eta * grad
        # ensure nonnegative: clip
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex (sum=1)
        a_new /= max(a_new.sum(), 1e-12)
        a_new = simplex_projection(a_new)  # ensure exact sum = 1

        a = a_new

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
            # optional: print improvement
    return best, best_sc

def upsample(a_small, n_big):
    x_small = np.linspace(0,1, len(a_small))
    x_big = np.linspace(0,1, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.clip(a_big, 0, None)
    a_big /= a_big.sum()
    return a_big

"""
Propose a low‑autocorrelation step function.

The search works in two resolutions:

* Phase 0 – small problem (n≈2048).  A diverse pool of
  deterministic and random initialisations is explored with
  L‑BFGS continuation (smooth‑max objective).  The best small‑n
  shape is then interpolated to the target size.

* Phase 1 – big problem (n≈8192).  The up‑sampled shape is refined
  by a sequence of fast gradient‑based stages (Adam top‑k, Adam
  smooth‑max) followed by classic heuristics (simulated annealing,
  greedy hill‑climb and a final projected sub‑gradient descent).

All optimisation respects the global time budget (default 1000 s)
and the returned list is a valid non‑negative step function for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Simple helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x, for FFT padding."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation (exact when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact 2·n·max(conv) for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30,
                       edge_weight: float = 0.0) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient.
    `lam_factor` multiplies the current max to obtain λ.
    Returns (objective, gradient) w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)

    # self‑convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z   # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # optional quadratic edge penalty
    if edge_weight > 0.0:
        pen_grad = np.zeros_like(a)
        pen_grad[0] = 2.0 * edge_weight * a[0]
        pen_grad[-1] = 2.0 * edge_weight * a[-1]
        g_vec = g_vec + pen_grad

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation families (deterministic & random)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike
        a = np.full(n, 1.0 / n)
        idx = rng.integers(0, n)
        a[idx] = rng.uniform(2.0, 5.0) / n
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 9.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.  Returns best probability vector and
    its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam, 0.0),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    """
    Adam optimiser that gradually focuses on the true maximum.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (full)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # derivative w.r.t a
        grad_a = 2.0 * n * g_a                         # full objective grad

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 5e-2,
                         lam_end: float = 1e-7):
    """
    Adam optimisation on the log‑sum‑exp smooth max with exponential λ decay.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(y, n, fft_len, fft_len,
                                       lam_factor=lam,
                                       edge_weight=0.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """
    Simple simulated‑annealing on the simplex with tiny mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


def subgradient_descent_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 max_iter: int = 20000):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = a_start.copy()
    best = a.copy()
    best_sc = fast_score(a, fft_len, n)

    v = np.zeros_like(a)          # momentum term
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # full convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_idx = int(np.argmax(conv))

        # compute gradient of the active diagonal
        grad = np.zeros(n)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        j = max_idx - idx_range
        grad[idx_range] = 2.0 * a[j]  # derivative of 2·n·max(conv) w.r.t a

        # momentum‑aided step
        v = beta * v + (1 - beta) * grad
        eta = 0.4 / np.sqrt(it)            # diminishing step size
        a = a - eta * v

        # projection onto simplex (non‑negative, sum = 1)
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    return best, best_sc


def upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a_small to length n_big, followed by simplex projection."""
    x_small = np.linspace(0, 1, len(a_small))
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.clip(a_big, 0.0, None)
    a_big = simplex_projection(a_big)
    return a_big

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises the
    official evaluation function.  The routine respects the given
    time budget and returns a Python list of floats (the sequence).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Phase 0 – small‑scale optimisation (n≈2048)
    # --------------------------------------------------------------
    n_small = 2048
    fft_len_small = next_pow2(2 * n_small - 1)

    # Build a diverse pool for the small problem
    small_candidates = []

    # optionally seed from the previous global result
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # down‑sample previous construction
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n_small)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = simplex_projection(a0)
            small_candidates.append(a0)

    # deterministic families (small‑n versions)
    small_candidates.append(spectral_init(n_small))
    small_candidates.append(triangular_init(n_small))
    small_candidates.append(hann_init(n_small))
    small_candidates.append(blackman_init(n_small))
    small_candidates.append(gaussian_init(n_small))
    small_candidates.append(kaiser_init(n_small, beta=9.0))

    # randomised families
    for _ in range(10):
        small_candidates.append(random_initial(n_small, rng))

    # pick the best start by cheap fast_score
    best_a_small = small_candidates[0]
    best_sc_small = fast_score(best_a_small, fft_len_small, n_small)
    for cand in small_candidates[1:]:
        sc = fast_score(cand, fft_len_small, n_small)
        if sc < best_sc_small:
            best_sc_small = sc
            best_a_small = cand.copy()
    print(f"[phase0‑init] n_small={n_small}, best fast_score={best_sc_small:.9f}")

    start_time = time.time()
    budget_end = start_time + budget_s - 0.5          # safety margin

    # allocate ~30 % of the total budget for the small problem
    small_phase_seconds = min(0.30 * budget_s, 250.0)
    small_deadline = start_time + small_phase_seconds

    lam_schedule_small = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02,
        0.01, 0.005, 0.002, 0.001, 5e-4,
        2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam_small = 1500

    # iterative L‑BFGS refinements on the small problem
    while time.time() < small_deadline:
        rand_vec = random_initial(n_small, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a_small + alpha * rand_vec
        a_start = simplex_projection(a_start)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule_small,
            maxiter_per_lam_small,
            small_deadline,
            fft_len_small,
            fft_len_small,
            n_small,
        )
        if sc_opt < best_sc_small:
            best_sc_small = sc_opt
            best_a_small = a_opt.copy()
            print(f"[phase0‑lbfgs] new best small fast_score={best_sc_small:.9f}")

    # --------------------------------------------------------------
    # Upsample to the target size
    # --------------------------------------------------------------
    n_big = 8192
    fft_len_big = next_pow2(2 * n_big - 1)

    best_a_big = upsample(best_a_small, n_big)
    best_sc_big = fast_score(best_a_big, fft_len_big, n_big)
    print(f"[upsample] big n={n_big}, fast_score={best_sc_big:.9f}")

    # --------------------------------------------------------------
    # Phase 1 – big‑scale refinement (remaining budget)
    # --------------------------------------------------------------
    remaining_deadline = budget_end

    # split the remaining time among the following stages (fractions)
    total_rem = remaining_deadline - time.time()
    if total_rem <= 0:
        total_rem = 0.0

    # 1) optional light L‑BFGS (only a few λ’s, to give a fresh start)
    l_bfgs_frac = 0.10
    l_bfgs_deadline = time.time() + l_bfgs_frac * total_rem
    lam_schedule_big = [
        0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 1e-6
    ]
    maxiter_per_lam_big = 1500
    if time.time() < l_bfgs_deadline:
        print("[big‑lbfgs] start")
        a_opt, sc_opt = lbfgs_continuation(
            best_a_big,
            lam_schedule_big,
            maxiter_per_lam_big,
            l_bfgs_deadline,
            fft_len_big,
            fft_len_big,
            n_big,
        )
        if sc_opt < best_sc_big:
            best_sc_big = sc_opt
            best_a_big = a_opt.copy()
            print(f"[big‑lbfgs] new best fast_score={best_sc_big:.9f}")

    # 2) Adam with top‑k focus
    if time.time() < remaining_deadline:
        print("[Adam‑topk] start")
        best_a_big, best_sc_big = adam_topk_optimize(
            best_a_big,
            remaining_deadline,
            n_big,
            fft_len_big,
            rng,
            start_top=0.12,
            end_top=0.0008,
        )
        print(f"[Adam‑topk] best fast_score={best_sc_big:.9f}")

    # 3) Adam on smooth‑max
    if time.time() < remaining_deadline:
        mid_deadline = time.time() + 0.15 * (remaining_deadline - time.time())
        print("[Adam‑smooth] start")
        best_a_big, best_sc_big = adam_smooth_optimize(
            best_a_big,
            mid_deadline,
            n_big,
            fft_len_big,
            rng,
            lam_start=5e-2,
            lam_end=1e-7,
        )
        print(f"[Adam‑smooth] best fast_score={best_sc_big:.9f}")

    # 4) Simulated annealing
    if time.time() < remaining_deadline:
        sa_deadline = time.time() + 0.15 * (remaining_deadline - time.time())
        print("[SA] start")
        best_a_big, best_sc_big = simulated_annealing(
            best_a_big,
            sa_deadline,
            n_big,
            fft_len_big,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=350000,
        )
        print(f"[SA] best fast_score={best_sc_big:.9f}")

    # 5) Greedy hill‑climb
    if time.time() < remaining_deadline:
        hill_deadline = time.time() + 0.15 * (remaining_deadline - time.time())
        print("[hill‑climb] start")
        best_a_big, best_sc_big = hill_climb(
            best_a_big,
            best_sc_big,
            fft_len_big,
            n_big,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04,
        )
        print(f"[hill‑climb] best fast_score={best_sc_big:.9f}")

    # 6) Final projected sub‑gradient descent (short but aggressive)
    if time.time() < remaining_deadline:
        subgrad_deadline = time.time() + 0.10 * (remaining_deadline - time.time())
        print("[sub‑grad] start")
        best_a_big, best_sc_big = subgradient_descent_optimize(
            best_a_big,
            subgrad_deadline,
            n_big,
            fft_len_big,
            rng,
            max_iter=25000,
        )
        print(f"[sub‑grad] best fast_score={best_sc_big:.9f}")

    # ------------------------------------------------------------------
    # Return the result (evaluate_sequence rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a_big.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
