def subgrad_exact_max(a_start, deadline, n, fft_len, init_lr=0.5, top_k=5):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        # get top indices
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        # compute gradient as average of subgradients for each idx
        grad = np.zeros_like(a)
        for k in top_idx:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2 * a[idx[mask]]
        grad /= top_k
        # update using exponentiated gradient (multiplicative)
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        # project onto simplex
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

"""
Improved search for a low‑autocorrelation step function.

The algorithm consists of four stages:

1. **Parameter sweep** – generate many classical windows (Beta,
   polynomial, Gaussian, Kaiser, DPSS, etc.) with a fine grid of
   parameters, keep the best (fast_score = 2·n·max(conv)).
2. **L‑BFGS refinement** – minimise a smooth log‑sum‑exp approximation
   of the max‑convolution with many restarts and a decreasing
   smoothing schedule.
3. **Exact sub‑gradient descent** – use the true gradient of the
   maximal convolution entry (averaged over the top‑k entries) with an
   exponentiated‑gradient update on the probability simplex.
4. **Directed hill‑climb** – greedy tiny mass‑transfer moves until the
   time budget expires.

All stages respect the 1000 s wall‑clock limit.  The final vector is
returned as a plain list; `evaluate_sequence` rescales it internally.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for internal optimisation)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len: int,
    lam_factor: float = 0.30,
):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient
    w.r.t. the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    # use same FFT length for both transforms
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # gradient wrt a (size n) – centre slice of length n
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Candidate generators (parameterised families)
# ----------------------------------------------------------------------


def beta_window(n, alpha, beta):
    x = np.linspace(0.0, 1.0, n)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def poly_window(n, p):
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def gaussian_window(n, sigma_factor):
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def exponential_edge(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def hann_pow_window(n, p):
    i = np.arange(n, dtype=np.float64)
    base = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = base ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def sin_pow_window(n, p):
    i = np.arange(n, dtype=np.float64)
    base = np.sin(np.pi * (i + 0.5) / n)  # zero at the ends
    a = base ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def kaiser_window(n, beta):
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def dpss_window(n, NW):
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len,
    n,
):
    """
    Run L‑BFGS over the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def subgrad_exact_max(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    init_lr: float = 0.5,
    top_k: int = 5,
):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximal convolution entry, averaged over the top‑k
    entries to obtain a more stable direction.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # select top‑k indices (might contain duplicates if max occurs many times)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in top_idx:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad /= top_k

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto the simplex
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 100000,
    delta_frac: float = 0.04,
):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.
    Accepts only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a random fraction of the mass at i to j
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – we use a fairly large n (8192) to give the optimizer
    # more freedom to flatten the autocorrelation.
    n = 8192
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool via a parameter sweep
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # deterministic windows (baseline)
    candidates.append(hann_pow_window(n, 1.0))  # plain Hann
    candidates.append(kaiser_window(n, 8.0))
    candidates.append(dpss_window(n, 3.5))

    # systematic grids
    alphas = np.linspace(1.2, 3.0, 7)   # >1 to keep ends at zero
    betas = alphas  # symmetric for now
    for a_val in alphas:
        for b_val in betas:
            candidates.append(beta_window(n, a_val, b_val))

    poly_ps = np.linspace(0.5, 2.5, 9)
    for p in poly_ps:
        candidates.append(poly_window(n, p))

    sigma_factors = np.linspace(0.12, 0.30, 9)
    for sf in sigma_factors:
        candidates.append(gaussian_window(n, sf))

    rates = np.linspace(0.0005, 0.02, 9)
    for r in rates:
        candidates.append(exponential_edge(n, r))

    hann_ps = np.linspace(0.5, 3.0, 9)
    for p in hann_ps:
        candidates.append(hann_pow_window(n, p))

    sin_ps = np.linspace(0.5, 3.0, 9)
    for p in sin_ps:
        candidates.append(sin_pow_window(n, p))

    kaiser_betas = np.linspace(3.0, 12.0, 10)
    for b in kaiser_betas:
        candidates.append(kaiser_window(n, b))

    dpss_nws = np.linspace(2.0, 5.0, 10)
    for nw in dpss_nws:
        candidates.append(dpss_window(n, nw))

    # random draws (Dirichlet‑like)
    for _ in range(15):
        mode = rng.choice(['uniform', 'exp', 'power', 'beta'])
        if mode == 'uniform':
            a = np.full(n, 1.0 / n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:  # beta‑like
            a = rng.beta(0.5, 0.5, size=n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a)

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score after sweep = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS refinement (≈50 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.50 * budget_s
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.1,
        0.05,
        0.02,
        0.01,
        5e-3,
        2e-3,
        1e-3,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
        5e-7,
        1e-7,
    ]
    maxiter_per_lam = 4000

    while time.time() < phase1_end:
        # mix current best with a fresh random draw to diversify restarts
        rand_vec = candidates[rng.integers(len(candidates))]
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = simplex_projection(a_start)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = deadline - 5.0  # keep a cushion for polishing
        print("[Sub‑grad] start exact sub‑gradient descent")
        best_a, best_score = subgrad_exact_max(
            best_a,
            subgrad_deadline,
            n,
            fft_len,
            init_lr=0.6,
            top_k=7,
        )
        print(f"[Sub‑grad] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print("[Hill‑climb] start greedy hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04,
        )
        print(f"[Hill‑climb] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation (official scoring)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
