"""
Improved step‑function optimiser.

The pipeline is:

1. **Broad candidate sweep** – generate a large and diverse set of
   window‑type sequences (Kaiser, DPSS, Gaussian, sine‑power, Tukey,
   Hann, Blackman, …) as well as many random Dirichlet draws and the
   previous best sequence (if available).  Each candidate is evaluated
   with the exact fast score (the same quantity that ``evaluate_sequence``
   returns, but without the internal clamping).  The top‑k candidates
   are kept for further refinement.

2. **Non‑smooth L‑BFGS continuation** – starting from the current best
   vector we run a multi‑stage L‑BFGS optimisation on a log‑sum‑exp
   smooth approximation of the max‑convolution.  The schedule of the
   smoothing parameter goes from 5·10⁻¹ down to 10⁻⁹, and we perform a
   large number of short restarts (convex combination of the incumbent
   and a fresh random draw).  All iterates stay on the probability simplex
   thanks to a soft‑max re‑parameterisation.

3. **Exact exponentiated‑gradient (EG) descent** – the true sub‑gradient
   of the non‑smooth objective is used to perform a multiplicative EG
   update with a 1/√t step‑size decay.  Several independent EG runs are
   launched (different random starts) and the best trajectory is kept.

4. **Annealed hill‑climb** – a greedy mass‑transfer local search that
   also accepts uphill moves with a slowly cooling temperature.  Tiny
   random transfers (down to 0.01 % of the donor mass) are tried many
   times; this fine‑tunes the sequence and typically squeezes a few
   thousandths out of the objective.

All stages respect a total wall‑clock budget of ``budget_s`` seconds.
The final sequence (a list of floats) is returned; it will be scored
by ``evaluate_sequence`` after the function exits.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows as win

# --------------------------------------------------------------
# Utility routines
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    out = e / e.sum()
    # numerical safety
    out = np.clip(out, 0.0, 1.0)
    out /= out.sum()
    return out

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # normalise to guard against tiny rounding errors
    s = w.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / n)
    return w / s

# --------------------------------------------------------------
# Exact fast score (the quantity we want to minimise)
# --------------------------------------------------------------

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# --------------------------------------------------------------
# Initialisation helpers (various windows / random draws)
# --------------------------------------------------------------

def kaiser_window(n: int, beta: float) -> np.ndarray:
    a = win.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def dpss_window(n: int, NW: float) -> np.ndarray:
    w = win.dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def gaussian_window(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def sine_power_window(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def tukey_window(n: int, alpha: float) -> np.ndarray:
    a = win.tukey(n, alpha)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    # parameters >0 give a Dirichlet distribution; use Gamma draws.
    alpha = rng.uniform(0.2, 5.0, size=n)
    a = rng.gamma(alpha, 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def previous_sequence(n: int, rng: np.random.Generator) -> np.ndarray | None:
    """If a global variable ``height_sequence_1`` exists, interpolate it."""
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
                return a0
    return None

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 iters: int,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent on the exact non‑smooth objective.
    Runs for a fixed number of iterations.
    Returns the best iterate found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    for t in range(1, iters + 1):
        # convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) wrt a (sub‑gradient)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update with decaying step‑size
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb_anneal(a: np.ndarray,
                      cur_score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      init_delta_frac: float = 0.1,
                      min_delta_frac: float = 0.005,
                      decay_rate: float = 0.94):
    """
    Greedy hill‑climb with a simple annealing schedule.
    ``delta_frac`` (the maximum fraction of a donor's mass that may be
    moved) decays exponentially from ``init_delta_frac`` to
    ``min_delta_frac``.  Moves that improve the exact score are always
    accepted; worsening moves are accepted with a Boltzmann probability
    that also decays.
    """
    best = a.copy()
    best_score = cur_score
    delta_frac = init_delta_frac
    temperature = 0.1  # controls acceptance of uphill moves

    while time.time() < deadline and delta_frac > min_delta_frac:
        # inner loop at a fixed delta_frac
        it = 0
        max_it = 20000          # per delta level
        while it < max_it and time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                it += 1
                continue
            # amount to transfer from i to j
            max_transfer = best[i] * delta_frac
            if max_transfer <= 0:
                it += 1
                continue
            delta = rng.random() * max_transfer

            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta

            sc = fast_score(cand, fft_len, n)
            diff = sc - best_score
            if diff < 0:
                best = cand
                best_score = sc
            else:
                # accept with probability exp(-diff / T)
                if rng.random() < np.exp(-diff / temperature):
                    best = cand
                    best_score = sc
            it += 1
        # decay parameters
        delta_frac *= decay_rate
        temperature *= decay_rate
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Phase 0 – generate a large pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # previous best, if any
    prev = previous_sequence(n, rng)
    if prev is not None:
        candidates.append(prev)

    # deterministic families with parameter sweeps
    # Kaiser β
    for beta in np.arange(0.5, 15.5, 0.5):
        candidates.append(kaiser_window(n, beta))

    # DPSS NW
    for NW in np.arange(1.0, 6.1, 0.5):
        candidates.append(dpss_window(n, NW))

    # Gaussian σ
    for sigma in np.linspace(0.12, 0.45, 9):
        candidates.append(gaussian_window(n, sigma))

    # Sine‑power windows
    for p in np.linspace(0.5, 3.5, 13):
        candidates.append(sine_power_window(n, p))

    # Tukey windows
    for a in np.linspace(0.1, 1.0, 10):
        candidates.append(tukey_window(n, a))

    # Classical windows (no parameters)
    candidates.append(win.hann(n))
    candidates.append(win.blackman(n))
    candidates.append(win.bartlett(n))          # triangular
    candidates.append(win.flattop(n))
    candidates.append(win.parzen(n))
    candidates.append(win.nuttall(n))

    # Random Dirichlet draws
    for _ in range(30):
        candidates.append(random_dirichlet(n, rng))

    # --------------------------------------------------------------
    # Quick screening – keep the best few
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score from pool = {best_score:.8f}")

    # Keep only the top 5 distinct candidates for the expensive phases
    # (including the current best)
    scored = [(fast_score(c, fft_len, n), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [c for _, c in scored[:top_k]]

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        5e-1, 3e-1, 2e-1, 1e-1, 5e-2, 2e-2, 1e-2,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8,
        5e-9, 1e-9
    ]
    maxiter_per_lam = 5000

    while time.time() < phase1_end:
        # pick a random base from the top candidates
        base = rng.choice(top_candidates)
        # mix it with a fresh random draw to escape plateaus
        rand_vec = random_dirichlet(n, rng)
        alpha = rng.random() * 0.4  # up‑to 40 % new mass
        a_start = (1.0 - alpha) * base + alpha * rand_vec
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – multiple EG runs (≈50 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0   # keep cushion for final polish
        remaining = eg_deadline - time.time()
        # how many independent EG runs? allocate roughly 10 s per run
        n_runs = max(1, int(remaining // 12))
        it_per_run = int(1e5)   # large enough to converge

        print(f"[EG‑max] launching {n_runs} independent runs, {it_per_run} iters each")
        for run in range(n_runs):
            # random start or a mixture with current best
            if run == 0:
                start_vec = best_a.copy()
            else:
                rand_vec = random_dirichlet(n, rng)
                mix = rng.random() * 0.3
                start_vec = (1 - mix) * best_a + mix * rand_vec
                start_vec /= start_vec.sum()
            a_opt, sc_opt = eg_exact_max(start_vec, it_per_run, n, fft_len, init_lr=0.6)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[EG] run {run+1}: improved best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – annealed hill‑climb (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_score = hill_climb_anneal(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            init_delta_frac=0.12,
            min_delta_frac=0.001,
            decay_rate=0.93
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # A final call to the official scorer for logging purposes.
    final_eval = evaluate_sequence(final_seq)   # assumes it is imported
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
