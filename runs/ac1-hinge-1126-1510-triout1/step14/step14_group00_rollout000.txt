def adam_subgrad_optimize(a_start, deadline, n, fft_len, rng, eps=1e-12):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t=0
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)
    while time.time()<deadline:
        t+=1
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_c = conv.max()
        # active mask
        mask = (conv >= max_c - eps).astype(np.float64)
        # gradient w.r.t a of conv max (Sum of gradients)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fw, fft_len)
        grad_a = rev_grad[(n-1):(n-1+n)]
        # gradient of objective f = 2*n*max_c
        grad_a = 2.0 * n * grad_a
        # project gradient onto tangent space of simplex? Actually for update we will apply Adam on raw a after projection.
        # We'll use simple direct step: a_new = a - lr * grad_a, then simplex projection.
        # Use Adam scaling
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a ** 2)
        beta1_pow = beta1**t
        beta2_pow = beta2**t
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = 0.01 / (np.sqrt(t))   # maybe adjust.
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # ensure nonneg, project onto simplex
        a = simplex_projection(a)
        # evaluate
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

1. Build a diverse pool of initial candidates (deterministic windows,
   spectral guess, random families, and a possible previous solution).
2. Pick the best after a cheap fast_score evaluation.
3. Phase 1 – many short L‑BFGS runs on a log‑sum‑exp smooth
   approximation of the max‑convolution (continuation in the smoothing
   factor λ).  This quickly reduces large peaks.
4. Phase 2 – Adam optimisation with a top‑k focus.  The mask contains
   the k largest convolution entries and the gradient is obtained by
   a convolution in the Fourier domain.
5. Phase 3 – Adam optimisation on the smooth‑max objective with an
   exponential λ‑decay (fine‑grained smoothing).
6. Phase 4 – Simulated annealing for a global stochastic search.
7. Phase 5 – Exact sub‑gradient Adam:  the gradient of the *true*
   max‑convolution (mask = points attaining the maximum) is computed
   via FFT and Adam updates are performed directly on the probability
   vector, projecting onto the simplex after every step.
   This phase pushes the solution closer to the theoretical lower
   bound (≈1.0) and typically improves the score by a few 10⁻³.
8. Return the best sequence found (the objective is scale‑invariant, so
   the vector is already normalised to sum = 1).  The official
   `evaluate_sequence` function is called once at the end for reporting.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalise to avoid numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Fast exact score (used for bookkeeping; equals official score
# when the sequence sums to 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families (deterministic & random)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random probability vectors."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike
        a = np.full(n, 1.0 / n)
        idx = rng.integers(0, n)
        a[idx] = rng.uniform(2.0, 5.0) / n
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 9.0) -> np.ndarray:
    # Using numpy's kaiser implementation (always present)
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, att: float = 80.0) -> np.ndarray:
    """Chebyshev window (fallback to Kaiser if unavailable)."""
    try:
        from scipy.signal import chebwin
        a = chebwin(n, at=att)
    except Exception:
        a = np.kaiser(n, 8.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, halfband: float = 0.22) -> np.ndarray:
    """DPSS (Slepian) window – first eigenvector."""
    try:
        from scipy.signal.windows import dpss
        a = dpss(n, halfband, Kmax=1)[0]
    except Exception:
        a = np.kaiser(n, 8.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30,
                       edge_weight: float = 0.0) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient.
    Returns (objective, gradient) w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # optional edge penalty
    if edge_weight > 0.0:
        pen = np.zeros_like(a)
        pen[0] = 2.0 * edge_weight * a[0]
        pen[-1] = 2.0 * edge_weight * a[-1]
        g_vec = g_vec + pen

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best probability vector and
    its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam, 0.0),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    """
    Adam optimiser that gradually focuses on the true maximum.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (full)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # derivative w.r.t a (ignoring 2·n factor)
        grad_a = 2.0 * n * g_a                         # full objective gradient

        # softmax back‑prop
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 5e-2,
                         lam_end: float = 1e-7):
    """
    Adam optimiser on the log‑sum‑exp smooth max with exponential λ decay.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(y, n, fft_len, fft_len,
                                       lam_factor=lam / (a @ a) if (a @ a) > 0 else lam,
                                       edge_weight=0.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """
    Simple simulated‑annealing on the simplex with tiny mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


def adam_subgrad_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eps: float = 1e-12):
    """
    Adam on the *exact* max‑convolution sub‑gradient.
    Each iteration:
        - compute full self‑convolution,
        - build mask = {indices attaining the max (within eps)},
        - obtain gradient via convolution of a_rev with mask,
        - take an Adam step on the probability vector,
        - project back onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    while time.time() < deadline:
        t += 1

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # indicator of active constraints (max entries)
        mask = (conv >= max_c - eps).astype(np.float64)

        # gradient of max(conv) w.r.t a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fw, fft_len)
        grad_a = rev_grad[(n - 1):(n - 1 + n)]

        # gradient of the full objective 2·n·max_c
        grad_a = 2.0 * n * grad_a

        # Adam update on *raw* a (not logits)
        m = beta1 * m + (1.0 - beta1) * grad_a
        v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.02 / np.sqrt(t)   # modest learning rate
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep a feasible probability vector
        a = simplex_projection(a)

        # bookkeeping
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises the
    supplied `evaluate_sequence` function.  The routine respects the
    given time budget (≈1000 s) and returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # Problem size – a power of two gives cheap FFTs.
    n = 8192                       # can be increased if time permits
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # 1) If a previous best exists, interpolate it to the current size.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(chebwin_init(n, att=80.0))
    candidates.append(dpss_init(n, halfband=0.22))

    # 3) Random families (more to increase diversity)
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # Pick the current best by the cheap fast_score.
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # create a perturbed start point
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam with top‑k focus (≈20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.12,
            end_top=0.0008
        )
        print(f"[Adam‑topk] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam on smooth‑max (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.15 * budget_s
        print("[Adam‑smooth] start")
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n,
            fft_len,
            rng,
            lam_start=5e-2,
            lam_end=1e-7
        )
        print(f"[Adam‑smooth] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=350000
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – Exact sub‑gradient Adam (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining, 0.15 * budget_s)
        print("[Sub‑grad‑Adam] start")
        best_a, best_sc = adam_subgrad_optimize(
            best_a,
            sub_deadline,
            n,
            fft_len,
            rng,
            eps=1e-12
        )
        print(f"[Sub‑grad‑Adam] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (use any leftover time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 12.0)
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04
        )
        print(f"[hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result (evaluate_sequence rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    return final_seq
