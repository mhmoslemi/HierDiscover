def annealing_hill_climb(a, cur_score, fft_len, n, deadline, rng, max_iter=200000, delta_frac=0.04, init_temp=0.01, final_temp=1e-6):

   T_init = init_temp; T_final = final_temp

   total_time = deadline - time.time()

   iteration = 0

   while time.time() < deadline and iteration < max_iter:

        # Temperature linearly decays with elapsed fraction
        elapsed = (time.time() - (deadline - total_time)) / total_time
        # Or compute T = max(T_final, T_init * (1 - elapsed))

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
1. **Diversified initial pool** – generate many candidate shapes:
   * Classic windows (Hann, Blackman, Kaiser, DPSS with many bandwidths,
     Tukey, Gaussian, triangular, etc.).
   * Parametric families (Beta, power‑law, exponential).
   * Random Dirichlet draws.
   * Resampled previous best (if available).

2. **Screening** – compute the exact objective
   `2·n·max(conv(a,a))` (fast via FFT) for every candidate and keep the
   best few (default 10).

3. **Local smoothing optimisation** – run a continuation of L‑BFGS on a
   log‑sum‑exp smooth approximation of the max‑convolution.  The schedule
   starts with strong smoothing and gradually tightens.  Each run starts
   from a mixture of the current best and a fresh random draw.

4. **Exact sub‑gradient descent** – an exponentiated‑gradient (mirror‑descent)
   method minimises the true nonsmooth objective using the exact sub‑gradient
   of the maximal convolution entry.

5. **Annealed hill‑climb** – a stochastic greedy search that occasionally
   accepts non‑improving moves (simulated annealing) to escape local minima.
   Tiny mass‑transfer steps are tried repeatedly until the time budget expires.

All phases respect a total wall‑clock budget of 1000 s.  The best sequence
found is returned; the official ``evaluate_sequence`` rescales internally.

The implementation has no closures, uses only NumPy/SciPy, and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def beta_init(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Beta‑distribution shaped window on the simplex."""
    x = (np.arange(n) + 0.5) / n  # avoid exact 0/1
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta_param - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def tukey_init(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (cosine‑taper) window."""
    try:
        w = windows.tukey(n, alpha)
    except Exception:  # fallback to Hann if unavailable
        w = windows.hann(n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ. Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT (fast)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a:
        #   g_j = 2 * a_{k_max - j}  (if the index is in range)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/sqrt(t))
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def annealing_hill_climb(a: np.ndarray,
                         cur_score: float,
                         fft_len: int,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 200000,
                         delta_frac: float = 0.04,
                         init_temp: float = 1e-2,
                         final_temp: float = 1e-6):
    """
    Greedy hill‑climb with simulated‑annealing acceptance.
    Tiny mass‑transfer moves are proposed; worsening moves are accepted
    with probability exp(−Δ/ T) where the temperature T decays linearly.
    """
    best = a.copy()
    best_score = cur_score
    start = time.time()
    total_time = deadline - start
    it = 0
    while time.time() < deadline and it < max_iter:
        elapsed = (time.time() - start) / total_time
        T = max(final_temp, init_temp * (1.0 - elapsed))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        diff = sc - best_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-16)):
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    candidates = []

    # reuse previous construction if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # classic deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(kaiser_init(n, beta=8.0))

    # DPSS with a sweep of bandwidth parameters
    for NW in np.linspace(2.0, 6.0, 9):
        candidates.append(dpss_init(n, NW))

    # Kaiser with several betas
    for beta in [4.0, 6.0, 8.0, 10.0, 12.0]:
        candidates.append(kaiser_init(n, beta))

    # Tukey windows with different tapers
    for alpha in [0.0, 0.2, 0.5, 0.8, 1.0]:
        candidates.append(tukey_init(n, alpha))

    # Gaussian with several sigma factors
    for sf in [0.15, 0.20, 0.30, 0.40, 0.45]:
        candidates.append(gaussian_init(n, sigma_factor=sf))

    # Beta‑distribution shaped windows
    for _ in range(6):
        a_par = rng.uniform(0.2, 5.0)
        b_par = rng.uniform(0.2, 5.0)
        candidates.append(beta_init(n, a_par, b_par))

    # random Dirichlet‑type draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # Pick the best few by exact fast_score
    # --------------------------------------------------------------
    scores = [fast_score(c, fft_len, n) for c in candidates]
    idx_sorted = np.argsort(scores)
    top_k = 10
    best_idx = idx_sorted[0]
    best_a = candidates[best_idx].copy()
    best_sc = scores[best_idx]
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – intensified L‑BFGS (≈35 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2500  # a balance between depth and time

    while time.time() < phase1_end:
        # Choose starting point: mostly the current best, sometimes a random top‑k
        if rng.random() < 0.7:
            a_start = best_a.copy()
        else:
            a_start = candidates[rng.choice(idx_sorted[:top_k])].copy()

        # Mix with a fresh random vector to keep diversity
        rnd = random_initial(n, rng)
        mix_alpha = rng.random() * 0.5
        a_mix = (1.0 - mix_alpha) * a_start + mix_alpha * rnd
        a_mix /= a_mix.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_mix,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent (≈45 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 30.0   # keep a small cushion for final polish
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_sc = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – annealed hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_sc = annealing_hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=120000,
            delta_frac=0.04,
            init_temp=1e-2,
            final_temp=1e-7
        )
        print(f"[final hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
