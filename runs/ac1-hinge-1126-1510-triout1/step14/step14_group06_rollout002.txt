"""
Propose a low‑score step sequence.

The search is split into a few phases:

1. Build a diverse pool of symmetric probability vectors (spectral guess,
   classic windows, random Dirichlet draws, simple parametric families).
   The best vector according to the exact fast_score is kept.

2. Refine the best vector with a long L‑BFGS run on a smoothed max‑convolution
   (log‑sum‑exp).  The smoothing parameter λ is gradually decreased.  After
   each L‑BFGS call the vector is symmetrised and re‑normalised.

3. Perform a few random restarts of the above refinement.

4. Run Adam on the same smoothed objective for many iterations.

5. Apply a projected sub‑gradient descent that directly minimises the exact
   max‑convolution, using a back‑tracking line‑search.

6. Perform a long simulated‑annealing phase with mass‑transfer moves.

7. Finish with a greedy hill‑climb that only accepts strict improvements
   (mass is moved symmetrically).

All phases respect the overall time budget (default 1000 s).  The final
vector is returned as a list of floats; ``evaluate_sequence`` will be called
by the grader to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def normalize(a: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1)."""
    s = a.sum()
    if s <= 0:
        return np.full_like(a, 1.0 / a.size)
    return a / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce symmetry a[i] = a[n‑1‑i]."""
    return (a + a[::-1]) * 0.5


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                              lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-15
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a = np.clip(a, 0.0, None)
    return normalize(a)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet (all entries positive)."""
    x = rng.gamma(1.0, 1.0, size=n)
    return normalize(x)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return normalize(a)


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return normalize(np.clip(a, 0.0, None))


def kaiser_init(n: int, beta: float = 9.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    return normalize(a)


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(a)


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return normalize(np.clip(a, 0.0, None))


def power_decay_init(n: int, rng: np.random.Generator, exp_range=(0.1, 0.7)) -> np.ndarray:
    exp = rng.uniform(*exp_range)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    return normalize(a)


def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow
    log_a = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    a = np.exp(log_a)
    a = np.clip(a, 0.0, None)
    return normalize(a)

def lbfgs_refine(a_start: np.ndarray, lam_schedule, maxiter_per_lam,
                 deadline: float, fft_len: int, n: int) -> tuple:
    """Run L‑BFGS on the smoothed max‑convolution for a series of λ."""
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=smooth_objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = symmetrize(softmax(y))
        a = normalize(a)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray, deadline: float, n: int,
                  fft_len: int, lam_factor: float = 0.001,
                  lr: float = 0.08, max_iter: int = 40000) -> tuple:
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = smooth_objective_and_grad(y, n, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = symmetrize(softmax(y))
        a = normalize(a)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray, deadline: float,
                               n: int, fft_len: int,
                               step0: float = 0.30,
                               backtrack_factors=(1.0, 0.5, 0.25)) -> tuple:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and max index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of the max conv wrt a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        for fac in backtrack_factors:
            a_new = a + fac * alpha * d
            a_new = symmetrize(np.clip(a_new, 0.0, None))
            a_new = normalize(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # accept the non‑improving backtrack if any
            a = symmetrize(np.clip(a + alpha * d, 0.0, None))
            a = normalize(a)
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        n: int, fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 800000) -> tuple:
    """Simple Simulated Annealing with symmetric mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        # pick a source index in the left half (symmetry will copy to the right)
        i = rng.integers(0, n // 2)
        # choose a different target index also in the left half
        j = rng.integers(0, n // 2)
        if i == j:
            move += 1
            continue

        # amount to move (up to 30 % of source mass)
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            move += 1
            continue

        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # mirror the changes to the right half
        ri, rj = n - 1 - i, n - 1 - j
        a_cand[ri] -= delta
        a_cand[rj] += delta

        a_cand = np.clip(a_cand, 0.0, None)
        a_cand = normalize(a_cand)

        sc_cand = fast_score(a_cand, fft_len, n)

        if sc_cand < best_score or rng.random() < np.exp(-(sc_cand - best_score) / max(T, 1e-15)):
            a = a_cand
            if sc_cand < best_score:
                best_score = sc_cand
                best_a = a_cand.copy()
        move += 1
    return best_a, best_score


def hill_climb(a: np.ndarray, cur_score: float,
               fft_len: int, n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 400000,
               delta_frac: float = 0.08) -> tuple:
    """Greedy hill‑climb that only keeps strictly improving symmetric moves."""
    best = a.copy()
    best_score = cur_score
    it = 0
    half = n // 2

    while time.time() < deadline and it < max_iter:
        i = rng.integers(0, half)
        j = rng.integers(0, half)
        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue

        # perform symmetric transfer
        best[i] -= delta
        best[j] += delta
        ri, rj = n - 1 - i, n - 1 - j
        best[ri] -= delta
        best[rj] += delta

        # ensure feasibility
        best = np.clip(best, 0.0, None)
        best = normalize(best)

        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
            best[ri] += delta
            best[rj] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)  # legacy safety

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000                     # proven effective size
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diversified pool of symmetric candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) use a previous solution if present (up‑sample / re‑interpolate)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = symmetrize(a0)
            a0 = normalize(a0)
            candidates.append(a0)

    # 2) deterministic window families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(gaussian_init(n))
    candidates.append(arcsine_init(n))

    # 3) random Dirichlet draws (ensuring symmetry)
    for _ in range(10):
        a = random_dirichlet(n, rng)
        a = symmetrize(a)
        candidates.append(a)

    # 4) power‑law decays
    for _ in range(6):
        a = power_decay_init(n, rng)
        a = symmetrize(a)
        candidates.append(a)

    # 5) simple Beta shapes (single Beta, symmetric)
    def try_beta(p, q):
        return beta_shape(n, p, q)

    for _ in range(12):
        p = rng.uniform(0.05, 3.0)
        q = rng.uniform(0.05, 3.0)
        a = try_beta(p, q)
        a = symmetrize(a)
        candidates.append(a)

    # ------------------------------------------------------------------
    # Select the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 8000

    best_a, best_score = lbfgs_refine(best_a, lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len, n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – a few random restarts of the L‑BFGS refinement
    # ------------------------------------------------------------------
    restarts = 3
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet(n, rng)
        rand_vec = symmetrize(rand_vec)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = normalize(a_start)
        a_opt, sc_opt = lbfgs_refine(a_start, lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase1_end,
                                     fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 3 – Adam on smoothed objective (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase3_end,
                                           n, fft_len,
                                           lam_factor=0.001,
                                           lr=0.06,
                                           max_iter=80000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient with line search (≈10 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.68 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n, fft_len,
                                                        step0=0.25)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.86 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n, fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-8,
                                                 max_moves=900000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=600000,
                                        delta_frac=0.09)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
