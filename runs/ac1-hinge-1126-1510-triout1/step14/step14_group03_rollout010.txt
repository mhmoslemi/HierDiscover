"""
propose_candidate – 2‑stage convex‐smooth + adaptive Adam + annealed hill‑climb

The search runs for the full 1000 s budget and is split into three
time‑bounded phases.

Phase 1 – many short L‑BFGS runs on a log‑sum‑exp smooth
approximation of   max(conv(a,a)).  Each run starts from a mixture of
the current best vector and a fresh random Dirichlet draw; the λ‑
schedule slowly drives the smooth approximation to the true max.

Phase 2 – Adam optimiser on the exact non‑smooth objective.
The gradient of  max(conv)  w.r.t. a is given analytically
(g_i = 2·a_{k_max‑i} where k_max is the index of the maximal
convolution entry).  Adam is performed on the soft‑max logits so the
simplex constraints are satisfied automatically.  The best vector
found by Adam replaces the incumbent.

Phase 3 – simulated‑annealing hill‑climb.  Random “mass‑transfer”
moves (i→j) of varying magnitude are proposed; improving moves are
always accepted, worsening moves are accepted with a temperature‑
dependent probability.  This lets the search escape shallow local
minima and polish the final solution.

All phases respect a global wall‑clock deadline; the best sequence
found is returned.  The official ``evaluate_sequence`` rescales the
vector internally, so we keep the sum normalised to 1 throughout.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss


# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (kernel objective)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    Convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective and gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_max(a_start: np.ndarray,
             deadline: float,
             n: int,
             fft_len: int,
             init_alpha: float = 0.02):
    """
    Adam optimisation on the exact max‑convolution objective.
    Operates on soft‑max logits so the simplex constraint remains satisfied.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (FFT based)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a = np.zeros_like(a)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through softmax
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_alpha * (np.sqrt(1.0 - beta2 ** t) /
                             (1.0 - beta1 ** t))
        y -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep best
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def simulated_annealing_climb(a: np.ndarray,
                              cur_score: float,
                              fft_len: int,
                              n: int,
                              deadline: float,
                              rng: np.random.Generator,
                              max_iter: int = 120000):
    """
    Annealed hill‑climb.  Random mass‑transfer moves are accepted
    if they improve the score, otherwise with a temperature‑dependent
    probability.  The temperature decays linearly to 0.
    """
    best = a.copy()
    best_score = cur_score
    start = time.time()
    total_time = deadline - start
    it = 0
    while time.time() < deadline and it < max_iter:
        # linear cooling
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total_time)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # propose moving a fraction p of a[i] to a[j]
        p = rng.random() * (0.2 + 0.8 * temperature)   # larger steps early
        delta = best[i] * p
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        improve = sc < best_score
        if improve or (temperature > 0 and rng.random() < np.exp(-(sc - best_score) / (temperature * 0.01 + 1e-9))):
            best = cand
            best_score = sc
        it += 1
    return best, best_score

    # deterministic windows – they give the search a good start
    candidates.append(dpss_init(n, NW=3.5))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.22))

    # random draws – wide coverage of the simplex
    for _ in range(16):
        candidates.append(random_initial(n, rng))

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS (≈35 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15,
        0.1, 0.07, 0.05, 0.03, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # mix current best with a fresh random draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.5
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the exact max‑convolution (≈45 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = deadline - 5.0   # leave a small cushion for polishing
        print("[Adam] start exact Adam optimiser")
        best_a, best_score = adam_max(
            best_a,
            adam_deadline,
            n,
            fft_len,
            init_alpha=0.02
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – annealed hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Anneal] final annealed hill‑climb")
        best_a, best_score = simulated_annealing_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=200000
        )
        print(f"[Anneal] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
