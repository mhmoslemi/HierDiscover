def subgradient_descent(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline:
        t += 1
        # Compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_idx = np.argmax(conv)
        # compute gradient
        i_min = max(0, max_idx - (n-1))
        i_max = min(n-1, max_idx)
        # assign gradient
        g = np.zeros_like(a)
        idx_i = np.arange(i_min, i_max+1)
        idx_j = max_idx - idx_i
        g[idx_i] = 2 * a[idx_j]
        # step size schedule, e.g., step = 0.1 / sqrt(t)
        step = 0.5 / np.sqrt(t) # bigger start?
        # gradient descent step
        a_new = a - step * g
        # project onto simplex
        a_new = simplex_projection(a_new)
        # evaluate
        sc = fast_score(a_new, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        # optional occasional random perturb
        if t % 500 == 0:
            # add small jitter
            a = a + rng.normal(scale=1e-5, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)
    return best_a, best_sc

"""
Optimise a non‑negative step function (probability vector) to minimise
the evaluation metric 2·n·max(conv)/(sum²).  The optimiser consists of:

1.  A rich pool of deterministic / random initialisations (window functions,
    spectral guess, power‑law, edge‑heavy, DPSS with many half‑band values…).
2.  A multi‑stage L‑BFGS continuation on a smooth log‑sum‑exp approximation
    of the max‑convolution.  A small negative quadratic penalty on the
    end‑points encourages mass to concentrate at the edges (they create few
    overlaps and thus lower the max‑convolution).
3.  Adam optimisation with a top‑k focus that progressively shrinks to the
    true maximum.
4.  Adam optimisation on the smooth‑max with exponential λ‑decay.
5.  A final sub‑gradient descent on the exact max‑convolution (projected
    onto the simplex) followed by a greedy hill‑climb.

All phases respect the global 1000 s budget.  The best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (used for internal bookkeeping; equals the official
# score when the sum of the sequence is 1).
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    edge_weight: float = 0.0,
) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient.
    `lam_factor` multiplies the current max to obtain the smoothing λ.
    `edge_weight` adds a quadratic penalty on the two end points.
    Returns (objective, gradient) with respect to logits y
    (where a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries (weights)
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # edge quadratic penalty (if any)
    if edge_weight != 0.0:
        # d (edge_weight * (a0² + a_{n-1}²)) / dy = edge_weight * 2 * a_i * da_i/dy
        # back‑prop through softmax later
        pen = np.zeros_like(a)
        pen[0] = 2.0 * edge_weight * a[0]
        pen[-1] = 2.0 * edge_weight * a[-1]
        g_vec = g_vec + pen

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diversified random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "spike",
            "logistic",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "spike":
        a = np.full(n, 1.0 / n)
        idx = rng.integers(0, n)
        a[idx] = rng.uniform(2.0, 5.0) / n
    else:  # logistic
        scale = rng.uniform(0.05, 0.3) * n
        loc = rng.integers(0, n)
        x = np.arange(n) - loc
        a = 1.0 / (1.0 + np.exp(x / scale))
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 9.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, att: float = 80.0) -> np.ndarray:
    try:
        from scipy.signal import chebwin

        a = chebwin(n, at=att)
    except Exception:
        a = np.kaiser(n, 8.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, halfband: float = 0.22) -> np.ndarray:
    try:
        from scipy.signal.windows import dpss

        a = dpss(n, halfband, Kmax=1)[0]
    except Exception:
        a = np.kaiser(n, 8.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
    edge_weight: float = 0.0,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous result.
    Returns the best probability vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam, edge_weight),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    start_top: float = 0.12,
    end_top: float = 0.0008,
):
    """
    Adam optimiser that gradually focuses on the true maximum.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (full)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1) : (n - 1 + n)]  # derivative w.r.t a
        grad_a = 2.0 * n * g_a  # full objective gradient

        # softmax back‑prop
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lam_start: float = 5e-2,
    lam_end: float = 1e-7,
    edge_weight: float = 0.0,
):
    """
    Adam optimiser on the log‑sum‑exp smooth max with exponential λ decay.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        # exponential decay of λ
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(
            y,
            n,
            fft_len,
            fft_len,
            lam_factor=lam / (a @ a) if (a @ a) > 0 else lam,
            edge_weight=edge_weight,
        )

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def subgradient_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # compute full convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # gradient of the selected max entry
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        g = np.zeros_like(a)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = max_idx - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # step size schedule (roughly 0.5 / sqrt(t))
        step = 0.5 / np.sqrt(t)

        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        # occasional random jitter to escape plateaus
        if t % 800 == 0:
            a = a + rng.normal(scale=1e-5, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_sc


def hill_climb(
    a: np.ndarray,
    cur_sc: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 150000,
    delta_frac: float = 0.04,
):
    """
    Greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    candidates.append(hann_init(n))

    candidates.append(chebwin_init(n, att=80.0))

    # Pick the best by cheap fast_score
    best_a = candidates[0].copy()

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS continuation (≈45 % of total budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    # λ‑schedule, long and down to 1e‑7
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.1,
        0.05,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
        5e-7,
        2e-7,
        1e-7,
    ]
    maxiter_per_lam = 2600  # a little more than before
    # small negative edge penalty to encourage mass at the boundaries
    edge_pen = -0.015

    while time.time() < phase1_end:
        # perturb the current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n,
            edge_weight=edge_pen,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam with top‑k focus (≈20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.12,
            end_top=0.0005,
        )
        print(f"[Adam‑topk] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam on smooth‑max (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.15 * budget_s
        print("[Adam‑smooth] start")
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n,
            fft_len,
            rng,
            lam_start=5e-2,
            lam_end=1e-8,
            edge_weight=edge_pen,
        )
        print(f"[Adam‑smooth] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – Projected sub‑gradient descent (≈10 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.10 * budget_s
        print("[Sub‑gradient] start")
        best_a, best_sc = subgradient_descent(
            best_a, sub_deadline, n, fft_len, rng
        )
        print(f"[Sub‑gradient] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (use any leftover seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 12.0)
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04,
        )
        print(f"[hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result (evaluate_sequence rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
