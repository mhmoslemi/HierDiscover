    n = len(a)

"""
Improved step‑function optimisation.

The search proceeds in three stages:

1. **Diverse initialisation** – a pool of deterministic windows,
   spectral guesses and random Dirichlet draws is built; the best
   (lowest exact fast_score) is kept as the starting point.

2. **Global optimisation** – a short L‑BFGS phase minimises a
   log‑sum‑exp smoothed version of the maximum auto‑correlation.
   Afterwards an exact exponentiated‑gradient descent on the
   non‑smooth max‑convolution further reduces the bound.

3. **Long‑run incremental hill‑climb** – the remaining time budget is
   spent in a massive greedy/local‑search loop that updates the full
   self‑convolution *incrementally* (O(n) per move) instead of via FFT.
   Moves consist of transferring a tiny amount of mass from one
   coordinate to another; a simple simulated‑annealing acceptance
   rule allows occasional uphill steps, helping escape shallow local
   minima.  The current best vector and its exact score are reported
   continuously.  The algorithm respects the 1000 s wall‑clock limit
   and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact fast score (used for bookkeeping)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)            # logits for softmax
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[EG‑max] iter {t}, best fast_score={best_sc:.8f}")
    return best_a, best_sc


def compute_full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full self‑convolution of a (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return conv


def hill_climb_incremental(a: np.ndarray,
                           conv: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           init_temp: float = 0.02,
                           min_temp: float = 1e-4):
    """
    Massive O(n) hill‑climb with simulated‑annealing acceptance.
    Moves transfer a small amount of mass from i to j.
    The convolution is updated incrementally, allowing millions of trials.
    """
    L = len(conv)  # 2·n‑1
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = best_conv.max()
    best_score = 2.0 * n * best_max
    cur_a = a.copy()
    cur_conv = conv.copy()
    iter_cnt = 0

    # temperature schedule: exponential decay
    temp = init_temp

    while time.time() < deadline:
        iter_cnt += 1

        # pick a source index with non‑zero mass
        i = rng.integers(0, n)
        if cur_a[i] <= 0.0:
            continue

        # step size fraction slowly shrinks
        frac = max(0.001, 0.04 * (0.999 ** (iter_cnt // 5000)))

        delta = cur_a[i] * rng.random() * frac
        if delta <= 0.0:
            continue

        # pick a distinct destination
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # build a candidate convolution incrementally
        cand_conv = cur_conv.copy()

        # contributions from the linear part: 2*delta*(a_shift_j - a_shift_i)
        cand_conv[i:i + n] -= 2.0 * delta * cur_a
        cand_conv[j:j + n] += 2.0 * delta * cur_a

        # quadratic delta² terms
        d2 = delta * delta
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        if idx_2i < L:
            cand_conv[idx_2i] += d2
        if idx_2j < L:
            cand_conv[idx_2j] += d2
        if idx_ij < L:
            cand_conv[idx_ij] -= 2.0 * d2

        # new objective
        new_max = cand_conv.max()
        new_score = 2.0 * n * new_max

        # acceptance rule
        accept = False
        if new_score < best_score - 1e-12:
            accept = True
        else:
            # annealed uphill acceptance
            if rng.random() < np.exp((best_score - new_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            # commit move
            cur_conv = cand_conv
            cur_a[i] -= delta
            cur_a[j] += delta

            # keep the invariant sum(cur_a) == 1 (mass is transferred)
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                best_conv = cur_conv.copy()
                best_max = new_max
                # occasional progress report
                if iter_cnt % 50000 == 0:
                    print(f"[hill‑inc] iter {iter_cnt}, best score={best_score:.8f}")

        # temperature decay
        if iter_cnt % 2000 == 0:
            temp = max(min_temp, temp * 0.995)

        # every few hundred thousand iterations recompute convolution from scratch
        if iter_cnt % 200000 == 0:
            cur_conv = compute_full_conv(cur_a, next_pow2(2 * n - 1))
            cur_max = cur_conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                best_conv = cur_conv.copy()
                print(f"[hill‑inc] recompute at iter {iter_cnt}, best score={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    evaluation score.  The function respects a total wall‑clock budget
    of `budget_s` seconds and returns the best sequence found before the
    deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool (same as the previous approach)
    # --------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # resample to current length preserving shape
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # random draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – short L‑BFGS (≈20 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new overall best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0   # keep a small cushion for phase 3
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – long incremental hill‑climb (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[hill‑inc] start massive incremental hill‑climb")
        # initialise full convolution for the incremental updater
        cur_conv = compute_full_conv(best_a, fft_len)
        best_a, best_score = hill_climb_incremental(
            best_a,
            cur_conv,
            n,
            deadline,
            rng,
            init_temp=0.02,
            min_temp=1e-4
        )
        print(f"[hill‑inc] final best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official evaluation
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
