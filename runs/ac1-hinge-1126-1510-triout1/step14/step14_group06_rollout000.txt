def l2_optimize(a_start, deadline, n, fft_len, init_lr=0.1):
    a = a_start.copy()
    target = 1.0 / (2*n - 1)  # constant conv value.
    lr = init_lr
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        diff = conv - target
        # gradient: 2 * (reverse(a) conv diff) slice.
        a_rev = a[::-1]
        # convolution via FFT
        # diff length = 2n-1
        # we need conv_rev = conv of a_rev and diff
        # length = len(a_rev) + len(diff) - 1 = n + 2n-1 -1 = 3n-2
        Lg = next_pow2(len(diff) + n - 1)  # for efficient FFT
        # We'll use fft_len maybe bigger; but we can compute using same L used for original: L = next_pow2(2*n-1). For convolution length > L maybe need bigger; but we can zero-pad and compute long conv using next_pow2(3*n-2)
        L2 = max(fft_len, next_pow2(3*n-2))
        # compute via rfft size L2
        fa_rev = np.fft.rfft(a_rev, L2)
        fdiff = np.fft.rfft(diff, L2)
        conv_rev = np.fft.irfft(fa_rev * fdiff, L2)
        # slice central portion indices n-1 to n-1+n-1 inclusive (n entries)
        grad = 2 * conv_rev[(n-1):(n-1+n)]
        # simple step size schedule: lr = init_lr / sqrt(it) (or constant)
        step = lr / np.sqrt(it)
        a_new = a - step * grad
        # projection onto simplex
        a_new = simplex_projection(a_new)
        # ensure no tiny negative
        # compute score
        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

"""
Search for a low‑score non‑negative step sequence.

Algorithm
---------
1. Build a diverse pool of candidate distributions (deterministic windows,
   random Dirichlet draws, Beta families, previous solution if any).
2. Select the best pool member using the exact fast objective
   (2·n·max(self‑convolution) for a probability vector).
3. Refine the selected vector with a custom L2‑error gradient descent:
   we minimise the squared deviation of the self‑convolution from the
   uniform target 1/(2·n‑1).  The gradient is obtained analytically via FFT,
   and each update is projected onto the probability simplex.
4. Finish with a greedy hill‑climb that only accepts strictly improving
   pairwise mass‑transfer moves.
5. Return the best sequence (as a list of floats) before the time budget
   expires.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (assumes sum(a)=1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral guess via square‑root of a uniform spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the simplex (Dirichlet(1,…,1))."""
    a = rng.dirichlet(np.ones(n))
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_mixture_shape(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    a1 = beta_shape(n, p1, p1)
    a2 = beta_shape(n, p2, p2)
    a = w * a1 + (1.0 - w) * a2
    # each component already sums to 1 → mixture also sums to 1
    return a


# ----------------------------------------------------------------------
# Gradient descent on L2 error of the self‑convolution
# ----------------------------------------------------------------------


def l2_descent(a_start: np.ndarray,
               deadline: float,
               n: int,
               fft_len: int,
               target: float,
               init_step: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Minimise ½·∑(conv - target)² using an adaptive projected gradient descent.
    Returns the best vector encountered and its fast_score.
    """
    # Length needed for the convolution of a_rev (len n) with diff (len 2n‑1)
    Lg = next_pow2(3 * n - 2)

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    step = init_step
    it = 0
    while time.time() < deadline:
        # self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # deviation from uniform target
        diff = conv - target

        # gradient = 2 * (a_rev * diff) restricted to n entries
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, Lg)
        fdiff = np.fft.rfft(diff, Lg)
        conv_rev = np.fft.irfft(fa_rev * fdiff, Lg)
        grad = 2.0 * conv_rev[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        improved = False
        cur_step = step
        for _ in range(10):
            a_new = a - cur_step * grad
            a_new = simplex_projection(a_new)
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                # slightly increase step for next iteration
                step = cur_step * 1.1
                break
            cur_step *= 0.5
        if not improved:
            # no improvement → shrink step size
            step *= 0.5
        it += 1
        if step < 1e-12:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.10) -> tuple[np.ndarray, float]:
    """Pairwise mass‑transfer hill‑climb that only accepts improvements."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    target_conv = 1.0 / (2 * n - 1)          # uniform target for the convolution

        best_a, best_score = l2_descent(
            best_a,
            phase1_end,
            n,
            fft_len,
            target=target_conv,
            init_step=1.0)
        print(f"[L2‑descent] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Final greedy hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=600000,
            delta_frac=0.09)
        print(f"[hill‑climb] final fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the host environment
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
