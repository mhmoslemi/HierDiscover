"""
Hybrid optimizer for the low‑autocorrelation step‑function challenge.

The algorithm consists of three stages:

1. **Small‑scale LP boost** –  The problem
   max ∑ g_j  s.t.  f⋆g ≤ M , g ≥ 0
   (where M = max(f⋆f)) is a linear program.
   Solving it on a modest size (n≈1024) yields a vector g whose
   ℓ₁‑norm can be far larger than 1 while still respecting the
   convolution bound.  By mixing the current solution f with the
   normalised g we obtain a strict decrease of the objective.
   Several LP iterations (t ≈ 0.2) are performed until no further
   improvement is seen.

2. **Upsampling** –  The improved small‑scale solution is interpolated
   to the target size (n=8192).  This provides a powerful starting
   point for the next stage.

3. **Continuous refinement** –  A cascade of gradient‑based methods
   (L‑BFGS on a smooth‑max surrogate, Adam with a top‑k focus,
   Adam on the log‑sum‑exp smooth max, a short simulated annealing,
   and a final greedy hill‑climb) is run on the full‑size vector.
   All phases respect the global time budget (≈1000 s).

The code uses only NumPy, SciPy and CVXPY (ECOS solver).  If CVXPY is
unavailable the optimizer falls back to the pure gradient‑based
pipeline.
"""

import time
import numpy as np
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_convolution(a: np.ndarray) -> float:
    """Return max(f⋆f) for a (assumed non‑negative, sum=1)."""
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) objective and gradient – used by L‑BFGS & Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam_factor: float = 0.30,
                       edge_weight: float = 0.0) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv) and its gradient w.r.t. logits y.
    a = softmax(y) is the probability vector.
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv entries
    w = exp_shift / Z                        # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # optional edge penalty
    if edge_weight > 0.0:
        pen = np.zeros_like(a)
        pen[0] = 2.0 * edge_weight * a[0]
        pen[-1] = 2.0 * edge_weight * a[-1]
        g_vec += pen

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous result.
    Returns best probability vector and its fast_score.
    """
    from scipy.optimize import minimize

    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam, 0.0),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    """Adam optimiser that gradually focuses on the true maximum."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len)

    start_time = time.time()
    total_allowed = deadline - start_time

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑fraction
        elapsed = time.time() - start_time
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 5e-2,
                         lam_end: float = 1e-7):
    """Adam optimiser on the log‑sum‑exp smooth max with exponential λ decay."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len)

    start_time = time.time()
    total_allowed = deadline - start_time

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(y, n, fft_len, lam_factor=lam, edge_weight=0.0)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """Tiny‑mass‑transfer SA on the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """Strictly improving greedy hill‑climb."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc

def lp_step(f: np.ndarray,
            M: float,
            mixing: float = 0.2,
            solver: str = 'ECOS'):
    """
    One LP step:
    maximise Σ g   subject to  f⋆g ≤ M , g ≥ 0.
    Returns the mixed vector (1‑mix)·f + mix·(g/Σg)  (still on the simplex)
    or None if the LP fails.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = f.shape[0]

    # Build the Toeplitz convolution matrix A where (A g)_k = Σ_i f_i g_{k-i}
    col = np.r_[f, np.zeros(n - 1)]
    row = np.r_[f[0], np.zeros(2 * n - 2)]
    A = toeplitz(col, row)                     # shape (2n‑1, n)

    g = cp.Variable(n)
    constraints = [A @ g <= M + 1e-12, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=solver, warm_start=True, max_iters=2000, verbose=False)
    except Exception:  # pragma: no cover
        return None

    if g.value is None:
        return None

    g_opt = np.maximum(0.0, np.asarray(g.value, dtype=np.float64))
    sum_g = g_opt.sum()
    if sum_g <= 1e-12:
        return None

    # Normalise g to a probability vector
    g_norm = g_opt / sum_g
    # Convex combination with a modest mixing factor
    f_new = (1.0 - mixing) * f + mixing * g_norm
    return f_new


def lp_boost(initial_f: np.ndarray,
             max_iters: int,
             time_limit: float,
             mixing: float = 0.2):
    """
    Repeated LP boost on a small problem.
    Returns the best probability vector found.
    """
    f = initial_f.copy()
    n = f.shape[0]
    fft_len = next_pow2(2 * n - 1)
    best_f = f.copy()
    best_sc = fast_score(f, fft_len)

    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        M = max_convolution(f)
        f_new = lp_step(f, M, mixing=mixing)
        if f_new is None:
            break
        f = f_new
        sc = fast_score(f, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[LP‑boost] iter {it+1}, new best small score={best_sc:.9f}")
        # stop if improvement stalls
        if it > 0 and best_sc / fast_score(best_f, fft_len) > 0.9999:
            break
    return best_f

def propose_candidate(seed: int = 42,

    fft_len_small = next_pow2(2 * n_small - 1)

    # Assemble initial candidates (probability vectors)
    # ------------------------------------------------------------------
    candidates_small = []

    # 1. Uniform baseline (always feasible)
    candidates_small.append(np.full(n_small, 1.0 / n_small))

    # 2. Remember the previous run, if any (downsized)
    if 'height_sequence_1' in globals():

            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates_small.append(a0)

    # 3. Random starts
    for _ in range(5):
        a = rng.random(n_small)
        a /= a.sum()
        candidates_small.append(a)

    # ------------------------------------------------------------------
    # Pick the best small candidate by cheap fast_score
    # ------------------------------------------------------------------
    best_small = candidates_small[0].copy()
    best_small_sc = fast_score(best_small, fft_len_small)
    for cand in candidates_small[1:]:
        sc = fast_score(cand, fft_len_small)
        if sc < best_small_sc:
            best_small_sc = sc
            best_small = cand.copy()
    print(f"[init‑small] best small fast_score={best_small_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP boost on the small problem (≈30 % of total budget)
    # ------------------------------------------------------------------
    start_time = time.time()

    best_small = lp_boost(best_small, max_iters=max_lp_iters,
                          time_limit=lp_deadline - time.time(),
                          mixing=0.18)
    best_small_sc = fast_score(best_small, fft_len_small)
    print(f"[LP‑final] best small fast_score={best_small_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Upsample to full size and initialise the continuous stage
    # ------------------------------------------------------------------
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, best_small)
    a_large = np.clip(a_large, 0.0, None)
    a_large /= a_large.sum()

    # ------------------------------------------------------------------
    # Phase 3 – Continuous refinement on the full‑size vector
    # ------------------------------------------------------------------
    # Respect the remaining time
    start_full = time.time()
    remaining = deadline - start_full

    # 1. L‑BFGS continuation (≈30 % of remaining time)
    phase1_end = start_full + 0.30 * remaining
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # perturb current best a bit to escape possible stagnation
        perturb = rng.random(a_large.shape) * 0.02
        a_start = a_large * (1.0 - perturb) + perturb / a_large.shape[0]
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_large,
            n_large
        )
        if sc_opt < fast_score(a_large, fft_len_large):
            a_large = a_opt
            print(f"[L‑BFGS‑large] improved fast_score={sc_opt:.9f}")

    # 2. Adam‑topk (≈20 % of remaining time)
    if time.time() < deadline:
        print("[Adam‑topk‑large] start")
        a_large, _ = adam_topk_optimize(
            a_large,
            deadline,
            n_large,
            fft_len_large,
            rng,
            start_top=0.12,
            end_top=0.0008
        )
        print(f"[Adam‑topk‑large] best fast_score={fast_score(a_large, fft_len_large):.9f}")

    # 3. Adam‑smooth (≈15 % of remaining time)
    if time.time() < deadline:
        mid = time.time() + 0.15 * budget_s
        print("[Adam‑smooth‑large] start")
        a_large, _ = adam_smooth_optimize(
            a_large,
            mid,
            n_large,
            fft_len_large,
            rng,
            lam_start=5e-2,
            lam_end=1e-7
        )
        print(f"[Adam‑smooth‑large] best fast_score={fast_score(a_large, fft_len_large):.9f}")

    # 4. Simulated annealing (≈15 % of remaining time)
    if time.time() < deadline:
        sa_deadline = time.time() + min(0.15 * budget_s,
                                        deadline - time.time())
        a_large, _ = simulated_annealing(
            a_large,
            sa_deadline,
            n_large,
            fft_len_large,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=350000
        )
        print(f"[SA‑large] best fast_score={fast_score(a_large, fft_len_large):.9f}")

    # 5. Final greedy hill‑climb (use any leftover seconds)
    if time.time() < deadline:
        hill_deadline = time.time() + min(12.0, deadline - time.time())
        cur_sc = fast_score(a_large, fft_len_large)
        a_large, cur_sc = hill_climb(
            a_large,
            cur_sc,
            fft_len_large,
            n_large,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04
        )
        print(f"[hill‑climb‑large] final fast_score={cur_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the final sequence (evaluate_sequence rescales internally)
    # ------------------------------------------------------------------
    final_seq = a_large.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
