"""
Propose a low‑score step function.

Improvements over the reference implementation
------------------------------------------------
* Wider hyper‑parameter sweep (Gaussian σ, Kaiser β, Beta α/β) to get a
  much better starting point.
* Longer, finer L‑BFGS schedule (λ down to 5e‑7) with more iterations
  per λ.
* Two passes of the L‑BFGS/Adam cycle – after the first pass we
  re‑initialise from the current best and run the whole continuation
  again (helps to escape shallow basins).
* LP‑refinement with a larger active set (max_active=60) and a smaller
  mixing step that decays more slowly.
* A final “global random search” that keeps drawing Dirichlet vectors
  for the remaining time and keeps any improvement.
* All phases share a single global deadline (default 1000 s) and
  respect a small safety margin.

The algorithm works on the simplex (∑ a_i = 1, a_i ≥ 0).  The scoring
function used internally is
    fast_score(a) = 2·n·max( a * a )           (convolution a·a )
which coincides with the official `evaluate_sequence` when the
vector is normalised to have L¹‑norm 1.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a point on the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective (used for ranking – a is on the simplex)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max( a * a )   (convolution performed via FFT)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max + gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y   (y are logits, a = softmax(y))
    """
    a = softmax(y)

    # convolution a*a (linear)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                # ≤ 0
    exp_shift = np.exp(shifted / lam)         # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                          # (2n‑1,)

    # back‑prop through convolution to obtain ∂max/∂a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]      # gradient wrt a

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Square‑root of a flat spectrum – a classic spectral start."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    pad = np.zeros(L)
    pad[:target_len] = u
    S = np.fft.fft(pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def tukey_init(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey window – cosine taper with adjustable flat region."""
    if alpha <= 0:
        return np.ones(n) / n
    if alpha >= 1:
        return hann_init(n)
    t = np.arange(n) / (n - 1)
    w = np.ones_like(t)
    idx1 = t < alpha / 2
    idx2 = t > 1 - alpha / 2
    w[idx1] = 0.5 * (1 + np.cos(2 * np.pi / alpha * (t[idx1] - alpha / 2)))
    w[idx2] = 0.5 * (1 + np.cos(2 * np.pi / alpha * (t[idx2] - 1 + alpha / 2)))
    w /= w.sum()
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, warm‑starting each stage.
    Returns the best vector found (still on the simplex) and its fast_score.
    """
    y = np.log(a_start + 1e-12)                # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:          # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution (logits → softmax).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # exact max‑convolution and mask of the active peak(s)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient of the exact max (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # chain rule through softmax
        mean_g = np.dot(a, g_a)
        grad_y = a * (g_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 2000 == 0:
            # tiny random jitter to escape plateaus
            y += rng.normal(scale=1e-4, size=y.shape)

        # evaluate
        sc = fast_score(softmax(y), fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = softmax(y).copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              max_active: int = 60,
              init_step: float = 0.15):
    """
    Repeated tiny LP: maximise Σb under the active constraints
    f * b ≤ M (M = current peak).  The solution b is mixed with f
    using a shrinking convex step.
    """
    f = a_start.copy()
    best_f = f.copy()
    best_sc = fast_score(best_f, fft_len, n)
    t = init_step
    iteration = 0

    while time.time() < deadline:
        # current autocorrelation and its peak
        fa = np.fft.rfft(f, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # top‑k active constraints (indices where conv is near the peak)
        if max_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            active_idx = np.argpartition(-conv, max_active - 1)[:max_active]

        # build sparse matrix A for the active constraints
        rows, cols, data = [], [], []
        f_rev = f[::-1]
        for r, k in enumerate(active_idx):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                b_idx = k - j
                rows.append(r)
                cols.append(b_idx)
                data.append(f_rev[j])
        A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M)

        # maximise Σ b  → minimise -Σ b
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print("[LP] linprog failed – exiting LP phase")
            break

        b_opt = res.x
        if b_opt.sum() <= 0:
            break
        g = b_opt / b_opt.sum()          # normalise to the simplex

        # convex mixing
        f = (1.0 - t) * f + t * g

        sc = fast_score(f, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[LP] iter={iteration}, new best fast_score={best_sc:.8f}")

        # decay mixing step very slowly – this helps the refinement settle
        t = max(0.02, t * 0.998)
        iteration += 1

    return best_f, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Pairwise mass‑transfer SA (cheap, works well for fine‑tuning).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


def global_random_search(best_a: np.ndarray,
                         best_sc: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         batch: int = 200):
    """
    Draw random Dirichlet vectors until time runs out, keep any improvement.
    """
    while time.time() < deadline:
        # generate a batch to amortise overhead of the RNG
        alphas = rng.uniform(0.2, 2.0, size=(batch, n))
        draws = rng.gamma(alphas)
        draws = draws / draws.sum(axis=1, keepdims=True)
        scores = np.apply_along_axis(fast_score, 1, draws, fft_len, n)
        idx = scores.argmin()
        if scores[idx] < best_sc:
            best_sc = scores[idx]
            best_a = draws[idx].copy()
            print(f"[Random] new best fast_score={best_sc:.8f}")
    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two FFT length works best
    # --------------------------------------------------------------
    n = 4096                         # keep the size used in the reference run
    fft_len = next_pow2(2 * n - 1)   # 8192 for n=4096

    # --------------------------------------------------------------
    # Build a diverse pool of candidates (including a fine parameter sweep)
    # --------------------------------------------------------------
    candidates = []

    # 1. May reuse a previous solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # 2. Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(tukey_init(n, alpha=0.4))

    # 3. Parameter sweeps (coarse – cheap)
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))
    for beta in np.linspace(4.0, 12.0, 7):
        candidates.append(kaiser_init(n, beta=beta))
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 2.0, 5):
            candidates.append(beta_init(n, alpha=a, beta=b))

    # 4. Randomised draws
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # Keep the best candidate (quick ranking using fast_score)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score after sweep={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time‑budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement (two independent passes)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # 30 % of total time
    lam_schedule = np.geomspace(0.4, 5e-7, num=70)   # finer λ schedule
    maxiter_per_lam = 2000

    # first pass
    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global‑1] new best fast_score={best_sc:.8f}")

    # second pass – re‑start from the current best to go deeper
    while time.time() < phase1_end:
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global‑2] new best fast_score={best_sc:.8f}")
        else:
            break   # small improvement stalls – exit early

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a, deadline, n, fft_len, rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑programming refinement (more aggressive)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(45.0, deadline - time.time())
        print("[LP] start aggressive refinement")
        best_a, best_sc = lp_refine(best_a,
                                    lp_deadline,
                                    n,
                                    fft_len,
                                    max_active=60,
                                    init_step=0.15)
        print(f"[LP] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing fine‑tuning (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(30.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=200000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + min(20.0, deadline - time.time())
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.03)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Global random search for any late improvement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Random‑search] start")
        best_a, best_sc = global_random_search(best_a,
                                                best_sc,
                                                deadline,
                                                n,
                                                fft_len,
                                                rng,
                                                batch=300)
        print(f"[Random‑search] final best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
