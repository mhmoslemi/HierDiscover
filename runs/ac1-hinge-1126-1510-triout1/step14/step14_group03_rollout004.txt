Propose a low‑score non‑negative step‑function.

The search combines several cheap diversification tricks and three
optimization phases that respect a strict 1000 s wall‑clock budget.

1. **Diversified initial pool** – deterministic windows (Hann, Blackman,
   Kaiser with several β, Gaussian with several σ, DPSS with several
   NW) plus many random Dirichlet draws, plus the previous best
   solution (if any) interpolated to the working length.  The best few
   candidates (by the exact fast score) are kept for later refinement.

2. **Smooth quasi‑Newton (L‑BFGS) stage** – we minimise a
   log‑sum‑exp smooth approximation of `max(conv(a,a))`.  A schedule of
   decreasing smoothing parameters (λ) drives the iterate toward the
   true objective.  Restarts are performed by mixing the current best
   with fresh random draws.

3. **Exact sub‑gradient descent** – using the analytical gradient of
   the active convolution entry (`∇max(conv)=2⋅a shifted`).  An
   exponentiated‑gradient update (mirror descent on the simplex) with
   a √t‑decay step size drives the solution into a sharp local optimum.

4. **Mass‑transfer simulated‑annealing hill‑climb** – a bespoke
   O(n) incremental update of the self‑convolution enables millions of
   cheap random moves.  The move consists of transferring a small
   amount of mass from one index to another; the new convolution is
   obtained by a simple vectorised update rather than a full FFT.  A
   temperature schedule permits occasional uphill steps to escape local
   minima.  The best vector encountered is returned.

All phases stop as soon as the global deadline (`budget_s` seconds) is
reached.  The final vector is normalised (sum = 1), converted to a
plain Python list and fed to `evaluate_sequence` for the official
score.

The code uses only `numpy` and `scipy.optimize.minimize`.  No file or
network I/O is performed.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact objective (shape factor) – used for all bookkeeping
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (∑a = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv  # also return the full conv


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (∂ smooth_max / ∂ conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: ∂ conv / ∂ a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (various deterministic windows)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    # Random concentration parameter in a log‑uniform range
    concentration = rng.uniform(0.1, 5.0)
    x = rng.gamma(concentration, 1.0, size=n)
    a = x / x.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    return a / a.sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered
    (and its fast_score).
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc, _ = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc, _ = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc, _ = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # exact convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc, _ = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb_incremental(a: np.ndarray,
                           conv: np.ndarray,
                           cur_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_iter: int = 300_000,
                           delta_frac: float = 0.04,
                           temp_start: float = 1e-3,
                           temp_end: float = 1e-6):
    """
    Simulated‑annealing hill‑climb with O(n) incremental convolution updates.
    Only moves that improve (or are accepted probabilistically) are kept.
    Returns the best vector found and its score.
    """
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = cur_score

    cur_a = a
    cur_conv = conv
    cur_sc = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        # linear temperature schedule
        temp = temp_start + (temp_end - temp_start) * (it / max_iter)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        ai = cur_a[i]
        if ai <= 0.0:
            it += 1
            continue
        delta = ai * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # build candidate by moving delta from i to j
        # --------------------------------------------------------------
        cand_a = cur_a.copy()
        cand_a[i] -= delta
        cand_a[j] += delta

        # incremental update of the convolution (copy + O(n) updates)
        cand_conv = cur_conv.copy()
        # linear contributions (2 * Δ * a_old)
        cand_conv[i:i + n] += -2.0 * delta * cur_a
        cand_conv[j:j + n] +=  2.0 * delta * cur_a
        # quadratic contributions
        cand_conv[2 * i] += delta * delta          # (−δ)²
        cand_conv[2 * j] += delta * delta          # ( +δ)²
        cand_conv[i + j] -= 2.0 * delta * delta    # 2·(−δ)·(+δ)

        cand_sc = 2.0 * n * cand_conv.max()

        # decide acceptance
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # Metropolis rule
            prob = np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = cand_a
            cur_conv = cand_conv
            cur_sc = cand_sc
            if cand_sc < best_score:
                best_score = cand_sc
                best_a = cand_a.copy()
                best_conv = cand_conv.copy()
        it += 1

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer approximation of the
    # continuous optimum; 8192 is still cheap for FFT‑based steps.
    # ------------------------------------------------------------------
    n = 8192
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1. Build a diverse initial pool
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous construction if available (interpolate to new length)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic windows (many parameter choices)
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))

    # Kaiser with several betas
    for beta in [4.0, 6.0, 8.0, 10.0, 12.0]:
        candidates.append(kaiser_init(n, beta=beta))

    # Gaussian with several sigma factors
    for sigma_factor in [0.20, 0.25, 0.30, 0.35]:
        candidates.append(gaussian_init(n, sigma_factor=sigma_factor))

    # DPSS with a few NW values
    for NW in [2.5, 3.0, 3.5, 4.0]:
        candidates.append(dpss_init(n, NW=NW))

    # random Dirichlet draws (different concentrations)
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # keep the top few (by exact fast_score)
    best_a = None
    best_score = np.inf
    best_conv = None
    for cand in candidates:
        sc, conv = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            best_conv = conv.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # mix current best with a fresh random draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            _, best_conv = fast_score(best_a, fft_len, n)
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient (≈45 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 10.0   # keep buffer for final polish
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.6
        )
        _, best_conv = fast_score(best_a, fft_len, n)
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (≈25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print("[SA‑hill] start simulated‑annealing hill‑climb")
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_conv,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=500_000,
            delta_frac=0.05,
            temp_start=5e-4,
            temp_end=1e-6
        )
        print(f"[SA‑hill] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
