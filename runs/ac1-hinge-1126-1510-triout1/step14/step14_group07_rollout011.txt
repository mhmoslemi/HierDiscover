"""
Propose a step sequence with a low evaluation score.

Algorithm Overview
------------------
1. **Initial pool** – generate a diverse set of probability vectors:
   classical windows (uniform, Hann, Blackman, Kaiser, Gaussian,
   arcsine, Tukey), DPSS windows, random Dirichlet samples and
   convex blends with the uniform vector.
2. **Select best starter** – evaluate the exact fast score
   (2·n·max(conv) with ∑a=1) and keep the best.
3. **Phase 1 – L‑BFGS on a smoothed max** – a log‑sum‑exp surrogate
   of the max‑convolution is minimized using L‑BFGS with a
   decreasing smoothing parameter λ.  Several random restarts
   explore different basins.
4. **Phase 2 – L2‐flattening** – minimise the squared deviation of
   the self‑convolution from the uniform distribution.
   The gradient is obtained by a single convolution in the
   frequency domain; projected gradient descent on the simplex
   drives the convolution towards flatness, which strongly reduces
   the maximal entry.
5. **Phase 3 – Sub‑gradient descent on the exact max** – a
   projected sub‑gradient method (exact max entry) fine‑tunes the
   solution.
6. **Phase 4 – Greedy hill‑climb** – strictly improving mass‑transfer
   moves finish the polishing.
7. **Final evaluation** – the official ``evaluate_sequence`` routine
   is called and the best list of floats is returned.

All phases respect a global time budget (default 1000 s) and
periodically print progress and current scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def tukey_init(n: int, alpha: float = 0.5) -> np.ndarray:
    # implemented directly to avoid additional SciPy import
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.ones_like(i)
    w = (1 - alpha) / 2
    left = i < w
    right = i > 1 - w
    a[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / alpha - 1)))
    a[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / alpha - 2 / alpha + 1)))
    a[left | right] = a[left | right]
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 2.5) -> np.ndarray:
    try:
        from scipy.signal import windows
        a = windows.dpss(n, NW, Kmax=1, sym=True)
    except Exception:          # pragma: no cover
        # fallback to Hann if DPSS not available
        a = hann_init(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def flatten_l2_optimize(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int):
    """
    Projected gradient descent minimising the L2 deviation of the
    self‑convolution from the uniform distribution.
    """
    avg = 1.0 / (2 * n - 1)          # desired flat value for each conv entry
    a = a_start.copy()
    best_a = a.copy()

    # initial error
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    err = np.sum((conv - avg) ** 2)
    best_err = err

    # size for the auxiliary convolution (a_rev * d_rev)
    L = next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline:
        it += 1
        # gradient of error w.r.t. a : 2 * (a * (conv-avg)_rev) reversed
        d = conv - avg                     # length 2n-1
        a_rev = a[::-1]
        d_rev = d[::-1]

        Fa_rev = np.fft.rfft(a_rev, L)
        Fd_rev = np.fft.rfft(d_rev, L)
        grad_full = np.fft.irfft(Fa_rev * Fd_rev, L)[: 3 * n - 2]
        # slice n‑1 … 2n‑2 (inclusive) gives the values for j = n‑1…0
        g_half = grad_full[n - 1:2 * n - 1]
        g = 2.0 * g_half[::-1]            # gradient vector (size n)

        # step size – diminishing with sqrt(it)
        alpha0 = 0.15
        alpha = alpha0 / np.sqrt(it)

        # back‑tracking line search (simple)
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_new = simplex_projection(a - factor * alpha * g)
            fa_new = np.fft.rfft(a_new, fft_len)
            conv_new = np.fft.irfft(fa_new * fa_new, fft_len)[: 2 * n - 1]
            err_new = np.sum((conv_new - avg) ** 2)
            if err_new < err:
                a = a_new
                conv = conv_new
                err = err_new
                accepted = True
                if err_new < best_err:
                    best_err = err_new
                    best_a = a_new.copy()
                break
        # if no improvement, still keep the projection (helps escape flats)
        if not accepted:
            a = simplex_projection(a - alpha * g)   # no guarantee of improvement

    best_score = fast_score(best_a, fft_len, n)
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = sub
        if idx % 2 == 0:
            i_mid = idx // 2
            if 0 <= i_mid < n:
                g[i_mid] += a[i_mid]   # extra term for i==j
        g *= 2.0

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        # simple back‑tracking
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # still move but keep feasibility
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 500000,
               delta_frac: float = 0.10):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
                candidates.append(a0)

    # --------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈20 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 2000

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # Additional random restarts while time permits (≈10 restarts)
    restarts = 10
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase1_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – L2 flattening (≈45 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.65 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = flatten_l2_optimize(best_a,
                                                 phase2_end,
                                                 n,
                                                 fft_len)
        print(f"[L2‑flatten] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact‑max sub‑gradient descent (≈25 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.35)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
