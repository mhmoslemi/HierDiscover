"""
A refined search for a low‐evaluation step sequence.

The core idea is the same as the reference solution – a multi‑phase
local optimisation pipeline – but we add a richer set of initial
candidates (including very sparse Dirichlet draws and power‑law
shapes), a longer L‑BFGS λ‑schedule, more random restarts and a more
aggressive simulated‑annealing phase.  The code respects the
time‑budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) max objective and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet(n: int, rng: np.random.Generator,
                     alpha_low: float = 0.05, alpha_high: float = 5.0) -> np.ndarray:
    """Sample a random Dirichlet point – concentration can be tuned."""
    alpha = rng.uniform(alpha_low, alpha_high)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def sparse_dirichlet(n: int, rng: np.random.Generator,
                     sparsity: float = 0.02) -> np.ndarray:
    """Very sparse Dirichlet by zeroing most entries."""
    a = np.zeros(n)
    m = max(1, int(sparsity * n))
    idx = rng.choice(n, size=m, replace=False)
    # concentrate mass uniformly on the chosen indices
    a[idx] = 1.0 / m
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def powerlaw_init(n: int, rng: np.random.Generator,
                  expo_low: float = 0.0, expo_high: float = 2.0) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with random exponent."""
    p = rng.uniform(expo_low, expo_high)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def exponential_decay_init(n: int, rng: np.random.Generator,
                           tau_low: float = 0.1, tau_high: float = 0.5) -> np.ndarray:
    """a_i ∝ exp(-i / τ) with random τ."""
    tau = rng.uniform(tau_low, tau_high) * n
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS over the whole λ schedule starting from a_start.
    Returns the best vector seen.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.05,
                  max_iter: int = 200000):
    """
    Adam on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    focusing on the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # indices i where i + (idx-i) = idx
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = sub
        # extra diagonal term when i == idx-i
        if idx % 2 == 0:
            i_mid = idx // 2
            if 0 <= i_mid < n:
                g[i_mid] += a[i_mid]
        g *= 2.0

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 500000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # propose moving a fraction of mass from i to j
        delta = a[i] * rng.random() * 0.25
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 400000,
               delta_frac: float = 0.10):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a few thousand gives a good trade‑off.
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # deterministic windows
    candidates.append(uniform_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(gaussian_init(n, sigma_factor=0.35))
    candidates.append(gaussian_init(n, sigma_factor=0.45))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(kaiser_init(n, beta=10.0))
    candidates.append(arcsine_init(n))

    # random Dirichlet draws – wide concentration range
    for _ in range(12):
        candidates.append(random_dirichlet(n, rng, 0.02, 2.0))
    for _ in range(8):
        candidates.append(random_dirichlet(n, rng, 2.0, 10.0))

    # very sparse draws
    for _ in range(6):
        candidates.append(sparse_dirichlet(n, rng, sparsity=rng.uniform(0.005, 0.02)))

    # power‑law / exponential families
    for _ in range(6):
        candidates.append(powerlaw_init(n, rng, 0.0, 2.5))
        candidates.append(exponential_decay_init(n, rng, 0.05, 0.5))

    # convex mixtures with the uniform vector
    uniform = uniform_init(n)
    for base in candidates[2:]:
        for w in np.linspace(0.0, 1.0, 9):
            a = (1.0 - w) * uniform + w * base
            a /= a.sum()
            candidates.append(a)

    # previous solution, if any (allows continuation across runs)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # --------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 5000

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # a few aggressive random restarts
    restarts = 8
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_dirichlet(n, rng, 0.01, 8.0)
        alpha = rng.random() * 0.5
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase1_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.0004,
                                           lr=0.04,
                                           max_iter=120000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.012)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient descent (≈10 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.65 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.40)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.85 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=8e-4,
                                                 final_temp=5e-10,
                                                 max_moves=1200000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
