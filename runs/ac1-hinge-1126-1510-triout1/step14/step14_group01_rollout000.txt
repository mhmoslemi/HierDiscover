"""
Propose a low‑score step function.

Algorithm outline
-----------------
0. Build a diverse pool of initial candidates (classical windows,
   random Dirichlet draws, Beta‑family, mixtures of Gaussians, etc.).
   Keep the best according to the exact fast_score.

1. Repeated L‑BFGS refinements on a log‑softmax parametrisation.
   A schedule of decreasing smoothing parameters (log‑sum‑exp)
   drives the optimizer towards the true max‑convolution.
   Each run starts from a random convex combination of the current
   best and a fresh random vector, and the best result is kept.
   After each L‑BFGS run the vector is symmetrised (a←(a+a[::-1])/2).

2. Simulated‑annealing / guided hill‑climb.
   Using the current best as a seed we repeatedly perform tiny mass‑transfer
   moves (δ transferred from a high‑gradient index to a low‑gradient one).
   The move is always accepted if it improves the fast_score, otherwise it
   is accepted with probability exp(−Δ/temperature).  Temperature and the
   move‑size decay slowly.  After each move the candidate is forced to be
   symmetric and renormalised.

3. Final polishing by projected sub‑gradient descent on the exact
   max‑convolution (the same routine as in the reference code, but run
   for the remaining budget).

All phases respect the global 1000 s budget; the best vector ever seen is
returned.  The final result is passed to ``evaluate_sequence`` (which
rescales internally) and the official score is printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector a (sum≈1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i]=a[-i-1] and renormalise to sum 1."""
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.shape[0])
    return a

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def gaussian_mixture_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Gaussians with random parameters."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0

    sigma1 = rng.uniform(0.03, 0.12) * n
    sigma2 = rng.uniform(0.03, 0.12) * n
    offset = rng.uniform(-0.2, 0.2) * n

    g1 = np.exp(-0.5 * ((i - centre) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((i - (centre + offset)) / sigma2) ** 2)

    weight = rng.uniform(0.2, 0.8)
    a = weight * g1 + (1 - weight) * g2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z

    # back‑prop through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimization primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        a = symmetrize(a)               # enforce symmetry after each stage
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator):
    """
    Simulated annealing with mass‑transfer moves guided by the
    gradient of the maximal convolution entry.
    """
    cur = a_start.copy()
    cur_score = fast_score(cur, fft_len, n)

    best = cur.copy()
    best_score = cur_score

    # temperature and move‑size schedules
    temp = cur_score * 0.02 + 1e-6
    move_frac = 0.08
    decay_temp = 0.9997
    decay_move = 0.9999

    while time.time() < deadline:
        # pick donor and receiver
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor_mass = cur[i]
        if donor_mass <= 1e-12:
            continue

        delta = donor_mass * move_frac * rng.random()
        if delta <= 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # optional small Gaussian smoothing to avoid extreme spikes
        # (commented out for speed; can be enabled if needed)
        # kernel = np.array([0.25, 0.5, 0.25])
        # cand = np.convolve(cand, kernel, mode='same')
        # cand = np.maximum(cand, 0.0)

        cand = symmetrize(cand)

        # re‑normalise (symmetrisation may introduce tiny drift)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_score = fast_score(cand, fft_len, n)

        # Metropolis accept/reject
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / temp):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # decay schedules
        temp *= decay_temp
        move_frac = max(0.001, move_frac * decay_move)

    return best, best_score


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size (decreasing)
        step = 0.5 / np.sqrt(t)

        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 4096 gives a good trade‑off between expressiveness and speed.
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, upsample it (helps continuation)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.22))
    candidates.append(gaussian_init(n, sigma_factor=0.28))
    for beta in (6.0, 9.0, 12.0):
        candidates.append(kaiser_init(n, beta))

    # Beta sweep (symmetric)
    for alpha in np.linspace(0.8, 4.0, 7):
        candidates.append(beta_init(n, alpha, alpha))

    # Additional randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(gaussian_mixture_init(n, rng))

    # ------------------------------------------------------------------
    # Pick the best initial candidate (exact fast_score)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Multiple L‑BFGS refinements (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2500

    # Perform restarts while we have time left in this phase
    while time.time() < phase1_end:
        # random mixture of current best and a fresh random vector
        rand_vec = random_initial(n, rng)
        mix = rng.random()
        a_start = (1.0 - mix) * best_a + mix * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing / guided hill‑climb (≈40 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = min(deadline, time.time() + 0.40 * budget_s)
        print("[SA] start simulated‑annealing refinement")
        best_a, best_score = simulated_annealing_optimize(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final sub‑gradient polishing (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        ft_deadline = time.time() + remaining
        print("[fine‑tune] start projected sub‑gradient polishing")
        best_a, best_score = fine_tune_subgradient(
            best_a,
            ft_deadline,
            n,
            fft_len
        )
        print(f"[fine‑tune] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
