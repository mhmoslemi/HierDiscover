"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Broad exploratory sweep** – generate a large pool of candidate
   probability vectors from many parameterised families
   (Beta, sinusoidal, exponential, Kaiser, DPSS, random complex‑phase,
   mixtures with the uniform distribution).  Compute the exact fast
   objective  2·n·max(conv)  for each and keep the best.

2. **Smooth optimisation with L‑BFGS** – treat the vector as a soft‑max
   of free logits and minimise a log‑sum‑exp approximation of the
   maximum convolution.  A continuation on a schedule of smoothing
   parameters (λ from 0.5 down to 1e‑7) is performed, with many random
   restarts that mix the current best with fresh Dirichlet draws.

3. **Exact sub‑gradient descent + multi‑scale hill‑climb** – apply an
   exponentiated‑gradient descent on the true non‑smooth objective
   (max of the autocorrelation) followed by several passes of greedy
   mass‑transfer moves with decreasing step sizes to polish the result.

All phases respect the global time budget (default 1000 s).  The
returned list is a non‑negative sequence; the official `evaluate_sequence`
will internally rescale it, but we keep the vector normalised to unit
sum throughout the optimisation.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss
from scipy.stats import beta

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic windows)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a handful of simple families on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            # In practice this rarely happens.
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT (fast)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/sqrt(t))
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.04):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.  Only accepts
    strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Phase 0 – broad candidate sweep (≈15 % of budget)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # deterministic windows
    candidates = [
        spectral_init(n),
        triangular_init(n),
        hann_init(n),
        blackman_init(n),
        gaussian_init(n, sigma_factor=0.25),
        kaiser_init(n, beta=8.0),
        dpss_init(n, NW=3.5),
    ]

    # --------------------------------------------------------------
    # Parameterised families
    # --------------------------------------------------------------
    x_grid = (np.arange(n) + 0.5) / n   # avoid exactly 0/1 for beta

    # Beta(α,α) sweep
    for α in np.linspace(0.2, 2.0, 19):
        pdf = beta.pdf(x_grid, α, α)
        a = np.clip(pdf, 0.0, None)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # sin^p sweep (p > 0)
    for p in np.linspace(0.5, 5.0, 19):
        a = np.sin(np.pi * x_grid) ** p
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # exponential decay sweep
    for rate in np.linspace(0.001, 0.02, 12):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        candidates.append(a)

    # Kaiser beta sweep
    for beta_val in np.linspace(4.0, 12.0, 9):
        a = np.kaiser(n, beta_val)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # DPSS (Slepian) sweep
    for NW in np.linspace(2.0, 5.0, 16):
        a = dpss_init(n, NW=NW)
        candidates.append(a)

    # Random complex‑phase magnitude‑square construction
    for _ in range(8):
        phi = rng.random(fft_len) * 2.0 * np.pi
        s = np.exp(1j * phi)
        a_complex = np.fft.ifft(s)
        a = np.abs(a_complex[:n]) ** 2
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # Mixtures with the uniform distribution (helps fine tuning)
    uniform_vec = np.full(n, 1.0 / n)
    mix_candidates = []
    for cand in candidates:
        mix = 0.6 * uniform_vec + 0.4 * cand
        mix /= mix.sum()
        mix_candidates.append(mix)
    candidates.extend(mix_candidates)

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best so far
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init‑sweep] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS (≈35 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7, 5e-8, 1e-8
    ]
    maxiter_per_lam = 4000

    while time.time() < phase1_end:
        # Generate a fresh starting point by mixing the current best with a random draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4  # mixing weight
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0   # keep a few seconds for polishing
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – multi‑scale greedy hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # run several passes with decreasing move size
        hill_deadline = deadline
        delta = 0.04
        while time.time() < hill_deadline and delta > 1e-5:
            passes = int(50000 * (delta / 0.04) + 1)  # fewer iterations for tiny steps
            best_a, best_score = hill_climb(
                best_a,
                best_score,
                fft_len,
                n,
                hill_deadline,
                rng,
                max_iter=passes,
                delta_frac=delta
            )
            print(f"[hill‑climb δ={delta:.5f}] best fast_score={best_score:.8f}")
            delta *= 0.5   # refine

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
