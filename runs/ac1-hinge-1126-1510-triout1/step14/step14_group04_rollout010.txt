"""
Propose a low‑score step function.

Algorithm Overview
------------------
1. **Initial pool** – a mix of deterministic windows (Hann, Blackman,
   Kaiser, Chebyshev, DPSS, Beta‑shaped) and several random Dirichlet‑like
   draws. The best of these is kept as the starting point.

2. **Fast exact objective** – `fast_score` evaluates the true objective
   `2·n·max(conv(a,a))` (the evaluation function is scale‑invariant when
   Σa=1, so we normalise the vector after every update).

3. **Peak‑focused Exponentiated‑Gradient (PE‑EG)** – the heaviest
   convolution entry (`k* = argmax conv`) is identified, the exact
   sub‑gradient of that entry is obtained by a single FFT‑based
   back‑propagation, and the probabilities are updated with an
   exponentiated‑gradient step.  This directly attacks the current
   bottleneck and has proved more effective than smoothing‑based L‑BFGS
   for the final polishing.

4. **Random‑restart outer loop** – many short PE‑EG runs are launched
   from random perturbations of the current best vector (simple
   convex‑combination with a fresh random draw).  The loop respects the
   global time budget.

5. **Optional L‑BFGS refinement** – a few continuation steps with a
   smooth log‑sum‑exp surrogate are performed on the current best
   candidate to squeeze any remaining gain.

6. **Final sanity‑check** – clip, renormalise and evaluate with the
   official `evaluate_sequence`.  The resulting list is returned.

The whole search runs within the 1000 s budget and prints intermediate

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a * a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    x → a = softmax(y)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a = a / a.sum()
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
        a = a / a.sum()
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (smooth surrogate)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage from
    the previous result.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Peak‑focused Exponentiated‑Gradient optimisation
# ----------------------------------------------------------------------


def peak_eg_optimize(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator,
                     max_inner: int = 3000):
    """
    Gradient descent that attacks the current maximal convolution entry.
    Uses an exponentiated‑gradient update with a decreasing step‑size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_inner:
        # compute full convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # index of the peak entry
        k_star = int(conv.argmax())

        # one‑hot mask for the peak (exact sub‑gradient)
        mask = np.zeros_like(conv)
        mask[k_star] = 1.0

        # back‑prop to obtain gradient wrt a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fw, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # learning rate schedule
        eta = 0.5 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (clip to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[Peak‑EG] it={it}, fast_score={best_sc:.8f}")

        it += 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep it moderate (4096) to balance quality & speed
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the current size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(chebwin_init(n, attenuation=rng.uniform(40.0, 80.0)))
    candidates.append(dpss_init(n, half_bw=rng.uniform(1.5, 5.0)))

    # randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families (including U‑shapes)
    for _ in range(6):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # --------------------------------------------------------------
    # Keep the best initial candidate (fast_score is exact for Σa=1)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Random‑restart Peak‑EG loops
    # ------------------------------------------------------------------
    pe_eg_deadline = time.time() + 0.55 * budget_s
    print("[Phase 1] Peak‑EG exploration")
    while time.time() < pe_eg_deadline:
        # blend current best with a fresh random vector to keep diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.25  # up to 25 % random perturbation
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        # inner optimisation run (time‑limited)
        sub_deadline = min(pe_eg_deadline, time.time() + 5.0)  # max 5 s per run
        a_opt, sc_opt = peak_eg_optimize(a_start,
                                          sub_deadline,
                                          n,
                                          fft_len,
                                          rng,
                                          max_inner=4000)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Light L‑BFGS refinement (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Phase 2] L‑BFGS continuation")
        lam_schedule = np.geomspace(5e-1, 5e-6, num=35)
        maxiter_per_lam = 800
        best_a, best_sc = lbfgs_continuation(best_a,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              deadline,
                                              fft_len,
                                              fft_len,
                                              n)
        print(f"[Phase 2] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (tiny deterministic moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 20.0)
        print("[Phase 3] Hill‑climb polishing")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.025)
        print(f"[Phase 3] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final sanity‑check & return
    # --------------------------------------------------------------
    final_seq = np.clip(best_a, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
