def lp_optimize(f, M, t):
    n = len(f)
    g = cp.Variable(n)
    conv = cp.conv(f, g)  # returns (2n-1) length variable.
    constraints = [conv <= M,
                   g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except cp.SolverError:
        prob.solve(solver=cp.SCS, max_iters=2000)

    g_opt = np.array(g.value).flatten()
    # handle if solution not found.
    if g_opt is None:
        # fallback: return f
        return f
    # mix
    new_f = (1-t)*f + t*g_opt
    new_f = np.maximum(new_f, 0)
    new_f_sum = new_f.sum()
    if new_f_sum > 0:
        new_f = new_f / new_f_sum
    else:
        new_f = uniform_init(len(f))
    return new_f

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n)
    conv = cp.conv(f, g)  # shape 2n-1
    constraints = [conv <= M,
                   g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8, reltol=1e-6)
    except Exception as e:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            return np.full(n, 1.0 / n)  # fallback

    if g.value is None:
        return np.full(n, 1.0 / n)
    g_val = np.array(g.value).flatten()
    # Ensure non-negativity due to numerical errors.
    g_val = np.maximum(g_val, 0.0)
    total = g_val.sum()
    if total > 0:
        g_val = g_val / total
    else:
        g_val = np.full(n, 1.0 / n)
    return g_val

"""
Propose a low‑score step sequence.

Algorithm
---------
1. Build a rich pool of initial candidates (classical windows,
   power‑law shapes, Dirichlet samples, convex combos with the uniform
   vector).  Keep the one with the smallest exact fast_score
   (fast_score(a)=2·n·max(a*a) for ∑a=1).

2. LP refinement (optional, uses cvxpy if available):
   For the current vector f compute M = max(f*f).
   Solve the linear program
        maximise   ∑ g_j
        subject to conv(f,g) ≤ M  (component‑wise)
                    g ≥ 0
   and replace f ← (1‑α)·f + α·g (α starts at 0.5 and is reduced
   when no improvement is seen).  Perform a handful of such
   alternating steps.

3. L‑BFGS minimisation of a smooth approximation of the max‑convolution.
   The objective is a log‑sum‑exp smoothed max; a schedule of decreasing
   smoothing parameters (λ) is traversed, each with a few thousand
   L‑BFGS iterations.  Random restarts with a small mixture of a Dirichlet
   draw are performed.

4. Adam optimisation on the same smoothed objective for fine‑grained
   descent.

5. Exact‑max sub‑gradient descent with a back‑tracking line‑search.
   The gradient is the sum of the sub‑gradients of all convolution entries
   whose value is within a tiny tolerance of the current maximum.

6. A final greedy hill‑climb that only accepts strictly improving
   mass‑transfer moves.

All phases respect the time budget (default 1000 s).  The best vector
found is returned as a plain list of floats; `evaluate_sequence` will be
called on it before returning.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def powerlaw_init(n: int, exponent: float) -> np.ndarray:
    """Monotonically decreasing power‑law shape."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** exponent
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.05,
                  max_iter: int = 200000):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f_val, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30,
                               eps: float = 1e-12):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution; aggregates all entries within a tiny
    tolerance of the maximum to obtain a more informative direction.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        # indices whose value is within eps of the maximum
        idxs = np.where(conv >= M - eps)[0]

        # accumulate gradient contributions
        g = np.zeros_like(a)
        for idx in idxs:
            j_low = max(0, idx - (n - 1))
            j_high = min(idx, n - 1)
            # contribution = 2 * a[idx - i] for i in [j_low, j_high]
            sub = a[idx - np.arange(j_low, j_high + 1)]
            g[j_low:j_high + 1] += 2.0 * sub
        # zero‑sum projection
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # feasible step length (keep a ≥ 0)
        neg = d < 0
        if np.any(neg):
            alpha_max = np.min(-a[neg] / d[neg])
        else:
            alpha_max = np.inf
        alpha = step0 / np.sqrt(it + 1.0)
        alpha = min(alpha, alpha_max)

        # simple back‑tracking line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # if no improvement, just project
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 500000,
               delta_frac: float = 0.10):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

try:
    import cvxpy as cp
except Exception:   # pragma: no cover
    cp = None


def lp_refine(f: np.ndarray, M: float, alpha: float) -> np.ndarray:
    """
    Solve  maximise   ∑ g_j
        s.t.        conv(f,g) ≤ M   (component‑wise)
                     g ≥ 0
    Return the mixture (1‑α)·f + α·g normalised to sum = 1.
    If cvxpy is unavailable the function just returns f unchanged.
    """
    if cp is None:   # pragma: no cover
        return f

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    # cvxpy's conv works with a constant and a variable
    conv_fg = cp.conv(f, g)          # length 2n‑1
    constraints = [conv_fg <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-6, feastol=1e-9)
    except Exception:   # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, max_iters=8000, verbose=False)
        except Exception:
            return f

    if g.value is None:
        return f

    g_val = np.array(g.value).flatten()
    g_val = np.maximum(g_val, 0.0)
    # mixture and re‑normalise
    new_f = (1.0 - alpha) * f + alpha * g_val
    new_f = simplex_projection(new_f)
    return new_f

                                           fft_len,
                                           lam_factor=0.0003,
                                           lr=0.04,
                                           max_iter=120000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact‑max sub‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
