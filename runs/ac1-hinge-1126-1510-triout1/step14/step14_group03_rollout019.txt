def pgd_optimize(a_start, start_deadline, end_deadline, n, fft_len, init_eta=0.5):
   a = a_start.copy()
   best = a.copy()
   best_score = fast_score(best, fft_len, n)
   t = 0
   while time.time() < end_deadline:
       t += 1
       fa = np.fft.rfft(a, fft_len)
       conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
       max_val = conv.max()
       k_max = int(np.argmax(conv))
       # gradient slice
       i_start = max(0, k_max - (n - 1))
       i_end = min(n - 1, k_max)
       g = np.zeros_like(a)
       # compute indices
       idx = np.arange(i_start, i_end + 1)
       g[idx] = 2 * a[k_max - idx]
       # step size
       eta = init_eta / np.sqrt(t)
       a = a - eta * g
       # enforce non-negativity
       a[a < 0] = 0.0
       s = a.sum()
       if s <= 0:

       else:
           a = a / s
       sc = fast_score(a, fft_len, n)
       if sc < best_score:
           best_score = sc
           best = a.copy()
   return best, best_score

"""
search for a low‐value step function.

The algorithm consists of four stages:

1.  **Diversified initial pool** – a handful of analytic windows
    (Hann, Blackman, Kaiser, DPSS, …) together with a few Dirichlet‑type
    random draws and the previously found construction (if any).  The
    best of these is taken as the starting point.

2.  **Heavy L‑BFGS smoothing** – we minimise a log‑sum‑exp
    smooth approximation of `max(conv(a,a))`.  Each run starts from a
    convex combination of the current best and a fresh random draw,
    runs through a long λ‑schedule and keeps the best vector discovered.

3.  **Projected sub‑gradient descent** – a simple additive sub‑gradient
    method on the exact nonsmooth objective.
    For the current maximal convolution index `k*` the gradient is
    `g_i = 2·a_{k*‑i}` (zero outside the range).  After each step we
    enforce non‑negativity and renormalise onto the simplex.
    Many restarts are performed while the budget permits.

4.  **Multi‑scale greedy hill‑climb** – a series of random mass‑transfer
    moves with decreasing step size (`δ = 4 %, 2 %, 1 %, 0.5 %, 0.1 %`).
    Only strictly improving moves are accepted.
    Because the move size shrinks the final stage can polish the
    solution to a higher precision.

All stages respect the global 1000 s wall‑clock limit.
The returned vector is a plain list of `float`s; the official
`evaluate_sequence` will rescale it internally, so we keep the sum
equal to 1 throughout the optimisation.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def conv_max_and_idx(a: np.ndarray, fft_len: int, n: int):
    """Return (max value, index of max) of a*a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Initialisation helpers (all return vectors with sum = 1)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    padded = np.zeros(L)
    padded[: 2 * n - 1] = target
    S = np.fft.fft(padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    tmp = np.fft.ifft(A).real[:n]
    tmp[tmp < 0] = 0.0
    s = tmp.sum()
    return tmp / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient
    w.r.t. the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run L‑BFGS over the whole λ‑schedule."""
    y = np.log(a_start + 1e-12)          # start in log‑space
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def pgd_optimize(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Simple additive sub‑gradient scheme.
    The gradient of `max(conv)` at the current maximiser `k*` is
    g_i = 2·a_{k*‑i} (zero outside the range).  After each step we set
    negative entries to zero and renormalise.
    """
    a = a_start.copy()
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        # convolution & index of maximal entry
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # gradient slice (use a copy so that updates do not interfere)
        a_old = a.copy()
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        idx = np.arange(i_lo, i_hi + 1, dtype=int)
        a[idx] = a_old[idx] - 2.0 * (init_lr / np.sqrt(t)) * a_old[k_max - idx]

        # enforce feasibility
        a[a < 0.0] = 0.0
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
    return best, best_sc


# ----------------------------------------------------------------------
# Multi‑scale greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb_multi(a: np.ndarray,
                     cur_score: float,
                     fft_len: int,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator):
    """
    Greedy hill‑climb with a sequence of shrinking move sizes.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    # move size schedule (as a fraction of the source entry)
    for delta_frac in (0.04, 0.02, 0.01, 0.005, 0.001):
        max_iter = 20000
        it = 0
        while time.time() < deadline and it < max_iter:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                it += 1
                continue
            # propose a tiny transfer from i to j
            delta = best[i] * rng.random() * delta_frac
            if delta <= 0.0:
                it += 1
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            # (non‑negativity is guaranteed by construction)
            sc = fast_score(cand, fft_len, n)
            if sc < best_score:
                best = cand
                best_score = sc
            it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    candidates = []

    # reuse previous construction if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # random draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # pick the best by exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # convex blend of current best and a fresh random draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈50 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        pgd_deadline = deadline - 8.0   # keep a few seconds for polishing
        print("[PGD] start projected sub‑gradient descent")
        while time.time() < pgd_deadline:
            # fresh start: blend best with a new random vector
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.35
            a_start = (1.0 - alpha) * best_a + alpha * rand_vec
            a_start /= a_start.sum()
            a_opt, sc_opt = pgd_optimize(
                a_start,
                pgd_deadline,
                n,
                fft_len,
                init_lr=0.5
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[PGD] improved fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – final multi‑scale hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Hill‑climb] start final greedy polishing")
        best_a, best_score = hill_climb_multi(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng
        )
        print(f"[Hill‑climb] final fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence globally for possible reuse
    globals()['height_sequence_1'] = final_seq

    return final_seq
