def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4096
    fft_len = next_pow2(2*n - 1)

    # initial candidates as before include more beta parameter sweeps
    best_a = None
    best_score = np.inf

    # generate candidates from beta param grid (maybe 200 combos)
    alphas = np.concatenate([np.linspace(0.05, 0.9, 20), np.linspace(1.0, 6.0, 20)])
    betas = alphas # start symmetric
    for a_ in alphas:
        cand = beta_init(n, a_, a_)
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    # also try asymmetric combos: random pairs
    for _ in range(200):
        a_ = 10**rng.uniform(-2, 1)  # sample log-uniform maybe
        b_ = 10**rng.uniform(-2, 1)
        cand = beta_init(n, a_, b_)
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    # also try random vectors
    for _ in range(30):
        cand = random_initial(n, rng)
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # Now annealing phase
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    cur_a = best_a.copy()
    cur_score = best_score

    # Temperature schedule
    T0 = 1.0
    T_min = 1e-5
    # Determine total iterations approximate
    # We'll loop while time left
    it = 0
    while time.time() < deadline:
        it += 1
        # propose move: select two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # fraction to move
        frac = rng.random() * 0.2  # up to 20% of mass at i
        delta = cur_a[i] * frac
        if delta <= 0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non-negative (should be)
        # compute score
        cand_score = fast_score(cand, fft_len, n)
        # acceptance
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                # print improvement
                print(f"[SA] iter {it} new best {best_score:.8f}")
        else:
            # temperature decreasing
            elapsed = time.time() - start_time
            total = deadline - start_time
            T = T0 * (1 - elapsed / total) ** 2 + T_min
            prob = np.exp(-(cand_score - cur_score) / T)
            if rng.random() < prob:
                cur_a = cand
                cur_score = cand_score

    # Final optional fine-tune with local hill climb for a bit
    # we can reuse hill_climb function with small moves
    try:
        remaining = deadline - time.time()
        if remaining > 5.0:
            hill_deadline = time.time() + min(remaining, 10.0)
            cur_a, cur_score = hill_climb(best_a, best_score, fft_len, n, hill_deadline, rng, max_iter=20000, delta_frac=0.05)
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
    except Exception as e:
        print("[hill after SA] exception:", e)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq

"""
Propose a low‑score step‑function.

This version starts from a wide collection of simple families (including
Beta‑distributions with exponents < 1, spectral and window shapes) and
keeps the best among them.  It then runs a Simulated‑Annealing (SA)
search that repeatedly moves a random fraction of mass from one index
to another, accepting worsening moves with a temperature that cools
linearly to zero.  After the SA phase a short greedy hill‑climb and a
final projected sub‑gradient polishing are performed.  The whole
pipeline respects the 1000 s time limit; the best sequence seen is
returned and evaluated with the official ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum≈1).
    Returns 2·n·max( a * a ) where * is the (linear) convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root) guess, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete symmetric or asymmetric Beta‑type density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A collection of inexpensive random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """Greedy hill‑climb with tiny mass‑transfer moves (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active shift (ignoring the 2·n factor)
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = 0.5 / np.sqrt(t)
        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (a power‑of‑two size yields fast FFTs)
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an extensive pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, up‑sample it (helps continuation)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=6.0))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(kaiser_init(n, beta=12.0))
    candidates.append(gaussian_init(n, sigma_factor=0.22))
    candidates.append(gaussian_init(n, sigma_factor=0.28))

    # Symmetric Beta sweep – now also covering exponents < 1
    alphas = np.concatenate([
        np.linspace(0.05, 0.7, 14),   # < 1   (U‑shaped)
        np.linspace(0.8, 4.0, 12)    # ≥ 1
    ])
    for a in alphas:
        cand = beta_init(n, a, a)
        candidates.append(cand)

    # A few random draws (diversity)
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Pick the best of the pool (fast exact score)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Simulated Annealing (≈45 % of budget)
    # ------------------------------------------------------------------
    sa_end = start_time + 0.45 * budget_s
    cur_a = best_a.copy()
    cur_score = best_score

    # SA parameters
    T0 = 1.0          # initial temperature
    T_min = 1e-5      # final temperature
    max_move_frac_initial = 0.30   # up to 30 % of the source mass
    max_move_frac_final = 0.001    # very fine moves at the end

    it = 0
    while time.time() < sa_end:
        it += 1
        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # linearly decay the maximal transferable fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        max_move_frac = max_move_frac_final + \
            (max_move_frac_initial - max_move_frac_final) * (1.0 - frac)

        delta = cur_a[i] * rng.random() * max_move_frac
        if delta <= 0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta   # mass is conserved, sum stays 1

        cand_score = fast_score(cand, fft_len, n)

        # greedy acceptance if we improve
        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"[SA] iter {it:,} new best {best_score:.8f}")
            continue

        # otherwise accept with probability exp(-Δ/T)
        elapsed = time.time() - start_time
        T = max(T_min, T0 * (1.0 - elapsed / total) ** 2)
        prob = np.exp(-(cand_score - cur_score) / T)
        if rng.random() < prob:
            cur_a, cur_score = cand, cand_score

    print(f"[SA] finished with best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Greedy hill‑climb (≈30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.30 * budget_s)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n,
            hill_deadline, rng,
            max_iter=200000,
            delta_frac=0.05
        )
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final sub‑gradient polishing (≈15 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        fine_deadline = time.time() + min(remaining, 30.0)
        best_a, best_score = fine_tune_subgradient(
            best_a, fine_deadline, n, fft_len
        )
        print(f"[fine‑tune] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
