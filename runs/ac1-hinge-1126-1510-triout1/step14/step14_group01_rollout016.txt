def init_alternating_projection(n, L, iterations, seed):
    rng = np.random.default_rng(seed)
    # start with random nonnegative vector (probability simplex)
    a = rng.random(n)
    a = a / a.sum()
    A_target = np.sqrt(2*n/(2*n-1))
    for it in range(iterations):
        # compute FFT of padded a
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        # enforce magnitude
        phase = np.angle(A)
        A_new = A_target * np.exp(1j*phase)
        # invert
        a_full = np.fft.irfft(A_new, L)
        a_new = a_full[:n]
        # project to nonnegative
        a_new = np.maximum(a_new, 0)
        # renormalize sum to 1
        if a_new.sum() > 0:
            a_new /= a_new.sum()
        else:
            a_new = np.full(n, 1.0/n)
        a = a_new
    return a

"""
Propose a low‑score step function.

Algorithm
---------

0. Build a diverse pool of initial candidates (classical windows, beta
   families, random draws).  Keep the best according to the exact
   fast_score (2·n·max(conv) – the evaluation is scale‑invariant).

1. Smoothed L‑BFGS (log‑sum‑exp approximation of the max‑convolution).

2. Adam optimisation with a top‑k mask (flattens the largest convolution
   entries).

3. Exponentiated‑gradient descent on the exact max‑convolution.

4. Greedy hill‑climb with tiny mass‑transfer moves.

5. Fine‑grained sub‑gradient descent on the true max entry.

6. **Active‑set linear‑programming refinement** – the most original
   part.  Given the current vector `f` (normalised to Σf = √(2·n)),
   compute M = max(conv(f,f)).  Build the linear constraints
   (f⋆g)_i ≤ M for the (few) indices i where the max is attained,
   maximise Σg subject to g ≥ 0 and the constraints, then mix
   `f ← (1‑t)·f + t·g` (with a decreasing step‑size).  This is the
   procedure described in the statement and typically pushes the
   bound below the previous best.

All phases respect the overall 1000 s budget; the best vector seen so
far is always retained.  The final sequence is returned on the simplex
(sum ≈ 1) – `evaluate_sequence` rescales it internally, so the score is
invariant to the absolute scale.
"""

import time

# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Fast exact objective (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum≈1).
    Returns 2·n·max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type density on the grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.02):
    """Adam optimiser that uses a top‑k mask."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score


def eg_optimize(a_start: np.ndarray,
                deadline: float,
                n: int,
                fft_len: int,
                rng: np.random.Generator,
                init_lr: float = 1.0,
                top_start: float = 0.12,
                top_end: float = 0.02):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    start_time = time.time()
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        top_frac = max(top_end, top_start * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        idxs = np.argpartition(conv, -k)[-k:]
        mask = np.zeros_like(conv)
        mask[idxs] = 1.0

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """Greedy hill‑climb with tiny mass‑transfer moves."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


def fine_tune_subgradient(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = 0.5 / np.sqrt(t)

        a_new = a - step * g
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def active_set_lp_improve(a_init: np.ndarray,
                          deadline: float,
                          n: int,
                          start_time: float):
    """
    Linear‑programming based active‑set refinement.
    The vector is kept on the scale Σa = √(2·n), so the score is simply max(conv).
    """
    L = next_pow2(2 * n - 1)
    target_sum = np.sqrt(2.0 * n)

    # scale to the required sum
    a = a_init * (target_sum / a_init.sum())
    best_a = a.copy()
    best_score = fast_score(a, L, n)  # = max(conv) * 2·n / (sum^2) = max(conv)

    # mixing schedule
    t_start = 0.30
    t_end = 0.02

    rng = np.random.default_rng()

    while time.time() < deadline:
        # current convolution and bound
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
        M = conv.max()

        # active indices (within a tiny tolerance)
        eps = max(1e-12, 1e-6 * M)
        active_idx = np.where(conv >= M - eps)[0]

        # limit the number of constraints for speed
        if active_idx.size > 200:
            active_idx = rng.choice(active_idx, size=200, replace=False)

        # build linear constraints A g ≤ M   (A shape: |active| × n)
        A = np.zeros((len(active_idx), n), dtype=np.float64)
        for row, i in enumerate(active_idx):
            k_min = max(0, i - (n - 1))
            k_max = min(i, n - 1)
            ks = np.arange(k_min, k_max + 1)
            # coefficient for g_k is a_{i‑k}
            A[row, ks] = a[i - ks]

        b = np.full(len(active_idx), M, dtype=np.float64)

        # maximise Σ g  → minimise -Σ g
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # If LP fails we abort the refinement phase.
            print("[LP] linprog failure:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Rescale g to the required sum
        g = g * (target_sum / sum_g)

        # adaptive mixing parameter
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        t_mix = max(t_end, t_start * (1.0 - elapsed_frac))

        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new * (target_sum / a_new.sum())

        new_score = fast_score(a_new, L, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] improved best fast_score={best_score:.8f}")
        else:
            # no improvement – try a smaller step or quit
            if t_mix > t_end * 2:
                t_mix *= 0.5
                a_new = (1.0 - t_mix) * a + t_mix * g
                a_new = np.clip(a_new, 0.0, None)
                a_new = a_new * (target_sum / a_new.sum())
                new_score = fast_score(a_new, L, n)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a_new.copy()
                    a = a_new
                    print(f"[LP] improved with smaller t, score={best_score:.8f}")
                else:
                    # stagnation – exit
                    break
            else:
                break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (trade‑off between granularity and speed)
    # ------------------------------------------------------------------
    n = 4096                         # power‑of‑two size – good FFT behaviour
    fft_len = next_pow2(2 * n - 1)   # length for zero‑padded FFTs

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, upsample it
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.22))
    candidates.append(gaussian_init(n, sigma_factor=0.28))
    for beta in (6.0, 9.0, 12.0):
        candidates.append(kaiser_init(n, beta))

    # Symmetric Beta sweep
    for alpha in np.linspace(0.8, 4.0, 7):
        cand = beta_init(n, alpha, alpha)
        candidates.append(cand)

    # Randomised families
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Evaluate all candidates with the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Smoothed L‑BFGS (≈15 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈30 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start exact optimisation")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.10,
            end_top=0.02
        )
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 2.0   # keep a couple of seconds free
        print("[EG] start exponentiated‑gradient optimisation")
        best_a, best_score = eg_optimize(
            best_a,
            eg_deadline,
            n,
            fft_len,
            rng,
            init_lr=1.0,
            top_start=0.10,
            top_end=0.02
        )
        print(f"[EG] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Greedy hill‑climb (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=180000,
            delta_frac=0.04
        )
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final sub‑gradient polishing (≈5 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        fine_deadline = time.time() + min(remaining, 30.0)
        best_a, best_score = fine_tune_subgradient(
            best_a,
            fine_deadline,
            n,
            fft_len
        )
        print(f"[fine‑tune] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Active‑set LP refinement (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = deadline
        print("[LP] start active‑set linear‑programming refinement")
        best_a_lp, best_score_lp = active_set_lp_improve(
            best_a,
            lp_deadline,
            n,
            start_time
        )
        if best_score_lp < best_score:
            best_score = best_score_lp
            best_a = best_a_lp
            print(f"[LP] final best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    # Convert back to a probability vector (sum≈1) for aesthetic output
    final_seq = (best_a / best_a.sum()).tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
