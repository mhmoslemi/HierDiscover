def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4096
    fft_len = next_pow2(2*n-1)
    start = time.time()
    deadline = start + budget_s - 0.1

    # candidate pool init
    candidates = []
    # deterministic windows: using range of sigma, beta, NW
    for sigma_factor in np.linspace(0.1, 0.5, 5):
        candidates.append(gaussian_init(n, sigma_factor))
    for beta in [4,6,8,10,12]:
        candidates.append(kaiser_init(n,beta))
    for NW in [2.5,3.0,3.5,4.0,4.5]:
        candidates.append(dpss_init(n,NW))
    # random
    for _ in range(20):
        candidates.append(random_initial(n, rng))
    # optionally previous best
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        # resample to n if needed
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0,0,None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # evaluate candidates via fast_score
    best_a = None; best_score = float('inf')
    for a in candidates:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc; best_a = a.copy()
    print(f"[init] best fast_score {best_score:.8f}")

    # Phase EG (main)
    # We'll run EG loops for about 60% of budget, maybe 500 secs
    # We'll run multiple restarts.
    phase_end = start + 0.6 * budget_s
    # schedule for restarts: after certain iterations we reset with random jitter.
    t_iter = 0
    a_curr = best_a.copy()
    best_overall = best_a.copy()
    best_overall_score = best_score
    while time.time() < phase_end:
        # compute convolution
        fa = np.fft.rfft(a_curr, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        k_max = int(np.argmax(conv))
        # gradient
        idx = k_max - np.arange(n)
        mask = (idx>=0) & (idx<n)
        grad = np.zeros_like(a_curr)
        grad[mask] = 2.0 * a_curr[idx[mask]]
        # step size
        t_iter += 1
        eta = 0.8 / np.sqrt(t_iter)  # adjust base 0.8
        # EG update
        a_curr = a_curr * np.exp(-eta * grad)
        sum_a = a_curr.sum()
        if sum_a <= 0 or not np.isfinite(sum_a):
            a_curr = np.full_like(a_curr, 1.0/n)
        else:
            a_curr /= sum_a
        # evaluate
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall = a_curr.copy()
            # print progress
            print(f"[EG] t={t_iter}, new best {best_overall_score:.8f}")
        # occasional random perturbation to escape
        if t_iter % 5000 == 0:
            # blend with random
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.3
            a_curr = (1-alpha) * a_curr + alpha * rand_vec
            a_curr /= a_curr.sum()
    # Phase hill climb + simulated annealing for rest
    if time.time() < deadline:
        remaining = deadline - time.time()
        # assign some to hill climb
        max_iter = 200000
        # start from best_overall
        a_hc = best_overall.copy()
        best_hc_score = best_overall_score
        it = 0
        # temperature schedule: start high
        temp0 = 0.1
        while time.time() < deadline and it < max_iter:
            # propose move: transfer delta from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # sample delta as fraction of a_hc[i]
            delta = a_hc[i] * rng.random() * 0.05
            if delta <= 0:
                continue
            cand = a_hc.copy()
            cand[i] -= delta
            cand[j] += delta
            # ensure nonnegativity
            if cand[i] < 0:
                continue
            cand_sum = cand.sum()
            cand = cand / cand_sum
            sc = fast_score(cand, fft_len, n)
            # accept if better or with prob exp(-(sc - best_hc_score)/temp)
            if sc < best_hc_score:
                a_hc = cand
                best_hc_score = sc
                best_overall = cand.copy()
                best_overall_score = sc
                # print progress
                print(f"[hill] it={it}, new best {best_hc_score:.8f}")
            else:
                # metropolis
                temp = temp0 * (1 - it / max_iter)
                if rng.random() < np.exp(-(sc - best_hc_score) / max(temp,1e-9)):
                    a_hc = cand
            it += 1

    # final evaluate sequence with official evaluate_sequence
    final_seq = best_overall.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The search works in three stages (within the 1000 s wall‑clock limit).

1. **Rich initial pool** – deterministic windows (Gaussian, Kaiser, DPSS,
   Hann, Blackman, triangular) with several parameter settings, several
   random Dirichlet draws and, if available, the previous best construction.
   The pool is scored with an exact fast convolution (FFT) and the best
   candidate is kept.

2. **Exponentiated‑gradient descent on the exact ℓ∞‑norm** – using the
   true sub‑gradient of `max(conv(a,a))`.  The update
   `a_i ← a_i·exp(‑η·g_i)` keeps the vector on the probability simplex.
   The learning rate decays as `η₀/√t`.  Every few thousand iterations the
   current iterate is mixed with a fresh random vector to escape possible
   basins.  The best iterate observed during this phase is retained.

3. **Simulated‑annealing hill‑climb** – tiny random mass‑transfer moves
   (`δ` taken from a donor index and added to a receiver index) are tried
   greedily; worsening moves are occasionally accepted with a temperature
   schedule.  This fine‑tunes the solution.

The final sequence is returned as a plain list of floats; the official
`evaluate_sequence` will internally rescale it, therefore we keep the sum
normalized to one throughout the search.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.signal.windows import dpss  # type: ignore

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))

    The convolution is performed via FFT for speed.
    """
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex."""
    # Use a mixture of simple distributions to increase diversity.
    mode = rng.integers(0, 5)
    if mode == 0:                     # uniform
        a = np.full(n, 1.0 / n)
    elif mode == 1:                   # exponential decay
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 2:                   # power‑law
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 3:                   # beta‑distribution like
        a = rng.beta(0.5, 0.5, size=n)
    else:                             # edge‑biased (few large entries)
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes `evaluate_sequence`.
    The implementation follows the three‑stage pipeline described in the
    module docstring.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we keep the proven size n = 4096 (fast enough)
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse initial pool
    # ------------------------------------------------------------------
    candidates = []

    # deterministic windows with a few parameter sweeps
    for sigma in np.linspace(0.12, 0.40, 5):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    for beta in (4.0, 6.0, 8.0, 10.0, 12.0):
        candidates.append(kaiser_init(n, beta=beta))

    for NW in (2.5, 3.0, 3.5, 4.0, 4.5):
        candidates.append(dpss_init(n, NW=NW))

    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(triangular_init(n))

    # random draws
    for _ in range(20):
        candidates.append(random_initial(n, rng))

    # reuse previous solution if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 /= a0.sum()
                candidates.append(a0)

    # ------------------------------------------------------------------
    # Pick the best by the exact fast score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent on the exact max‐conv
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * budget_s
    a_curr = best_a.copy()
    best_overall = best_a.copy()
    best_overall_score = best_score

    t_iter = 0
    while time.time() < phase1_end:
        t_iter += 1

        # convolution (full length 2n‑1)
        fa = rfft(a_curr, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]

        # index of the peak
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a_i:
        #   g_i = 2 * a_{k_max‑i}  for indices that are in range
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a_curr)
        grad[mask] = 2.0 * a_curr[idx[mask]]

        # learning rate schedule (η₀ = 0.8 works well in practice)
        eta = 0.8 / np.sqrt(t_iter)

        # multiplicative EG update + simplex re‑normalisation
        a_curr = a_curr * np.exp(-eta * grad)
        s = a_curr.sum()
        if s <= 0 or not np.isfinite(s):
            a_curr = np.full_like(a_curr, 1.0 / n)
        else:
            a_curr /= s

        # evaluate
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall = a_curr.copy()
            print(f"[EG] t={t_iter:,}  new best = {best_overall_score:.8f}")

        # every few thousand iterations inject fresh randomness to avoid stagnation
        if t_iter % 5000 == 0:
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.30
            a_curr = (1.0 - alpha) * a_curr + alpha * rand_vec
            a_curr /= a_curr.sum()

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing hill‑climb (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # remaining time is used for hill‑climb
        a_hc = best_overall.copy()
        best_hc_score = best_overall_score

        it = 0
        max_iter = 300_000            # generous upper bound; loop stops on deadline
        delta_frac = 0.04             # maximal fraction of donor mass to move
        temp0 = 0.12                  # initial temperature for Metropolis acceptance

        while time.time() < deadline and it < max_iter:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # propose moving a small amount of mass from i to j
            delta = a_hc[i] * rng.random() * delta_frac
            if delta <= 0.0:
                continue

            cand = a_hc.copy()
            cand[i] -= delta
            cand[j] += delta
            # non‑negativity safeguard
            if cand[i] < 0:
                continue
            cand /= cand.sum()               # stays on the simplex

            sc = fast_score(cand, fft_len, n)

            # greedy acceptance
            if sc < best_hc_score:
                a_hc = cand
                best_hc_score = sc
                best_overall = cand.copy()
                best_overall_score = sc
                print(f"[hill] it={it:,}  new best = {best_hc_score:.8f}")
            else:
                # Metropolis rule with a simple linear cooling schedule
                temp = temp0 * (1.0 - it / max_iter)
                if rng.random() < np.exp(-(sc - best_hc_score) / max(temp, 1e-12)):
                    a_hc = cand

            it += 1

    # ------------------------------------------------------------------
    # Final evaluation (official function) and return
    # ------------------------------------------------------------------
    final_seq = best_overall.tolist()
    final_eval = evaluate_sequence(final_seq)   # official scoring
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
