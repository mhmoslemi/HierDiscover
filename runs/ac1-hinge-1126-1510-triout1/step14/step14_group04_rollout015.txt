"""
Improved step‑function search.

The algorithm builds a rich initial pool (including a parametrised Gaussian
window), refines the best candidate with several stages of optimisation and
adds a few extra tricks compared with the reference implementation:

* **Gaussian pre‑search** – a one‑dimensional sweep over the Gaussian width
  (σ as a fraction of the length) often lands extremely close to the
  theoretical optimum (≈ 1.5).  The best Gaussian is used as a starting
  point for the later gradient‑based phases.

* **Symmetrisation** – after every update the vector is forced to be
  symmetric (a_i = a_{n‑1‑i}); the optimum is known to be symmetric,
  and this reduces the search space by a factor of two.

* **Longer L‑BFGS schedule** – a finer geometric λ‑schedule and more
  iterations per λ give the quasi‑Newton method more time to converge.

* **Extended gradient phases** – the EG, sub‑gradient and Adam phases run
  longer and use a slightly larger “top fraction” for the mask.

* **Final smoothing / hill‑climb** – a cheap low‑pass filter (moving‑average)
  is applied repeatedly; each smoothing step is accepted only if it
  lowers the exact score.  This removes tiny high‑frequency ripples that
  can cause a marginal increase of the max‑convolution.

All phases respect the global time budget (≈ 1000 s).  The returned
sequence is normalised to sum = 1, non‑negative and is finally scored
with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric (a_i = a_{n-1-i})."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Fast exact objective (sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    x → a = softmax(y)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def gaussian_init(n: int, sigma_frac: float) -> np.ndarray:
    """
    Gaussian window centred at (n‑1)/2,
    σ = sigma_frac * n.
    """
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS runs for each λ in lam_schedule, starting each stage from the
    previous result.  Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        a = symmetrize(a)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.015):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Only the top `top_fraction` entries receive a non‑zero weight.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)

        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = symmetrize(a)
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[EG] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses an exponentiated‑gradient style update on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # mask of entries that are (almost) maximal
        tol = 1e-12
        mask = (conv >= max_c - tol).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 0.7 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = symmetrize(a)
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[SubGrad] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits (a = softmax(y)) so that the simplex constraint
    is satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)
        a = symmetrize(a)
        a /= a.sum()

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        a_curr = symmetrize(a_curr)
        a_curr /= a_curr.sum()
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Pairwise mass‑transfer SA (always symmetrised after a move).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = symmetrize(cand)
        cand /= cand.sum()

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Greedy improvement with tiny pairwise transfers (symmetrised).
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = symmetrize(cand)
        cand /= cand.sum()
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = cand
        it += 1
    return best, best_sc


def smoothing_pass(a: np.ndarray,
                   fft_len: int,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   kernel_width: int = 7):
    """
    Apply a small moving‑average filter repeatedly.
    Each accepted smoothing must lower the exact score.
    """
    a_cur = a.copy()
    best_sc = fast_score(a_cur, fft_len, n)
    kernel = np.ones(kernel_width, dtype=np.float64) / kernel_width

    while time.time() < deadline:
        cand = np.convolve(a_cur, kernel, mode='same')
        cand = symmetrize(cand)
        cand /= cand.sum()
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            a_cur = cand
        else:
            # stop when no improvement for a few attempts
            break
    return a_cur, best_sc


# ----------------------------------------------------------------------
# Gaussian width search (cheap 1‑D scan)
# ----------------------------------------------------------------------


def gaussian_width_search(n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          deadline: float) -> np.ndarray:
    """
    Scan sigma_frac in [0.15, 0.55] with a simple golden‑section style.
    Returns the Gaussian (symmetrised and normalised) with lowest fast_score.
    """
    lower, upper = 0.15, 0.55
    phi = (np.sqrt(5) - 1) / 2   # 0.618...
    best_a = None
    best_sc = np.inf

    # initialise two interior points
    x1 = lower + (1 - phi) * (upper - lower)
    x2 = lower + phi * (upper - lower)

    while time.time() < deadline:
        a1 = gaussian_init(n, x1)
        a1 = symmetrize(a1)
        a1 /= a1.sum()
        sc1 = fast_score(a1, fft_len, n)

        a2 = gaussian_init(n, x2)
        a2 = symmetrize(a2)
        a2 /= a2.sum()
        sc2 = fast_score(a2, fft_len, n)

        if sc1 < sc2:
            upper, x2 = x2, x1
            x1 = lower + (1 - phi) * (upper - lower)
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1
        else:
            lower, x1 = x1, x2
            x2 = lower + phi * (upper - lower)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2

        # stop when interval is sufficiently small
        if upper - lower < 0.001:
            break

    return best_a if best_a is not None else np.full(n, 1.0 / n)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power‑of‑two FFT length works best)
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            a0 = symmetrize(a0)
            a0 /= a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(chebwin_init(n, attenuation=rng.uniform(40.0, 80.0)))
    candidates.append(dpss_init(n, half_bw=rng.uniform(1.5, 5.0)))

    # Gaussian family – we will also run a cheap 1‑D search later
    candidates.append(gaussian_init(n, sigma_frac=0.35))

    # Random families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families (including U‑shapes)
    for _ in range(8):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # ------------------------------------------------------------------
    # Choose the best among the pool (exact fast_score, sum=1 enforced)
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_a = symmetrize(best_a)
    best_a /= best_a.sum()
    best_sc = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        a = symmetrize(cand)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – quick Gaussian width sweep (makes a strong start)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gauss_deadline = time.time() + 0.07 * budget_s
        print("[Gaussian sweep] start")
        gauss_candidate = gaussian_width_search(n, fft_len, rng, gauss_deadline)
        sc_g = fast_score(gauss_candidate, fft_len, n)
        if sc_g < best_sc:
            best_sc = sc_g
            best_a = gauss_candidate.copy()
            print(f"[Gaussian sweep] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements (aggressive)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase1_end = start_time + 0.30 * budget_s
        lam_schedule = np.geomspace(5e-1, 5e-6, num=55)   # finer schedule
        maxiter_per_lam = 2500

        while time.time() < phase1_end:
            # random convex combination of current best and a fresh random start
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.30
            a_start = (1.0 - alpha) * best_a + alpha * rand_vec
            a_start = symmetrize(a_start)
            a_start /= a_start.sum()
            a_opt, sc_opt = lbfgs_continuation(a_start,
                                               lam_schedule,
                                               maxiter_per_lam,
                                               phase1_end,
                                               fft_len, fft_len, n)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.17 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.012)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.17 * budget_s
        print("[SubGrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                               sub_deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[SubGrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=200000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Smoothing pass (remove tiny ripples)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        smooth_deadline = time.time() + min(12.0, deadline - time.time())
        print("[Smooth] start")
        best_a, best_sc = smoothing_pass(best_a,
                                         fft_len,
                                         n,
                                         smooth_deadline,
                                         rng,
                                         kernel_width=9)
        print(f"[Smooth] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + min(20.0, deadline - time.time())
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=100000,
                                     delta_frac=0.025)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final sanity‑check & return
    # --------------------------------------------------------------
    final_seq = np.clip(best_a, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
