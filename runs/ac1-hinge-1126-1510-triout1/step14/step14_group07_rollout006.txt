"""
LP‑based iterative improvement of a step function.

The algorithm works on a probability vector a (non‑negative, sum = 1).
For a given a we compute the current worst self‑convolution value
    M = max_k (a * a)[k].
We then solve a linear program:
    maximize   Σ b_j
    subject to A·b ≤ M·1,   b ≥ 0,
where A is the (2 n‑1)×n convolution matrix such that A·b = a * b.
The optimizer returns a vector b with the largest possible mass while
still keeping the mixed convolution a*b below the current worst value M.

After scaling b to a probability vector g we move a a little towards g:
    a ← (1‑t)·a + t·g      (t ≈ 0.25‑0.5)

The move never increases the worst self‑convolution and usually
strictly decreases it.  The process is repeated until the time budget
expires.  A final short greedy hill‑climb polishes the result.

The whole search runs in a single call to ``propose_candidate`` and
produces a list of floats that is fed to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from math import ceil

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_convolution(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return max value of the self‑convolution of a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Initialisers – all return a probability vector of length n
# ----------------------------------------------------------------------


def arcsine_init(n: int) -> np.ndarray:
    """Discrete arcsine window."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Conv. matrix builder (fixed pattern, only data change each iteration)
# ----------------------------------------------------------------------


def build_conv_matrix_pattern(n: int):
    """
    Pre‑compute row/col index arrays for the (2·n‑1)×n convolution matrix.
    The actual numeric data are filled later as ``np.repeat(a, n)``.
    """
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.repeat(i, n)
    L = 2 * n - 1
    return rows, cols, L


# ----------------------------------------------------------------------
# LP step – given current a and its worst convolution value M,
# compute a new candidate g.
# ----------------------------------------------------------------------


def lp_step(a: np.ndarray, M: float, rows, cols, L: int):
    """
    Solve
        maximise Σ b_j
        subject to A·b ≤ M,  b ≥ 0
    where A is the convolution matrix for a.
    Returns the scaled probability vector g.
    """
    n = a.shape[0]
    # data = a repeated for each column
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    ub = np.full(L, M, dtype=np.float64)

    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if res.status != 0:
        # LP failed – signal by returning None
        return None
    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    # Scale to a probability vector (sum = 1)
    g = b / s
    return g


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass shuffles)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, fft_len: int, n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300_000,
               delta_frac: float = 0.12) -> np.ndarray:
    """Strictly improving greedy hill‑climb."""
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 2500 provides a good trade‑off between quality
    # and runtime (≈6.2 M non‑zeros in the LP matrix).
    # --------------------------------------------------------------
    n = 2500
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise the vector (arcsine window is a strong start)
    # --------------------------------------------------------------
    a = arcsine_init(n)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Pre‑compute the sparse matrix pattern for the convolution matrix
    # --------------------------------------------------------------
    rows, cols, L = build_conv_matrix_pattern(n)

    # --------------------------------------------------------------
    # Time handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Main LP‑based improvement loop
    # --------------------------------------------------------------
    iteration = 0
    max_lp_iters = 30                     # generous but safe bound
    while time.time() < deadline and iteration < max_lp_iters:
        M = max_convolution(best_a, fft_len, n)

        # Solve the LP to get a new feasible vector g
        g = lp_step(best_a, M, rows, cols, L)
        if g is None:
            # LP failed – break out and fall back to random restarts
            print("[LP] iteration {} failed – aborting LP loop".format(iteration))
            break

        # Try a few mixing coefficients; pick the first that improves
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1):
            candidate = (1.0 - t) * best_a + t * g
            sc = fast_score(candidate, fft_len, n)
            if sc < best_score - 1e-12:
                best_a = candidate
                best_score = sc
                improved = True
                print(f"[LP {iteration}] t={t:.2f} fast_score={best_score:.8f}")
                break

        if not improved:
            # No mixing improved the score – maybe we are stuck.
            # Perform a small random perturbation as a restart seed.
            rand_vec = random_dirichlet(n, rng)
            # blend a bit with random vector and continue
            best_a = 0.9 * best_a + 0.1 * rand_vec
            best_a /= best_a.sum()
            best_score = fast_score(best_a, fft_len, n)
            print(f"[restart] random blend – fast_score={best_score:.8f}")

        iteration += 1

    # --------------------------------------------------------------
    # Final greedy polishing (uses remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a = hill_climb(best_a, fft_len, n, deadline, rng,
                            max_iter=500_000, delta_frac=0.12)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final hill‑climb] fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
