"""
A multi‑phase optimizer for the non‑negative step‑function problem.

The algorithm builds a diverse pool of initial designs, then performs
large‑scale global optimisation with L‑BFGS on a smoothly‑approximated
max‑convolution, followed by several exact (sub‑gradient / Adam) refinements.
A new LP refinement step mixes the current design with a mass‑maximising
candidate obtained from a tiny linear program; the mixing factor is
selected by a cheap line‑search.

After the LP phase a short simulated‑annealing / hill‑climb polish is applied.
All phases respect the global 1000 s deadline; the best sequence found is
returned and finally evaluated with the provided `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective (sum of coeffs is 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f   = 2·n·smooth_max(conv)
       grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, starting each stage
    from the previous result. Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.018):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Only the top `top_fraction` convolution entries receive a non‑zero weight.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)

        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[EG] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses an exponentiated‑gradient style update on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        tol = 1e-12
        mask = (conv >= max_c - tol).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 0.7 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[SubGrad] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits (a = softmax(y)) so that the simplex constraint
    is satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional noise to escape stagnation
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 120000):
    """
    Cheap SA with pairwise mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def mix_with_best_alpha(f: np.ndarray,
                        g: np.ndarray,
                        alphas,
                        fft_len,
                        n,
                        cur_best_sc):
    """
    Given current design `f` and candidate `g`,
    try a few mixing coefficients and keep the best.
    Returns (new_f, new_sc).
    """
    best_f = f.copy()
    best_sc = cur_best_sc
    for a in alphas:
        mixed = (1.0 - a) * f + a * g
        sc = fast_score(mixed, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_f = mixed.copy()
    return best_f, best_sc


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              max_active: int = 30,
              init_step: float = 0.2):
    """
    Repeatedly solve a tiny LP that maximises Σb under the active
    convolution constraints f * b ≤ M, where M is the current peak.
    The solution b is mixed into the current design f using a short
    line‑search over a few step sizes.
    """
    f = a_start.copy()
    best_f = f.copy()
    best_sc = fast_score(best_f, fft_len, n)

    # a small set of mixing factors for the line‑search
    mix_alphas = [0.02, 0.05, 0.1, 0.2, 0.35]

    while time.time() < deadline:
        # current convolution and peak
        fa = np.fft.rfft(f, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active constraints: indices with values close to the peak
        if max_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            topk = max_active
            active_idx = np.argpartition(-conv, topk - 1)[:topk]

        # Build sparse matrix A (rows = active constraints)
        rows = []
        cols = []
        data = []

        f_rev = f[::-1]
        for r, k in enumerate(active_idx):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                b_idx = k - j
                coeff = f_rev[j]
                rows.append(r)
                cols.append(b_idx)
                data.append(coeff)

        A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_ub = np.full(len(active_idx), M)

        c = -np.ones(n)                     # maximise Σb  → minimise -Σb
        bounds = [(0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print("[LP] linprog failed – terminating LP refinement")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b                     # normalise to simplex

        # line‑search over a few α values
        f, sc = mix_with_best_alpha(f, g, mix_alphas,
                                    fft_len, n, best_sc)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[LP] new best fast_score={best_sc:.8f}")

    return best_f, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it
    prev_seq = globals().get('height_sequence_1')
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        prev = np.asarray(prev_seq, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # Randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families
    for _ in range(4):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # keep the best candidate from the pool
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split (seconds)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.4  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Global L‑BFGS search (≈30 % of time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = np.geomspace(0.4, 2e-6, num=35)   # go closer to exact max
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # blend best with a fresh random start to keep diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement (≈12 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.12 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.02)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (≈10 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.10 * budget_s
        print("[SubGrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                               sub_deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[SubGrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimiser (≈15 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Linear‑programming refinement (≈8 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(35.0, deadline - time.time())
        if time.time() < lp_deadline:
            print("[LP] start")
            best_a, best_sc = lp_refine(best_a,
                                        lp_deadline,
                                        n,
                                        fft_len,
                                        max_active=40,
                                        init_step=0.25)
            print(f"[LP] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Simulated annealing fine‑tuning (≈7 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=100000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Greedy hill‑climb polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=50000,
                                     delta_frac=0.035)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Optional final mixing with the uniform distribution (tiny tweak)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    mix_alphas = [0.005, 0.01, 0.02]
    best_a, best_sc = mix_with_best_alpha(best_a, uniform,
                                           mix_alphas,
                                           fft_len,
                                           n,
                                           best_sc)
    print(f"[final‑mix] best fast_score after uniform mix={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
