"""
A multi‑phase optimiser for the non‑negative step‑function problem.

Strategy
--------
1. generate a rich pool of candidates (window families, Dirichlet draws,
   linear mixes) and keep the best according to the fast objective
   `2·n·max(conv(a,a))` (the official score for a normalised vector).

2. Refine the best candidate with a continuation of L‑BFGS on a
   smooth log‑sum‑exp approximation of the max‑convolution.  A geometric
   schedule of smoothing parameters is used and a few random restarts
   are performed.

3. Perform a directed stochastic local search that repeatedly
   identifies the current maximal convolution entry, computes the
   contributing pairwise products, and transfers a small amount of mass
   from the most contributing index to a low‑impact index.
   Improvements are kept.

4. Finish with a long simulated‑annealing / greedy hill‑climb phase that
   makes cheap pairwise mass‑transfer moves with an exponential cooling
   schedule.

All phases respect the global time budget (≈ 1000 s).  The final
vector is normalised to sum 1 (the score is invariant to scaling) and
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact fast objective (used throughout the search)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex (∑a=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    return np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    The variable y are logits; a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic / random families)
# ----------------------------------------------------------------------


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = windows.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attenuation: float = 60.0) -> np.ndarray:
    a = windows.chebwin(n, at=attenuation)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, half_bw: float = 3.0) -> np.ndarray:
    a = windows.dpss(n, half_bw, sym=True)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def power_init(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)"""
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dirichlet_init(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Dirichlet with concentration alpha."""
    a = rng.gamma(alpha, size=n)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Diverse random draws on the simplex.
    Mixes several simple families.
    """
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation on smooth max‑convolution
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule` starting from the previous result.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Directed, sub‑gradient style local search
# ----------------------------------------------------------------------


def directed_local_search(a: np.ndarray,
                          fft_len: int,
                          n: int,
                          deadline: float,
                          rng: np.random.Generator):
    """
    Repeatedly reduces the maximum convolution entry by moving a tiny
    amount of mass from the most responsible index to a low‑impact one.
    """
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    iteration = 0
    while time.time() < deadline:
        # current convolution and its maximising lag
        conv = full_conv(best_a, fft_len)
        k_max = int(conv.argmax())
        max_val = float(conv.max())

        # compute contribution w_i = a_i * a_{k_max-i} where defined
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(best_a)
        w[mask] = best_a[i[mask]] * best_a[j[mask]]

        # index with largest contribution
        i_src = int(np.argmax(w))

        # choose a target with small contribution (random among bottom 30%)
        thresh = np.percentile(w, 30)
        low_idxs = np.where(w <= thresh)[0]
        if low_idxs.size == 0:
            i_tgt = rng.integers(0, n)
        else:
            i_tgt = int(rng.choice(low_idxs))

        if i_src == i_tgt:
            continue

        # transfer a modest fraction of the source mass
        delta = best_a[i_src] * rng.random() * 0.08  # up to 8 %
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            print(f"[directed] iter={iteration}, fast_score={best_sc:.8f}")

        iteration += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing / greedy hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Pairwise mass‑transfer SA with an exponential cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best candidate found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0. Initial pool – deterministic families, parameter sweeps, random draws
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # uniform, a safe starter
    best_sc = fast_score(best_a, fft_len, n)

    def try_candidate(a):
        nonlocal best_a, best_sc
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[init] new best fast_score={best_sc:.8f}")

    # a) previous solution (if any)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            try_candidate(a0)

    # b) deterministic windows with coarse grids
    for beta in np.linspace(4.0, 12.0, 17):
        try_candidate(kaiser_init(n, beta))
    for att in np.linspace(30.0, 80.0, 11):
        try_candidate(chebwin_init(n, att))
    for bw in np.linspace(1.5, 5.0, 15):
        try_candidate(dpss_init(n, bw))
    try_candidate(hann_init(n))
    try_candidate(blackman_init(n))

    # c) power‑law family
    for p in np.linspace(0.5, 2.5, 21):
        try_candidate(power_init(n, p))

    # d) random draws (Dirichlet‑like)
    for _ in range(600):
        # symmetric Dirichlet with random concentration
        alpha = rng.uniform(0.2, 4.0)
        try_candidate(dirichlet_init(n, alpha, rng))

    # e) mixtures of two deterministic windows
    base_funs = [
        lambda: hann_init(n),
        lambda: blackman_init(n),
        lambda: kaiser_init(n, rng.uniform(5.0, 11.0)),
        lambda: chebwin_init(n, rng.uniform(40.0, 80.0)),
        lambda: dpss_init(n, rng.uniform(1.5, 5.0)),
    ]
    base_windows = [fun() for fun in base_funs]
    for i in range(len(base_windows)):
        for j in range(i + 1, len(base_windows)):
            w_i, w_j = base_windows[i], base_windows[j]
            for alpha in np.linspace(0.1, 0.9, 9):
                mix = alpha * w_i + (1.0 - alpha) * w_j
                mix = np.clip(mix, 0.0, None)
                mix /= mix.sum()
                try_candidate(mix)

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement (aggressive, multiple restarts)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = np.geomspace(5e-1, 5e-7, num=25)   # finer schedule
    maxiter_per_lam = 250

    while time.time() < phase1_end:
        # random blend of the current best and a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Directed sub‑gradient‑style local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = time.time() + 0.25 * budget_s
        print("[directed] start")
        best_a, best_sc = directed_local_search(best_a,
                                                fft_len,

        print(f"[directed] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (broad exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + 0.20 * budget_s
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,

                                              fft_len,
                                              rng,
                                              init_temp=6e-4,

        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Greedy hill‑climb (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=120000,
                                     delta_frac=0.03)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final normalisation & official evaluation

    final_seq = np.clip(best_a, 0.0, None)
    if final_seq.sum() == 0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / final_seq.sum()

    final_seq_list = final_seq.tolist()
    final_eval = evaluate_sequence(final_seq_list)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq_list
