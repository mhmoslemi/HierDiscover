import cvxpy as cp

def solve_lp_max_sum_under_conv_bound(f, M):
    n = len(f)
    L = 2*n - 1
    A = toeplitz_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [A @ g <= M,
                   g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-6, reltol=1e-6)
    # or use solver=cp.OSQP if faster.
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0) # ensure nonnegative

def solve_lp_max_sum_under_conv_bound(f, M):
    n = len(f)
    Lc = 2*n - 1
    # Build sparse matrix A
    data = []
    rows = []
    cols = []
    for d in range(n):
        # row indices = d + np.arange(n)
        rows.extend(d + np.arange(n))
        cols.extend(np.arange(n))
        data.extend([f[d]] * n)
    # create CSR
    A = sparse.csr_matrix((data, (rows, cols)), shape=(Lc, n))
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds (0, None) for each var
    bounds = [(0, None)] * n
    # constraints: A @ g <= M (vector)
    # However linprog expects A_ub @ x <= b_ub with shape (m, n)
    # So we pass A matrix, b = M * np.ones(Lc)
    res = linprog(c, A_ub=A, b_ub=np.full(Lc, M), bounds=bounds,
                  method='highs', options={'time_limit': max_seconds})
    if res.success:
        g = res.x
        return g
    else:
        return None

"""
Propose a low‑score step sequence.

The algorithm builds a diverse set of initial probability vectors (standard
windows, random Dirichlet draws, generalized‑Gaussian shapes, etc.), picks
the best one according to the exact fast score, and then refines it with a
combination of:

* a finely‑scheduled L‑BFGS optimisation on a log‑sum‑exp smoothed
  max‑convolution objective,
* short “Adam” – style runs,
* a few random restarts,
* a final greedy hill‑climb with many mass‑transfer moves.

All work is done inside the time budget (default 1000 s).  The returned
list is a non‑negative step sequence whose evaluation with
``evaluate_sequence`` is expected to be lower than the previous best
(~1.50594) and hopefully below the target 1.503.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import minimize, differential_evolution, linprog
from scipy import signal

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ------------------------------------------------------------

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution. Returns (obj, grad) w.r.t. logits y."""
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                 # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: ∂conv/∂a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def dirichlet_init(n: int, rng: np.random.Generator,
                   concentration: float) -> np.ndarray:
    """Dirichlet draw with equal concentration parameter."""
    alpha = np.full(n, concentration)
    a = rng.gamma(alpha, 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def generalized_gaussian_init(n: int, p: float,
                              sigma_frac: float) -> np.ndarray:
    """Shape a_i ∝ exp(-| (i‑centre)/σ |^p)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-np.abs((i - centre) / sigma) ** p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def scipy_window_init(name: str, n: int) -> np.ndarray:
    """Return a window from scipy.signal.get_window, normalised to sum = 1."""
    try:
        w = signal.get_window(name, n, fftbins=False)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        return w
    except Exception:
        return uniform_init(n)


def beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Beta density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_mixture_shape(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    a1 = beta_shape(n, p1, p1)
    a2 = beta_shape(n, p2, p2)
    a = w * a1 + (1.0 - w) * a2
    return a

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule, keeping the best vector.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 40000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 500000,
               delta_frac: float = 0.10):
    """Greedy hill‑climb that only accepts strictly improving mass‑transfer moves."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Assemble an initial pool of candidates.
    # ------------------------------------------------------------------
    candidates = []

    # 1 – use any previous solution if available

        if prev.ndim == 1 and prev.size > 0:

            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else uniform_init(n)
            candidates.append(a0)

    # 2 – deterministic standard windows
    candidates.extend([
        uniform_init(n),
        hann_init(n),
        blackman_init(n),
        kaiser_init(n, beta=9.0),
        gaussian_init(n, sigma_factor=0.3),
        triangular_init(n),
    ])

    # 3 – a few extra SciPy windows that were not used before

                 'flattop']:
        candidates.append(scipy_window_init(name, n))

    # 4 – random Dirichlet draws with various concentrations
    for _ in range(8):
        conc = 10.0 ** rng.uniform(-2, 2)   # between 0.01 and 100
        candidates.append(dirichlet_init(n, rng, concentration=conc))

    # 5 – random mixture families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # 6 – generalized‑Gaussian family with random parameters
    for _ in range(6):
        p = rng.uniform(0.5, 6.0)
        sigma_frac = rng.uniform(0.05, 0.45)
        candidates.append(generalized_gaussian_init(n, p, sigma_frac))

    # 7 – low‑dimensional Beta / mixture families (DE optimisation)
    # single Beta
    def beta_obj(params):
        p, q = params
        a = beta_shape(n, p, q)
        return fast_score(a, fft_len, n)

    bounds_beta = [(0.1, 2.5), (0.1, 2.5)]
    res_beta = differential_evolution(beta_obj,
                                      bounds=bounds_beta,
                                      maxiter=120,
                                      popsize=12,
                                      seed=int(rng.integers(1e9)),
                                      polish=False,
                                      updating='deferred')
    candidates.append(beta_shape(n, *res_beta.x))

    # Beta mixture
    def beta_mix_obj(params):
        p1, p2, w = params
        a = beta_mixture_shape(n, p1, p2, w)
        return fast_score(a, fft_len, n)

    bounds_mix = [(0.1, 2.5), (0.1, 2.5), (0.0, 1.0)]
    res_mix = differential_evolution(beta_mix_obj,
                                     bounds=bounds_mix,
                                     maxiter=120,
                                     popsize=12,
                                     seed=int(rng.integers(1e9)),
                                     polish=False,
                                     updating='deferred')
    candidates.append(beta_mixture_shape(n, *res_mix.x[:2], res_mix.x[2]))

    # ------------------------------------------------------------------
    # Pick the best start according to the exact fast_score.
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 8000   # a bit more than before

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – a few random restarts (≈10 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    restarts = 3
    while time.time() < phase2_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        # blend a little bit of randomness into the current best
        alpha = rng.random() * 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase2_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 3 – Adam (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase3_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.05,
                                           max_iter=30000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=800000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
