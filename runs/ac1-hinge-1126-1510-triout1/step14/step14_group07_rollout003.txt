    i = np.arange(n, dtype=np.float64)

"""
A focused optimisation for the maximising‑convolution problem.

Main ideas
----------
* Build a diverse pool of initial windows (classical, Gaussian, Kaiser,
  arcsine, and a set of symmetric Beta distributions with varying
  concentration) together with a few random Dirichlet draws.
* Pick the best start according to the exact fast_score
  (2·n·max(self‑convolution) for a probability vector).
* Apply a long run of exact‑max sub‑gradient descent (the same update as
  the earlier implementation but with a larger step‑size and many more
  iterations) to quickly drive the maximum of the convolution down.
* Finish with a *targeted* greedy hill‑climb: at each step the index that
  contributes most to the current maximal convolution entry is identified
  and a small amount of mass is transferred to an index that contributes
  the least to that entry.  This directly attacks the bottleneck and is
  much more effective than a blind random hill‑climb.
* All work respects the 1000 s budget and the data‑validation rules of
  ``evaluate_sequence``.  The final sequence is returned as a plain list
  of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_fft_len(n: int) -> int:
    """FFT length used for all convolution‑related transforms."""
    return next_pow2(2 * n - 1)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (2·n·max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def full_convolution(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full (linear) self‑convolution of a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Initialisation families (return probability vectors of length n)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a point on the simplex via Gamma draws."""
    alpha = rng.uniform(0.05, 5.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a

def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         step0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent (many iterations, decreasing step).
    Returns the best vector seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and index of the current maximum
        conv = full_convolution(a, fft_len, n)
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        # overlapping range for the index
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # contributions a[idx - i] for i in [j_low, j_high]
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = sub
        # the special i = idx/2 term (double contribution)
        if idx % 2 == 0:
            i_mid = idx // 2
            if 0 <= i_mid < n:
                g[i_mid] += a[i_mid]
        g *= 2.0  # each pair counted twice

        # zero‑sum projected sub‑gradient (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        # tentative update + projection onto simplex
        a_new = simplex_projection(a + alpha * d)

        sc_new = fast_score(a_new, fft_len, n)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        # always move (helps to escape plateaus)
        a = a_new
        it += 1
    return best_a, best_sc


def targeted_hill_climb(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        eps_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current maximal convolution entry.
    A small fraction (eps_frac) of the mass at the worst‑contributing index
    is transferred to the least‑contributing index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    # inactivity counter – after many failed attempts we randomise a little
    fail_counter = 0
    while time.time() < deadline:
        conv = full_convolution(a, fft_len, n)
        idx = int(conv.argmax())

        # range of i that actually contributes to conv[idx]
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        i_range = np.arange(low, high + 1, dtype=np.int64)
        partner = a[idx - i_range]                     # a_{idx-i}
        contributions = a[i_range] * partner           # product a_i * a_{idx-i}
        i_max_local = i_range[np.argmax(contributions)]

        # partner values for *all* positions (0 if out of range)
        partner_all = np.zeros(n, dtype=np.float64)
        j_low = low
        j_high = high
        partner_all[j_low:j_high + 1] = a[idx - np.arange(j_low, j_high + 1)]

        # choose a destination with the smallest partner value
        j_min = int(np.argmin(partner_all))
        if j_min == i_max_local:
            # pick the second smallest if the absolute minimum coincides
            sorted_idx = np.argsort(partner_all)
            for cand in sorted_idx:
                if cand != i_max_local:
                    j_min = int(cand)
                    break

        # amount to move – a small fraction of the source mass
        epsilon = eps_frac * a[i_max_local]
        if epsilon <= 0.0:
            # fallback to a random tiny move
            pos = rng.choice(np.where(a > 1e-12)[0])
            epsilon = eps_frac * a[pos]
            i_max_local = int(pos)

        # ensure we never make a negative entry
        epsilon = min(epsilon, a[i_max_local])

        # perform the transfer
        a[i_max_local] -= epsilon
        a[j_min] += epsilon

        sc = fast_score(a, fft_len, n)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            fail_counter = 0
        else:
            # revert – we only keep strictly improving moves
            a[i_max_local] += epsilon
            a[j_min] -= epsilon
            fail_counter += 1

        # occasional random jitter to escape stagnation
        if fail_counter >= 2000:
            # mix a little with a fresh Dirichlet draw
            rand = random_dirichlet(n, rng)
            mix = 0.97
            a = (mix * a + (1.0 - mix) * rand)
            a /= a.sum()
            fail_counter = 0
    return best_a, best_sc

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) with a low
    ``evaluate_sequence`` score.
    The routine follows the plan described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the same as the reference solution for fairness.
    # --------------------------------------------------------------
    n = 5000
    fft_len = compute_fft_len(n)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates.
    # --------------------------------------------------------------
    candidates = [
        uniform_init(n),
        triangular_init(n),
        hann_init(n),
        blackman_init(n),
        gaussian_init(n, sigma_factor=0.20),
        gaussian_init(n, sigma_factor=0.30),
        gaussian_init(n, sigma_factor=0.40),
        kaiser_init(n, beta=6.0),
        kaiser_init(n, beta=8.0),
        kaiser_init(n, beta=10.0),
        arcsine_init(n),
    ]

    # beta‑distribution family (symmetric)
    for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
        candidates.append(beta_init(n, alpha))

    # a handful of Dirichlet samples
    for _ in range(30):
        candidates.append(random_dirichlet(n, rng))

    # reuse a previous solution if it exists
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
                candidates.append(a0)

    # --------------------------------------------------------------
    # Choose the best start according to the exact fast_score.
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] best fast_score = {best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget handling.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – exact‑max sub‑gradient descent (≈40 % of budget).
    # --------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    if time.time() < phase1_end:
        best_a, best_sc = subgradient_optimize(best_a,
                                               phase1_end,
                                               n,
                                               fft_len,
                                               step0=0.65)
        print(f"[sub‑grad] best fast_score = {best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – aggressive targeted hill‑climb (remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = targeted_hill_climb(best_a,
                                              deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              eps_frac=0.04)
        print(f"[targeted hill‑climb] best fast_score = {best_sc:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
