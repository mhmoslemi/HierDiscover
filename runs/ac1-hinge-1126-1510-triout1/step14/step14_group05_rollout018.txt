def flatten_spectrum(a, steps, L, rng, verbose=False):
    best_a = a.copy()
    best_sc = fast_score(best_a, L, n)
    for it in range(steps):
        # compute current energy
        E = np.dot(a, a)
        # FFT on zero padded vector
        # pad to length L
        a_padded = np.concatenate([a, np.zeros(L - n, dtype=np.float64)])
        A = np.fft.rfft(a_padded, L)
        # Keep original phases
        phases = np.angle(A)
        # Set magnitude to sqrt(E)
        mag = np.sqrt(E)
        A_new = mag * (np.cos(phases) + 1j*np.sin(phases))
        a_full = np.fft.irfft(A_new, L)
        a = a_full[:n]
        # Non-negativity
        a = np.maximum(a, 0.0)
        # Renormalize sum to 1
        s = a.sum()
        if s == 0:
            # reinitialize random
            a = np.full(n, 1.0/n)
            continue
        a /= s
        # optionally add tiny random jitter to avoid stagnation
        if it % 10 == 0:
            # small random mixing
            rand = random_initial(n, rng) * 0.001
            a = a * 0.999 + rand
            a /= a.sum()
        # Evaluate fast_score
        sc = fast_score(a, L, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def alternating_projection(a_start, steps, n, L, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, L, n)
    for it in range(steps):
        # compute energy
        E = np.dot(a, a)
        # compute FFT of full length
        a_pad = np.concatenate((a, np.zeros(L - n, dtype=np.float64)))
        A = np.fft.rfft(a_pad, L)
        # phase
        phase = np.angle(A)
        # replace zero magnitude phases with random
        mag = np.abs(A)
        zero_idx = mag < 1e-14
        # random phases for those indices
        phase[zero_idx] = rng.random(size=zero_idx.sum()) * 2 * np.pi
        # set all magnitudes to sqrt(E)
        A_new = np.sqrt(E) * (np.cos(phase) + 1j*np.sin(phase))
        # Ensure DC component is real positive (phase zero)
        # rfft DC is index 0, Nyquist maybe at -1.
        A_new[0] = np.sqrt(E)  # real positive
        # If L is even, Nyquist freq at index L//2 has only real part; set to sqrt(E) (real)
        if L % 2 == 0:
            A_new[-1] = np.sqrt(E)  # real
        # Inverse FFT
        a_full = np.fft.irfft(A_new, L)
        # keep first n entries
        a_new = a_full[:n]
        # Project onto non-negative simplex
        # set negatives to 0, renorm to sum 1
        a_new = np.maximum(a_new, 0.0)
        sum_a = a_new.sum()
        if sum_a < 1e-12:
            # reinitialize uniform distribution to avoid division by zero
            a_new = np.full(n, 1.0 / n)
            sum_a = 1.0
        a = a_new / sum_a

        # optionally add small jitter every few iter to avoid stagnation
        if it % 50 == 0:
            # jitter: small random mixing
            jitter = random_initial(n, rng) * 0.001
            a = a * 0.999 + jitter
            a /= a.sum()

        # evaluate
        sc = fast_score(a, L, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # maybe print
        # we could break early if sc < some threshold, but continue
    return best_a, best_sc

def alternating_projection(a_start, steps, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    for it in range(steps):
        E = np.dot(a, a)
        # Pad
        a_pad = np.concatenate([a, np.zeros(fft_len - n, dtype=np.float64)])
        A = np.fft.rfft(a_pad, fft_len)
        # Determine phases
        mags = np.abs(A)
        phases = np.angle(A)
        # Randomize phases where magnitude close to zero
        mask = mags < 1e-14
        if np.any(mask):
            phases[mask] = rng.random(mask.sum()) * 2.0 * np.pi
        # Ensure DC component real
        phases[0] = 0.0
        # For even length FFT, Nyquist frequency must be real; ensure phase = 0
        if fft_len % 2 == 0:
            phases[-1] = 0.0
        # New coefficients
        new_mag = np.sqrt(E)
        A_new = new_mag * (np.cos(phases) + 1j * np.sin(phases))
        a_full = np.fft.irfft(A_new, fft_len)
        a = a_full[:n]
        # Project onto non-negative simplex: clip negatives, normalize by sum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            # reset to uniform
            a = np.full(n, 1.0 / n)
            s = 1.0
        a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

"""
Propose a low‑score step function.

Algorithm
---------
1. Build a diverse pool of initial windows (spectral, Hann, Gaussian, …) and
   keep the best one (fast_score = 2·n·max(conv) for a normalized sequence).

2. **Spectral flattening (alternating projection).**
   Repeatedly force the Fourier magnitude (except the DC term) to be constant
   while projecting the time‑domain signal onto the non‑negative simplex.
   This is an inexpensive way to lower the peak of the autocorrelation
   a*a.  Many short restarts are performed.

3. **L‑BFGS on a smoothed max.**
   With the best sequence from step 2 as start, run L‑BFGS over a
   log‑sum‑exp smooth approximation of the max‑convolution.  The λ‑schedule
   is geometric, each stage restarts from the previous result.

4. **Exact sub‑gradient descent** on the true max‑convolution (exponentiated‑
   gradient style).  This removes the smoothing bias.

5. **Adam optimisation** on the exact objective (logits → softmax).

6. **Greedy hill‑climb** for final polishing.

All phases respect a global 1000 s deadline; the best sequence found is
returned and evaluated with the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact fast objective (sum of coeffs is 1)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f   = 2·n·smooth_max(conv)
       grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage
    from the previous result. Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.01):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Only the top `top_fraction` convolution entries receive a non‑zero weight.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask of top entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient via reversed a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)

        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[EG] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses an exponentiated‑gradient style update on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        tol = 1e-12
        mask = (conv >= max_c - tol).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 0.7 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[SubGrad] it={it}, fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits (a = softmax(y)) so that the simplex constraint
    is satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# New Phase: Alternating‑projection spectral flattening
# ----------------------------------------------------------------------


def alternating_projection(a_start: np.ndarray,
                           steps: int,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator):
    """
    Repeatedly project onto:
      * constant Fourier magnitude (except DC)  – “flat spectrum”
      * non‑negative simplex in the time domain.
    Returns the best vector seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for it in range(steps):
        # total energy (sum of squares)
        E = np.dot(a, a)

        # FFT of zero‑padded vector
        a_pad = np.concatenate([a, np.zeros(fft_len - n, dtype=np.float64)])
        A = np.fft.rfft(a_pad, fft_len)

        # phases – randomise where magnitude is essentially zero
        mags = np.abs(A)
        phases = np.angle(A)
        zero_mask = mags < 1e-14
        if np.any(zero_mask):
            phases[zero_mask] = rng.random(zero_mask.sum()) * 2.0 * np.pi

        # enforce the DC term (sum of a) exactly
        S = a.sum()                     # should be 1.0 after normalization

        # build new spectrum
        k_len = A.shape[0]              # = fft_len//2 + 1
        A_new = np.empty_like(A, dtype=np.complex128)

        # DC component (real)
        A_new[0] = S

        # Nyquist component (real) if present
        if fft_len % 2 == 0:
            # index -1 corresponds to Nyquist frequency
            A_new[-1] = np.sqrt(E)
            if k_len > 2:
                A_new[1:-1] = np.sqrt(E) * (np.cos(phases[1:-1]) + 1j * np.sin(phases[1:-1]))
        else:
            if k_len > 1:
                A_new[1:] = np.sqrt(E) * (np.cos(phases[1:]) + 1j * np.sin(phases[1:]))

        # inverse transform and truncate
        a_full = np.fft.irfft(A_new, fft_len)
        a = a_full[:n]

        # project onto non‑negative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
            s = 1.0
        a = a / s

        # occasional jitter to avoid stagnation
        if it % 50 == 0:
            jitter = random_initial(n, rng) * 5e-4
            a = a * 0.9995 + jitter
            a = a / a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)      # 8192

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # Interpolate a previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # Randomised families
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families
    for _ in range(4):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # keep the best candidate so far
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase A – Spectral flattening (alternating projection)
    # ------------------------------------------------------------------
    phase_a_end = start_time + 0.20 * budget_s   # 20 % of total time
    ap_steps = 800                               # iterations per restart
    while time.time() < phase_a_end:
        # start from a mixture of the current best and a random draw
        start = best_a * 0.85 + random_initial(n, rng) * 0.15
        start = np.maximum(start, 0.0)
        start /= start.sum()
        a_ap, sc_ap = alternating_projection(start,
                                              steps=ap_steps,
                                              n=n,
                                              fft_len=fft_len,
                                              rng=rng)
        if sc_ap < best_sc:
            best_sc = sc_ap
            best_a = a_ap.copy()
            print(f"[AP] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase B – L‑BFGS on smoothed max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_b_end = time.time() + 0.30 * budget_s
        lam_schedule = np.geomspace(0.5, 2e-6, num=35)
        maxiter_per_lam = 1200
        while time.time() < phase_b_end:
            # perturb the current best a little before each L‑BFGS run
            pert = best_a * (1 - 0.1 * rng.random()) + random_initial(n, rng) * 0.1
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            a_opt, sc_opt = lbfgs_continuation(pert,
                                               lam_schedule,
                                               maxiter_per_lam,
                                               phase_b_end,
                                               fft_len, fft_len, n)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_end = time.time() + 0.15 * budget_s
        print("[SubGrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                               sub_end,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[SubGrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase D – Adam optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase E – Greedy hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.05)
        print(f"[Hill‑climb] final fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
