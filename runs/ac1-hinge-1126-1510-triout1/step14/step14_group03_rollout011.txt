def gradient_descent_fine(a, fft_len, n, deadline):
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        k_max = int(np.argmax(conv))
        # compute gradient wrt a
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(best_a)
        grad[mask] = 2.0 * best_a[idx[mask]]
        # try step sizes
        for factor in [0.5, 0.1, 0.02, 0.005, 0.001]:
            step = factor * np.mean(best_a)  # maybe scale
            a_new = best_a - step * grad
            # project onto simplex: guarantee nonneg and sum=1
            a_new = simplex_projection(a_new)
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a_new
                break
        # else break if no improvement

"""
Enhanced step‑function optimizer.

The search proceeds in four stages:

1.  **Diverse initial pool** – several deterministic windows,
    a spectral construction, and many Dirichlet draws.
2.  **Heavy L‑BFGS smoothing** – minimise a log‑sum‑exp
    approximation of the maximum self‑convolution, restarting
    from mixtures of the current best and a fresh random draw.
3.  **Sub‑gradient descent** – both a classic exponentiated‑gradient
    (EG) descent and a projected‑gradient method with momentum are
    run on the exact non‑smooth objective.
4.  **Multi‑scale hill‑climbing** – first a generic random mass‑transfer
    hill‑climb, then a *targeted* hill‑climb that always steals mass
    from the indices that contribute most to the current maximal
    convolution entry.  The sequence is kept on the probability simplex
    throughout, so the evaluation score remains scale‑invariant.

All stages respect a total wall‑clock budget of 1000 s.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def momentum_subgrad(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    init_lr: float = 0.5,
                    gamma: float = 0.9):
    """
    Projected sub‑gradient descent with momentum on the exact max‑conv.
    """
    a = a_start.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # compute convolution and the shift attaining the maximum
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size
        eta = init_lr / np.sqrt(t)

        # momentum update (we move opposite to the gradient)
        v = gamma * v + eta * grad
        a = a - v

        # project back onto the simplex
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 200000,
               delta_frac: float = 0.06):
    """
    Greedy hill‑climb with random mass‑transfer moves.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 500000,
                        delta_frac: float = 0.02):
    """
    Fine‑grained hill‑climb that always removes mass from the indices
    that contribute most to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        # current convolution and its maximiser
        fa = np.fft.rfft(best, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # indices that actually take part in the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # product contribution a_i * a_{k_max-i}
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i = int(np.argmax(contrib))                       # worst offender
        # try to move mass to a location that does *not* overlap with k_max
        non_overlap = np.where(~mask)[0]
        if non_overlap.size == 0:
            j = rng.integers(0, n)
        else:
            j = int(rng.choice(non_overlap))

        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # keep non‑negative (projection would also work but cheap to check)
        if cand[i] < 0:
            it += 1
            continue

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse initial pool
    # ------------------------------------------------------------------
    candidates = []

    # Include a previously found construction if present.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
                candidates.append(a0)

    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Pick the best by exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS (≈20 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # split remaining time equally between EG and momentum
        eg_deadline = time.time() + 0.5 * remaining
        print("[EG] start exact exponentiated‑gradient")
        a_eg, sc_eg = eg_exact_max(best_a, eg_deadline, n, fft_len, init_lr=0.5)
        if sc_eg < best_score:
            best_score = sc_eg
            best_a = a_eg.copy()
            print(f"[EG] improved best fast_score={best_score:.8f}")

        if time.time() < deadline:
            mom_deadline = deadline
            print("[Momentum] start projected sub‑gradient")
            a_mom, sc_mom = momentum_subgrad(best_a, mom_deadline, n, fft_len,
                                            init_lr=0.5, gamma=0.9)
            if sc_mom < best_score:
                best_score = sc_mom
                best_a = a_mom.copy()
                print(f"[Momentum] improved best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Generic hill‑climb (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline - 5.0  # keep a tiny cushion
        print("[Hill‑climb] start random mass‑transfer")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=200000,
            delta_frac=0.06
        )
        print(f"[Hill] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted fine hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Targeted‑hill‑climb] start fine‑grained optimisation")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=500000,
            delta_frac=0.02
        )
        print(f"[Targeted] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
```

This version expands the original pipeline with a momentum‑based projected sub‑gradient method and a targeted hill‑climb that directly attacks the dominant convolution term.  By allocating the 1000 s budget across all stages and using increasingly fine‑grained local moves, it aims to push the evaluation score below the previous 1.505876 benchmark.
