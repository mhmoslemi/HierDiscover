def exp_grad_descent(a_start, deadline, n, fft_len, rng, base_eta=0.5, tol=1e-12):

   a = a_start.copy()

   best_a = a.copy(); best_sc = fast_score(a, fft_len, n)

   iter = 0

   while time.time() < deadline:

       iter += 1

       # compute conv

       fa = np.fft.rfft(a, fft_len)

       conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]

       max_val = conv.max()

       # build mask: I = (conv >= max_val - tolerance)

       mask = (conv >= max_val - 1e-12).astype(np.float64)

       # compute gradient

       a_rev = a[::-1]

       fa_rev = np.fft.rfft(a_rev, fft_len)

       fmask = np.fft.rfft(mask, fft_len)

       grad_full = np.fft.irfft(fa_rev * fmask, fft_len)

       grad = 2.0 * grad_full[n-1:n-1+n]

       # step size schedule

       eta = base_eta / np.sqrt(iter)

       # update via exponentiated gradient

       a_new = a * np.exp(-eta * grad)

       # avoid all zero

       sum_a = a_new.sum()

       if sum_a == 0.0:

           # fallback: add small noise

           a_new = a + 1e-12

           sum_a = a_new.sum()

       a_new /= sum_a

       # optionally jitter: occasionally random small move

       if iter % 2000 == 0:

           # jitter to avoid stagnation

           jitter = rng.normal(scale=1e-5, size=n)

           a_new = np.maximum(a_new + jitter, 0.0)

           a_new /= a_new.sum()

       a = a_new

       sc = fast_score(a, fft_len, n)

       if sc < best_sc:

           best_sc = sc

           best_a = a.copy()

           # logging

           if iter % 1000 == 0:

               print(f"[exp-grad] iter {iter}, best fast_score={best_sc:.9f}")

   return best_a, best_sc

"""
Propose a low‑autocorrelation step‑function.

The optimizer works in two stages:

1.  A fast L‑BFGS continuation on a log‑sum‑exp smooth approximation of
    the maximal self‑convolution (the classic “smooth‑max” technique).
    This stage runs on a moderate size (n=8192) to quickly find a good
    shape.

2.  The shape is interpolated to a larger resolution (n=16384) and a
    mirror‑descent style optimisation with the exact sub‑gradient of the
    non‑smooth max‑convolution is performed.  The update is
    exponentiated (a_i ← a_i·exp(−η·∂max/∂a_i)) which automatically
    respects non‑negativity, after which the vector is projected onto the
    probability simplex.  The step size η decays as 1/√t, allowing a
    very large number of cheap iterations (thousands per second).

A final short greedy hill‑climb cleans up any remaining local
irregularities.  The whole pipeline respects the 1000‑second budget
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(convolution) for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral guess – take square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diversified random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "spike",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike
        a = np.full(n, 1.0 / n)
        idx = rng.integers(0, n)
        a[idx] = rng.uniform(2.0, 5.0) / n
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Gaussian window centred on the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    edge_weight: float = 0.0,
) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient.
    The optimisation variables are logits y (a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    if edge_weight > 0.0:
        pen_grad = np.zeros_like(a)
        pen_grad[0] = 2.0 * edge_weight * a[0]
        pen_grad[-1] = 2.0 * edge_weight * a[-1]
        g_vec = g_vec + pen_grad

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (stage 1)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best probability vector.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam, 0.0),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Upsampling (interpolation) from a coarse to a fine grid
# ----------------------------------------------------------------------


def upsample_interpolate(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a_small → a_big (preserving mass)."""
    x_small = np.linspace(0, 1, a_small.size)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.clip(a_big, 0.0, None)
    s = a_big.sum()
    return a_big / s if s > 0 else np.full(n_big, 1.0 / n_big)


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (stage 2)
# ----------------------------------------------------------------------


def conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution using FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    return np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]


def grad_max_convolution(a: np.ndarray, mask: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    mask has shape (2·n‑1,) and contains 1 where conv equals the max.
    """
    n = a.size
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fmask = np.fft.rfft(mask, fft_len)
    rev = np.fft.irfft(fa_rev * fmask, fft_len)
    # central segment gives the gradient
    grad = 2.0 * rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exponentiated mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_mirror_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    base_eta: float = 0.5,
):
    """Run many cheap iterations of a↦a·exp(−η·∂max) with projection."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and mask of maximal entries
        conv = conv_full_fft(a, fft_len)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        # exact sub‑gradient of the max
        grad = grad_max_convolution(a, mask, fft_len)

        # step‑size schedule (1/√t decay)
        eta = base_eta / np.sqrt(it)

        # exponentiated update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            # fallback to a tiny uniform jitter
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional tiny random jitter to escape flat plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=5e-6, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        # keep the best ever
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[exp‑descent] iter {it}, best fast_score={best_sc:.9f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy hill‑climb (tiny mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_sc: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 150_000,
    delta_frac: float = 0.04,
):
    """Deterministic improving moves; stops when time runs out."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc
