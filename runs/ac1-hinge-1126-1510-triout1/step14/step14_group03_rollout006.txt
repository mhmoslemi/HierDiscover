def propose_candidate(seed=42, budget_s=1000, **kwargs):
    # Setup RNG
    rng = np.random.default_rng(seed)

    # Problem size
    n = 2048  # maybe 2048 for speed

    # Compute FFT length
    fft_len = next_pow2(2*n - 1)

    # Build initial candidate list
    candidates = []

    # earlier height_sequence_1 if exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Interpolate to n points
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # deterministic windows
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # random draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    best_a = None; best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # start timing
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Phase 0: LP improvement
    # allocate 15% of budget, but if we have enough time else skip
    phase0_deadline = start_time + 0.15 * budget_s
    # Precompute row_idx and col_idx for LP matrix (constant for n)
    row_idx = np.concatenate([np.arange(j, j+n) for j in range(n)])
    col_idx = np.repeat(np.arange(n), n)
    # We'll iterate until time.
    a = best_a.copy()
    M = (np.convolve(a, a).max())
    # store best across LP
    best_lp_a = a.copy()
    best_lp_score = fast_score(a, fft_len, n)
    lp_iter = 0
    while time.time() < phase0_deadline:
        # Solve LP: maximize sum(g)
        data = np.tile(a, n)
        A_ub = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        c = -np.ones(n)  # minimize -sum(g)
        b = np.full(2*n-1, M, dtype=np.float64)
        # call linprog
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Scale to sum=1
        g = g0 / sum_g0
        # Evaluate new M
        conv_g = np.convolve(g, g)
        M_new = conv_g.max()
        # Choose update rule: if M_new < M (improvement) accept g; else maybe blend
        if M_new < M - 1e-9:

            sc = fast_score(a, fft_len, n)

"""
Propose a low‑score non‑negative step‑function.

The search is split into four phases (total wall‑clock ≤ budget_s):

1. **Diverse initialisation** – deterministic windows + random Dirichlet draws,
   plus the previous best sequence (if available).  The best is kept as a start.

2. **LP‑based improvement** – given a current vector `a`, solve the linear program
   ``max Σ g_i  s.t. a * g ≤ M`` (convolution ≤ current max M) and accept the
   new vector if its self‑convolution max decreases.  This implements the
   “alternating LP” idea from the reference and quickly moves out of poor local
   minima.

3. **Heavy L‑BFGS smoothing** – minimise a log‑sum‑exp approximation of
   `max(conv(a,a))` with a continuation on the smoothing parameter λ.
   The variables are the softmax logits, guaranteeing non‑negativity and
   sum‑to‑one.

4. **Exact sub‑gradient descent (exponentiated‑gradient)** and a final
   greedy hill‑climb polish.

All phases respect a global timer; the best sequence discovered (as a Python
list) is returned just before the deadline.  The official ``evaluate_sequence``
is called once at the end to report the true score.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
from scipy.signal.windows import dpss


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping and final comparison)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(conv(a,a))
    The convolution is performed via FFT for speed.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient w.r.t.
    the softmax logits y (where a = softmax(y)).
    Returns (objective, gradient_y).
    """
    a = softmax(y)

    # Full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max (d smooth_max / d conv_i)
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic windows + random draws)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Random spectral guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:  # edge‑biased
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, NW: float = 3.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, projected onto simplex."""
    w = dpss(n, NW, return_ratios=False)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for the whole λ‑schedule, starting each stage from the
    result of the previous λ.  Returns the best vector discovered
    (and its exact fast_score).
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def eg_exact_max(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 init_lr: float = 0.5):
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    (non‑smooth) maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        # convolution via FFT (fast)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # EG update (step size decays as 1/sqrt(t))
        eta = init_lr / np.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.04):
    """
    Greedy hill‑climb with tiny mass‑transfer moves.
    Only accepts strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# LP‑based improvement (phase 0)
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray,
               row_idx: np.ndarray,
               col_idx: np.ndarray,
               M: float,
               n: int,
               deadline: float):
    """
    Solve the LP  max Σ g_i  s.t.  conv(a,g) ≤ M   (g ≥ 0),
    then return a scaled version of the LP solution if it strictly reduces
    the self‑convolution maximum.  If no improvement is possible, returns
    the original vector unchanged.
    """
    # Build the (2n‑1)×n constraint matrix with data = tiled a
    data = np.tile(a, n)
    A_ub = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    c = -np.ones(n)                       # maximise Σ g_i  <=> minimise -Σ g_i
    b = np.full(2 * n - 1, M, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'time_limit': max(0.0, deadline - time.time())})

    if not res.success:
        # LP failed – fall back to identity
        return a, M

    g0 = res.x
    if g0.sum() <= 0:
        return a, M

    # Scale to have unit sum (the same normalization as a)
    g = g0 / g0.sum()

    # Check whether the new self‑convolution improves M
    conv_g = np.convolve(g, g)
    M_new = conv_g.max()
    if M_new < M - 1e-12:          # genuine improvement
        return g, M_new
    else:
        # Try a modest convex blend; sometimes a direct jump is too aggressive.
        t = 0.2
        a_blend = (1 - t) * a + t * g
        a_blend /= a_blend.sum()
        conv_blend = np.convolve(a_blend, a_blend)
        M_blend = conv_blend.max()
        if M_blend < M - 1e-12:
            return a_blend, M_blend
        else:
            return a, M

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a modest size keeps the LP tractable while still
    # providing enough degrees of freedom.
    # ------------------------------------------------------------------
    n = 2048
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Assemble a diverse initial pool
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous construction if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates.append(a0)

    # deterministic window functions
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))
    candidates.append(dpss_init(n, NW=3.5))

    # a handful of random draws
    for _ in range(14):
        candidates.append(random_initial(n, rng))

    # pick the best by the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Global timer
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – LP based improvement (≈15 % of budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.15 * budget_s

    # Pre‑compute the index pattern for the convolution‑matrix (independent of a)
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    col_idx = np.repeat(np.arange(n), n)

    # Current bound M = max(conv(a,a))
    M = np.convolve(best_a, best_a).max()

    lp_iter = 0
    while time.time() < phase0_deadline:
        a_new, M_new = lp_improve(best_a, row_idx, col_idx, M, n, phase0_deadline)
        if M_new < M - 1e-9:
            M = M_new
            best_a = a_new
            best_score = fast_score(best_a, fft_len, n)
            print(f"[LP] iter {lp_iter:02d}, new best fast_score={best_score:.8f}, M={M:.8f}")
        else:
            # No improvement – exit early
            break
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS smoothing (≈35 % of remaining budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # random start = convex combination of current best and a fresh draw
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient on max‑conv (≈30 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0      # keep a small cushion for final polish
        print("[EG‑max] start exact exponentiated‑gradient")
        best_a, best_score = eg_exact_max(
            best_a,
            eg_deadline,
            n,
            fft_len,
            init_lr=0.5
        )
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        # tighter step size for fine polishing
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=120000,
            delta_frac=0.02
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
