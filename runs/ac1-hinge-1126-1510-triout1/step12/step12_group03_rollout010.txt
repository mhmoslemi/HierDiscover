"""
Propose a low‑score step function.

Algorithm
---------
1.  Build a diverse pool of starting vectors (spectral, classic
    windows, parametric families and random draws).  Evaluate them with the
    exact fast‑score (2·n·max self‑convolution) and keep the best.
2.  Intensive L‑BFGS polishing.  Many random restarts are performed; each
    restart starts from a convex blend of the current best and a fresh
    random vector.  A fine geometric schedule for the smoothing factor
    λ is used and a relatively large iteration budget per λ is given.
3.  Adam optimisation on the smoothed max‑convolution (log‑sum‑exp) to
    escape shallow plateaux.
4.  A long targeted hill‑climb that always moves mass away from the
    positions contributing most to the current peak of the convolution.
    The step size shrinks automatically as the involved masses become
    small.
5.  Final evaluation with the official ``evaluate_sequence`` routine.

All steps respect the global 1000 s budget and stop early when the
deadline is reached.  The returned list sums to 1 (the score is invariant
under scaling).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    """Power‑law decay a_i ∝ i^{‑exponent} (i = 1…n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.04):
    """Exponentiated‑gradient descent on a top‑fraction of the convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # Full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # Mask top‑fraction entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # Gradient of smoothed max‑convolution (exact, not smoothed)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 500000):
    """
    Greedy hill‑climb that preferentially moves mass away from the
    positions contributing most to the current peak of the self‑convolution.
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        # convolution and current peak
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())
        # Determine the overlapping index range that contributes to conv[idx]
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        if i_start > i_end:
            # fallback to a random move
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            if products.sum() == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / products.sum()
                i = rng.choice(i_range, p=probs)

        # Destination index (different from source)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # Transfer a modest fraction of the source mass
        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep n moderately large (5000 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Use a previously exported solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows with several sigmas
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows with a grid of betas
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Power‑law decays with a grid of exponents
    for expo in np.linspace(0.15, 0.70, 12):
        candidates.append(power_decay_init(n, exponent=expo))

    # Randomised families (multiple draws)
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # Choose the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-9, num=40)   # decreasing λ‑fractions
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35               # ≤ 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.06,
                                           max_iter=50000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.65 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.03)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(best_a,
                                                best_score,
                                                fft_len,
                                                n,
                                                deadline,
                                                rng,
                                                max_moves=800000)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
