"""
Optimiser for the step‑function bound (improved version).

The search consists of five stages:

1. **Broad initial pool** – many families (uniform, exponentials,
   power‑laws, Gaussians, cosine‑powers, sincs, Betas, edge‑heavy,
   sparse).  Each vector is projected onto the probability simplex.

2. **Multiple aggressive L‑BFGS runs on a smoothed log‑sum‑exp proxy**.
   The optimizer works in log‑space (soft‑max) so the simplex constraints
   are automatic.  A long λ‑schedule (from 1 down to 1e‑7) and a generous
   per‑λ iteration budget allow the algorithm to escape shallow basins.

3. **Adam refinement** on the same surrogate with a finer λ‑schedule.

4. **Exact sub‑gradient descent (exponentiated‑gradient)** – the true
   L∞‑objective is tackled directly; the loop runs until the time budget
   for this phase is exhausted (typically a few hundred thousand steps).

5. **Long Metropolis annealing** – three move types (global transfer,
   local transfer, swap) with a linearly decreasing temperature polish
   the last few digits.

All stages share a common `fast_score` implementation that evaluates
`2·n·max(conv(a,a))` using a single FFT (the denominator of
`evaluate_sequence` is 1 because the vectors are normalised to sum 1).
The final sequence is stored in the global `height_sequence_1` variable
and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    Uses an FFT of length ``fft_len`` (a power of two ≥ 2·n‑1).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from a diverse set of styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian‑shaped density."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power (non‑negative)."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass at the two edges, light interior."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Squared sinc shape (bell‑like)."""
    x = np.arange(n) - (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.15) * n
    a = np.sinc(x / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangular_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular ramp up then down."""
    peak_frac = rng.uniform(0.2, 0.6)
    peak = int(peak_frac * n)
    if peak == 0:
        return np.full(n, 1.0 / n)
    a = np.concatenate([
        np.linspace(1, peak, peak, endpoint=False),
        np.full(n - 2 * peak, peak),
        np.linspace(peak, 1, peak, endpoint=True)
    ])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    alpha = rng.uniform(0.3, 3.0)
    beta = rng.uniform(0.3, 3.0)
    x = np.linspace(0.0, 1.0, n)
    # add a small epsilon to avoid zero at the ends
    eps = 1e-10
    a = (x + eps) ** (alpha - 1) * (1 - x + eps) ** (beta - 1)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    """
    # softmax
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    # convolution a*a (via FFT)
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best probability vector encountered and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_sched,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.05):
    """
    Adam optimisation (adaptive moment) on the log‑sum‑exp surrogate.
    Returns the best probability vector encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

# Exact sub‑gradient exponentiated‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          eta0: float = 30.0):
    """
    Mirror‑descent on the exact L∞‑objective.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (exponentially damped)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Metropolis annealer with three move types:
      * global mass transfer,
      * swap,
      * local neighbour transfer (distance ≤ 5).
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]          # pure swap
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # move from larger entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

            # resample to current n
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) classic windows (if SciPy is present)
    try:
        from scipy.signal import chebwin, kaiser  # noqa: F401
        have_windows = True
    except Exception:
        have_windows = False

    if have_windows:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

    # 2) Random families
    for _ in range(8):
        pool.append(random_initial(n, rng))

    # 3) Parametric families (including the new Beta family)
    for _ in range(3):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))
        pool.append(sinc_initial(n, rng))
        pool.append(triangular_initial(n, rng))
        pool.append(beta_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget splits (total budget_s seconds)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.30 * budget_s        # aggressive L‑BFGS
    phase2_end = start_time + 0.50 * budget_s        # Adam refinement
    phase3_end = start_time + 0.80 * budget_s        # exp‑gradient descent
    phase4_end = start_time + 0.90 * budget_s        # fine L‑BFGS
    # remaining time → annealing / mixing

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule_l1 = [
        1.0, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05, 0.03, 0.02, 0.015,
        0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 5e-4, 2e-4,
        1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7,
        2e-7, 1e-7
    ]
    maxiter_per_lam = 1200  # a bit more than before

    best_a = None
    best_score = float('inf')
    best_logit = None

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # occasional jitter to explore neighbours
        if rng.random() < 0.35:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.008),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule_l1,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_logit = np.log(best_a + 1e-12)
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed surrogate (fine λ‑schedule)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_adam = [
            5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
        ]
        iters_per_lam = 300
        a_adam, sc_adam = adam_optimize(
            best_logit,
            lam_schedule_adam,
            iters_per_lam,
            phase2_end,
            fft_len,
            n,
            lr=0.07,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            print(f"[Adam] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            eta0=30.0,
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS with ultra‑small λ values
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        fine_lam_sched = [1e-7, 5e-8, 2e-8, 1e-8, 5e-9, 2e-9, 1e-9]
        a_fine, sc_fine = lbfgs_cascade(
            best_a,
            fine_lam_sched,
            maxiter_per_lam=1800,
            deadline=phase4_end,
            fft_len=fft_len,
            n=n,
        )
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – extended simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional random mixing of top candidates (cheap extra polish)
    # ------------------------------------------------------------------
    candidates = [best_a]
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape == best_a.shape:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            candidates.append(prev)

    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for _ in range(7):
                w = rng.random()
                mix = w * candidates[i] + (1 - w) * candidates[j]
                mix = np.maximum(mix, 0.0)
                if mix.sum() == 0.0:
                    continue
                mix /= mix.sum()
                sc = fast_score(mix, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = mix.copy()
                    print(f"[mix] new best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the platform
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
