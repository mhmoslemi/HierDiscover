"""
Hybrid optimisation for the step‑function attack.

Algorithm
---------
1. **Diversified warm‑starts** –  generate a large pool of symmetric
   probability vectors from many parametric families (uniform,
   edge‑exponential, power‑edge, symmetric beta, triangular, quadratic,
   bimodal, sparse, mixed).  Each candidate is normalised to sum‑one and
   scored with a fast FFT based surrogate ``fast_score``.  The best
   candidate is kept.

2. **Multi‑stage L‑BFGS cascade** – minimise a smoothed log‑sum‑exp
   surrogate of the L∞‑objective in log‑space.  The smoothing factor
   λ is decreased along a long schedule (≈25 values) and each stage is
   given a few thousand L‑BFGS iterations.  The cascade is run twice:
   a coarse pass (large λ) followed by a fine pass (very small λ).

3. **Momentum mirror‑SGD (exponentiated‑gradient)** – a long
   sub‑gradient descent on the exact max‑convolution, using a
   momentum‑averaged sub‑gradient and a η₀/√t step‑size schedule.
   This stage runs until the allocated time slice is exhausted.

4. **Aggressive simulated‑annealing** – random mass transfers (both
   local and long‑range) are accepted with a Metropolis rule.  The
   temperature is linearly cooled; moves that reduce the fast score are
   always kept.

All stages respect a global 1000 s deadline.  The best vector w.r.t.
the official ``evaluate_sequence`` is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    Convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add i.i.d. Gaussian noise and re‑project onto the probability simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation (expanded families)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a symmetric probability vector from a diverse family."""
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "beta",
            "triangular",
            "quadratic",
        ]
    )
    x = np.arange(n)
    half = n // 2
    # distance to the nearer edge (used by several families)
    d = np.minimum(x, n - 1 - x)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        a = (d + 1.0) ** (-p)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        rate = rng.uniform(0.001, 0.02)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == "beta":
        # symmetric Beta(α,α) distribution, α∈[0.5,5]
        alpha = rng.uniform(0.5, 5.0)
        # avoid divide‑by‑zero when n==1
        if n == 1:
            a = np.ones(1)
        else:
            xv = x / (n - 1)
            a = (xv ** (alpha - 1)) * ((1.0 - xv) ** (alpha - 1))
            # enforce a tiny floor to stay away from exact zeros
            a = np.maximum(a, 1e-12)

    elif mode == "triangular":
        # linear increase to centre then decrease
        a = d.astype(float) + 1.0

    else:  # quadratic (p‑th power of the triangular shape)
        p = rng.uniform(1.5, 4.0)
        a = (d.astype(float) + 1.0) ** p

    # final normalisation – ensure proper probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed L∞ surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Underlying probabilities a = softmax(y).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution via FFT ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # size 2n-1

    # back‑propagation through the convolution:
    # d (a*a) / d a = 2 * (a reversed convolved with mask)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the relevant gradient components correspond to the original length‑n part
    g_vec = conv_rev[n - 1 : n - 1 + n]  # shape (n,)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its ``fast_score``.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        # L‑BFGS on the smoothed surrogate
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirrored exponentiated gradient)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
    max_iters: int = 5_000_000,
):
    """
    Mirror‑SGD on the exact L∞‑objective.
    Uses momentum‑averaged sub‑gradient and a η₀/√t schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.85  # a bit more smoothing than before

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        # momentum
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (stay in the simplex)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        # cheap periodic check
        if (it & 31) == 0:  # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score

# Simulated‑annealing hill‑climb (more aggressive moves)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Aggressive simulated‑annealing on the exact fast_score.
    Allows both local and long‑range mass transfers.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.07, 1e-12)  # a higher initial temperature

    # fraction of source mass that may be moved in a single proposal
    max_frac_local = 0.12
    max_frac_global = 0.03

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac_time = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac_time), 1e-12)

        # Choose source and destination – sometimes far apart
        if rng.random() < 0.6:
            # local move – pick neighbours within a window
            i = rng.integers(0, n)
            # window size ~5 % of n
            win = max(1, int(0.05 * n))
            low = max(0, i - win)
            high = min(n, i + win + 1)
            j = rng.integers(low, high)
        else:
            # global move – completely random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        if i == j:
            continue

        # Ensure we move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        # Choose fraction according to move type
        max_frac = max_frac_local if (abs(i - j) <= win) else max_frac_global
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < math.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[SA] new best fast_score={best_score:.9f}")

    return best_a, best_score

        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] previous best fast_score={best_score:.9f}")

    # generate candidates until phase‑1 deadline
    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)

        # occasional jitter around the current best
        if best_a is not None and rng.random() < 0.12:
            a = add_noise(best_a, sigma=rng.uniform(0.001, 0.006), rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Stage 2 – first L‑BFGS cascade (coarse λ schedule)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_coarse = [
            0.7,
            0.5,
            0.35,
            0.25,
            0.15,
            0.1,
            0.07,
            0.05,
            0.03,
            0.02,
            0.01,
            0.005,
            0.003,
            0.0015,
            0.0008,
            0.0004,
            0.0002,
            0.0001,
            5e-05,
            1e-05,
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule_coarse,
            maxiter_per_lam=2500,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS coarse] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 3 – second L‑BFGS cascade (very fine λ schedule) + exp‑grad
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        lam_schedule_fine = [
            5e-06,
            2e-06,
            1e-06,
            5e-07,
            2e-07,
            1e-07,
            5e-08,
            2e-08,
            1e-08,
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule_fine,
            maxiter_per_lam=3000,
            deadline=phase3_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS fine] best fast_score={best_score:.9f}")

        # short exponentiated‑gradient run to polish
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=15.0,
            max_iters=2_000_000,
        )
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 4 – final aggressive simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑SA] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
