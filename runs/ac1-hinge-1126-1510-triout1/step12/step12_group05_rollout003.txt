def adam_optimize(a_start, deadline, n, fft_len, rng, lr0=0.5, beta1=0.9, beta2=0.999):
    a = a_start.copy()
    y = np.log(a + 1e-12)  # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # grad of max conv
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv_full[n-1:n-1+n]  # derivative of max conv w.r.t a
        # gradient of objective J = 2*n*max_val (since sum=1)
        grad_J = 2.0 * n * g
        # Chain rule into logits
        mean_grad = np.dot(a, grad_J)
        grad_y = a * (grad_J - mean_grad)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + 1e-8)
        y -= step * m_hat
        # re-softmax
        y_max = np.max(y)
        e = np.exp(y - y_max)
        a = e / e.sum()
        # periodic evaluate
        if (t & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t} fast_score={best_score:.9f}")
    return best_a, best_score

def lp_refine(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    # parameters
    t = 0
    while time.time() < deadline:
        # compute convolution and find active indices where conv near max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # Choose active indices: e.g., top K highest values, maybe all within 0.001*max?
        idx_active = np.where(conv >= max_val - max_val * 1e-4)[0]
        if idx_active.size == 0:
            idx_active = np.array([np.argmax(conv)])
        # Build linear constraints matrix A (len idx_active x n)
        # Use vectorized method: each active index k leads to row A[k] where A[k,j] = a[k-j] if 0<=k-j<n else 0.
        # We'll build using loops for each active index due to moderate size.
        rows = []
        for k in idx_active:
            # produce row
            row = np.zeros(n)
            j_min = max(0, k - (n-1))
            j_max = min(k, n-1)
            # fill row
            # vectorized: indices = np.arange(j_min, j_max+1)
            # row[indices] = a[k - indices]
            indices = np.arange(j_min, j_max+1)
            row[indices] = a[k - indices]
            rows.append(row)
        A = np.vstack(rows)  # shape (len(idx_active), n)
        M = max_val
        # Solve LP
        try:
            import cvxpy as cp
            g_var = cp.Variable(n, nonneg=True)
            constraints = [A @ g_var <= M]
            # also ensure (f*g) <= M for the rest? ignore.
            objective = cp.Maximize(cp.sum(g_var))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
        except Exception as e:
            print(f"[LP] solver error: {e}")
            break
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        sum_g = g.sum()
        sum_a = a.sum()
        if sum_g <= 0:
            break
        # scale to keep sum.
        g = g * (sum_a / sum_g)
        # convex combination
        t += 1
        alpha = min(0.5, 1.0 / (t + 1))
        a = (1 - alpha) * a + alpha * g
        # re-normalize (tiny errors)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        # evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter={t} fast_score={best_score:.9f}")
    return best_a, best_score

"""
Step‑function optimiser (≈1000 s budget)

Algorithm
---------
1. **Warm‑starts** – Generate many diverse shapes (uniform, edge‑exponential,
   edge‑power, bimodal, sparse, mixed, inverse‑sqrt, edge‑geometric) and keep the
   best one according to a fast proxy score 2·n·max(conv(a,a)).
2. **L‑BFGS cascade** – Refine the warm‑start with a sequence of smoothed
   log‑sum‑exp objectives (decreasing λ).  The optimisation works in log‑space
   (softmax) so that the coefficients stay non‑negative and sum to one.
3. **Adam on the exact L∞‑objective** – Perform a long mirror‑gradient descent
   on the true max‑convolution (no smoothing).  Gradients are obtained from the
   sub‑gradient of the max‑convolution; Adam’s adaptive steps accelerate
   convergence.
4. **Linear‑programming refinement** – At the current point, collect the indices
   where the convolution attains (or almost attains) its maximum.  Solve a tiny
   LP (maximise ∑g_j subject to (f∗g)_k ≤ M for those active k) and mix the
   solution back into the current vector.
5. **Optional simulated‑annealing polish** – A cheap hill‑climb finishes the
   search.
6. Return the best sequence (stored globally as ``height_sequence_1``) and
   its official evaluation.

All stages respect the global 1000 s deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` is a probability vector:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add isotropic Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation (new families added)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a symmetric probability vector using a randomly chosen family.
    The families include uniform, edge‑exponential, edge‑power,
    inverse‑sqrt‑edge, edge‑geometric, bimodal, sparse and mixed.
    """
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "inverse_sqrt",
            "geom_edge",
            "bimodal",
            "sparse",
            "mixed",
        ]
    )
    x = np.arange(n)
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "inverse_sqrt":
        # a_i ∝ 1/√((i+1)*(n-i))
        prod = (x + 1) * (n - x)
        a = 1.0 / np.sqrt(prod + 1e-12)

    elif mode == "geom_edge":
        # a_i ∝ decay^{distance to nearest edge}
        decay = rng.uniform(0.9, 0.9995)
        d = np.minimum(x, n - 1 - x)
        a = decay ** d

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    else:  # mixed
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    # avoid exact zeros, normalise to a probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    Returns (objective, gradient_y).
    """
    # ---- softmax ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution via FFT ----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12,
                         "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the exact L∞‑objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
):
    """
    Mirror‑Adam on the true max‑convolution objective.
    a is kept as a probability vector via a softmax parametrisation.
    """
    a = a_start.copy()
    # logits y so that a = softmax(y)
    y = np.log(a + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t_iter = 0
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        # ----- convolution and mask of active max locations -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the (hard) max‑convolution -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1 : n - 1 + n]   # d max(conv) / d a

        # ----- gradient of the exact objective J = 2·n·max(conv) -----
        grad_J = 2.0 * n * g  # shape (n,)

        # ----- chain rule to logits (softmax) -----
        mean_grad = np.dot(a, grad_J)
        grad_y = a * (grad_J - mean_grad)

        # ----- Adam update -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr0 / (np.sqrt(v_hat) + 1e-8)
        y -= step * m_hat

        # ----- re‑softmax -----
        y_max = np.max(y)
        expy = np.exp(y - y_max)
        a = expy / expy.sum()

        # ----- periodic cheap score check -----
        if (t_iter & 31) == 0:     # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t_iter} fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (tiny LP on active constraints)
# ----------------------------------------------------------------------


def lp_refine(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    max_active: int = 8,
):
    """
    At the current point, collect the (few) indices where the convolution
    attains its maximum and solve a tiny LP:
        maximise   ∑ g_j
        subject to (a ∗ g)_k ≤ M   for those active k,
                   g_j ≥ 0.
    The solution is mixed back into the current vector.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[LP] cvxpy not available – skipping LP refinement")
        return a_start, fast_score(a_start, fft_len)

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    t_iter = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # indices where conv is within a tiny relative slack of the max
        slack = max_val * 1e-5
        active_idx = np.where(conv >= max_val - slack)[0]
        # keep at most max_active indices (choose the largest ones)
        if active_idx.size > max_active:
            # sort by value descending, keep top max_active
            order = np.argsort(-conv[active_idx])[:max_active]
            active_idx = active_idx[order]

        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)])

        # Build constraint matrix A (rows = active constraints)
        rows = []
        for k in active_idx:
            row = np.zeros(n)
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            idx = np.arange(j_min, j_max + 1)
            row[idx] = a[k - idx]          # coefficient a_{k-j}
            rows.append(row)
        A = np.vstack(rows)               # shape (len(active_idx), n)

        # ----- solve LP -----
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= max_val]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except Exception as e:
            print(f"[LP] solver failure: {e}")
            break

        if g.value is None:
            # infeasible or solver gave up
            break
        g_vec = np.maximum(g.value, 0.0)

        # rescale to keep total mass unchanged
        sum_a = a.sum()
        sum_g = g_vec.sum()
        if sum_g == 0:
            break
        g_vec *= sum_a / sum_g

        # convex combination with a shrinking step size
        t_iter += 1
        alpha = min(0.5, 1.0 / (t_iter + 1))
        a = (1.0 - alpha) * a + alpha * g_vec
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum

        # cheap score update
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter={t_iter} fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (as a final polish)
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly towards the deadline.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size works well for the time budget
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Timing allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.15 * budget_s       # warm‑starts
    phase2_end = start_time + 0.45 * budget_s       # L‑BFGS cascade
    phase3_end = start_time + 0.80 * budget_s       # Adam optimisation
    # remaining time will be used for LP + final anneal

    # ------------------------------------------------------------------
    # Stage 1 – generate many parametric shapes and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] previous best fast_score={best_score:.9f}")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)

        # occasional jitter of a promising candidate
        if best_a is not None and rng.random() < 0.12:
            a = add_noise(a, sigma=rng.uniform(0.001, 0.006), rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS cascade with a finer λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.5,
            0.35,
            0.2,
            0.1,
            0.05,
            0.02,
            0.01,
            0.005,
            0.002,
            0.001,
            5e-04,
            2e-04,
            1e-04,
            5e-05,
            2e-05,
            1e-05,
            5e-06,
            2e-06,
            1e-06,
            5e-07,
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam=3000,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 3 – long Adam optimisation on the exact L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            lr0=0.8,
        )
        print(f"[Adam] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 4 – LP‑based refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = lp_refine(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[LP] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 5 – final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
