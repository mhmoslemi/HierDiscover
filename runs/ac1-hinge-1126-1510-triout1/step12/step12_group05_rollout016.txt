def generate_random_phase(n, fft_len, rng):
    # create spectrum of size fft_len (complex)
    # only half needed due to hermitian symmetry
    # First set all to zero
    spec = np.exp(1j * rng.uniform(0, 2*np.pi, size=fft_len))
    # enforce conjugate symmetry:
    # for k=1..fft_len//2-1, set spec[-k] = conj(spec[k])
    half = fft_len//2
    for k in range(1, half):
        spec[-k] = np.conj(spec[k])
    # ensure real at k=0 and k=half if even
    spec[0] = np.real(spec[0]) + 0j
    if fft_len % 2 == 0:
        spec[half] = np.real(spec[half]) + 0j
    # compute IDFT
    seq = np.real(np.fft.ifft(spec))
    seq = seq[:n]   # take first n entries (the rest are periodic)
    # make nonnegative
    seq = np.abs(seq)   # ensure nonnegative
    # Avoid zeros
    seq = np.maximum(seq, 1e-12)
    seq /= seq.sum()
    return seq
```

"""
step‑function optimiser – 1000 s budget (enhanced)

Algorithm sketch
----------------
1. **Rich warm‑starts** – generate a large pool of candidates:
   • the original parametric families (exp, power, bimodal, …)
   • a new “random‑phase” family: a is the absolute value of an
     inverse‑FFT of a unit‑modulus spectrum with random phases.
   All candidates are normalised to a probability vector.
   Keep the best according to the fast proxy
        2·n·max(conv(a,a))

2. **L‑BFGS cascade** – starting from the best warm‑start, minimise a
   smooth log‑sum‑exp surrogate of the ‑∞‑objective.
   The smoothing parameter λ is decreased from 5e‑1 to 1e‑6,
   each stage allowed up to 3000 BFGS iterations.

3. **Exponentiated‑gradient (mirror‑SGD)** – run a long momentum‑averaged
   sub‑gradient descent on the true L∞‑objective.  The update is
   a_i ← a_i·exp(−η·g_i) with a decaying step‑size η≈η₀/√t.
   The iteration continues until the bulk of the time budget is spent.

4. **Fine polishing** – a short final L‑BFGS run with very low λ,
   followed by a quick simulated‑annealing hill‑climb.

All stages respect the 1000 s deadline; the best sequence is stored in
the global ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse symmetric probability vector using several families."""
    mode = rng.choice(
        ['uniform', 'exp_edge', 'power_edge', 'bimodal', 'sparse', 'mixed']
    )
    x = np.arange(n)
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_edge':
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == 'power_edge':
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == 'bimodal':
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == 'sparse':
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    else:   # mixed
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def generate_random_phase_sequence(n: int, fft_len: int,
                                   rng: np.random.Generator) -> np.ndarray:
    """
    Generate a non‑negative probability vector by taking the absolute value
    of the inverse real FFT of a unit‑modulus spectrum with random phases.
    """
    half = fft_len // 2 + 1
    phases = rng.uniform(0.0, 2 * np.pi, size=half)
    spectrum = np.exp(1j * phases)               # |spectrum| == 1
    seq = np.fft.irfft(spectrum, n=fft_len)      # real‑valued of length fft_len
    seq = seq[:n]                                # keep first n entries
    seq = np.abs(seq)                            # enforce non‑negativity
    seq = np.maximum(seq, 1e-12)                 # avoid exact zeros
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed maximum of the self‑convolution (log‑sum‑exp) and its gradient.
    The optimisation variables are the log‑softmax parameters y,
    i.e. a = softmax(y) is a probability vector.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase together with its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑softmax variables
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          max_iters: int = 8_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    Uses a momentum‑averaged sub‑gradient and a step‑size η≈η₀/√t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped exponent)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional fast_score check
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted by a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) with a low
    evaluation score.  Returns the best list of floats found before the
    deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate‑large size works well for the time budget.
    # Larger n yields a lower theoretical bound (≈1) but increases per‑
    # iteration cost.  n=8000 is a good compromise.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Timing allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.18 * budget_s   # diverse warm‑starts
    phase2_end = start_time + 0.48 * budget_s   # L‑BFGS cascade
    phase3_end = start_time + 0.85 * budget_s   # exponentiated‑gradient

    # ------------------------------------------------------------------
    # Stage 1 – generate many candidate shapes (parametric + random‑phase)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')

    # reuse previous construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] previous best fast_score={best_score:.9f}")

    # main warm‑start loop
    while time.time() < phase1_end:
        # pick a generation method probabilistically
        if rng.random() < 0.55:   # parametric families (more diverse)
            a = generate_candidate_shape(n, rng)
        else:                     # random‑phase family (cheap & many)
            a = generate_random_phase_sequence(n, fft_len, rng)

        # occasional jitter of the current best
        if best_a is not None and rng.random() < 0.10:
            a = add_noise(a, sigma=rng.uniform(0.001, 0.006), rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)
        print("[warm‑start] fell back to uniform")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS cascade with a fine λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.5, 0.3, 0.2, 0.15, 0.10, 0.07, 0.05,
            0.03, 0.02, 0.01, 0.005, 0.002, 0.001,
            5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 1e-6
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam=3000,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 3 – long exponentiated‑gradient descent (momentum)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=25.0,
            max_iters=10_000_000
        )
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 4 – final polishing: low‑λ L‑BFGS + short anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,

        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
