def exponentiated_gradient(a0, max_iters, deadline):
   a = a0.copy()
   best = a; best_score = fast_score(a)
   it = 0
   while it < max_iters and time.time() < deadline:
       # compute conv
       fa = np.fft.fft(a, fft_len)
       conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
       M = conv.max()
       # subgradient w.r.t a: g = 2 * (conv of mask with reversed a)
       eps = M * 1e-12 + 1e-12
       mask = (conv >= M - eps).astype(np.float64)
       # compute gradient
       mask_fft = np.fft.fft(mask, fft_len)
       a_rev = a[::-1]
       a_rev_fft = np.fft.fft(a_rev, fft_len)
       grad_full = np.real(np.fft.ifft(mask_fft * a_rev_fft))
       g = 2 * grad_full[n-1: n-1+n]  # size n
       # step size eta: decreasing
       eta = 1.0 / (np.sqrt(it+1) + 1)  # maybe scaled
       # update multiplicatively
       factor = np.exp(-eta * g)
       a = a * factor
       a /= a.sum()
       # evaluate
       score = fast_score(a)
       if score < best_score:
           best_score = score; best = a.copy()
       it +=1

def lp_mix(a, max_iter, deadline):
   a_current = a.copy()
   best_score = fast_score(a_current)
   best = a_current.copy()
   while time.time() < deadline and iter_count < max_iter:
       # compute M
       fa = np.fft.fft(a_current, fft_len)
       conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
       M = conv.max()
       # if M is extremely low unreachable?
       # Build A
       A = np.zeros((2*n-1, n), dtype=np.float64)
       for j in range(n):
           A[j:j+n, j] = a_current
       # LP solve
       c = -np.ones(n)
       bounds = [(0, None)] * n
       b_ub = np.full(2*n-1, M)
       # using linprog
       res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
       if not res.success:
           break
       g = res.x
       sum_g = g.sum()
       if sum_g <= 0:
           break
       g_norm = g / sum_g
       # try mixing with decreasing t
       improved = False
       for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
           a_new = (1 - t) * a_current + t * g_norm
           # ensure sum 1
           a_new /= a_new.sum()
           score_new = fast_score(a_new)
           if score_new < best_score:
               best_score = score_new; best = a_new.copy()
           # check conv max
           fa_new = np.fft.fft(a_new, fft_len)
           conv_new = np.real(np.fft.ifft(fa_new * fa_new))[:2*n-1]
           M_new = conv_new.max()
           if M_new < M:
               a_current = a_new
               improved = True
               break
       if not improved:
           # maybe reduce step size further
           break
       iter_count += 1
   return best, best_score

def lp_mixed_optimize(a_start, deadline, n, fft_len, rng):
    a_cur = a_start.copy()
    best_a = a_cur.copy()
    best_score = fast_score(best_a, fft_len)
    iter_count = 0
    while time.time() < deadline:
        # compute M
        fa = np.fft.fft(a_cur, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        M = conv.max()
        # Build A matrix
        # allocate zeros
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # Fill matrix
        for j in range(n):
            A[j:j+n, j] = a_cur
        # Solve LP
        c = -np.ones(n)  # maximize sum g
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n
        # call linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # attempt mixing
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            a_new = (1 - t) * a_cur + t * g_norm
            a_new /= a_new.sum()
            # compute score
            sc = fast_score(a_new, fft_len)
            if sc < best_score:
                best_score = sc; best_a = a_new.copy()
            # compute new M to ensure not increase
            fa_new = np.fft.fft(a_new, fft_len)
            conv_new = np.real(np.fft.ifft(fa_new * fa_new))[:2*n-1]
            M_new = conv_new.max()
            if M_new < M - 1e-12:  # improvement
                a_cur = a_new
                improved = True
                break
        if not improved:
            # try a smaller t or exit
            break
        iter_count += 1
        # optional break after many iterations
        if iter_count >= 30:
            break
    return best_a, best_score
```

"""
Propose a low‑value step‑function sequence.

The search pipeline:
1. Small‑scale LP‑based improvement (n≈2000).  Starting from a diverse
   pool we solve a linear program that maximises the total mass of a
   candidate `g` under the constraint `conv(f,g) ≤ M`.  Mixing the
   current `f` with the normalised `g` provably does not increase the
   maximal self‑convolution `M`.  Repeating this yields a strong reduction
   of the bound at very low computational cost.

2. Upsample the refined small‑scale solution to a larger grid
   (`n≈8000`).  A finer discretisation gives the optimiser more freedom
   to lower the max‑convolution.

3. Adam optimisation on a smooth log‑sum‑exp surrogate of the
   max‑convolution (the same surrogate used in the previous state‑of‑the‑art
   code).  The surrogate is far cheaper to evaluate than the exact
   `max` and provides a useful gradient for many iterations.

4. Short Metropolis‑annealing polishing using mass‑transfer moves.

The whole routine respects a 1000 s budget and stores the best found
sequence globally as `height_sequence_1`.  All vectors are kept on the
probability simplex (sum = 1), so the final evaluation is simply
`2·n·max(conv)` – the same quantity returned by the fast scorer.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n) - (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.15) * n
    a = np.sinc(x / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangular_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    peak_frac = rng.uniform(0.2, 0.6)
    peak = int(peak_frac * n)
    a = np.concatenate([
        np.linspace(1, peak, peak, endpoint=False),
        np.full(n - 2 * peak, peak),
        np.linspace(peak, 1, peak, endpoint=True)
    ])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (used by Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    """
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv entries
    w = exp_shift / Z
    # back‑prop through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def adam_optimize(y_start: np.ndarray,
                  lam_sched,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.07):
    """
    Adam optimiser on the smoothed surrogate.
    Returns the best probability vector encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of the max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

# Simulated‑annealing polishing (same as previous version)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
    * mass transfer between two arbitrary positions
    * swapping two entries
    * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]          # pure swap
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # move from larger entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        # fast evaluation (FFT based)
        fft_len = next_pow2(2 * n - 1)
        new_score = fast_score(cand, fft_len)
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing step (core of the alternating scheme)
# ----------------------------------------------------------------------


def lp_mixed_optimize(a_start: np.ndarray,
                      phase_end: float,
                      n: int,
                      fft_len: int,
                      rng: np.random.Generator,
                      max_iters: int = 30):
    """
    Repeatedly solve
        max   sum(g)
        s.t.  conv(a,g) ≤ M    ,   g ≥ 0
    where M = max(conv(a,a)).  The normalized g is mixed into a with a
    small convex coefficient.  The bound `2·n·M` never increases.
    """
    a_cur = a_start.copy()
    best_a = a_cur.copy()
    best_score = fast_score(best_a, fft_len)

    rows = 2 * n - 1
    it = 0
    while time.time() < phase_end and it < max_iters:
        # current maximum convolution value
        fa = np.fft.fft(a_cur, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: rows]
        M = conv.max()

        # build convolution matrix A where (A @ g)[k] = sum_i a[i] * g[k-i]
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a_cur

        # linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # should not happen, but break to avoid endless loops
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # try mixing with decreasing step sizes (small t guarantees
        # non‑increase of the bound)
        improved = False
        for t in (0.05, 0.1, 0.2, 0.3, 0.5):
            a_new = (1 - t) * a_cur + t * g_norm
            a_new /= a_new.sum()
            # compute new bound
            fa_new = np.fft.fft(a_new, fft_len)
            conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: rows]
            M_new = conv_new.max()
            if M_new <= M + 1e-14:          # tolerance for numerical noise
                score_new = 2.0 * n * M_new
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
                a_cur = a_new
                improved = True
                break
        if not improved:
            # no admissible t found → stop the LP loop
            break
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # small safety margin

    # --------------------------------------------------------------
    # Phase 0 – small scale LP‑based optimisation (n≈2000)
    # --------------------------------------------------------------
    n_small = 2000
    fft_small = next_pow2(2 * n_small - 1)

    pool = []

    # incorporate previous best if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n_small:
                xs_old = np.linspace(0, 1, old.size)
                xs_new = np.linspace(0, 1, n_small)
                a0 = np.interp(xs_new, xs_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # random families
    for _ in range(8):
        pool.append(random_initial(n_small, rng))

    # parametric families
    for _ in range(2):
        pool.append(gaussian_initial(n_small, rng))
        pool.append(cosine_power_initial(n_small, rng))
        pool.append(edge_heavy_initial(n_small, rng))
        pool.append(sinc_initial(n_small, rng))
        pool.append(triangular_initial(n_small, rng))

    # pick the best starter from the pool (fast_score assumes sum=1)
    best_small = None
    best_score_small = np.inf
    for a in pool:
        a = a / a.sum()
        sc = fast_score(a, fft_small)
        if sc < best_score_small:
            best_score_small = sc
            best_small = a.copy()
    print(f"[init‑small] best fast_score = {best_score_small:.9f}")

    # --------------------------------------------------------------
    # Phase 1 – LP‑based iterative improvement (use ~40 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    a_lp, sc_lp = lp_mixed_optimize(best_small, phase1_end,
                                    n_small, fft_small, rng,
                                    max_iters=30)
    if sc_lp < best_score_small:
        best_score_small = sc_lp
        best_small = a_lp.copy()
    print(f"[LP‑small] best fast_score = {best_score_small:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – up‑sample to a larger grid (n≈8000)
    # --------------------------------------------------------------
    n_large = 8000
    fft_large = next_pow2(2 * n_large - 1)

    xs_small = np.linspace(0, 1, n_small)
    xs_large = np.linspace(0, 1, n_large)
    a_large = np.interp(xs_large, xs_small, best_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()

    # --------------------------------------------------------------
    # Phase 3 – Adam optimisation on smoothed surrogate (≈20 % budget)
    # --------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 5.0:
        phase3_end = time.time() + 0.20 * budget_s
        lam_schedule_adam = [
            5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
        ]
        y0 = np.log(a_large + 1e-12)
        a_adam, sc_adam = adam_optimize(
            y0,
            lam_schedule_adam,
            iters_per_lam=400,
            deadline=phase3_end,
            fft_len=fft_large,
            n=n_large,
            lr=0.07,
        )
        if sc_adam < best_score_small:
            best_score_small = sc_adam
            a_large = a_adam.copy()
        print(f"[Adam‑large] fast_score = {fast_score(a_large, fft_large):.9f}")

    # --------------------------------------------------------------
    # Phase 4 – final annealing polish (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_score = fast_score(a_large, fft_large)
        a_final, sc_final = simulated_annealing_improved(
            a_large,
            cur_score,
            deadline,
            n_large,
            rng,
        )
        if sc_final < best_score_small:
            best_score_small = sc_final
            a_large = a_final.copy()
        print(f"[anneal‑final] fast_score = {fast_score(a_large, fft_large):.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = a_large.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
