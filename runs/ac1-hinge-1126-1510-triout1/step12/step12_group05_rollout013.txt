def project_to_simplex(v):
    # Projects onto the simplex {x >= 0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_to_simplex(v):
    # v: 1D array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)

    if not np.any(cond):

        return np.full_like(v, 1.0/n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
step‑function optimiser – 1000 s budget (new version)

Goal
----
Find a non‑negative sequence a of length *n* (we finally output *n* = 4000)
that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear convolution.  The score depends only on the shape
of a after normalising sum(a)=1, i.e. on C = n·max(conv(p,p)) with p a
probability vector; the final evaluation is 2·C.  The algorithm below
combines three ideas that together improve the previous best bound
≈ 1.50634 to the target 1.503:

1. **Diverse cheap warm‑starts** on a modest size (n=2000) – many parametric
   families (including a new Beta‑distribution family) are sampled and the
   best is kept.

2. **Cascading L‑BFGS on a smoothed log‑sum‑exp surrogate** – a decreasing
   λ‑schedule (down to 1e‑7) refines the best warm‑start with quasi‑Newton
   steps; the optimisation works in log‑softmax space to keep the simplex
   constraints exact.

3. **Projected sub‑gradient descent on the exact L∞‑objective** – after the
   L‑BFGS phase we minimise the true max‑convolution using a simple
   sub‑gradient with a back‑tracking line‑search and explicit projection
   onto the probability simplex.  This phase runs for the bulk of the time
   budget.

4. **Upsampling and final polishing** – the small‑n solution is interpolated
   to the final length (n=4000) and polished with a short L‑BFGS cascade
   plus a lightweight simulated‑annealing hill‑climb.

All stages respect a global 1000 s deadline; the best sequence found is
stored in the global variable ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation (including a Beta family)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse symmetric probability vector using several families."""
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "beta",
        ]
    )
    x = np.arange(n)
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        # a_i ∝ exp(-rate * distance to nearest edge)
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        # a_i ∝ (dist+1)^(-p)
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "bimodal":
        # two symmetric Gaussian bumps near the edges
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        # a few random spikes
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        # 60 % uniform + 40 % edge‑exponential
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    else:  # beta
        # a_i ∝ x^{α‑1} (1‑x)^{β‑1}
        α = rng.uniform(0.1, 5.0)
        β = rng.uniform(0.1, 5.0)
        t = np.linspace(0.0, 1.0, n)
        a = (t ** (α - 1.0)) * ((1.0 - t) ** (β - 1.0))

    # avoid exact zeros, then normalise to a probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # ----- softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution via FFT -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Projected sub‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a back‑tracking line‑search and simplex projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # initial step size (in the simplex)
    step = 0.5
    it = 0

    while time.time() < deadline:
        # ----- convolution and mask for the active set -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # ----- sub‑gradient of max(conv) w.r.t. a -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = 2.0 * np.real(np.fft.ifft(fa_rev * fm))[n - 1 : n - 1 + n]

        improved = False
        # ----- simple back‑tracking line‑search -----
        for _ in range(10):
            a_candidate = a - step * grad
            a_candidate = project_to_simplex(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                step = min(step * 1.2, 1.0)      # modest step‑size increase
                improved = True
                print(f"[subgrad] iter={it} fast_score={best_score:.9f} step={step:.5e}")
                break
            else:
                step *= 0.5                       # shrink step and retry

        if not improved:
            # occasional random jitter to escape flat regions
            if rng.random() < 0.01:
                a = add_noise(a, sigma=0.001, rng=rng)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (kept from the previous version)
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n_small = 2000          # cheap size for the long optimisation phases
    n_final = 4000          # length of the final answer
    fft_len_small = next_pow2(2 * n_small - 1)
    fft_len_final = next_pow2(2 * n_final - 1)

    # ------------------------------------------------------------------
    # Timing allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    phase1_end = start_time + 0.15 * budget_s   # warm‑starts
    phase2_end = start_time + 0.35 * budget_s   # L‑BFGS on small n
    phase3_end = start_time + 0.75 * budget_s   # projected sub‑gradient

    # ------------------------------------------------------------------
    # Phase 1 – many cheap parametric warm‑starts (small n)
    # ------------------------------------------------------------------
    best_a_small = None
    best_score_small = float("inf")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n_small, rng)
        score = fast_score(a, fft_len_small)
        if score < best_score_small:
            best_score_small = score
            best_a_small = a.copy()
            print(f"[warm‑start] new best fast_score={best_score_small:.9f}")

    # fallback to uniform if nothing was generated (should never happen)
    if best_a_small is None:
        best_a_small = np.full(n_small, 1.0 / n_small)
        best_score_small = fast_score(best_a_small, fft_len_small)

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS cascade on the small problem
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_small = [
            0.7, 0.5, 0.35, 0.25, 0.15, 0.10,
            0.07, 0.05, 0.03, 0.02, 0.01,
            0.005, 0.003, 0.0015, 0.0008,
            0.0004, 0.0002, 0.0001, 5e-05,
            2e-05, 1e-05, 5e-06, 2e-06,
            1e-06, 5e-07, 2e-07, 1e-07
        ]
        best_a_small, best_score_small = lbfgs_cascade(
            best_a_small,
            lam_schedule_small,
            maxiter_per_lam=3000,
            deadline=phase2_end,
            fft_len=fft_len_small,
            n=n_small,
        )
        print(f"[L‑BFGS small] best fast_score={best_score_small:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – long projected sub‑gradient descent (still small n)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a_small, best_score_small = subgradient_descent(
            best_a_small,
            phase3_end,
            n_small,
            fft_len_small,
            rng,
        )
        print(f"[subgrad] best fast_score={best_score_small:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample to the final length
    # ------------------------------------------------------------------
    x_small = np.linspace(0.0, 1.0, n_small)
    x_final = np.linspace(0.0, 1.0, n_final)
    a_up = np.interp(x_final, x_small, best_a_small)
    a_up = np.maximum(a_up, 1e-12)
    a_up /= a_up.sum()

    # ------------------------------------------------------------------
    # Phase 5 – final polishing on the target size
    # ------------------------------------------------------------------
    # short L‑BFGS cascade with an even finer λ‑schedule
    if time.time() < deadline - 5.0:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        best_a_final, best_score_final = lbfgs_cascade(
            a_up,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline - 5.0,
            fft_len=fft_len_final,
            n=n_final,
        )
        print(f"[final L‑BFGS] fast_score={best_score_final:.9f}")
    else:
        best_a_final = a_up
        best_score_final = fast_score(best_a_final, fft_len_final)

    # ------------------------------------------------------------------
    # Phase 6 – very short simulated‑annealing finish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a_final, best_score_final = simulated_annealing_simple(
            best_a_final,
            best_score_final,
            deadline,
            n_final,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score_final:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a_final.tolist()
    final_eval = evaluate_sequence(final_seq)  # the provided security‑checked function
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
