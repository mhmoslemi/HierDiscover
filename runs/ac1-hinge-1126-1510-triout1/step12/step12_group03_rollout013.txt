def hill_climb(a, timeout, n, fft_len, rng):
    best = a.copy()
    best_score = fast_score(best, fft_len)
    while time.time() < timeout:

        cand[i] -= delta

        cand_score = fast_score(cand, fft_len)

            best_score = cand_score
    return best, best_score

"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
1. Choose a modest length (n = 500).  The theoretical lower bound for the
   objective is ~2·n/(2·n‑1) ≈ 1, so a good solution can easily beat 1.503.
2. Build a small collection of diverse initial candidates (uniform,
   triangular, random, down‑sampled previous best).  Keep the one with the
   smallest exact score 2·n·max(self‑convolution).
3. Iterate a linear‑programming improvement:
   – Let f be the current vector and M = max(f*f).
   – The linear constraint (f ∗ g)_k ≤ M is linear in g.  Build the
     (2·n‑1)×n Toeplitz matrix A such that A·g = f∗g.
   – Solve   maximise 1ᵀg   subject to A·g ≤ M, g ≥ 0.
   – Normalise the solution g to Σg = 1.
   – If max(g*g) < M replace f←g, otherwise mix f←(1‑α)f+αg
     (α≈0.3) and keep the better of the two.
4. After the LP phase perform a fast greedy hill‑climb:
   repeatedly move a random fraction of mass from one index to another
   and keep the move only if the exact score decreases.
5. Return the best vector found (as a plain list).  The final score is
   obtained with the official `evaluate_sequence` routine.

All steps respect the 1000 s budget and use only NumPy/SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def max_conv(a: np.ndarray, fft_len: int) -> float:
    """Return max entry of the linear self‑convolution."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random point on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        peak = rng.uniform(0.5, 1.5) / n
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def triangular_init(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build sparse matrix A (size (2n‑1)×n) such that A @ g = f ∗ g
    (linear convolution with zero padding).
    """
    n = f.shape[0]
    # total number of non‑zero entries = n * n
    N = n * n
    rows = np.empty(N, dtype=np.int32)
    cols = np.empty(N, dtype=np.int32)
    data = np.empty(N, dtype=np.float64)

    # fill by blocks
    idx = 0
    for i in range(n):
        rows[idx:idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx:idx + n] = i
        data[idx:idx + n] = f
        idx += n

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lp_improvement(f: np.ndarray, fft_len: int, rng: np.random.Generator,
                   deadline: float, max_iters: int = 20, alpha: float = 0.30):
    """
    Repeatedly solve the linear program
        maximise Σ g_j   s.t.   (f ∗ g) ≤ M , g ≥ 0
    where M = max(f ∗ f).  The candidate g is normalised and either
    replaces f (if it improves the max‑convolution) or is mixed with it.
    Returns the best vector found.
    """
    n = f.size
    best_f = f.copy()
    best_score = fast_score(best_f, fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = max_conv(best_f, fft_len)

        # build the convex‑convolution matrix for the current f
        A = build_conv_matrix(best_f)

        # linear programme: minimise -1ᵀg  (i.e. maximise Σg)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (sparse support)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # If the solver failed, abort the LP phase.
            print("[LP] linprog failed:", res.message)
            break

        g_raw = res.x
        s = g_raw.sum()
        if s <= 0.0:
            break
        g = g_raw / s                                      # normalise to Σg = 1

        # check whether g alone already improves the bound
        M_g = max_conv(g, fft_len)
        if M_g < M:
            best_f = g
            best_score = fast_score(best_f, fft_len)
            print(f"[LP] iteration {it}: direct improvement, new score {best_score:.8f}")
            continue

        # otherwise mix a fraction of g into f
        f_candidate = (1.0 - alpha) * best_f + alpha * g
        f_candidate = np.maximum(f_candidate, 0.0)
        f_candidate /= f_candidate.sum()
        M_new = max_conv(f_candidate, fft_len)
        if M_new < M:
            best_f = f_candidate
            best_score = fast_score(best_f, fft_len)
            print(f"[LP] iteration {it}: mixed improvement, new score {best_score:.8f}")
        else:
            # no improvement – shrink the mixing step a little
            alpha = max(0.05, alpha * 0.7)
        # continue
    return best_f, best_score


def hill_climb(a: np.ndarray, fft_len: int, deadline: float,
               rng: np.random.Generator, max_iter: int = 200000,
               delta_frac: float = 0.08):
    """
    Greedy hill‑climb that only accepts strictly improving mass‑transfer moves.
    """
    n = a.size
    best = a.copy()
    best_score = fast_score(best, fft_len)

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a random fraction (up to delta_frac) of mass from i to j
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # safeguard against negative due to rounding
        if cand[i] < -1e-12:
            it += 1
            continue
        cand_score = fast_score(cand, fft_len)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        it += 1
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a modest length makes the linear programme cheap.
    # ------------------------------------------------------------------
    n = 500                       # 500 gives a theoretical lower bound ≈1.0
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Interpolate a previously found solution if it exists.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(triangular_init(n))
    # Random families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Choose the best start (exact fast_score)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming driven improvement
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    best_a, best_score = lp_improvement(best_a, fft_len, rng,
                                        deadline=phase1_end,
                                        max_iters=30, alpha=0.30)
    print(f"[LP phase] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – simple greedy hill‑climb for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, fft_len,
                                        deadline, rng,
                                        max_iter=300000,
                                        delta_frac=0.07)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the sequence in the global namespace for possible reuse
    globals()['height_sequence_1'] = final_seq

    return final_seq
