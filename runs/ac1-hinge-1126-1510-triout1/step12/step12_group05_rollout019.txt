"""
serpentine‑optimiser – 1000 s budget (alternative)

Goal
----
Search a non‑negative step‑function (a list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The theoretical optimum for a
given length `n` is attained by a uniform distribution (score = 2).  The
reference solution beats this bound due to floating‑point quirks; this
implementation explores the search space with a different set of ideas:

* **Rich warm‑starts** – many parametric families (including sinusoidal,
  quadratic, and piecewise‑linear shapes) are sampled.
* **Exact sub‑gradient descent** – the true `L∞`‑objective is optimised by
  projected gradient steps on the probability simplex.
* **Heavy‑tailed simulated annealing** – a final stochastic hill‑climb
  removes remaining artefacts.
* **Time‑aware phases** – the 1000 s budget is split into three stages;
  the algorithm always respects the deadline.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT, which is much faster than
    ``np.convolve`` for large ``n``.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ_i x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if np.all(v <= 0):
        # All non‑positive – project to the vertex e_0
        out = np.zeros_like(v)
        out[0] = 1.0
        return out

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric warm‑starts (new families added)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse symmetric probability vector using several families."""
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "cosine",
            "quadratic",
            "piecewise",
        ]
    )
    x = np.arange(n, dtype=float)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * (1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == "cosine":
        # a_i ∝ 1 + cos(2πi/(n‑1)) – a smooth bump in the centre
        a = 1.0 + np.cos(2.0 * np.pi * x / (n - 1))

    elif mode == "quadratic":
        # a_i ∝ 1 - ((2i/(n‑1) – 1)²)  – a parabola opening downwards
        t = 2.0 * x / (n - 1) - 1.0
        a = 1.0 - t * t
        a = np.maximum(a, 0.0)

    else:  # piecewise – a random combination of a few linear ramps
        pts = sorted(rng.integers(0, n, size=rng.integers(3, 6)))
        heights = rng.random(len(pts) + 1) + 0.5
        a = np.empty_like(x)
        last = 0
        for i, pt in enumerate(pts):
            a[last:pt] = np.linspace(heights[i], heights[i + 1], pt - last, endpoint=False)
            last = pt
        a[last:] = np.linspace(heights[-2], heights[-1], n - last)

    # avoid exact zeros (log‑softmax would break), then normalise
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient step (projected onto the simplex)
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, n: int, fft_len: int, eta: float) -> np.ndarray:
    """
    One projected sub‑gradient step on the true L∞‑objective.
    a is a probability vector (sum = 1).
    """
    # convolution via FFT (exact up to round‑off)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)  # (2n‑1,) indicator

    # sub‑gradient of max(conv) w.r.t. a
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1 : n - 1 + n]                     # size n

    # take a step against the sub‑gradient and project back onto simplex
    a_next = a - eta * g
    a_next = project_onto_simplex(a_next)
    return a_next

# Simulated‑annealing hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator) -> (np.ndarray, float):
    """
    Simple Metropolis hill‑climb that randomly transfers a fraction of mass
    from one index to another.  Temperature decays linearly to zero.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    max_frac = 0.30               # how much mass may be moved in a single step

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = max(1e-12, cur_score * (1.0 - frac))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[anneal] new best fast_score={best_score:.9f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size works well for the time budget
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Time allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    phase1_end = start_time + 0.25 * budget_s   # warm‑starts
    phase2_end = start_time + 0.70 * budget_s   # projected sub‑gradient
    phase3_end = deadline                       # final annealing

    # ------------------------------------------------------------------
    # Phase 1 – generate many parametric warm‑starts
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # if a previous construction exists, use it as a seed
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] warm‑start from previous best fast_score={best_score:.9f}")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)

        # occasional jitter around the current best (helps escape plateaus)
        if best_a is not None and rng.random() < 0.20:
            sigma = rng.uniform(5e-4, 2e-3)
            noisy = a + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            a = noisy

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on the L∞‑objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    it = 0
    while time.time() < phase2_end:
        # simple step‑size schedule η_t = η₀ / sqrt(t+1)
        eta0 = 12.0
        eta = eta0 / np.sqrt(it + 1.0)

        a = subgradient_step(a, n, fft_len, eta)
        sc = fast_score(a, fft_len)

        if sc < cur_score:
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[sub‑grad] iter={it} new best fast_score={best_score:.9f}")

        # occasional random jitter to avoid stagnation
        if it % 1000 == 0 and rng.random() < 0.05:
            a = a + rng.normal(0.0, 5e-4, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            phase3_end,
            n,
            rng,
        )
        print(f"[final‑anneal] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # expose the result for possible warm‑restart in the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
