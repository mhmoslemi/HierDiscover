```python
import time, numpy as np, math
from scipy.optimize import minimize

def next_pow2(x):
    return 1 << ((x-1).bit_length())

def fast_score(a, fft_len):
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    return float(2.0 * n * conv.max())

def simplex_projection(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1.0)
    w = np.maximum(v - theta, 0.0)
    return w
```

```
def random_shape_candidate(n, rng):
    family = rng.choice(['power','exp','beta','mixed','lin'])
    if family == 'power':
        p = rng.uniform(0.2, 4.0)
        shift = rng.uniform(0.0, 5.0)
        arr = (np.arange(1, n+1) + shift) ** (-p)
    elif family == 'exp':
        lam = rng.uniform(0.0001, 0.05)
        arr = np.exp(-lam * np.arange(n))
    elif family == 'beta':
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        x = (np.arange(n) + 0.5) / n
        arr = (x**(a-1)) * ((1-x)**(b-1))
    elif family == 'mixed':
        w = rng.random()
        lam1 = rng.uniform(0.0001, 0.05)
        lam2 = rng.uniform(0.0001, 0.05)
        arr = w * np.exp(-lam1*np.arange(n)) + (1-w)*np.exp(-lam2*np.arange(n))
    else:  # lin
        slope = rng.uniform(-3.0, 0.0)
        intercept = rng.uniform(0.0, 2.0)
        arr = intercept + slope * np.arange(n)
        arr = np.maximum(arr, 1e-9)
    arr = np.maximum(arr, 0.0)
    if arr.sum() == 0:
        arr = np.full(n, 1.0/n)
    a = arr / arr.sum()
    return a
```

"""
Step‑function optimiser – 1000 s budget (improved)

The optimiser is a multi‑phase pipeline:

1. **Parametric random‑shape search** – generate a huge variety of
   probability vectors from simple analytic families (power law,
   exponential, Beta, mixtures, linear).  Keep the best according to the
   fast surrogate 2 n·max(conv).

2. **Smoothed ‑∞ optimisation** – minimise a log‑sum‑exp surrogate of the
   true ‑∞‑objective with L‑BFGS.  A decreasing λ‑schedule (down to 5e‑8)
   and many internal iterations tighten the solution.

3. **Multiplicative (exponentiated) gradient descent** – mirror‑descent on
   the exact max‑convolution using an adaptive 1/√t step size.

4. **Additive projected gradient descent** – classic sub‑gradient descent
   on the exact max‑convolution, followed by Euclidean projection onto the
   simplex (simplex_projection).

5. **Fine L‑BFGS** – another short L‑BFGS run with ultra‑small λ to polish
   the result.

6. **Greedy mass‑transfer refinement** – hill‑climbing moves that
   transfer a random fraction of mass from one position to another and
   keep only improving steps.

All phases respect a global 1000 s deadline.  The best sequence found
throughout is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return the exact 2·n·max(conv(a,a)) value (sum(a) assumed = 1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Shape families for the random‑shape phase
# ----------------------------------------------------------------------


def random_power_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 4.0)
    shift = rng.uniform(0.0, 5.0)
    arr = (np.arange(1, n + 1, dtype=float) + shift) ** (-p)
    arr = np.maximum(arr, 0.0)
    return arr / arr.sum()


def random_exponential_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(1e-4, 5e-2)  # decay rate
    arr = np.exp(-lam * np.arange(n, dtype=float))
    return arr / arr.sum()


def random_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    x = (np.arange(n, dtype=float) + 0.5) / n  # avoid exact 0/1
    arr = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    arr = np.maximum(arr, 0.0)
    return arr / arr.sum()


def random_mixed_exponential_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    lam1 = rng.uniform(1e-4, 5e-2)
    lam2 = rng.uniform(1e-4, 5e-2)
    arr = w * np.exp(-lam1 * np.arange(n, dtype=float)) + (1.0 - w) * np.exp(
        -lam2 * np.arange(n, dtype=float)
    )
    return arr / arr.sum()


def random_linear_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    slope = rng.uniform(-3.0, 0.0)  # decreasing line
    intercept = rng.uniform(0.0, 2.0)
    arr = intercept + slope * np.arange(n, dtype=float)
    arr = np.maximum(arr, 1e-9)
    return arr / arr.sum()


def random_shape_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random family and generate a normalised probability vector."""
    fam = rng.choice(
        [
            random_power_shape,
            random_exponential_shape,
            random_beta_shape,
            random_mixed_exponential_shape,
            random_linear_shape,
        ]
    )
    return fam(n, rng)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y are free variables (unconstrained); a = softmax(y) lives on the simplex.
    """
    # softmax → probability vector
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT (real‑valued)
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c           # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)  # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through a*a → a (use FFT for speed)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]   # size n, gradient w.r.t. a

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS problems for a decreasing λ‑schedule.
    Returns the best vector (normalised) and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 20.0,
):
    """
    Mirror (multiplicative) sub‑gradient descent on the exact max‑convolution.
    Update: a ← a · exp(−η·g)   then project onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clipped for safety)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def additive_gradient_optimize(

    n: int,
    fft_len: int,
    rng: np.random.Generator,

    Update: a ← a − η·g   then Euclidean projection onto the simplex.
    """

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:

        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * g
        a = simplex_projection(a)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def local_refinement(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """
    Simple hill‑climbing: repeatedly transfer a random fraction of mass
    from one position to another. Only moves that strictly improve the
    `2·n·max(conv)` score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20  # up to 20 % of source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

        fine_lam_sched = [5e-8, 2e-8, 1e-8, 5e-9, 2e-9, 1e-9]
        a_fine, sc_fine = lbfgs_cascade(
            best_a,
            fine_lam_sched,
            maxiter_per_lam=3000,
            deadline=deadline - 20.0,
            fft_len=fft_len,
            n=n,
        )
        if sc_fine < best_score:
            best_score, best_a = sc_fine, a_fine.copy()
            print("[fine‑L‑BFGS] fast_score =", best_score)

    # ------------------------------------------------------------------
    # Phase 6 – greedy local refinement (the last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        print("[local‑refine] fast_score =", best_score)

    # ------------------------------------------------------------------
    # Final evaluation with the provided scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # evaluate_sequence is pre‑imported
    print("[final] evaluated score =", final_eval)

    # store for potential reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
