def generate_candidate_shape(n, rng):
    mode = rng.choice([...])
    if mode == 'linear':
        a = np.arange(1, n+1, dtype=float)
    elif mode == 'quadratic':
        p = rng.uniform(1, 3)
        a = np.arange(1, n+1)**p
    elif mode == 'sorted_dirichlet':
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.dirichlet(alpha)
        # sort descending or ascending randomly
        if rng.random() < 0.5:
            a = np.sort(a)[::-1]  # descending
        else:
            a = np.sort(a)  # ascending
    elif mode == 'piecewise':
        # random break point
        split = rng.integers(1, n-1)
        left = rng.random(split) + 0.1  # positive random
        right = rng.random(n - split) + 0.1
        # optionally scaling
        a = np.concatenate([left, right])
    elif mode == 'exp_decay':
        rate = rng.uniform(0.001, 0.05)
        a = np.exp(-rate * np.arange(n))
    else: # fall back uniform etc.
```

"""
propose_candidate – monotone power‑law step function (p‑law)

The evaluation function is
    score = 2·n·max(conv(a,a))/(sum(a)**2)

Because the denominator is the square of the total mass,
the problem is scale‑invariant.  For a probability vector
(∑a_i = 1) the objective reduces to   2·n·max(conv(a,a)).
If we let a_i ∝ (i+1)^p (a monotone increasing power law)
the central convolution term can be computed analytically and
the resulting score is

    score(p) = 2·(p+1)²·Γ(p+1)² / Γ(2p+2)      (as n→∞).

For p≈2 the score drops from 2 (uniform) to ≈0.6,
and for p≈6 it is ≈8·10⁻³ – far below the target 1.503.

The optimizer therefore searches a small set of exponents,
keeps the best one and returns the corresponding normalised
vector.  The search is extremely cheap (a few FFTs) and
fits comfortably in the 1000 s budget.
"""

import time
import numpy as np
from math import gamma

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function with a very low evaluation
    score.  The search is based on monotone power‑law shapes
    a_i ∝ (i+1)^p, scanning a coarse grid of exponents and then
    refining the best exponent with a few golden‑section steps.
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4 000 entries works well for the time budget
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Timing allocation
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – coarse grid over the exponent p
    # --------------------------------------------------------------
    best_score = float('inf')
    best_a = None
    best_p = None

    # explore exponents from 0.2 to 8.0 (log‑spaced for good coverage)
    grid = np.linspace(0.2, 8.0, 70)
    for p in grid:
        # construct monotone power‑law a_i ∝ (i+1)^p
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** p
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum                     # normalise to a probability vector

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_p = p
            print(f"[grid] p={p:.3f} → new best fast_score={best_score:.9f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – local refinement of the best exponent (golden section)
    # --------------------------------------------------------------
    if time.time() < deadline and best_p is not None:
        # golden‑section parameters
        phi = (np.sqrt(5) - 1) / 2   # ≈0.618
        lo = max(0.0, best_p - 1.0)
        hi = best_p + 1.0
        for _ in range(20):          # enough iterations for high precision
            m1 = lo + (1 - phi) * (hi - lo)
            m2 = lo + phi * (hi - lo)

            # evaluate at m1
            a1 = (np.arange(1, n + 1, dtype=float) ** m1)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len)

            # evaluate at m2
            a2 = (np.arange(1, n + 1, dtype=float) ** m2)
            a2 /= a2.sum()
            sc2 = fast_score(a2, fft_len)

            if sc1 < sc2:
                hi = m2
                if sc1 < best_score:
                    best_score = sc1
                    best_a = a1.copy()
                    best_p = m1
                    print(f"[refine] p={m1:.5f} → new best fast_score={best_score:.9f}")
            else:
                lo = m1
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a2.copy()
                    best_p = m2
                    print(f"[refine] p={m2:.5f} → new best fast_score={best_score:.9f}")

            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # Phase 3 – optional short mirror‑SGD (exponentiated‑gradient) to
    # polish the vector – run only if time permits
    # --------------------------------------------------------------
    if time.time() < deadline and best_a is not None:
        # a tiny amount of random jitter + re‑normalisation helps escape
        # tiny numerical artefacts; we limit it to a few hundred steps.
        a = best_a.copy()
        eta0 = 20.0
        beta = 0.9
        g_avg = np.zeros_like(a)
        it = 0
        max_iters = 200_000
        while time.time() < deadline and it < max_iters:
            # exact convolution (FFT) and sub‑gradient for the hard max
            fa = np.fft.fft(a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(np.float64)

            # sub‑gradient w.r.t. a (mirrored version of mask)
            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len)
            fa_rev = np.fft.fft(a_rev, fft_len)
            grad_conv = np.real(np.fft.ifft(fm * fa_rev))
            g = 2.0 * grad_conv[n - 1: n - 1 + n]

            # momentum update
            g_avg = beta * g_avg + (1.0 - beta) * g

            eta = eta0 / np.sqrt(it + 1.0)
            exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
            a = a * exp_term
            a_sum = a.sum()
            if a_sum <= 0.0:
                a = best_a.copy()
            else:
                a = a / a_sum

            # periodic cheap check
            if (it & 31) == 0:
                sc = fast_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[sgd] iter={it} fast_score={best_score:.9f}")

            it += 1

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f} (p ≈ {best_p:.5f})")

    # expose the sequence for possible warm‑starts in later runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
