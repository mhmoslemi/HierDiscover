"""
search for a step‑function (non‑negative vector) that minimizes
evaluate_sequence(sequence) = 2·n·max(conv(a,a)) / (sum a)^2.

The algorithm combines:
  • a rich set of deterministic & random initial guesses,
  • L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution,
  • Adam on the same smoothed objective,
  • exponentiated‑gradient descent on the exact max‑convolution,
  • a dedicated exact sub‑gradient method with line‑search,
  • simulated‑annealing and a final greedy hill‑climb.

All vectors are kept on the probability simplex (sum = 1) – the
objective is scale‑invariant – and the implementation works for any
length *n*.  Using n=8000 (instead of 5000) gives the optimizer a finer
grid to flatten the autocorrelation and typically yields a lower bound.
"""
import time
import numpy as np
from scipy.optimize import minimize

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast evaluation (exact for probability vectors)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(conv) for a probability vector a (sum = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient for L‑BFGS/Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution using log‑sum‑exp."""
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. the smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def random_phase_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Construct a vector with (approximately) flat Fourier magnitude:
    draw random phases, invert, shift to non‑negative and renormalise.
    """
    L = next_pow2(2 * n - 1)
    # random phases respecting Hermitian symmetry
    phases = rng.random(L // 2 - 1) * 2 * np.pi
    ft = np.ones(L, dtype=np.complex128)
    ft[1:L // 2] = np.exp(1j * phases)
    ft[-(L // 2 - 1):] = np.conj(ft[1:L // 2])
    a_raw = np.fft.ifft(ft).real[:n]
    a_raw -= a_raw.min()               # make non‑negative
    a_raw += 1e-12                     # avoid exact zeros (helps simplex)
    a_raw /= a_raw.sum()
    return a_raw

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS for every λ in lam_schedule, carrying the logits forward."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.07):
    """Adam optimisation of the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """Exponentiated‑gradient descent on the exact max‑convolution (focus on top entries)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient via convolution with reversed a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def exact_max_subgrad_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               max_line_search: int = 5):
    """
    Projected sub‑gradient descent on the *exact* max entry of the
    self‑convolution, with a simple back‑tracking line‑search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    # pre‑compute powers of two for fast convolution
    while time.time() < deadline:
        # compute full self‑convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k0 = int(conv.argmax())               # index of the peak

        # sub‑gradient of the peak entry w.r.t. a
        g = np.zeros_like(a)
        low = max(0, k0 - (n - 1))
        high = min(k0, n - 1)
        idx = np.arange(low, high + 1)
        g[idx] = 2.0 * a[k0 - idx]

        # zero‑sum projection (stay on simplex after renormalisation)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # simple geometric line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            a_candidate = a + factor * d
            a_candidate = np.clip(a_candidate, 0.0, None)
            s = a_candidate.sum()
            if s <= 0:
                continue
            a_candidate /= s
            sc = fast_score(a_candidate, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
        if not improved:
            # if nothing improved, shrink step permanently
            # (this simple scheme is enough for the allotted time)
            break
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """Simple SA with mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.20   # allow larger jumps
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb (strictly improving mass transfers)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step sequence within the given time budget.
    Returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # --------------------------------------------------------------
    n = 8000                     # a bit larger than the reference 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a rich pool of initial candidates.
    # --------------------------------------------------------------
    candidates = []

    # Warm‑start from a previous run (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(5):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(3):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Extra systematic explorations
    for _ in range(150):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))

    for _ in range(80):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    for _ in range(30):
        candidates.append(gaussian_mixture_init(n, rng, components=rng.integers(2, 6)))

    for _ in range(30):
        candidates.append(random_phase_init(n, rng))

    # --------------------------------------------------------------
    # Pick the best candidate by exact fast_score.
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈35 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # Occasionally restart from a fresh random vector to escape stagnation.
        if rng.random() < 0.25:
            a_start = random_initial(n, rng)
        else:
            # blend current best with a fresh random vector
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.5          # up‑to‑50 % fresh mass
            a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of total time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=8000,
                                           lam_init=0.30,
                                           lam_decay=0.9992,
                                           lr=0.07)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (≈15 % of total time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.02)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – exact‑max sub‑gradient with line‑search (≈10 % of total time)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.80 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = exact_max_subgrad_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        max_line_search=5)
        print(f"[exact‑sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-7,
                                                 max_moves=250000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=120000,
                                        delta_frac=0.07)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose for potential future runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
