"""
Improved search for a low‑evaluation non‑negative step function.

The algorithm proceeds in three stages:
1️⃣  A short L‑BFGS cascade on a log‑sum‑exp surrogate to get a decent
   starting point from a diverse pool (random, window‑functions, previous best).
2️⃣  Mirror‑Adam (exponentiated‑gradient with Adam momentum) directly on the
   true max‑convolution objective.  The parameters are kept in log‑space,
   a softmax guarantees non‑negativity and a unit sum.  The step‑size is
   adapted automatically, and the best vector seen is recorded.
3️⃣  A final low‑λ L‑BFGS polish followed by an optional brief simulated‑annealing
   refinement.

The whole search respects a 1000 s deadline and keeps the best sequence
found in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution and gradient w.r.t. log‑variables y
    (a = softmax(y)).
    """
    # ---- softmax ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution (FFT) ----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑Adam on the true max‑convolution objective
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def adam_optimize(
    y0: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_sigma: float = 1e-3,
):
    """
    Mirror‑Adam (exponentiated‑gradient with Adam momentum) on the
    exact L∞‑objective.  Works in log‑space; the iterates are kept
    normalised via a softmax.
    """
    y = y0.copy()
    m1 = np.zeros_like(y)
    m2 = np.zeros_like(y)
    beta1_pow = beta1
    beta2_pow = beta2
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        # ----- current vector and its convolution -----
        a = softmax(y)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- sub‑gradient of max(conv) -----
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a[::-1], fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g_a = 2.0 * grad_conv[:n]                 # size n

        # ----- gradient w.r.t. log‑variables -----
        dot_ag = np.dot(a, g_a)
        grad_y = a * (g_a - dot_ag)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_y
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_y ** 2)

        m1_hat = m1 / (1.0 - beta1_pow)
        m2_hat = m2 / (1.0 - beta2_pow)

        y -= lr * m1_hat / (np.sqrt(m2_hat) + eps)

        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----- bookkeeping -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[Adam] it={it:,} fast_score={best_score:.9f}")

        # periodic symmetrisation (helps to keep the shape balanced)
        if (it & 511) == 0:
            a_sym = symmetrize(a)
            score_sym = fast_score(a_sym, fft_len)
            if score_sym < best_score:
                best_score = score_sym
                best_a = a_sym.copy()
                # re‑initialise log‑variables from the symmetrised vector
                y = np.log(a_sym + 1e-12)
                m1.fill(0.0)
                m2.fill(0.0)
                beta1_pow = beta1
                beta2_pow = beta2
                print(f"[Adam‑sym] it={it:,} fast_score={best_score:.9f}")

        # jitter if stagnation lasts too long
        if (it - last_improve) > 150_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            y = np.log(a + 1e-12)
            m1.fill(0.0)
            m2.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb for a final polish
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # fast_score using the already defined helper (FFT based)
        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate dimension keeps the FFT cheap while
    # still allowing a flexible shape.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Use the previous best if present (interpolated)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Classical windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman

        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        w = hann(n)
        pool.append(w / w.sum())
        w = blackman(n)
        pool.append(w / w.sum())
    except Exception:
        pass

    # 3) Randomly generated families
    for _ in range(15):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.15 * budget_s   # short L‑BFGS warm‑up
    phase2_end = start_time + 0.80 * budget_s   # long Adam phase
    # remaining time → final L‑BFGS polish + tiny anneal

    # ------------------------------------------------------------------
    # Phase 1 – rapid L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6,
        0.4,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.022,
        0.012,
        0.006,
        0.003,
        0.0015,
        7e-4,
        3e-4,
        1.5e-4,
        8e-5,
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.4:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.015), rng=rng)
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # enforce symmetry once – often useful
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Mirror‑Adam optimisation on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            y0,
            phase2_end,
            n,
            fft_len,
            rng,
            lr=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        print(f"[Phase2‑Adam] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polish + optional anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    # tiny moving‑average smoothing before annealing (helps a little)
    if time.time() < deadline:
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        s_score = fast_score(smoothed, fft_len)
        if s_score < best_score:
            best_a = smoothed
            best_score = s_score
            print(f"[Phase3‑smooth] fast_score={best_score:.9f}")

    # short simulated‑annealing hill‑climb
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
