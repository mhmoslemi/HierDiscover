"""
Propose a low‑evaluation step‑function.

The search works on a moderately large problem (n≈8000) and consists of
three phases:

1. **Coarse multi‑restart L‑BFGS** on the smooth log‑sum‑exp
   surrogate.  The λ‑schedule goes from 0.5 down to 1e‑8; each restart
   runs a cascade of short L‑BFGS optimisations (max ≈ 1500 iterations
   per λ).  The best vector from many random starts is kept.

2. **Long exponentiated‑gradient (mirror‑SGD)** on the true
   L∞‑objective.  The update uses an adaptive step‑size η_t≈η₀/√t and
   occasional Gaussian jitter to escape plateaus.  This phase runs
   until the allocated time budget is exhausted.

3. **Final polishing** – a short low‑λ L‑BFGS cascade followed by a
   tiny moving‑average smoothing and a brief simulated‑annealing hill‑climb.

All phases respect the global 1000 s deadline.  The final sequence is
evaluated with the official `evaluate_sequence` and stored in the
global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import chebwin, kaiser, hann, blackman, tukey, gaussian

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when sum(a)=1:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # ---- softmax ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution (FFT) ----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # effective smoothing parameter
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  verbose: bool = False):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 20_000_000,
                          jitter_every: int = 150_000,
                          verbose: bool = False):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    Runs until the deadline or ``max_iters`` is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of entries attaining the max (or near‑max)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # periodic evaluation
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                if verbose:
                    print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter to escape stagnation
        if (it - last_improve_it) > jitter_every:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            last_improve_it = it

        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator,
                               verbose: bool = False):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose:
                    print(f"[Anneal] new best fast_score={best_score:.9f}")

    return best_a, best_score

            max_iters=30_000_000,
            jitter_every=120_000,
            verbose=False,
        )
        print(f"[Phase 2] exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Very low‑λ L‑BFGS refinement
        fine_lam_schedule = [5e-9, 2e-9, 1e-9, 5e-10, 2e-10, 1e-10]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
            verbose=False,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase 3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # Tiny moving‑average smoothing before annealing
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        smoothed_score = fast_score(smoothed, fft_len)
        if smoothed_score < best_score:
            best_a = smoothed
            best_score = smoothed_score
            print(f"[Phase 3‑smooth] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
            verbose=False,
        )
        print(f"[Phase 3‑Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    # Store for possible reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
