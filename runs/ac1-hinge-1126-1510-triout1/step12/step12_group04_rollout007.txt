def propose_candidate(seed=43, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    n = 5000
    fft_len = next_pow2(2*n - 1)

    # obtain starting a
    if "height_sequence_1" in globals():
        a_start = np.array(globals()["height_sequence_1"], dtype=float)
        a_start = np.maximum(a_start, 0.0)
        a_start = a_start / a_start.sum()
    else:
        a_start = np.full(n, 1/n)

    # create additional random starts to allow restarts
    pool = [a_start]
    for _ in range(5):
        pool.append(random_initial(n, rng))

    best_a = None
    best_score = float('inf')
    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    # Phase 0: quick L-BFGS warm start on a few random pool vectors to get good starting a
    # We'll run a few L-BFGS cascades with moderate lambdas
    lam_schedule = [0.05, 0.02, 0.008, 0.003, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    for a0 in pool:
        if time.time() >= deadline: break
        a_opt, sc_opt = lbfgs_cascade(a0, lam_schedule, maxiter_per_lam=1200,
                                      deadline=deadline, fft_len=fft_len, n=n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Warm] new best fast_score={best_score:.9f}")

    # Phase 1: Adam optimization on smoothed max with decreasing lambda
    # lambda schedule for this phase
    lambdas = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    # parameters for Adam
    alpha = 0.05   # base learning rate for y
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Convert best_a to log-space
    y = np.log(best_a + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for lam in lambdas:
        # decide number of steps for this lam based on remaining time
        # e.g., target ~30,000 steps per lambda but limited by time
        steps_target = 50000
        for _ in range(steps_target):
            if time.time() >= deadline:
                break
            # compute objective and gradient
            f_val, grad_y = objective_and_grad(y, n, fft_len, lam)
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y = y - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

            # evaluate every 500 steps
            if (t % 500) == 0:
                a = np.exp(y - np.max(y))
                a = a / a.sum()
                sc = fast_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[Adam] λ={lam:.1e} iter={t} best={best_score:.9f}")
            # optional occasional symmetrize to encourage balanced shape
            # we may skip to speed

    # Phase 2: final short exponentiated gradient (hard max) for fine tuning
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(best_a, deadline, n,
                                                   fft_len, rng,
                                                   eta0=15.0, max_iters=5_000_000)

    # symmetrize and evaluate
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len)
    print(f"[FinalFast] fast_score={best_score:.9f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[FinalEval] score={final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq

"""
Propose a low‑evaluation step‑function.

The algorithm consists of three stages:
1.  A quick L‑BFGS cascade on a log‑sum‑exp (smooth max) surrogate.
    This yields a good starting shape from several random restarts.
2.  An Adam optimiser that works directly on the *hard* max‑convolution
    using the sub‑gradient (mask of the maximal entries).  Adam’s adaptive
    learning‑rate provides faster convergence than the plain exponentiated‑gradient
    used previously.
3.  A final short L‑BFGS refinement (very small λ) and a symmetry‑enforcement
    pass before the official evaluation.

All stages respect the global 1000 s budget; the best vector found is stored
in the global ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution and its gradient w.r.t. the log‑variables ``y``
    (where ``a = softmax(y)``).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # scale lambda relative to the magnitude of the max value
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade (few λ values, modest iteration count)
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase together with its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} new best fast_score={best_score:.9f}")

    return best_a, best_score

# Adam optimiser on the hard‑max (sub‑gradient) objective
# ----------------------------------------------------------------------


def adam_subgrad_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 0.12,
                          max_iters: int = 15_000_000):
    """
    Adam optimiser on the true L∞‑objective.
    Works in log‑space (y = log a) to keep the iterate on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        # reconstruct a from y
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # gradient w.r.t. log‑variables y
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning rate schedule – start with eta0 and decay slowly
        lr = eta0 / np.sqrt(it + 1.0)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation
        if (it & 31) == 0:   # every 32 iterations
            a_cur = np.exp(y - np.max(y))
            a_cur /= a_cur.sum()
            cur_score = fast_score(a_cur, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()
                last_improve_it = it
                print(f"[Adam‑sub] it={it} fast_score={best_score:.9f}")

        # jitter when stuck
        if (it - last_improve_it) > 300_000:
            # add gentle isotropic noise in log‑space
            y = y + rng.normal(0.0, 0.02, size=y.shape)
            last_improve_it = it

    return best_a, best_score

    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter before optimisation
        if rng.random() < 0.3:
            a_start = add_noise(a_start, sigma=rng.uniform(0.0005, 0.005), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule_phase1,
            maxiter_per_lam=1300,
            deadline=phase1_end,
            fft_len=fft_len,
            n=n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:                    # fallback – uniform vector
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the true max‑convolution.
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_subgrad_optimize(
            best_a,
            deadline=phase2_end,
            n=n,
            fft_len=fft_len,
            rng=rng,
            eta0=0.12,
            max_iters=20_000_000,
        )
        print(f"[Phase2] Adam‑sub best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polish + symmetry enforcement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_sched = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_sched,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    # final symmetry pass (often beneficial)
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len)
    print(f"[FinalFast] fast_score after symmetrize={best_score:.9f}")

    # --------------------------------------------------------------
    # Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[FinalEval] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
