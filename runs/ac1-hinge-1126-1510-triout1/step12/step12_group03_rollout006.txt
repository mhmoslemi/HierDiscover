"""
Search for a non‑negative step function that minimizes
      2·n·max(self‑convolution) / (∑ a_i)²

The algorithm works in several phases:

1. Build a diverse pool of initial candidates (window functions,
   spectral guess, randomized families, exponential decays, etc.).
2. Keep the best candidate according to the *exact* fast_score.
3. Phase‑1 – many short L‑BFGS continuations with a decreasing
   smooth‑max parameter λ.  Each continuation starts from a blend of
   the current best vector and a fresh random vector, and after every
   λ‐stage we keep the best vector seen.
4. Phase‑2 – Adam optimisation on the smoothed max (very small λ).
5. Phase‑3 – Projected sub‑gradient descent on the *exact* max entry.
6. Phase‑4 – Simulated annealing with mass‑transfer moves.
7. Phase‑5 – Greedy hill‑climb for final polishing.
8. Return the best vector found (as a plain list) and print the final
   official evaluation.

The whole search respects the given time budget (default 1000 s).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast objective (sum = 1 → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess (inverse‑FFT of sqrt of uniform PSD)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random distributions on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'logistic'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # logistic
        loc = rng.uniform(0, n)
        scale = rng.uniform(0.1 * n, 0.3 * n)
        i = np.arange(n, dtype=float)
        a = 1.0 / (1.0 + np.exp(-(i - loc) / scale))
        a = a - a.min()
        a = a / a.max()
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a /= a.sum()
    return a


def exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0005, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def reverse_exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0005, 0.02)
    a = np.exp(-rate * np.arange(n - 1, -1, -1))
    a /= a.sum()
    return a


def symmetric_double_exp_init(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0005, 0.02)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * np.abs(i - centre))
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       fft_len_g,
                       n):
    """
    Run a sequence of L‑BFGS optimisations over a decreasing λ schedule.
    Returns the best vector and its exact fast_score observed during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent on the exact max entry of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a (2 * a[idx - j] where defined)
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)

        # simple backtracking line‑search
        improved = False
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 500000):
    """
    Simulated annealing with mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer up to 15 % of source mass
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 150000,
               delta_frac: float = 0.09):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ------------------------------------------------------------------
    # Problem size – stick to n = 5000 (a good trade‑off)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families and exponentials
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(3):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))
    for _ in range(3):
        candidates.append(exponential_init(n, rng))
    for _ in range(2):
        candidates.append(reverse_exponential_init(n, rng))
    for _ in range(2):
        candidates.append(symmetric_double_exp_init(n, rng))

    # Choose the best start according to the *exact* fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – abundant L‑BFGS continuations (≈45 % of time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 800   # fairly short but many restarts

    while time.time() < phase1_end:
        # fresh start: blend current best with a random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

        # also try a symmetric version (may help)
        sym = (a_opt + a_opt[::-1]) * 0.5
        sym /= sym.sum()
        sc_sym = fast_score(sym, fft_len, n)
        if sc_sym < best_score:
            best_score = sc_sym
            best_a = sym.copy()
            print(f"[symmetrised] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.08,
                                           max_iter=25000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact‑max sub‑gradient descent (≈15 % time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.75 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (≈15 % time)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.90 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase4_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-8,
                                                 max_moves=600000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=200000,
                                        delta_frac=0.09)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
