def adam_optimize(a_start, deadline, n, fft_len, rng, lr0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iters=int(1e7)):
    w = np.log(a_start + 1e-24)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    best_a = a_start
    best_score = fast_score(best_a, fft_len)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a from w via softmax
        # stable softmax
        mx = np.max(w)
        e = np.exp(w - mx)
        a = e / np.sum(e)
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # mask for max positions (tolerance relative)
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)
        # gradient wrt a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_a = 2.0 * grad_conv[n-1:n-1+n]   # shape n
        # gradient wrt w
        mean_g = np.dot(a, g_a)
        grad_w = a * (g_a - mean_g)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w = w - lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # track best
        if (t & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t} fast_score={best_score:.9f}")
    return best_a, best_score

"""
Optimization of step‑function heights to minimise the L∞ convolution bound.

The algorithm combines:
  * diverse parametric warm‑starts,
  * a cascade of L‑BFGS on a log‑sum‑exp smoothed objective,
  * a long Adam optimiser on the exact max‑convolution sub‑gradient,
  * a quick simulated‑annealing hill‑climb,
  * and a final low‑λ L‑BFGS polish.

All phases respect a global 1000 s deadline.  The best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list of
floats.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score for a probability vector a (sum(a)==1):
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# parametric warm‑starts
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse (possibly asymmetric) probability vector."""
    mode = rng.choice(
        ['uniform', 'exp_edge', 'power_edge', 'bimodal', 'sparse', 'mixed',
         'ramp', 'cosine']
    )
    x = np.arange(n, dtype=float)
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_edge':
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == 'power_edge':
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == 'bimodal':
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == 'sparse':
        m = rng.integers(int(0.015 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == 'mixed':
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == 'ramp':
        # strictly decreasing linear ramp, then flipped to keep some mass on both sides
        a = np.linspace(1.0, 0.0, n)

    else:   # cosine (smooth bump centred)
        a = 0.5 * (1.0 + np.cos(np.pi * (x - (n - 1) / 2) / ((n - 1) / 2)))
        a = np.maximum(a, 0.0)

    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smooth approximation of the max‑convolution objective and its gradient.
    The underlying probability vector is a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z               # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # length n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations using a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr0: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iters: int = 10_000_000):
    """
    Adam (first‑order) optimisation on the exact L∞ objective.
    Parameters are the usual Adam hyper‑parameters.
    """
    w = np.log(a_start + 1e-24)               # log‑space to enforce positivity
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # softmax to obtain a
        mx = np.max(w)
        e = np.exp(w - mx)
        a = e / e.sum()

        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # mask for active (max) indices
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # gradient of max conv w.r.t. a (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g_a = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # gradient w.r.t. log‑parameters w
        mean_g = np.dot(a, g_a)
        grad_w = a * (g_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w = w - lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # periodic score check
        if (t & 31) == 0:  # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t} fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

Potential pitfalls:

- In warm start interpolation, we need to avoid division by zero. Already check.

- In generate_candidate_shape, ramp with a descending linear may produce zeros at end, after normalization some values may be zero - but we enforce min 1e-12, then sum normalized still >0. So ok.

- Use of `np.exp(y - mx)` may overflow if y large. But w from log of probabilities; each w values likely moderate (around -8), exponent stable.

- Adam may cause w to become large magnitude, potentially causing overflow in exp. But we use safe clip by subtract max before exp, so stable.

"""
Optimization of step‑function heights to minimise the L∞ convolution bound.

The algorithm combines:
  * diverse parametric warm‑starts,
  * a cascade of L‑BFGS on a log‑sum‑exp smoothed objective,
  * a long Adam optimiser on the exact max‑convolution sub‑gradient,
  * a quick simulated‑annealing hill‑climb,
  * and a final low‑λ L‑BFGS polish.

All phases respect a global 1000 s deadline. The best sequence found is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score for a probability vector a (sum(a)==1):
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# parametric warm‑starts
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse (possibly asymmetric) probability vector."""
    mode = rng.choice(
        ['uniform', 'exp_edge', 'power_edge', 'bimodal', 'sparse', 'mixed',
         'ramp', 'cosine']
    )
    x = np.arange(n, dtype=float)

    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_edge':
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == 'power_edge':
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == 'bimodal':
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == 'sparse':
        m = rng.integers(int(0.015 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == 'mixed':
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == 'ramp':
        a = np.linspace(1.0, 0.0, n)

    else:   # cosine
        a = 0.5 * (1.0 + np.cos(np.pi * (x - (n - 1) / 2) / ((n - 1) / 2)))
        a = np.maximum(a, 0.0)

    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a

        if best_a is not None and rng.random() < 0.12:
            a = add_noise(a, sigma=rng.uniform(0.001, 0.006), rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS cascade (moderately fine λ‑schedule)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.15, 0.10,
            0.07, 0.05, 0.03, 0.02, 0.01,
            0.005, 0.003, 0.0015, 0.0008,
            0.0004, 0.0002, 0.0001, 5e-05, 1e-05,
            5e-06, 2e-06, 1e-06, 5e-07
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam=2500,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a,
            deadline=phase3_end,
            n=n,
            fft_len=fft_len,
            rng=rng,
            lr0=0.07,
            max_iters=12_000_000
        )
        print(f"[Adam] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – final annealing + tiny L‑BFGS polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # short simulated‑annealing refinement
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

        # final ultra‑fine L‑BFGS (λ as low as 5e‑7)
        if time.time() < deadline:
            fine_lambda = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08]
            a_ref, sc_ref = lbfgs_cascade(
                best_a,
                fine_lambda,
                maxiter_per_lam=3000,
                deadline=deadline,
                fft_len=fft_len,
                n=n,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_a = a_ref.copy()
                print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
