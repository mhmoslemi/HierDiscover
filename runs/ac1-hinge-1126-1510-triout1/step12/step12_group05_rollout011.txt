def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4000
    fft_len = next_pow2(2*n-1)
    deadline = time.time() + budget_s - 0.05
    phase1_end = time.time() + 0.4 * budget_s  # allocate 40%
    best_a = None
    best_score = np.inf

    # baseline: uniform for reference
    uniform = np.full(n, 1.0/n)
    best_a, best_score = uniform, fast_score(uniform, fft_len)

    while time.time() < phase1_end:
        # random m
        m = rng.integers(150, 301)   # choose between 150 and 300
        idx = rng.choice(n, size=m, replace=False)
        # equal weights
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0/m
        # maybe a small uniform background to avoid zeros? Not needed.
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[rand sparse] m={m} new best fast_score={best_score:.9f}")

    # Phase 2: local refinement (hillclimbing) on best_a
    # while time remaining:
    #   propose move: pick i with mass > eps, pick j random (maybe zero)
    #   transfer fraction (e.g., half of a_i) or random fraction.
    #   evaluate new score. accept if better; if not, accept with prob exp(-Δ / T) where T decays.

    def refine(a_start, start_score, deadline):
         a = a_start.copy()
         cur_score = start_score
         best_a_local = a.copy()
         best_score_local = cur_score
         temperature0 = max(best_score_local*0.05, 1e-12)
         while time.time() < deadline:
            elapsed = time.time() - start_time
            frac = (deadline - start_time - elapsed) / (deadline - start_time)  # maybe decreasing T
            T = temperature0 * frac
            # propose move
            # choose source idx with a_i > 1e-12
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src = rng.choice(src_candidates)
            # choose destination index (different)
            dst = rng.integers(0, n)
            if dst == src:
                continue
            # amount to move:
            delta = a[src] * rng.random() * 0.5  # up to half of src
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            # normalize sum? sum unchanged.
            # compute fast_score
            new_score = fast_score(a_new, fft_len)
            dE = new_score - cur_score
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                if new_score < best_score_local:
                    best_score_local = new_score
                    best_a_local = a.copy()
                    print(f"[refine] new best fast_score={best_score_local:.9f}")

         return best_a_local, best_score_local

    # allocate time for refinement and final gradient
    phase2_end = deadline - 0.3*budget_s  # allocate remaining for refinement+final
    # start_time global variable defined earlier

    if time.time() < phase2_end:
        best_a, best_score = refine(best_a, best_score, phase2_end)
        print(f"[refine] best fast_score after refinement={best_score:.9f}")

    # Phase 3: maybe run L-BFGS from previous code
    # but we could just run exp_gradient as before

"""
step‑function optimiser – 1000 s budget (new approach)

Algorithm sketch
----------------
1. **Sparse random warm‑starts** – generate many “spike” vectors with a
   moderate number (150‑250) of non‑zero entries.  We draw the support
   uniformly and assign either equal weights or a Dirichlet‑distributed
   weight vector (sum=1).  These give a very low Ⓛ₂‑norm (hence low
   self‑convolution) while the pair‑wise sums are spread over many bins,
   reducing the maximal convolution entry.

2. **Simple simulated‑annealing refinement** – starting from the best
   sparse candidate we repeatedly move a random amount of mass from a
   heavily‑loaded index to a random destination.  Moves are accepted
   with a Metropolis rule; the temperature is linearly cooled.  This
   quickly drives down the maximum convolution value.

3. **Momentum exponentiated‑gradient descent** – a mirror‑SGD (exponentiated
   gradient) on the exact Ⓛ∞‑objective is run for the remaining time.
   It further smooths the distribution while keeping the sum fixed.

4. **Final polishing** – a short L‑BFGS cascade on a log‑sum‑exp
   surrogate (as in the previous state‑of‑the‑art) and a tiny
   simulated‑annealing run finish the optimisation.

All phases respect a global 1000 s deadline.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed maximum of the auto‑convolution (log‑sum‑exp) and its gradient.
    The underlying probability vector is a = softmax(y).
    """
    # ------- softmax -------
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ------- convolution -------
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_iters: int = 5_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    Uses a momentum‑averaged sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # cheap periodic score check
        if (it & 15) == 0:          # every 16 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (focuses on sparse vectors)
# ----------------------------------------------------------------------


def sparse_annealing_refine(a_start: np.ndarray,
                            start_score: float,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator):
    """
    Moves mass between entries (mostly from heavy to light locations)
    with a linearly cooling temperature.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)

    while time.time() < deadline:
        # cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        # pick source (must have some mass) and destination
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # move a random fraction of the source mass
        delta = a[src] * rng.random() * rng.uniform(0.05, 0.30)
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, fft_len)
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[sparse‑anneal] new best fast_score={best_score:.9f}")

    return best_a, best_score

    if time.time() < deadline:

        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
