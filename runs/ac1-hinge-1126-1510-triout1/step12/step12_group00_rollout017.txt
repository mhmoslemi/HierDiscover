        conv = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), L)[:2*n-1]
        M = conv.max()
        # build constraint matrix
        row_ind = []
        col_ind = []
        data = []
        for k in range(2*n-1):
            # i indices for a: i from max(0, k-(n-1)) to min(k, n-1)
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # For each i, coefficient for g_{k-i} = a_i
            # So j = k-i, loop over i
            # Equivalent j from max(0, k-n+1) to min(k, n-1)
            # We'll loop i
            for i in range(i_start, i_end+1):

                # add entry (k, j) coefficient a[i]
                row_ind.append(k)
                col_ind.append(j)
                data.append(a[i])

    fft_len = next_pow2(2*n-1) # maybe reuse global

        fa = np.fft.rfft(a, fft_len)

        M = conv.max()

"""
Algorithm:
- Generate several diverse initial distributions (deterministic spectral, triangular, Hann, Gaussian, edge‑heavy, random).
- Run a continuation of smoothed L‑BFGS (log‑sum‑exp approximation) to quickly lower a smooth max.
- Refine by Adam on the exact max using a top‑k mask, keeping the best ever seen.
- Apply a linear‑programming “boost” step: for the current distribution f compute its maximal sum‑distribution value M,
  solve a LP maximising the total mass of a new distribution g under the linear constraints f∗g ≤ M, normalise g,
  and blend f and g (trying several blend factors).  Repeat a few times – this frequently reduces the max
  convolution further than pure gradient methods.
- Finish with a short greedy hill‑climb on the simplex.
The whole search respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Fast exact score (used for bookkeeping only)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.02):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    The top‑k fraction linearly decays from start_top to end_top.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    # total window for the phase
    phase_start = time.time()
    total_allowed = deadline - phase_start
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        # mask of largest entries
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked max (linear combination of entries)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # d max/ d a (ignoring the 2·n factor)

        grad_a = 2.0 * n * g_a                         # full objective gradient

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # a slightly larger initial step
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake off plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.05):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(a_start: np.ndarray,
              n: int,
              fft_len: int,
              max_iter: int,
              t_blend: float,
              deadline: float):
    """
    Linear‑programming boost step.
    For the current distribution a, compute M = max(conv(a,a)).
    Solve   maximise Σ g_i   s.t. a * g ≤ M , g ≥ 0,
    scale g to a simplex, blend a←(1‑t)·a + t·g (trying a few t‑values),
    and repeat a few times.
    Returns the best a seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    # ensure all entries are strictly positive – otherwise the LP is unbounded
    eps = 1e-12

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current max convolution value
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # build Toeplitz‑like constraint matrix A (size (2n‑1) × n)
        a_pos = np.maximum(a, eps)

        # each diagonal offset i (0 ≤ i < n) contains the vector a_pos
        offsets = np.arange(n, dtype=int)
        # repeat a_pos for every diagonal
        data = np.tile(a_pos, (n, 1))   # shape (n, n)
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        # linear programme: maximise Σ g_i   ⇔ minimise -Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # fallback: abort the refinement
            print(f"[LP] iteration {it} linprog failed: {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 0:
            break
        g = g_raw / sum_g   # normalise to sum 1

        # try a few blend factors and keep the best
        candidates = [(1 - t_blend) * a + t_blend * g,
                      (1 - 0.5 * t_blend) * a + 0.5 * t_blend * g,
                      (1 - 2 * t_blend) * a + 2 * t_blend * g]  # may overshoot a bit
        for cand in candidates:
            # re‑normalise (tiny numerical drift)
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s == 0:
                continue
            cand = cand / s
            sc = fast_score(cand, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                # optional early exit if improvement is significant
        # if no candidate improved, still move a towards g to allow future progress
        if np.allclose(a, g, atol=1e-12):
            # already aligned – break
            break
        # otherwise continue with the blended a (even if not best) to give the LP a new centre
        # use simple convex blend
        a = (1 - t_blend) * a + t_blend * g
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum == 0:
            break
        a = a / a_sum

    return best_a, best_score

"""
Algorithm for minimising `evaluate_sequence`:

1. **Diverse initial pool** – deterministic (spectral, triangular, Hann,
   Blackman, Gaussian, Kaiser) and random probability vectors are built;
   the best of them (exact fast_score) seeds the optimisation.

2. **Smoothed L‑BFGS** – minimise a log‑sum‑exp (smooth) approximation of
   the max‑convolution.  The smoothing parameter λ is decreased stepwise;
   each λ is given a few hundred L‑BFGS iterations.  The result is a strong
   starting point for the exact optimiser.

3. **Exact Adam with a top‑k mask** – Adam is applied to the true objective
   `2·n·max(conv)`.  The gradient is built from the top‑k largest
   convolution entries (k decays from ~12 % to 2 %).  This flattens the
   whole autocorrelation.

4. **Linear‑programming boost** – with the current distribution `f`,
   compute `M = max(conv(f,f))`.  Solve the LP  
   `max Σ g_i   s.t.   f * g ≤ M ,  g ≥ 0`.  
   The LP solution `g` respects the same maximal convolution bound.
   After normalising `g` to a simplex we blend `f ← (1‑t)·f + t·g`
   (trying a few values of `t`).  Repeating this step a few times often
   reduces the max‑convolution more than pure gradient steps.

5. **Final greedy hill‑climb** – tiny mass‑transfer moves are accepted only
   if they improve the exact score.

All phases respect the 1000 s budget and the best sequence found is
returned.  The code works with any size `n` that is a power of two; `n=4096`
offers a good trade‑off between granularity and speed.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (used for bookkeeping only)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation helpers (deterministic & random families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.12,
                       end_top: float = 0.02):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    The top‑k fraction linearly decays from start_top to end_top.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.05):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine(a_start: np.ndarray,
              n: int,
              fft_len: int,
              max_iter: int,
              t_blend: float,
              deadline: float):
    """
    Linear‑programming boost:
    maximise Σ g_i  s.t. a * g ≤ M , g ≥ 0.
    After solving, g is normalised to a simplex and blended with a.
    The process is repeated a few times.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    eps = 1e-12

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound M
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # build Toeplitz‑like constraint matrix A (size (2n‑1) × n)
        a_pos = np.maximum(a, eps)
        offsets = np.arange(n, dtype=int)
        data = np.tile(a_pos, (n, 1))            # shape (n, n)
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        c = -np.ones(n)                          # minimise -Σg_i  ⇔ maximise Σg_i
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 0:
            break
        g = g_raw / sum_g

        # try a few blend factors
        candidates = [
            (1 - t_blend) * a + t_blend * g,
            (1 - 0.5 * t_blend) * a + 0.5 * t_blend * g,
            (1 - 2 * t_blend) * a + 2 * t_blend * g
        ]
        improved = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s == 0:
                continue
            cand = cand / s
            sc = fast_score(cand, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True

        if not improved:
            # still move a a little toward g so the LP gets a new centre
            a = (1 - t_blend) * a + t_blend * g
            a = np.maximum(a, 0)
            s = a.sum()
            if s == 0:
                break
            a = a / s

        # early termination if already very close to the optimum
        if best_score <= 1.0 + 1e-4:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – power‑of‑two length for fast FFTs
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    # Interpolate previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Pick the best initial candidate (exact fast_score)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Smoothed L‑BFGS (≈30% of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈45% of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start exact optimisation")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.12,
            end_top=0.02
        )
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑programming boost (≈20% of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[LP‑refine] start")
        # reserve a small slice for the final hill‑climb
        max_lp_iter = 6
        blend_t = 0.25
        best_a, best_score = lp_refine(
            best_a,
            n,
            fft_len,
            max_iter=max_lp_iter,
            t_blend=blend_t,
            deadline=deadline
        )
        print(f"[LP‑refine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=40000,
            delta_frac=0.04
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
