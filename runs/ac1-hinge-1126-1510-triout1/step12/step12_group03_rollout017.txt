"""
Propose a step sequence with a low evaluation score.

The method combines three ideas that differ from the reference implementation:

1.  A larger problem size (n≈10000) – the theoretical lower bound
    2 n / (2 n−1) → 1 is much smaller than 2, giving more “room” for improvement.

2.  A custom Adam optimiser that minimises a *combined* objective:
        • a smoothed max‑convolution (log‑sum‑exp, as before);
        • the variance of the convolution (∑(c−mean)²), which pushes the
          self‑convolution towards a flat profile.
    The gradient of the variance term is obtained analytically via FFTs,
    and both gradients are merged on the log‑softmax (logits) variables
    that guarantee a probability simplex (sum = 1) constraint.

3.  An intensified L‑BFGS polishing phase (more iterations / restarts)
    followed by a short greedy hill‑climb for final polishing.

All steps respect the 1000 s budget and keep the sequence non‑negative.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial families (scaled to sum = 1)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based guess – a smooth spectrum projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    # target: flat magnitude -> take sqrt of flat spectrum
    # uniform spectrum:
    spectrum = np.ones(L, dtype=np.complex128)
    # random phases improve diversity
    rng = np.random.default_rng()
    phases = np.exp(1j * 2 * np.pi * rng.random(L // 2 - 1))
    spectrum[1:L // 2] = phases
    spectrum[-(L // 2 - 1):] = np.conj(phases)
    a_complex = np.fft.ifft(spectrum)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m = max(1, int(edge_frac * n))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        l = max(0, centre - width)
        r = min(n, centre + width)
        a[l:r] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Custom Adam optimiser (smooth‑max + variance)
# ----------------------------------------------------------------------


def combined_adam_optimize(a_start: np.ndarray,
                           deadline: float,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator,
                           lam_factor: float = 1e-4,
                           lr: float = 0.06,
                           max_iter: int = 300000,
                           alpha_start: float = 0.8,
                           alpha_end: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a *combined* objective:
        - smoothed max‑convolution (log‑sum‑exp);
        - variance of the convolution (∑(c−mean)²).

    The mixture weight α goes linearly from `alpha_start` to `alpha_end`
    during the run, moving the focus from the max‑term to the variance term.
    """
    y = np.log(a_start + 1e-18)                     # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # ----- forward pass -----
        a = softmax(y)

        # ----- FFT of a (used many times) -----
        fa = np.fft.rfft(a, fft_len)

        # full self‑convolution (length 2·n‑1)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # ---------- smoothed max ----------
        max_c = conv.max()
        lam = max_c * lam_factor + 1e-12
        shifted = conv - max_c
        # prevent overflow / underflow
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        w = exp_shift / Z                     # ∂ smooth_max / ∂ conv

        # gradient of smooth max w.r.t. a
        a_rev = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
        g_vec = conv_rev[(n - 1):(n - 1 + n)]

        grad_y_max = a * (g_vec - np.dot(a, g_vec))

        # ---------- variance term ----------
        mean_conv = conv.mean()
        diff = conv - mean_conv
        fdiff = np.fft.rfft(diff, fft_len)
        # reuse a_rev_fft (already computed as fa_rev)
        conv_rev_var = np.fft.irfft(fdiff * fa_rev, fft_len)
        g_var = conv_rev_var[(n - 1):(n - 1 + n)]
        grad_a_var = 4.0 * g_var               # ∂∑(c−mean)² / ∂ a
        grad_y_var = a * (grad_a_var - np.dot(a, grad_a_var))

        # ---------- blend the two gradients ----------
        # linear schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        grad_y = alpha * grad_y_max + (1.0 - alpha) * grad_y_var

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        # ----- monitor progress -----
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # occasional logging
        if it % 20000 == 0:
            print(f"[Adam‑combined] it={it}, best_score={best_score:.8f}, α={alpha:.3f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS polishing (exactly as before – only the schedule is longer)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    f = 2.0 * n * (max_c + lam * np.log(Z))

    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run a cascade of L‑BFGS solves over a decreasing λ schedule."""
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (strictly improving mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.09):
    """Greedy hill‑climb that only accepts improvements."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine runs three phases:
        1. L‑BFGS polishing on the smoothed max‑convolution.
        2. Adam optimisation on a combined (max + variance) objective.
        3. Final greedy hill‑climb.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------
    n = 10000                     # 10 k entries – still cheap FFT‑wise
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the new size.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(inv_sqrt_init(n))
    for _ in range(3):
        # random β for Kaiser window
        beta = rng.uniform(4.0, 12.0)
        a = np.kaiser(n, beta)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        candidates.append(a)

    # --------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # a fresh start: blend best with a random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on combined objective (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = combined_adam_optimize(best_a,
                                                    phase2_end,
                                                    n,
                                                    fft_len,
                                                    rng,
                                                    lam_factor=1e-4,
                                                    lr=0.06,
                                                    max_iter=500000,
                                                    alpha_start=0.8,
                                                    alpha_end=0.2)
        print(f"[Adam‑combined] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=250000,
                                        delta_frac=0.09)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # the provided routine
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
