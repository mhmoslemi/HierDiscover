"""
step_function_optimiser_v2 – 1000 s budget

Strategy
--------
* **Diverse warm‑starts** – generate many shapes (uniform,
  edge‑exponential, power‑edge, bimodal, Dirichlet‑sparse,
  mixed) and keep a small pool of the best K.
* **Multi‑restart L‑BFGS** – each pool entry receives a short
  L‑BFGS cascade (few λ‑values) to escape poor basins; the best
  result seeds a long cascade with a fine λ‑schedule.
* **Adaptive exponentiated‑gradient descent** – mirror‑SGD on the
  exact L∞‑objective, with momentum, step‑size decay and occasional
  “jumps’’ (Gaussian re‑normalisation) when progress stalls.
* **Aggressive simulated‑annealing** – large mass‑transfer moves
  with a linear‑cooling schedule, run after the gradient phase.
* **Final polishing** – extra very‑low‑λ L‑BFGS + a short anneal.

All phases respect the global 1000 s deadline; the best vector found
is stored in the global variable ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast surrogate for the official score when ∑a = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive: return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector (sum = 1) created from a random family."""
    mode = rng.choice(
        ['uniform', 'exp_edge', 'power_edge', 'bimodal',
         'sparse', 'mixed', 'dirichlet']
    )
    x = np.arange(n, dtype=float)

    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_edge':
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == 'power_edge':
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == 'bimodal':
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == 'sparse':
        # a handful of spikes – Dirichlet with low concentration
        m = rng.integers(int(0.01 * n), int(0.05 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == 'mixed':
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    else:   # dirichlet – very sparse, controllable sparsity
        conc = rng.uniform(0.02, 0.2)            # low concentration → sparse
        a = rng.gamma(conc, 1.0, size=n)

    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    a = softmax(y) – a probability vector of length n.
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # effective smoothing proportional to the current max value
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient wrt convolution entries ---
    w = exp_shift / Z                       # d smooth_max / d conv_i
    # the convolution of w with reversed a gives the gradient w.r.t. a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
    verbose: bool = False,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score

# Exponentiated‑gradient descent with adaptive restarts
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
    max_iters: int = 8_000_000,
    verbose: bool = False,
):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution.
    Includes momentum, adaptive step‑size and sporadic Gaussian jumps.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.9            # momentum decay
    no_impr = 0           # counter for stagnation

    it = 0
    while time.time() < deadline and it < max_iters:
        # --- convolution and mask for the (almost) active set ---
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # --- sub‑gradient of the hard max ---
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size: slowly decays, but we boost after long stagnation
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrored on the simplex)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        a /= a.sum()

        # --- periodic progress check ---
        if (it & 31) == 0:               # every 32 iterations
            cur_score = fast_score(a, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_impr = 0
                if verbose:
                    print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")
            else:
                no_impr += 1

        # --- occasional random jump to escape a flat region ---
        if no_impr >= 100_000 and rng.random() < 0.6:
            sigma = rng.uniform(0.003, 0.010)
            noisy = a + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            a = noisy
            no_impr = 0
            if verbose:
                print(f"[exp‑grad] random restart at iter={it}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing with larger mass moves
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
    verbose: bool = False,
):
    """
    Random mass‑transfer moves (1‑% to 20‑% of source) accepted by
    a Metropolis rule with a linear cooling schedule.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.08, 1e-12)   # initial temperature
    max_frac = 0.20                         # up‑to 20 % of src mass can be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        # pick a source index with non‑zero mass and a destination
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(0, n)
        if src == dst:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose:
                    print(f"[anneal] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that
    minimises ``evaluate_sequence``.  Returns the best list of floats
    discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – moderate dimension works well with FFT
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Timing allocation (percent of the total budget)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.15 * budget_s  # warm‑starts
    phase2_end = start_time + 0.45 * budget_s  # multi‑restart L‑BFGS
    phase3_end = start_time + 0.80 * budget_s  # long exp‑grad
    phase4_end = start_time + 0.95 * budget_s  # final low‑λ LBFGS
    # phase5 runs until deadline (final anneal)

    # --------------------------------------------------------------
    # 1) Warm‑starts – keep a small pool of the top‑K candidates
    # --------------------------------------------------------------
    K = 5                       # pool size
    pool = []                   # list of (score, vector)
    best_a = None
    best_score = float('inf')

    # use previous result if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            sc = fast_score(a0, fft_len)
            pool.append((sc, a0.copy()))
            best_a, best_score = a0, sc
            print(f"[init] warm‑start from previous best fast_score={sc:.9f}")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)
        sc = fast_score(a, fft_len)

        # keep in a sorted pool of size ≤ K
        if len(pool) < K:
            pool.append((sc, a.copy()))
        else:
            # replace worst if this one is better
            worst_idx = np.argmax([p[0] for p in pool])
            if sc < pool[worst_idx][0]:
                pool[worst_idx] = (sc, a.copy())

        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[warm‑start] new global best fast_score={best_score:.9f}")

    # sort pool for convenience
    pool.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2) Multi‑restart L‑BFGS – short cascades on the pool entries
    # --------------------------------------------------------------
    lam_schedule_short = [0.7, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01]
    for idx, (_, a0) in enumerate(pool):
        if time.time() >= phase2_end:
            break
        # short cascade: few iterations per λ
        a_ref, sc_ref = lbfgs_cascade(
            a0,
            lam_schedule_short,
            maxiter_per_lam=800,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
            verbose=False,
        )
        if sc_ref < best_score:
            best_score, best_a = sc_ref, a_ref.copy()
            print(f"[restart‑L‑BFGS #{idx}] improved fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # 3) Long L‑BFGS cascade on the current best (fine λ‑schedule)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        lam_schedule_fine = [
            0.7, 0.5, 0.35, 0.25, 0.15, 0.10,
            0.07, 0.05, 0.03, 0.02, 0.01,
            0.005, 0.003, 0.0015, 0.0008,
            0.0004, 0.0002, 0.0001, 5e-05, 1e-05,
            5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
            5e-08, 2e-08, 1e-08
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule_fine,
            maxiter_per_lam=2500,
            deadline=phase3_end,
            fft_len=fft_len,
            n=n,
            verbose=True,
        )
        print(f"[fine L‑BFGS] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # 4) Adaptive exponentiated‑gradient descent (long run)
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase4_end,
            n,
            fft_len,
            rng,
            eta0=30.0,
            max_iters=10_000_000,
            verbose=True,
        )
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # 5) Final polishing – ultra‑low‑λ L‑BFGS + short anneal
    # --------------------------------------------------------------
    if time.time() < deadline:
        fine_lam = [5e-09, 2e-09, 1e-09, 5e-10, 2e-10]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam,
            maxiter_per_lam=3000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
            verbose=False,
        )
        if sc_refined < best_score:
            best_score, best_a = sc_refined, a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
            verbose=True,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # 6) Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
