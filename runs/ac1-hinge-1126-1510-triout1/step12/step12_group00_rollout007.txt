def solve_lp(f, M, S):
    n = len(f)
    K = len(S)
    A = np.zeros((K, n))
    for row_idx, k in enumerate(S):
        # for i from max(0, k - (n-1)) to min(n-1, k)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # Then for i in [i_min, i_max], g_i coefficient is f_{k-i}
        # So fill: A[row_idx, i] = f[k-i]
        idx = np.arange(i_min, i_max+1)
        A[row_idx, idx] = f[k - idx]
    # Solve LP: max sum(g) s.t. A g <= M, g >=0
    c = -np.ones(n)  # minimize -sum(g) -> maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: use uniform g
        g = np.full(n, 1.0 / n)
    else:
        g = res.x
    # Normalize
    if g.sum() <= 0:
        g = np.full(n, 1.0 / n)
    else:
        g = g / g.sum()
    return g, res

"""
LP‑augmented descent for the step‑function problem.

The algorithm works with a probability vector a (∑a=1) because the
objective `evaluate_sequence` is homogeneous:
    score = 2·n·max(conv(a,a)).
Starting from a diverse pool of deterministic and random initial
vectors we iteratively improve the best candidate:

1. Compute M = max(conv(a,a)).
2. Build a (small) set of the K largest convolution entries.
   For each selected shift k the inequality
        Σ_i a_i·g_{k‑i} ≤ M
   is linear in the unknown vector g.
3. Solve a linear program that maximises Σ g_i subject to those
   inequalities and g_i ≥ 0 (high‑sparsity “highs” solver).
   The optimal g tends to spread its mass so that the overlap with a
   is uniformly small.
4. Mix the current a and the LP‑solution g with a few trial
   mixing rates t∈{0.1,0.2,0.3,0.5}.  Keep the best among the mixes,
   the raw g, and the previous a.
5. Occasionally perturb the current best vector with a tiny
   mass‑transfer move to escape plateaus.
6. After the time budget a short greedy hill‑climb cleans up any
   remaining slack.

All steps respect the 1000 s budget; the function returns the best
list of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified draws on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return np.maximum(a, 0.0) / np.maximum(a, 0.0).sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimisation helpers
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (∑a=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def build_constraint_matrix(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Build a dense matrix A where A[k,i] = a_{shift - i}
    for each shift ∈ shifts (only entries inside bounds are non‑zero).
    """
    n = a.shape[0]
    K = shifts.size
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(shifts):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A[row, idx] = a[k - idx]
    return A


def lp_step(a: np.ndarray, M: float, top_k: int, fft_len: int, n: int):
    """
    Solve the LP
        maximise Σ g_i
        s.t.   (a * g)[k] ≤ M   for the `top_k` largest shifts k
               g_i ≥ 0
    Returns a feasible g (normalised to sum 1).
    """
    # choose the most restrictive shifts
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    if top_k >= conv.size:
        idx_set = np.arange(conv.size)
    else:
        idx_set = np.argpartition(conv, -top_k)[-top_k:]
    idx_set.sort()  # small to large – not required but tidy

    A = build_constraint_matrix(a, idx_set)
    bounds = [(0.0, None)] * n
    c = -np.ones(n)                     # maximise Σ g_i  → minimise -Σ g_i
    res = linprog(c, A_ub=A, b_ub=np.full(len(idx_set), M),
                  bounds=bounds, method='highs', options={'presolve': True})

    if not res.success:
        # fallback to a tiny random perturbation of `a`
        g = a * (1.0 + 0.01 * np.random.randn(n))
        g = np.clip(g, 0, None)
        if g.sum() == 0:
            g = np.full(n, 1.0 / n)
        else:
            g /= g.sum()
    else:
        g = res.x
        s = g.sum()
        if s == 0:
            g = np.full(n, 1.0 / n)
        else:
            g /= s
    return g

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    rng = np.random.default_rng(seed)

    n = 2048

    def score_vec(a):

    best_a = np.full(n, 1.0 / n)
