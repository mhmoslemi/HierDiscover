"""
Propose a low‑evaluation step‑function.

The algorithm consists of three stages:

1. **Quick L‑BFGS refinement** – start from a diverse pool of
   probability vectors (including the previous best), run a cascade of
   L‑BFGS optimisations on a smoothed log‑sum‑exp surrogate of the
   max‑convolution.  This gives a reasonably good configuration quickly.

2. **Shift‑mix hill‑climbing** – repeatedly pick a random offset, shift the
   current vector by that offset (zero‑padding the empty part) and mix it
   back with a decaying step size.  The mixture reduces the overlap of
   large coefficients and therefore lowers the peak of the autocorrelation.
   A simple simulated‑annealing acceptance rule is used to escape plateaus.
   Occasional symmetrisation and moving‑average smoothing help the search.

3. **Final low‑λ L‑BFGS polish** – a short cascade of L‑BFGS with very
   small smoothing parameters finishes the optimisation.

All stages respect the global 1000 s deadline.  The best sequence found
(its entries sum to 1) is stored in the global ``height_sequence_1``
variable and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns the smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # ----- softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution (FFT) -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build the pool of starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Use previous best if it exists (interpolated to current n)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.15 * budget_s   # quick L‑BFGS warm‑up
    phase2_end = start_time + 0.80 * budget_s   # long shift‑mix hill‑climb
    # remaining time → final L‑BFGS polish

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.022,
        0.012,
        0.006,
        0.003,
        0.0015,
        7e-4,
        3e-4,
        1.5e-4,
        8e-5,
    ]
    maxiter_per_lam = 800

    best_a = None
    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter before L‑BFGS
        if rng.random() < 0.35:
            a_start = add_noise(
                a_start, sigma=rng.uniform(0.001, 0.015), rng=rng
            )
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – shift‑mix hill‑climbing (simulated annealing style)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        cur_a = best_a.copy()
        cur_score = best_score
        # temperature parameters
        init_temp = max(cur_score * 0.05, 1e-8)
        while time.time() < phase2_end:
            # progress fraction for temperature decay and mixing weight
            elapsed = time.time() - phase1_end
            total = phase2_end - phase1_end
            frac = min(1.0, elapsed / total)

            # pick a random offset (shift direction randomised)
            offset = rng.integers(1, n)               # 1 … n‑1
            if rng.random() < 0.5:
                # shift right
                shifted = np.zeros_like(cur_a)
                shifted[offset:] = cur_a[:-offset]
            else:
                # shift left
                shifted = np.zeros_like(cur_a)
                shifted[:-offset] = cur_a[offset:]

            # mixing coefficient – larger at the start, smaller later
            tau = 0.30 * (1.0 - frac) + 0.05

            cand = (1.0 - tau) * cur_a + tau * shifted
            s = cand.sum()
            if s <= 0.0:
                continue
            cand /= s

            cand_score = fast_score(cand, fft_len)

            # simulated‑annealing acceptance
            temperature = init_temp * (1.0 - frac)
            if cand_score < cur_score or rng.random() < np.exp(
                -(cand_score - cur_score) / max(temperature, 1e-12)
            ):
                cur_a, cur_score = cand, cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = cur_a.copy()
                    print(f"[ShiftMix] new best fast_score={best_score:.9f}")

            # occasional extra tricks
            if (rng.random() < 0.01) and (time.time() < phase2_end):
                # try symmetrisation
                sym = symmetrize(cur_a)
                sym_score = fast_score(sym, fft_len)
                if sym_score < cur_score:
                    cur_a, cur_score = sym, sym_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = cur_a.copy()
                        print(f"[ShiftMix] symmetrised best fast_score={best_score:.9f}")

                # small Gaussian jitter to avoid stagnation
                cur_a = add_noise(cur_a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)

    # ------------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional tiny smoothing before final evaluation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        smoothed_score = fast_score(smoothed, fft_len)
        if smoothed_score < best_score:
            best_a = smoothed
            best_score = smoothed_score
            print(f"[Smooth] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
