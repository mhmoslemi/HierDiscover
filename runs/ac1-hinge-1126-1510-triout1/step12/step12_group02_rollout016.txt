    positions = []

    for i in rng.permutation(n):
        # test collisions

        # check self sum

                    possible = False

            # add new sums

    used = np.zeros(2*n-1, dtype=bool)

                break

        # add i
        positions.append(i)
        used[i + i] = True
        for j in positions[:-1]:
            used[i + j] = True
    return np.array(positions, dtype=int)

"""
search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The code builds a very large pool of diverse initial vectors (classical windows,
random Dirichlet draws, an extensive random search over the Beta family and
other analytic shapes).  The best of those candidates is then refined by many
restarts of a smoothed‑max L‑BFGS optimisation (the smoothing parameter is
decreased down to 1e‑9).  After the L‑BFGS phase we run a few steps of Adam,
exponentiated‑gradient descent on the exact max‑convolution and finish with a
light hill‑climb.  All operations respect the non‑negativity and sum‑to‑one
constraints required by ``evaluate_sequence``.  The routine works within the
allocated budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact for probability vectors)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(conv) for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient for L‑BFGS/Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution using log‑sum‑exp."""
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """A spectral guess (inverse‑fft of a flat spectrum)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(amp)
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified Dirichlet‑type draws on the simplex."""
    mode = rng.choice(['uniform', 'dirichlet', 'exp', 'power', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        a = rng.gamma(0.5, 1.0, size=n)          # small concentration → spiky
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:  # edge
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Beta‑density discretisation."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta_ - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous result."""
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 4000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.07):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-10)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.01):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """Simple greedy hill‑climb that only accepts improving moves."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a low‑score step sequence and return it as a list of floats."""
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ------------------------------------------------------------------
    # problem size – keep the historic size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # build a rich pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # random families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # extensive random Beta search – the most promising family
    # --------------------------------------------------------------
    for _ in range(1200):          # a few thousand draws fit easily into the budget
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_ = rng.exponential(scale=0.8) + 0.1
        cand = beta_init(n, alpha, beta_)
        candidates.append(cand)

    # --------------------------------------------------------------
    # pick best candidate by exact fast_score
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] considered {len(candidates)} candidates – best fast_score={best_score:.8f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS restarts (≈55 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # blend current best with a fresh random vector for diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.45          # up to 45 % random mass
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of total time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=5000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient on the exact max (≈15 % of total time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.01)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_score = fast_score(best_a, fft_len, n)
        best_a, best_score = hill_climb(best_a,
                                        cur_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=120000,
                                        delta_frac=0.05)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # return the sequence (evaluate_sequence is called once at the end)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
