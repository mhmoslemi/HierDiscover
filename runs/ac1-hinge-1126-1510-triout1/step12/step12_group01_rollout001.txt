"""
Propose a low‑score step function.

Algorithm
---------
* Use a richer set of initialisations (parameter sweeps for Kaiser,
  Gaussian and power‑law windows) and interpolate the previous best
  sequence if available.
* Run a more aggressive L‑BFGS continuation on a log‑sum‑exp smoothed
  max‑convolution.  The λ‑schedule goes down to 1e‑5 and each stage is
  allowed more iterations.
* Refine with exponentiated‑gradient descent that focuses on an ever
  shrinking top‑fraction of the convolution entries.
* Optimise the exact max‑convolution with Adam on the soft‑max logits.
* Perform a longer simulated‑annealing phase with pairwise mass
  transfers.
* Finish with a high‑precision greedy hill‑climb (tiny transfers).
* The whole pipeline fits into the 1000 s budget and works for a larger
  problem size (n=8192) which gives the optimiser more freedom to
  approach the theoretical limit (~1.5).

The function returns a plain list of non‑negative floats; the
`evaluate_sequence` function (provided by the runner) is called once at the
end to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact objective (fast_score) for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(conv) for a on the simplex (sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth max of the convolution.
    Returns (f, grad) where f = 2·n·smooth_max and grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution values
    w = exp_shift / Z    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def powerlaw_init(n: int, exponent: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+1)^(-exponent) + (n-i)^(-exponent)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (-exponent)
    right = (n - i + 1) ** (-exponent)
    a = left + right
    a[a < 0] = 0.0
    a /= a.sum()
    return a


def sweep_kaiser(n: int, betas: np.ndarray) -> tuple[np.ndarray, float]:
    """Return the best Kaiser window among the supplied betas."""
    best_a = None
    best_sc = np.inf
    for b in betas:
        a = kaiser_init(n, beta=b)
        sc = fast_score(a, next_pow2(2 * n - 1), n)
        if sc < best_sc:
            best_sc = sc
            best_a = a
    return best_a, best_sc


def sweep_gaussian(n: int, sigmas: np.ndarray) -> tuple[np.ndarray, float]:
    """Return the best Gaussian window among the supplied sigma factors."""
    best_a = None
    best_sc = np.inf
    for s in sigmas:
        a = gaussian_init(n, sigma_factor=s)
        sc = fast_score(a, next_pow2(2 * n - 1), n)
        if sc < best_sc:
            best_sc = sc
            best_a = a
    return best_a, best_sc


def sweep_powerlaw(n: int, exps: np.ndarray) -> tuple[np.ndarray, float]:
    """Return the best power‑law window among the supplied exponents."""
    best_a = None
    best_sc = np.inf
    for e in exps:
        a = powerlaw_init(n, exponent=e)
        sc = fast_score(a, next_pow2(2 * n - 1), n)
        if sc < best_sc:
            best_sc = sc
            best_a = a
    return best_a, best_sc

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, each stage starting from the
    previous result (logits).  Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction_start: float = 0.02,
                          top_fraction_end: float = 0.001):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The top_fraction linearly decays from start to end.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # linearly decay top_fraction
        frac = max(top_fraction_end,
                   top_fraction_start * (deadline - time.time()) /
                   (deadline - (deadline - (deadline - time.time()))))
        # Actually we want a simple linear decay based on iteration count
        # For simplicity we recompute a decreasing fraction every 5000 steps
        if it % 5000 == 0 and it > 0:
            # exponential decay of the fraction
            top_fraction_start *= 0.7

        kth = max(1, int(np.ceil(frac * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient wrt a via reversed a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)   # diminishing step size

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        a = a / s if s > 0 else best_a.copy()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[EG] it={it}, fast_score={best_sc:.9f}")
        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y (a = softmax(y)).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and mask of the (almost) maximal entries
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # sub‑gradient w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a  # derivative of 2·n·max(conv)

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)      # slowly decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional Gaussian jitter to escape plateaux
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.9f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Pairwise mass‑transfer simulated annealing.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.12   # smaller transfers for fine tuning
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.03):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives the optimiser more freedom.
    # --------------------------------------------------------------
    n = 8192                         # try a larger sequence
    fft_len = next_pow2(2 * n - 1)   # FFT size for zero‑padded transforms

    # --------------------------------------------------------------
    # Build an initial pool of candidates (including parameter sweeps)
    # --------------------------------------------------------------
    candidates = []

    # Interpolate previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))

    # Parameter sweeps for Kaiser, Gaussian, power‑law
    betas = np.linspace(4.0, 14.0, 21)       # 4.0 .. 14.0 step 0.5
    best_kaiser, _ = sweep_kaiser(n, betas)
    candidates.append(best_kaiser)

    sigma_factors = np.linspace(0.18, 0.35, 18)   # 0.18 .. 0.35
    best_gauss, _ = sweep_gaussian(n, sigma_factors)
    candidates.append(best_gauss)

    exponents = np.linspace(0.3, 3.0, 28)        # 0.3 .. 3.0
    best_pow, _ = sweep_powerlaw(n, exponents)
    candidates.append(best_pow)

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # Keep the best candidate so far (fast_score exact for sum=1)
    best_a = None
    best_sc = np.inf
    for cand in candidates:
        if cand is None:
            continue
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Aggressive L‑BFGS (smoothed max)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # 30 % of total time
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.06, 0.04, 0.03, 0.02, 0.015,
                    0.01, 0.006, 0.004, 0.002, 0.001,
                    5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.33
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len, fft_len,
            n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.18 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(
            best_a,
            eg_deadline,
            n,
            fft_len,
            top_fraction_start=0.02,
            top_fraction_end=0.001
        )
        print(f"[EG] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng
        )
        print(f"[Adam] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-7,
            max_moves=150000
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final greedy hill‑climb (high‑precision polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=120000,
            delta_frac=0.015
        )
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Final evaluation (official scoring)
    # --------------------------------------------------------------
    # Clip tiny negatives that may appear due to numerical noise
    final_seq = np.clip(best_a, 0.0, None).tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
