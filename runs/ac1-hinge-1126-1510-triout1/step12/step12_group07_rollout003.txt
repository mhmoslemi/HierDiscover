"""
Propose a step‑function using the discrete arcsine law.

The arcsine distribution f(i) ∝ 1/√(i·(n‑i)) (with i shifted by 0.5) has the
property that the sum of two independent draws has (asymptotically) a
uniform distribution.  Consequently the self‑convolution of the discrete
approximation is almost flat, giving a very small value of
2·n·max(conv(a,a)).  The construction is followed by a short
exponentiated‑gradient descent that directly reduces the exact L∞‑norm
of the convolution while staying on the probability simplex.

The routine works for any time budget ≤1000 s, returns the best sequence
found and stores it in the global ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a discrete arcsine distribution, then run a short
    exponentiated‑gradient descent on the true L∞‑objective.
    The best sequence found (as a list of floats) is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality & speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Initialise with the discrete arcsine law.
    #    a_i ∝ 1/√((i+½)·(n‑i‑½)), i=0,…,n‑1.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    a /= a.sum()                      # normalise to sum‑one

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # 2) Exponentiated‑gradient descent on the exact max‑convolution.
    #    This is a simple sub‑gradient method that stays on the simplex.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin
    it = 0

    # parameters for EG
    eta0 = 35.0                # initial step size (absolute scale)
    max_iter = 10_000_000      # just a huge upper bound; loop stops by deadline

    while time.time() < deadline and it < max_iter:
        # convolution a*a
        fa = np.fft.fft(best_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max (sub‑gradient)
        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]       # size n

        # step size decays like 1/√t
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        cand = best_a * exp_term
        s = cand.sum()
        if s <= 0.0:                 # numerical safeguard
            cand = best_a.copy()
        else:
            cand /= s

        cur_score = fast_score(cand, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # occasional progress report
            if it % 500 == 0:
                print(f"[EG {it:06d}] new best fast_score = {best_score:.9f}")

        it += 1

    # --------------------------------------------------------------
    # 3) Final official evaluation and expose the result.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
