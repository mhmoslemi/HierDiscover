def variance_objective_and_grad(y, n, fft_len):
    # y unrestricted; a = softmax(y)
    a = np.exp(y - np.max(y))
    a /= a.sum()
    # convolution
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    sum_a = a.sum()   # =1
    avg = sum_a * sum_a / (2*n-1)   # =1/(2*n-1)
    diff = conv - avg
    loss = np.dot(diff, diff)  # L2 loss
    # gradient w.r.t conv entries = 2*diff
    w = 2.0 * diff
    # backprop
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_vec = np.real(np.fft.ifft(fa_rev * fw))
    g = grad_vec[n-1:n-1+n]       # size n
    # chain rule softmax similar to before
    mean_g = np.dot(a, g)
    grad_y = a * (g - mean_g)
    return loss, grad_y
```

"""
Step‑function optimiser – 1000 s budget (enhanced)

The optimiser works in four phases, each respecting the global deadline.

1. **Variance pre‑optimisation** – minimise the L2‑variance of the
   auto‑convolution (which encourages a flat convolution).  This uses
   L‑BFGS on a smooth quadratic loss and a simple simplex projection.
2. **Smoothed‑max L‑BFGS cascade** – minimise a log‑sum‑exp surrogate of the
   true ∞‑objective with a decreasing λ‑schedule that goes down to 5e‑8.
3. **Mirrored exponentiated‑gradient descent** – a long sub‑gradient run on
   the exact max‑convolution, using a 1/√t step‑size.
4. **Greedy local refinement** – random mass‑transfer moves that strictly
   improve the exact score.

The search is performed on a *large* problem size (`n = 8000`), which
allows a finer approximation of a flat auto‑convolution and thus a lower
final bound.  An existing best sequence (`height_sequence_1`) is
interpolated to the new size and used as a warm start.  The code prints
progress reports and finally stores the best sequence in the global
variable `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (a is assumed non‑negative, sum>0)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def upsample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length, then simplex‑project."""
    old_n = seq.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum == 0.0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up /= up_sum
    return up

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a

def smooth_max_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and gradient.
    y – unrestricted variables, a = softmax(y) lives on the simplex.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                # ≤0
    exp_shift = np.exp(shifted / lam_eff)      # safe: exponent ≤0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def variance_and_grad(y: np.ndarray, n: int, fft_len: int):
    """
    L2 variance of the auto‑convolution (target: flat convolution) and gradient.
    y – unrestricted variables, a = softmax(y) lives on the simplex.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # mean of convolution (for sum_a == 1)
    avg = 1.0 / (2 * n - 1)

    diff = conv_full - avg
    loss = np.dot(diff, diff)  # L2 loss

    # gradient wrt conv entries = 2*diff
    w = 2.0 * diff
    # back‑propagation to a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return loss, grad_y

def lbfgs_minimize(a_start: np.ndarray,
                   objective_fun,
                   maxiter: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a single L‑BFGS call on a given objective (expects (f,grad) return).
    Returns the (possibly improved) a and its fast_score.
    """
    if time.time() >= deadline:
        return a_start.copy(), fast_score(a_start, fft_len)

    y0 = np.log(a_start + 1e-12)  # work in log‑space
    try:
        res = minimize(fun=objective_fun,
                       x0=y0,
                       args=(n, fft_len),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter,
                                'gtol': 1e-12,
                                'disp': False})
    except Exception:
        return a_start.copy(), fast_score(a_start, fft_len)

    y_opt = res.x
    a_opt = np.exp(y_opt - np.max(y_opt))
    a_opt /= a_opt.sum()
    score = fast_score(a_opt, fft_len)
    return a_opt, score


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam,
                  deadline,
                  fft_len,
                  n):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break

        # wrap the smooth‑max objective with the current λ
        def obj(y, n=n, fft_len=fft_len, lam=lam):
            return smooth_max_and_grad(y, n, fft_len, lam)

        # run a short L‑BFGS from current y
        try:
            res = minimize(fun=obj,
                           x0=y,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score = {best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 18.0):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: repeatedly transfer a small random fraction of mass
    from one position to another.  Only moves that strictly improve the
    `2·n·max(conv)` score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20  # up to 20 % of source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    pool = []

    # Warm‑start from previous best (interpolated)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            interp = upsample_seq(old, n)
            pool.append(interp)

    # Classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # Parametric families
    for _ in range(5):
        pool.append(random_initial(n, rng))
    for _ in range(3):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.22 * budget_s  # variance pre‑optimisation
    phase2_end = start_time + 0.70 * budget_s  # smoothed‑max L‑BFGS
    phase3_end = start_time + 0.95 * budget_s  # EG descent
    # remaining time → local refinement

    best_a = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # Phase 1 – variance minimisation (L‑BFGS on quadratic loss)
    # --------------------------------------------------------------
    print("[phase 1] variance minimisation start")
    while time.time() < phase1_end:
        a0 = rng.choice(pool)
        if rng.random() < 0.30:
            a0 = add_noise(a0,
                           sigma=rng.uniform(0.001, 0.006),
                           rng=rng)

        a_opt, sc_opt = lbfgs_minimize(a0,
                                       variance_and_grad,
                                       maxiter=600,
                                       deadline=phase1_end,
                                       fft_len=fft_len,
                                       n=n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[phase 1] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – aggressive smooth‑max L‑BFGS cascade
    # --------------------------------------------------------------
    print("[phase 2] smoothed‑max L‑BFGS cascade start")
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7,
        2e-7, 1e-7, 5e-8
    ]
    maxiter_per_lam = 800

    if time.time() < phase2_end:
        best_a, best_score = lbfgs_cascade(best_a,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase2_end,
                                           fft_len,
                                           n)
        print(f"[phase 2] best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent on exact max
    # --------------------------------------------------------------
    print("[phase 3] exponentiated‑gradient descent start")
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_descent(best_a,
                                                  phase3_end,
                                                  n,
                                                  fft_len,
                                                  rng,
                                                  eta0=20.0)
        print(f"[phase 3] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy local refinement (mass‑transfer hill‑climb)
    # --------------------------------------------------------------
    print("[phase 4] local refinement start")
    if time.time() < deadline:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[phase 4] final fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
