The goal is to minimise  

\[
\text{Eval}(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of a non‑negative vector `a` with itself.
Because the objective is homogeneous we may restrict ourselves to probability
vectors `∑a = 1`.  The problem then reduces to minimising the *peak* of the
autocorrelation `c_k = Σ_i a_i a_{i+k}`.

The theoretical lower bound is  

\[
\displaystyle\frac{2n}{2n-1}\approx 1,
\]

so a value around 1.5 (the current best) leaves a large margin for improvement.

The optimiser works in four phases:

1. **Diverse random warm‑starts** – many parametrised families (exponential
   edges, double‑exponential, power‑law, sinusoid, Gaussian, Dirichlet
   random) are sampled, smoothed and projected onto the probability simplex.
   The best candidate is kept.

2. **Smoothed L∞ optimisation** – a cascade of L‑BFGS runs on a
   log‑sum‑exp surrogate of the `max`‐operator.  The smoothing parameter
   λ is decreased from 0.7 to 1e‑7, allowing a fine local search.

3. **Mirror‑SGD (exponentiated gradient)** – a momentum‑averaged sub‑gradient
   descent on the exact `max`‑objective, with a slowly decaying learning‑rate.
   Occasional Gaussian jitter helps escape shallow local minima.

4. **Final polishing** – a short low‑λ L‑BFGS pass followed by a simulated‑annealing
   hill‑climb.

All phases respect the global 1000 s deadline.  The best sequence found is
stored in the global variable `height_sequence_1` and returned as a plain

```python
"""Step‑function optimiser – 1000 s budget (enhanced)

The algorithm proceeds in four stages:
1. Generate many diverse random shapes (edge‑exponential, double‑exponential,
   power‑law, sinusoidal, Gaussian, Dirichlet, sparse, mixed) and keep the best.
2. Refine the best shape with a cascade of L‑BFGS optimisations on a
   log‑sum‑exp smooth approximation of the L∞‑objective, decreasing the
   smoothing λ from 0.7 down to 1e‑7.
3. Perform a long mirror‑SGD (exponentiated‑gradient) optimisation on the
   exact max‑convolution, using momentum and a decaying step‑size.
4. Finish with a short low‑λ L‑BFGS run and a simple simulated‑annealing
   hill‑climb.

All phases are time‑controlled; the current best sequence is stored in the
global variable ``height_sequence_1``.  The returned list is a non‑negative
probability vector (sum = 1)."""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / (np.arange(1, len(v) + 1)))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    return project_onto_simplex(noisy)


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector using a variety of simple parametric
    families.  The output is guaranteed to lie on the simplex.
    """
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "double_exp",
            "power_edge",
            "sinusoid",
            "gaussian_center",
            "dirichlet",
            "sparse",
            "mixed",
        ]
    )
    x = np.arange(n, dtype=float)
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "double_exp":
        rate_l = rng.uniform(0.001, 0.020)
        rate_r = rng.uniform(0.001, 0.020)
        a = np.exp(-rate_l * x) + np.exp(-rate_r * (n - 1 - x))

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "sinusoid":
        # sin(π·(i+0.5)/n) is positive on (0,π)
        a = np.sin(np.pi * (x + 0.5) / n)

    elif mode == "gaussian_center":
        # single bump centred in the middle
        mu = (n - 1) / 2.0
        sigma = rng.uniform(0.05, 0.20) * n
        a = np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))

    elif mode == "dirichlet":
        alpha = rng.uniform(0.1, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)

    elif mode == "sparse":
        m = rng.integers(max(2, int(0.01 * n)), max(3, int(0.05 * n)))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    else:  # mixed
        # 60 % uniform + 40 % edge‑exponential
        rate = rng.uniform(0.001, 0.020)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * (1.0 / n) + 0.4 * np.exp(-rate * d)

    # avoid exact zeros which can break some gradients
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_iters: int = 5_000_000,
                          jitter_interval: int = 200_000):
    """
    Mirror‑SGD with momentum on the exact max‑convolution.
    Occasionally adds a tiny Gaussian jitter to avoid stagnation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # momentum variables
    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        a = project_onto_simplex(a)

        # cheap periodic score check
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        # occasional jitter to escape flat regions
        if jitter_interval and (it % jitter_interval) == 0 and it > 0:
            a = add_noise(a, sigma=rng.uniform(0.0005, 0.0025), rng=rng)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score
```
