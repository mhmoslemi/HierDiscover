"""
Candidate generator for the step‑function minimisation task.

The algorithm builds a diverse pool of starting vectors (uniform,
random families, classic windows, and several deterministic
parameterised families such as exponential‐ or power‑law decays).
All candidates are normalised to a unit sum and the cheap
``fast_score`` (2·n·max(conv)) is used to keep the best few (≈30)
for an intensive optimisation phase.

Optimisation proceeds in three timed phases:

* Phase 1 – a short L‑BFGS cascade on the smooth log‑sum‑exp
  surrogate (decreasing λ‑schedule).  The best effort of several
  restarts is kept.
* Phase 2 – long exponentiated‑gradient descent (mirrored SGD) on the
  exact max‑convolution objective, with a square‑root stepsize decay
  and occasional Gaussian jitter to escape plateaus.
* Phase 3 – final polishing: a very low‑λ L‑BFGS refinement,
  a tiny moving‑average smoothing, and a short simulated‑annealing
  hill‑climb.

The whole search respects a 1000 s global deadline; the best
sequence found is stored in the global ``height_sequence_1``
variable and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score (sum of sequence = 1):
        2·n·max(conv(a, a))
    Conv is computed by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector using a variety of heuristics."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def shift_pad(a: np.ndarray, shift: int) -> np.ndarray:
    """Shift the whole vector to the right by ``shift`` positions,
    filling the left with zeros (mass is conserved by normalisation)."""
    if shift <= 0:
        return a.copy()
    n = a.shape[0]
    shifted = np.concatenate((np.zeros(shift), a[: n - shift]))
    return shifted / shifted.sum()

def make_power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power law)."""
    x = np.arange(1, n + 1, dtype=float) ** (-p)
    return x / x.sum()


def make_exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (decreasing exponential)."""
    x = np.exp(-lam * np.arange(n, dtype=float))
    return x / x.sum()


def mix_vectors(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex blend α·a + (1‑α)·b, re‑normalised to sum = 1."""
    m = alpha * a + (1.0 - alpha) * b
    return m / m.sum()


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns the smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 30_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    Runs until the deadline or ``max_iters`` is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter if stuck
        if (it - last_improve_it) > 250_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            last_improve_it = it

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of candidate vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) previous best (if any) – interpolated to the new length
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) classic windows (if SciPy is present)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        w = hann(n)
        pool.append(w / w.sum())
        w = blackman(n)
        pool.append(w / w.sum())
    except Exception:
        pass

    # 3) random families (as before)
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # 4) systematic deterministic families
    #    a) power‑law decays
    for p in np.linspace(0.1, 2.5, 25):
        pool.append(make_power_decay(n, p))

    #    b) exponential decays
    for lam in np.logspace(-4, -1, 25):
        pool.append(make_exp_decay(n, lam))

    #    c) mixtures with uniform background
    uniform = np.full(n, 1.0 / n)
    for p in (0.5, 0.8, 1.2):
        base = make_power_decay(n, p)
        for alpha in (0.1, 0.3, 0.5):
            pool.append(mix_vectors(base, uniform, alpha))

    #    d) mixtures of a decay and its reverse (asymmetric U‑shapes)
    for p in (0.5, 1.0):
        base = make_power_decay(n, p)
        rev = base[::-1]
        for beta in (0.2, 0.5):
            pool.append(mix_vectors(base, rev, beta))

    #    e) shifted versions of a few strong candidates (edge‑focused)
    for _ in range(5):
        src = rng.choice(pool)
        shift = rng.integers(0, n // 10 + 1)
        pool.append(shift_pad(src, shift))

    # ------------------------------------------------------------
    # Scan the pool quickly and retain the top 30 candidates
    # ------------------------------------------------------------
    scored = []
    for vec in pool:
        sc = fast_score(vec, fft_len)
        scored.append((sc, vec))
    scored.sort(key=lambda x: x[0])
    top_k = 30
    best_candidates = [vec for _, vec in scored[:top_k]]
    best_score = scored[0][0]
    best_a = scored[0][1].copy()
    print(f"[Init] best fast_score from pool = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.15 * budget_s   # L‑BFGS warm‑up
    phase2_end = start_time + 0.78 * budget_s   # long EG phase

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS cascade on many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.022,
        0.012,
        0.006,
        0.003,
        0.0015,
        7e-4,
        3e-4,
        1.5e-4,
        8e-5,
    ]
    maxiter_per_lam = 1000

    for start_vec in best_candidates:
        if time.time() >= phase1_end:
            break
        a_opt, sc_opt = lbfgs_cascade(
            start_vec,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1] new overall best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=36.0,
            max_iters=40_000_000,   # stop by time
        )
        print(f"[Phase 2] exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Very low‑λ L‑BFGS refinement
        fine_lam_schedule = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase 3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # a light moving‑average smoothing before annealing
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        smoothed_score = fast_score(smoothed, fft_len)
        if smoothed_score < best_score:
            best_a = smoothed
            best_score = smoothed_score
            print(f"[Phase 3‑smooth] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
