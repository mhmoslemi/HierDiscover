* more aggressive Adam/EG settings and a final exhaustive hill‑climb.

All phases respect a total time budget (default 1000 s) and the function
`propose_candidate` returns the best list of non‑negative heights found.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ----------------------------------------------------------------------
# Exact fast objective (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a * a) for a normalised to sum = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Return (f, grad) where f = 2·n·smooth_max(conv) and grad = ∂f/∂y."""
    a = softmax(y)

    # linear convolution a*a via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam)               # (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Generate a spectral start‑vector (square‑root of a flat spectrum)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using several families."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def left_exp_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay placed completely in the left half of the interval."""
    decay = rng.uniform(0.002, 0.02)
    a = np.exp(-decay * np.arange(n))
    # enforce zeros on the right half to keep the left‑hand mass away from the tail
    a[n // 2:] = 0.0
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS for each λ in `lam_schedule` starting from `a_start`."""
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.02):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask of the largest entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max‑conv w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0 else best_a.copy()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[EG] iter={it}, best fast_score={best_sc:.8f}")
        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """Adam optimiser on the exact max‑convolution objective."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         step0: float = 0.05,
                         decay: float = 0.995):
    """
    Simple sub‑gradient descent that directly reduces the current worst
    convolution entry.  The update is projected back onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    step = step0
    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # gradient of conv[k_max] w.r.t. a (sub‑gradient)
        grad = np.zeros_like(a)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = a[j]
        grad *= 2.0

        # step in opposite direction
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else best_a.copy()

        # occasional step decay
        if it % 200 == 0 and it > 0:
            step *= decay

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[SGD] it={it}, best fast_score={best_sc:.8f}")

        it += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """Pure greedy hill‑climb with tiny pairwise mass transfers."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two length works well with FFT
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # warm‑start from a previous solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(left_exp_init(n, rng))

    # randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # pick the best initial candidate (exact fast_score)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Global time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep half‑second safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Extended L‑BFGS (more λ‑steps, more iterations)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1,
                    0.07, 0.05, 0.03, 0.02, 0.01, 0.005,
                    0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Random Dirichlet boost (cheap exhaustive search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        boost_end = time.time() + 0.05 * budget_s
        print("[Boost] random Dirichlet sampling")
        while time.time() < boost_end:
            a = rng.dirichlet(np.ones(n) * rng.uniform(0.1, 5.0))
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[Boost] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient refinements
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.15 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.02)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Sub‑gradient descent targeting the current peak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + 0.10 * budget_s
        print("[Sub‑grad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                               sg_deadline,
                                               n,
                                               fft_len,
                                               rng,
                                               step0=0.08,
                                               decay=0.998)
        print(f"[Sub‑grad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 18.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=40000,
                                     delta_frac=0.03)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate with the official routine
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
