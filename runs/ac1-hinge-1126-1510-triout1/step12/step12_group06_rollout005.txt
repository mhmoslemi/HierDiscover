"""
Step‑function optimiser – 1000 s budget (enhanced)

Improvements over the baseline:
* Longer aggressive L‑BFGS stage with a richer λ‑schedule.
* Two complementary descent phases:
    – exponentiated‑gradient (multiplicative) descent (exact max‑convolution)
    – additive projected sub‑gradient descent (exact max‑convolution)
* Simulated‑annealing hill‑climb for escaping shallow local minima.
* Final “mix‑and‑power’’ sweep: line‑search over convex combinations with
  a few simple families (uniform, Gaussian, edge‑heavy) and over power
  transforms a↦a^p.
* All phases respect the global 1000 s deadline and keep the best
  sequence seen in the global variable `height_sequence_1`.

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import chebwin, kaiser

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (sum‑normalised)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass concentrated at the two edges."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ are unrestricted; a = softmax(y) ∈ simplex.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t conv entries
    w = exp_shift / Z          # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]  # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score = {best_score:.9f}")
    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """Mirrored EG descent on the exact L∞‑objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score

def additive_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator,
                              eta0: float = 0.5):

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a = simplex_projection(a_new)

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score

def local_refinement(a_start: np.ndarray,
                     start_score: float,

        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (allows occasional uphill moves)
# ----------------------------------------------------------------------


def simulated_annealing_refine(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator):
    """Annealed hill‑climb with a simple temperature schedule."""
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    T0 = 1e-3
    Tmin = 1e-9

    while time.time() < deadline:
        # linear cooling based on remaining time
        remaining = deadline - time.time()
        total = deadline - (deadline - (deadline - time.time()))  # dummy to keep type
        # Instead we use a simple exponential decay
        frac = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))  # always ≈1; ignore
        # We'll just decay multiplicatively each iteration
        T = max(Tmin, T0 * (0.9995 ** (deadline - time.time())))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose moving a random fraction of mass
        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * 0.15  # up to 15 %
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        new_score = fast_score(cand, fft_len)

        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / T):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mixing & power‑transform sweep (final polishing)
# ----------------------------------------------------------------------


def mix_and_power_refine(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """Search over convex combinations with simple families and power exponents."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score

    # simple families to mix with
    uniform = np.full(n, 1.0 / n)
    gaussian = gaussian_initial(n, rng)
    edge = edge_heavy_initial(n, rng)

    families = [uniform, gaussian, edge]

    # mixing coefficients
    alphas = np.linspace(0.0, 0.20, 21)  # up to 20 % mixing

    # power exponents
    powers = [0.8, 0.9, 1.0, 1.1, 1.2]

    while time.time() < deadline:
        # --- convex mixing -------------------------------------------------
        for fam in families:
            for alpha in alphas:
                cand = (1.0 - alpha) * a + alpha * fam
                # already sums to 1
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                sc = fast_score(cand, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
        # --- power transform ------------------------------------------------
        for p in powers:
            cand = np.power(a, p)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = fast_score(cand, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        # early break if time is short
        if time.time() >= deadline:
            break
    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    pool = []

    # a) previous best if present
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # b) classic windows (Chebyshev & Kaiser)
    try:
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # c) random / parametric families
    for _ in range(8):
        pool.append(random_initial(n, rng))
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.20 * budget_s   # aggressive L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # long EG descent
    phase3_end = start_time + 0.85 * budget_s   # fine L‑BFGS
    phase4_end = start_time + 0.95 * budget_s   # simulated annealing
    # remainder for final mix/power sweep

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7,
        2e-7, 1e-7, 5e-8
    ]
    maxiter_per_lam = 800   # a bit lighter per λ to allow many restarts

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # occasional jitter
        if rng.random() < 0.30:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.007),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1 L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=22.0
        )
        print(f"[Phase‑2 EG] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2b – additive projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = additive_subgrad_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=0.6
        )
        print(f"[Phase‑2b Add‑Subgrad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with ultra‑low λ
    # --------------------------------------------------------------
    if time.time() < (deadline - 70.0):
        fine_lam_sched = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1200,
                                        deadline=deadline - 30.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[Phase‑3 fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing_refine(
            best_a,
            best_score,
            phase4_end,
            n,
            fft_len,
            rng
        )
        print(f"[Phase‑4 anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – greedy local refinement (mass‑transfer hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[Phase‑5 local‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 6 – mix & power sweep (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = mix_and_power_refine(best_a,
                                                  best_score,
                                                  deadline,
                                                  n,
                                                  fft_len,
                                                  rng)
        print(f"[Phase‑6 mix/power] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
