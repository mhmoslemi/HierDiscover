"""
Optimise a non‑negative step‑function for the objective

    score = 2 * n * max(convolve(a, a)) / (sum(a)**2)

The algorithm consists of three stages.

Stage 1 – many short L‑BFGS runs on a log‑sum‑exp surrogate.
Stage 2 – a long “Adam‑style” mirror descent on the true
          L∞‑objective (the gradient of the max‑convolution is computed
          by FFT).  If progress stalls we jitter the logits.
Stage 3 – a final low‑λ L‑BFGS polish and a short simulated‑annealing
          refinement.

All work is done under the 1000 s deadline; the best sequence found
is returned and stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score when ``a`` sums to 1:
    2·n·max(conv(a,a))."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector on ``n`` entries."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# L‑BFGS on the smooth surrogate
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution objective.
    y – log‑variables; a = softmax(y).
    Returns (value, gradient w.r.t. y).
    """
    # ----- softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution via FFT -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns best vector (probability simplex) and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑variables
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        # back‑to probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score

def adam_max_conv_optimize(a_start: np.ndarray,
                           deadline: float,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator,
                           alpha0: float = 0.12,
                           max_iters: int = 20_000_000,
                           patience: int = 300_000):
    """
    Adam optimiser on the exact L∞‑objective (max‑convolution).
    Works in the softmax (log‑probability) space.
    """
    # initialise log‑variables
    y = np.log(a_start + 1e-12)
    m1 = np.zeros_like(y)      # first moment
    m2 = np.zeros_like(y)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        # ----- probability vector -----
        mx = np.max(y)
        e = np.exp(y - mx)
        a = e / e.sum()

        # ----- convolution -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # ----- gradient of the (hard) max -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_rev = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_rev[n - 1: n - 1 + n]          # size n

        # ----- back‑prop through softmax -----
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)                     # shape (n,)

        # ----- Adam update -----
        it1 = it + 1
        m1 = beta1 * m1 + (1.0 - beta1) * grad_y
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_y * grad_y)

        m1_hat = m1 / (1.0 - beta1 ** it1)
        m2_hat = m2 / (1.0 - beta2 ** it1)

        step = alpha0 / np.sqrt(it1)                  # decaying base step
        y = y - step * m1_hat / (np.sqrt(m2_hat) + eps)

        # ----- occasional evaluation -----
        if (it & 63) == 0:        # every 64 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                print(f"[Adam] it={it} fast_score={best_score:.9f}")

        # ----- stagnation handling -----
        if (it - last_improve_it) > patience:
            # jitter the logits to escape a possible plateau
            sigma = rng.uniform(8e-4, 3e-3)
            y += rng.normal(0.0, sigma, size=y.shape)
            # reset moments to avoid stale direction
            m1.fill(0.0)
            m2.fill(0.0)
            last_improve_it = it
            print(f"[Adam] jitter at it={it}")

        it += 1

    return best_a, best_score

def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """Random mass‑exchange moves accepted with a linear‑cooled Metropolis rule."""
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.04, 1e-9)
    max_frac = 0.18

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

                                rng=rng)
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # Symmetry sometimes helps
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam mirror descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_max_conv_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            alpha0=0.14,
            max_iters=40_000_000,
            patience=250_000,
        )
        print(f"[Phase2‑Adam] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS polish + tiny anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_ref, sc_ref = lbfgs_cascade(
            best_a,
            fine_lam,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_ref < best_score:
            best_a = a_ref
            best_score = sc_ref
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # very light smoothing before anneal
        kernel = np.ones(3) / 3.0
        sm = np.convolve(best_a, kernel, mode="same")
        sm = np.maximum(sm, 0.0)
        sm /= sm.sum()
        sm_score = fast_score(sm, fft_len)
        if sm_score < best_score:
            best_a = sm
            best_score = sm_score
            print(f"[Phase3‑smooth] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final verification with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
