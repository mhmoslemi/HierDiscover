def lp_step(a, M, target_sum):
    n = len(a)
    from scipy.sparse import diags
    from scipy.optimize import linprog
    # Build constraint matrix
    diag_vals = [np.full(n, a[i]) for i in range(n)]
    offsets = np.arange(n)
    A = diags(diag_vals, offsets, shape=(2*n-1, n), format='csr')
    # RHS
    b = np.full(2*n-1, M)
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return same a
        return a.copy()
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a.copy()
    scale = target_sum / sum_g
    g *= scale
    return g
```

"""
Propose a low‑score step function.

The algorithm works in three stages:

1. **Smooth L‑BFGS** – Optimise a probability vector on the simplex
   using a log‑sum‑exp smoothed version of the peak of the
   autocorrelation.  This quickly drives the objective down from
   the initial construction.

2. **Linear‑programming mixing** – Given the current sequence `a`,
   solve a linear program that finds a sequence `g` with the same
   sum but with a *larger* L1‑norm while respecting the bound
   `a * g ≤ max(a*a)`.  By binary‑searching a mixing factor `t`
   we can replace `a ← (1‑t)·a + t·g` without increasing the
   peak autocorrelation.  Repeating this step steadily reduces the
   bound.

3. **Final greedy polishing** – Tiny pairwise mass transfers are
   applied whenever they lower the peak value, giving a last
   refinement.

All phases respect the 1000 s budget.  The final list of floats is
returned; `evaluate_sequence` is called only once at the end.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Fast evaluation helpers (sum = 1 ⇔ score = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(prob: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    prob = np.asarray(prob, dtype=np.float64)
    fa = np.fft.rfft(prob, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_conv(a: np.ndarray, fft_len: int) -> float:
    """Maximum of a*a (no normalisation)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f    = 2·n·smooth_max(conv)
        grad = ∂f/∂y
    y are the logits (a = softmax(y)).
    """
    a = softmax(y)

    # a * a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam)              # (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smoothed max
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (only the central n entries are needed)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple λ values)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, starting from the
    previous result.  Returns best probability vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming step (maximize L1 norm under convolution bound)
# ----------------------------------------------------------------------


def lp_step(a: np.ndarray, M: float, target_sum: float):
    """
    Solve
        max   sum(g)
        s.t.  a * g ≤ M   (convolution)
              g ≥ 0
    Scale the solution to have sum = target_sum.
    """
    n = a.shape[0]
    # Build sparse Toeplitz matrix A where each row k contains a_i
    # at column j = k - i (if 0 ≤ j < n).
    # This is exactly a set of n constant diagonals.
    diag_data = [np.full(n, a[i]) for i in range(n)]
    offsets = np.arange(n)
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')

    b = np.full(2 * n - 1, M * (1.0 + 1e-12))   # tiny slack for numeric robustness
    c = -np.ones(n)                            # minimise -sum(g)  → maximise sum(g)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        # fall back to the current sequence
        return a.copy()

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return a.copy()
    # Scale to the required total mass
    g *= target_sum / s
    return g


# ----------------------------------------------------------------------
# Helper for binary‑search blending (guaranteed non‑increase)
# ----------------------------------------------------------------------


def blend_with_g(a: np.ndarray, g: np.ndarray, M: float,
                 fft_len: int, max_iter: int = 12):
    """
    Find the largest t∈[0,1] such that
        max_conv((1‑t)·a + t·g) ≤ M .
    Returns the blended vector and its new maximal convolution value.
    """
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        cand = (1.0 - mid) * a + mid * g
        m = max_conv(cand, fft_len)
        if m <= M:
            lo = mid
        else:
            hi = mid
    t = lo
    a_new = (1.0 - t) * a + t * g
    new_M = max_conv(a_new, fft_len)
    return a_new, new_M


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (tiny pairwise moves)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_M: float,
               fft_len: int, deadline: float,
               rng: np.random.Generator, max_iter: int = 50000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny mass transfers.
    """
    best = a.copy()
    best_M = cur_M
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        m = max_conv(cand, fft_len)
        if m < best_M:
            best = cand
            best_M = m
            # occasional progress output
            if it % 5000 == 0:
                print(f"[hill] iter {it}, improved M = {best_M:.8f}")
        it += 1
    return best, best_M

        a0 = prob_best * target_sum
        cur_M = max_conv(a0, fft_len)
        print(f"[Phase‑1] post‑L‑BFGS max_conv = {cur_M:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – LP‑based mixing (monotone decrease of the peak)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s        # use up to 80 % of time
    while time.time() < phase2_end:
        # solve LP for the current a0
        g = lp_step(a0, cur_M, target_sum)

        # blend with binary search to keep the peak non‑increasing
        a_new, new_M = blend_with_g(a0, g, cur_M, fft_len)

        # accept if we made strict progress
        if new_M < cur_M - 1e-12:
            a0 = a_new
            cur_M = new_M
            print(f"[LP] improved max_conv = {cur_M:.8f}")
        else:
            # if no progress, break to avoid endless loop
            break

    # --------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (tiny moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        a0, cur_M = hill_climb(a0, cur_M,
                               fft_len,
                               hill_deadline,
                               rng,
                               max_iter=80000,
                               delta_frac=0.035)
        print(f"[Hill‑climb] final max_conv = {cur_M:.8f}")

    # --------------------------------------------------------------
    # Return the sequence (evaluate_sequence rescales internally)
    # --------------------------------------------------------------
    final_seq = a0.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.8f}")
    return final_seq
