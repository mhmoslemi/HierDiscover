def smooth_max_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = np.sum(exp_shift)
    w = exp_shift / Z
    # gradient wrt a via reversed a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[n-1:n-1+n]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    smooth_max = max_c + lam * np.log(Z)
    f = 2 * n * smooth_max
    return f, grad_y, a, max_c
```

"""
Optimisation of a non‑negative step function (probability vector) that
minimises 2·n·max(conv(a,a)) / (sum a)^2.
The algorithm consists of:

1. Generate a diverse pool of initial candidates (deterministic windows,
   random Dirichlet‑like vectors, and the previous best if available).
2. Pick the best candidate w.r.t. the exact objective (fast_score).
3. Refine the candidate with a continuation method:
   * Smooth the maximum via a log‑sum‑exp approximation.
   * Optimise the smoothed objective with Adam on the soft‑max logits.
   * Decrease the smoothing parameter gradually (λ‑schedule) while
     preserving the current iterate.
4. Apply a short, deterministic greedy hill‑climb that moves a tiny amount
   of mass between two coordinates whenever it reduces the exact score.
5. Return the best sequence found (list of floats).  All values are
   non‑negative and the sum is exactly 1, thus the official
   `evaluate_sequence` returns the same number as our fast_score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Fast exact objective (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact value of 2·n·max(a*a) for a on the simplex (sum a = 1).
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisers (a family of plausible shapes)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Take square‑root of a uniform spectrum, project to non‑negative simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex from several heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – Adam optimisation
# ----------------------------------------------------------------------


def smooth_max_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam_factor: float) -> tuple[float, np.ndarray, np.ndarray, float]:
    """
    Returns (smooth_obj, grad_y, a, exact_max) where
    - smooth_obj = 2·n·smooth_max(conv)
    - grad_y is the gradient w.r.t. logits y (softmax parametrisation)
    - a = softmax(y) (probability vector, sum = 1)
    - exact_max = max(conv) used for fast_score and diagnostics
    """
    a = softmax(y)

    # full linear convolution via zero‑padded FFT
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c            # ≤ 0
    exp_shift = np.exp(shifted / lam)     # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z                     # (2n‑1,) weights, sum to 1

    # back‑prop through convolution a*a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # gradient wrt a (size n)
    g_vec = conv_rev[n - 1:n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    smooth_obj = float(2.0 * n * smooth_max)
    return smooth_obj, grad_y, a, float(max_c)


def smooth_max_adam_optimize(a_start: np.ndarray,
                             n: int,
                             fft_len: int,
                             lam_schedule: list[float],
                             iters_per_lambda: int,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Continuation optimisation on the smoothed max‑convolution.
    Parameters
    ----------
    a_start : initial probability vector (sum = 1)
    lam_schedule : list of λ‑factors (smaller → less smoothing)
    iters_per_lambda : number of Adam steps per λ value
    deadline : absolute time after which we must stop
    Returns best (a, fast_score) seen during the run.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam_factor in lam_schedule:
        # reset Adam accumulators at each new smoothing level
        m.fill(0.0)
        v.fill(0.0)

        for _ in range(iters_per_lambda):
            if time.time() >= deadline:
                return best_a, best_sc

            smooth_obj, grad_y, a, max_c = smooth_max_and_grad(
                y, n, fft_len, lam_factor)

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.2 / np.sqrt(t)          # simple decay
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # exact score (no smoothing)
            sc = 2.0 * n * max_c
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        # optional brief print
        # print(f"[smooth‑Adam] λ={lam_factor:.5g}, best_sc={best_sc:.8f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple greedy hill‑climb for final polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that transfers a tiny amount of mass between
    two random coordinates whenever it strictly reduces the exact score.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = cand
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function within the given time budget.
    Returns a plain Python list of floats (the heights).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power‑of‑two FFT length works best)
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)   # 8192

    # --------------------------------------------------------------
    # Build a pool of plausible starting vectors
    # --------------------------------------------------------------
    candidates = []

    # 1) previous best if it exists (global variable set by a prior run)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to current length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # 3) a few random draws
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # Choose the best initial candidate (exact score)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Continuation optimisation on smoothed max (Adam)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s   # ~55 % of total time
    lam_schedule = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    iters_per_lambda = 1500   # modest number of Adam steps per λ level

    # we will run a few restarts, each blending the current best with fresh noise
    restarts = 4
    for r in range(restarts):
        if time.time() >= phase1_end:
            break
        # blend current best with a random Dirichlet‑like vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = smooth_max_adam_optimize(
            a_start,
            n,
            fft_len,
            lam_schedule,
            iters_per_lambda,
            phase1_end,
            rng)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[smooth‑Adam‑restart {r+1}] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient Adam (focus on current max entry)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = time.time() + 0.25 * budget_s   # approx 25 % of total time
        # simple Adam on the true max (sub‑gradient)
        y = np.log(best_a + 1e-12)
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t = 0
        while time.time() < phase2_end:
            a = softmax(y)
            # convolution and identify the (possibly several) maximal entries
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
            max_c = conv.max()
            # sub‑gradient mask (focus on entries that are (almost) maximal)
            mask = (conv >= max_c - 1e-14).astype(np.float64)

            # gradient of the max entry w.r.t. a
            a_rev = a[::-1]
            fa_rev = np.fft.rfft(a_rev, fft_len)
            fm = np.fft.rfft(mask, fft_len)
            rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
            g_a = 2.0 * rev_grad[n - 1:n - 1 + n]

            # full gradient of the objective 2·n·max_c
            grad_a = 2.0 * n * g_a

            # back‑prop through softmax
            mean_g = np.dot(a, grad_a)
            grad_y = a * (grad_a - mean_g)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = 0.15 / np.sqrt(t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep best
            sc = 2.0 * n * max_c
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[exact‑Adam] iter={t}, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 18.0)
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=40000,
                                     delta_frac=0.04)
        print(f"[hill‑climb] final best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Convert to plain list and evaluate with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official scoring (includes clipping)
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
