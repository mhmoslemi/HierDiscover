def lp_refine(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0.5  # mixing weight, may adjust later
    while time.time() < deadline:
        # Compute max conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n-1]
        M = conv.max()
        # Build convolution matrix A
        rows = np.empty(n*n, dtype=np.int32)
        cols = np.empty(n*n, dtype=np.int32)
        data = np.empty(n*n, dtype=np.float64)
        idx = 0
        for i in range(n):
            rows[idx:idx+n] = np.arange(i, i+n, dtype=np.int32)
            cols[idx:idx+n] = i
            data[idx:idx+n] = a[i]
            idx += n
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

        # Solve LP
        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n
        ub = np.full(2*n-1, M, dtype=np.float64)
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] exception:", e)
            break
        if not res.success:
            print("[LP] not successful")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            break
        g = g0 / sum_g0

        # mix
        a_new = (1 - t) * a + t * g
        # compute new score
        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] improved fast_score={best_score:.8f}")
        else:
            # if no improvement, maybe reduce t
            t *= 0.5
            if t < 1e-3:
                break
    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm outline
-----------------
1. **Diverse initialization** – a rich pool (deterministic windows,
   random draws and high‑beta Kaiser windows) is generated and the best
   candidate (by the exact fast_score) is kept.

2. **Smoothed L‑BFGS** – the log‑sum‑exp smoothed max of the
   self‑convolution is minimised.  The smoothing factor (lam_factor)
   is decreased through a long schedule and each stage runs many
   L‑BFGS iterations.  Random restarts are mixed with the current
   best to avoid poor basins.

3. **Exact Adam with top‑k mask** – Adam optimises the true objective
   using a mask that contains the largest *k* convolution entries.
   The fraction *k* decays from a large value (30 % of all shifts) to
   a tiny value (0.5 %).  This flattens the whole autocorrelation
   surface.

4. **Simulated‑annealing & greedy hill‑climb** – simple mass‑transfer
   moves finish off the local optimisation.

5. **Linear‑programming refinement** – a convex LP (maximise ∑g
   under the linear constraints (f∗g)≤M) is solved.  The optimal g
   is mixed back into f; each successful mix provably does not raise
   the maximal convolution value.  A few such refinements are
   performed at the end.

All phases respect the 1000 s budget.  The returned list is a
non‑negative step‑function; ``evaluate_sequence`` rescales it
internally, so the absolute scale is irrelevant.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse
from scipy.signal import kaiser, tukey

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective used for bookkeeping (probability vector ⇒ sum=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    # objective = 2·n·max(conv) (since sum(a)=1)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_high_beta_init(n: int, rng: np.random.Generator) -> np.ndarray:
    beta = rng.uniform(12.0, 22.0)   # more aggressive tapering
    return kaiser_init(n, beta)


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def tukey_init(n: int, rng: np.random.Generator) -> np.ndarray:
    # alpha in (0,1]; higher alpha = more cosine taper (broader)
    alpha = rng.uniform(0.4, 0.9)
    a = tukey(n, alpha)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = scipy.optimize.minimize(fun=objective_and_grad,
                                          x0=y,
                                          args=(n, fft_len_conv, fft_len_g, lam),
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': maxiter_per_lam,
                                                   'gtol': 1e-12,
                                                   'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.30,
                       end_top: float = 0.005):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    The top‑k fraction linearly decays from start_top to end_top.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_iter = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t_iter += 1
        a = softmax(y)

        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction (decay)
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]  # d(max)/d a (ignoring 2·n factor)
        grad_a = 2.0 * n * g_a                     # full objective gradient

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t_iter)          # decreasing learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t_iter % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 250000):
    """
    Simple simulated‑annealing over the simplex.  Small mass‑transfer
    moves are proposed; the acceptance probability follows a geometric
    temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.18   # up to ~18 % of source mass
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        # accept if better or with Metropolis probability
        if sc < best_score:
            a = cand
            best_score = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ g = conv(f,g) where f=a.
    Each column i contains the constant value a[i] on the rows i … i+n‑1.
    """
    n = a.shape[0]
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel().astype(np.int32)
    cols = np.repeat(np.arange(n), n).astype(np.int32)
    data = np.repeat(a, n).astype(np.float64)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int):
    """
    Perform a few LP refinements:
    maximise ∑g subject to (a∗g) ≤ M, g ≥ 0.
    The resulting g is mixed back into a.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    mix_factor = 0.5  # start with a 50/50 mix; will be reduced on failure

    while time.time() < deadline:
        # current maximal autocorrelation
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build linear constraint matrix
        A = build_conv_matrix(a)

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve LP with HiGHS (sparse support)
        try:
            res = scipy.optimize.linprog(c,
                                         A_ub=A,
                                         b_ub=ub,
                                         bounds=bounds,
                                         method='highs',
                                         options={'presolve': True})
        except Exception as e:  # pragma: no cover
            print("[LP] exception:", e)
            break

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # scale g to unit sum (our normalisation for fast_score)
        g = g0 / sum_g0

        # convex combination
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        sc = fast_score(a_new, fft_len, n)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            print(f"[LP] improved fast_score={best_score:.8f}")
            # keep the same mixing factor for the next iteration
        else:
            # no improvement – reduce mixing factor to stay closer to current a
            mix_factor *= 0.5
            if mix_factor < 1e-3:
                break

    return best_a, best_score

    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------

    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # random restart mixed with current best
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ---------- Phase 2 – Adam with top‑k mask (≈45 % of total time) ----------
    if time.time() < deadline:
        print("[Adam‑topk] start exact optimisation")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.30,
            end_top=0.005
        )
        print(f"[Adam‑topk] best fast_score={best_score:.8f}")

    # ---------- Phase 3 – Simulated annealing (≈20 % of total time) ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.20 * budget_s)
        best_a, best_score = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-8,
            max_moves=250000
        )
        print(f"[Sim‑Anneal] best fast_score={best_score:.8f}")

    # ---------- Phase 4 – final greedy hill‑climb (≈5 % of total time) ----------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=60000,
            delta_frac=0.04
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ---------- Phase 5 – LP refinement (≈5 % of total time) ----------
    if time.time() < deadline:
        lp_deadline = deadline
        best_a, best_score = lp_refine(best_a, lp_deadline, n, fft_len)
        print(f"[LP‑refine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
