"""
Step‑function optimiser – 1000 s budget (enhanced, LP‑guided)

Algorithm outline
-----------------
0. Global set‑up – choose a moderate length (8000) and pre‑compute FFT size.
1. Warm‑starts – generate a large set of diverse probability vectors
   (uniform, edge‑exponential, edge‑power, bimodal, sparse, mixed,
   triangular).  Keep the best according to the cheap fast_score.
2. L‑BFGS cascade – minimise a smooth log‑sum‑exp surrogate of the
   L∞‑objective.  Lambda is reduced from 0.7 down to 1e‑8, each stage
   allowed up to 3000 iterations.  The best vector is retained.
3. Linear‑programme (LP) improvement – with the current vector *f*,
   compute its worst convolution value *M* and collect the active
   shifts (where the autocorrelation is close to *M*).  Solve a single
   LP that maximises the L1‑mass of a new vector *g* under the linear
   constraints  (f∗g)[k] ≤ M  for all active shifts.  Mix *g* into *f*
   with a small step‑size *t* and repeat a few times.
4. Exponentiated‑gradient descent – a momentum‑averaged sub‑gradient
   descent on the exact L∞‑objective.  The sub‑gradient is taken as
   the average of the active gradients (instead of their sum) for a
   smoother descent.
5. Final polish – a short low‑λ L‑BFGS pass followed by a quick
   simulated‑annealing hill‑climb.

All steps respect the global 1000 s deadline; the best sequence found
is stored in ``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse probability vector using several families."""
    mode = rng.choice(
        ['uniform', 'exp_edge', 'power_edge', 'bimodal', 'sparse',
         'mixed', 'triangular']
    )
    x = np.arange(n)
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_edge':
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == 'power_edge':
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == 'bimodal':
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == 'sparse':
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == 'mixed':
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    else:   # triangular
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(x - centre) / centre
        a = np.maximum(a, 0.0)

    # avoid exact zeros, then normalise
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, lam),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_per_lam,
                     'gtol': 1e-12,
                     'disp': False}
        )
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 3_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution.
    The sub‑gradient is the *average* of all active gradients,
    which gives a valid (and often smoother) sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - eps)[0]

        if active.size == 0:
            mask = np.zeros_like(conv)
        else:
            mask = np.zeros_like(conv)
            mask[active] = 1.0 / active.size   # average over active set

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # cheap periodic score check
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programme improvement (active‑set LP)
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               t: float = 0.2,
               eps_factor: float = 1e-6,
               max_iters: int = 5):
    """
    Given a current probability vector a, solve a small LP that maximises
    the L1‑mass of a new vector g under the active convolution constraints
    (f∗g)[k] ≤ M for all k where the current autocorrelation is within
    eps_factor·M of its maximum M.  Mix the solution back into a.
    """
    n = len(a)
    fft_len = next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current autocorrelation
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        eps = eps_factor * M
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            break

        # build A_ub (|active| × n) where each row k encodes (a∗g)[k] ≤ M
        rows = []
        for k in active:
            row = np.zeros(n)
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            js = np.arange(start, end + 1)
            coeffs = a[k - js]          # a[i] where i = k - j
            row[js] = coeffs
            rows.append(row)

        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M)
        c = -np.ones(n)                   # maximise sum(g)  ≡  minimise -sum(g)

        # bounds: g_j ≥ 0
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs')
        except Exception as e:
            print(f"[LP] linprog exception: {e}")
            break

        if not res.success:
            # No feasible improvement – stop
            print(f"[LP] iteration {it}: linprog failed ({res.message})")
            break

        g = res.x
        sumg = g.sum()
        if sumg <= 0.0:
            break

        # normalise g to be a probability vector, then mix
        g_norm = g / sumg
        a = (1.0 - t) * a + t * g_norm
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_score = fast_score(a, fft_len)
        print(f"[LP] iter={it} fast_score={cur_score:.9f}")

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – larger n gives a finer discretisation
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Timing allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.18 * budget_s       # warm‑starts
    phase2_end = start_time + 0.50 * budget_s       # L‑BFGS cascade
    phase3_end = start_time + 0.65 * budget_s       # LP improvement
    phase4_end = start_time + 0.90 * budget_s       # exponentiated‑gradient

    # ------------------------------------------------------------------
    # Stage 1 – generate many parametric shapes and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')

    # reuse previous construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] previous best fast_score={best_score:.9f}")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)

        # jitter a promising candidate occasionally
        if best_a is not None and rng.random() < 0.15:
            a = add_noise(a, sigma=rng.uniform(0.001, 0.008), rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS cascade with a fine λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.15, 0.10,
            0.07, 0.05, 0.03, 0.02, 0.01,
            0.005, 0.003, 0.0015, 0.0008,
            0.0004, 0.0002, 0.0001, 5e-05, 1e-05,
            5e-06, 2e-06, 1e-06, 5e-07, 2e-07,
            1e-07, 5e-08, 1e-08
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam=3000,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 3 – small LP‑guided improvement (active‑set only)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a = lp_improve(
            best_a,
            deadline=phase3_end,
            rng=rng,
            t=0.25,
            eps_factor=1e-6,
            max_iters=6
        )
        best_score = fast_score(best_a, fft_len)
        print(f"[LP‑improve] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 4 – long exponentiated‑gradient descent on the true L∞ objective
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase4_end,
            n,
            fft_len,
            rng,
            eta0=50.0,
            max_iters=3_000_000
        )
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Stage 5 – final polish: very low‑λ L‑BFGS + short anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07, 5e-08]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
