"""
Propose a low‑evaluation step‑function.

The optimiser works in four stages:

1. **Massive warm‑start generation** – many parametric families
   (edge‑exponential, edge‑power, beta‑shaped, linear, random
   Dirichlet, mixtures).  The 5 best probability vectors (fast L∞
   proxy) are kept.

2. **L‑BFGS cascade on the log‑sum‑exp surrogate** – each kept vector is
   refined through a decreasing λ‑schedule.  The best refined vector
   is retained.

3. **Long exponentiated‑gradient descent** – a mirror‑SGD (AdaGrad‑style
   momentum) runs on the *exact* max‑convolution objective for the
   remaining time slice.

4. **Final polishing** – a short low‑λ L‑BFGS run followed by a simple
   simulated‑annealing mass‑transfer hill‑climb.

All stages respect the global 1000 s deadline; the best
sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.  ``a`` must be non‑negative.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start families
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random symmetric (or not) probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "linear_edge",
            "beta",
            "bimodal",
            "sparse",
            "mixed",
            "dirichlet",
            "gauss_mixture",
        ]
    )
    x = np.arange(n, dtype=float)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        # a_i ∝ exp(-rate * distance to nearest edge)
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "linear_edge":
        # triangular decreasing to centre
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 1e-12)

    elif mode == "beta":
        # a_i ∝ (i+1)^{alpha-1} (n-i)^{beta-1}
        alpha = rng.uniform(0.2, 3.0)
        beta = rng.uniform(0.2, 3.0)
        a = (x + 1) ** (alpha - 1) * (n - x) ** (beta - 1)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.04, 0.15) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), max(int(0.07 * n), 1))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        # 70% uniform + 30% edge‑exponential
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.7 * np.full(n, 1.0 / n) + 0.3 * np.exp(-rate * d)

    elif mode == "dirichlet":
        alpha_vec = rng.uniform(0.1, 5.0, size=n)
        a = rng.dirichlet(alpha_vec)

    else:  # gauss_mixture
        # mixture of up to 3 Gaussians with random weights
        k = rng.integers(1, 4)
        a = np.zeros_like(x)
        total_w = 0.0
        for _ in range(k):
            mu = rng.uniform(0.0, 1.0) * (n - 1)
            sigma = rng.uniform(0.02, 0.12) * n
            w = rng.random()
            a += w * np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))
            total_w += w
        a /= total_w

    # avoid exact zeros (log‑softmax needs positivity) and normalise
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    Returns (objective, gradient) as required by scipy.optimize.minimize.
    """
    # ---- softmax to obtain a valid probability vector ----
    m_y = np.max(y)
    e = np.exp(y - m_y)
    a = e / e.sum()

    # ---- convolution via FFT ----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter that scales with magnitude (helps numerical stability)
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. convolution entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i   (size 2n‑1)

    # gradient w.r.t. a : conv = a * a  (linear convolution)
    # We need d f / d a = 2 * (w * a_rev)  (where * is convolution)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part corresponding to indices n‑1 … n‑1+n‑1 (size n)
    g_vec = grad_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  verbose: bool = False):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue

        y = res.x
        # convert back to probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(
                    f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          beta: float = 0.9,
                          max_iters: int = 8_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) directly on the max‑convolution.
    Uses a momentum‑averaged sub‑gradient and step‑size η₀/√t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # momentum accumulator
    g_avg = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- convolution ----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12

        # find all entries that are (almost) maximal
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution wrt a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        # the gradient of max(conv) w.r.t. a is 2 * the slice of grad_conv
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clamped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        # cheap periodic check of the fast proxy score
        if (it & 31) == 0:  # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb (mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers with a Metropolis acceptance rule.
    Temperature is linearly cooled from an initial value.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.08, 1e-12)
    max_frac = 0.25  # maximum fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j

        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score
