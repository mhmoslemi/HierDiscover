"""
Step‑function optimiser (≈1000 s)

The optimiser works in four stages:

1. **Initialisation** – reuse the previous best (`height_sequence_1`);
   otherwise start from a random “probability vector” (sum = 1).

2. **Brief L‑BFGS polishing** – minimise a smooth log‑sum‑exp surrogate
   of the true ‑∞‑objective.  A short cascade (few λ‑values) gives a
   decent starting point for the later stages.

3. **Linear‑programming refinement** – given the current sequence `a`,
   we solve a linear programme that maximises the total mass `∑b`
   subject to the convolution bound `a * b ≤ max(a * a)`.
   The solution `b` is mixed back into `a` (small convex step)
   which provably never increases the current maximum convolution.
   Repeating this step drives the peak down.

4. **Exact sub‑gradient descent (mirrored EG)** – after the LP stage we
   run a long exponentiated‑gradient descent on the exact max‑convolution
   to fine‑tune the shape.

5. **Greedy local refinement** – final hill‑climbing moves that
   transfer a tiny fraction of mass between two positions and accept
   only strict improvements.

All phases respect the global 1000 s deadline and the best sequence
found is stored in the global variable `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) – fast surrogate (assumes sum(a)=1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ unrestricted; a = softmax(y) ∈ simplex.
    """
    # softmax → simplex
    mmax = np.max(y)
    e = np.exp(y - mmax)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Aggressive L‑BFGS on a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 18.0):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    Update: a := a * exp(-η·g) followed by simplex renormalisation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: repeatedly transfer a small random fraction of mass
    from one position to another.  Only moves that strictly improve the
    `2·n·max(conv)` score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

# Linear‑programming refinement (core novelty)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              fft_len: int,
              deadline: float,
              n: int):
    """
    Perform a few iterations of LP‑based refinement.
    Each iteration:
        * compute the current maximal convolution value M,
        * collect the (few) active indices where the bound is tight,
        * solve   max ∑b   s.t.  a * b ≤ M   and b ≥ 0,
        * mix a towards the normalised solution b (line‑search).
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        # current convolution and bound
        fa = np.fft.fft(best_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # active indices (within a tiny relative tolerance)
        tol = max(1e-12, M * 1e-8)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break

        # Build a sparse matrix A (|active| × n) where
        #   (A b)_k = sum_i a_i * b_{k-i}
        rows = active.size
        A = lil_matrix((rows, n), dtype=np.float64)
        for r, k in enumerate(active):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            idx = np.arange(j_start, j_end + 1, dtype=int)
            A[r, idx] = best_a[k - idx]
        A = A.tocsr()

        # Linear programme – maximise sum(b)  (i.e. minimise -∑b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        if b.sum() <= 0.0:
            break

        # Normalise the LP solution (probability vector)
        b_norm = b / b.sum()

        # Line search on the mixing factor α
        improved = False
        for alpha in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - alpha) * best_a + alpha * b_norm
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                best_a = cand
                best_score = cand_score
                improved = True
                break
        if not improved:
            # No progress – exit early
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 gives a good speed/quality trade‑off
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    # ------------------------------------------------------------------
    # 0) Starting point – reuse previous best if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(globals()['height_sequence_1'], dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = random_initial(n, rng)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # 1) Short L‑BFGS polishing (aggressive λ‑schedule)
    # ------------------------------------------------------------------
    lam_schedule = [0.6, 0.33, 0.18, 0.09, 0.04, 0.015, 0.006, 0.0025, 0.001, 5e-4, 2e-4]
    phase1_deadline = start_time + 0.12 * budget_s
    best_a, best_score = lbfgs_cascade(best_a,
                                       lam_schedule,
                                       maxiter_per_lam=800,
                                       deadline=phase1_deadline,
                                       fft_len=fft_len,
                                       n=n)
    print(f"[phase‑1 L‑BFGS] best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement (core improvement)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.55 * budget_s
    best_a, best_score = lp_refine(best_a, fft_len, lp_deadline, n)
    print(f"[LP‑refine] best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Exact mirror‑EG descent (long fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        exp_deadline = deadline - 80.0
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   exp_deadline,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   eta0=22.0)
        print(f"[exp‑grad] best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 4) Final greedy hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[local‑refine] best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 5) Final evaluation with the provided scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
