def projected_subgradient_optimize(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()

        mask = (conv >= max_val - eps).astype(np.float64)  # shape (2n-1,)

        fa_rev = np.fft.fft(a_rev, fft_len)

                best_a = a.copy()

def projected_subgradient_descent(a_start, deadline, n, fft_len, rng, max_iters=5_000_000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    eta0 = 0.1
    beta = 0.9
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        grad = 2 * np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len))) [n-1:n-1+n]
        # momentum
        eta = eta0 / np.sqrt(it+1)
        v = beta * v + (1-beta) * grad
        a_new = a - eta * v
        # project onto simplex
        a_new = simplex_projection(a_new)
        a = a_new
        if (it & 31) == 0:
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[PSGD] it={it} best={best_score:.9f}")
        it += 1
    return best_a, best_score

"""
Propose a lower‐evaluation step function.

Algorithm
---------
1. Build a diverse pool of initial probability vectors (window functions,
   random families, the previous best if any).
2. Phase 1 – quick L‑BFGS cascade on a log‑sum‑exp surrogate with a
   decreasing λ‑schedule (few hundred iterations per λ).  The best
   candidate is kept.
3. Phase 2 – long exponentiated‑gradient (mirror‑SGD) descent on the
   true max‑convolution using an adaptive √‑decay step‑size and occasional
   Gaussian jitter to escape plateaus.
4. Phase 3 – low‑λ L‑BFGS polish (more iterations, finer λ‑schedule).
5. Phase 4 – projected sub‑gradient descent on the exact max‑convolution.
   We use a simple momentum scheme, a 1/√t step‑size, and project onto the
   probability simplex after each update.
6. Phase 5 – cheap random “mass‐transfer” hill‑climb (moving a tiny
   fraction of mass from a random source index to a random destination).
7. Final evaluation with the official scorer.  The best sequence found
   (as a list of floats) is stored in the global variable
   ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score (sum of coefficients is assumed 1):
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w with w_i >= 0 and sum w_i = 1.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy) and renorm."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# 1) Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(
                f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# 2) Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
    max_iters: int = 20_000_000,
):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    last_improve_it = 0

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        # occasional evaluation
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter to escape plateaus
        if (it - last_improve_it) > 200_000:
            a = add_noise(
                a,
                sigma=rng.uniform(5e-4, 2e-3),
                rng=rng,
            )
            last_improve_it = it

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 3) Projected sub‑gradient descent (additive updates)
# ----------------------------------------------------------------------


def projected_subgradient_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 0.1,
    beta: float = 0.9,
    max_iters: int = 5_000_000,
):
    """
    Additive projected sub‑gradient descent on the exact max‑convolution.
    Uses a simple momentum term and projects onto the probability simplex each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    v = np.zeros_like(a)
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of active indices
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

        # step‑size
        eta = eta0 / np.sqrt(it + 1.0)

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # additive step and simplex projection
        a = simplex_projection(a - eta * v)

        # evaluation every 64 steps
        if (it & 63) == 0:
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve_it = it
                print(f"[PSGD] it={it} best fast_score={best_score:.9f}")

        # jitter if stuck for a while
        if (it - last_improve_it) > 150_000:
            a = add_noise(
                a,
                sigma=rng.uniform(2e-4, 1e-3),
                rng=rng,
            )
            last_improve_it = it

        it += 1

    return best_a, best_score

# 4) Tiny random mass‑transfer hill‑climb (fast score)
# ----------------------------------------------------------------------


def random_move_hillclimb(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
    max_frac: float = 0.05,
):
    """
    Very cheap local search: move a random fraction of mass from a source
    index to a destination index, accept only if the fast score improves.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst or a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        if new_score < best_score:
            best_score = new_score
            best_a = cand.copy()
            a = cand
            cur_score = new_score
            print(f"[MoveHill] new best fast_score={best_score:.9f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Start from the previous best (if it exists) – interpolated to current n.
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)

            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    try:

        w = blackman(n)

    # ------------------------------------------------------------------
