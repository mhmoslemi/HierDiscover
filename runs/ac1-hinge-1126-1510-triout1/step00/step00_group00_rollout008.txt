    # The evaluation function works with any scaling, therefore we keep the sum ≈1.
    return best_seq.tolist()

"""
Heuristic stochastic local search for a step‐function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on the full length of the reference construction
(`height_sequence_1` if it exists, otherwise 6402).  It starts from that
construction (or a random Dirichlet vector) and repeatedly moves a small
amount of mass from one index to another.  Moves are biased towards indices
that currently contribute to the maximal convolution value, and a simple
Metropolis acceptance rule (simulated annealing) allows occasional uphill
steps to escape local minima.  The best sequence seen before the time budget
expires is returned as a plain list of floats.
"""

import numpy as np
import random
import time
import math

def _score(seq: np.ndarray) -> float:
    """
    Compute the evaluation metric exactly as ``evaluate_sequence`` does,
    but without the defensive checks (they are unnecessary for the
    generated non‑negative, finite values).
    """
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (total * total))


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Stochastic local search for a low‑value height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.  The function will try to return a sequence
        before the budget is exhausted.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Determine the problem size and initial sequence.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # use the published construction length
    except NameError:
        n = 6402                               # fallback length used in the description

    # start from the known construction with a 70 % probability, otherwise random.
    if 'height_sequence_1' in globals() and random.random() < 0.7:
        cur = np.array([max(0.0, float(v)) for v in height_sequence_1],
                       dtype=np.float64)
        if cur.sum() == 0.0:
            cur = np.random.rand(n)
    else:
        cur = np.random.rand(n)

    # normalise to unit sum (the metric is scale‑invariant)
    cur = cur / cur.sum()

    cur_score = _score(cur)
    best_seq = cur.copy()
    best_score = cur_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # keep a small safety margin

    # --------------------------------------------------------------
    # 2. Simulated‑annealing parameters.
    # --------------------------------------------------------------
    T0 = 0.1                                   # initial temperature
    decay = 0.99995                           # exponential decay per iteration
    temperature = T0
    iteration = 0
    print(f"[init] n={n}, start_score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        temperature *= decay

        # ----- locate the index giving the current maximal convolution -----
        conv_full = np.convolve(cur, cur)
        k_max = int(np.argmax(conv_full))

        # ----- indices that actually participate in that maximal term -----
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n) & (cur > 1e-14) & (cur[j_vals] > 1e-14)
        contributors = np.where(mask)[0]

        # ----- propose a move -------------------------------------------------
        if contributors.size > 0 and random.random() < 0.7:

            if j >= i:
                j += 1                     # guarantee i != j
        else:

            i, j = random.sample(range(n), 2)

        # Amount to transfer – a fraction of the source entry scaled by temperature.

        if source_mass <= 0.0:

    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
