def project_onto_simplex(v, s=1.0):
    v = np.asarray(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_optimize(n, max_iters, deadline):
    import numpy as np, time
    # initial a uniform
    a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # pre compute FFT length
    n_fft = 1 << ((2*n-1).bit_length())
    step0 = 0.4  # can tune
    eval_every = 10
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution
        fft_a = np.fft.rfft(a, n_fft)
        conv = np.fft.irfft(fft_a * fft_a, n_fft)[:2*n-1]
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # gradient
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        i_range = np.arange(lo, hi+1)
        j_range = max_idx - i_range
        grad[i_range] = a[j_range]
        grad *= 2.0
        # step size
        step = step0 / np.sqrt(it+1)
        a = a - step * grad
        # project onto simplex
        a = project_onto_simplex(a)
        if (it % eval_every) == 0:
            # evaluate score
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
        # occasional random perturb: with small prob
        if np.random.rand() < 0.001:
            # add small random noise and reproject
            a = a + np.random.randn(n) * 0.01
            a = np.maximum(a, 0)
            a = project_onto_simplex(a)
    return best_a, best_score

def local_search(a, deadline, time_per=5.0):
    import numpy as np, time
    a = a.copy()
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    n = len(a)
    while time.time() < deadline - 0.01:
        # propose random swap of epsilon
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        eps = np.random.exponential(scale=0.01)  # small positive
        if a[i] < eps:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # ensure nonnegative
        # Projection onto simplex not needed because sum remains same and all nonneg.
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
        else:
            # accept with small probability for annealing?
            if np.random.rand() < 0.01:
                a = a_new
    return best, best_score

def solve_convex_opt(n, max_seconds):
    import numpy as np, cvxpy as cp, time
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]
    # Build quadratic constraints
    Q_list = []
    for k in range(2*n-1):
        # create Q matrix in sparse format maybe
        Q = np.zeros((n,n))
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        for i in range(lo, hi+1):
            j = k - i
            Q[i, j] = 1.0
        Q_list.append(Q)
        constraints.append(cp.quad_form(a, Q) <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iters=int(max_seconds*2000), verbose=False)
    # Retrieve solution
    if a.value is not None:
        a_opt = np.maximum(a.value, 0)  # clip negative due to numeric issues
        a_opt = a_opt / np.sum(a_opt)  # ensure sum=1
        # evaluate score
        score = evaluate_sequence(a_opt.tolist())
        return a_opt, score
    else:
        return None, None

    step = 0.02
    while time.time() < deadline - 0.1:

"""Search for a step function with low evaluation value.

Uses a soft‑max based projected gradient descent on the
max of the auto‑convolution, combined with random restarts
and a simple hill‑climbing refinement.
"""

import numpy as np
import time
import math

def project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def softmax_weights(x, gamma):
    """Return soft‑max weights for vector x with temperature 1/gamma."""
    # shift for numerical stability
    shift = x - np.max(x)
    exp_gamma = np.exp(gamma * shift)
    w = exp_gamma / np.sum(exp_gamma)
    return w

def gradient_step(a, gamma, lr):
    """One projected gradient descent step on the soft‑max objective."""
    # convolution of a with itself
    conv = np.convolve(a, a)
    # soft‑max weights on the convolution values
    w = softmax_weights(conv, gamma)
    # gradient = 2 * (a_rev * w)_{centered}
    rev = a[::-1]
    # full convolution (rev length n, w length 2n-1)
    grad_full = np.convolve(rev, w)
    # extract n entries that correspond to i+j indexing
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    # gradient descent update
    a_new = a - lr * grad
    # project back onto simplex
    a_new = project_onto_simplex(a_new)
    return a_new, conv

def hill_climb(a, deadline, max_no_improve=2000):
    """Simple random edge‑swap hill climbing that respects the simplex."""
    n = len(a)
    best = a.copy()
    best_score = 2 * n * np.max(np.convolve(best, best))
    no_improve = 0

"""
Search for a step function (list of non‑negative heights) that minimises
`evaluate_sequence`.  The algorithm runs many random restarts of a projected
gradient descent that flattens the auto‑convolution using a soft‑max surrogate
for the max‑operator, followed by a short hill‑climbing phase.

The search respects the simplex constraint (non‑negative entries with unit sum);
because `evaluate_sequence` is homogeneous (scaling the vector does not change
the score) this is sufficient – the returned vector may have any total sum.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def project_onto_simplex(v, s=1.0):
    """
    Euclidean projection of a vector onto the simplex {x >= 0, sum(x) = s}.
    Returns a new array with the same shape as v.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]

    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Soft‑max weights for a given temperature (parameter gamma).
# ----------------------------------------------------------------------
def softmax_weights(x, gamma):
    """
    Compute soft‑max weights w_k = exp(gamma * x_k) / sum_k exp(gamma * x_k).
    The input x is a 1‑D array.  For numerical stability the maximum of x
    is subtracted before exponentiation.
    """
    shift = x - np.max(x)          # avoid overflow
    exp_gamma = np.exp(gamma * shift)
    w = exp_gamma / np.sum(exp_gamma)
    return w

# ----------------------------------------------------------------------
# One gradient descent step (projected) using the soft‑max surrogate.
# ----------------------------------------------------------------------
def gradient_step(a, gamma, lr):
    """
    Perform a single projected gradient descent step on the smooth objective
    F(a) = (1/gamma) * log(sum_k exp(gamma * (a*a)[k])).
    Returns the updated vector and the latest (unsmoothed) convolution.
    """
    # auto‑convolution of a with itself
    conv = np.convolve(a, a)                 # length 2n‑1
    # soft‑max weights on the convolution values
    w = softmax_weights(conv, gamma)         # same length 2n‑1

    # gradient of F w.r.t. a:
    #   ∇F_i = 2 * sum_j a_j * w_{i+j}
    # which is a correlation of a (reversed) with w.
    rev = a[::-1]                            # reversed a
    full_grad = np.convolve(rev, w)          # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * full_grad[n - 1 : n - 1 + n]   # extract the n components we need

    # gradient step followed by projection onto the simplex
    a_new = a - lr * grad
    a_new = project_onto_simplex(a_new)
    return a_new, conv

# ----------------------------------------------------------------------
# Light hill‑climbing refinement (edge‑swap moves) that stays on the simplex.
# ----------------------------------------------------------------------
def hill_climb(a, deadline, max_no_improve=2000):
    """Randomly move a tiny amount of mass between two coordinates."""
    n = len(a)
    best = a.copy()
    best_score = 2 * n * np.max(np.convolve(best, best))
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        eps = 0.01 * np.random.rand()    # small transfer amount
        if best[i] < eps:
            continue
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = project_onto_simplex(cand)
        score = 2 * n * np.max(np.convolve(cand, cand))
        if score < best_score:
            best_score = score
            best = cand
            no_improve = 0
        else:
            no_improve += 1
    return best, best_score

# ----------------------------------------------------------------------
# Core optimisation for a fixed length n.
# ----------------------------------------------------------------------
def optimise_length(n, max_iters, deadline):
    """
    Optimise a vector of length n.  The vector is kept on the simplex
    (sum == 1) so that the final `evaluate_sequence` score is simply
    2 * n * max(conv).  The routine returns the best vector found and its
    score.
    """
    # start from a random Dirichlet point (guaranteed sum == 1)
    a = np.random.dirichlet(np.ones(n))
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))

    # hyper‑parameters for the schedule
    lr0 = 0.5          # initial learning rate
    gamma0 = 1.0
    gamma_max = 20.0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # learning‑rate and temperature schedules
        lr = lr0 / math.sqrt(it + 1)
        gamma = gamma0 + (gamma_max - gamma0) * (it / max_iters)

        a, conv = gradient_step(a, gamma, lr)

        # occasional evaluation of the true objective
        if it % 20 == 0:
            cur_score = 2 * n * np.max(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final local refinement
    refined_a, refined_score = hill_climb(best_a, deadline)
    if refined_score < best_score:
        return refined_a, refined_score
    return best_a, best_score

    """

    """

        best_seq = np.full(128, 1.0 / 128)
        best_score = 2 * 128 * np.max(np.convolve(best_seq, best_seq))

    result = best_seq.tolist()
    print(f"Returning best sequence (length={len(result)}), final score={best_score:.6f}")
    return result
