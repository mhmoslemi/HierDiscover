def fast_convolve(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2 >= n
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:n]
    return conv

def cross_correlation(a, w):
    n = len(a) + len(w) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)

"""
Propose a better step‑function sequence by directly minimising a smooth
approximation of the maximum self‑convolution.
The algorithm works on a probability vector `a` (sum = 1, a_i≥0) and
optimises the objective

    J(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

which approaches the true maximum `max_k (a*a)_k` as β→∞.
The gradient of J is computed via FFT‑based cross‑correlation,
a step is taken opposite to the gradient, the vector is projected back
onto the probability simplex and the process is repeated with slowly
increasing β and decreasing step size.

Multiple random restarts with different lengths are performed until the
time budget expires.  The best sequence (as evaluated by the provided
`evaluate_sequence`) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT based convolution / correlation (real valued, uses rfft)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (returns length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    return np.fft.irfft(X * Y, n=N)[:n]


def fft_cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[k] = Σ_j a_j·w_{j+k}.
    Returns the full linear correlation (length len(a)+len(w)-1).
    """
    n = len(a) + len(w) - 1
    N = _next_pow2(n)
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    return np.fft.irfft(A * np.conj(W), n=N)[:n]


# ----------------------------------------------------------------------
# Gradient descent step on the smooth max objective
# ----------------------------------------------------------------------
def _gd_step(a: np.ndarray, beta: float, step: float) -> np.ndarray:
    """
    Perform one gradient‑descent step on J(a) described in the module docstring.
    The vector `a` is assumed to be non‑negative and sum to 1.
    """
    n = a.shape[0]

    # 1) self‑convolution a*a
    conv = fft_convolve(a, a)               # length 2n‑1
    # 2) soft‑max weights for a smooth max
    shift = conv.max()                       # for numerical stability
    w = np.exp(beta * (conv - shift))
    w /= w.sum()                             # w_k ≥0, Σ w_k = 1

    # 3) gradient = 2·(a ⋆ w)  (only first n components are needed)
    corr = fft_cross_corr(a, w)              # length 3n‑2, first n are needed
    grad = 2.0 * corr[:n]

    # 4) move opposite to the gradient (normalise direction)
    norm = np.linalg.norm(grad) + 1e-20
    direction = grad / norm
    a_new = a - step * direction

    # 5) projection onto the non‑negative simplex (clip & renormalise)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s < 1e-12:                     # re‑initialise if we lost all mass
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new

    """
