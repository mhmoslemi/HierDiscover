def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # start with uniform sequence length n0=200 (or maybe 100?).
    n = 200
    seq = np.ones(n) / n  # sum=1

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    temperature = 1.0
    while time.time() < deadline:
        # propose new seq by random mutation
        new_seq = seq.copy()
        # With probability 0.5 do local mass transfer
        if np.random.rand() < 0.5:
            i, j = np.random.choice(len(new_seq), 2, replace=False)
            # move delta from i to j
            delta = np.random.exponential(scale=0.1) * (np.mean(new_seq)+1e-6)
            delta = min(delta, new_seq[i])
            new_seq[i] -= delta
            new_seq[j] += delta
        else:
            # slight perturbation
            idx = np.random.randint(len(new_seq))
            new_seq[idx] += np.random.randn() * 0.01
            new_seq[idx] = max(0.0, new_seq[idx])

        # optional length change: with small prob, insert zeros at random position then adjust length up to max allowed
        if np.random.rand() < 0.01 and len(new_seq) < 2000:
            pos = np.random.randint(len(new_seq)+1)
            new_seq = np.insert(new_seq, pos, 0.0)

        # compute score
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # accept if better, else maybe accept worse according to temperature
        if score < best_score or np.random.rand() < np.exp((best_score - score) / (temperature+1e-9)):
            seq = new_seq
            if score < best_score:
                best_score = score
                best_seq = new_seq.copy()
                print(f"New best {best_score:.6f}")

        # cool down temperature gradually
        temperature *= 0.99995

    return best_seq.tolist()

"""
Propose candidate step‑function heights.

The evaluation metric is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

for a non‑negative vector `a`.  By construction the metric is invariant
under a global scaling of `a`; the only way to improve it is to keep the
supremum of the auto‑convolution small while increasing the ℓ¹‑norm.
The algorithm works as follows

1.  Initialise a random vector `a` (or re‑use the previous best if it is
    available).  The vector is normalised to have ℓ¹‑norm 1 – this fixes the
    scaling without affecting the score.

2.  Repeatedly try two complementary local moves:

    * **Targeted hill‑climbing** – find a shift `k` where the convolution
      attains its maximum and move a tiny amount of mass from a pair of
      indices that contribute most to this peak to a far away index.  This
      tends to flatten the convolution.

    * **LP‑guided boost** – solve a linear program that looks for a vector
      `g ≥ 0` with the *largest possible* sum under the constraint
      `conv(a, g) ≤ max(conv(a, a))`.  Adding a small fraction of `g` to `a`
      increases the ℓ¹‑norm without raising the current supremum, which
      reduces the score.

    Candidate vectors created by both moves are evaluated;
    improvements are kept.  The step size for the interpolation with `g`
    is chosen from a short list `[1.0, 0.5, 0.2, 0.1, 0.05, 0.01]` and the
    best one is applied.

3.  Occasionally (≈1 % of the iterations) the current vector is replaced by a
    fresh random one – this avoids stagnation in local minima.

The loop runs until the allocated time budget (≈ 990 s to leave a margin
for the final return) expires and the best sequence found so far is
returned as a plain Python list.  All heavy work is performed with NumPy
arrays; the linear programme is solved by SciPy’s high‑performance
“highs” interior‑point method.

No file‑IO or external network calls are performed.
"""

import time
import numpy as np
from scipy import optimize

def _conv_matrix(f):
    """Return the (2n‑1)×n matrix A where A[k, j] = f[k‑j] (zero if out of range)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        # f[i] contributes to all positions k = i + j
        A[i : i + n, :] += np.diagflat(np.full(n, f[i]), k=i)
        # More directly: fill row k = i + j with f[i] at column j
        # but the loop above is simple and still O(n²) for n ≤ 600.
    # The above double‑loop is acceptable for the modest n we use.
    # A more compact version could use Toeplitz, but clarity beats micro‑optimisation here.
    return A

def _solve_lp(f, max_conv):
    """
    Solve:
        maximise   sum(g)
        subject to conv(f, g) ≤ max_conv   (for every shift)
                   g ≥ 0
    Returns the optimal g as a NumPy array or None if the LP fails.
    """
    n = f.shape[0]
    # build convolution matrix (2n‑1) × n
    A = np.zeros((2 * n - 1, n), dtype=float)
    # each row k corresponds to Σ_i f[i] * g[k‑i]
    for i in range(n):
        A[i : i + n, i] = f  # broadcasting f over the appropriate slice

    c = -np.ones(n)               # maximise sum(g)  → minimise –sum(g)
    b = np.full(2 * n - 1, max_conv * (1.0 + 1e-12))   # tiny slack for numerical safety

    # Use bounds to enforce non‑negativity
    bounds = [(0.0, None)] * n

    try:
        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds and returns the best sequence discovered.
    """

    deadline = time.time() + budget_s - 5   # a small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    # If a previous construction is present, start there with 50 % probability.
    try:

        init_seq = list(height_sequence_1)   # may be huge – we truncate for speed
        use_prev = np.random.rand() < 0.5
    except Exception:
        init_seq = None
        use_prev = False

    if use_prev and init_seq is not None and len(init_seq) <= 800:
        seq = np.array(init_seq, dtype=float)
    else:
        # A moderate size random start (ℓ¹‑norm 1)
        n0 = np.random.randint(150, 251)          # length between 150 and 250
        seq = np.random.random(n0)
        seq /= seq.sum()                           # normalise ℓ¹‑norm to 1

    # Make sure the vector respects the evaluate‑function’s clipping rules.
    seq = np.clip(seq, 0.0, 1000.0)

    best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 2️⃣  Main optimisation loop -----------------------------------------
    # ------------------------------------------------------------------
    # Step‑size candidates for the interpolation with the LP‑direction.
    t_candidates = [1.0, 0.5, 0.2, 0.1, 0.05, 0.01]

    # Simple temperature for occasional downhill moves (simulated annealing flavour).
    temperature = 1.0

    while time.time() < deadline:
        n = seq.shape[0]

        # ------------------------------------------------------------------
        # 2a – compute current convolution and its maximum
        # ------------------------------------------------------------------
        conv = np.convolve(seq, seq)
        max_conv = conv.max()
        max_k = conv.argmax()

        # ------------------------------------------------------------------
        # 2b – try a targeted hill‑climbing move that reduces the peak
        # ------------------------------------------------------------------
        # Identify a pair (i, j) that contributes to the maximal entry.
        # j = max_k - i  (if inside the array)
        i_candidates = np.where(seq > 0)[0]
        if len(i_candidates) > 0:
            i = np.random.choice(i_candidates)
            j = max_k - i
            if 0 <= j < n and seq[j] > 0:
                # move a tiny amount ε from the larger of the two to a random slot
                eps = 0.01 * min(seq[i], seq[j]) * np.random.rand()
                if seq[i] >= seq[j]:
                    src, dst = i, np.random.randint(n)
                else:
                    src, dst = j, np.random.randint(n)
                new_seq = seq.copy()
                if new_seq[src] >= eps:
                    new_seq[src] -= eps
                    new_seq[dst] += eps
                else:
                    new_seq = seq.copy()   # fallback to a pure random move
                new_seq = np.clip(new_seq, 0.0, 1000.0)
                cand_score = evaluate_sequence(new_seq.tolist())
                # accept if better or via Metropolis criterion
                if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / (temperature + 1e-12)):
                    seq = new_seq
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = new_seq.copy()
                        print(f"[hill] new best {best_score:.6f} (len={len(seq)})")
        # ------------------------------------------------------------------
        # 2c – LP‑guided move (once every few iterations)
        # ------------------------------------------------------------------
        if np.random.rand() < 0.2:   # 20 % of the loop iterations
            g = _solve_lp(seq, max_conv)
            if g is not None:
                # Normalise g to the same ℓ¹‑norm as seq (the LP already respects the bound)
                # but we keep its raw magnitude – the interpolation will control the exact
                # contribution.
                for t in t_candidates:
                    cand = (1.0 - t) * seq + t * g
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand_score = evaluate_sequence(cand.tolist())
                    if cand_score < best_score:
                        seq = cand
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[lp]  t={t:.2f} improved score to {best_score:.6f}")
                        break   # stop trying larger t's once we found improvement

        # ------------------------------------------------------------------
        # 2d – occasional random restart to avoid dead‑ends
        # ------------------------------------------------------------------
        if np.random.rand() < 0.01:
            n_restart = np.random.randint(150, 251)
            seq = np.random.random(n_restart)
            seq /= seq.sum()
            seq = np.clip(seq, 0.0, 1000.0)
            print(f"[restart] new random length={n_restart}")

        # ------------------------------------------------------------------
        # 2e – temperature cooling
        # ------------------------------------------------------------------
        temperature *= 0.9999

    # ------------------------------------------------------------------
    # 3️⃣  Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} length {len(best_seq)}")
    return best_seq.tolist()
