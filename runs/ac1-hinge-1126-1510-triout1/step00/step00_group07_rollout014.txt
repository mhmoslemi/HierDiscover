def propose_candidate(seed=42, budget_s=1000, **kwargs):
     np.random.seed(seed)
     deadline = time.time() + budget_s - 10
     best_seq = None
     best_score = float('inf')
     # try several lengths

          if time.time() > deadline: break
          # compute optimal r
          if n == 1:
               seq = [1.0]

               r = math.exp(-math.log(n) / (n - 1))
               # avoid r=1 which yields uniform; for n>1 it's less than 1.
               # compute C
               r_n = r**n
               C = (1 - r) / (1 - r_n)
               # generate sequence
               seq = [C * (r**i) for i in range(n)]
          # evaluate
          score = evaluate_sequence(seq)
          if score < best_score:
               best_score = score
               best_seq = seq
               print(f"New best: n={n}, score={score:.6f}")
     # optionally try random perturbations around best
     while time.time() < deadline:
          # small random perturbation
          seq = [max(0.0, x + np.random.normal(scale=0.01)) for x in best_seq]
          # renormalize to sum = 1 (or any positive)
          total = sum(seq)
          if total < 1e-6:
               continue
          seq = [x / total for x in seq]
          score = evaluate_sequence(seq)
          if score < best_score:
               best_score = score
               best_seq = seq
               print(f"Improved by perturbation: score={score:.6f}")
     return best_seq

"""
Propose a low‑score step‑function using a fast parametric search
followed by a simple local hill‑climb.  The objective is
    score = 2·n·max_k (a * a)[k] / (sum a)^2 .
Since the score is homogeneous, we fix sum(a)=1 and minimise
    max_k (a * a)[k] .
A family a_i = exp(-c·(i/(n‑1))**p) (normalised) contains many
different shapes – from almost uniform (p≈1,c≈0) to steeply
decreasing (p<1 or p>1).  By sweeping (c,p) we quickly discover a
distribution with max‑convolution well below the uniform value
(score≈2).  A cheap hill‑climbing stage then fine‑tunes the best
candidate found.

The routine respects the time budget (default 1000 s) and returns
the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# evaluate_sequence is assumed to be already imported in the judge environment.


def _exp_shape(n: int, c: float, p: float) -> list[float]:
    """Return a normalised sequence a_i = exp(-c * (i/(n-1))**p)."""
    if n == 1:
        return [1.0]
    # use float array for speed
    idx = np.arange(n, dtype=float)
    # avoid division by zero when n==1 (handled above)
    x = (idx / (n - 1)) ** p
    a = np.exp(-c * x)
    total = a.sum()
    # normalise to sum‑1 (the evaluation function is scale‑invariant)
    a /= total
    return a.tolist()


def _pow_exp_shape(n: int, alpha: float, beta: float) -> list[float]:
    """Shape a_i = (i+1)**alpha * exp(-beta*i)  (normalised)."""
    if n == 1:
        return [1.0]
    i = np.arange(n, dtype=float)
    a = (i + 1.0) ** alpha * np.exp(-beta * i)
    total = a.sum()
    a /= total
    return a.tolist()


def _local_optimize(start_seq: list[float],
                    start_score: float,
                    deadline: float) -> tuple[list[float], float]:
    """
    Very light hill‑climbing: repeatedly move a small amount of mass
    from one index to another, keeping the sum = 1.
    Accept only improving moves (occasional random jump can be added).
    """
    n = len(start_seq)
    curr_seq = start_seq[:]
    curr_score = start_score
    best_seq = start_seq[:]
    best_score = start_score
    it = 0

    while time.time() < deadline:
        # choose two distinct indices
        i, j = random.sample(range(n), 2)

        # move at most 20 % of the source entry
        if curr_seq[i] == 0.0:
            continue
        delta = curr_seq[i] * (random.random() * 0.2)

        # propose new sequence
        new_seq = curr_seq[:]
        new_seq[i] -= delta
        new_seq[j] += delta

        # renormalise (tiny numerical drift)
        total = sum(new_seq)
        if total == 0.0:
            continue
        new_seq = [x / total for x in new_seq]

        new_score = evaluate_sequence(new_seq)
        it += 1
        # always keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq[:]
            # also adopt it as the current point
            curr_seq = new_seq[:]
            curr_score = new_score
            print(f"  [local] iter={it}  new best={best_score:.6f}")
        # otherwise accept only if it improves the current point
        elif new_score < curr_score:
            curr_seq = new_seq[:]
            curr_score = new_score

    return best_seq, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The search consists of:
      1. a coarse grid / random sweep over the parametrised family
         a_i = exp(-c·(i/(n‑1))**p) for several lengths,
      2. a second family a_i = (i+1)^α·exp(-β·i),
      3. a brief local hill‑climb from the best candidate found.
    The routine stops just before the allocated budget and returns
    the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 2.0   # a tiny safety margin
    best_seq: list[float] | None = None
    best_score = float('inf')

    # Different lengths give different granularity – try a few.
    lengths = [200, 400, 800, 1823]

    # ------------------------------------------------------------------
    # 1) Exponential‑power family
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        # systematic grid (coarse)
        for p in np.linspace(0.2, 2.0, 9):          # 9 values
            for c in np.linspace(0.5, 6.0, 12):      # 12 values
                if time.time() > deadline:
                    break
                seq = _exp_shape(n, c, p)
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"[grid] n={n:4d} p={p:4.2f} c={c:4.2f} → score={score:.6f}")

        # random sampling in the same family
        rmsamples = 2000 if n <= 400 else 1200
        for _ in range(rmsamples):
            if time.time() > deadline:
                break
            p = random.uniform(0.1, 2.5)
            c = random.uniform(0.2, 8.0)
            seq = _exp_shape(n, c, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[rand] n={n:4d} p={p:4.2f} c={c:4.2f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Power‑exponential family (adds extra shape freedom)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        # grid over (alpha,beta)
        alphas = np.linspace(-1.0, 2.0, 7)   # may be negative → larger early terms
        betas = np.linspace(0.0, 2.5, 8)
        for alpha in alphas:
            for beta in betas:
                if time.time() > deadline:
                    break
                seq = _pow_exp_shape(n, alpha, beta)
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"[grid2] n={n:4d} α={alpha:4.2f} β={beta:4.2f} → score={score:.6f}")

        # random samples in this family
        rmsamples = 1500 if n <= 400 else 800
        for _ in range(rmsamples):
            if time.time() > deadline:
                break
            alpha = random.uniform(-1.5, 2.5)
            beta = random.uniform(0.0, 3.0)
            seq = _pow_exp_shape(n, alpha, beta)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[rand2] n={n:4d} α={alpha:4.2f} β={beta:4.2f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final local refinement (if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # ensure we still have time for a brief local optimisation
        if time.time() + 10.0 < deadline:
            print(f"[refine] starting local optimisation, current best={best_score:.6f}")
            best_seq, best_score = _local_optimize(best_seq, best_score, deadline)

    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        n = 200
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    print(f"[final] best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
