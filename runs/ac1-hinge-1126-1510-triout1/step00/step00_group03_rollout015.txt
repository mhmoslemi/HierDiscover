"""search algorithm for step function heights.

We start from a continuous–fraction power‑law shape a_i≈(i+1)^‑½
(which is known to give the asymptotic bound π/2≈1.57 once the
sequence is normalized).  From this seed we repeatedly solve the linear
program used in the published state‑of‑the‑art (maximise the sum while
keeping the cross‑convolution bounded) and then perform a line search
over the mixing parameter.  Small random perturbations are added from
time to time to escape plateaus.  The algorithm stops after the given
budget and returns the best sequence found.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _build_convolution_ub_matrix(f):
    """Return A_ub (2n‑1)×n for the constraints  conv(f,b) ≤ rhs."""
    n = len(f)
    rows = np.zeros((2 * n - 1, n), dtype=float)
    # For each output index k, fill the coefficients of b.
    # conv_k = sum_i f_i * b_{k-i}
    for k in range(2 * n - 1):
        # i runs over indices where 0 ≤ i < n and 0 ≤ k‑i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vectorised fill
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        rows[k, j_vals] = f[i_vals]
    return rows

def solve_convolution_lp(f_sequence, rhs):
    """Linear program:
       maximise sum(b)  s.t. conv(f,b) ≤ rhs, b ≥ 0.
       Returns the optimal b as a numpy array, or None if infeasible.
    """
    n = len(f_sequence)
    # objective: minimise –sum(b)  (linprog minimises)
    c = -np.ones(n)

    # upper‑bound constraints for each convolution entry
    a_ub = _build_convolution_ub_matrix(f_sequence)
    b_ub = np.full(a_ub.shape[0], rhs, dtype=float)

    # non‑negativity is given by bounds, but we also add explicit rows
    # for completeness (some SciPy versions need them).
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # solve – use the HiGHS interior‑point solver (fast for dense problems)
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"time_limit": 5.0, "presolve": True},
    )
    if res.success:
        # numerical noise may produce tiny negatives – clip
        return np.maximum(res.x, 0.0)
    return None

def normalize_to_target(seq, target):
    """Scale seq so that its L1‑norm equals target (target > 0)."""
    s = np.sum(seq)
    if s == 0:
        return seq.copy()
    return seq * (target / s)

def line_search_mix(f, g, n, best_score):
    """Try several α∈[0,1] for the convex combination (1‑α)f+αg.
       Returns (new_seq, new_score) if an improvement is found,
       otherwise (None, None)."""
    # f and g are already normalised (sum = sqrt(2n))
    alphas = np.linspace(0.0, 1.0, 11)
    improved_seq = None
    improved_score = None
    for alpha in alphas:
        cand = (1.0 - alpha) * f + alpha * g
        # tiny numerical errors may drift the sum – re‑normalise
        cand = normalize_to_target(cand, np.sqrt(2 * n))
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            improved_seq = cand
            improved_score = score
            # keep the best α – continue searching in case a later α is even better
    return improved_seq, improved_score

def random_local_perturb(seq, magnitude=0.02):
    """Add a small random perturbation (Gaussian) and re‑normalise."""
    n = len(seq)
    # pick a subset of coordinates to perturb (10 % of them)
    idx = np.random.choice(n, size=max(1, n // 10), replace=False)
    noise = np.random.randn(len(idx)) * magnitude * np.mean(seq)
    cand = seq.copy()
    cand[idx] += noise
    cand = np.maximum(cand, 0.0)
    return normalize_to_target(cand, np.sqrt(2 * n))

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a list of non‑negative heights that (approximately) minimise
    the evaluation function.  The algorithm tries a power‑law seed,
    then performs a few LP‑based mixing steps with line‑search and occasional
    random perturbations.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # a tiny safety margin

    # ------------------------------------------------------------------
    # 1.  Build an initial family of candidates (power‑law shapes)
    # ------------------------------------------------------------------
    # Choose a moderate length – the objective is scale‑invariant,
    # a length of a few thousand gives a good trade‑off between quality
    # and speed.
    n = 2500
    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    print("---- initial grid search (power‑law) ----")
    for p in np.linspace(0.30, 0.70, 41):
        seq = (np.arange(1, n + 1, dtype=float) ** (-p))
        seq = normalize_to_target(seq, target_sum)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"p={p:.3f} → score={score:.6f}")

"""search algorithm for step‐function heights.

We start from a power‑law seed a_i ≈ (i+1)^‑½ (the known asymptotic
shape giving the constant π/2≈1.57 once the sequence is normalized).  
From this seed we repeatedly

1. solve the linear programme used in the published SOTA (maximise the
   ℓ₁‑sum while keeping the cross‑convolution bounded),
2. run a line‑search over the mixing coefficient α in the convex
   combination (1‑α)·f + α·g,
3. occasionally apply a small random perturbation to escape plateaus.

The routine stops after the allocated time budget and returns the best
sequence found (as a plain list of floats).  All numbers are non‑negative
and the sequence is automatically normalised to the required ℓ₁‑norm,
so the evaluation function reduces to the maximum convolution value.
"""

import numpy as np
import time
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _build_convolution_ub_matrix(f):
    """Build the matrix A_ub for the constraints  conv(f,b) ≤ rhs.

    Each row k (0 ≤ k < 2·n‑1) corresponds to the inequality
        sum_i f_i * b_{k‑i} ≤ rhs,
    where out‑of‑range indices are ignored.
    """
    n = len(f)
    rows = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        rows[k, j_vals] = f[i_vals]
    return rows


def solve_convolution_lp(f_sequence, rhs):
    """Linear programme:

          maximise   Σ b_j
          subject to conv(f,b) ≤ rhs   (elementwise)
                     b_j ≥ 0

    Returns the optimal b as a NumPy array, or None if the problem is
    infeasible or the solver times out.
    """
    n = len(f_sequence)
    # linprog minimises; we maximise Σ b_j by minimising –Σ b_j
    c = -np.ones(n)

    a_ub = _build_convolution_ub_matrix(f_sequence)
    b_ub = np.full(a_ub.shape[0], rhs, dtype=float)

    # explicit non‑negativity rows (helpful for the HiGHS solver)
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # solve – HiGHS is fast for dense problems
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"time_limit": 5.0, "presolve": True},
    )
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def normalize_to_target(seq, target):
    """Scale seq so that its ℓ₁‑norm equals *target* (target > 0)."""
    s = np.sum(seq)
    if s == 0:
        return seq.copy()
    return seq * (target / s)


def line_search_mix(f, g, n, current_best):
    """Try several α∈[0,1] for (1‑α)·f + α·g.
    Returns (new_seq, new_score) if an improvement is found, else (None, None)."""
    alphas = np.linspace(0.0, 1.0, 11)      # 0, 0.1, …, 1.0
    best_seq = None
    best_score = current_best
    for a in alphas:
        cand = (1.0 - a) * f + a * g
        cand = normalize_to_target(cand, np.sqrt(2.0 * n))
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
    return best_seq, best_score


def random_local_perturb(seq, magnitude=0.02):
    """Gaussian noise on a random 10 % subset, followed by re‑normalisation."""
    n = len(seq)
    idx = np.random.choice(n, size=max(1, n // 10), replace=False)
    noise = np.random.randn(len(idx)) * magnitude * np.mean(seq)
    cand = seq.copy()
    cand[idx] += noise
    cand = np.maximum(cand, 0.0)
    return normalize_to_target(cand, np.sqrt(2.0 * n))
