import numpy as np
import time

def generate_uniform(n):
    """Uniform distribution (all ones)."""
    return np.ones(n).tolist()

def generate_arcsine(n):
    """Discrete arcsine distribution: weights ~ 1/sqrt(i*(n+1-i)). Normalized to sum 1."""
    i = np.arange(1, n+1)
    w = 1.0/np.sqrt(i * (n + 1 - i))
    w /= w.sum()
    return w.tolist()

def generate_sin(n):
    """Discrete sin distribution: weights ~ sin(pi*i/(n+1)). Normalized."""
    i = np.arange(1, n+1)
    w = np.sin(np.pi * i / (n + 1))
    w /= w.sum()
    return w.tolist()

def generate_cosine_squared(n):
    """Discrete cosine squared distribution: weights ~ sin^2(pi*i/(n+1)). Normalized."""
    i = np.arange(1, n+1)
    w = np.sin(np.pi * i / (n + 1))**2
    w /= w.sum()
    return w.tolist()

def generate_triangular(n):
    """Symmetric triangular distribution rising then falling."""
    half = (n+1)//2
    if n % 2 == 0:
        # even length: increasing then decreasing
        inc = np.arange(1, half+1)
        dec = np.arange(half, 0, -1)
        w = np.concatenate((inc, dec))
    else:
        inc = np.arange(1, half+1)
        dec = np.arange(half, 0, -1)
        w = np.concatenate((inc, dec))
    w = w.astype(float)
    w /= w.sum()
    return w.tolist()

"""
propose_candidate – search for a step‑function that minimizes the
`evaluate_sequence` score.

The self‑convolution of a probability vector is lowest when the
distribution is the discrete arcsine law
    f_i ∝ 1 / √(i·(n+1‑i)) .
Its convolution is almost uniform, giving an objective close to the
theoretical optimum (≈ 1).  The routine builds a few candidate shapes
(arcsine, sine, cosine‑squared, triangular, uniform) for several
lengths, evaluates them, keeps the best and then performs a tiny
stochastic hill‑climb (simulated‑annealing) from the current best
sequence.  All work stops before the allocated time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for generating candidate sequences
# ----------------------------------------------------------------------


def generate_uniform(n):
    """Uniform distribution – all entries equal."""
    return np.ones(n).tolist()


def generate_arcsine(n):
    """Discrete arcsine law:  w_i ∝ 1/√(i·(n+1‑i)),  i=1..n.  Normalized to sum 1."""
    i = np.arange(1, n + 1)
    w = 1.0 / np.sqrt(i * (n + 1 - i))
    w /= w.sum()
    return w.tolist()


def generate_sin(n):
    """Sinusoidal shape:  w_i ∝ sin(π·i/(n+1))."""
    i = np.arange(1, n + 1)
    w = np.sin(np.pi * i / (n + 1))
    w /= w.sum()
    return w.tolist()


def generate_cosine_squared(n):
    """Cosine‑squared shape:  w_i ∝ sin²(π·i/(n+1))."""
    i = np.arange(1, n + 1)
    w = np.sin(np.pi * i / (n + 1)) ** 2
    w /= w.sum()
    return w.tolist()


def generate_triangular(n):
    """Symmetric triangular shape: w_i = min(i+1, n‑i)."""
    idx = np.arange(n)
    w = np.minimum(idx + 1, n - idx).astype(float)
    w /= w.sum()
    return w.tolist()


def evaluate(seq):
    """Safely compute `evaluate_sequence` – return ∞ on error."""
    try:
        return evaluate_sequence(seq)
    except Exception as e:
        print("Evaluation error:", e)
        return float("inf")


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that achieves
    a low `evaluate_sequence` score.  The algorithm consists of:

    1.  Trying a handful of analytically motivated shapes (arcsine,
        sine, cosine‑squared, triangular, uniform) for several
        lengths.
    2.  Keeping the best among them.
    3.  Running a lightweight simulated‑annealing hill‑climb from the
        current best sequence until the time budget runs out.

    The returned list is guaranteed to contain only non‑negative numbers.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # leave a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Use the previously discovered construction (if available)
    # ------------------------------------------------------------------
    try:
        init_seq = list(height_sequence_1)  # defined in the global scope
        init_score = evaluate(init_seq)
        print(f"Initial height_sequence_1 score = {init_score:.6f}")
        if init_score < best_score:
            best_score = init_score
            best_seq = init_seq
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1)  Exhaustively test a few analytic families for several lengths
    # ------------------------------------------------------------------
    lengths = [80, 120, 200, 400, 800, 1200, 1542, 2000]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"--- Testing length n={n} ---")

        # Build a list of candidate sequences for this length
        candidates = [
            generate_uniform(n),
            generate_arcsine(n),
            generate_sin(n),
            generate_cosine_squared(n),
            generate_triangular(n),
        ]

        # Add a few noisy perturbations of the arcsine shape
        base = np.array(generate_arcsine(n))
        for _ in range(3):
            pert = base.copy()
            noise = np.random.randn(n) * 0.02
            pert = np.maximum(pert + noise, 0.0)
            if pert.sum() > 0:
                pert /= pert.sum()
            candidates.append(pert.tolist())

        # Evaluate all candidates
        for seq in candidates:
            if time.time() > deadline:
                break
            sc = evaluate(seq)
            if np.isfinite(sc):
                print(f"len={len(seq):4d}  score={sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"*** New best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Optional stochastic hill‑climb from the current best point
    # ------------------------------------------------------------------
    if best_seq is not None:

        n = len(current)

            # Occasionally reset to a fresh arcsine vector (helps escape stagnation)
            if iteration % 5000 == 0:
                current = np.array(generate_arcsine(n))
                current /= current.sum()
                temperature = 0.1
                step = 0.02

    # ------------------------------------------------------------------
    # 3)  Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        print("No valid sequence found – returning a trivial uniform vector.")
        best_seq = generate_uniform(1)

    print(f"Returning best sequence (len={len(best_seq)}) with score {best_score:.6f}")
    return best_seq
