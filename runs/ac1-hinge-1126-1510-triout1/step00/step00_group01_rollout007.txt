def compute_convolution(seq):
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n - 1]
    return conv
```

def gradient_step(seq, lr):
    n = len(seq)
    sum_val = seq.sum()
    if sum_val == 0:
        return seq
    conv = compute_convolution(seq)
    max_val = conv.max()
    active = np.where(np.abs(conv - max_val) < 1e-12)[0]
    # Compute dM_avg_i
    dM = np.zeros_like(seq)
    for k in active:
        # contributions for i in range(max(0,k-n+1), min(k+1,n))
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # i from start to end inclusive: j = k - i, which is in range [0,n-1]
        # So for each i, add 2 * seq[j] to dM[i]
        # Since j = k - i, we can vectorize: indices i range; j = k - i.
        i_indices = np.arange(start, end + 1)
        j_indices = k - i_indices
        dM[i_indices] += 2.0 * seq[j_indices]
    # Average over active count
    dM /= len(active)
    # Compute gradient of objective O
    const_factor = 2.0 * n / (sum_val ** 2)
    const_sub = 4.0 * n * max_val / (sum_val ** 3)
    grad = const_factor * dM - const_sub
    # Update
    new_seq = seq - lr * grad
    # Ensure nonneg
    new_seq = np.maximum(new_seq, 0.0)
    # Renormalize to target sum
    new_sum = new_seq.sum()
    if new_sum > 0:
        new_seq = new_seq * (target_sum / new_sum)
    else:
        # if zero, reinitialize random
        new_seq = np.random.rand(n)
        new_seq = new_seq * (target_sum / new_seq.sum())
    return new_seq
```

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    # Build convolution constraints: For each k, sum_{i} f[i] * b_{k-i} <= rhs
    # Represent as A_ub * b <= b_ub
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    a_ub = np.vstack([a_ub, -np.eye(n)])  # non-negativity as -b <= 0
    b_ub = np.hstack([b_ub, np.zeros(n)])
    result = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if result.success:
        return result.x
    else:
        return None
```

    sum_val = seq.sum()
    if sum_val == 0:
        # reinitialize random
        seq = np.random.rand(n)
        seq = seq * (target_sum / seq.sum())
        sum_val = target_sum
    conv = compute_convolution(seq)
    max_val = conv.max()

    eps = 1e-12
    active = np.where(np.abs(conv - max_val) <= eps)[0]
    # compute dM_i avg
    dM = np.zeros_like(seq)
    # For each active k, compute contributions
    for k in active:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i_idx = np.arange(start, end + 1)
        j_idx = k - i_idx
        dM[i_idx] += 2.0 * seq[j_idx]
    dM /= len(active)
    # gradient of objective O
    const_factor = 2.0 * n / (sum_val ** 2)
    const_sub = 4.0 * n * max_val / (sum_val ** 3)
    grad = const_factor * dM - const_sub
    new_seq = seq - lr * grad
    new_seq = np.maximum(new_seq, 0.0)
    # renormalize to target sum
    new_sum = new_seq.sum()
    if new_sum > 0:

    else:
        # all zeros, reinitialize random
        new_seq = np.random.rand(n)
        new_seq = new_seq * (target_sum / new_seq.sum())

"""
Simple hybrid optimizer for the step‐function bound.

The optimizer works on a single candidate sequence (numpy array) of length *n*.
It repeatedly applies three kinds of moves:

1. **Gradient descent step** – a sub‑gradient of the evaluation function
   (2·n·max(conv)/sum²) is computed analytically and a small step is taken.
   After the step the sequence is projected onto the non‑negative orthant
   and re‑normalised to the target sum √(2n) (the evaluation is scale‑invariant).

2. **Random perturbation** – adds Gaussian noise, clips negative values
   and re‑normalises. This helps to escape shallow plateaus.

3. **LP improvement** – solves the linear programme used in the SOTA
   construction to obtain a direction that does not increase the convolution
   bound. The new sequence is a convex combination of the current one and the
   LP solution.

The algorithm keeps the best sequence seen so far and stops after the
allocated time budget (≈ 1000 s).  It returns the best sequence as a list of
Python floats, ready for `evaluate_sequence`.
"""

import math
import random
import time
import numpy as np
from scipy.optimize import linprog


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Compute the full convolution a*a using a real FFT.
    Returns a 1‑D array of length 2*len(seq)-1.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    return conv[: 2 * n - 1]


def _score(seq: np.ndarray) -> float:
    """
    Direct evaluation of the objective 2·n·max(conv)/sum².
    The function mirrors `evaluate_sequence` but avoids the security checks.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max() / (s * s))


def _gradient_step(seq: np.ndarray, lr: float, target_sum: float) -> np.ndarray:
    """
    One gradient descent step.

    Parameters
    ----------
    seq : ndarray
        Current non‑negative sequence (already normalised to target_sum).
    lr : float
        Learning rate.
    target_sum : float
        Desired ℓ₁‑norm ( = √(2·n) ) after the step.

    Returns
    -------
    ndarray
        Updated sequence (projected onto non‑negative orthant and re‑normalised).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s == 0.0:
        # re‑initialise if we ever hit zero
        seq = np.random.rand(n)
        seq = seq * (target_sum / seq.sum())
        s = target_sum

    conv = _conv_fft(seq)
    max_val = conv.max()
    # indices where the maximum is attained (allow tiny numerical tolerance)
    eps = 1e-12
    active = np.where(np.abs(conv - max_val) <= eps)[0]

    # build the average sub‑gradient of the max‑convolution term
    dM = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        dM[i_idx] += 2.0 * seq[j_idx]
    dM /= len(active)

    # gradient of the whole objective
    const_factor = 2.0 * n / (s * s)
    const_sub = 4.0 * n * max_val / (s * s * s)
    grad = const_factor * dM - const_sub

    # gradient descent update, projection, and re‑normalisation
    new_seq = seq - lr * grad
    new_seq = np.maximum(new_seq, 0.0)
    new_sum = new_seq.sum()
    if new_sum > 0.0:
        new_seq = new_seq * (target_sum / new_sum)
    else:
        # fall back to a random point if everything vanished
        new_seq = np.random.rand(n)
        new_seq = new_seq * (target_sum / new_seq.sum())
    return new_seq


def _perturb(seq: np.ndarray, sigma: float, target_sum: float) -> np.ndarray:
    """
    Additive Gaussian noise followed by projection onto the feasible set.
    """
    n = seq.shape[0]
    noisy = seq + sigma * np.random.randn(n)
    noisy = np.maximum(noisy, 0.0)
    total = noisy.sum()
    if total > 0.0:
        noisy = noisy * (target_sum / total)
    else:
        noisy = np.random.rand(n)
        noisy = noisy * (target_sum / noisy.sum())
    return noisy


def _solve_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the linear programme
        maximise   Σ b_j
        subject to conv(f_seq, b) ≤ rhs   (for every convolution index)
                   b_j ≥ 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = f_seq.shape[0]
    c = -np.ones(n)                      # maximise Σb → minimise -Σb
    # Build the convolution constraints A_ub @ b ≤ rhs
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill row[j] = f_seq[i] whenever i + j == k
        # i varies, j = k - i
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
        rhs_vec.append(rhs)
    # non‑negativity: -b ≤ 0
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(rhs_vec)

    # Use the HiGHS solver with a small time limit (seconds)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'time_limit': 2.0})
    if res.success:
        return res.x
    return None


def _lp_improve(seq: np.ndarray, t: float, target_sum: float):
    """
    Perform the LP move described in the reference algorithm.
    Returns a new sequence or None if the LP failed.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s == 0.0:
        return None
    # normalise the current sequence as required by the LP formulation
    f_norm = seq * (math.sqrt(2.0 * n) / s)
    rhs = _conv_fft(f_norm).max()
    g = _solve_lp(f_norm, rhs)
    if g is None:
        return None
    g = np.asarray(g, dtype=float)
    sum_g = g.sum()
    if sum_g == 0.0:
        return None
    g_norm = g * (math.sqrt(2.0 * n) / sum_g)
    # convex combination and re‑normalisation
    new_seq = (1.0 - t) * seq + t * g_norm
    new_seq = new_seq * (target_sum / new_seq.sum())
    return new_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Maximum wall‑clock time (seconds) allowed for the search.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    # ------------------------------------------------------------------ #
    # Setup
    # ------------------------------------------------------------------ #
    np.random.seed(seed)
    random.seed(seed)

    # Length of the sequence – use the known construction if available,
    # otherwise fall back to a medium‑size value.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 1500

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------ #
    # Initial candidate (random, not uniform – uniform would be a stationary point)
    # ------------------------------------------------------------------ #
    cur = np.random.rand(n)
    cur = cur * (target_sum / cur.sum())
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------ #
    # Hyper‑parameters
    # ------------------------------------------------------------------ #
    lr = 0.5                # initial gradient learning rate
    sigma = 0.02            # Gaussian noise magnitude for perturbations
    t_lp = 0.05             # mixing factor for the LP move
    no_improve = 0          # counter for stagnation detection
    iteration = 0

    deadline = time.time() + budget_s - 2.0   # small safety margin

    print(f"[INIT] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        iteration += 1
        rnd = np.random.rand()

        if rnd < 0.55:
            # Gradient descent step
            cand = _gradient_step(cur, lr, target_sum)
        elif rnd < 0.85:
            # Random perturbation
            cand = _perturb(cur, sigma, target_sum)
        else:
            # LP improvement (rare, because it is relatively expensive)
            cand = _lp_improve(cur, t_lp, target_sum)
            if cand is None:
                cand = _perturb(cur, sigma, target_sum)

        cand_score = _score(cand)

        # -------------------------------------------------------------- #
        # Acceptance logic
        # -------------------------------------------------------------- #
        if cand_score < cur_score - 1e-12:
            # Accept as new current point (always move downhill)
            cur, cur_score = cand, cand_score
            # increase learning rate modestly after a successful step
            lr = min(lr * 1.03, 2.0)
            no_improve = 0
        else:
            # No improvement – shrink learning rate a bit
            lr = max(lr * 0.90, 1e-4)
            no_improve += 1

        # -------------------------------------------------------------- #
        # Track the global best
        # -------------------------------------------------------------- #
        if cand_score < best_score - 1e-12:
            best, best_score = cand.copy(), cand_score
            elapsed = time.time() - (deadline - budget_s)
            print(f"[{iteration:06d}] New best score={best_score:.6f} (elapsed {elapsed:.1f}s)")

        # -------------------------------------------------------------- #
        # Simple restart mechanism if we are stuck for a long time
        # -------------------------------------------------------------- #
        if no_improve >= 2000:
            cur = np.random.rand(n)
            cur = cur * (target_sum / cur.sum())
            cur_score = _score(cur)
            lr = 0.5
            no_improve = 0
            print(f"[{iteration:06d}] Restarted due to stagnation (score {cur_score:.6f})")

        # occasional status report
        if iteration % 5000 == 0:
            print(f"[{iteration:06d}] Current score={cur_score:.6f}, best={best_score:.6f}")

    # ------------------------------------------------------------------ #
    # Finish
    # ------------------------------------------------------------------ #
    print(f"[DONE] Iterations={iteration}, best score={best_score:.6f}")
    # Convert to plain Python list as required by the evaluator
    return best.tolist()
