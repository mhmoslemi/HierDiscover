Pseudo:

def simplex_projection(v):
    """Project vector v onto simplex {x>=0, sum x = 1}."""
    if np.sum(v) == 1 and np.all(v>=0): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def subgradient_max(a, k):
    n = len(a)
    idx = np.arange(n)
    rev_idx = k - idx
    mask = (rev_idx >= 0) & (rev_idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2 * a[rev_idx[mask]]
    return grad

        return None

"""
propose_candidate
=================

Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution.  
The function builds a family of power‑law shaped sequences
a_i = 1 / sqrt(i + 1) (the known near‑optimal continuous solution) and
refines them with a simple projected sub‑gradient descent that directly
reduces the maximal convolution value.  Several lengths and a few random
re‑starts are tried within the time budget; the best sequence found is
returned.

The algorithm is completely self‑contained, uses only NumPy (no file or
network access) and respects the 1000 s runtime limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _normalize(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform if everything became zero
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _fast_score(v: np.ndarray) -> float:
    """
    Compute the evaluation function for a *already normalised* vector.
    Because sum(v) == 1 the denominator disappears and the score is
        2 * n * max(conv(v, v))
    """
    conv = np.convolve(v, v)
    return 2.0 * v.size * conv.max()


def _subgradient_step(v: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective
    f(v) = max_k (v * v)[k].

    The sub‑gradient at the current maximising index k* is
        g_i = 2 * v_{k* - i}   (when the shifted index is in range)
    The step moves opposite to the sub‑gradient and the result is
    projected back onto the simplex.
    """
    n = v.size
    conv = np.convolve(v, v)
    k_star = int(conv.argmax())          # index where the maximum occurs

    idx = np.arange(n)
    partner = k_star - idx                # partner index for each i
    mask = (partner >= 0) & (partner < n)

    grad = np.zeros_like(v)
    grad[mask] = 2.0 * v[partner[mask]]

    new_v = v - step * grad
    return _normalize(new_v)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a good height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (default 1000).
    kwargs : dict
        Ignored – present only to match the expected signature.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    # A few problem sizes – longer vectors give a better asymptotic bound,
    # but very long ones are slower to optimise.
    lengths = [800, 1200, 1600, 1823, 2000, 2500]

    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # 1. Start from the classic 1/sqrt(i) shape (power‑law with p=0.5)
        # ------------------------------------------------------------------
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
        a = _normalize(a)

        cur = a.copy()
        cur_score = _fast_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[init] n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 2. Projected sub‑gradient descent to lower the maximal
        #    convolution value.
        # ------------------------------------------------------------------
        step0 = 0.5                     # initial step size
        it = 0
        while time.time() < deadline and it < 3000:
            # decreasing step size (standard sub‑gradient schedule)
            step = step0 / (np.sqrt(it + 1) + 1.0)
            cur = _subgradient_step(cur, step)

            cur_score = _fast_score(cur)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[improve] n={n} it={it} score={best_score:.6f}")

            it += 1

        # ------------------------------------------------------------------
        # 3. A few random restarts – sometimes a different basin gives a
        #    slightly better flatness.
        # ------------------------------------------------------------------
        for restart in range(3):
            if time.time() > deadline:
                break
            rand = np.random.rand(n)
            cur = _normalize(rand)

            it = 0
            while time.time() < deadline and it < 800:
                step = step0 / (np.sqrt(it + 1) + 1.0)
                cur = _subgradient_step(cur, step)

                cur_score = _fast_score(cur)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"[rand] n={n} restart={restart} it={it} "
                          f"score={best_score:.6f}")

                it += 1

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list (the harness expects it)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform vector (never happens in practice)
        best_seq = np.full(100, 1.0 / 100.0)

    print(f"Final best score (fast estimate): {best_score:.6f}")
    return best_seq.tolist()
