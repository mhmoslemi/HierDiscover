import numpy as np
import time
import math
# evaluate_sequence is assumed to be imported

def _convolution_max(seq):
    """Return the maximum value of the (full) convolution seq*seq."""
    # using numpy's convolve which is fast enough for lengths <= 5000
    b = np.convolve(seq, seq)
    return np.max(b)

def _generate_random_sequence(n, density, rng):
    """
    Generates a random non‑negative sequence of length n.
    A fraction `density` of the entries are set to the same value,
    the rest are zero. The values are scaled so that the L1‑norm of the
    sequence equals sqrt(2n) (the natural normalisation used in the
    evaluation function). Small random perturbations are added to the
    non‑zero entries to break symmetry.
    """
    s = max(2, int(round(density * n)))
    # ensure we do not exceed n
    s = min(s, n)
    # random distinct indices
    indices = rng.choice(n, size=s, replace=False)
    # baseline weight to satisfy the sum constraint
    target_sum = math.sqrt(2.0 * n)
    w = target_sum / s
    # add a tiny multiplicative noise (10% std) to each non‑zero entry
    noise = 1.0 + rng.normal(loc=0.0, scale=0.1, size=s)
    values = w * np.maximum(0.0, noise)
    # renormalise to keep the exact L1‑norm (scaling does not affect the score)
    total = values.sum()
    if total < 1e-12:
        values[:] = w
        total = values.sum()
    values *= target_sum / total

"""
Fast random‑sparse search for a step function with a small
self‑convolution maximum.

The evaluation metric is invariant under scaling of the whole vector,
so we can normalise every candidate to have L1‑norm sqrt(2 n)
(the normalisation used in the reference implementation).

The algorithm works in two phases:

1. **Random sampling** – generate many sequences that are zero everywhere
   except on a randomly chosen subset of indices.  All non‑zero entries
   receive the same weight (with a tiny multiplicative noise to break
   perfect symmetry).  The density of the support is sampled uniformly
   in [0.2, 0.8]; theory predicts that a moderate density (≈ 0.5) gives a
   good trade‑off between the number of terms that overlap in a convolution
   and the size of each term, typically yielding scores ≈ 1.5 or less.

2. **Local optimisation** – starting from the best sequence found in
   phase 1 we perform a simple simulated‑annealing style hill‑climb.
   At each step a small amount of mass is transferred from a non‑zero
   position to a zero position (the L1‑norm is preserved).  Moves that
   improve the score are always accepted; worsening moves are accepted
   with a temperature‑dependent probability.

The whole search stops after the allotted time budget (default 1000 s)
or when a sufficiently low score is reached.  The function returns the
best sequence as a plain list of floats.
"""

import math
import time

import numpy as np


def _generate_random_sequence(n: int, density: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative sequence of length *n*.
    Exactly ``density * n`` entries are non‑zero; they all share the same
    weight up to a small Gaussian perturbation.  The vector is scaled so
    that its L1‑norm equals ``sqrt(2 n)``.
    """
    s = max(2, int(round(density * n)))          # number of non‑zero entries
    s = min(s, n)                                 # cannot exceed length
    # choose distinct positions
    idx = rng.choice(n, size=s, replace=False)

    # weight that would give the exact normalisation if all entries were equal
    target_sum = math.sqrt(2.0 * n)
    base_weight = target_sum / s

    # add a modest noise (10 % std) to avoid perfectly symmetric patterns
    noise = 1.0 + rng.normal(loc=0.0, scale=0.1, size=s)
    values = base_weight * np.maximum(0.0, noise)

    # renormalise to keep the exact L1‑norm (scaling does not affect the score)
    total = values.sum()
    if total <= 0.0:                     # degenerate case – fall back to uniform weight
        values[:] = base_weight
        total = values.sum()
    values *= target_sum / total

    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = values
    return seq
