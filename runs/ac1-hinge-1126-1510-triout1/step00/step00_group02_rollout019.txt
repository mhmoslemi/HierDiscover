"""
Propose a coefficient sequence that minimizes the evaluation function.

The algorithm starts from a smooth “square‑root‑difference” profile
a_i = sqrt(i+1) - sqrt(i) (which is known to give a low autocorrelation
peak) and then performs a simple projected gradient descent on the
position k where the current convolution peak occurs.  Small random
perturbations and a short simulated‑annealing phase are added to avoid
getting stuck in a local minimum.  The procedure runs for the allotted
time budget and returns the best sequence found.

All numbers are kept non‑negative and below the 0‑1000 clipping limit
used inside `evaluate_sequence`.  The returned sequence is scaled to the
full 0‑1000 range (scaling does not affect the score because the metric
is homogeneous).
"""

import time
import numpy as np

def _conv_max_and_index(seq: np.ndarray):
    """Return (max_value, index_of_max) of the full convolution of seq with itself."""
    conv = np.convolve(seq, seq)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    # ------------------------------------------------------------------
    # 1. Determine the length of the step function
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                     # uses the global if present
    except Exception:                                 # noqa: BLE001
        n = 1542                                        # default length from the statement

    # ------------------------------------------------------------------
    # 2. Initialise a promising starting point
    #    a_i = sqrt(i+1) - sqrt(i)  (the “optimal” continuous kernel)
    # ------------------------------------------------------------------
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    # Bring the vector into the [0, 1000] range (scaling does not change the score)
    if a.max() > 1000.0:
        a = a * (1000.0 / a.max())
    # Add a tiny random jitter to break exact symmetry
    a += np.random.randn(n) * 1e-5
    a = np.clip(a, 0.0, None)

    # Normalise to unit L1 – the score is scale invariant, this avoids huge numbers
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (projected gradient descent on the peak index)
    # ------------------------------------------------------------------
    # parameters – they slowly decay so the method is aggressive early on
    step_scale = 0.02          # base step size (fraction of the vector norm)
    temperature = 1.0
    min_step_scale = 1e-5

    while time.time() < deadline:
        # compute the convolution peak
        max_b, k_max = _conv_max_and_index(a)

        # ---- gradient of the peak with respect to a -------------------------------------------------
        # only entries that appear in the active convolution term contribute
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        if i_low <= i_high:
            # vectorised contribution
            i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
            j_vals = k_max - i_vals
            grad[i_vals] = a[i_vals] + a[j_vals]

        # The objective also depends on the denominator (sum a)^2, which
        # contributes a constant term to the gradient; we enforce a zero‑sum
        # move by subtracting the mean of the computed gradient.
        mean_grad = grad.mean()
        direction = -grad + mean_grad                     # ensures sum(direction)=0

        # ---- step length -------------------------------------------------
        # adapt step size to the magnitude of the direction and to the schedule
        step = step_scale * temperature * (np.linalg.norm(direction) / n + 1e-12)
        step = max(step, min_step_scale)

        a_candidate = a + step * direction
        a_candidate = np.clip(a_candidate, 0.0, None)
        if a_candidate.sum() == 0.0:                     # safeguard against collapse
            continue
        a_candidate = a_candidate / a_candidate.sum()    # re‑normalise to keep scale fixed

        score = evaluate_sequence(a_candidate.tolist())

            step_scale = max(step_scale * 0.99, min_step_scale)

            delta = a[idx] * (np.random.rand() - 0.5) * 0.1
            a[idx] = max(0.0, a[idx] + delta)

    print(f"Returning sequence with final score {best_score:.6f}")
