def project_onto_simplex(v):
    """Project a vector onto the probability simplex (nonnegative, sum = 1)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    """Projects a vector onto the probability simplex (nonnegative, sum = 1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]

    return w

"""
Propose a sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a simple projected gradient descent on a
smooth approximation of the max‑convolution.
We use a soft‑max (log‑sum‑exp) weighting of the convolution
values, compute the gradient via a correlation, take a small
step, project back onto the probability simplex (non‑negative,
sum = 1) and repeat.  The temperature of the soft‑max is
gradually increased so the optimisation focuses more on the
true maximum as it proceeds.  Random perturbations and restarts
help to avoid local minima.

The method works for moderate lengths (≈150‑300) and fits easily
into the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum(x) = 1}
    Implementation follows the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation score.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Choose a starting length (150‑300 works well for speed/quality)
    n = np.random.randint(150, 301)

    # Start from a uniform distribution on the simplex (sum = 1)
    a = np.full(n, 1.0 / n)

    # If a previous construction is available, occasionally use it.
    if "height_sequence_1" in globals():
        if np.random.rand() < 0.3:   # 30 % chance to start from SOTA
            a = np.array(height_sequence_1, dtype=float)
            a = np.clip(a, 0.0, 1000.0)
            a = _project_onto_simplex(a)  # renormalise to sum = 1
            n = a.shape[0]

    # Hyper‑parameters (tuned empirically)
    beta = 0.5          # temperature of the soft‑max (grows)
    lr0 = 0.30          # initial learning‑rate (decays like 1/√t)
    last_improve = 0    # iteration of the last improvement

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current sequence ---------------------------------
        conv = np.convolve(a, a)               # full auto‑convolution
        max_conv = conv.max()
        # since sum(a) == 1 after projection,
        # the evaluation formula reduces to 2 * n * max_conv
        cur_val = 2.0 * n * max_conv

        # keep the best sequence seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] iter {iteration:6d} – new best {best_val:.6f}")

        # ----- increase temperature slowly (focus more on the true max) --
        if iteration % 200 == 0:
            beta = min(beta * 1.02, 10.0)

        # ----- compute the smoothed gradient ----------------------------
        # soft‑max weights (max weight = 1)
        w = np.exp(beta * (conv - max_conv))

        # gradient of the smoothed objective:
        #   grad_i = 2 * Σ_j w_{i+j} * a_j
        # This is exactly 2 * correlation(w, a) with mode='valid'.
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # ----- take a projected gradient step -----------------------------
        # Normalise the direction to keep steps of comparable magnitude.
        max_abs = np.max(np.abs(grad)) + 1e-12
        direction = grad / max_abs

        # Learning rate that decays like 1/√iteration.
        lr = lr0 / np.sqrt(iteration)

        new_a = a - lr * direction
        new_a = np.maximum(new_a, 0.0)          # enforce non‑negativity
        new_a = _project_onto_simplex(new_a)    # renormalise to sum = 1

        # ----- accept the new point if it improves (or occasionally) -----
        new_conv = np.convolve(new_a, new_a)
        new_val = 2.0 * n * new_conv.max()

        # simple Metropolis criterion to allow occasional uphill moves
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability decreasing with the increase and with time
            prob = np.exp(-(new_val - cur_val) /
                          (0.01 * np.sqrt(iteration) + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = new_a

        # ----- occasional random perturbation to escape plateaus -------
        if iteration % 1000 == 0:
            noise = np.random.randn(n) * 0.004
            a = np.maximum(a + noise, 0.0)
            a = _project_onto_simplex(a)

        # ----- restart if we have not improved for a while -------------
        if iteration - last_improve > 8000:
            # re‑initialise to a fresh uniform distribution
            a = np.full(n, 1.0 / n)
            last_improve = iteration

    # ------------------------------------------------------------------
    # Return the best sequence that was discovered.
    if best_seq is None:
        # Fallback – a simple uniform vector of length 100
        best_seq = np.full(100, 1.0 / 100.0)

    # The evaluation routine will cap values to [0,1000] and
    # enforce non‑negativity, so we simply return the raw floats.
    return best_seq.tolist()
