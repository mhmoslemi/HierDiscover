"""
propose_candidate
-----------------

Search for a step‑function (a non‑negative list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution of ``a`` with itself.
The implementation uses a one‑parameter family of sequences:
    a_i = exp(-alpha * i)   (i = 0 … n‑1)

The parameter ``alpha`` controls how fast the heights decay.
For a fixed length ``n`` the evaluation value behaves roughly like
    2 * n * alpha / e,
so a small positive ``alpha`` can push the score well below the uniform
value of 2.  The optimizer performs a fast golden‑section search on ``alpha`
to find the value that gives the smallest score, then returns the
corresponding sequence.

The routine respects the 1000‑second wall‑time limit, caches evaluations
to avoid duplicate work and prints progress information.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def exponential_sequence(n: int, alpha: float) -> List[float]:
    """Return a normalized exponential sequence of length *n* with decay *alpha*."""
    # a_i = exp(-alpha * i)
    idx = np.arange(n, dtype=float)
    seq = np.exp(-alpha * idx)
    # normalise so the sum is 1 (any scaling leaves the score unchanged)
    seq /= seq.sum()
    return seq.tolist()


def evaluate_alpha(alpha: float, n: int, cache: dict) -> float:
    """Evaluate the score for a given alpha, using a cache."""
    if alpha in cache:
        return cache[alpha]
    seq = exponential_sequence(n, alpha)
    score = evaluate_sequence(seq)          # evaluate_sequence is provided by the judge
    cache[alpha] = score
    return score


def golden_section_search(
    func,
    a: float,
    b: float,
    n: int,
    max_iter: int,
    deadline: float,
    cache: dict,
) -> float:
    """
    Find alpha in [a, b] that minimises *func(alpha, n, cache)*
    using the golden‑section method. Stops early if the time limit is hit.
    Returns the best alpha encountered.
    """
    phi = (math.sqrt(5) - 1) / 2          # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c, n, cache)
    fd = func(d, n, cache)
    best_alpha = c if fc < fd else d
    best_score = min(fc, fd)

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if fc < fd:
            b, fd = d, fc                # move the interval left
            d = c
            c = b - phi * (b - a)
            fc = func(c, n, cache)
            if fc < best_score:
                best_score = fc
                best_alpha = c
        else:
            a, fc = c, fd                # move the interval right
            c = d
            d = a + phi * (b - a)
            fd = func(d, n, cache)
            if fd < best_score:
                best_score = fd
                best_alpha = d
        # stop when interval is sufficiently small
        if abs(b - a) < 1e-12:
            break
    return best_alpha


def local_random_refine(seq: List[float], time_limit: float) -> List[float]:
    """
    Starting from *seq*, perform cheap random “mass‑transfer” moves while
    the overall wall‑time allows. Accept only moves that improve the score.
    This can sometimes push the result a little further below the optimum
    found by the 1‑parameter search.
    """
    n = len(seq)
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    start = time.time()
    while time.time() - start < time_limit:
        # pick two indices and move a small amount from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = 0.01 * best_seq[i]                # 1 % of the source value
        if delta <= 0:
            continue
        new_seq = best_seq[:]
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] = new_seq[j] + delta
        # renormalise to keep the sum (the score is scale‑invariant)
        s = sum(new_seq)
        if s < 1e-12:   # protect against accidental zero sum
            continue
        new_seq = [x / s for x in new_seq]
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
    return best_seq
