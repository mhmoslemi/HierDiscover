def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random

    np.random.seed(seed)
    random.seed(seed)

    # Choose length n. Could try multiple lengths and keep best.

    best_seq = None
    best_score = np.inf

    time_deadline = time.time() + budget_s - 0.5

    # Let's try different n values: e.g., [200, 400, 800, 1500] (or up to len(height_sequence_1) maybe 1542)
    n_options = [200, 400, 800, 1200]  # adapt to budget.

    # We'll loop over lengths until time runs out.
    for n in n_options:
        # initialize with uniform.
        a = np.full(n, 1.0/n)
        # optionally jitter small random noise
        a += np.random.randn(n) * 1e-4
        a = np.maximum(a, 0)
        a /= a.sum()

        # parameters
        iteration = 0
        # hyperparameters
        gamma = 0.1   # initial step size scaling factor.

        while time.time() < time_deadline and iteration < 50000:
            iteration += 1

            # Compute convolution
            b = np.convolve(a, a)
            max_b = b.max()
            score = 2 * n * max_b / (a.sum()**2)  # sum is 1 so 2*n*max_b
            # note: evaluate_sequence may clip min 0 & max 1000 but we guarantee positive small values.

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best score {best_score:.6f} at n={n} iter={iteration}")

            # Subgradient
            k = int(b.argmax())
            # compute gradient vector: grad_i = 2*a[k-i] if in bounds else 0
            grad = np.zeros_like(a)
            # vectorized: indices where i from max(0,k-(n-1)) to min(n-1,k)
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            # For i in that range:
            indices = np.arange(i_min, i_max+1)
            grad[indices] = 2 * a[k - indices]

            # step size schedule (like diminishing)
            step = gamma / np.sqrt(iteration)  # diminishing step

            # Update a
            a_new = a - step * grad
            # ensure non-negative:
            a_new = np.maximum(a_new, 0.0)
            # renormalize:
            sum_new = a_new.sum()
            if sum_new < 1e-12:
                # fallback to uniform
                a_new = np.full(n, 1.0/n)
            else:
                a_new /= sum_new

            # Occasionally add random jitter to escape local minima
            if iteration % 1000 == 0:
                a_new += np.random.randn(n) * 1e-3
                a_new = np.maximum(a_new, 0)
                a_new /= a_new.sum()

            a = a_new

        # after finishing for this n, maybe try random restart
        # We could also try random start more.

        if time.time() >= time_deadline:
            break

    # Return best sequence as list
    return best_seq.tolist()

This construction uses a linear decreasing distribution
