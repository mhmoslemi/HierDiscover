    start_time = time.time()
    deadline = start_time + budget_s - 5

        base_seq = rng.random(n)
    n = len(base_seq)

    # Normalize base_seq to sum sqrt(2n)
    target_sum = np.sqrt(2 * n)
    base_seq = np.clip(base_seq, 0, None)
    base_sum = base_seq.sum()
    if base_sum == 0:
        base_seq = np.full(n, 1.0)
    base_seq = base_seq * (target_sum / base_seq.sum())

    best_seq = base_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Current sequence
    curr_seq = best_seq.copy()
    curr_score = best_score

    # Hyperparameters
    eta = 0.5  # initial step size
    alpha = 8.0  # smoothing of max
    it = 0
    while time.time() < deadline:
        # Compute convolution and max
        conv = np.convolve(curr_seq, curr_seq)
        max_conv = conv.max()
        # Softmax weights
        # Use exponent of (conv - max_conv) * alpha to ensure non-negative and numerically stable
        exps = np.exp(alpha * (conv - max_conv))  # values ≤ 1

        grad_norm = np.linalg.norm(grad)

            idx = rng.integers(n)
            curr_seq[idx] += rng.normal(scale=0.1)
            curr_seq = np.clip(curr_seq, 0, None)
            curr_seq = curr_seq * (target_sum / curr_seq.sum())
            continue

        if it % 200 == 0 and it > 0:
            eta *= 0.9

        step = eta * grad / grad_norm  # unit direction scaled
        new_seq = curr_seq - step
        # Ensure nonnegative
        new_seq = np.clip(new_seq, 0, None)
        # Project onto simplex (sum = target_sum)
        sum_new = new_seq.sum()

            sum_new = new_seq.sum()

            new_score = evaluate_sequence(new_seq.tolist())

            # fallback: skip

"""
Propose a step‑function (non‑negative heights) that improves the
evaluation score.

The algorithm is a simple projected gradient descent on a smooth
approximation of the max‑convolution objective.
*   The sequence is always projected onto the simplex
    Σ a_i = sqrt(2·n) (the scale used by the evaluator).
*   A soft‑max (temperature α) is used to turn the max of the
    convolution into a differentiable objective.
*   The gradient of the soft‑max objective is obtained by a single
    correlation (np.correlate).  After a descent step we clip negatives
    and renormalise.
*   Multiple random restarts are performed while time permits; the
    best sequence found is returned.
"""

import time
import numpy as np

def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max weights w_k that approximate the max of conv."""
    # conv is non‑negative; shift by its maximum for numerical stability.
    max_c = conv.max()
    # exponent values are in [0,1] after the shift, so no overflow.
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()
    return w


def _gradient_softmax(seq: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximated max‑convolution.
    grad_i = 2 * Σ_k w_k * a_{k-i}   (indices out of range contribute 0)
    This is exactly 2 * np.correlate(w, a, mode='valid').
    """
    grad = 2.0 * np.correlate(weights, seq, mode="valid")
    return grad


def _project_onto_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that it sums to target_sum.
    Zero entries stay zero; the scaling factor is applied to the whole vector.
    """
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s == 0.0:
        # fallback to a tiny positive vector to avoid division by zero
        x = np.full_like(x, 1e-12)
        s = x.sum()
    return x * (target_sum / s)

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # give a small safety margin before the hard timeout
    deadline = start_time + float(budget_s) - 5.0

    # ----- length of the sequence ------------------------------------------

    target_sum = np.sqrt(2.0 * n)          # the normalization used by the evaluator
    best_seq = None
    best_score = np.inf
    iteration = 0

    # hyper‑parameters (tuned empirically)
    alpha = 6.0            # temperature for the soft‑max (higher ⇒ sharper)
    eta_initial = 0.6      # initial step‑size (norm of the descent direction)
    restart_min_steps = 80
    restart_max_steps = 150

    while time.time() < deadline:
        # ----- random start -------------------------------------------------
        seq = rng.random(n, dtype=float)
        seq = _project_onto_simplex(seq, target_sum)

        # start with a relatively large step size, shrink over the descent
        eta = eta_initial

        # number of gradient steps for this restart (randomised a bit)
        steps_this_restart = int(rng.integers(restart_min_steps, restart_max_steps + 1))

        for _ in range(steps_this_restart):
            if time.time() >= deadline:

            # full convolution of the current sequence with itself
            conv = np.convolve(seq, seq)
            max_conv = conv.max()                     # this is the objective value
