def generate_sidon_set(n, target_m):
    positions = []
    used_sums = set()
    # we also may want to prepopulate used_sums with 0..maybe none.

    while len(positions) < target_m:
        # random candidate from range
        cand = np.random.randint(n)
        if cand in positions:
            continue
        # compute sums
        new_sums = [cand + p for p in positions] + [cand + cand]
        # check collision
        if any(s in used_sums for s in new_sums):
            # try again
            continue
        # add
        positions.append(cand)
        for s in new_sums:
            used_sums.add(s)
    return positions

"""
Propose a step‑function (non‑negative sequence) that improves the bound
2·n·max(conv)/sum².  The idea is to place the same height on a subset of
indices such that every unordered pair of indices produces a *different*
sum i+j.  If all sums are distinct, each entry of the convolution contains
at most one product a_i·a_j, so the maximal convolution value is just a_i².
For a set of size m with equal heights v we have

    max(conv) = v² = (√(2n)/m)² = 2n / m² .

Thus the score is 2n·max(conv)/sum² = 2n/m².  To beat the target
1.503 we only need m > √(2n/1.503) ≈ 93 for n≈6400.  We construct a
large “Sidon‑type” set (all sums i+j distinct) by greedy random
permutations and keep the largest one found before the time budget
expires.

The returned sequence has length n (taken from the existing
`height_sequence_1` if present, otherwise 6402) and contains the value
v at the selected positions, zeros elsewhere.
"""
import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: build a set of indices with pairwise distinct sums (i+j).
# ----------------------------------------------------------------------
def greedy_distinct_sum_set(n, rnd_state, max_trials=1):
    """
    Return a list of indices 0<=i<n such that all unordered sums i+j
    (including i=j) are distinct.  The function performs `max_trials`
    greedy attempts with different random orders and keeps the largest
    set found.
    """
    best = []
    for _ in range(max_trials):
        order = rnd_state.permutation(n)
        used = set()
        cur = []
        for cand in order:
            # self‑sum already used?
            s_self = cand + cand
            if s_self in used:
                continue
            # any collision with already chosen elements?
            conflict = False
            for p in cur:
                if cand + p in used:
                    conflict = True
                    break
            if conflict:
                continue
            # add this index
            for p in cur:
                used.add(cand + p)
            used.add(s_self)
            cur.append(cand)
        if len(cur) > len(best):
            best = cur
    return best


# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence up to rounding).
# ----------------------------------------------------------------------
def fast_score(seq):
    """
    Compute the same score as evaluate_sequence but using FFT for speed.
    The returned value equals 2·n·max(conv)/sum².
    """
    n = len(seq)
    # zero‑pad to the next power of two for an efficient FFT
    pad = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(seq, n=pad)
    conv = np.fft.irfft(a_fft * a_fft, n=pad)[: 2 * n - 1]
    max_b = conv.max()
    s = np.sum(seq)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Build a Sidon‑type sequence of length n.  The algorithm repeatedly
    tries to enlarge a set of indices with distinct pairwise sums, then
    puts an equal height on those positions.  The best sequence found
    before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    # ------------------------------------------------------------------
    # Determine the problem size n.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6402                       # fallback length used by the SOTA code

    # Target score (the smaller the better)
    TARGET = 1.503

    # Minimum number of positions needed to beat the target.
    # score = 2*n / m²  ⇒  m > sqrt(2*n / TARGET)
    required_m = int(math.ceil(math.sqrt(2.0 * n / TARGET)))
    print(f"[info] n = {n}, required size ≥ {required_m}")

    best_seq = None
    best_score = float('inf')
    attempts = 0

    # Random state object for reproducibility inside the loop.
    rnd = np.random.RandomState(seed)

    while time.time() < deadline:
        # Try to grow a distinct‑sum set; a few trials per iteration give
        # a chance to escape local stagnation.
        positions = greedy_distinct_sum_set(n, rnd, max_trials=3)

        m = len(positions)
        if m < required_m:
            # Not enough positions – try again.
            attempts += 1
            continue

        # Equal height on the chosen positions.
        v = math.sqrt(2.0 * n) / m
        seq = np.zeros(n, dtype=float)
        seq[positions] = v

        # Fast scoring (identical to evaluate_sequence up to tiny rounding).
        score = fast_score(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[improve] attempt={attempts}, m={m}, score={score:.6f}")

        # Early exit if we already beat the target.
        if best_score <= TARGET:
            break

        attempts += 1

    # If nothing satisfactory was found, fall back to the uniform sequence.
    if best_seq is None:
        print("[fallback] using uniform sequence")
        uniform_val = math.sqrt(2.0 * n) / n
        best_seq = [uniform_val] * n
        best_score = float(evaluate_sequence(best_seq))

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
