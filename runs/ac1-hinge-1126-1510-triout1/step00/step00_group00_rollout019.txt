"""
Propose a low‑value step‑function sequence.

The objective to minimise is
    f(a) = 2·n·max(conv(a,a)) / (sum(a))**2
(scale‑invariant).  Two complementary heuristics are used:

1.  A high‑cardinality “Sidon‑like’’ set of large entries.
    Positions are added while keeping the count of each pair sum
    (i+j) below a small threshold (1 or 2).  For a set of size m we
    assign value 1 to the “large’’ indices and a uniform small value
    ε≈1/m to the remaining positions.  The resulting sequence is
    evaluated and the best ε is kept.

2.  Gradient‑descent on the exact objective.
    Starting from a random vector (sum normalised to 1) we repeatedly
    compute the convolution, locate the index of its maximum and take
    a projected gradient step.  The gradient of max(conv) is easy:
    ∂/∂a_j max = 2·a_{k−j} (0 ≤ k−j < n), where k is the
    arg‑max index.  The full gradient of f is
        ∇f = (2n / S²)·∇max  –  (4n·M / S³)·1,
    with S = sum(a) and M = max(conv).  After each step we clip to
    non‑negative values and renormalise the sum (scale‑invariance makes
    this harmless).  Several restarts are performed within the time
    budget.

The best sequence found by either method is returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: build a (almost) Sidon set, limiting how many times each sum may appear.
def build_sid_set(length: int, max_collisions: int, trials: int) -> list[int]:
    """Return a set of indices with pairwise sums occurring at most max_collisions."""
    best_set: list[int] = []
    for _ in range(trials):
        order = list(range(length))
        random.shuffle(order)

        sums = {}                     # sum -> count
        cur: list[int] = []

        for p in order:
            ok = True
            # check sums with existing elements
            for q in cur:
                s = p + q
                if sums.get(s, 0) + 1 > max_collisions:
                    ok = False
                    break
            if not ok:
                continue
            # self–sum
            s = p + p
            if sums.get(s, 0) + 1 > max_collisions:
                continue

            # accept p
            for q in cur:
                s = p + q
                sums[s] = sums.get(s, 0) + 1
            sums[p + p] = sums.get(p + p, 0) + 1
            cur.append(p)

        if len(cur) > len(best_set):
            best_set = cur
            # early exit when we reach a good size (≈sqrt(n)·1.5)
            if len(best_set) > int(1.5 * (length ** 0.5)):
                break
    return best_set


# ----------------------------------------------------------------------
# Helper: evaluate a candidate built from a Sidon set and a given epsilon.
def evaluate_sid_candidate(n: int, large_pos: list[int], eps: float):
    """Construct the sequence (1 on large_pos, eps elsewhere) and evaluate."""
    seq = np.full(n, eps, dtype=float)
    seq[large_pos] = 1.0
    # enforce the evaluation limits (clamping already done inside evaluate_sequence)
    return evaluate_sequence(seq.tolist()), seq.tolist()


# ----------------------------------------------------------------------
# Gradient‑descent optimisation on the exact objective.
def gradient_descent_opt(n: int, max_iters: int, step0: float):
    """
    Run a simple projected gradient descent on a random start.
    Returns the best (score, sequence) obtained.
    """
    # start from a random point, normalise sum to 1 (scale does not matter)
    a = np.random.rand(n)
    a = np.maximum(a, 0.0)
    a /= a.sum() + 1e-12

    best_score = float('inf')
    best_seq = None
    step = step0

    for it in range(max_iters):
        # convolution and max index
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        M = conv[k]
        S = a.sum()
        if S < 1e-9:
            break

        # gradient of the max‑convolution term:
        dM = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        dM[mask] = 2.0 * a[j[mask]]

        # full gradient of the objective
        grad = (2.0 * n / (S * S)) * dM - (4.0 * n * M / (S * S * S)) * np.ones_like(a)

        # projected gradient step
        a -= step * grad
        a = np.clip(a, 0.0, None)

        # renormalise (scale invariance)
        total = a.sum()
        if total < 1e-12:
            break
        a /= total

        # occasional evaluation
        if it % 50 == 0:
            try:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                # simple step‑size decay
                step *= 0.995
            except Exception:
                pass

    # final evaluation
    try:
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a.tolist()
    except Exception:
        pass

    return best_score, best_seq

                sidon_best_score = score
                sidon_best_seq = seq
                print(f"[Sidon] new best {score:.6f} (m={m}, eps={eps:.6e})")
                if score < 1.503:          # early exit if target reached
                    return sidon_best_seq

    # ------------------------------------------------------------------
    # 2) Gradient‑descent refinement (multiple restarts)
    gd_best_score = sidon_best_score
    gd_best_seq = sidon_best_seq

    # try a few different lengths – shorter vectors allow many iterations
    lengths = [200, 300, 400]
    for n_gd in lengths:
        # number of inner iterations calibrated to the remaining time budget
        max_iters = 5000
        while time.time() < deadline:
            score, seq = gradient_descent_opt(n_gd, max_iters, step0=0.2)
            if score < gd_best_score:
                gd_best_score = score
                gd_best_seq = seq
                print(f"[GD] new best {score:.6f} (len={n_gd})")
                if score < 1.503:
                    return gd_best_seq

    # ------------------------------------------------------------------
    # Return the best sequence found by any method
    result_seq = gd_best_seq if gd_best_seq is not None else sidon_best_seq
    # Guard against returning None (should not happen)
    if result_seq is None:
        result_seq = [1.0]                     # fallback trivial sequence
    return result_seq
