def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math, random
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    # Let's try multiple n values
    for n in [200, 400, 800, 1500]:
        if time.time() > deadline: break
        # Create several initial sequences:
        init_seqs = []
        # uniform
        init_seqs.append(np.ones(n))
        # geometric with c in [0.8,1.0,1.2,1.5]
        for c in [0.5,0.8,1.0,1.2,1.5,2.0]:
            if time.time() > deadline: break
            eps = c / n
            r = 1 - eps
            seq = (r ** np.arange(n))
            init_seqs.append(seq)
        # power law: a_i = (i+1)**(-p)
        for p in [0.5,1.0,1.5]:
            seq = (np.arange(1, n+1) ** (-p))
            init_seqs.append(seq)
        # sine shape
        seq = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))
        init_seqs.append(seq)

        for seq in init_seqs:
            if time.time() > deadline: break
            # Normalize to avoid blowups: scale to max <=1000
            if seq.max() > 0:
                seq = seq * (1000.0 / seq.max())
            else:
                continue
            # Evaluate
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq = score, seq.tolist()
            # Perform LP based refinement
            curr = np.array(seq)
            # Normalize to sum sqrt(2n) for LP
            target_sum = math.sqrt(2 * n)
            curr = curr * (target_sum / np.sum(curr))
            for it in range(30):
                if time.time() > deadline: break
                # compute bound
                conv_f = np.convolve(curr, curr)
                M = conv_f.max()
                # Solve LP to max sum b
                b = solve_convolution_lp(curr, M)
                if b is None:
                    break
                # Scale new b to target_sum
                sum_b = np.sum(b)
                new_seq = b * (target_sum / sum_b)
                # Clip to 0..1000 (post scaling)
                new_seq = np.clip(new_seq, 0, 1000)
                # Evaluate
                val = evaluate_sequence(new_seq.tolist())
                if val < best_score:
                    best_score, best_seq = val, new_seq.tolist()
                # If improvement not great, break
                # Accept new_seq as current for next iteration
                curr = new_seq
                # Accept only if val <= M (should be <=) else break
                if val >= best_score + 1e-6:
                    # maybe stop early
                    pass
                # optional early termination if val low enough (target 1.5)
                if best_score < 1.5:
                    break

        # End loops

    # As fallback if nothing found, return uniform with length maybe 100
    if best_seq is None:
        best_seq = [1.0] * 100
    return best_seq

def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n) # maximize sum = -c
    # build A_ub
    rows = []
    for k in range(2*n-1):
        # row = [f_seq[i] where i+j=k => j = k - i ; but j index must be within [0,n-1].
        # So for i from max(0,k-(n-1)) to min(k, n-1): row[j] = f_seq[i].
        # Implementation: create row vector zeros length n.
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # For each i in [i_min, i_max], set j = k - i:
        # row[j] = f_seq[i]
        # else 0.
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M, dtype=float)
    # non-negativity of b: -I b <= 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if result.success:
        return result.x
    else:
        return None

def normalize_to_sum(seq, target):
    s = np.sum(seq)
    if s == 0:
        return seq
    factor = target / s
    seq = seq * factor
    # clip to [0, 1000] after scaling
    seq = np.clip(seq, 0.0, 1000.0)
    # re-normalize if clipping changed sum, repeat until convergence or small change
    new_s = np.sum(seq)
    if new_s == 0:
        return seq
    # If clipping reduced sum, rescale again but keep within bounds
    # simple loop: adjust factor limited times
    for _ in range(5):

            break
        factor = target / new_s
        seq = seq * factor
        seq = np.clip(seq, 0.0, 1000.0)

"""
Propose a good step-height sequence for the provided evaluate_sequence().

Algorithm
---------
1.  Choose a moderate length n (200‑800).  All sequences are scaled so that
    their L1‑norm equals √(2n).  For that scaling the evaluation equals the
    maximal auto‑convolution value M = max(a*a) (because
        eval = 2·n·M / (∑a)² = M  when ∑a = √(2n)).
2.  Initialise a set of candidate sequences:
        – uniform,
        – geometric a_i = r^i  with r = 1–c/n for several c,
        – power‑law a_i = (i+1)⁻ᵖ,
        – a half‑sine shape,
        – a few random vectors.
3.  For each candidate run a short “LP‑refinement’’ loop.
    * Let f be the current sequence and M = max(convolve(f,f)).
    * Solve the linear programme
          maximise   Σ b_j
          subject to  (f * b)_k ≤ M   for all k = 0 … 2n‑2
                     b_j ≥ 0
      (the convolution (f*b) is linear in b, therefore the problem is an LP).
      The optimal b tries to put as much mass as possible while keeping the
      convolution bounded by the same M.
    * Scale b back to the prescribed L1‑norm (√(2n)) and mix it with f:
          f ← (1‑t)·f + t·b      (t starts at 0.5 and is reduced if the mix
                                 does not improve the score).
    * Evaluate the new f; keep the best overall sequence.
4.  Continue the outer search while time permits (budget ≈ 1000 s).
   The best sequence found is returned.

The LP is solved with SciPy’s “highs’’ method which is fast for the
moderate matrix sizes (n ≤ 800).  All values are clipped to the
[0,1000] interval required by evaluate_sequence().

def _clip_and_rescale(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale so that the L1‑norm equals target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s == 0:
        return seq
    factor = target_sum / s
    seq = seq * factor
    # clipping may have changed the sum; iterate a few times to stay inside bounds
    for _ in range(5):
        s = np.sum(seq)
        if np.isclose(s, target_sum, rtol=1e-9, atol=1e-12):
            break
        factor = target_sum / s
        seq = seq * factor
        seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  (f * b)_k ≤ rhs   for k = 0 … 2·n‑2
                    b_j ≥ 0
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    # objective: maximise sum(b)  ⇒ minimise –sum(b)
    c = -np.ones(n)

    # build the convolution matrix A_ub * b ≤ rhs, one row per shift k
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]          # (f * b)_k = Σ_i f_i * b_{k-i}
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs, dtype=float)

    # non‑negativity of b (b ≥ 0) is expressed as –I·b ≤ 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_) -> list[float]:
    """Search for a sequence with a low evaluate_sequence() value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # reserve a little margin

    best_seq: list[float] = []
    best_score = float('inf')

    # a few different lengths – larger n gives a better “continuous’’
    # behaviour but also larger LPs; 400‑800 is a good trade‑off.
    lengths = [300, 500, 800]

    for n in lengths:
        if time.time() > deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------
        # 1) generate a family of initial sequences
        # ------------------------------------------------------------
        candidates = []

        # uniform
        candidates.append(np.full(n, target_sum / n))

        # geometric families (c = ε·n)
        for c in [0.6, 0.8, 1.0, 1.2, 1.5, 2.0]:
            eps = c / n
            r = 1.0 - eps
            seq = r ** np.arange(n, dtype=float)
            candidates.append(_clip_and_rescale(seq, target_sum))

        # power‑law families
        for p in [0.5, 1.0]:
            seq = (np.arange(1, n + 1, dtype=float) ** (-p))
            candidates.append(_clip_and_rescale(seq, target_sum))

        # half‑sine shape (positive on [0,π])
        seq = np.sin(np.pi * (np.arange(1, n + 1, dtype=float) / (n + 1)))
        candidates.append(_clip_and_rescale(seq, target_sum))

        # a few random vectors (helps escaping local minima)
        for _ in range(3):
            seq = np.random.rand(n)
            candidates.append(_clip_and_rescale(seq, target_sum))

        # ------------------------------------------------------------
        # 2) evaluate each start and launch a short LP‑refinement loop
        # ------------------------------------------------------------
        for start_idx, f in enumerate(candidates):
            if time.time() > deadline:
                break

            cur = f.copy()
            cur_score = float('inf')
            # initial evaluation (no clipping needed – the sequence already respects the limits)
            try:
                cur_score = evaluate_sequence(cur.tolist())
            except Exception:
                continue

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.tolist()
                print(f"[Init] n={n}, candidate={start_idx}, score={cur_score:.6f}")

            # simple schedule for the mixing parameter
            t = 0.5

            # up to ~15 refinement steps (adjustable)
            for it in range(15):
                if time.time() > deadline:
                    break

                # current bound M = max(conv(cur,cur))
                conv_cur = np.convolve(cur, cur)
                M = conv_cur.max()

                # solve LP to get a direction that can increase the total mass
                b = _solve_convolution_lp(cur, M)
                if b is None:
                    # LP failed – abort this candidate
                    break

                # rescale b to the required L1 norm
                b = _clip_and_rescale(b, target_sum)

                # mix – convex combination keeps the bound non‑increasing in practice
                new = (1.0 - t) * cur + t * b
                new = np.clip(new, 0.0, 1000.0)  # safety clipping

                # evaluate the new point
                try:
                    new_score = evaluate_sequence(new.tolist())
                except Exception:
                    # any numerical mishap – stop refining this candidate
                    break

                # keep the best sequence ever seen
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new.tolist()
                    print(f"[Update] n={n}, cand={start_idx}, it={it}, t={t:.3f}, score={new_score:.6f}")

                # accept the step if it improves (or does not worsen) the local score
                if new_score <= cur_score:
                    cur = new
                    cur_score = new_score
                else:
                    # step was not good – shrink the mixing factor
                    t *= 0.5
                    if t < 0.02:
                        # too small to make progress; quit the refinement
                        break

                # early exit if we have already passed the target threshold
                if best_score < 1.503:
                    break

            # early exit from outer loops when the target is reached
            if best_score < 1.503:
                break

        if best_score < 1.503:
            break

    # sanity: ensure the returned list contains only non‑negative numbers
    best_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[Result] best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
