Simple heuristic to lower the self‑convolution supremum.

The method starts from the best known sequence (if available) or a random
initialisation.  In each iteration we locate the index `k` where the
self‑convolution `a*a` attains its maximum.  Among the pairs `(i, j)` with
`i+j=k` we pick the pair that contributes most to that peak and move a small
fraction of mass from the larger member of the pair to the currently smallest
entry of the sequence.  This locally reduces the dominant peak while keeping
the total `ℓ¹`‑norm unchanged.  The move is accepted whenever the
`evaluate_sequence` score improves (or with a tiny Metropolis probability
to escape plateaux).  The algorithm runs until the allotted time expires and
returns the best sequence found.

The approach is deliberately lightweight, avoids solving large LPs and can be
run for many thousands of cheap iterations within the 1000 s budget.
"""

import numpy as np
import time

def _conv(seq: np.ndarray) -> np.ndarray:
    """Return the linear convolution of `seq` with itself."""
    # Use FFT for speed on larger arrays
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[: 2 * n - 1]
    # Numerical errors may produce tiny negative values – clip them
    conv = np.clip(conv, 0.0, None)
    return conv


def _max_convolution_index(seq: np.ndarray):
    """Return (max_index, max_value, convolution_array) for `seq`."""
    conv = _conv(seq)
    idx = int(np.argmax(conv))
    return idx, float(conv[idx]), conv


def _move_mass(seq: np.ndarray, i_src: int, i_dst: int, amount: float) -> np.ndarray:
    """Move `amount` from `i_src` to `i_dst` (both indices inside `seq`)."""
    new_seq = seq.copy()
    # Ensure we never make a negative entry
    amount = min(amount, new_seq[i_src])
    new_seq[i_src] -= amount
    new_seq[i_dst] += amount
    return new_seq

    idx = int(np.argmax(conv))
    return idx, float(conv[idx]), conv


def _move_mass(seq: np.ndarray, i_src: int, i_dst: int, amount: float) -> np.ndarray:
    """Return a copy of `seq` where `amount` is moved from i_src to i_dst."""
    new_seq = seq.copy()
    # Never make a negative entry.
    amount = min(amount, new_seq[i_src])
    new_seq[i_src] -= amount
    new_seq[i_dst] += amount
    return new_seq


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a step‑function height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    kwargs
        Ignored – present for compatibility.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + float(budget_s) - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise a starting sequence.
    # ------------------------------------------------------------------
    try:
        # A previously discovered good solution may be available as a global.
        init = np.array(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise NameError
    except Exception:
        # No good seed – fall back to a random normalised vector.
        n = np.random.randint(500, 2000)          # moderate length for speed
        init = np.random.random(n)
        # Normalise to the usual ℓ¹‑norm (the objective is scale‑invariant).
        init = init / np.sum(init) * np.sqrt(2.0 * n)

    # Enforce non‑negativity and the evaluator's hard caps.
    init = np.clip(init, 0.0, 1000.0)

    seq = init.copy()
    best_seq = seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0          # counter for stagnation detection
    move_frac = 0.05        # fraction of a donor entry that we relocate

    while time.time() < deadline:
        it += 1

        # 1. Locate the dominant peak of the self‑convolution.
        max_k, max_val, conv = _max_convolution_index(seq)

        # 2. Determine the pair (i, j) with i+j = max_k that contributes most.
        n = seq.size
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = max_k - i_vals
        contrib = seq[i_vals] * seq[j_vals]
        pair_idx = int(np.argmax(contrib))
        i_src = int(i_vals[pair_idx])
        j_src = int(j_vals[pair_idx])

        # 3. Choose a destination index that currently holds the least mass.
        dst = int(np.argmin(seq))

        # 4. Pick the larger contributor as the donor.
        donor = i_src if seq[i_src] >= seq[j_src] else j_src

        # Ensure donor and destination differ.
        if donor == dst:
            candidates = np.where(np.arange(n) != donor)[0]
            dst = int(np.random.choice(candidates))

        # 5. Amount to relocate – a modest fraction of the donor's value.
        amount = move_frac * seq[donor]
        if amount < 1e-12:                     # fallback to a tiny random tweak
            i_rand = np.random.randint(n)
            j_rand = np.random.randint(n)
            if i_rand == j_rand:
                continue
            amount = move_frac * seq[i_rand]
            amount = min(amount, seq[i_rand])
            cand_seq = _move_mass(seq, i_rand, j_rand, amount)
        else:
            cand_seq = _move_mass(seq, donor, dst, amount)

        # 6. Evaluate the candidate.
        try:
            cand_score = evaluate_sequence(cand_seq.tolist())
        except Exception:
            continue

        # 7. Accept if the score improves; otherwise occasionally accept a worse move.
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            seq = cand_seq
            no_improve = 0
            print(f"Iter {it:06d} – new best score: {best_score:.6f}")
            # If we are making progress we can be slightly more aggressive.
            move_frac = min(0.2, move_frac * 1.02)
        else:
            # Simple Metropolis acceptance with a slowly cooling temperature.
            temperature = max(1e-4, 0.1 * (1.0 - it / 1e6))
            if np.random.rand() < np.exp((best_score - cand_score) / temperature):
                seq = cand_seq
            no_improve += 1
            # Reduce step size gradually if we see no improvement.
            if no_improve % 500 == 0:
                move_frac = max(0.01, move_frac * 0.9)

        # 8. Restart from a fresh random point after prolonged stagnation.
        if no_improve > 5000:
            seq = np.random.random(n)
            seq = np.clip(seq, 0.0, 1000.0)
            total = np.sum(seq)
            if total > 0:
                # Keep overall ℓ¹‑norm comparable to the best sequence seen.
                scale = np.sum(best_seq) / total
                seq *= scale
            no_improve = 0
            print(f"Iter {it:06d} – restart after stagnation")

    print(f"Finished after {it} iterations, best score found: {best_score:.6f}")
    return best_seq.tolist()
