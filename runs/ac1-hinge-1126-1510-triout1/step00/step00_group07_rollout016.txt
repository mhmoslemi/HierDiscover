def max_convolution(seq):
    n = len(seq)
    # Compute convolution using FFT
    size = 1 << ((2*n - 1).bit_length())
    # Use next power of two for speed
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)
    conv = conv[:2*n-1]  # trim
    return np.max(conv.real)

def random_move(seq):
    n = len(seq)
    i, j = np.random.randint(n, size=2)
    if i==j: return seq
    amt = seq[i] * np.random.uniform(0, 0.2)  # Move up to 20% of i's mass
    new_seq = seq.copy()
    new_seq[i] -= amt
    new_seq[j] += amt
    return new_seq
```

def project_to_simplex(v):
    # Assumes v >= 0 after clipping earlier
    # We want to find w = max(v + tau, 0) such that sum(w) = 1
    # Actually we want to preserve positivity automatically.
    # Use algorithm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate sequence that improves the evaluation score.

The algorithm combines three complementary ideas:
1. **Random initialization** – start from a Dirichlet distribution (uniform on the
   simplex) and optionally from the known state‑of‑the‑art sequence.
2. **Projected sub‑gradient descent** – repeatedly locate the convolution lag that
   attains the maximal value and move mass away from the indices that contribute
   most to this lag.  After each step the vector is projected back onto the
   probability simplex (non‑negative and sum‑to‑one), which keeps the objective
   scale‑invariant.
3. **LP refinement** – every few iterations a linear program is solved that
   maximises the total mass under the current maximal convolution bound.
   The resulting direction is mixed with the current sequence, testing several
   mixture coefficients to obtain the best improvement.

The loop runs until the allotted time budget expires (default 1000 s).  The best
sequence found during the search is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_simplex(v):
    """Project a non‑negative vector onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(seq):
    """Return both the maximal convolution value and its index."""
    # use FFT‑based convolution for speed (zero‑padding to power of two)
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    f_hat = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f_hat * f_hat, size)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx


def _evaluate_score(seq):
    """Direct implementation of evaluate_sequence (fast, no clamping)."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    max_b, _ = _max_convolution(seq)
    return float(2.0 * n * max_b / (s * s))


def _solve_lp(f_seq, rhs):
    """
    Solve the linear program:
        maximise   sum(g)
        subject to  (f * g)[k] <= rhs   for all k,
                    g >= 0.
    Returns the optimal g as a NumPy array or None on failure.
    """
    n = f_seq.shape[0]
    # Objective: maximise sum(g)  -> minimise -sum(g)
    c = -np.ones(n)

    # Build the convolution constraints A_ub @ g <= b_ub
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # For i in [i_min, i_max] we have term f[i] * g[k-i]
        # This contributes coefficient f[i] to column (k-i)
        # Fill row accordingly
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], rhs)

    # Non‑negativity constraints  -g <= 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # Solve with the high‑performance HiGHS interior‑point solver
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None

    except Exception:
        base_len = 200
    n = max(100, min(base_len, 400))   # between 100 and 400 entries

    # ------------------------------------------------------------------
    # 2. Initial candidate (random Dirichlet) – always feasible.
    # ------------------------------------------------------------------
    cur = np.random.rand(n)
    cur = cur / cur.sum()
    best_seq = cur.copy()
    best_score = _evaluate_score(best_seq)

    # Some mixing parameters
    mix_ts = [0.01, 0.05, 0.1, 0.2, 0.35, 0.5]
    eta = 0.05          # step size for sub‑gradient updates
    it = 0

    print(f"Starting search (n={n}), initial score = {best_score:.6f}")

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 3. Sub‑gradient descent step targeting the current maximal lag
        # --------------------------------------------------------------
        max_b, max_k = _max_convolution(cur)
        # gradient of the selected lag (2 * a_{k-i} when index valid)
        grad = np.zeros_like(cur)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * cur[max_k - idx]

        # take a step away from the gradient, keep non‑negative, renormalise
        nxt = cur - eta * grad
        nxt = np.maximum(nxt, 0.0)
        if nxt.sum() == 0.0:
            nxt = np.random.rand(n)
        nxt = nxt / nxt.sum()

        score_nxt = _evaluate_score(nxt)
        if score_nxt < best_score:
            best_score = score_nxt
            best_seq = nxt.copy()
            print(f"[{it:05d}] Gradient step improved score to {best_score:.6f}")

        # simple acceptance rule (always move)
        cur = nxt

        # --------------------------------------------------------------
        # 4. Every few iterations perform an LP refinement + line search
        # --------------------------------------------------------------
        if it % 12 == 0:
            # Normalise the current vector (sum = 1) for the LP
            f = cur.copy()
            rhs = max_b
            g = _solve_lp(f, rhs)
            if g is not None:
                # Normalise g to sum = 1 (scaling does not affect the bound)
                g = g / g.sum()
                # Try a handful of mixing coefficients and keep the best one
                local_best = cur
                local_score = best_score
                for t in mix_ts:
                    mixed = (1.0 - t) * cur + t * g
                    mixed = mixed / mixed.sum()
                    sc = _evaluate_score(mixed)
                    if sc < local_score:
                        local_score = sc
                        local_best = mixed
                if local_score < best_score:
                    best_score = local_score
                    best_seq = local_best.copy()
                    cur = local_best
                    print(
                        f"[{it:05d}] LP mix improved score to {best_score:.6f}"
                    )
        # --------------------------------------------------------------
        # 5. Occasionally inject a small random perturbation to escape plateaus
        # --------------------------------------------------------------
        if it % 31 == 0 and np.random.rand() < 0.2:
            # move a random fraction (up to 10 %) from one index to another
            i, j = np.random.choice(n, 2, replace=False)
            delta = cur[i] * np.random.uniform(0.0, 0.1)
            cur[i] -= delta
            cur[j] += delta
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
        # --------------------------------------------------------------
        # 6. Decay the step size slowly
        # --------------------------------------------------------------
        eta *= 0.9999
        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    print(f"Search finished after {it} iterations, best score = {best_score:.6f}")
    # Convert to regular Python floats to avoid any NumPy scalar issues
    return [float(x) for x in best_seq]
