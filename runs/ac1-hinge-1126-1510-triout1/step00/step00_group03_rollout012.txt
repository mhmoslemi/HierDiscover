        grad_y = a * (g - mean_g)
        return f, grad_y

    while time.time() < deadline:

        y0 = np.random.randn(n)  # normal init
        # optionally start from best refined y if we saved
        # minimize

                       jac=True,

    print(f'Best score after search: {best_score}')

def hill_climb(seq, best_score, deadline):
    seq = seq.copy()
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving delta
        delta = seq[i] * np.random.rand() * 0.1  # up to 10% of i's mass
        if delta > seq[i]:
            delta = seq[i]
        seq[i] -= delta
        seq[j] += delta
        new_score = evaluate_sequence(seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = seq.copy()
        else:
            # revert
            seq[i] += delta
            seq[j] -= delta
    return best_seq, best_score
```

    # define softmax & conv functions.
    def softmax(y):
        m = np.max(y)
        e = np.exp(y - m)
        return e / e.sum()

    def conv_fft(a):
        m = 1 << ((2*n - 1 - 1).bit_length())
        fa = np.fft.fft(a, m)
        conv = np.real(np.fft.ifft(fa * fa))
        return conv[:2*n-1]

    def fast_score(a):
        conv = conv_fft(a)
        return 2.0 * n * conv.max()

    def objective_and_grad(y):
        a = softmax(y)
        # conv
        m = 1 << ((2*n - 1 - 1).bit_length())
        fa = np.fft.fft(a, m)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        lam = max_c * 0.02 + 1e-12
        exp_shift = np.exp((conv - max_c) / lam)
        Z = exp_shift.sum()
        smooth_max = max_c + lam * np.log(Z)
        f = 2.0 * n * smooth_max
        # weights w
        w = exp_shift / Z
        # compute g

        g = conv_rev[(n-1):(n-1+n)]
        mean_g = np.dot(a, g)
        grad = a * (g - mean_g)
        return f, grad

    best_a = np.full(n, 1.0 / n)

"""
Propose a low‐evaluation step sequence.

The algorithm works on the simplex (sum = 1, non‑negative) by
optimising a smooth approximation of the maximum convolution value.
The variables are unconstrained real numbers y and the sequence a is
obtained by a soft‑max mapping a = softmax(y).  The objective

    f(y) = 2 * n * smooth_max( a * a )

with smooth_max(t) = λ·log Σ exp(t_i/λ) (λ > 0) is differentiable.
Its gradient with respect to y can be expressed analytically using
convolutions that are evaluated by FFTs, so each iteration is O(n log n).

Multiple random restarts are performed within the allotted time.
After each L‑BFGS run a short greedy hill‑climb (mass transfer between
two positions) refines the best candidate.  The final sequence is
returned as a plain list of floats; it will be evaluated by the
provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Circular convolution of a with itself using FFT,
    returning the linear convolution of length 2*n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2*n*max(conv) for a sequence that already sums to 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective and its gradient w.r.t. y.

    Returns
    -------
    f : float
        Smooth objective value (2*n*smooth_max).
    g : np.ndarray
        Gradient w.r.t. y (shape n).
    """
    # Transform to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12          # smoothing width
    # stable soft‑max over the convolution entries
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    # weights w_i = ∂smooth_max/∂conv_i
    w = exp_shift / Z

    # g_i = Σ_j a_j * w_{i+j}   (i = 0..n‑1)
    # This is the (a reversed) * w convolution.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice the needed part: indices (n‑1) … (n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climb.
    At each step move a random fraction of one coordinate to another.
    The sum stays exactly 1, so the score can be evaluated with the fast
    O(n log n) convolution.
    """
    best = a.copy()
    best_score = current_score

    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a safety margin

    # problem size – 2000 gives a good trade‑off between quality and speed
    n = 2000

    # pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # start with the uniform distribution (the known baseline)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    iteration = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline:
        # ----------- pick a random starting point for L‑BFGS -------------
        mode = rng.choice(['random', 'exp', 'power', 'edge'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # edge‑heavy heuristic
            # number of heavy edge entries (1 % – 3 % of n)
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ----------- run L‑BFGS on the smooth objective ---------------
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, 0.30),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-10,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  improved fast score = {best_score:.6f}")

        # ----------- local greedy refinement (hill climbing) ----------
        # use whatever time is left, but keep a modest number of moves
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=2000)

        iteration += 1

    # final verification with the official evaluator
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
