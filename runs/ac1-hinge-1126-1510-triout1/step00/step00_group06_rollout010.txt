def conv_max_fft(v):
    n = v.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(A*A, n=size)[:2*n-1]
    max_val = conv.max()
    idx = conv.argmax()
    return max_val, idx

def approx_score(v):
    n = v.shape[0]
    M, _ = conv_max_fft(v)
    # sum may not be 1, adjust
    s = v.sum()
    if s <= 0:
        return np.inf
    return float(2 * n * M / (s**2))

def gradient_step(v, k, lr):

"""
Propose a step‐function (non‑negative list) that minimizes the
`evaluate_sequence` objective.

The idea:
* Treat the list as a probability distribution (its sum is normalised to 1).
  For a distribution `p`, the convolution `p*p` is the distribution of the sum
  of two independent draws.  The objective is proportional to the maximum
  entry of this convolution, i.e. we want a distribution whose self‑convolution
  is as flat as possible.
* Start from a variety of simple parametric families (uniform, power‑law,
  exponential, edge‑heavy “piecewise”, randomly smoothed vectors).  After
  normalisation we evaluate the objective using an FFT based convolution
  (much faster than `np.convolve`).
* For each candidate we perform a few gradient‑descent refinements.
  The gradient of the maximal convolution entry w.r.t. the vector is known:
  for the index `k` where the maximum occurs,
      ∂(conv_k)/∂a_i = 2·a_{k‑i}  (if the partner index lies in range).
  A step `a ← a – η·gradient` reduces that maximal entry; after the step the
  vector is projected back onto the simplex (non‑negative, summed to 1).
* After the initial sampling we continue a longer refinement loop that keeps
  improving the best vector found, occasionally applying extra refinement,
  random perturbations, or a fresh random restart if progress stalls.
* Finally the best vector is returned as a plain Python list.  The returned
  sequence is evaluated with the official `evaluate_sequence` for the final
  score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def normalize_vec(v: np.ndarray) -> np.ndarray:
    """Project vector onto the probability simplex (non‑negative, sum = 1)."""
    s = v.sum()
    if s == 0.0:
        return v
    return v / s


def conv_max_fft(v: np.ndarray):
    """
    Return (max_value, argmax_index) of the linear convolution v*v.
    Computed via FFT for speed.
    """
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def approx_score(v: np.ndarray) -> float:
    """
    Approximate `evaluate_sequence` when the vector is already normalised
    to sum = 1.  The exact objective is
        2 * n * max(conv) / (sum(v)^2)
    """
    n = v.shape[0]
    max_val, _ = conv_max_fft(v)
    # sum(v) == 1 after normalisation, therefore the denominator is 1.
    return float(2.0 * n * max_val)


def gradient_step(v: np.ndarray, k: int, lr: float) -> np.ndarray:
    """
    Perform one projected gradient step that tries to reduce the convolution
    entry at index `k`.  Gradient of conv_k w.r.t. v_i is
        2 * v_{k-i}  (if the partner index is inside the array, otherwise 0).
    After the step the vector is projected back onto the simplex.
    """
    n = v.shape[0]
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(v)
    grad[mask] = 2.0 * v[partner[mask]]
    new_v = v - lr * grad
    new_v = np.maximum(new_v, 0.0)
    return normalize_vec(new_v)


def refine(v: np.ndarray, steps: int = 20, lr_start: float = 0.1,
           lr_decay: float = 0.9) -> np.ndarray:
    """
    Apply a short sequence of gradient steps (with decreasing learning rate)
    to the vector `v`.
    """
    cur = v.copy()
    for t in range(steps):
        _, k = conv_max_fft(cur)
        lr = lr_start * (lr_decay ** t)
        cur = gradient_step(cur, k, lr)
    return cur

# Main search routine required by the platform
# ----------------------------------------------------------------------

    """
    Search for a good step‑function within the given time budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a safety margin

    # Length of the sequence – use the length of the provided construction
    # if it exists, otherwise a large but manageable size.
    try:
        n = len(height_sequence_1)                # type: ignore[name-defined]
    except Exception:
        n = 5600
    if n > 8000:                                 # avoid excessively long runs
        n = 8000

    best_seq = None
    best_score = np.inf

    shape_pool = ["uniform", "power", "expo", "piecewise", "mix"]

    # ------------------------------------------------------------------
    # 1.  Random initial families + quick refinements
    # ------------------------------------------------------------------
    for trial in range(300):
        if time.time() > deadline:
            break

        shape = np.random.choice(shape_pool)

        if shape == "uniform":
            a = np.ones(n)

        elif shape == "power":
            alpha = np.random.uniform(0.0, 3.0)
            i = np.arange(n)
            a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)

        elif shape == "expo":
            beta = np.random.uniform(0.0, 5.0)
            i = np.arange(n)
            a = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))

        elif shape == "piecewise":

            edge_len = max(1, int(edge_frac * n))

        if np.random.rand() < 0.005:

            delta = np.random.rand() * best_seq[i] * 0.2
