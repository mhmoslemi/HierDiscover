"""
Propose a candidate sequence that beats the uniform (value = 2.0) using the
half‑order generating function (central binomial coefficients / 4**i).
The series a_i = C(2i,i)/4**i satisfies
    (∑ a_i z^i)^2 = 1/(1−z)   (up to degree n−1)
so the first n entries of the self‑convolution are exactly constant.
After normalising the sum to 1 this gives a bound ≈ π/2 ≈ 1.57,
already better than the previous upper bound of 2.0.
A simple simulated‑annealing loop then tries to flatten the remaining
peaks by random perturbations and occasional gradient steps.
The function always returns a valid non‑negative sequence before the
timeout expires.
"""

import time
import math
import numpy as np


def _half_order_series(n: int) -> np.ndarray:
    """
    Generate the first n coefficients of the series
        a_i = C(2i,i) / 4**i
    using a stable recurrence to avoid huge integers.
    The result is normalised so that its sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence: a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    # normalise to sum = 1 (the optimiser works with a fixed sum)
    a /= a.sum()
    return a


def _gradient_step(seq: np.ndarray, max_index: int, step: float) -> np.ndarray:
    """
    Perform a single gradient–descent step aiming to reduce the
    convolution value at `max_index`.  The gradient of
        c_k = sum_{i+j=k} a_i a_j
    w.r.t. a_i is 2*a_{k-i} (when the index is in range).
    """
    n = len(seq)
    grad = np.zeros_like(seq)
    # fill gradient where the pair contributes to the max‑index
    # grad[i] = 2 * a_{k-i}  if 0 <= k-i < n
    lo = max(0, max_index - (n - 1))
    hi = min(n - 1, max_index)
    # vectorised version
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = max_index - idx
    grad[idx] = 2.0 * seq[rev]
    # take a small step, keep non‑negative, then renorm
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    if new_seq.sum() == 0.0:
        return seq
    new_seq /= new_seq.sum()
    return new_seq


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a sequence of non‑negative heights that minimises
        2 * n * max(conv(seq, seq)) / (sum(seq)^2)

    The algorithm:
    1. Initialise with the half‑order series (length ≈ 1000).
    2. Keep the best sequence seen so far.
    3. Run a cheap simulated‑annealing loop:
        – random multiplicative noise on the whole vector,
        – occasional gradient step on the currently maximal convolution entry.
    4. Return the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin

    # ------------------------------------------------------------------
    # 1) initial construction
    # ------------------------------------------------------------------
    n = 1000                     # length – a trade‑off between quality & speed
    seq = _half_order_series(n)  # shape that already gives ~1.57 bound

    # evaluate the starter
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    curr_seq = seq.copy()
    curr_score = best_score

    # parameters for the annealing loop
    initial_temp = 0.5
    final_temp = 1e-5
    total_runtime = budget_s - 2.0
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # linear progress (0 … 1) for temperature scheduling
        progress = (time.time() - start_time) / total_runtime
        temp = initial_temp * (final_temp / initial_temp) ** progress

        # ------------------------------------------------------------------
        # 2) random multiplicative perturbation (local search)
        # ------------------------------------------------------------------
        sigma = 0.02 * (1.0 - progress) + 0.001   # shrink noise slowly
        noise = np.random.randn(n) * sigma
        cand = curr_seq * (1.0 + noise)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_score = evaluate_sequence(cand.tolist())

        # Metropolis acceptance (we minimise)
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            # small probability to escape plateaus
            prob = math.exp(-(cand_score - curr_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            curr_seq = cand
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{iteration:6d}] New best (random) score: {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) occasional gradient step on the current maximal entry
        # ------------------------------------------------------------------
        if iteration % 250 == 0:
            conv = np.convolve(best_seq, best_seq)
            max_val = conv.max()
            # pick one of the (possibly several) maximal indices at random
            max_idxs = np.where(conv == max_val)[0]
            k = np.random.choice(max_idxs)

            # take a modest step size – tuned empirically
            step_size = 0.005
            cand = _gradient_step(best_seq, k, step_size)

            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                curr_seq = cand
                curr_score = cand_score
                print(f"[{iteration:6d}] Gradient improvement score: {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    return best_seq.tolist()
