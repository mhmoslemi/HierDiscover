def project_simplex(v):
    # Project onto simplex {x>=0, sum x = 1}
    # Use technique from https://arxiv.org/pdf/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n - 1]
    return conv.max()

"""
Propose a candidate step function sequence.

This implementation searches for a non‑negative sequence `a`
which minimises the evaluation metric

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution.  Since the metric is
scale‑invariant we fix `sum(a)=1` and minimise `max(conv(a,a))`.

The search combines three ideas:
  * A cheap estimate of `max(conv)` via FFT.
  * Projected sub‑gradient descent on the max‑convolution term.
  * Simulated‑annealing style acceptance of occasional worsening moves
    and random perturbations to escape local minima.

The algorithm runs until the supplied time budget expires and returns the
best sequence found.
"""

import time
import numpy as np
from scipy import optimize  # only for the optional LP fallback (not used here)

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------
def _fft_conv_max(a: np.ndarray):
    """
    Compute the maximal value of the linear convolution a*a
    using a real FFT (fast for moderate n).

    Returns (max_value, max_index, full_convolution_array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the function max_k conv_k(a) at the active index k.
    For each i :  ∂conv_k/∂a_i = a_i + a_{k-i} (if the second term exists).
    """
    n = a.shape[0]
    g = a.copy()                     # contribution a_i
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    for i in range(low, high + 1):
        j = k - i
        g[i] += a[j]                 # contribution a_{k-i}
    return g


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i >= 0 , sum_i x_i = 1 }.
    Implementation follows the algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The routine stops a few seconds
        before the deadline to allow safe termination.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0.01).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – use the length of the known construction if available,
    # otherwise fall back to a moderately large size.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the evaluation harness
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = int(init_seq.shape[0])
        a = np.maximum(init_seq, 0.0)
        if a.sum() < 1e-9:                # degenerate fallback
            a = np.random.rand(n)
    except Exception:
        # No external sequence – pick a size that gives a good trade‑off
        n = 2048
        a = np.random.rand(n)

    # Normalise to a probability simplex (sum = 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.random.rand(n)
        a_sum = a.sum()
    a = a / a_sum

    # Best solution tracking
    max_val, max_idx, _ = _fft_conv_max(a)
    best_a = a.copy()
    best_score = 2.0 * n * max_val           # evaluation with sum == 1
    cur_a = a.copy()
    cur_score = best_score

    # Hyper‑parameters
    init_eta = 0.20          # initial step size for sub‑gradient descent
    eta = init_eta
    temp = 1.0               # temperature for occasional uphill moves
    temp_decay = 0.9995
    iter_cnt = 0
    report_interval = 10.0   # seconds between progress prints
    next_report = time.time() + report_interval

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Evaluate current max‑convolution and its active index
        # --------------------------------------------------------------
        max_val, max_idx, _ = _fft_conv_max(cur_a)

        # --------------------------------------------------------------
        # 2) Compute sub‑gradient for the active index and take a step
        # --------------------------------------------------------------
        g = _subgradient_max(cur_a, max_idx)
        # Adaptive step size – slowly decay to refine the solution
        eta = init_eta * (0.9995 ** iter_cnt)

        # Gradient descent step (project back onto simplex)
        proposal = cur_a - eta * g
        proposal = np.maximum(proposal, 0.0)
        if proposal.sum() == 0.0:
            # degenerate – re‑initialise a tiny random perturbation
            proposal = np.random.rand(n) * 1e-6 + cur_a * 0.99
        proposal = _project_to_simplex(proposal)

        # --------------------------------------------------------------
        # 3) Evaluate the proposal
        # --------------------------------------------------------------
        new_max, _, _ = _fft_conv_max(proposal)
        new_score = 2.0 * n * new_max

        # Keep track of the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_a = proposal.copy()
            print(f"[{time.time():.2f}] New best score: {best_score:.6f} "
                  f"(iteration {iter_cnt})")

        # --------------------------------------------------------------
        # 4) Acceptance rule – always accept improving moves,
        #    occasionally accept worsening ones (Simulated Annealing)
        # --------------------------------------------------------------
        if new_score <= cur_score:
            cur_a = proposal
            cur_score = new_score
        else:
            # uphill acceptance probability
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                cur_a = proposal
                cur_score = new_score
            # otherwise keep the old state

        # --------------------------------------------------------------
        # 5) Temperature decay and occasional random perturbation
        # --------------------------------------------------------------
        temp *= temp_decay

        if iter_cnt % 500 == 0 and iter_cnt > 0:
            # Random jitter to help escape plateaus
            jitter = (np.random.rand(n) - 0.5) * 0.02
            cur_a = cur_a + jitter
            cur_a = np.maximum(cur_a, 0.0)
            cur_a = _project_to_simplex(cur_a)

        # --------------------------------------------------------------
        # 6) Periodic progress report
        # --------------------------------------------------------------
        if time.time() >= next_report:
            print(f"[{time.time():.2f}] Iter {iter_cnt:6d} | "
                  f"best={best_score:.6f} | cur={cur_score:.6f} | "
                  f"temp={temp:.6f}")
            next_report = time.time() + report_interval

        iter_cnt += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Search finished after {iter_cnt} iterations. "
          f"Best score = {best_score:.6f}")
    return best_a.tolist()
