def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    b = np.convolve(a, a)  # length 2n -1
    # compute softmax weights
    maxb = np.max(b)  # for numerical stability, subtract max

    grad = conv_rev[n-1: n-1 + len(a)]   # slice of length n
    grad = 2 * grad
    return grad

        # find pair (i, j) maximizing a_i * a_j under i+j=k_max

        i_end = min(k_max, len(a)-1)

        prod = a[i_candidates] * a[j_candidates]

        j_star = j_candidates[idx]

def linear_convolve(a, b):

    size = 1 << (n - 1).bit_length()

    fb = np.fft.rfft(b, size)

"""
Propose a sequence of non‑negative heights that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a simple simulated‑annealing hill climber.  The algorithm
  * starts from a random Dirichlet vector (sum = 1),
  * repeatedly identifies the convolution index k that attains the current
    maximum,
  * moves a small amount of probability mass from the pair (i, j) that
    contributes most to that peak to two low‑density locations,
  * accepts the move if it improves the score or with a temperature‑dependent
    probability (to escape local minima),
  * periodically adds a little random noise,
  * runs until the allocated time expires and returns the best sequence found.

All arithmetic stays inside the allowed range, the final sequence is projected
onto the probability simplex (non‑negative, sum = 1) so that `evaluate_sequence`
produces the same value as the internal fast score computation.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def linear_convolve(a, b):
    """Linear convolution via FFT (returns length len(a)+len(b)-1)."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    conv = np.fft.irfft(fc, size)[:n]
    return conv


def hill_climb(initial_a, time_budget):
    """
    Simulated‑annealing hill climbing.
    Returns the best vector found and its score (2*n*max(conv)).
    """
    a = initial_a.copy()
    n = a.shape[0]

    # pre‑computed FFT size (used inside linear_convolve)
    # (linear_convolve recomputes the size each call – cheap enough)

    # current best
    b = linear_convolve(a, a)
    best_score = 2.0 * n * np.max(b)
    best_a = a.copy()

    start = time.time()
    iteration = 0

    # main loop
    while time.time() - start < time_budget:
        iteration += 1

        # 1) convolution and the index of the current peak
        b = linear_convolve(a, a)
        k_max = int(np.argmax(b))

        # 2) locate the pair (i, j) that contributes most to this peak
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k_max - i_vals
        products = a[i_vals] * a[j_vals]
        pair_idx = int(np.argmax(products))
        i_star = int(i_vals[pair_idx])
        j_star = int(j_vals[pair_idx])

        # 3) choose two low‑density positions that will receive the transferred mass
        low_thr = np.percentile(a, 20)               # bottom 20 %
        low_idx = np.where(a <= low_thr)[0]
        if low_idx.size >= 2:
            i_target, j_target = np.random.choice(low_idx, 2, replace=False)
        else:
            # fall back to random distinct indices
            i_target, j_target = np.random.choice(n, 2, replace=False)

        # 4) amount of probability to move (a small fraction of the smaller of the two)
        if i_star == j_star:
            delta = 0.05 * a[i_star]          # 5 % of the single entry
        else:
            delta = 0.05 * min(a[i_star], a[j_star])
        if delta <= 1e-12:
            continue

        # 5) build the candidate vector
        a_new = a.copy()
        a_new[i_star] -= delta
        a_new[j_star] -= delta
        # keep non‑negative
        if a_new[i_star] < 0 or a_new[j_star] < 0:
            continue
        a_new[i_target] += delta / 2.0
        a_new[j_target] += delta / 2.0

        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0:
            continue
        a_new /= total

        # 6) score of the candidate
        b_new = linear_convolve(a_new, a_new)
        cand_score = 2.0 * n * np.max(b_new)

        # 7) acceptance rule (simulated annealing)
        #    temperature decays linearly from 0.5 to 0 over ~5000 iterations
        T = max(0.0, 0.5 * (1.0 - iteration / 5000.0))
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # probability of accepting a worse move
            prob = math.exp(-(cand_score - best_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_new.copy()

        # 8) occasional random perturbation to keep diversity
        if iteration % 1000 == 0:
            noise = np.random.normal(scale=0.01, size=n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 0:
                a = a_pert / a_pert.sum()

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    current best bound.  The routine runs a fast stochastic hill climber for
    the allotted time and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # keep a safety margin

    # Choose a moderate length (large enough to give a good bound,
    # small enough to keep the optimisation fast).
    n = np.random.randint(1500, 2500)

    # Start from a random Dirichlet (exponential) vector and normalise.
    start_vec = np.random.exponential(scale=1.0, size=n)
    start_vec /= start_vec.sum()

    # Run the hill‑climber for the remaining time.
    remaining = max(0.0, deadline - time.time())
    best_vec, best_score = hill_climb(start_vec, remaining)

    # A tiny sanity check – evaluate with the provided function.
    # (We compute the score ourselves to avoid the extra convolution.)
    # print is allowed for diagnostics.
    print(f"Final length = {n}, best score ≈ {best_score:.6f}")

    # Convert to plain Python list as required.
    return best_vec.tolist()
