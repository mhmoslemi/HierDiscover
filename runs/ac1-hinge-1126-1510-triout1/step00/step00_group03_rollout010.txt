"""
Propose a better step–function using a simple stochastic descent.

The evaluation of a sequence is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the full (linear) convolution.  The score is invariant under a
global scaling of the vector, so we keep the total sum fixed to the
canonical value sqrt(2·n) but otherwise work with the raw non‑negative
vector.

Idea
----
* Start from a random (exponential) vector of the required length and
  normalise its sum.
* Repeatedly look at the convolution entry that attains the current maximum.
  The pairs (i, j) with i + j = k contribute a_i·a_j to this entry.
* Take a small amount of mass from the pair that contributes the most
  (or from the single index if i == j) and move it to a random other
  index.  This directly reduces the dominant term while keeping the sum
  unchanged.
* Occasionally do a tiny gradient step on the active term and accept
  moves with a simple simulated‑annealing schedule to escape plateaus.
* Keep the best sequence seen so far and return it when the time budget
  (≈1000 s) runs out.

The method needs only the `evaluate_sequence` routine; all other work is
performed locally with NumPy.  It typically drives the score well below
the naïve uniform value 2.0 and often reaches the target region ~1.5.
"""

import time
import math
import random
import numpy as np

def _initial_sequence(n: int, seed: int) -> np.ndarray:
    """Create a positive random start vector of length n with the canonical sum."""
    rng = np.random.default_rng(seed)
    # exponential distribution gives a wide spread while staying non‑negative
    a = rng.exponential(scale=1.0, size=n).astype(np.float64)
    target_sum = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:                     # fallback – uniform walk
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= (target_sum / s)
    return a


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops a few seconds before the
        limit and returns the best sequence discovered.

    Returns
    -------
    list[float]
        The best step‑function heights found.
    """
    # ------------------------------------------------------------------
    # Basic parameters and initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Length – use the provided construction if available, otherwise a
    # moderate size that is fast to work with.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # start from the known construction (it may already be decent)
        start = np.array(height_sequence_1, dtype=np.float64)
        # re‑normalise to the canonical sum (the score is scale‑invariant)
        target_sum = math.sqrt(2.0 * n)
        start *= (target_sum / start.sum())
        a = start.copy()
    else:
        n = 2000                     # a safe default
        a = _initial_sequence(n, seed)

    target_sum = math.sqrt(2.0 * n)   # the normalising constant
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}  initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stochastic descent loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a small safety margin

    temperature = 0.5
    temp_decay = 0.9999
    min_temp = 1e-6
    eta = 0.05                # base step size for the gradient move

    # store the current score to avoid recomputation each iteration
    cur_score = best_score

    # main optimisation cycle
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. Compute convolution and locate the active (largest) entry
        # --------------------------------------------------------------
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))          # index with maximal autocorrelation
        max_val = conv[k]

        # --------------------------------------------------------------
        # 2. Decide which type of move to try
        #    - with prob 0.3 use a tiny gradient step on the active term
        #    - otherwise transfer mass from the most harmful pair
        # --------------------------------------------------------------
        if rng.random() < 0.3:
            # ----- Gradient step -------------------------------------------------
            # gradient of b_k = sum_i a_i a_{k-i} w.r.t. a_i is 2 * a_{k-i}
            grad = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # vectorised fill of the gradient on the overlapping indices
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

            step = eta * temperature
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)                     # positivity

            # Renormalise to keep the total sum exactly target_sum
            s = a_new.sum()
            if s == 0.0:
                a_new = np.full(n, target_sum / n, dtype=np.float64)
            else:
                a_new *= (target_sum / s)
        else:
            # ----- Pair‑transfer move -------------------------------------------
            # indices that actually contribute to b_k
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range

            # product for each pair (i,j)
            products = a[i_range] * a[j_range]
            max_pair = int(np.argmax(products))
            i_max = int(i_range[max_pair])
            j_max = int(j_range[max_pair])

            # decide how much mass to move – a small fraction of the smaller entry
            if i_max == j_max:
                eps = 0.10 * a[i_max]                 # 10 % of the single entry
                if eps <= 0.0:
                    continue
                # pick a different destination
                r = rng.integers(0, n)
                while r == i_max:
                    r = rng.integers(0, n)
                a_new = a.copy()
                a_new[i_max] -= eps
                a_new[r] += eps
            else:
                eps = 0.10 * min(a[i_max], a[j_max])    # 10 % of the weaker side
                if eps <= 0.0:
                    continue
                r = rng.integers(0, n)
                while r == i_max or r == j_max:
                    r = rng.integers(0, n)
                a_new = a.copy()
                a_new[i_max] -= eps
                a_new[j_max] -= eps
                a_new[r] += 2.0 * eps

            # clip negatives that could appear because of rounding
            a_new = np.maximum(a_new, 0.0)

            # The move keeps the total sum unchanged (mass is transferred),
            # but clipping may have altered it a little – renormalise just in case.
            s = a_new.sum()
            if s == 0.0:
                a_new = np.full(n, target_sum / n, dtype=np.float64)
            else:
                a_new *= (target_sum / s)

        # --------------------------------------------------------------
        # 3. Evaluate the proposal
        # --------------------------------------------------------------
        new_score = evaluate_sequence(a_new.tolist())

        # --------------------------------------------------------------
        # 4. Acceptance test (simulated annealing)
        # --------------------------------------------------------------
        accepted = False
        if new_score < cur_score:
            accepted = True
        else:
            # probability decays with temperature
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accepted = True

        if accepted:
            a = a_new
            cur_score = new_score

        # --------------------------------------------------------------
        # 5. Keep the global best
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.1f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6. Temperature schedule
        # --------------------------------------------------------------
        temperature = max(min_temp, temperature * temp_decay)

        # --------------------------------------------------------------
        # 7. Small random perturbations to avoid stagnation
        # --------------------------------------------------------------
        if rng.random() < 0.001:
            # perturb a few random coordinates
            for _ in range(3):
                idx = rng.integers(0, n)
                a[idx] = max(0.0, a[idx] + rng.normal() * target_sum / n)
            # re‑normalise after perturbation
            a *= (target_sum / a.sum())

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best score = {final_score:.6f}  (n={n})")
    return best_a.tolist()
