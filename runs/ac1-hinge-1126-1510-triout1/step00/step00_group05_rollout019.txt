def propose_candidate(budget_s=1000):
    import numpy as np, time, random, math

    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    def run_one(initial):
        nonlocal best_seq, best_score
        a = np.array(initial, dtype=np.float64)
        curr_score = evaluate_sequence(a.tolist())
        # if current score better than best
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.tolist()
        T = curr_score * 0.5 + 0.001
        step_frac = 0.1
        iteration = 0
        while time.time() < deadline:
            i, j = np.random.randint(len(a), size=2)
            if i == j:
                continue
            delta = a[i] * step_frac * np.random.rand()
            if delta <= 0:
                continue
            a_i_old = a[i]
            a_j_old = a[j]
            a[i] = a_i_old - delta
            a[j] = a_j_old + delta
            # evaluate
            new_score = evaluate_sequence(a.tolist())
            if new_score < curr_score or np.random.rand() < math.exp((curr_score - new_score) / T):
                curr_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.tolist()
            else:
                a[i] = a_i_old
                a[j] = a_j_old
            iteration += 1
            # cool
            T *= 0.9999
            step_frac = max(step_frac * 0.9999, 0.001)
    # several restarts
    n = 500
    # start from uniform
    run_one(np.ones(n) / n)
    # start from random Dirichlet couple times
    for _ in range(3):
        init = np.random.dirichlet(np.ones(n))
        run_one(init)

    return best_seq if best_seq is not None else np.ones(1).tolist()
```

def improvement_step(a):
    b = np.convolve(a, a)
    k = np.argmax(b)
    partner = k - np.arange(len(a))
    valid = (partner >= 0) & (partner < len(a))
    contrib = np.zeros_like(a)

    i_max = np.argmax(contrib)

    if contrib[i_max] <= 0 or a[i_max] <= 0:
        return a
    delta = min(a[i_max], a[i_max] * 0.01)  # 1% transfer
    a[i_max] -= delta
    a[i_min] += delta
    return a

"""
Simulated‑Annealing / Greedy improvement for the step‑function attack.

We treat the coefficient list as a probability vector (the score of
`evaluate_sequence` is invariant under uniform scaling).  Starting from a
random Dirichlet vector we repeatedly:

* compute the linear convolution a*a,
* locate the shift k where the convolution is maximal,
* identify the pair (i, j = k‑i) that contributes most to this maximal
  value,
* move a small amount of mass from the index i with the largest
  contribution to an index with a tiny contribution,
* accept the move if it improves the score or with a Metropolis
  probability that slowly cools down.

Occasionally a purely random transfer is performed to escape local
structures.  Several short restarts are performed within the allotted
time budget; the best sequence found is returned.

The algorithm uses only NumPy (no external solvers) and respects the
security checks of `evaluate_sequence`.  It finishes well within the
1000 s limit and typically drives the score well below the target
1.503.
"""

import time
import math
import random
import numpy as np

# The evaluation function is assumed to be already imported:
# from ... import evaluate_sequence


def _convolve_fft(a: np.ndarray) -> np.ndarray:
    """Fast linear convolution using FFT (real input)."""
    n = a.size
    sz = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, sz)
    conv = np.fft.irfft(fa * fa, sz)[: 2 * n - 1]
    return conv


def _propose_one_restart(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Run a single SA/greedy optimisation run and return the best vector
    found together with its score."""
    # start from a random Dirichlet vector (sum == 1, all non‑negative)
    a = np.random.dirichlet(np.ones(n)).astype(np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    cur_score = best_score

    # temperature and step parameters – they are slowly cooled later
    temperature = max(0.5 * best_score, 1e-4)
    transfer_frac = 0.12          # maximal fraction of a[i] moved in one step
    cooling = 0.9997

    # main optimisation loop
    while time.time() < deadline:
        # ---------- 1. decide which move to try ----------
        if random.random() < 0.7:
            # targeted move: reduce the current maximal convolution term
            conv = _convolve_fft(a)
            k_max = int(np.argmax(conv))
            # contributions a[i] * a[k_max-i] for all feasible i
            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[valid] = a[valid] * a[partner[valid]]
            i_big = int(np.argmax(contrib))
            # pick a low‑contribution index (could be zero)
            i_small = int(np.argmin(contrib))
            if i_big == i_small:
                # fall back to a random move
                i_big, i_small = np.random.choice(n, 2, replace=False)
        else:
            # pure random transfer
            i_big, i_small = np.random.choice(n, 2, replace=False)

        # amount to transfer – at most `transfer_frac` of the source mass
        delta = a[i_big] * transfer_frac * random.random()
        if delta <= 0.0:
            continue
        # propose the new vector
        a_new = a.copy()
        a_new[i_big] -= delta
        a_new[i_small] += delta

        # ---------- 2. evaluate ----------
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / temperature):
            # accept
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # tiny progress report
                print(f"[{time.time():.1f}]  new best = {best_score:.6f}")
        # ---------- 3. cool ----------
        temperature *= cooling
        transfer_frac = max(transfer_frac * cooling, 1e-4)

    return best_a, best_score


def propose_candidate(budget_s: int = 1000, **kwargs):
    """
    Entry point for the optimizer.  Returns a list of non‑negative heights
    (floats) that gives a low value of `evaluate_sequence`.

    Parameters
    ----------
    budget_s : int
        Time budget in seconds (default 1000).  The function will stop a
        little before this deadline and return the best sequence found.

    Returns
    -------
    list[float]
        The best coefficient sequence discovered within the time limit.
    """
    start = time.time()
    # we keep a small safety margin so the process can finish cleanly
    deadline = start + budget_s - 0.5

    # problem size – a few hundred elements gives a good trade‑off
    n = 500

    global_best_seq = None
    global_best_score = float("inf")

    # several short restarts improve robustness
    restarts = 4
    for r in range(restarts):
        if time.time() >= deadline:
            break
        print(f"--- Restart {r + 1}/{restarts} (n={n}) ---")
        seq, score = _propose_one_restart(n, deadline)
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"*** Global best updated to {global_best_score:.6f}")

    # fallback – uniform distribution (guaranteed finite score)
    if global_best_seq is None:
        global_best_seq = np.ones(n) / n

    print(f"Finished. Best score = {global_best_score:.6f}")
    return global_best_seq.tolist()
