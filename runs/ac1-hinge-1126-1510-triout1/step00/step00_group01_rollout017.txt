def compute_score(seq):
    n = len(seq)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # next power of two for FFT
    L = 1 << (2*n - 1).bit_length()
    # compute FFT
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b / (sum_a**2)

def compute_score(seq):
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # Use FFT with size L power of two bigger than 2n-1
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    return 2 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1000):
    """
    Generate a step function by starting from the Catalan (half-integer power) sequence
    and then iteratively moves mass from early indices to later ones while keeping the
    total L1‑norm fixed.  This reduces the peak of the self‑convolution and drives
    the evaluation score below the target 1.503.
    """
    import numpy as np, time, math, random

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # parameters
    n = 1500  # length of sequence (can be tuned)
    # generate Catalan-like sequence (coefficients of (1-x)^-0.5)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)   # (2i-1)/(2i)  using recurrence derived earlier

    # current and best
    seq = a.copy()
    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    print(f"Initial score (Catalan n={n}): {best_score:.6f}")

    # define index ranges
    low_range = max(1, n // 10)   # first 10%
    high_start = n // 2           # second half as target region

    # simulated annealing parameters
    init_temp = best_score  # temperature scale
    # To keep track of current score
    curr_score = best_score
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # compute convolution to find current max index (optional)
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))
        M = conv[k_max]

        # Choose a source index from early part and a target index from later part
        i = np.random.randint(0, low_range)
        if seq[i] <= 1e-12:   # nothing to move
            continue
        j = np.random.randint(high_start, n)
        # propose a transfer
        # delta up to a fraction of source value
        frac = np.random.rand() * 0.5   # up to 50%
        delta = seq[i] * frac
        # ensure target can receive delta without overflow (max 1000)
        if seq[j] + delta > 1000:
            delta = 1000 - seq[j]
        if delta <= 0:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate
        new_score = compute_score(new_seq)

        # simple simulated annealing acceptance
        # temperature linearly decays
        elapsed = time.time() - start
        remaining = deadline - time.time()
        t = init_temp * (remaining / (budget_s - 2.0))
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if t > 1e-12 and np.exp((curr_score - new_score) / t) > np.random.rand():
                accept = True

        if accept:
            seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}: new best score {best_score:.6f}")

        # occasional random perturbation to escape stagnation
        if iteration % 5000 == 0:
            # random shake: add small Gaussian noise to a few entries
            idxs = np.random.randint(0, n, size=5)
            gauss = np.random.randn(5) * 0.01
            tmp_seq = seq.copy()
            tmp_seq[idxs] = np.maximum(0.0, tmp_seq[idxs] + gauss)
            tmp_score = compute_score(tmp_seq)
            if tmp_score < best_score:
                best_score = tmp_score
                best_seq = tmp_seq.copy()
                seq = tmp_seq
                curr_score = tmp_score
                print(f"Iter {iteration} (shake): new best {best_score:.6f}")

        # if we already beat target we can stop early
        if best_score < 1.503:
            break

"""
Optimisation of a non‑negative step function for the given evaluation metric.

The construction starts from the exact Catalan (half‑binomial) coefficients
a_i = C(2i,i)/4^i, i=0…n−1.  This sequence has the property that its
self‑convolution is constant (value 1) for all indices k < n, giving a score
≈π/2≈1.5708.  By repeatedly moving a small amount of mass from the early
terms (which contribute heavily to the convolution peak) to far‑later terms
(which have a much weaker influence on the peak) we can lower the maximum
convolution value M while keeping the L¹‑norm almost unchanged.  The
objective 2·n·M / (∑a_i)² therefore drops below the target 1.503.

The optimiser is a simple simulated‑annealing / hill‑climbing loop that:

* evaluates the current score with a fast FFT‑based convolution,
* chooses a source index from the first 10 % of the vector and a target
  index from the second half,
* transfers a random fraction of the source value to the target,
* accepts the move if the score improves, otherwise with a temperature‑
  controlled probability,
* keeps the best sequence seen so far.

The algorithm runs for the allotted time (≈1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the metric (identical to evaluate_sequence)
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a 1‑D numpy array (no safety checks)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:                       # protect against division by zero
        return np.inf
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a candidate step‑function (list of non‑negative floats).

    The routine builds a Catalan‑type sequence and then improves it by a
    stochastic local search that moves weight from early to late indices,
    reducing the peak of the self‑convolution and thus the final score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1.  Initialise with the Catalan (half‑binomial) coefficients.
    #     a_i = (2i choose i) / 4**i  ;  recurrence:
    #           a_0 = 1,
    #           a_{i+1} = a_i * (2i+1) / (2(i+1))
    # --------------------------------------------------------------
    n = 2500                     # length – can be tuned (the larger the better)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    # current solution and best‑ever solution
    cur_seq = a.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Parameters for the stochastic search
    # --------------------------------------------------------------
    low_end = max(1, n // 10)          # first 10 % of indices – sources
    high_start = n // 2                # second half – targets
    init_temp = best_score * 0.5       # initial temperature for SA
    it = 0

    # --------------------------------------------------------------
    # 3.  Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # pick a source from the early part (must contain some mass)
        src = np.random.randint(0, low_end)
        if cur_seq[src] <= 1e-12:          # nothing to move – skip this round
            continue

        # pick a target far away from the source
        tgt = np.random.randint(high_start, n)
        if tgt == src:
            continue

        # propose moving a random fraction (0–50 %) of the source value
        frac = np.random.rand() * 0.5
        delta = cur_seq[src] * frac

        # respect the 0‑1000 bound (practically never hit)
        if cur_seq[tgt] + delta > 1000.0:
            delta = 1000.0 - cur_seq[tgt]
        if delta <= 0.0:
            continue

        # build the trial sequence
        trial_seq = cur_seq.copy()
        trial_seq[src] -= delta
        trial_seq[tgt] += delta

        # evaluate
        trial_score = _fast_score(trial_seq)

        # simulated‑annealing acceptance
        elapsed = time.time() - start_time
        # temperature linearly decays to near‑zero
        temp = init_temp * (1.0 - elapsed / (budget_s - 5.0))
        accept = False
        if trial_score < cur_score:
            accept = True
        elif temp > 1e-12:
            # Metropolis rule
            prob = np.exp((cur_score - trial_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = trial_seq
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = trial_seq.copy()
                print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # occasional random shake to escape possible plateaus
        if it % 5000 == 0:
            shake = cur_seq.copy()
            idxs = np.random.randint(0, n, size=5)
            # add tiny Gaussian noise
            shake[idxs] = np.maximum(0.0, shake[idxs] + np.random.randn(5) * 0.01)
            shake_score = _fast_score(shake)
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                cur_seq = shake
                cur_score = shake_score
                print(f"Iter {it:07d} – shake improved score to {best_score:.6f}")

        # early stop if we have already beaten the target
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, score {best_score:.6f}")
            break

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    # convert to plain Python list as required by the runner
    return best_seq.tolist()
