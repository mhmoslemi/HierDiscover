"""
Propose a better step‑function sequence by a fast projected gradient descent on the
max‑convolution objective.  The evaluation function is invariant under scaling, so we
fix the sum of the coefficients to 1 (the optimizer works on the simplex).  The
algorithm repeatedly

1. computes the self‑convolution b = a * a,
2. selects the indices where b is close to its maximum,
3. builds a sub‑gradient of the (soft‑max) peak‑value w.r.t. a,
4. tries a handful of step sizes, projects the update onto the probability simplex
   (non‑negative and sum‑to‑1) and keeps the best move,
5. if no improvement is found a small random perturbation is applied.

The loop runs until the time budget expires (≈ 1000 s).  The best sequence found
is returned as a plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# Helper utilities
# --------------------------------------------------------------

def _project_to_simplex(v):
    """
    Projects a non‑negative vector onto the probability simplex {x ≥ 0, sum(x)=1}.
    Uses the O(n log n) sorting method (a standard algorithm).
    """
    if v.sum() == 0:
        # return uniform distribution if the vector is all zeros
        return np.full_like(v, 1.0 / v.shape[0])
    # ensure non‑negative before projection
    v = np.maximum(v, 0.0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # extreme case, fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    # re‑normalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _convolution(a):
    """Linear convolution a*a (full length)."""
    return np.convolve(a, a)


def _max_convolution_info(a):
    """
    Returns (max_value, list_of_max_indices, full_convolution).
    """
    b = _convolution(a)
    max_val = b.max()
    # include all indices that are within 0.1 % of the maximum (to catch ties)
    eps = max_val * 1e-3
    max_inds = np.where(b >= max_val - eps)[0]
    return max_val, max_inds, b


def _gradient_from_max(a, max_inds, weight=1.0):
    """
    Sub‑gradient of the (soft) max‑convolution at the current a.
    The gradient of b_k = sum_i a_i a_{k-i} w.r.t. a_j equals 2*a_{k-j}
    (zero if the index is out of range).  For a set of indices we sum the
    contributions and average them.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    wk = weight / len(max_inds)  # uniform weight for the selected indices
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        # contribution = -2 * a_j  (negative because we want to *decrease* the peak)
        grad[i_range] -= 2.0 * wk * a[j_range]
    return grad


def _random_perturbation(a, magnitude=0.01):
    """
    Apply a tiny random move that preserves the sum.
    A random index loses a small amount, another gains it.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = magnitude * np.random.rand()
    delta = min(delta, a[i])  # cannot go negative
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑project to the simplex (takes care of any tiny negatives)
    return _project_to_simplex(a_new)

        grad = _gradient_from_max(a, max_inds, weight=1.0)

        # 3️⃣ try several step sizes and keep the best improvement
        improved = False
        best_candidate = None
        best_candidate_score = curr_score

        for eta in step_sizes:
            a_new = a - eta * grad
            # enforce non‑negativity
            a_new = np.maximum(a_new, 0.0)
            # project back onto the simplex (sum = 1)
            a_new = _project_to_simplex(a_new)

            # evaluate
            score = _evaluate(a_new.tolist())
            if score < best_candidate_score:
                best_candidate_score = score
                best_candidate = a_new
                improved = True
                # early exit for large improvements
                if score < curr_score * 0.99:
                    break

        if improved and best_candidate is not None:
            a = best_candidate
            curr_score = best_candidate_score
            if curr_score < best_score:
                best_score = curr_score
                best_seq = a.tolist()
                print(f"[{iteration:05d}] New best: {best_score:.6f} (len={len(a)})")
            last_improve_iter = iteration
        else:
            # No progress – apply a tiny random move to escape a flat region.
            a = _random_perturbation(a, magnitude=0.005)
            curr_score = _evaluate(a.tolist())
            # occasionally try a completely fresh random start if stuck too long
            if iteration - last_improve_iter > 2000:
                n_new = np.random.randint(1500, 2500)
                a = np.random.rand(n_new)
                a = a / a.sum()
                curr_score = _evaluate(a.tolist())
                last_improve_iter = iteration
                print(f"[{iteration:05d}] Restarted with new length {n_new}")

        iteration += 1

    # ---------------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    return best_seq
