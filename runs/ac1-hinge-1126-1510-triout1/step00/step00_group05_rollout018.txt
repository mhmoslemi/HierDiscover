def compute_convolution(p): return np.convolve(p,p)

def step(p, step_factor):
    conv = compute_convolution(p)
    k_max = np.argmax(conv)
    # compute contributions
    n = len(p)
    # create boolean mask for i s.t. 0 <= k_max - i < n
    i_indices = np.arange(n)
    j = k_max - i_indices
    valid = (j >= 0) & (j < n)
    contributions = np.zeros(n)
    contributions[valid] = p[i_indices[valid]] * p[j[valid]]
    # contributions from both (i,j) and (j,i) will be accounted separately when i iterates other side.
    # Pick i_high = argmax contributions
    # i_low = argmin contributions (maybe zero)
    i_high = np.argmax(contributions)
    i_low = np.argmin(contributions)  # could be zero contributions at edges
    # Transfer mass delta.
    delta = step_factor * p[i_high]  # portion of high mass
    # Ensure delta not exceed p[i_high]
    if delta <= 0: return p
    p_new = p.copy()
    p_new[i_high] -= delta
    p_new[i_low] += delta
    # Ensure non-negative
    if p_new[i_high] < 0:
        p_new[i_low] += p_new[i_high]  # add back negative to low
        p_new[i_high] = 0.0
    # Renormalize to sum=1 (should stay same but due to rounding)
    p_new /= p_new.sum()
    return p_new

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random, math
    np.random.seed(seed)
    random.seed(seed)

    # Determine length n
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000  # fallback

    # Initialize uniform sequence
    seq = np.full(n, 1.0 / n, dtype=np.float64)

    # optionally randomize a few entries to break symmetry
    # small random noise
    seq += np.random.uniform(-1e-4, 1e-4, size=n)
    seq = np.clip(seq, 0, None)
    seq /= seq.sum()

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    deadline = time.time() + budget_s - 5.0
    step_frac = 0.1
    temperature = 1e-3  # for simulated annealing acceptance
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        # compute convolution
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        # contributions for each index
        idx = np.arange(n)
        partner = max_idx - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[valid] = seq[idx[valid]] * seq[partner[valid]]
        # source index with largest contribution (must be >0)
        if np.all(contrib == 0):
            # fallback to random move
            src = np.random.randint(n)
            dst = np.random.randint(n)
        else:
            src = int(np.argmax(contrib))
            dst = int(np.argmin(contrib))

        # move fraction from src to dst
        delta = step_frac * seq[src]
        if delta <= 0:
            continue
        # ensure dst not same as src
        if src == dst:
            # pick another dst
            dst = (dst + 1) % n

        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # ensure non-negativity (should hold)
        new_seq = np.clip(new_seq, 0, None)
        # renormalize to sum 1 (scaling invariant)
        new_seq /= new_seq.sum()

        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq  # continue from improved
            print(f"Iter {iter_count}: improved score {best_score:.6f}")
            # optionally increase step size modestly
            step_frac = min(0.5, step_frac * 1.02)
        else:
            # probabilistic acceptance
            prob = math.exp(-(new_score - best_score) / temperature) if temperature > 0 else 0.0
            if random.random() < prob:
                seq = new_seq
            # slowly cool
            temperature *= 0.9999
        # decay step size
        step_frac = max(1e-5, step_frac * 0.9999)

        # occasional random perturbation
        if iter_count % 1000 == 0:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                eps = 0.01 * seq[i]
                seq[i] -= eps
                seq[j] += eps
                seq = np.clip(seq, 0, None)
                seq /= seq.sum()

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished after {iter_count} iterations, best score {final_score:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------

Search for a probability distribution (non‑negative step heights) whose
self‑convolution has a small maximum value.  The evaluation function

    E = 2 * n * max_k (a * a)[k] / (sum(a)**2)

is scale‑invariant, so we constrain the heights to sum to one.
The algorithm starts from a (almost) uniform distribution and performs a
simple hill‑climbing / simulated‑annealing loop:

* At each iteration the convolution is computed and the index `k_max`
  where the maximum occurs is identified.
* The contribution of each coefficient to this peak is
  `c_i = a_i * a_{k_max-i}` (zero if the partner index is out of range).
* Mass is moved from the coefficient with the largest contribution to the
  one with the smallest contribution (a small fraction `step` of the source
  mass).  The sequence is re‑normalised and the new score evaluated.
* Improvements are kept; occasional probabilistic moves allow escaping
  shallow plateaux.  The step size and a temperature parameter decay over
  time.

The loop runs until the allotted time (default 1000 s) is exhausted and
returns the best sequence found.
"""

import time
import random
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative floats representing the step heights.
    The routine tries to minimise ``evaluate_sequence`` within the given
    time budget.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # Length of the sequence: use the provided construction if available,
    # otherwise fall back to a reasonable default.
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000                     # safe default when no global is present

    # Initialise a nearly‑uniform distribution (sum = 1).
    seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Small random perturbation to break perfect symmetry early on.
    seq += np.random.uniform(-1e-4, 1e-4, size=n)
    seq = np.clip(seq, 0.0, None)
    seq /= seq.sum()

    # Store the best solution seen so far.
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the search
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin
    step_frac = 0.10                           # fraction of source mass to move
    temperature = 1e-3                         # annealing temperature
    iter_count = 0
    no_improve = 0

    # Main optimisation loop
    while time.time() < deadline:
        iter_count += 1

        # --------------------------------------------------------------
        # 1) Evaluate the current convolution and locate the peak
        # --------------------------------------------------------------
        conv = np.convolve(seq, seq)            # linear convolution, length 2n-1
        k_max = int(np.argmax(conv))            # index of the maximal entry

        # --------------------------------------------------------------
        # 2) Compute each coefficient's contribution to that peak
        # --------------------------------------------------------------
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(seq)
        contrib[valid] = seq[idx[valid]] * seq[partner[valid]]

        # --------------------------------------------------------------
        # 3) Choose source (largest contribution) and destination (smallest)
        # --------------------------------------------------------------
        if np.all(contrib == 0):
            # Fallback – pick two random distinct positions
            src = random.randrange(n)
            dst = random.randrange(n-1)
            if dst >= src:
                dst += 1
        else:
            src = int(np.argmax(contrib))
            dst = int(np.argmin(contrib))

        if src == dst:
            # pick a different destination if they coincide
            dst = (dst + 1) % n

        # --------------------------------------------------------------
        # 4) Propose a move: shift a bit of mass from src to dst
        # --------------------------------------------------------------
        delta = step_frac * seq[src]            # amount to move
        if delta <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # Numerical safety – clip negatives (should not happen) and renormalise
        new_seq = np.clip(new_seq, 0.0, None)
        new_seq /= new_seq.sum()                # scaling does not affect the score

        # --------------------------------------------------------------
        # 5) Evaluate the new candidate
        # --------------------------------------------------------------
        new_score = evaluate_sequence(new_seq.tolist())

        # --------------------------------------------------------------
        # 6) Acceptance logic
        # --------------------------------------------------------------
        if new_score < best_score:
            # Strict improvement – keep the move
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            no_improve = 0
            # Slightly enlarge step after a good move
            step_frac = min(0.5, step_frac * 1.02)
            print(f"[{iter_count:06d}] improvement -> score = {best_score:.6f}")
        else:
            # Probabilistic acceptance (simulated annealing)
            prob = math.exp(-(new_score - best_score) / temperature) if temperature > 0 else 0.0
            if random.random() < prob:
                seq = new_seq
            # Cool down and shrink the step
            temperature *= 0.9999
            step_frac = max(1e-5, step_frac * 0.9999)
            no_improve += 1

        # --------------------------------------------------------------
        # 7) Periodic random perturbation to escape stagnation
        # --------------------------------------------------------------
        if no_improve and no_improve % 5000 == 0:
            i, j = random.sample(range(n), 2)
            eps = 0.01 * seq[i]
            seq[i] = max(0.0, seq[i] - eps)
            seq[j] += eps
            seq /= seq.sum()

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] iterations = {iter_count}, best score = {final_score:.6f}")
    return best_seq.tolist()
