def simplex_projection(v):
    # v may have negative entries
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # ensure v shape (n,)
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Algorithm

2. Build several initial probability vectors (they sum to 1, i.e. a

   best one is kept.

   is performed.  At each iteration the index `k*` where the convolution
   (`p * p`) attains its maximum is found.  The gradient of that maximal

   back onto the probability simplex (simply by renormalisation – the

4. The loop runs until the allotted time (≈ 1000 s) is exhausted or the
   target value 1.503 is reached.  The best sequence found is returned
   as a plain list of floats (the values are already non‑negative and
   summed to 1, which is sufficient because the evaluation function is
   scale‑invariant).

The implementation uses only `numpy` and the supplied `evaluate_sequence`
function – no filesystem or network access is performed.
"""

import time

# Helper utilities

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a (possibly negative) vector onto the probability simplex:
    {x >= 0, sum x = 1}.  The implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013) and runs in O(n log n).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to guard against tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _linear_conv(p: np.ndarray) -> np.ndarray:
    """
    Compute the linear (non‑circular) convolution p * p using an FFT.
    Returns an array of length 2*len(p)-1.
    """
    n = p.shape[0]
    # size for FFT – next power of two for speed
    fft_len = 1 << ((2 * n - 1).bit_length())
    P = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(P * P, n=fft_len)
    return conv[: 2 * n - 1]


def _gradient_of_max(p: np.ndarray, k_max: int) -> np.ndarray:
    """
    Analytic gradient of the maximal convolution entry (index k_max)
    w.r.t. the probability vector `p`.  The gradient is
        g_i = 2 * p_{k_max - i}   if the partner index is inside [0,n)
    and 0 otherwise.
    """
    n = p.shape[0]
    g = np.zeros_like(p)
    # Number of overlapping positions for this shift
    overlap = min(k_max + 1, 2 * n - 1 - k_max)
    if k_max < n:
        # i runs from 0 .. overlap-1
        i = np.arange(overlap)
        j = k_max - i
        g[i] = 2.0 * p[j]
    else:
        start = k_max - (n - 1)
        i = np.arange(start, start + overlap)
        j = k_max - i
        g[i] = 2.0 * p[j]
    return g

            break
        i = np.arange(n, dtype=float)
        x = (i + 0.5) / n                     # ∈ (0,1)
        # p_i ∝ x^{a-1} (1-x)^{a-1}
        p = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
        p += 1e-12                            # avoid exact zeros
        p /= p.sum()
        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"[BETAS] a={a:5.2f} → new best score {sc:.6f}")
        # early stop if we already beat the target
        if best_score <= 1.503:
            break

    # a few random Dirichlet starts (helps escape local minima)
    for _ in range(3):
        if time.time() >= deadline:
            break
        p = np.random.dirichlet(np.ones(n))
        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"[RANDOM] new best score {sc:.6f}")
        if best_score <= 1.503:
            break

    if best_seq is None:
        # ultimate fallback – uniform sequence
        best_seq = np.full(n, 1.0 / n)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣  Refine the current best with projected gradient descent
    # ------------------------------------------------------------------
    p_curr = best_seq.copy()
    iter_cnt = 0
    # Parameters for the descent
    step0 = 0.5
    decay = 1e-4                       # controls how fast the step shrinks

    while time.time() < deadline:
        iter_cnt += 1
        conv = _linear_conv(p_curr)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_curr.copy()
            print(f"[ITER {iter_cnt:05d}] improved best score {cur_score:.6f}")

        # stop early if we have already reached the target
        if best_score <= 1.503:
            print("[STOP] target achieved")
            break

        # gradient step that tries to shrink the current maximal entry
        k_star = int(conv.argmax())
        g = _gradient_of_max(p_curr, k_star)

        # simple decreasing step size
        alpha = step0 / (1.0 + decay * iter_cnt)

        p_next = p_curr - alpha * g
        # enforce non‑negativity and project back to the simplex
        p_next = np.maximum(p_next, 0.0)
        s = p_next.sum()
        if s == 0.0:
            p_next = np.random.dirichlet(np.ones(n))
        else:
            p_next /= s

        # occasional random shake to avoid stagnation
        if np.random.rand() < 0.001:
            shake = np.random.dirichlet(np.full(n, 0.5))
            p_next = 0.9 * p_next + 0.1 * shake
            p_next = p_next / p_next.sum()

        p_curr = p_next

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    final_score = _score(best_seq)
    print(f"[RESULT] best score after optimisation = {final_score:.6f}")
    # Convert to plain Python list (the evaluator accepts a list)
    return best_seq.tolist()
