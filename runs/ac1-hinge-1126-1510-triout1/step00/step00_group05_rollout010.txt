def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    target_time = time.time() + budget_s - 5

    # initial best: simple two-element ratio 2:1
    best_seq = [2.0, 1.0]  # yields evaluate about 1.777...
    best_score = evaluate_sequence(best_seq)

    # main loop
    while time.time() < target_time:
        # pick a generator type randomly
        gen_type = np.random.choice(['geo','pow','rand','two'])

        if gen_type == 'geo':
            n = np.random.randint(5, 2000)  # longer lengths for better potential
            r = np.random.uniform(0.01, 0.99)
            seq = [r**i for i in range(n)]
        elif gen_type == 'pow':
            n = np.random.randint(5, 2000)
            p = np.random.uniform(0.1, 3.0)
            seq = [1.0/(i+1)**p for i in range(n)]
        elif gen_type == 'rand':
            n = np.random.randint(5, 2000)
            # draw from Dirichlet
            alpha = np.ones(n)
            seq = np.random.dirichlet(alpha).tolist()
        else:  # two-step
            n = np.random.randint(2, 2000)
            split = np.random.randint(1, n)
            a = np.random.rand()*10+0.1
            b = np.random.rand()*10+0.1
            seq = [a]*split + [b]*(n-split)

        # optional normalization (not needed)
        # evaluate
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best from generator {gen_type} n={len(seq)} score={best_score:.6f}")

        # local refinement on current best_seq
        # run a small number of directed updates
        # skip if time low
        if time.time() < target_time - 0.02:
            seq_refined = local_refine(best_seq, steps=100)

                score2 = evaluate_sequence(seq_refined)

                    best_score = score2
                    best_seq = seq_refined.copy()

            except Exception:
                pass

    return best_seq

def local_refine(seq, steps=200):
    a = np.array(seq, dtype=float)
    n = len(a)
    for _ in range(steps):
        # compute conv and get max indices
        b = np.convolve(a, a)
        max_val = b.max()
        max_idx = np.flatnonzero(b == max_val)
        # compute weight per index
        weight = np.zeros(n)
        for k in max_idx:
            # contributions of each i where both i and k-i in [0,n-1]
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    weight[i] += a[j]
        # choose i from top weighted, j from low weighted
        i = np.argmax(weight)
        j = np.argmin(weight)
        if a[i] <= 1e-12 or i==j:
            continue
        # move some mass
        delta = 0.05 * a[i]  # 5% step
        a[i] -= delta
        a[j] += delta
        # ensure non-negative
        if a[i] < 0:
            a[j] += a[i]
            a[i] = 0.0
        # optional early break if no improvement?
    return a.tolist()

def reduce_max_seq(seq, step=0.05):
    a = np.array(seq, dtype=float)
    n = len(a)
    b = np.convolve(a, a)
    max_idx = int(np.argmax(b))
    # compute contributions for each i
    contrib = np.zeros(n)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]
    # pick i with max contribution
    i = int(np.argmax(contrib))
    # pick j random that yields small contribution (or not overlapping)
    # avoid i
    candidates = list(range(n))
    candidates.remove(i)
    # find j with low contribution to same max shift (maybe a[j]*a[max_idx - j] minimal)
    contrib_j = [contrib[j] for j in candidates]
    j = int(candidates[np.argmin(contrib_j)])
    delta = step * a[i]
    if delta <= 0: return seq
    a[i] -= delta
    a[j] += delta
    return a.tolist()

"""
Propose a new step‑function sequence that beats the uniform bound (value = 2).

Algorithm
---------

1.  Initialise a collection of candidate sequences.
    * Uniform vector (the old SOTA).
    * Very short two‑element vectors (the analytically optimal [2,1] gives ~1.78).
    * Random points on the simplex (Dirichlet samples) of several lengths.
    * Simple parametric families (geometric decay, power‑law, two‑level blocks).

2.  For every candidate try a _local optimisation_ that reduces the peak of the
    self‑convolution.
    * Compute the convolution and locate the shift(s) where the maximum occurs.
    * The contribution of element i to that peak is `c_i = Σ_{k∈K} a_{k−i}`
      (K = set of maximal shifts).  Elements with large `c_i` are the main
      culprits.
    * Move a small fraction of mass from the most harmful index to the least
      harmful one (mass is preserved, all entries stay non‑negative).
    * Perform a few hundred such moves; after each move the new value of
      `evaluate_sequence` is tested and kept only if it improves the score.
      This simple greedy step is extremely fast (O(n) per move).

3.  An additional improvement pass based on linear programming (the same idea
    used by the previous SOTA) is performed on the current best sequence.
    The LP finds a direction `g` that can increase the total sum while keeping
    the convolution bound unchanged.  The current sequence is then updated by a
    convex combination `f ← (1‑t)·f + t·g` with several trial values of `t`
    (0.1, 0.2, 0.4, 0.8).  The best `t` that yields a lower score is kept.
    Only a handful of LP calls are made (the lengths used are ≤ 800, which is
    fast enough).

4.  The whole process runs until the allotted time (≈ 990 s) expires.
    The best sequence found so far is returned.

The routine never exceeds the built‑in limits of `evaluate_sequence`
(non‑negative, finite numbers, sum ≥ 0.01) and therefore never returns
`inf`.  It is completely self‑contained and uses only NumPy and SciPy.

"""

import time

from scipy import optimize

def _evaluate(seq):
    """Wrapper for the provided evaluate_sequence with error handling."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")


def _geometric_sequence(n, r):
    """a_i = r**i  (r in (0,1))."""
    return [r ** i for i in range(n)]


def _powerlaw_sequence(n, p):
    """a_i = 1/(i+1)**p  (p > 0)."""
    return [1.0 / ((i + 1) ** p) for i in range(n)]


def _two_level_sequence(n, split, a, b):
    """first `split` entries = a, remaining = b (both ≥0)."""
    return [a] * split + [b] * (n - split)


def _random_simplex(n):
    """Draw a random point on the simplex of dimension n (Dirichlet with α=1)."""
    return np.random.dirichlet(np.ones(n)).tolist()


def _local_refine(seq, steps=200, step_ratio=0.07):
    """
    Greedy refinement that shifts a small fraction of mass from
    the indices that contribute most to the current maximal convolution
    towards the indices that contribute least.
    """
    a = np.array(seq, dtype=float)
    n = len(a)

    for _ in range(steps):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # Take the first shift achieving the maximum (any works)
        k = int(conv.argmax())

        # contribution of each entry to this peak (a_i * a_{k-i})
        contrib = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # indices with largest / smallest contribution
        i = int(contrib.argmax())
        j = int(contrib.argmin())
        if i == j or a[i] <= 0.0:
            break

        # move a modest fraction of the mass from i to j
        delta = step_ratio * a[i]
        a[i] -= delta
        a[j] += delta

    return a.tolist()


def _solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   Σ b_j
        subject to Σ_i f_i * b_{k-i} ≤ rhs    (for every shift k)
                   b_j ≥ 0
    Returns the optimal `b` as a NumPy array, or None on failure.
    """
    n = len(f_seq)

    # Objective: maximise 1·b  → minimise -1·b
    c = -np.ones(n)

    # Build the convolution constraints (2n‑1 of them)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  -b ≤ 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # Use the high‑performance default method ("highs")
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                          bounds=(0, None),
                          method='highs')
    if res.success:
        return res.x
    return None


def _lp_improvement(seq, max_iters=6, t_candidates=None):
    """
    Perform a few LP‑based improvement steps on `seq`.
    Returns the (possibly) improved sequence.
    """
    if t_candidates is None:
        t_candidates = [0.1, 0.2, 0.4, 0.8]

    a = np.array(seq, dtype=float)
    n = len(a)

    # Normalise to the "standard" L2‑norm used in the paper:
    #   Σ a_i = √(2n)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    # current bound M = max(conv(a,a))
    conv = np.convolve(a, a)
    M = conv.max()

    for _ in range(max_iters):
        g = _solve_convolution_lp(a, M)
        if g is None:
            break
        # scale g to the same L1‑norm as a
        g = g * (target_sum / g.sum())

        # try several mixing ratios and keep the best improvement
        best_local = a
        best_val = _evaluate(best_local.tolist())
        improved = False

        for t in t_candidates:
            cand = (1.0 - t) * a + t * g
            val = _evaluate(cand.tolist())
            if val < best_val:
                best_local = cand
                best_val = val
                improved = True

        if not improved:
            # no t yielded improvement → stop early
            break

        a = best_local
        # recompute M for the new iterate
        M = np.convolve(a, a).max()

    return a.tolist()

        # ------------------------------------------------------------------
        refined = _local_refine(cand, steps=250, step_ratio=0.06)
        score_ref = _evaluate(refined)
        if score_ref < best_score:
            best_seq, best_score = refined, score_ref
            print(f"Refined (greedy) ({kind}) n={n}: score={score_ref:.6f}")

        # ------------------------------------------------------------------
        # 2b️⃣  Occasional LP‑based improvement on the global best
        # ------------------------------------------------------------------
        if np.random.rand() < 0.12:   # low frequency – LP is expensive
            candidate = best_seq
            # limit the length for LP to keep solve time reasonable
            if len(candidate) > 800:
                # down‑sample by picking a random contiguous block
                start = np.random.randint(0, len(candidate) - 800)
                candidate = candidate[start:start + 800]
            improved = _lp_improvement(candidate, max_iters=5)
            score_imp = _evaluate(improved)
            if score_imp < best_score:
                best_seq, best_score = improved, score_imp
                print(f"LP improvement success: score={score_imp:.6f}, n={len(improved)}")

        # ------------------------------------------------------------------
        # safety break if time is about to run out
        # ------------------------------------------------------------------
        if time.time() > deadline - 5.0:
            break

    # final sanity clamp – ensure a tiny positive sum (required by evaluator)
    best_seq = [float(x) for x in best_seq]
    if sum(best_seq) < 0.01:
        # add a tiny constant to avoid the “sum too close to zero” guard
        best_seq[0] = best_seq[0] + 0.02

    print(f"Finished search – best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
