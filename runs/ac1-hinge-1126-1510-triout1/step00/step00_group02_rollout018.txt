def project_to_simplex(v):
    # ensure nonnegative and sum to 1
    # algorithm: sort v descending, find rho s.t. v_i - ( (cumsum - 1) / i ) > 0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')

    # Define lengths to try
    lengths = [800, 1500, 2000]
    for n in lengths:
        # initial candidates: power law with p in list, mixture, random.
        # 1. generate power-law candidates
        p_vals = np.linspace(0.3, 0.7, 9)  # 0.3..0.7
        for p in p_vals:
            base = np.arange(1, n+1, dtype=float) ** (-p)
            base = base / base.sum()
            score = compute_score(base, n)
            if score < best_score:
                best_score = score
                best_seq = base.copy()
                print(f"Init pow p={p:.3f}, n={n}, score={score:.6f}")
            # generate mixture with uniform
            uniform = np.ones(n) / n
            for alpha in np.linspace(0, 1, 11):
                cand = alpha * base + (1 - alpha) * uniform
                cand /= cand.sum()
                score = compute_score(cand, n)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"Init mix p={p:.3f}, alpha={alpha:.2f}, n={n}, score={score:.6f}")

        # random initializations
        for _ in range(5):
            cand = np.random.rand(n)
            cand = cand / cand.sum()
            score = compute_score(cand, n)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"Random init, n={n}, score={score:.6f}")

        # now run local optimization from best_seq for this n
        a = best_seq.copy()
        # ensure length matches n
        if len(a) != n:
            a = np.random.rand(n)
            a /= a.sum()
        # run iteration loop
        iteration = 0
        while time.time() < deadline:
            # compute convolution via FFT
            fa = np.fft.rfft(a, n=2*n-1)
            b = np.fft.irfft(fa*fa, n=2*n-1)
            # consider only real part
            b = np.real(b)
            max_idx = int(np.argmax(b))
            max_val = b[max_idx]

            # compute score
            score = 2.0 * n * max_val / (a.sum()**2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter n={n} iter={iteration} new best {best_score:.6f}")

            # balancing step
            count_k = max_idx + 1 if max_idx < n else 2*n-1-max_idx
            target = max_val / count_k  # target product per term

            # compute partner indices vectorized
            i = np.arange(n)
            j = max_idx - i
            mask = (j >= 0) & (j < n)
            partner = np.zeros(n, dtype=float)
            partner[mask] = a[j[mask]]

            # products
            prod = a * partner
            # delta product (positive if too large)
            delta = prod - target
            # update a_i
            # avoid division by zero
            step = 0.5  # step size
            # compute update: a_i = a_i - step * delta / (partner + eps)
            eps = 1e-12
            a_new = a - step * delta / (partner + eps)
            # clip negatives
            a_new = np.maximum(a_new, 0.0)
            # renormalize to sum 1
            s = a_new.sum()
            if s == 0:
                # reinitialize
                a_new = np.random.rand(n)
                a_new /= a_new.sum()
            else:
                a_new /= s
            a = a_new
            iteration += 1
            # optional break if no improvement after some iterations
            if iteration > 5000:
                break

    if best_seq is None:
        return [1.0]
    return list(best_seq)

"""
Sequence optimizer for the convolution‐norm objective.

The algorithm combines a simple parametric search (power‑law families) with a
custom “balance” refinement that tries to flatten the convolution
(a*a).  The refinement iterates a small number of steps that equalise the
contributions of the pairwise products belonging to the currently maximal
convolution index.  All updates keep the vector non‑negative and re‑normalise
it to unit ℓ¹–norm, so the evaluation score
    2·n·max(a*a)/(sum(a)²)
is simply 2·n·max(a*a) during the search.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` function, and respects the 1000 s time budget.
"""

import numpy as np
import time

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _conv_max(a: np.ndarray) -> float:
    """
    Fast convolution using FFT (real -> real) and return the maximum entry.
    The vector ``a`` is assumed to be non‑negative.
    """
    n = a.shape[0]
    # real FFT of length 2*n‑1 (next power of two is not required, rfft works)
    fa = np.fft.rfft(a, n=2 * n - 1)
    b = np.fft.irfft(fa * fa, n=2 * n - 1)
    # small negative values from numerical noise are harmless for max()
    return float(b.max())


def _score(a: np.ndarray) -> float:
    """
    Compute the objective value for a non‑negative sequence a.
    Since the sum is re‑normalised to 1 before the call, the denominator
    equals 1 and the score reduces to 2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


def _normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑norm equal to 1."""
    s = v.sum()
    if s == 0.0:
        # should never happen – fall back to a uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# --------------------------------------------------------------
# Core optimisation routine
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current benchmark.  The routine respects a wall‑clock budget
    (default 1000 s).  It works in three stages:

    1.  A quick parametric scan over power‑law sequences
        a_i ∝ (i+1)^(-p) for several exponents p and several
        mixture weights with the uniform vector.

    2.  Random initialisations (a few dozen) to diversify the search
        space.

    3.  A lightweight “balance” refinement applied to the currently
        best candidate.  The refinement repeatedly identifies the
        convolution index k_max with the largest value, computes the
        target product per contributing pair (max_value / count(k_max)),
        and moves mass from indices whose local product exceeds the target
        to those whose product is below the target.  After each move the
        vector is clipped to non‑negative values and renormalised.
        The process is stopped when the time budget expires.

    The function finally returns the best sequence (as a plain Python list)
    found within the allocated time.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 5.0)  # leave a tiny safety margin

    best_seq = None
    best_val = np.inf

    # ----------------------------------------------------------
    # 1)  Parametric scan – power laws and simple mixtures
    # ----------------------------------------------------------
    lengths_to_try = [800, 1500, 2000]          # a few representative lengths
    power_exponents = np.linspace(0.30, 0.70, 9)  # 0.30 … 0.70

    for n in lengths_to_try:
        # uniform baseline (useful for mixtures)
        uniform = np.full(n, 1.0 / n)

        # power‑law families
        for p in power_exponents:
            base = np.arange(1, n + 1, dtype=float) ** (-p)
            base = _normalise(base)

            # pure power‑law candidate
            val = _score(base)
            if val < best_val:
                best_val, best_seq = val, base.copy()
                print(f"[init] n={n} power={p:.3f} → {best_val:.6f}")

            # mixtures with the uniform vector
            for alpha in np.linspace(0.0, 1.0, 11):
                cand = alpha * base + (1.0 - alpha) * uniform
                cand = _normalise(cand)
                val = _score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    print(f"[init] n={n} power={p:.3f} α={alpha:.2f} → {best_val:.6f}")

        # ------------------------------------------------------
        # 2) Random starts – a few dozen fresh candidates
        # ------------------------------------------------------
        for _ in range(5):
            cand = np.random.rand(n)
            cand = _normalise(cand)
            val = _score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[rand] n={n} → {best_val:.6f}")

        # ------------------------------------------------------
        # 3) Balance refinement – apply to the best vector of this length
        # ------------------------------------------------------
        if best_seq is None or best_seq.shape[0] != n:
            # safety: if something went wrong, start from uniform
            a = uniform.copy()
        else:
            a = best_seq.copy()

        iteration = 0
        # parameters for the refinement
        max_iters = 5000           # hard cap (usually far below the time limit)
        step_factor = 0.7          # how aggressively to move mass
        eps = 1e-12                # avoid division by zero

        while time.time() < deadline and iteration < max_iters:
            # current convolution and its worst index
            n = a.shape[0]
            fa = np.fft.rfft(a, n=2 * n - 1)
            b = np.fft.irfft(fa * fa, n=2 * n - 1)
            k_max = int(np.argmax(b))
            max_val = float(b[k_max])

            # current score (used only for acceptance tracking)
            cur_val = 2.0 * n * max_val
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[refine] n={n} iter={iteration} → {best_val:.6f}")

            # ----- balancing step ---------------------------------
            # number of ordered pairs that contribute to index k_max
            if k_max < n:
                pair_cnt = k_max + 1
            else:
                pair_cnt = 2 * n - 1 - k_max

            # target product per ordered pair (the average for this k_max)
            target = max_val / pair_cnt

            # partner index for each i (j = k_max - i)
            idx = np.arange(n)
            j = k_max - idx
            valid = (j >= 0) & (j < n)

            partner = np.empty_like(a)
            partner.fill(0.0)
            partner[valid] = a[j[valid]]

            # current product a_i * a_{k_max-i}
            prod = a * partner

            # error = current - target; we want to push it towards zero
            err = prod - target

            # update rule: a_i ← a_i - step * err / (partner + eps)
            #   (moves a_i opposite to its excess contribution)
            delta = -step_factor * err / (partner + eps)
            a_new = a + delta

            # enforce non‑negativity, then renormalise to unit sum
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalise(a_new)

            a = a_new
            iteration += 1

    # --------------------------------------------------------------
    # Final safety – fall back to a simple uniform vector if nothing found
    # --------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # Return a plain Python list as required by the evaluator
    return list(best_seq)
