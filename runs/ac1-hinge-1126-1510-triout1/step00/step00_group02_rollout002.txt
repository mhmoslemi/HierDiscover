import time

def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution of two real vectors using FFT.
    The result has length len(x)+len(y)-1.
    """
    m = len(x) + len(y) - 1
    n_fft = 1 << (m - 1).bit_length()          # next power of two

    Y = np.fft.rfft(y, n=n_fft)
    z = np.fft.irfft(X * Y, n=n_fft)[:m]
    return z

        iteration += 1

        # ----- convolution of the current iterate ---------------------------
        b = _full_convolution(a, a)          # length 2n‑1
        max_b = b.max()

        # ----- soft‑max weights concentrating on the biggest entries ---------
        # shift by max_b for numerical stability (all values ≤ 0)
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # normalised weights

        # ----- gradient of the soft‑max of the convolution ------------------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]                       # reversed a
        conv_w_rev = _full_convolution(w, a_rev)   # length 3n‑2
        grad = 2.0 * conv_w_rev[n - 1 : n - 1 + n]

        # ----- momentum ----------------------------------------------------
        direction = grad + momentum * direction_prev
        direction_prev = direction

        # ----- back‑tracking line search (up to a few reductions) ----------
        eta_try = eta
        for _ in range(8):
            a_new = a - eta_try * direction
            a_new = np.maximum(a_new, 0.0)        # project onto ℝ₊ⁿ
            s = a_new.sum()
            if s == 0.0:                         # safeguard against the zero vector
                a_new[:] = 1.0 / n
                s = 1.0
            a_new /= s                            # renormalise (sum = 1)

            b_new = _full_convolution(a_new, a_new)
            score_new = 2.0 * n * b_new.max()

            if score_new <= best_score:          # accept if not worse
                a = a_new
                if score_new < best_score:
                    best_score = score_new
                    best_seq = a_new.copy()
                # slightly enlarge step if we succeeded
                eta = eta_try * 1.05
                break
            else:
                eta_try *= 0.5                    # shrink step and retry

        # ----- occasional random jitter to break symmetries -----------------
        if iteration % 150 == 0:
            i, j = np.random.choice(n, size=2, replace=False)
            eps = np.random.rand() * 0.02       # up to 2 % of total mass
            delta = min(eps, a[i])
            a_tmp = a.copy()
            a_tmp[i] -= delta
            a_tmp[j] += delta
            a_tmp /= a_tmp.sum()

            b_tmp = _full_convolution(a_tmp, a_tmp)
            score_tmp = 2.0 * n * b_tmp.max()
            if score_tmp < best_score:
                best_score = score_tmp
                best_seq = a_tmp.copy()
                a = a_tmp                         # keep the improvement

        # ----- occasional progress report ------------------------------------
        if iteration % 500 == 0:
            print(f"[Iter {iteration}] best score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------------
    # 4. Return the best discovered sequence ---------------------------------
    # --------------------------------------------------------------------
    # Ensure the format matches the expected input of `evaluate_sequence`.
    return best_seq.tolist()
