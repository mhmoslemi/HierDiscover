def fast_evaluate(seq):
    arr = np.array(seq, dtype=np.float64)
    n = len(arr)
    # compute convolution using FFT: zero pad to length N = 2^ceil(log2(2n-1))
    N = 1 << ((2*n-2).bit_length())
    fft_seq = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_seq * fft_seq, n=N)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def fast_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    n = arr.size
    # computing aperiodic convolution via FFT with zero padding length N = next power of 2 >= 2n-1
    N = 1 << ((2*n-2).bit_length())
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)
```

"""
Propose a candidate step‑function sequence.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a))²
where `conv` is the linear (aperiodic) convolution.
The minimum possible value tends to 1 when the convolution is almost flat.
A flat convolution can be approached by a random 0/1 (binary) pattern with a
moderate density of ones: the number of ordered pairs that sum to a given
shift follows approximately a Poisson distribution whose mean is
k²/(2n) (k = number of ones).  For large k this mean is large and the
relative fluctuations become small, giving a convolution whose maximum is
close to the average and therefore a low evaluation score.

The algorithm therefore repeatedly draws binary sequences of length n
(4000 – 8000) with a density of ones in the interval [0.18, 0.30],
keeps the best one (according to a fast FFT‑based score), and performs a
short local hill‑climb that swaps a random 1 with a random 0 and accepts
the move if it improves the score.  The whole search runs for the allotted
time (default 1000 s) and finally returns the best sequence found.
"""

import time
import numpy as np

# -------------------------------------------------------------
# fast evaluation (FFT based) – used for the search
# -------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same metric as `evaluate_sequence` but much faster."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:                     # same safeguard as evaluate_sequence
        return np.inf
    # next power of two for efficient FFT, at least 2·n-1
    N = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# -------------------------------------------------------------
# local hill‑climbing on a binary pattern
# -------------------------------------------------------------
def _hill_climb(
    seq: np.ndarray,
    cur_score: float,
    max_iter: int,
    best_score: float,
) -> tuple[np.ndarray, float, float]:
    """
    Perform a very cheap hill‑climb: repeatedly swap a random 1 with a random 0.
    Accept a swap only if it does not increase the score (or with a tiny
    probability to escape plateaus).  Return the (possibly) improved sequence,
    its score and the current global best score.
    """
    n = seq.shape[0]
    a = seq.copy()
    score = cur_score
    for _ in range(max_iter):
        ones = np.flatnonzero(a)          # indices where a == 1
        zeros = np.flatnonzero(1 - a)     # indices where a == 0
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        a[i] = 0.0
        a[j] = 1.0
        new_score = _fast_score(a)

        # accept if better, otherwise revert (occasionally accept worse)
        if new_score < score or np.random.rand() < 0.001:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                # keep the global best sequence inside the caller
        else:
            a[i] = 1.0
            a[j] = 0.0
    return a, score, best_score


# -------------------------------------------------------------
# main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The method repeatedly samples binary vectors with
    a moderate density of ones and refines them by a short hill‑climb.
    The best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a few seconds guard

    best_seq = None
    best_score = float("inf")
    iter_count = 0

    # length candidates – larger lengths give a lower theoretical bound,
    # but also a higher per‑iteration cost.
    length_choices = [3000, 4000, 5000, 6000, 8000]

    while time.time() < deadline:
        # -------------------------------------------------
        # draw a random binary pattern
        # -------------------------------------------------
        n = int(np.random.choice(length_choices))
        density = np.random.uniform(0.18, 0.30)        # fraction of ones
        k = max(2, int(density * n))                # number of ones (at least 2)
        idx = np.random.choice(n, size=k, replace=False)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = 1.0

        cur_score = _fast_score(seq)

        # keep the best random draw
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{iter_count:5d}] new best (rand)  score={best_score:.6f}  n={n}  k={k}")

        # -------------------------------------------------
        # short local optimisation (swap‑based hill‑climb)
        # -------------------------------------------------
        # limit the number of swaps – 200 is cheap and enough to improve most draws
        seq_opt, opt_score, best_score = _hill_climb(
            seq, cur_score, max_iter=200, best_score=best_score
        )
        if opt_score < cur_score:
            # a genuine improvement after hill‑climb
            if opt_score < best_score:
                best_score = opt_score
                best_seq = seq_opt.copy()
                print(f"[{iter_count:5d}] new best (hill) score={best_score:.6f}  n={n}  k={k}")

        iter_count += 1

    # -----------------------------------------------------------------
    # if nothing succeeded (unlikely) fall back to a uniform sequence
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # convert to plain Python list of floats as required by the evaluator
    return best_seq.tolist()
