def simplex_projection(v):
    # From https://github.com/OptimizationModelsInPython/Cvxpy/blob/master/cvxpy/... not needed
    # Input v: 1d numpy array.
    # Output: projection onto simplex {x>=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scaling‑invariant; we therefore optimise a probability
vector p (sum(p)=1, p_i≥0) and finally output a = p * sqrt(2*n).
The problem is to minimise   max(p * p)   where * denotes discrete
convolution.  This is a non‑convex min‑max problem.  We use a smooth
approximation of the max (log‑sum‑exp) and perform projected gradient
descent on the probability simplex.  The gradient of the smoothed
objective can be computed by a single convolution, which makes each
iteration O(n log n) (or O(n²) with the simple np.convolve for moderate
sizes).  An annealing schedule slowly sharpens the soft‑max (increase
α) while the step‑size decays.  Random perturbations and occasional
re‑starts help escape shallow local minima.  The best sequence found
during the allotted time is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based indexing
    # find the largest rho satisfying u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative; return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation score.  The routine runs for ~budget_s seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # Larger n gives finer discretisation; n≈3000‑5000 is a good trade‑off.
    # ------------------------------------------------------------------
    n = 3500                              # length of the step‑function
    # ------------------------------------------------------------------

    # If a previous solution is available (height_sequence_1) we start
    # from its normalised version, otherwise use the uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(height_sequence_1, dtype=float)
        s = prev.sum()
        if s > 0:
            p = prev / s
        else:

        # adjust length if necessary
        if prev.shape[0] != n:
            # resample/trim to the chosen length
            if prev.shape[0] > n:
                p = p[:n]
                p = p / p.sum()
            else:
                # pad with tiny uniform mass
                extra = np.full(n - prev.shape[0], 1e-12)
                p = np.concatenate([p, extra])
                p = p / p.sum()
    else:
        p = np.full(n, 1.0 / n)

    # Gradient‑descent hyper‑parameters
    step0 = 0.6                # base step size
    alpha = 1.0                # initial soft‑max temperature
    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(p)
    best_score = float("inf")
    best_p = None
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- convolution p * p (full length 2n‑1) --------------------
        conv = np.convolve(p, p)                # O(n²) but fine for n≈3500
        max_conv = conv.max()
        # ----- true objective (scaling‑invariant) ----------------------
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_p = p.copy()

        # ----- annealing schedule: slowly sharpen the soft‑max ---------
        if it % 150 == 0 and it > 0:

        # subtract max for numerical stability

        rev_p = p[::-1]

    # ------------------------------------------------------------------

    heights = (best_p * scale).tolist()
    # Ensure the returned list is pure Python floats (no NumPy scalars)
    heights = [float(v) for v in heights]

    # ------------------------------------------------------------------
    # Final logging
    # ------------------------------------------------------------------
    final_score = 2.0 * n * np.max(np.convolve(heights, heights))
    print(f"Finished after {it} iterations. Best score = {best_score:.6f}")
    print(f"Re‑computed score from returned heights = {final_score:.6f}")

    return heights
