"""
recommendation:
    A fast stochastic optimizer that works directly on the objective
    2·n·max(a*a)/sum(a)².  The algorithm keeps the vector on the probability
    simplex (sum = 1) – the objective is scale‑invariant – and iteratively
    applies two types of moves:

    1.  **Gradient descent on the active convolution index** – the index k
        where the current convolution attains its maximum.  The sub‑gradient
        of max(a*a) at that index is 2·a shifted by k, so we step away from
        the peak, renormalise and keep the best improvement.

    2.  **Random mass‑exchange** – move a small amount of mass from one
        coordinate to another while preserving the total sum.  This adds
        stochasticity and helps escape plateaux.

    Convolution is evaluated by FFT (O(n log n)) so that many thousands of
    iterations fit comfortably into the 1000 s budget.  The best sequence
    found is returned as a plain Python list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _max_convolution(a: np.ndarray) -> float:
    """
    Compute max(a * a) using a real FFT.  The length of the full convolution is
    2·len(a)‑1; only the maximum value is required.
    """
    n = a.shape[0]
    # size is next power‑of‑two ≥ 2·n‑1 for efficient FFT
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑x = 1}.  Implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
