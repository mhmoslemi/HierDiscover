def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 10
    # Choose length
    n = kwargs.get('n', 1200) # maybe adjust
    # initialize f: optionally from height_sequence_1 if provided else random
    if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
        f = np.array(height_sequence_1[:n])  # ensure length n maybe adjust if shorter.
        # Pad/trim
        if len(f) < n:
            f = np.pad(f, (0, n - len(f)), constant_values=0.0)
        elif len(f) > n:
            f = f[:n]
    else:
        # random positive entries
        f = np.random.rand(n)
    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2 * n)
    f = f * (target_sum / np.sum(f))
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    iteration = 0
    while time.time() < deadline:
        # calculate M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Solve LP for g
        g = solve_convolution_lp(f.tolist(), M)
        if g is None:
            # random perturbation: add small gaussian noise to random element(s)
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * (np.max(f) * 0.01 + 1e-6))
            # renorm to target sum
            f = f * (target_sum / np.sum(f))
            continue
        g = np.array(g)
        # normalize g sum to target_sum
        g = g * (target_sum / np.sum(g))
        # compute conv_gg
        conv_g = np.convolve(g, g)
        M_g = conv_g.max()
        # compute optimal t
        if M + M_g == 0:
            t_opt = 0.0
        else:
            t_opt = np.sqrt(M / (M + M_g))
        # enforce limits
        t_opt = max(0.0, min(1.0, t_opt))
        # mixture candidates
        for t_factor in [t_opt, t_opt * 0.5, min(1.0, t_opt * 1.5)]:
            f_candidate = (1 - t_factor) * f + t_factor * g
            # ensure non-negative (should be)
            # compute score
            score = evaluate_sequence(f_candidate.tolist())
            if score < best_score:
                best_score = score
                best_f = f_candidate.copy()
                # accept as current for next iteration (use f_candidate)
                f = f_candidate.copy()
                break
        else:
            # No improvement. Try a small random tweak to escape.
            idx = np.random.randint(n)
            delta = np.random.randn() * (np.max(f) * 0.02 + 1e-6)
            f[idx] = max(0.0, f[idx] + delta)
            f = f * (target_sum / np.sum(f))
        iteration += 1
        if iteration % 10 == 0:
            print(f"Iter {iteration}, best_score={best_score:.6f}, n={n}")
    # finalize: ensure non-negative, sum positive.
    best_seq = best_f.tolist()
    # Clip to [0, 1000] though evaluate_sequence will clamp.
    print(f"Finished after {iteration} iterations. Best score: {best_score}")
    return best_seq

def geometric_sequence(n, alpha):
    r = 1 - alpha / n
    # ensure r>0
    if r <= 0:
        r = 0.5  # fallback
    arr = r ** np.arange(n)
    target_sum = math.sqrt(2 * n)
    scale = target_sum / arr.sum()
    arr = arr * scale
    return arr

"""
Propose a step‑function sequence with a very low evaluation score.

The evaluation score for a sequence `a` is
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

If we normalise the sequence so that sum(a)=sqrt(2*n), the denominator becomes
2*n and the score equals the maximal value of the convolution.
Therefore we only need a shape that makes the maximal autocorrelation as
small as possible.

A decreasing geometric progression
    a_i = C * r**i ,   r = 1 - α/n ,   α > 0
has a closed‑form convolution.
For large n the maximal entry is approximately

    max(conv) ≈ max( 2*α² / n , 2/(α·e) )

(the first term comes from the very first coefficient, the second from the
peak of (k+1)·r**k).  Minimising the larger of the two terms gives the
optimal α ≈ (n/e)^{1/3} and a resulting maximal convolution of order
O(n^{-1/3}) – far below the target 1.5 even for modest n.

This function builds several geometric sequences for different lengths
(1500, 3000, 5000) and α values around the theoretical optimum, evaluates
them and returns the best one found within the time budget.

The method is deterministic (except for a tiny random perturbation that
helps avoid pathological exactness) and requires only NumPy and the
provided `evaluate_sequence` function.
"""

import math
import time
import numpy as np

def _geometric_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence a_i = C * r**i,
    where r = 1 - alpha/n and C rescales the sum to sqrt(2*n).

    Parameters
    ----------
    n : int
        Length of the sequence.
    alpha : float
        Shape parameter (must be >0 and < n).

    Returns
    -------
    np.ndarray
        The generated sequence.
    """
    # Ensure a valid ratio
    if alpha <= 0:
        alpha = 1e-6
    if alpha >= n:
        alpha = n - 1e-6

    r = 1.0 - alpha / n          # 0 < r < 1
    # geometric series
    exponents = np.arange(n, dtype=float)
    raw = np.power(r, exponents)   # r**i

    # Normalise to the required sum (sqrt(2*n))
    target_sum = math.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    seq = raw * scale
    # Tiny random jitter to break perfect symmetry (helps the LP based
    # benchmark if it is ever called later)
    jitter = np.random.randn(n) * 1e-8 * scale
    seq = np.maximum(seq + jitter, 0.0)
    # Re‑normalise after jitter
    seq *= target_sum / seq.sum()
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function sequence with a low evaluation score.

    The function tries a few geometric sequences of different lengths and
    shape parameters, keeps the best one and returns it as a plain list of
    floats.  It respects the 1000‑second time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds.
    **kwargs :
        Ignored (present for compatibility with the surrounding framework).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)   # leave a few seconds margin

    best_seq = None
    best_score = float('inf')

    # Lengths to try – larger gives a lower theoretical bound but costs more.
    lengths = [1500, 3000, 5000]

    for n in lengths:
        # Theoretical optimal α ≈ (n/e)^{1/3}
        alpha_opt = (n / math.e) ** (1.0 / 3.0)

        # Try a few values around the optimum
        trial_alphas = [
            alpha_opt * 0.8,
            alpha_opt,
            alpha_opt * 1.2,
            max(0.1, alpha_opt - 1.0),
            alpha_opt + 1.0
        ]

        # Add a handful of random α values for robustness
        for _ in range(5):
            # α should stay well below n to keep r > 0
            trial_alphas.append(np.random.uniform(2.0, min(50.0, n / 2.0)))

        for alpha in trial_alphas:
            if time.time() > deadline:
                break
            seq = _geometric_sequence(n, alpha)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                # Defensive: if evaluate_sequence crashes we skip this candidate
                continue

            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:6.2f}s] New best – n={n}, "
                      f"α={alpha:.3f}, score={score:.6f}")

        if time.time() > deadline:
            break

    # Fallback: if for some reason no sequence was accepted, return a simple uniform one.
    if best_seq is None:
        n = 1500
        uniform = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = uniform.tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Finished. Best score = {best_score:.6f} with length = {len(best_seq)}")
    return best_seq
