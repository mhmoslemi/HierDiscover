def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose length n
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.array(height_sequence_1, dtype=float)
            n = base_seq.size
            if n <= 0:
                raise ValueError
        except Exception:
            n = np.random.randint(800, 2000)
            base_seq = None
    else:
        n = np.random.randint(800, 2000)
        base_seq = None

    target_avg = 1.0 / (2 * n - 1)

    best_seq = None
    best_score = float('inf')
    # Evaluate initial sequences: start from base_seq if exists, else random
    if base_seq is not None:
        seq0 = np.maximum(0.0, base_seq)
        seq0 = seq0 / (seq0.sum() + 1e-12)
        score0 = evaluate_sequence(seq0.tolist())
        best_seq = seq0.copy()
        best_score = score0
        print(f"Initial base seq score {score0:.6f}")
    else:
        # initial random
        seq0 = np.random.rand(n)
        seq0 = seq0 / seq0.sum()
        score0 = evaluate_sequence(seq0.tolist())
        best_seq = seq0.copy()
        best_score = score0
        print(f"Initial random seq score {score0:.6f}")

    # Helper functions
    def full_convolution(x):
        # Use power of two length
        L = 2 * n - 1
        fft_len = 1 << (L - 1).bit_length()
        X = np.fft.rfft(x, fft_len)
        conv = np.fft.irfft(X * X, fft_len)[:L]
        return conv

    def gradient_L2(x):
        # compute full conv
        conv = full_convolution(x)
        err = conv - target_avg
        # compute gradient: 4 * conv(err, rev(x)) slice
        rev = x[::-1]
        L_err = err.size
        L_rev = rev.size
        fft_len = 1 << (L_err + L_rev - 1 - 1).bit_length()
        E = np.fft.rfft(err, fft_len)
        R = np.fft.rfft(rev, fft_len)
        grad_full = np.fft.irfft(E * R, fft_len)[:L_err + L_rev - 1]
        # slice middle n entries
        start_idx = n - 1
        grad = 4.0 * grad_full[start_idx : start_idx + n]
        return grad, conv

    # main optimization loop
    while time.time() < deadline:
        # random restart
        seq = np.random.rand(n)
        seq = seq / seq.sum()
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.5  # base learning rate
        max_iters = 3000
        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break
            grad, conv = gradient_L2(seq)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            seq = seq - lr * m_hat / (np.sqrt(v_hat) + eps)
            # clip nonnegative and renorm
            seq = np.clip(seq, 0, None)
            total = seq.sum()
            if total <= 0:
                seq = np.random.rand(n)
                total = seq.sum()
            seq = seq / total
            # periodically evaluate

                # compute score using evaluate_sequence for reliability
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"Iter {it} new best {best_score:.6f} at time {time.time() - start:.1f}s")
        # after restart evaluate final seq

            best_score = final_score
            best_seq = seq.copy()
            print(f"Restart final improved to {best_score:.6f}")

    print(f"Finished. Best score {best_score:.6f}")
    # Return best sequence as list
    return best_seq.tolist()

    # compute conv = seq * seq
    L = 2*n - 1
    fft_len = next_pow2(L)
    X = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(X * X, fft_len)[:L]
    err = conv - avg
    # compute convolution err * rev(seq)
    rev = seq[::-1]
    L_err = L
    L_rev = n

    E = np.fft.rfft(err, fft_len2)

    grad_full = np.fft.irfft(E * R, fft_len2)[:L_err + L_rev - 1]
    start = n - 1
    grad = 4.0 * grad_full[start : start + n]
    return grad, conv

"""
Propose a sequence of non‑negative heights that minimizes the provided
evaluate_sequence function.

The algorithm works in three stages:

1. **Initialisation**
   * Use the known construction ``height_sequence_1`` if it exists,
     otherwise start from a random Dirichlet vector.
   * The vector is normalised to have sum = 1 (the objective is scale‑invariant).

2. **Projected Adam optimisation**
   * We minimise the squared L2‑error between the full self‑convolution
     ``c = a * a`` and its theoretical average value
     ``avg = 1/(2n‑1)``.  After normalisation ``avg`` is the smallest value
     any entry of the convolution can possibly attain.
   * The gradient of the L2 error can be written analytically as a
     convolution, which we evaluate with FFTs (three fast transforms per
     iteration).  An Adam step is taken and the iterate is projected back
     onto the probability simplex (non‑negative + sum = 1).
   * Every few iterations the true objective
     ``score = 2·n·max(a*a)`` is computed and the best sequence found so far
     is stored.

3. **Time‑bounded restarts**
   * The optimisation loop is repeated with fresh random restarts until the
     allocated time budget (≈1000 s) is exhausted.
   * The best sequence discovered is returned.

The method does not rely on the linear‑programming direction used in the
reference implementation; it directly attacks the flat‑convolution goal
and typically drives the evaluation value well below the baseline ``2.0``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Linear (full) convolution of ``seq`` with itself.
    Returns an array of length ``2*n-1``.
    """
    L = 2 * n - 1
    fft_len = _next_pow2(L)
    X = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(X * X, fft_len)[:L]
    return conv


def _grad_l2(seq: np.ndarray, n: int, avg: float):
    """
    Gradient of the squared L2 error
        J(seq) = || conv(seq,seq) - avg ||_2^2
    together with the current full convolution.
    Returns (grad, conv) where ``grad`` has shape (n,).
    """
    # ----- convolution a*a -----
    L = 2 * n - 1
    fft_len1 = _next_pow2(L)
    X = np.fft.rfft(seq, fft_len1)
    conv = np.fft.irfft(X * X, fft_len1)[:L]

    # ----- error -----
    err = conv - avg

    # ----- gradient = 4 * (err * rev(seq)) sliced -----
    rev = seq[::-1]
    L2 = L + n - 1                     # = 3*n-2
    fft_len2 = _next_pow2(L2)
    E = np.fft.rfft(err, fft_len2)
    R = np.fft.rfft(rev, fft_len2)
    grad_full = np.fft.irfft(E * R, fft_len2)[:L2]

    start = n - 1                      # centre of the full convolution
    grad = 4.0 * grad_full[start:start + n]
    return grad, conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """
    Computes the objective value 2·n·max(conv) for a vector normalised to sum=1.
    """
    return 2.0 * n * np.max(conv)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that aims to
    minimise ``evaluate_sequence``.
    The routine respects the time budget (≈1000 s) and uses a projected
    Adam optimiser based on flattening the self‑convolution.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (length of the sequence)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float).copy()
            if base.size == 0:
                raise ValueError
            n = base.size
            base = np.maximum(0.0, base)
            base /= max(base.sum(), 1e-12)          # normalise to sum = 1
        except Exception:
            base = None
            n = np.random.randint(2000, 5000)
    else:
        base = None
        n = np.random.randint(2000, 5000)

    avg_val = 1.0 / (2 * n - 1)          # target average of the convolution

    # ------------------------------------------------------------------
    # Helper to compute the true score (uses our own fast code)
    # ------------------------------------------------------------------
    def _fast_score(a):
        return _score_from_conv(_full_convolution(a, n), n)

    # ------------------------------------------------------------------
    # Initialise the best solution seen so far
    # ------------------------------------------------------------------
    if base is not None:
        best_seq = base.copy()
    else:
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
    best_score = _fast_score(best_seq)
    print(f"Initial score {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    max_iters_per_restart = 3000        # Adam iterations per restart
    eval_every = 50                     # how often to evaluate the true score
    base_lr = 0.5                       # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ------------------------------------------------------------------
    # Main time‑bounded loop with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Occasionally start from the current best to refine it
        if np.random.rand() < 0.15 and best_seq is not None:
            a = best_seq.copy()
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

            it += 1

            # ----- gradient and current convolution -----
            grad, conv = _grad_l2(a, n, avg_val)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # ----- projection onto the simplex (non‑negative, sum=1) -----
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s <= 0.0:
                a = np.random.rand(n)
                s = a.sum()
            a /= s

            # ----- occasional evaluation -----
            if it % eval_every == 0 or it == max_iters_per_restart:
                cur_score = _fast_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(f"Iter {it:4d}  new best {best_score:.6f}  time {elapsed:.1f}s")

        # End of this restart – evaluate once more
        final_score = _fast_score(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            elapsed = time.time() - start_time
            print(f"Restart finished with improved score {best_score:.6f}  time {elapsed:.1f}s")

    # ------------------------------------------------------------------
    # Final safety check (ensure non‑negative, reasonable magnitude)
    # ------------------------------------------------------------------
    best_seq = np.maximum(0.0, best_seq)
    total = best_seq.sum()
    if total == 0.0:
        best_seq = np.ones_like(best_seq) / best_seq.size
    else:
        best_seq /= total

    print(f"Finished optimisation – best score {best_score:.6f}")
