"""
Propose a near‑optimal step‑function for the given evaluation.

The evaluation function is 2 n·max(conv(a,a))/sum(a)^2 and is invariant under a
global scaling of the sequence.  For a fixed length n the uniform sequence
(a_i = 1/n) gives a score of exactly 2.
A two‑color alternating pattern (different values on even and odd indices)
can improve this value slightly.  The optimal alternating pattern for an even
length n = 2·m solves the simple quadratic
(3 + c)·t² – 4·t + 1 = 0  with  c = (m‑1)/m,
giving

    t = (2·m – √m) / (4·m – 1)          (0 ≤ t ≤ 1)

If we put `a_even = 2·t / n` on even positions and
`a_odd  = 2·(1 – t) / n` on odd positions, the sum of the sequence is 1,
the entries are non‑negative and the score becomes

    score = max(8·t·(1–t), 4·[t² + c·(1–t)²], 4·[c·t² + (1–t)²])  ≤ 2.

For large n this improves the score by approximately 1/n (e.g.
n=1526 → score≈1.99935).  The function builds this alternating sequence for the
length of the supplied `height_sequence_1` (or a default length) and returns it.
"""

import time
import numpy as np

# height_sequence_1 may be provided by the environment; if not, we fall back.
try:
    _default_len = len(height_sequence_1)
except NameError:
    _default_len = 200  # a modest default length


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence that improves the current best score.

    Parameters
    ----------
    seed : int, optional
        Random seed (unused for the deterministic construction but kept for API compatibility).
    budget_s : float, optional
        Time budget in seconds.  The algorithm is deterministic and finishes instantly.
    kwargs : dict
        Ignored keyword arguments.

    Returns
    -------
    list[float]
        The constructed non‑negative sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # small safety margin

    # ------------------------------------------------------------------
    # Choose the length.
    # ------------------------------------------------------------------
    n = _default_len
    # Ensure an even length for the alternating construction.
    if n % 2 == 1:
        n -= 1
    if n <= 2:
        # fallback to a uniform tiny sequence if n is too small
        seq = [1.0 / n] * n
        print("Fallback uniform sequence, length:", n)
        return seq

    m = n // 2                      # number of even (and odd) positions
    c = (m - 1) / m                 # coefficient that appears in the formula

    # ------------------------------------------------------------------
    # Optimal t for the two‑color alternating pattern.
    # ------------------------------------------------------------------
    # Solve (c+3)·t² - 4·t + 1 = 0 and take the smaller root in [0, 0.5].
    # Exact solution:
    sqrt_term = np.sqrt(1.0 / m)    # sqrt(1 - c) = sqrt(1/m)
    t_opt = (4.0 - 2.0 * sqrt_term) / (2.0 * (c + 3.0))
    # Equivalent closed form: (2*m - sqrt(m)) / (4*m - 1)
    # Use the more stable computed version.
    t_opt = max(0.0, min(1.0, t_opt))  # safety clamp

    # ------------------------------------------------------------------
    # Build the alternating sequence.
    # ------------------------------------------------------------------
    a_even = (2.0 * t_opt) / n
    a_odd = (2.0 * (1.0 - t_opt)) / n

    # Ensure non‑negativity with a tiny epsilon guard.
    a_even = max(a_even, 0.0)
    a_odd = max(a_odd, 0.0)

    sequence = [a_even if (i % 2 == 0) else a_odd for i in range(n)]

    # ------------------------------------------------------------------
    # Optional local refinement (tiny hill‑climbing) within the time budget.
    # ------------------------------------------------------------------
    # The improvement over the analytical optimum is marginal, but we
    # perform a few quick random swaps to possibly lower the score a bit.
    best_seq = list(sequence)
    best_score = evaluate_sequence(best_seq)

    # Simple hill‑climbing: move a small mass between two random indices.
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving a fraction of the smaller entry to the larger one
        delta = 0.01 * min(best_seq[i], best_seq[j])
        if delta == 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # maintain non‑negativity (already guaranteed by construction)
        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq
            # Occasionally print progress
            print(f"Improved score: {best_score:.6f}")

    print(f"Final length: {n}, final score: {best_score:.6f}")
    return best_seq
