            # indices where M achieved (allow slight tolerance)
            max_idx = np.where(b >= M - 1e-12)[0]
            # compute gradient

            g = np.zeros_like(cur_seq)
            N = len(cur_seq)

                i_start = max(0, k - (N - 1))
                i_end = min(N - 1, k)
                i_range = np.arange(i_start, i_end + 1)
                g[i_range] += 2 * cur_seq[k - i_range]
            # step size adapt
            # compute gradient norm
            norm = np.linalg.norm(g)
            if norm == 0:
                break
            # step factor relative to current values
            # scale such that maximum update < 10% of max(a)
            max_a = cur_seq.max()
            step = 0.2 * max_a / (norm / N)  # approx average step per entry
            # ensure step not too large
            step = min(step, 0.5)  # bound
            new_seq = cur_seq - step * g
            # clip and renormalize
            new_seq = np.maximum(new_seq, 0)
            s = new_seq.sum()
            if s == 0:
                # stay in current
                break
            new_seq = new_seq / s
            new_val = evaluate_sequence(new_seq.tolist())
            if new_val < cur_val:
                cur_seq = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = new_seq.copy()
                    print(f"New global best: {best_val:.6f}, time {time.time()-start_time:.2f}s")

                # not improvement, try smaller step or random perturb

                if np.random.rand() < 0.1:

                    i, j = np.random.choice(N, 2, replace=False)

import numpy as np
import time
import math

def safe_evaluate(seq):
    """Call the provided ``evaluate_sequence`` safely."""
    try:
        # ``evaluate_sequence`` is supplied in the execution environment.
        return evaluate_sequence(seq.tolist())
    except Exception:                     # pragma: no cover
        return math.inf

def gradient_max(a, max_idx):
    """
    Gradient of the maximal convolution entry w.r.t. the vector ``a``.
    For a given index ``k`` the derivative ∂b_k/∂a_i = 2·a_{k‑i}
    (zero when the index is out of range).  The gradient is the sum of
    these contributions over all indices attaining the current maximum.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        # i_range  = i_start … i_end  (inclusive)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        g[i_range] += 2.0 * a[k - i_range]
    return g
