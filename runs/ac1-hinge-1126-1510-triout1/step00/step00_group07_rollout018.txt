"""
Propose a candidate sequence that (approximately) minimizes the evaluation function.

The algorithm is a simple projected sub‑gradient descent on the
max‑convolution.  For a given sequence `a` we compute the convolution
`b = a * a`, find the index `k` where `b` attains its maximum and take a
sub‑gradient of this maximum w.r.t. `a` (which is simply `2 * a` shifted
by `k`).  We then take a small step opposite to this gradient,
project back onto the probability simplex (non‑negative entries that
sum to 1) and keep the best sequence seen so far.  Several random
initialisations (including the provided `height_sequence_1`) are tried
until the time budget expires.

The method guarantees a valid return (a non‑empty list of non‑negative
floats) and works within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a 1‑D numpy array onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the largest rho satisfying the KKT condition
    rho_array = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_array.size == 0:
        theta = 0.0
    else:
        rho = rho_array[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

            b = np.convolve(a, a)
            k = int(np.argmax(b))                     # index of the maximal entry

            # Gradient of b_k w.r.t. a_i is 2 * a_{k-i} (if the index exists)
            g = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            g[idx] = 2.0 * a[k - idx]

            # --------------------------------------------------------------
            # Take a step opposite to the gradient and re‑project onto the simplex.
            # A slowly decreasing step size works well in practice.
            # --------------------------------------------------------------
            step = 0.02 / (1.0 + it * 0.001)           # diminishing learning rate
            a_new = a - step * g

            # Clip negatives before projection (projection will also enforce non‑negativity)
            a_new = np.maximum(a_new, 0.0)

            if a_new.sum() == 0.0:
                # Rare pathological case – fall back to a fresh random point
                a_new = np.random.dirichlet(np.ones(n))
            else:
                a_new = _project_onto_simplex(a_new)

            # Occasionally add a tiny random jitter to escape shallow plateaux
            if it % 250 == 0 and it > 0:
                jitter = np.random.normal(scale=0.001, size=n)
                a_new = np.maximum(a_new + jitter, 0.0)
                a_new = _project_onto_simplex(a_new)

            a = a_new

    # ----------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) return a uniform vector.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = [1.0 / n] * n
        print("Fallback to uniform sequence.")

    return best_seq
