def conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1]
```

"""
Propose a candidate step‑function sequence.

The algorithm performs a fast gradient‐descent on a smooth approximation of
the max‑convolution objective.  For a sequence a of length n we keep the
ℓ₁ norm fixed to √(2 n) (the scale used by the evaluator).  The objective
is   max_k (a*a)[k] .  We approximate the max by a soft‑max with temperature β
and compute the gradient analytically.  Each iteration:

1. Compute the convolution a*a via an FFT (O(n log n)).
2. Build soft‑max weights w_k ∝ exp(β·(conv_k – max(conv))).
3. The gradient of the smoothed max is
      ∇L = 2·conv_rev[n‑1 : n‑1+n]   where conv_rev = a_rev * w .
   (a_rev is a reversed copy of a.)  This also uses an FFT.
4. Take a step a ← a – η·∇L, clip to non‑negative, and renormalise the sum
   back to √(2 n).

The step size η is scaled by the inverse gradient norm; β is slowly
increased so the smooth approximation becomes tighter.  Random restarts
and an occasional tiny random jitter help escape plateaus.  The best
sequence found (lowest observed max‑convolution) is returned as a plain
list of floats.

The routine respects the 1000 s budget, prints progress, and works with a
default length of 2000 (fast enough to run many thousands of iterations
while still allowing the theoretical bound ≈ 1.5 to be approached).
"""

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the (full) convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]                       # length 2n‑1


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a and a weight vector w (length 2n‑1), compute
    grad = 2 * (a_rev * w)[n‑1 : n‑1+n]   (real convolution).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A_rev * W, n=size)
    start = n - 1
    return 2.0 * conv[start : start + n]
