def propose_candidate(seed=..., budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 2000  # can also vary n perhaps 3000 etc.

    # Initialize a random initial vector: small random values, normalized.
    a = np.random.rand(n)
    a = a / a.sum()  # sum=1

    # Compute objective and best.
    best = a.copy()
    best_score = compute_obj(a)

    # Set parameters:
    step = 0.1  # initial step size
    decay = 0.9995

    # Main loop
    while time.time() < deadline:
        # compute convolution
        conv = np.fft.ifft(np.fft.fft(a, n*2) * np.fft.fft(a, n*2)).real
        # conv length 2n-1 (we can ignore extra zero tail)
        conv = conv[:2*n-1]
        # find max index and value
        k_max = np.argmax(conv)
        max_val = conv[k_max]

        # compute gradient g: for each i, if 0 <= k_max - i < n: g[i] = a[k_max-i], else 0.
        # We can compute as reversed slice
        # g_i = a[k_max - i] if k_max - i in [0, n-1] else 0
        # Equivalent to reversed version of a for relevant shift.

        # vectorized:
        g = np.zeros_like(a)
        low = max(0, k_max - (n-1))
        high = min(k_max, n-1)
        # i ranges from low to high
        i_range = np.arange(low, high+1)
        j = k_max - i_range  # indices in a
        g[i_range] = a[j]

        # gradient of max_b wrt a is 2*g (including factor for i=j case where i=j => g = a_i).
        # Use step size
        eta = step
        # Update a_new = a - eta * 2 * g
        a_new = a - 2 * eta * g
        a_new = np.maximum(a_new, 0.0)
        # Renormalize sum to 1
        sum_new = a_new.sum()
        if sum_new == 0:
            # Reset to random
            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / sum_new

        # compute new objective
        new_score = compute_obj(a_new)
        # Accept if better or with prob
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            print(f"New best {best_score:.6f}")
        # Simple acceptance: if new_score < current_score, accept; else reject
        # we need to keep current a for next iteration
        if new_score < compute_obj(a):
            a = a_new
        else:
            # maybe occasional random restart
            if np.random.rand() < 0.01:
                a = np.random.rand(n)
                a = a / a.sum()
        # decay step
        step *= decay

    result = best.tolist()
    return result

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # Build a_ub matrix: each shift k yields sum_i f_i * b_{k-i} <= rhs
    # For each k (0..2n-2), row_j = f_i where j = k-i, i from 0..n-1
    # Equivalent to building Toeplitz matrix.
    rows = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # For i in [i_min,i_max], j = k - i
        row_indices = np.arange(i_min, i_max+1)
        j_indices = k - row_indices
        row[j_indices] = f_seq[row_indices]
        rows.append(row)
        b_ub.append(rhs)
    # Nonnegativity constraints: -I * b <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a candidate sequence for the step‑function attack.

This implementation combines two simple optimisation ideas:

1. **Linear‑programming improvement (Frank‑Wolfe step).**
   From a current non‑negative sequence `f` we solve an LP that finds a
   direction `g` with maximal ℓ₁‑mass while keeping the convolution
   `f * g` bounded by the current worst convolution value `M`.
   Mixing `f` and `g` (with a decreasing step size) never increases `M`
   and typically reduces it.

2. **Fast stochastic refinement.**
   After a number of LP steps the algorithm switches to a cheap simulated‑annealing
   that perturbs the coefficients, accepts improving moves (or occasional
   uphill moves) and slowly cools the temperature.  The score used during the
   stochastic phase is computed with an FFT‑based convolution, which is much
   faster than `evaluate_sequence`.

Both phases respect the non‑negativity and the scale‑invariance of the
objective (the evaluation function is invariant under scaling, therefore we
renormalise the sequence to have unit ℓ₁‑sum after each update).

The routine runs for the allotted `budget_s` (default 1000 s) and returns
the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------
# Helper: fast evaluation of the objective (same as evaluate_sequence
# but without the security checks).  Used only for internal optimisation.
def _fast_score(seq: np.ndarray) -> float:
    """Return the value of 2 * n * max(conv) / (sum(seq)^2) using FFT."""
    n = len(seq)
    # length for linear convolution
    fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    max_b = conv[: 2 * n - 1].max()
    s = seq.sum()
    if s < 0.01:               # never happens in our runs, but keeps same guard
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# --------------------------------------------------------------
# Linear programming step: maximise sum(b) s.t. conv(f,b) <= M, b>=0.
def _solve_convolution_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise    sum(b)
        subject to  (f * b)[k] <= M   for all k
                    b >= 0
    Returns the optimal b (as a 1‑d ndarray) or None if the LP fails.
    """
    n = f.shape[0]
    # maximise sum(b) -> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    # For each shift k we have   Σ_i f[i] * b[k-i] <= M,
    # where the term is present only if both indices are in [0,n).
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals  # corresponding column indices
        row[j_vals] = f[i_vals]
        rows.append(row)
        rhs.append(M)

    # Non‑negativity constraints  -b <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([rhs, np.zeros(n)])

    # Use the modern HiGHS solver – fast and robust for these dense systems.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs", options={"presolve": True})
    if res.success:
        return res.x
    return None


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int

    Returns

        The best height sequence found within the time budget.
    """
    np.random.seed(seed)

    #     but also larger linear programmes.  n = 500 is a good compromise.
    n = 500

    #     Start from a random positive vector and normalise it to unit ℓ₁‑sum.
    cur = np.random.rand(n) + 1e-6
    cur = cur / cur.sum()

    # Keep the best solution seen so far.
    best_seq = cur.copy()
    best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # 3.  Phase‑1 : deterministic LP improvement (Frank‑Wolfe style).
    #
    #    We perform a decreasing number of LP steps.  The mixing factor `t`
    #    starts at 0.2 and is multiplied by 0.9 every 20 successful steps.
    t = 0.20

        cur_score = _fast_score(cur)
        nxt_score = _fast_score(nxt)

        if nxt_score < cur_score:
            cur = nxt
            # Record a new best if it improves the true objective.
            true_score = evaluate_sequence(cur.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = cur.copy()
                print(f"[LP] step {step_counter:3d} – new best = {best_score:.6f}")

        # Reduce the step size slowly.
        if step_counter % 20 == 0 and t > 0.005:
            t *= 0.90

        step_counter += 1

    # ------------------------------------------------------------------
    # 4.  Phase‑2 : cheap stochastic refinement (simulated annealing).
    #
    #    Temperatures are reduced geometrically; each iteration either
    #    swaps two coefficients or perturbs a single coefficient.
    temperature = 0.1
    iteration = 0
    while time.time() < deadline:
        # Propose a neighbour.
        cand = cur.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.choice(n, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # local perturbation
            i = np.random.randint(n)
            delta = np.random.randn() * 0.05 * cand[i]   # relative noise
            cand[i] = max(0.0, cand[i] + delta)

        cand = cand / cand.sum()                       # renormalise

        cur_score = _fast_score(cur)
        cand_score = _fast_score(cand)

        # Metropolis acceptance.
        if cand_score < cur_score or np.random.rand() < math.exp(
            (cur_score - cand_score) / max(temperature, 1e-12)
        ):
            cur = cand
            # Update the global best if needed (using the official evaluator).
            true_score = evaluate_sequence(cur.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = cur.copy()
                print(f"[SA] iter {iteration:5d} – new best = {best_score:.6f}")

        # Cooling schedule.
        if iteration % 200 == 0 and temperature > 1e-4:
            temperature *= 0.95

        iteration += 1

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found (as a plain Python list of floats).
    return best_seq.tolist()
