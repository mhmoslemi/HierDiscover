"""
Propose a better step‑function sequence.

The algorithm works directly on the probability vector p that sums to 1.
The evaluation of a sequence a is

    score = 2 * n * max(conv(p,p))

where conv(p,p) is the (non‑cyclic) convolution of p with itself.
For a uniform p we obtain score = 2, which is the current best.
To improve, we minimise the maximum entry of the convolution.
This is equivalent to minimising a smooth approximation of the max:
    f(p) = (1/α)·log( Σ_k exp(α·conv_k) ).

∇f(p) can be written in closed form using a correlation,
so a projected (multiplicative) gradient descent is cheap:
    p ← p * exp( -η·∇f(p) )
    p ← simplex‑project(p)               (non‑negative, Σp = 1)

α starts small (very smooth) and is increased slowly,
so the optimisation first moves towards a flatter distribution
and then concentrates on the true maximum.
The procedure runs until the allocated time runs out and returns the
best sequence found (scaled arbitrarily – the score is scale‑invariant).
"""

import time
import numpy as np
from scipy import signal  # for fast correlation (cross‑correlation)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, Σ x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]           # last true index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        try:
            cur_score = evaluate_sequence(p.tolist())
        except Exception:               # any unexpected failure – skip this iteration
            continue

        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[{iteration:05d}] improved score: {best_score:.6f}")

        # ---- schedule ----------------------------------------------------
        # slowly tighten the smooth‑max (increase α) and gradually shrink η
        if alpha < 200.0:
            alpha *= 1.02
        eta = max(eta * 0.999, 0.001)

    # Return the best sequence found (any positive scaling works;
    # we keep the probability vector itself).
    return best_seq.tolist()
