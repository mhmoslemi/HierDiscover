def adam_optimize_extended(y_start, lam_factors, steps_per, deadline, n, fft_len, best_a, best_score, rng):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per):
            if time.time() >= deadline:
                return best_a, best_score
            _, grad_y = objective_and_grad(y, n, fft_len, lam)
            t += 1
            m = beta1*m + (1-beta1)*grad_y
            v = beta2*v + (1-beta2)*(grad_y**2)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            lr = 0.3/ (np.sqrt(v_hat)+eps)   # initial lr scaling
            y = y - lr*m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            s = fast_score(a, fft_len)
            if s < best_score:
                best_score = s
                best_a = a.copy()
    return best_a, best_score

def pgd_optimize_max(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    while time.time() < deadline:
        # compute convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # mask of values within epsilon fraction of max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # gradient: 2 * reverse a * mask (as earlier)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = 2.0 * np.real(np.fft.ifft(fa_rev * fm))[n-1:n-1+n]   # shape n
        # step size schedule
        eta = 0.5 / np.sqrt(it + 1)
        a_new = a - eta * grad
        # project
        a_new = simplex_projection(a_new)
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

def blend_with_uniform(a, current_score, n, fft_len):
    u = np.full(n, 1.0/n)
    best_local = a
    best_local_score = current_score

"""
Propose a low‑evaluation step‑function sequence.

The algorithm works in several phases within the 1000 s budget:

1. **Random starts** – many diverse initial probability vectors are drawn.
2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp smooth
   approximation of the max‑convolution while staying on the simplex.
3. **Projected sub‑gradient descent** on the true max‑convolution
   (Euclidean projection onto the simplex after each step).
4. **Greedy hill‑climb** that moves a tiny fraction of mass from a heavy
   index to a lighter one, updating the full convolution incrementally.
5. **Uniform blending** – a cheap 1‑D search of convex combinations of the
   best vector with the uniform distribution.

All phases keep the vector on the probability simplex, therefore the final
score is simply `2·n·max(conv(a,a))`.  The best sequence found before the
deadline is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "sin",
            "bell",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = math.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "sin":
        i = np.arange(n)
        a = np.sin(np.pi * (i + 0.5) / (n + 0.5))
    else:  # bell
        sigma = rng.uniform(0.1, 0.35) * n
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) w.r.t. y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]  # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (many steps, adaptive schedule)
# ----------------------------------------------------------------------


def adam_optimize_extended(
    y_start: np.ndarray,
    lam_schedule,
    steps_per_lam: int,
    deadline: float,
    n: int,
    fft_len: int,
    best_a: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    lr0: float = 0.15,
):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Runs through the whole lam_schedule, many steps per λ.
    Returns the overall best a and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            _, grad_y = objective_and_grad(y, n, fft_len, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)
            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def pgd_optimize_max(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 1.0,
):
    """
    Projected gradient descent on the true max‑convolution.
    Returns the best a encountered.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Greedy hill‑climb that moves a small fraction of mass from a heavy index
    to a lighter one, accepting only improvements.  Convolution updates
    are performed incrementally.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)  # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff = np.zeros_like(conv)
        diff[src : src + n] -= best
        diff[dst : dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Blend with the uniform distribution (tiny 1‑D line search)
# ----------------------------------------------------------------------


def blend_with_uniform(a: np.ndarray, cur_score: float, n: int, fft_len: int):
    """Search convex combinations (1‑α)*a + α*uniform for a better score."""
    uniform = np.full(n, 1.0 / n)
    best_a = a.copy()
    best_score = cur_score
    for alpha in np.linspace(0.0, 1.0, 51):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        s = fast_score(cand, fft_len)
        if s < best_score:
            best_score = s
            best_a = cand.copy()
    return best_a, best_score
