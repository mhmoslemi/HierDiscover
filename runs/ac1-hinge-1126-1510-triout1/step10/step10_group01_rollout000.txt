"""
Propose a step‑function sequence that (approximately) minimizes

        2·n·max(a*a) / (sum(a)**2)

under the constraints of ``evaluate_sequence``.

The algorithm consists of three main phases:

1. **Rich parametric sweep** (≈15 % of the time budget)  
   Randomly draws vectors from a collection of one‑parameter and two‑parameter
   families (Beta, power‑law, exponential, Gaussian, tent, etc.) as well as
   random convex combinations of two families.  All vectors are normalised to
   unit sum and the best (lowest fast score) is kept.

2. **Smooth‑max L‑BFGS refinement** (≈45 % of the time budget)  
   The current best vector is re‑parameterised by a soft‑max of logits, thus
   enforcing non‑negativity and unit sum.  A log‑sum‑exp smoothing of the
   max‑convolution is minimised with a long schedule of decreasing smoothing
   parameters.  The L‑BFGS routine runs many iterations per ``lam`` value,
   progressively approaching the true non‑smooth objective.

3. **Greedy stochastic local search** (remaining time)  
   Simple pair‑wise mass transfers are attempted; a move is accepted only if
   it strictly improves the true objective (computed via an FFT based
   convolution).  The step size gradually shrinks, allowing a fine‑grained
   hill‑climb around the L‑BFGS optimum.

All phases respect the global deadline (default 1000 s).  The best candidate
seen at any moment is finally evaluated with the official ``evaluate_sequence``
function and returned as a list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(a*a) assuming ∑a = 1.
    a is a non‑negative vector of length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Parametric families (return a normalised non‑negative vector)
# ----------------------------------------------------------------------


def family_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(-4.0, 4.0)
    inc = rng.choice([True, False])
    i = np.arange(n, dtype=float)
    a = (i + 1 if inc else n - i) ** p
    return a


def family_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0003, 0.025)
    i = np.arange(n, dtype=float)
    return np.exp(-rate * i)


def family_gauss(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(150.0, 2500.0)
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def family_cos(n: int, rng: np.random.Generator) -> np.ndarray:
    width = rng.uniform(200.0, n / 2.0)
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
    return np.maximum(a, 0.0)


def family_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑like shape: x^{α‑1} (1‑x)^{β‑1} on [0,1]."""
    a = rng.uniform(0.2, 8.0)
    b = rng.uniform(0.2, 8.0)
    x = np.linspace(0.0, 1.0, n)
    # avoid exact 0/1 singularities
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    return (x ** (a - 1)) * ((1 - x) ** (b - 1))


def family_tent(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric tent (triangular) possibly raised to a power."""
    mu = rng.uniform(0.0, n - 1)
    w = rng.uniform(0.2 * n, 0.6 * n)
    p = rng.uniform(0.5, 3.0)
    i = np.arange(n, dtype=float)
    a = np.maximum(0.0, 1.0 - np.abs(i - mu) / w) ** p
    return a


def family_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    mu1 = rng.uniform(0.0, n / 2.0)
    mu2 = rng.uniform(n / 2.0, n - 1)
    sigma1 = rng.uniform(80.0, 500.0)
    sigma2 = rng.uniform(80.0, 500.0)
    i = np.arange(n, dtype=float)
    a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    return a1 + a2


def family_random_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    prob = rng.uniform(0.02, 0.25)
    mask = rng.random(n) < prob
    if not np.any(mask):
        mask[0] = True
    return mask.astype(float)


def family_random(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def generate_family(name: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Dispatch to a family generator."""
    if name == "power":
        return family_power(n, rng)
    if name == "exp":
        return family_exp(n, rng)
    if name == "gauss":
        return family_gauss(n, rng)
    if name == "cos":
        return family_cos(n, rng)
    if name == "beta":
        return family_beta(n, rng)
    if name == "tent":
        return family_tent(n, rng)
    if name == "bimodal":
        return family_bimodal(n, rng)
    if name == "binary_sparse":
        return family_random_sparse(n, rng)
    if name == "random":
        return family_random(n, rng)
    # fallback
    return rng.random(n)


def mixture_two_families(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a convex combination of two independent family samples."""
    fams = [
        "power",
        "exp",
        "gauss",
        "cos",
        "beta",
        "tent",
        "bimodal",
        "binary_sparse",
        "random",
    ]
    f1 = generate_family(rng.choice(fams), n, rng)
    f2 = generate_family(rng.choice(fams), n, rng)
    w = rng.random()
    a = w * f1 + (1.0 - w) * f2
    return a

# Parametric sweep (cheap)
# ----------------------------------------------------------------------


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Explore a rich set of parametric families (including mixtures)."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + 0.15 * budget_s  # 15 % of total budget

    family_names = [
        "power",
        "exp",
        "gauss",
        "cos",
        "beta",
        "tent",
        "bimodal",
        "binary_sparse",
        "random",
    ]

    while time.time() < deadline:
        if rng.random() < 0.4:
            a = mixture_two_families(n, rng)
        else:
            a = generate_family(rng.choice(family_names), n, rng)

        total = a.sum()
        if total == 0.0:
            continue
        a = a / total
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS refinement on smoothed max
# ----------------------------------------------------------------------


def lbfgs_refinement(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for a series of decreasing smoothing parameters.
    The vector is internally represented by logits y = log(a+ε) to enforce
    positivity and the simplex constraint via softmax.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] improved fast_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb on the true objective
# ----------------------------------------------------------------------


def stochastic_hillclimb(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass‑transfer hill‑climb.
    A move is accepted only if it strictly improves the true objective.
    The step size shrinks as time runs out.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial maximal step fraction (relative to the donor mass)
    step_frac = 0.4

    while time.time() < deadline:
        # linearly decay the step fraction
        remaining = max(deadline - time.time(), 0.0)
        step_frac = 0.4 * (remaining / (deadline - 0.0))

        i = rng.integers(n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # amount to transfer
        delta = a[i] * rng.random() * step_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
                print(f"[hill] new best fast_score={best_score:.6f}")

    return best_a, best_score

            a_start = np.maximum(a_start, 0.0)
            if a_start.sum() == 0.0:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start /= a_start.sum()
        else:
            # fresh random family start
            a_start = generate_family("random", n, rng)
            a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_refinement(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – greedy stochastic hill‑climb on the true objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = stochastic_hillclimb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase2_end,
            rng,
        )
        print(f"[phase2] hill‑climb best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
