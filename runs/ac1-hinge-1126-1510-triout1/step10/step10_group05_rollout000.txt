"""
Creator:  ChatGPT – 2024‑06‑27
Purpose:  Search for a non‑negative step‑function (probability vector) that
          minimises 2·n·max(conv(a,a))  (equivalently the score returned by
          ``evaluate_sequence`` after normalising the sum to 1).

Algorithm (≈1000 s budget)

1. **Initial pool** – start from the previously best construction (if any)
   and a few hand‑crafted shapes (exponential, triangular, uniform).

2. **L‑BFGS on a log‑sum‑exp smoothed max** – many random restarts.
   For each restart a cascade of decreasing smoothing parameters
   ``lam_factor`` is applied; each stage is solved with ``scipy.optimize.minimize``.
   The best vector found is kept.

3. **Exponentiated‑gradient descent on the true max‑convolution** – three
   runs with different initial learning‑rates (η₀).  This mirrors the
   sub‑gradient step while staying on the simplex.

4. **Simulated‑annealing hill‑climb** – the remaining time is spent on a
   temperature‑controlled random walk that moves a fraction of mass from a
   source index to a destination index.  The convolution is updated
   incrementally (FFT based) and uphill moves are occasionally accepted.

The programme prints intermediate scores and finally returns the best
sequence (as a plain ``list[float]``) found before the deadline.

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.

    Variables are logits ``y`` where ``a = softmax(y)`` lives on the simplex.
    Returns a tuple (objective, gradient w.r.t. y).
    """
    # soft‑max
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter – proportional to the current max value
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z
    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # gradient w.r.t. a before softmax

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in ``lam_schedule``,
    starting each stage from the previous result.
    Returns the best vector and its fast_score achieved during the cascade.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,

    """

        else:

            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb with simulated annealing.  Moves a random fraction
    of mass from a source index to a destination index, updates the
    convolution incrementally, and occasionally recomputes the exact
    convolution to avoid drift.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)            # exact convolution (length 2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule parameters
    T0 = max((best_score - 1.0) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1200
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # linear cooling
        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # ensure we move from a larger entry to a smaller one (helps convergence)
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (exact up to O(delta²) which is negligible for small delta)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries gives a good balance between quality
    #                and per‑iteration cost.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run (if any) and from a few hand‑crafted shapes.
    # --------------------------------------------------------------
    init_candidates = []

    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the current length if needed.
            if old.size != n:
                x_old = np.linspace(0.0, 1.0, old.size)
                x_new = np.linspace(0.0, 1.0, n)
                old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            old /= old.sum()
            init_candidates.append(old)

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # exponential centred at the middle
    centre = (n - 1) / 2.0
    for alpha in [0.001, 0.002, 0.003, 0.004, 0.005]:
        a_exp = np.exp(-alpha * np.abs(np.arange(n) - centre))
        a_exp /= a_exp.sum()
        init_candidates.append(a_exp)

    # simple triangular shape
    tri = np.maximum(0.0, 1.0 - np.abs(np.arange(n) - centre) / (centre + 1.0))
    tri /= tri.sum()
    init_candidates.append(tri)

    # pick the best of the initial pool
    best_a = None
    best_score = np.inf
    for a in init_candidates:
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # 55 % for intensive L‑BFGS, 20 % for exponentiated‑gradient,
    # remainder for simulated annealing.
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.75 * budget_s
    # phase3 goes up to the final deadline

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS with many restarts
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
                    0.0015, 0.0008, 0.0004, 0.0002, 0.0001,
                    0.00005]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # With 30 % probability perturb the current best, otherwise random start
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (multiple η₀)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        eta0_values = [12.0, 20.0, 30.0]
        for eta0 in eta0_values:
            a_exp, score_exp = exp_gradient_optimize(
                best_a,
                phase2_end,
                n,
                fft_len,
                rng,
                eta0=eta0,
            )
            if score_exp < best_score:
                best_score = score_exp
                best_a = a_exp.copy()
                print(f"[exp‑grad η0={eta0}] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        start_score = fast_score(best_a, fft_len)
        a_sa, score_sa = simulated_hill_climb(
            best_a,
            start_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        if score_sa < best_score:
            best_score = score_sa
            best_a = a_sa.copy()
            print(f"[anneal] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
