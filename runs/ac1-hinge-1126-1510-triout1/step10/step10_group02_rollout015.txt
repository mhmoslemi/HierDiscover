def linear_program_step(f, M, t, max_constraints, deadline):
    n = f.shape[0]
    # compute conv f*f
    conv = np.convolve(f, f)
    # select indices with conv >= M - rel_eps * M (e.g., 0.1)
    rel_eps = 0.1
    active = np.where(conv >= M - rel_eps * M)[0]
    if active.size > max_constraints:
        # keep largest conv values indices
        idx = np.argsort(-conv[active])[:max_constraints]
        active = active[idx]
    A = np.zeros((len(active), n), dtype=float)
    for row_idx, s in enumerate(active):
        # compute row of convolution linear coefficients
        # for each j, coefficient = f[s - j] if within range
        lo = max(0, s - (n - 1))
        hi = min(n - 1, s)
        # indices j in [lo, hi]
        js = np.arange(lo, hi + 1, dtype=int)
        coefs = f[s - js]
        A[row_idx, js] = coefs
    b = np.full(len(active), M, dtype=float)
    c = -np.ones(n)
    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs', options={'time_limit': max(0, deadline - time.time())})
    if not res.success:
        return None
    g0 = res.x
    sum_g = g0.sum()
    if sum_g == 0:
        return None
    g = g0 / sum_g
    # mix
    new_f = (1 - t) * f + t * g
    return new_f

    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Work in the log‑space y where a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # keep the central n entries (which correspond to derivatives w.r.t. a)
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print progress – not essential inside the phase
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0,
                          max_iters: int = 10_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact L∞‑objective.
    Runs until the deadline or max_iters is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and active mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming tightening step (single shot)
# ----------------------------------------------------------------------


def lp_tighten_step(f: np.ndarray,
                    M: float,
                    t: float,
                    n: int,
                    deadline: float,
                    max_constraints: int = 1500,
                    rel_eps: float = 0.10):
    """
    Solve a linear programme that finds a vector g (non‑negative) such that
    (f * g)_max ≤ M while maximizing Σ g_j.  The resulting g is normalised
    and mixed back into f by convex combination (1‑t)·f + t·g.
    """
    # current self‑convolution
    conv_self = np.convolve(f, f)
    # active constraints – indices where the bound is nearly tight
    active = np.where(conv_self >= M - rel_eps * M)[0]
    if active.size == 0:
        return None
    # limit the number of constraints for speed
    if active.size > max_constraints:
        # keep the largest values (most critical)
        order = np.argsort(-conv_self[active])[:max_constraints]
        active = active[order]

    # Build A_ub matrix (|active| × n)
    A = np.zeros((len(active), n), dtype=float)
    for row, s in enumerate(active):
        lo = max(0, s - (n - 1))
        hi = min(n - 1, s)
        js = np.arange(lo, hi + 1, dtype=int)
        A[row, js] = f[s - js]

    b = np.full(len(active), M, dtype=float)
    c = -np.ones(n)                     # maximise Σ g_j  ⇔ minimise −Σ g_j
    bounds = [(0, None)] * n

    # Simple time safeguard – if we are about to run out of budget, skip LP
    if time.time() > deadline - 1.0:
        return None

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if not res.success:
        return None
    g0 = res.x
    total = g0.sum()
    if total <= 0.0:
        return None
    g = g0 / total                # normalise to a probability vector
    new_f = (1.0 - t) * f + t * g
    return new_f


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but effective)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled from a fraction of the initial best score.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))

        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic refinement (zero‑temperature hill‑climb)
# ----------------------------------------------------------------------


def greedy_local_refine(a_start: np.ndarray,
                       start_score: float,
                       deadline: float,
                       n: int,
                       rng: np.random.Generator):
    """
    Repeatedly tries a few deterministic mass‑transfer moves that only
    accept improvements.  The move set is small (largest→smallest element)
    and the transferred fraction is reduced over time.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    # candidate fractions (starting relatively large)
    fractions = np.array([0.5, 0.2, 0.1, 0.05, 0.02, 0.01])

    while time.time() < deadline:
        # indices of the currently largest and smallest entries
        src = int(np.argmax(a))
        dst = int(np.argmin(a))
        if a[src] <= a[dst] + 1e-12:
            # nothing to move – pick random pair as fallback
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        improved = False
        for frac in fractions:
            delta = a[src] * frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            new_score = fast_score(cand, next_pow2(2 * n - 1))
            if new_score < cur_score:
                a = cand
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                improved = True
                break           # accept first improvement for this iteration
        if not improved:
            # random perturbation to escape possible plateaus
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                src2, dst2 = (i, j) if a[i] > a[j] else (j, i)
                delta = a[src2] * rng.random() * 0.01
                cand = a.copy()
                cand[src2] -= delta
                cand[dst2] += delta
                new_score = fast_score(cand, next_pow2(2 * n - 1))
                if new_score < cur_score:
                    a = cand
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 1) previous construction (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) deterministic random families
    for _ in range(10):
        pool.append(random_initial(n, rng))

    # 3) classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass  # windows optional

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.20 * budget_s   # L‑BFGS (aggressive)
    phase2_end = start_time + 0.55 * budget_s   # exponentiated‑gradient
    phase3_end = start_time + 0.90 * budget_s   # simulated annealing
    # remaining → greedy refinement

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed surrogate
    # --------------------------------------------------------------
    lam_schedule = [
        0.8, 0.6, 0.45, 0.33, 0.24, 0.18,
        0.13, 0.09, 0.06, 0.04, 0.025,
        0.015, 0.009, 0.005, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.35:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.012),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:                     # fallback – uniform vector
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=22.0,
            max_iters=10_000_000
        )
        print(f"[exp‑grad] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2b – single LP tightening step
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        M = np.max(np.convolve(best_a, best_a))
        lp_candidate = lp_tighten_step(best_a,
                                       M,
                                       t=0.12,
                                       n=n,
                                       deadline=phase2_end,
                                       max_constraints=1200,
                                       rel_eps=0.08)
        if lp_candidate is not None:
            lp_score = fast_score(lp_candidate, fft_len)
            if lp_score < best_score:
                best_score = lp_score
                best_a = lp_candidate.copy()
                print(f"[LP‑tighten] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            phase3_end,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy deterministic refinement (zero‑temp)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = greedy_local_refine(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[greedy] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
