def smoothed_gradient_descent_multiple(start_a, end_time, n, fft_len, rng):
    best_a = start_a.copy()
    best_score = fast_score(best_a, fft_len)
    # initial runs
    while time.time() < end_time:
        # start from a perturbation of best_a or random
        if rng.random() < 0.3:
            a0 = start_a.copy()  # warm start
        else:
            # random Dirichlet: sample gamma entries then normalize
            a0 = rng.gamma(shape=1.0, size=n)
            a0 = a0 / a0.sum()
        # run gradient descent for limited time slice
        a_opt, score_opt = smoothed_gradient_descent(a0, end_time, n, fft_len, rng)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()

        # break out if time nearly up
        if time.time() + 0.5 >= end_time:
            break
    return best_a, best_score

def smoothed_gradient_descent(a_start, deadline, n, fft_len, rng, eta0=5.0, alpha0=10.0, alpha_max=50.0):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 1
    while time.time() < deadline:
        # compute conv and gradient
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_conv = conv.max()
        # weighted softmax
        # shift to max to avoid overflow
        shifted = conv - max_conv
        w = np.exp(alpha0 * shifted)   # alpha0 acts as temperature
        # Normalize to sum 1 to get probability weights
        w_sum = w.sum()
        if w_sum == 0:
            # if all zero (should not happen), fallback to mask
            mask = (conv >= max_conv - 1e-12).astype(np.float64)
            w = mask
            w_sum = w.sum()
        w_norm = w / w_sum
        fw = np.fft.rfft(w_norm, fft_len)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # update parameters
        eta = eta0 / np.sqrt(it)
        # exponentiated gradient multiplicative update
        exponents = -eta * grad
        exponents = np.clip(exponents, -50, 50)
        a = a * np.exp(exponents)
        # projection onto simplex
        a = simplex_projection(a)

        # evaluate fast_score
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # adjust alpha gradually
        if alpha0 < alpha_max:
            alpha0 = min(alpha_max, alpha0 * 1.001)   # slowly increase

        it += 1
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative sequence) with a small evaluation score.

The algorithm works in three phases within the 1000 s budget:

1. **Broad parametric sweep** – cheap exploration of several analytic families
   (exponential, linear, mixtures, power‑law and a two‑exponential mixture).
   The best candidate becomes the seed for the later stages.

2. **Smoothed gradient descent** – we minimise a smooth approximation of
   `max(conv(a,a))` using a log‑sum‑exp (soft‑max) surrogate.
   The gradient is obtained analytically via FFTs and the update is a
   projected exponentiated‑gradient step on the probability simplex.
   Several random restarts are performed; the best sequence is kept.

3. **Simulated‑annealing fine‑tuning** – random mass transfers between two
   indices are accepted according to a Metropolis rule.  Incremental
   convolution updates keep the routine fast.

All phases respect the global time budget.  The final sequence is evaluated
with the provided `evaluate_sequence` function and stored in the global
variable `height_sequence_1` for possible reuse by a subsequent run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the official score assuming the sum of `a` is 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families (extended)
# ----------------------------------------------------------------------


def param_scan_extended(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    best_a: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Explore a few low‑dimensional families (including a two‑exponential mix)."""
    # uniform baseline (already in best_a)
    # --- pure exponential -------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- linear decreasing ------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=250):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=70):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=13):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=180):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- double exponential mixture ---------------------------------------
    # a = alpha * exp(-lam1*i) + (1-alpha) * exp(-lam2*i)
    lam_grid = np.logspace(np.log10(1e-5), np.log10(0.02), num=30)
    for lam1 in lam_grid:
        base1 = np.exp(-lam1 * np.arange(n))
        base1 /= base1.sum()
        for lam2 in lam_grid:
            base2 = np.exp(-lam2 * np.arange(n))
            base2 /= base2.sum()
            for alpha in np.linspace(0.0, 1.0, num=9):
                a = alpha * base1 + (1.0 - alpha) * base2
                a_sum = a.sum()
                if a_sum == 0.0:
                    continue
                a /= a_sum
                score = fast_score(a, fft_len)
                if score < best_score:
                    best_score, best_a = score, a.copy()
                if time.time() >= deadline:
                    return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smoothed (log‑sum‑exp) gradient descent on the true objective
# ----------------------------------------------------------------------


def smoothed_gradient_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
    alpha0: float = 12.0,
    alpha_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated) gradient descent on a smooth surrogate
    of `max(conv(a,a))`.  The surrogate is `log‑sum‑exp(alpha*conv)`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 1
    alpha = alpha0

    while time.time() < deadline:
        # forward FFT of a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # soft‑max weights (avoid overflow by shifting)
        max_c = conv.max()
        shifted = conv - max_c          # ≤ 0
        w = np.exp(alpha * shifted)     # max element becomes 1
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safeguard
            w = (conv >= max_c - 1e-12).astype(np.float64)
            w_sum = w.sum()
        w_norm = w / w_sum

        # gradient = 2 * (a_rev * w_norm) at the proper slice
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w_norm, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it)
        exponents = -eta * grad
        exponents = np.clip(exponents, -50, 50)        # avoid overflow
        a = a * np.exp(exponents)

        # projection onto the probability simplex
        a = simplex_projection(a)

        # evaluate cheap score
        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        # slowly increase the smoothing temperature (alpha)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.001)

        it += 1

    return best_a, best_score


def smoothed_gradient_descent_multiple(
    a_start: np.ndarray,
    end_time: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run several independent gradient descents (different random starts)
    within the allocated time window.
    """
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    # guarantee at least one run from the supplied start
    while time.time() < end_time:
        # decide whether to restart from a random Dirichlet or from the current best
        if rng.random() < 0.35:
            a0 = best_a.copy()
        else:
            # random Dirichlet (Gamma(1,1))
            a0 = rng.gamma(shape=1.0, size=n)
            a0 /= a0.sum()

        # allocate a small time slice for this run (≈ 5 % of remaining time)
        sub_deadline = time.time() + min(30.0, (end_time - time.time()) * 0.3)

        a_opt, score_opt = smoothed_gradient_descent(
            a0, sub_deadline, n, fft_len, rng, eta0=6.0, alpha0=12.0, alpha_max=80.0
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[smooth‑GD] new best fast_score={best_score:.6f}")

        # stop if very little time left
        if time.time() + 0.2 >= end_time:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated annealing (same routine as before, tuned a little)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass transfers with a Metropolis rule and linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.25   # how much of the source mass can be moved
    recompute_every = 5000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_score = new_score
            accepted += 1
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.01   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a balance between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the previous run if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0.0:
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Split the time budget (≈5 % / 73 % / 22 %)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.78 * budget_s
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (extended)
    # ------------------------------------------------------------------
    if time.time() < phase0_end:
        best_a, best_score = param_scan_extended(
            n, fft_len, phase0_end, rng, best_a, best_score
        )
        print(f"[param‑scan] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed gradient descent with several restarts
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = smoothed_gradient_descent_multiple(
            best_a, phase1_end, n, fft_len, rng
        )
        print(f"[smooth‑GD] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
