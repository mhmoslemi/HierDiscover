"""
Improved step‑function optimisation.

The routine searches for a non‑negative sequence a of length *n* (default 3000)
with ∑a=1 that minimises

    score = 2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence`` after the
security checks.  The algorithm consists of four phases:

1. **Parametric sweep** – cheap enumeration of several analytic families
   (exponential, linear, power‑law, cosine‑bell, mixtures with the uniform
   distribution and a few random Dirichlet draws).  The best candidate becomes
   the seed for the next phase.

2. **Exponentiated‑gradient (Adam‑style) descent** – a mirror‑descent on the
   true max‑convolution objective.  The sub‑gradient is obtained from the
   current maximal convolution entries and an Adam‑style learning‑rate
   adapts the multiplicative updates.

3. **Simulated‑annealing fine‑tuning** – random mass transfers between
   indices using an incremental O(n) update of the convolution.  The
   temperature cools linearly and the transferred fraction shrinks with time.

4. **Local refinement** – deterministic smoothing (moving‑average kernels),
   mixing with the uniform distribution and short random perturbations.
   All refinements are evaluated with the fast FFT based score and accepted
   only when they improve the best value.

The whole search respects a user supplied time budget (default 1000 s)
and returns the best sequence found before the deadline.  The best sequence
is stored in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) under the assumption ∑a = 1.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """Sub‑gradient of 2·n·max(conv(a,a)) w.r.t. *a* (and the current max)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val

def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force cheap families and a few random Dirichlet samples."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    def try_candidate(a_raw):
        nonlocal best_a, best_score
        a = np.maximum(a_raw, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a /= s
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    # exponential decay
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=200):
        if time.time() >= deadline:
            return best_a, best_score
        a = np.exp(-lam * np.arange(n))
        try_candidate(a)

    # linear decreasing to zero
    for s in np.linspace(0.0, 1.0 / n, num=200):
        if time.time() >= deadline:
            return best_a, best_score
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        try_candidate(a)

    # power‑law
    for p in np.linspace(0.2, 5.0, num=150):
        if time.time() >= deadline:
            return best_a, best_score
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        try_candidate(a)

    # cosine‑bell (zero at ends, peak in centre)
    for power in [1, 2, 3, 4]:
        if time.time() >= deadline:
            return best_a, best_score
        a = np.cos(np.linspace(0, np.pi, n)) ** power
        try_candidate(a)

    # mixture exponential + uniform
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=30):
        if time.time() >= deadline:
            return best_a, best_score
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=7):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            try_candidate(a)

    # random Dirichlet draws
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = rng.random(n)
        try_candidate(a)

    return best_a, best_score

# Phase 1 – exponentiated‑gradient descent (Adam style)

def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
):
    """Mirrored Adam‑style SGD on the true max‑convolution objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 1
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decreasing global learning‑rate
        eta = eta0 / np.sqrt(it)
        step = eta / (np.sqrt(v_hat) + eps)

        # multiplicative (exponentiated) update
        exponents = -step * g
        exponents = np.clip(exponents, -50, 50)   # avoid overflow
        a = a * np.exp(exponents)

        # projection onto simplex (preserves ∑a = 1)
        a = simplex_projection(a)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        it += 1

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers with a linear cooling schedule."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # initial temperature (a few percent of current score)
    temp0 = max(best_score * 0.05, 1e-6)
    start_time = time.time()
    recompute_every = 2000
    steps = 0
    accepted = 0

    while time.time() < deadline:
        # linear cooling factor
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        if a[i] >= a[j]:

        diff = np.zeros_like(conv)

        accept = False
        if dE < 0.0:

        else:

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        steps += 1
        if steps % recompute_every == 0:
            conv = np.convolve(a, a)
            curr_score = 2.0 * n * conv.max()

    return best_a, best_score

def local_refinement(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """Smoothing, mixing with uniform and tiny random perturbations."""
    best_a = a_start.copy()
    best_val = start_score

    # 1) moving‑average smoothing with several kernels
    widths = [3, 5, 7, 11, 15, 21, 31, 51, 71]
    for w in widths:
        if time.time() >= deadline:
            break
        kernel = np.ones(w) / w
        a_s = np.convolve(best_a, kernel, mode='same')
        a_s = np.maximum(a_s, 0.0)
        s = a_s.sum()
        if s == 0.0:
            continue
        a_s /= s
        val = fast_score(a_s, fft_len)
        if val < best_val:
            best_val, best_a = val, a_s

    # 2) convex mixture with the uniform distribution
    uniform = np.full(n, 1.0 / n)
    alphas = np.linspace(0.0, 0.30, 31)   # up to 30 % uniform
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a_mix = (1.0 - alpha) * best_a + alpha * uniform
        a_mix = np.maximum(a_mix, 0.0)
        a_mix /= a_mix.sum()
        val = fast_score(a_mix, fft_len)
        if val < best_val:
            best_val, best_a = val, a_mix

    # 3) short random perturbations (Gaussian noise, re‑projected)
    max_iter = 5000
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        eps = 0.003  # relative noise magnitude
        noise = eps * rng.standard_normal(size=n) * best_a
        a_new = best_a + noise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s
        val = fast_score(a_new, fft_len)
        if val < best_val:
            best_val, best_a = val, a_new

    return best_a, best_val

    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
