def adam_optimize(a_start, deadline, n, fft_len, rng, lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=1
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    while time.time()<deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        max_val = conv.max()
        # cut to produce subgradient approximating max: mask near max
        eps_sub = max_val * 1e-12 + 1e-12
        mask = np.where(conv >= max_val - eps_sub, 1.0, 0.0)  # same as before but maybe use soft weights
        # Could use soft weights:
        # temp = max_val * 0.01  # some smoothing
        # w = np.exp((conv - max_val)/temp)
        # w = w / w.sum()
        # compute gradient via w.

        # Let's do soft weights for smoother gradient.
        temp = max_val * 0.01 + 1e-12
        w = np.exp((conv - max_val)/temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = mask
            w_sum = w.sum()
        w = w / w_sum

        # gradient = 2 * conv of a_rev and w
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[:2*n-1]
        grad = 2.0 * grad_full[n-1:n-1+n]

        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non-negativity (by projection onto simplex, but we can first set negatives to zero)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        t += 1

    return best_a, best_score
```

"""
Improved search for a step‑function with a low evaluation score.

Algorithm Overview
------------------
1. **Parametric sweep** – Explore several low‑dimensional families
   (exponential, symmetric exponential, linear, power‑law, sinusoidal,
   and a simple “U‑shape”).  The best candidate becomes the seed for the
   next phases.

2. **Adam‑based smoothed gradient descent** –  We minimise a smooth
   surrogate of the true objective

        F(a) = 2·n·(1/β)·log Σ_k exp(β·(a*a)_k)

   whose gradient can be computed efficiently by FFTs.  Adam’s
   adaptive step‑size and momentum give a much stronger descent than
   the plain exponentiated‑gradient used before.

3. **Simulated‑annealing polishing** –  Random mass transfers with a
   Metropolis rule are performed for a short fine‑tuning stage.

All phases are time‑controlled; the total budget is 1000 s.  The best
sequence found is returned and stored in the global variable
`height_sequence_1` for possible warm‑starts on later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate of the official score.
    Assumes Σ a_i = 1.
    Returns 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a (real) using FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Phase 0 – parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Cheap scan over several analytic families; return the best candidate."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # 1) Pure exponential (left‑skewed)
    # ------------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ------------------------------------------------------------------
    # 2) Symmetric double‑exponential (mass at both ends)
    # ------------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=180):
        left = np.exp(-lam * np.arange(n))
        right = np.exp(-lam * np.arange(n - 1, -1, -1))
        a = left + right
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ------------------------------------------------------------------
    # 3) Linear decreasing (already present in the original code)
    # ------------------------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ------------------------------------------------------------------
    # 4) Mixture exponential + uniform
    # ------------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=50):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # ------------------------------------------------------------------
    # 5) Power‑law decay
    # ------------------------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=150):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ------------------------------------------------------------------
    # 6) Sinusoidal (smooth bump in the centre)
    # ------------------------------------------------------------------
    for amp in np.linspace(0.5, 2.0, num=30):
        a = 1.0 + amp * np.cos(2.0 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ------------------------------------------------------------------
    # 7) Simple “U‑shape” (zero at centre, larger at ends)
    # ------------------------------------------------------------------
    idx = np.arange(n)
    a = np.sqrt(idx * (n - 1 - idx))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    score = fast_score(a, fft_len)
    if score < best_score:
        best_score, best_a = score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on a smoothed max‑convolution surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lr: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    beta_temp: float = 30.0,
):
    """
    Adam optimisation of a smoothed surrogate of the max‑convolution.
    The surrogate is  (1/β)·log Σ exp(β·conv) ; its gradient is
    obtained by a soft‑max weighting of the true sub‑gradient.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 1
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        # ----------- convolution ----------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # ----------- smooth weighting ----------
        # sigma is a small fraction of the current maximum; the
        # soft‑max becomes increasingly sharp as the optimisation proceeds.
        sigma = max_val * 0.03 + eps
        w = np.exp(-(max_val - conv) / sigma)   # larger weight near the max
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to hard mask (extremely rare)
            w = (conv == max_val).astype(float)
            w_sum = w.sum()
        w /= w_sum

        # ----------- gradient (2·a * w) ----------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # central slice

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce feasibility: non‑negative & lies on the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # ---------- keep best ----------
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        t += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Incremental mass transfers with a linear cooling schedule.
    Uses O(n) updates of the convolution for each proposal.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # temperature proportional to the current score
    temp0 = max(best_score * 0.02, 1e-7)
    start_time = time.time()
    max_fraction = 0.25          # at most this fraction of source mass moves
    recompute_every = 5000

    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score
