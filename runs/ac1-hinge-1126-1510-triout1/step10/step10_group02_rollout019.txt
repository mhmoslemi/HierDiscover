    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] fast_score={best_score}")

    # RL-like iteration
    # allocate some fraction of time, say 0.5 budget
    rl_deadline = start + 0.5 * budget_s
    eps = 1e-12
    it = 0
    while time.time() < rl_deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        # ratio
        ratio = target / (conv + eps)
        # clip ratio to avoid huge values
        ratio = np.clip(ratio, 0.0, 1e6)

        # compute update factor: convolution of ratio with reversed a
        rev_a = a[::-1]
        fr = np.fft.fft(ratio, fft_len)
        far = np.fft.fft(rev_a, fft_len)
        upd = np.real(np.fft.ifft(fr * far))[: n]  # length n
        # update a multiplicatively
        a = a * upd
        # projection onto simplex
        a = simplex_projection(a)
        # evaluate score occasionally

            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[RL] iter {it} new best fast_score={best_score:.9f}")
        it += 1

    # after RL, fine-tune with exponentiated gradient (same as earlier but maybe more iterations)
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(best_a, deadline, n, fft_len, rng, eta0=10.0, max_iters=2_000_000)

"""
propose_candidate – “flatten‑the‑convolution” optimiser

The objective to minimise is

    2·n·max_k (a*a)[k] / (∑a)^2

which is invariant under scaling, so we can restrict the vector a to the
probability simplex (∑a = 1, a ≥ 0).  For a uniform a we obtain the score 2.
The theoretical lower bound is 1 (the average value of the convolution is
1/(2n‑1)).  The optimiser works in three stages:

1. **RL flattening** – an EM / Richardson‑Lucy style iteration that tries to
   make the full convolution as flat as possible (≈ constant).  This drives
   the maximum down towards the average value.

2. **Exponentiated‑gradient fine tuning** – a mirrored‑SGD on the exact
   L∞‑objective (the same sub‑gradient used in the reference code) to polish
   the result.

3. **Final evaluation** – the best vector found is fed to the supplied
   ``evaluate_sequence`` function and stored globally as ``height_sequence_1``.

The whole procedure respects a 1000 s budget and always returns a valid
non‑negative step‑function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate of the official score when the vector sums to 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0,
                          max_iters: int = 5_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) directly on the max‑convolution.
    Keeps the iterate on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# “Flatten” the convolution (EM / Richardson‑Lucy style)
# ----------------------------------------------------------------------


def flatten_convolution(a: np.ndarray,
                        target: np.ndarray,
                        fft_len: int,
                        max_iters: int,
                        deadline: float,
                        rng: np.random.Generator) -> np.ndarray:
    """
    Iteratively modifies ``a`` so that its self‑convolution approaches ``target``.
    The update is multiplicative (a ← a·(rev(a) * (target/conv))) and the
    result is projected onto the simplex after every step.
    """
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = fast_score(a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # ratio of desired to current convolution (avoid division by zero)
        ratio = target / (conv + eps)
        # limit extremely large ratios that would destabilise the iteration
        ratio = np.clip(ratio, 0.0, 1e6)

        # multiplicative correction factor = rev(a) * ratio
        rev_a = a[::-1]
        fr = np.fft.fft(ratio, fft_len)
        far = np.fft.fft(rev_a, fft_len)
        upd = np.real(np.fft.ifft(fr * far))[: n]

        # multiplicative update + simplex projection
        a = a * np.maximum(upd, eps)
        a = simplex_projection(a)

        # occasional bookkeeping
        if (it & 15) == 0:          # every 16th iteration
            sc = fast_score(a, fft_len)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[flatten] iter {it:05d} best fast_score={best_sc:.9f}")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Target convolution (constant – the theoretical optimum)
    # --------------------------------------------------------------
    target_val = 1.0 / (2 * n - 1)               # average value of the convolution
    target = np.full(2 * n - 1, target_val, dtype=np.float64)

    # --------------------------------------------------------------
    # Initialise with a uniform probability vector
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 1) EM / Richardson‑Lucy style flattening of the convolution
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin
    # allocate roughly half of the budget to the flattening phase
    flatten_deadline = start_time + 0.48 * budget_s

    a = flatten_convolution(
        a_start=a,
        target=target,
        fft_len=fft_len,
        max_iters=30_000,            # generous upper bound – loop stops earlier by time
        deadline=flatten_deadline,
        rng=rng,
    )

    # --------------------------------------------------------------
    # 2) Fine‑tune with exponentiated‑gradient descent on the true L∞
    # --------------------------------------------------------------
    if time.time() < deadline:
        a, final_fast = exp_gradient_optimize(
            a_start=a,
            deadline=deadline,
            n=n,
            fft_len=fft_len,
            rng=rng,
            eta0=18.0,               # a slightly larger start step
            max_iters=4_000_000,
        )
        print(f"[fine‑grad] final fast_score={final_fast:.9f}")

    # --------------------------------------------------------------
    # 3) Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # expose the result for possible reuse in subsequent runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
