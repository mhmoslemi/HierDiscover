def fast_score(a, fft_len):
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return float(2*n * conv.max())
```

"""
Improved step‑function optimiser

* Uses a larger problem size (n≈8000 when the time budget permits) – a larger
  search space gives a lower theoretical bound on the max‑auto‑convolution.
* Three optimisation phases:
    1. Broad random search + a quick L‑BFGS cascade on a smooth
       log‑sum‑exp surrogate of the max‑convolution.
    2. Mirror‑descent (exponentiated‑gradient) on the true L∞‑objective
       with an adaptive step‑size schedule.
    3. Simple but powerful simulated‑annealing that moves mass between
       any two entries (including long‑range transfers) and accepts
       deteriorations with a linearly cooling temperature.
* All vectors are kept on the probability simplex, i.e. they sum to one.
  The official scorer therefore evaluates `2·n·max(conv(a,a))`.
* The code is fully self‑contained, uses only NumPy/SciPy and respects the
  1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    2·n·max(a*a) assuming a sums to 1.
    Convolution via real FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Diverse random initialisers (all return a probability vector)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse', 'gauss', 'cos']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'gauss':
        sigma = rng.uniform(0.1, 0.4) * n
        x = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
    else:  # cosine bump
        x = np.linspace(-np.pi, np.pi, n)
        a = np.maximum(np.cos(x), 0.0)  # non‑negative half‑cosine
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smoothed max‑convolution and its gradient w.r.t. y,
    where a = softmax(y) lives on the simplex.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (real FFT)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                              # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
    # extract the part that multiplies a (size n)
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑space variables
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        # back to simplex
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # early‑exit if improvement stalls
        if time.time() >= deadline:
            break
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_iters: int = 2_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact L∞‑objective.
    Returns the best vector and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and mask of maximal entries
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step‑size schedule (decreasing)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip for stability)
        exp_term = np.exp(np.clip(-eta * g, -60, 60))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """
    Simple Metropolis hill‑climb that transfers a random fraction of mass
    from a source index to a destination index. Temperature cools linearly.
    """
    a = a_start.copy()
    cur_sc = start_score
    best_a = a.copy()
    best_sc = cur_sc

    start_time = time.time()
    # initial temperature: a few percent of the current score
    temp0 = max(best_sc * 0.05, 1e-9)

    max_frac = 0.30          # up to 30 % of the source mass may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        # pick source and destination
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # always move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # evaluate proposal
        cand_sc = fast_score(cand, next_pow2(2 * n - 1))
        dE = cand_sc - cur_sc
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_a = a.copy()

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a better theoretical bound)
    # --------------------------------------------------------------
    # Allocate the larger size only if we have enough total budget.
    n = 8000 if budget_s >= 400 else 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 1) Upsample previous best if it exists
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 1:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_up = np.interp(x_new, x_old, old, left=0.0, right=0.0)
            a_up = np.maximum(a_up, 0.0)
            a_up /= a_up.sum()
            pool.append(a_up)

    # 2) Random families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # 3) Classic window functions (if SciPy is present)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Quick screening – keep the few best candidates
    # --------------------------------------------------------------
    scored = []
    for a in pool:
        sc = fast_score(a, fft_len)
        scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    # retain the top three for deeper optimisation
    top_candidates = [a for _, a in scored[:3]]
    best_a = top_candidates[0].copy()
    best_sc = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Timing splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin
    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS heavy phase
    phase2_end = start_time + 0.75 * budget_s   # EG phase

    # --------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS cascade on the best few starts
    # --------------------------------------------------------------
    lam_schedule = np.logspace(np.log10(0.5), -5, num=18)  # decreasing λ

    for a0 in top_candidates:
        if time.time() >= phase1_end:
            break
        a_opt, sc_opt = lbfgs_cascade(
            a0,
            lam_schedule,
            maxiter_per_lam=1200,
            deadline=phase1_end,
            fft_len=fft_len,
            n=n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on true max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=20.0,
            max_iters=3_000_000,
        )
        print(f"[exp‑grad] fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_sc:.9f}")
