"""
step‑function optimiser – GA + exponentiated‑gradient + anneal

* Phase 1 – a simple genetic algorithm (population of probability vectors)
  explores a very diverse set of shapes (uniform, exponential, power‑law,
  linear, sinusoid, classic windows, sparse).  Elites are kept, children are
  created by convex combinations and Gaussian noise, then projected back onto
  the simplex.  The fast objective 2·n·max(conv) is used for cheap evaluation.

* Phase 2 – exponentiated‑gradient (mirrored SGD) on the exact max‑convolution
  refines the best individual found by the GA.  The update stays on the simplex
  and uses a diminishing step‑size schedule.

* Phase 3 – a short simulated‑annealing mass‑transfer hill‑climb finishes the
  search, accepting worsening moves with a linearly cooled temperature.

The whole routine respects a 1000 s budget, always returns a non‑negative
sequence and stores it in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    2·n·max(a*a) assuming a sums to 1.
    The convolution is evaluated by FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "linear",
            "sinusoid",
            "edge",
            "bimodal",
            "sparse",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "linear":
        # decreasing linear from left to right, shifted to be non‑negative
        a = np.linspace(1.0, 0.0, n)
        a = np.maximum(a, 0.0)
    elif mode == "sinusoid":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # positive half‑sine
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_projection(noisy)


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirrored SGD (exponentiated‑gradient) directly on the  L∞‑objective.
    Returns the best vector seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but effective)
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of starter vectors (including previous best)
    # ------------------------------------------------------------------
    pool = []
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
            pool.append(a0)

    # add a bunch of random families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    ga_end = start_time + 0.55 * budget_s   # genetic algorithm phase
    eg_end = start_time + 0.90 * budget_s   # exponentiated‑gradient phase
    # remaining → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – Genetic Algorithm on the fast_score
    # ------------------------------------------------------------------
    pop_size = 150
    elite_size = 12

    # initialise population from the pool (with some extra random draws)
    population = [rng.choice(pool) for _ in range(pop_size)]
    pop_scores = [fast_score(ind, fft_len) for ind in population]

    best_idx = int(np.argmin(pop_scores))
    best_a = population[best_idx].copy()
    best_score = pop_scores[best_idx]

    generation = 0
    while time.time() < ga_end:
        generation += 1
        # select elites
        elite_order = np.argsort(pop_scores)
        elites = [population[i] for i in elite_order[:elite_size]]

        # produce children
        children = []
        while len(children) < pop_size - elite_size:
            p1, p2 = rng.choice(elites, size=2, replace=False)
            w = rng.random()
            child = w * p1 + (1.0 - w) * p2
            # occasional mutation
            if rng.random() < 0.7:
                sigma = rng.uniform(0.0005, 0.008)
                child = add_noise(child, sigma=sigma, rng=rng)
            child = simplex_projection(child)
            children.append(child)

        # new generation
        population = elites + children
        pop_scores = [fast_score(ind, fft_len) for ind in population]

        # update global best
        cur_best = int(np.argmin(pop_scores))
        if pop_scores[cur_best] < best_score:
            best_score = pop_scores[cur_best]
            best_a = population[cur_best].copy()
            print(f"[GA gen {generation}] new best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < eg_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, eg_end, n, fft_len, rng, eta0=20.0, max_iters=5_000_000
        )
        print(f"[exp‑grad] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer and store result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
