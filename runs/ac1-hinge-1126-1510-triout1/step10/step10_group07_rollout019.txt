"""
Optimise a non‑negative step sequence for the bound

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution.
The algorithm is a multi‑phase stochastic optimiser:

1.  Initialise a diverse pool (uniform, random, exponential,
    power‑law, spectral, previous best).
2.  Run many Adam optimisations on a smooth “log‑sum‑exp” surrogate.
    Each run uses a long decreasing λ‑schedule and a moderate number
    of steps, keeping the best vector.
3.  Refine the champion with a long exponentiated‑gradient descent
    on the true (hard) max‑convolution objective.
4.  Deterministic hill‑climbing that moves a fraction of mass from the
    most damaging coordinate toward the least damaging one, with a
    geometrically shrinking step size.
5.  A cheap simulated‑annealing style perturbation loop that repeatedly
    adds Gaussian noise and accepts improvements (or probabilistic
    uphill moves) while the time budget permits.
6.  Convex‑combination (mixture) line‑search among the top‑K vectors
    discovered so far.
7.  Final evaluation with the provided `evaluate_sequence` routine.
   The best vector is stored globally as `height_sequence_1` for reuse.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0) by
using a soft‑max parametrisation (Adam) or an explicit normalisation
(EG, hill‑climb, perturbations).  The code contains no closures or
lambda expressions and complies with the 2‑CPU limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random initial probability vectors."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Gaussian noise and re‑projection onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    steps_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    rng: np.random.Generator,
    scale: float = 5.0,
):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Deterministic hill‑climb with geometric step reduction
# ----------------------------------------------------------------------


def fine_hill_climb(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_step: float = 0.15,
    min_step: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.7, min_step)

    return a, score

def eval_mix(t: float, a: np.ndarray, b: np.ndarray, n: int, fft_len: int) -> tuple[float, np.ndarray]:
    """Linear mixture of two probability vectors and its fast score."""
    mix = (1.0 - t) * a + t * b
    mix /= mix.sum()
    sc = fast_score(mix, fft_len, n)
    return sc, mix


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates, perform a cheap golden‑section line search
    over the mixing weight t ∈ [0,1] to minimise the bound.
    Returns the best vector and its score found during this phase.
    """
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b, n, fft_len)
            fd, _ = eval_mix(d, a, b, n, fft_len)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b, n, fft_len)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b, n, fft_len)
            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b, n, fft_len)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()

    return best_a, best_score

# Simulated‑annealing style random perturbation phase
# ----------------------------------------------------------------------


def anneal_phase(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Random Gaussian perturbations with temperature decay.
    Returns the best vector found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = score_start

    T0 = 0.5
    T1 = 1e-4
    total_it = 0
    while time.time() < deadline:
        # linearly decay temperature
        elapsed = deadline - time.time()
        frac = max(0.0, (deadline - time.time()) / (deadline - (deadline - 0.9 * (deadline - time.time()))))
        T = T0 * (frac) + T1 * (1 - frac)

        sigma = T * 0.05
        noisy = add_noise(a, sigma, rng)
        sc = fast_score(noisy, fft_len, n)

        if sc < best_score or rng.random() < np.exp((best_score - sc) / (T + 1e-12)):
            a = noisy
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        total_it += 1
        if total_it % 2000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    height_sequence_1 = final_seq

    return final_seq
