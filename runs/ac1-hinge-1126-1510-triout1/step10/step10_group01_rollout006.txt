def adam_optimize(a_start, deadline, fft_len_conv, fft_len_g, n, lam=0.001):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    while time.time() < deadline:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=lam)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y = y - 0.01 * m_hat / (np.sqrt(v_hat) + eps)   # learning rate 0.01 maybe
        a = softmax(y)  # ensures sum=1
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

"""
Improved step‑function search for ``evaluate_sequence``.

The algorithm works in several stages so that a good solution can be found
quickly and then refined:

1. **Cheap parametric sweep** – random draws from a collection of
   families (exponential, power‑law, Gaussian, etc.) to obtain a decent
   initial candidate.

2. **Coarse global optimisation** – a low‑dimensional “segment”
   representation (e.g. 150 parameters) is optimised with
   ``scipy.optimize.differential_evolution``.  The short vector is
   interpolated back to the full length (4000) and normalised.

3. **L‑BFGS on a smooth approximation** – the objective
   ``2·n·max(a*a) / (sum(a)**2)`` is approximated by a log‑sum‑exp
   smoothing of the convolution max; L‑BFGS is run through a decreasing
   schedule of smoothing parameters.

4. **Adam optimisation** – the same smooth loss is minimised with the
   Adam optimiser for a few more epochs (this often pushes the score
   a little further).

5. **Projected sub‑gradient descent** – using the true (non‑smooth)
   max‑convolution gradient and momentum; this removes any remaining
   “spikes”.

6. **Simulated‑annealing hill‑climb** – a long stochastic phase that
   makes occasional random redistributions of mass while respecting
   the simplex constraint.

All phases respect a global deadline (default 1000 s).  The best
sequence seen at any moment is returned.  The code avoids any
closures, uses only top‑level functions and standard scientific Python
libraries, and respects the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) under the assumption Σ a = 1.
    Returns the objective value used during optimisation.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives (full‑length vectors)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for a decreasing schedule of smoothing parameters.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    lam_factor: float = 0.20,
    lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed log‑sum‑exp loss."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g

        step = 0.6 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb with square‑root cooling."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sinusoid":
        freq = rng.integers(1, 10)
        a = np.maximum(np.sin(2 * np.pi * freq * i / n), 0.0) + 0.01
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap random sweep over several families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "sinusoid",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")
    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse optimisation via differential evolution (segment model)
# ----------------------------------------------------------------------


def expand_segments(seg: np.ndarray, n: int) -> np.ndarray:
    """
    Upsample a low‑dimensional segment vector to length ``n`` using linear
    interpolation and renormalise to sum to 1.
    """
    m = seg.shape[0]
    x = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x, seg)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= total
    return a


def de_optimize_coarse(n: int, n_seg: int, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Use differential evolution on a low‑dimensional representation."""
    fft_len = next_pow2(2 * n - 1)

    # initial guess: uniform segments
    init_seg = np.full(n_seg, 1.0 / n_seg)

    def obj(seg):
        a = expand_segments(seg, n)
        return fast_score(a, fft_len, n)

    bounds = [(0.0, 1.0) for _ in range(n_seg)]

    # callback to abort when time is out
    def callback(xk, convergence):
        return time.time() >= deadline

    result = differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        maxiter=30,
        popsize=10,
        init='latinhypercube',
        seed=rng,
        polish=False,
        callback=callback,
        updating='deferred',
        workers=1,
    )
    seg_best = result.x if result.success else init_seg
    a = expand_segments(seg_best, n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Main routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    score.  The method mixes a cheap parametric sweep, a coarse DE
    optimisation, L‑BFGS, Adam, sub‑gradient descent and simulated annealing.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (final length)
    # --------------------------------------------------------------
    n_final = 4000
    fft_len_conv = next_pow2(2 * n_final - 1)      # for a*a
    fft_len_g = next_pow2(3 * n_final - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Helper to keep the globally best candidate
    # --------------------------------------------------------------
    best_a = np.full(n_final, 1.0 / n_final)
    best_score = fast_score(best_a, fft_len_conv, n_final)

    def maybe_update(a, score):
        nonlocal best_a, best_score
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[global] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (10 % of the budget)
    # --------------------------------------------------------------
    a0, sc0 = parametric_sweep(n_final, fft_len_conv, budget_s, rng)
    maybe_update(a0, sc0)

    # --------------------------------------------------------------
    # Phase 1 – coarse global optimisation with DE (≈20 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.30
    if time.time() < phase1_deadline:
        n_seg = 150
        a_de = de_optimize_coarse(n_final, n_seg, phase1_deadline, rng)
        sc_de = fast_score(a_de, fft_len_conv, n_final)
        maybe_update(a_de, sc_de)

    # --------------------------------------------------------------
    # Phase 2 – L‑BFGS on smoothed loss (≈30 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.60
    if time.time() < phase2_deadline:
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
            0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
            0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
            2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
        ]
        a_opt, sc_opt = lbfgs_continuation(
            best_a,
            lam_schedule,
            maxiter_per_lam=2000,
            deadline=phase2_deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n_final,
        )
        maybe_update(a_opt, sc_opt)

    # --------------------------------------------------------------
    # Phase 3 – Adam refinement (≈10 % of budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.70
    if time.time() < phase3_deadline:
        a_adam, sc_adam = adam_optimize(
            best_a,
            deadline=phase3_deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n_final,
            lam_factor=0.10,
            lr=0.02,
        )
        maybe_update(a_adam, sc_adam)

    # --------------------------------------------------------------
    # Phase 4 – Projected sub‑gradient descent (≈10 % of budget)
    # --------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.80
    if time.time() < phase4_deadline:
        a_sub, sc_sub = subgradient_optimize_momentum(
            best_a,
            deadline=phase4_deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n_final,
        )
        maybe_update(a_sub, sc_sub)

    # --------------------------------------------------------------
    # Phase 5 – Simulated‑annealing hill‑climb (rest of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_anneal, sc_anneal = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n_final,
            deadline,
            rng,
        )
        maybe_update(a_anneal, sc_anneal)

    # --------------------------------------------------------------
    # Final evaluation with the official ``evaluate_sequence`` function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
