def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Analytic optimal solution: the objective
        2·n·max(conv(a,a)) / (sum(a)**2)
    is lower‑bounded by 2, with equality iff a is a constant vector.
    Therefore we simply return a uniform step function of the required length.
    The function respects the security checks of `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # Determine length: use existing best if available, else 4000
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        n = old.shape[0] if old.ndim == 1 and old.size > 0 else 4000
    else:
        n = 4000

    # Uniform distribution (probability vector)
    a = np.full(n, 1.0 / n, dtype=float)

    # Just in case the security checks do any clipping, enforce the limits explicitly

        print(f"[uniform] n={n}, evaluated score = {score:.12f}")
    except Exception as e:
        print("[uniform] evaluate_sequence failed:", e)

    return a.tolist()

"""
Uniform step function – provably optimal for the given objective.

The evaluation function is

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

For any non‑negative vector a of length n the Cauchy–Schwarz inequality gives

    max(conv(a, a)) = sum_i a_i**2  ≥  (sum_i a_i)**2 / n .

Consequently

    score ≥ 2·n·((sum a)**2 / n) / (sum a)**2 = 2,

with equality iff all entries are equal.
Therefore the globally optimal step function is the uniform distribution.
The function below simply returns that distribution (clipped to the allowed
range) and prints the resulting score for diagnostics.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a uniform (constant) non‑negative step sequence of length n.
    The length n is taken from a previously stored solution (if any) or
    defaults to 4000.  The returned sequence achieves the theoretical
    lower bound of the objective (score = 2.0).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the size of the vector.
    # If a previous best solution exists, reuse its length; otherwise
    # use the default problem size of 4000.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            n = int(old.size)
        else:
            n = 4000
    else:
        n = 4000

    # ------------------------------------------------------------------
    # Construct the uniform vector.
    # All entries are identical and sum to 1 (a probability vector).
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=float)

    # Apply the same clipping that evaluate_sequence performs,
    # purely for safety (has no effect for the uniform vector).
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)

    # ------------------------------------------------------------------
    # Optional – report the official score for verification.
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence is provided by the host environment.
        score = evaluate_sequence(a.tolist())
        print(f"[uniform] n={n}, evaluated score = {score:.12f}")
    except Exception as exc:
        # In the unlikely event of an error, still return the sequence.
        print("[uniform] evaluate_sequence raised:", exc)

    # ------------------------------------------------------------------
    # Return the sequence as a plain Python list of floats.
    # ------------------------------------------------------------------
    return a.tolist()
