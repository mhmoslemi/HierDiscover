"""
Improved step‑function optimiser.

The optimiser works in three phases on a moderately large problem size
(n≈6000).  It first runs a long Adam optimiser on a smoothed (log‑sum‑exp)
approximation of the L∞‑objective.  The optimisation is performed in the
soft‑max (log‑probability) domain, which automatically enforces the
non‑negativity and ∑a=1 constraints.  After the smoothed phase a
mirrored‑SGD (exponentiated‑gradient) sweep is performed on the exact
max‑convolution, followed by a long simulated‑annealing hill‑climb.
All phases respect the 1000 s budget and the best sequence found is
returned.
"""

import time
import numpy as np
from scipy.signal import chebwin, kaiser

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) assuming a sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector of length n."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) lives on the probability simplex.
    """
    # softmax a
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                              # d smooth_max / d conv_i
    # back‑propagation through convolution: (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lam_schedule,
                  iters_per_lam: int = 3000):
    """
    Adam optimiser on the smoothed objective.
    Returns the best a seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # log‑probabilities
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.05                         # base learning rate
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    it_tot = 0

    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len, lam)

            it_tot += 1
            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it_tot)
            v_hat = v / (1.0 - beta2 ** it_tot)

            y -= alpha * m_hat / (np.sqrt(v_hat) + eps)

            # occasional evaluation
            if it_tot % 100 == 0:
                a = np.exp(y - np.max(y))
                a /= a.sum()
                sc = fast_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[Adam] lam={lam:.2e} it={it_tot} best_score={best_score:.9f}")

    return best_a, best_score

# Exponentiated‑gradient (mirrored SGD) with exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 10_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact L∞‑objective.
    Returns best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of (almost) maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]      # size n

        # step‑size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update (exponentiated‑gradient)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # keep the best
        if it % 50 == 0:
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb (improved move set)

def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass‑transfer moves accepted with a Metropolis rule.
    Temperature decays linearly from a fraction of the current best score.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.25

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        # choose a source index (prefer larger a) and a destination
        src = int(rng.choice(n, p=a / a.sum()))  # bias towards larger entries
        dst = rng.integers(0, n)
        if src == dst:
            continue

        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a good compromise between quality and speed
    # --------------------------------------------------------------
    n = 6000                     # ≈6k entries gives a finer search space
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 1) Use the previous best if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families
    for _ in range(6):
        pool.append(random_initial(n, rng))

    # 3) Classical windows (if SciPy is present)
    try:
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass  # optional – ignore if signal package is missing

    # --------------------------------------------------------------
    # Quick pre‑selection: keep the best few from the pool
    # --------------------------------------------------------------
    best_pool = []
    for vec in pool:
        sc = fast_score(vec, fft_len)
        best_pool.append((sc, vec))
    best_pool.sort(key=lambda x: x[0])
    # keep up to 4 promising starters
    starters = [vec for _, vec in best_pool[:4]]

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin
    phase1_end = start_time + 0.40 * budget_s
    phase2_end = start_time + 0.80 * budget_s

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed objective (large number of steps)
    # --------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 1e-6, num=25)   # descending λ
    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(starters)
        # occasional small noise to escape plateaus
        if rng.random() < 0.25:
            a_start = a_start + 1e-4 * rng.standard_normal(a_start.shape)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()

        a_opt, sc_opt = adam_optimize(
            a_start,
            phase1_end,
            n,
            fft_len,
            rng,
            lam_schedule,
            iters_per_lam=2500
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1 Adam] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent on the exact max‑conv.
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=30.0,
            max_iters=8_000_000
        )
        print(f"[Phase‑2 Exp‑Grad] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng
        )
        print(f"[Phase‑3 Anneal] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # expose for possible reuse
    globals()['height_sequence_1'] = final_seq
    return final_seq
