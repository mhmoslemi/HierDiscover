"""
Propose a step‑function that (approximately) minimises

        2·n·max(a*a) / (sum(a)**2)

under the constraints of evaluate_sequence.

Algorithm
---------
* Choose a moderately large dimension (n≈8000) – larger n gives a
  lower theoretical bound (≈1) while still allowing fast FFTs.
* Begin with a cheap random sweep over a few parametric families.
* Refine the best candidate by *variance minimisation*:
  minimise the L2‑distance between the self‑convolution a*a and the
  flat distribution (its average 1/(2n‑1)).  The objective is smooth,
  its gradient is obtained by a single FFT‑based cross‑correlation,
  and we optimise it with L‑BFGS‑B on the soft‑max (log‑it) parametrisation.
* Use a projected sub‑gradient method (with momentum) on the true
  max‑convolution to squeeze the remaining peak.
* Finish with a simple simulated‑annealing hill‑climber.
* The best vector found is finally evaluated with evaluate_sequence.

All phases respect a global 1000 s deadline and keep the sequence on
the simplex (non‑negative, sum = 1).  The code contains no closures,
no file I/O and works with only numpy/scipy.
"""

import time
import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0,∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2) + np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sinusoid":
        freq = rng.integers(1, 10)
        a = np.maximum(np.sin(2 * np.pi * freq * i / n), 0.0) + 0.01
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

    deadline = time.time() + 0.05 * budget_s  # 5 % of total budget

# Variance (L2) minimisation – smooth objective
# ----------------------------------------------------------------------


def variance_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
) -> tuple[float, np.ndarray]:
    """
    Minimise sum_k ( (a*a)_k - 1/(2n‑1) )².
    Returns (loss, gradient w.r.t. logits y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = np.sum(diff * diff)

    # gradient wrt a : 4 * cross‑correlation(diff, a)
    fd = np.fft.fft(diff, fft_len_corr)
    fa_big = np.fft.fft(a, fft_len_corr)
    corr = np.real(np.fft.ifft(fd * np.conj(fa_big)))  # linear correlation

    # slice to obtain lags 0 … n‑1  (index offset = n‑1)
    g = 4.0 * corr[(n - 1) : (n - 1 + n)]

    # chain‑rule through softmax
    mean_g = np.dot(a, g)
    grad_y = a * (g - mean_g)

    return loss, grad_y

def subgradient_optimize_momentum(

    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # sub‑grad of max

        # gradient of max w.r.t a (same as in the original code)
        a_rev = a[::-1]

        step = 0.6 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (simple hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Very simple stochastic hill‑climb with square‑root cooling."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (larger n gives a lower theoretical bound)
    # --------------------------------------------------------------
    n = 8000 if budget_s >= 500 else 4000      # keep runtime reasonable
    fft_len_conv = next_pow2(2 * n - 1)        # for a*a
    fft_len_corr = next_pow2(3 * n - 2)        # for correlations needed in gradients

    # --------------------------------------------------------------
    # Initial uniform candidate
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional warm‑start from a previously saved construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # allocate phase end‑times
    phase1_end = start_time + 0.55 * budget_s   # variance L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient
    phase3_end = deadline                       # simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – variance minimisation with L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    while time.time() < phase1_end:
        # pick a start point – either a jittered best or a fresh random Dirichlet
        if rng.random() < 0.4:
            a_start = best_a * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = rng.random(n)
            a_start /= a_start.sum()

        y0 = np.log(a_start + 1e-12)
        # allocate a modest number of L‑BFGS iterations (time‑adaptive)
        remaining = max(0.0, phase1_end - time.time())
        maxiter = int(min(3000, max(200, remaining / 0.02)))  # aim ≈0.02 s per iter
        try:
            res = minimize(
                fun=variance_obj_and_grad,
                x0=y0,
                args=(n, fft_len_conv, fft_len_corr),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[l2] L‑BFGS exception:", e)
            break

        a_opt = softmax(res.x)
        sc_opt = fast_score(a_opt, fft_len_conv, n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[l2] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max‑conv.
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_corr, n
        )
        print(f"[subgrad] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (long hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
