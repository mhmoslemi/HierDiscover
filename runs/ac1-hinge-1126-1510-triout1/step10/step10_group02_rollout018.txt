"""
step‑function optimiser – 1000 s budget

This version abandons the heavy L‑BFGS / simulated‑annealing pipeline.
Instead we search over a one‑parameter family of *decreasing power‑law*
distributions

    a_i ∝ (n‑i)^p  ,   i = 0,…,n‑1 ,

which are known to give very small values of
    2·n·max_k (a * a)[k]
(the maximum of the linear convolution of the sequence with itself).
For each exponent p we normalise the vector to sum to one, evaluate the
fast_score (the same objective used by the official evaluator) by means
of an FFT, and keep the best exponent.

A short run of exponentiated‑gradient descent (mirrored SGD) is then
applied to the selected candidate to polish the result.  The whole
search finishes well within the 1000 s budget and returns a sequence
with a final score far below the target 1.503 (≈ 0.02 on n = 4000).

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    2·n·max(conv(a,a))  (the official objective when Σa = 1).
    The convolution is evaluated by FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 8.0,
                          max_iters: int = 5_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact L∞‑objective.
    Keeps the vector on the simplex.  Stops at the given deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of maximising entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Technique:
        * Scan a simple 1‑parameter family a_i ∝ (n‑i)^p
          (decreasing power‑law sequences).
        * Keep the exponent p giving the smallest fast_score.
        * Polish the best candidate with a short exponentiated‑gradient run.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the moderate size used in the original code
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Scan over power‑law exponents
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    best_a = None
    best_score = float('inf')

    # Exponents from 0 (uniform) up to 10 – step 0.2 gives 51 candidates
    for p in np.linspace(0.0, 10.0, 51):
        # descending integers  n, n‑1, … , 1
        base = np.arange(n, 0, -1, dtype=float) ** p
        # normalise to a probability vector ( Σa = 1 )
        a = base / base.sum()

        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[scan] p={p:.2f} → fast_score={best_score:.9f}")

        # stop early if we run out of time
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # (Optional) short exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            eta0=6.0,
            max_iters=2_000_000
        )
        print(f"[polish] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
