"""
A fast, multi‑phase solver for the step‑function bound.

The routine builds a rich pool of candidates (deterministic windows,
parameter sweeps for Kaiser, Gaussian, exponential and power laws,
plus random vectors).  It keeps the best according to the fast
objective 2·n·max(conv) (identical to ``evaluate_sequence`` when the
vector is normalised to sum 1).

After the pool the algorithm spends a fraction of the time sampling
random vectors and random permutations of the current best (this tends
to decorrelate the sequence and lowers the peak of the self‑convolution).
A long L‑BFGS polish with a schedule of decreasing smoothing
parameters finishes the optimisation, followed by a short greedy
hill‑climb.  The final list of floats is returned; its official score
is printed just before returning.

The implementation respects the 1000‑second budget and uses only
NumPy, SciPy and the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (exact when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (Σa=1):
        2·n·max(conv(a,a))
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution (log‑sum‑exp).
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers (wide variety)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    # start from a flat spectrum
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_c = np.fft.ifft(A)
    a = np.real(a_c[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def exponential_init(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def power_init(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    using a mask of the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # top‑fraction mask
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
            a_sum = a.sum()
        a = a / a_sum

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.07):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function runs for at most ``budget_s`` seconds and returns the best
    list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (trade‑off between granularity and speed)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool (deterministic windows + parameter sweeps)
    # --------------------------------------------------------------
    candidates = []

    # Optional previous solution – interpolate to the current length
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))

    # Parameter sweeps for Kaiser, Gaussian, exponential, power
    for beta in np.arange(2.0, 20.01, 0.5):
        candidates.append(kaiser_init(n, beta))

    for sigma in np.linspace(0.10, 0.50, 9):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    for rate in np.logspace(-4, -1, 10):
        candidates.append(exponential_init(n, rate))

    for p in np.linspace(0.2, 3.0, 10):
        candidates.append(power_init(n, p))

    # Some pure random draws (Dirichlet‑like)
    for _ in range(5):
        a = rng.random(n)
        a /= a.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] pool best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive random sampling / permutation phase
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s   # 35 % of total time
    while time.time() < phase1_end:
        # a) fresh random Dirichlet‑type vector
        a = rng.random(n)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

        # b) random permutation of the current best (decorrelates)
        perm = rng.permutation(n)
        a_perm = best_a[perm]
        sc_perm = fast_score(a_perm, fft_len, n)
        if sc_perm < best_score:
            best_score, best_a = sc_perm, a_perm.copy()
        # c) a tiny jittered version of the best (helps escape flat regions)
        jitter = rng.normal(scale=1e-4, size=n)
        a_jit = best_a + jitter
        a_jit = np.clip(a_jit, 0.0, None)
        s = a_jit.sum()
        if s > 0.0:
            a_jit /= s
            sc_jit = fast_score(a_jit, fft_len, n)
            if sc_jit < best_score:
                best_score, best_a = sc_jit, a_jit.copy()

    print(f"[random_sampling] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS polishing (deep schedule)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s   # up to 70 % of total time
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 2000

    while time.time() < phase2_end:
        # Blend current best with a fresh random vector (helps diversity)
        rand_vec = rng.random(n)
        rand_vec /= rand_vec.sum()
        alpha = rng.random() * 0.25          # ≤ 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase2_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   deadline,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.02)
        print(f"[exp‑grad] best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=80000,
                                        delta_frac=0.06)
        print(f"[final hill‑climb] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Return the final sequence (official evaluation for safety)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
