def simplex_projection(v):
    # v is array (not necessarily non-negative)
    # algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def pgd_subgradient(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    it = 0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_val = conv.max()
        # tolerance
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)  # shape (2n-1)
        # compute subgradient
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad = 2.0 * np.fft.irfft(fa_rev * fm, fft_len)[:2*n-1]
        # extract gradient component for a (positions n-1 to n-1+n-1)
        g = grad[n-1:n-1+n]
        # step size schedule
        # initial scale maybe 1.0, reducing
        eta = 1.0 / np.sqrt(it + 1)
        # try line search
        # line search reducing step until improvement
        improved = False
        # attempt up to e.g., 5 reductions
        for ls in range(5):
            # candidate via projection after gradient step
            a_new = a - eta * g
            # project onto simplex
            a_new = simplex_projection(a_new)

"""
Propose a low‑score step‑function sequence.

The algorithm builds on the previous approach but adds:
  • More aggressive Adam runs with a finer λ‑schedule.
  • A deterministic projected sub‑gradient descent on the true max‑convolution.
  • A final random local‑search that moves tiny mass chunks between
    coordinates.
All vectors stay on the probability simplex (∑a = 1, a ≥ 0).  The code
uses only NumPy and the already‑imported `evaluate_sequence`.  It respects
the 1000 s wall‑time budget and returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) under the assumption ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. logits y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """Adam optimiser on the log‑sum‑exp surrogate."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05          # base learning rate

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (same as before)
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.7, min_step)

    return a, score

# Convex‑combination search among top candidates (unchanged)
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """Golden‑section line search over convex combinations of pairs."""
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)

            # inner helper
            def eval_mix(t, a, b):
                mix = (1.0 - t) * a + t * b
                mix /= mix.sum()
                return fast_score(mix, fft_len, n), mix

            fc, _ = eval_mix(c, a, b)
            fd, _ = eval_mix(d, a, b)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b)

            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def pgd_subgradient(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """Projected sub‑gradient descent on max(a*a) (∑a = 1)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)   # shape (2n‑1)

        # sub‑gradient of the (hard) max
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
        g = 2.0 * grad_full[n - 1:n - 1 + n]   # gradient w.r.t. a

        # step‑size schedule (square‑root decay)
        eta = 1.0 / np.sqrt(it + 1)

        # back‑tracking line search (max 5 reductions)
        improved = False
        for _ in range(5):
            a_new = a - eta * g
            a_new = simplex_projection(a_new)
            score_new = fast_score(a_new, fft_len, n)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # occasional jitter to escape plateaus
            if it % 100 == 0:
                pert = rng.normal(0.0, 0.001, size=n)
                a = simplex_projection(a + pert)

        it += 1
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Random local search (tiny mass transfers)
# ----------------------------------------------------------------------
def random_local_search(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """Move tiny fractions of mass between random coordinates."""
    a = a_start.copy()
    score = start_score
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.uniform(0.001, 0.01)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged, non‑negativity guaranteed
        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
    return a, score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size (keep the historic length)
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from previous global result if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build a small pool of diverse starting vectors
    # ------------------------------------------------------------------
    candidates = [
        a_start,
        spectral_init(n),
        random_initial(n, rng),
        add_noise(a_start, rng.uniform(0.001, 0.02), rng)
    ]

    # extra exponential shape
    exp_rate = rng.uniform(0.0005, 0.02)
    a_exp = np.exp(-exp_rate * np.arange(n))
    a_exp /= a_exp.sum()
    candidates.append(a_exp)

    # evaluate quickly and keep the best
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – multiple Adam restarts with a finer λ‑schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005
    ]
    steps_per_lam = 200
    restarts = 8
    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Adam restart {r+1}] new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = pgd_subgradient(best_a, deadline, n, fft_len, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[PGD subgradient] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.2,
            min_step=1e-7,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Hill‑climb] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – random local search (tiny transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = random_local_search(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Random local] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – convex‑combination exploration among top candidates
    # ------------------------------------------------------------------
    pool = [best_a]
    for _ in range(4):
        pool.append(add_noise(best_a, rng.uniform(0.001, 0.02), rng))
    pool_scores = [fast_score(v, fft_len, n) for v in pool]
    # keep top‑5
    ordered = sorted(zip(pool_scores, pool), key=lambda x: x[0])
    top_vectors = [v for _, v in ordered[:5]]
    top_scores = [s for s, _ in ordered[:5]]

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            top_vectors,
            top_scores,
            n,
            fft_len,
            deadline,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Mixture] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
