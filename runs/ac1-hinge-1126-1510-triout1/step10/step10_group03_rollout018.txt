def smoothed_gradient_optimize(a_start, deadline, n, fft_len_conv, fft_len_grad, rng,
                               alpha_start=2.0, alpha_end=40.0,
                               eta0=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter_cnt = 1
    phase_start = time.time()
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        # compute softmax weights
        progress = (time.time() - phase_start) / (deadline - phase_start)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        # using max for numerical stability
        scaled = conv - max_val
        w = np.exp(alpha * scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient via convolution with reversed a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_grad)
        fw = np.fft.rfft(w, fft_len_grad)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)
        eta = eta0 / (np.sqrt(iter_cnt) + 1)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = simplex_projection(a)
        # evaluate
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
        iter_cnt += 1
    return best_a, best_score
```

"""
Propose a sequence of non‑negative heights that minimizes
evaluate_sequence.  The approach consists of three stages:

  0.  A cheap parametric scan (exponentials, power‑laws, mixtures …)
      to obtain a decent seed.

  1.  Smoothed gradient descent on a log‑sum‑exp surrogate of
      max(conv(a,a)).  The surrogate is differentiable; the gradient
      is obtained by a single convolution with a weight vector that
      concentrates on the near‑max entries.  An Adam‑style update
      combined with Euclidean projection onto the probability simplex
      keeps the iterate feasible.

  2.  A final simulated‑annealing fine‑tuner that moves tiny amounts
      of mass between two indices while keeping the exact convolution
      up‑to‑date (O(n) per move).  The temperature schedule is slower
      than the original code so that uphill moves are accepted more
      often, allowing the search to escape shallow plateaus.

The whole routine respects a 1000 s budget, splits the time roughly
as 5 % / 70 % / 25 % among the three phases and returns the best
sequence found (also stored in the global variable
`height_sequence_1` for warm‑starting the next run).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ∥a∥₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force over a few low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # pure exponential
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=200):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # linear decreasing
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # mixture exponential + uniform
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=50):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # power‑law decay
    for p in np.linspace(0.2, 5.0, num=150):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smoothed_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_grad: int,
    rng: np.random.Generator,
    alpha_start: float = 2.0,
    alpha_end: float = 40.0,
    eta0: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam‑style projected gradient descent on
    J(a)=log‑sum‑exp(α·conv(a,a))/α.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 1
    phase_start = time.time()

    while time.time() < deadline:
        # --- convolution ---
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        # --- soft‑max weights (smooth max) ---
        max_val = conv.max()
        progress = (time.time() - phase_start) / (deadline - phase_start)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        scaled = conv - max_val               # ≤ 0, prevents overflow
        w = np.exp(alpha * scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum                      # shape (2*n-1,)

        # --- gradient of the smooth surrogate ---
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_grad)
        fw = np.fft.rfft(w, fft_len_grad)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # length n

        # --- Adam update + projection onto simplex ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a)

        # --- track best ---
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers with Metropolis acceptance and linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # a larger initial temperature gives more freedom to escape local minima
    temp0 = max(best_score * 0.08, 1e-6)
    start_time = time.time()
    max_fraction = 0.20          # max fraction of source mass moved per proposal
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n) work)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 is a good trade‑off; change if you want more freedom
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = next_pow2(3 * n - 2)          # for w*a_rev

    # initialise from previous run if possible
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.02   # a tiny safety margin

    # very rough time split
    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.78 * budget_s
    # remaining -> simulated annealing

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len_conv, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = smoothed_gradient_optimize(
            best_a, phase1_end, n,
            fft_len_conv, fft_len_grad,
            rng,
            alpha_start=2.0,
            alpha_end=40.0,
            eta0=0.2,
        )
        print(f"[smooth‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
