def propose_candidate(seed=0,budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4000
    # compute FFT lengths
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    # Initialize candidates
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # exponential with rate ~1/n
    rate = 1.0 / n
    a_exp = np.exp(-rate * np.arange(n))
    a_exp /= a_exp.sum()
    candidates.append(a_exp)
    # power law p = 0.5
    p = 0.5
    a_pow = (np.arange(1, n+1) ** (-p))
    a_pow /= a_pow.sum()
    candidates.append(a_pow)
    # random Dirichlet with alpha = 0.5
    a_rand = rng.gamma(0.5, 1.0, size=n)
    a_rand /= a_rand.sum()
    candidates.append(a_rand)
    # spectral
    candidates.append(spectral_init(n))
    # Add previous solution if exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'])
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.maximum(a_prev,0)
            a_prev /= a_prev.sum()
            candidates.append(a_prev)
    # Evaluate initial best
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best_score={best_score:.6f}")
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    # Phase 1: Adam restarts
    lam_schedule = [0.5,0.35,0.25,0.18,0.12,0.08,0.05,0.03,0.018,0.01,0.006,0.003,0.0015,0.0008,0.0004,0.0002,0.0001,0.00005,0.00002,0.00001]
    steps_per_lam = 200
    restarts = 8
    for r in range(restarts):
        if time.time() >= deadline:
            break
        # perturb best for each restart
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001,0.02), rng)
        a_opt, sc_opt = adam_optimize(a0, lam_schedule, steps_per_lam, deadline, fft_len_conv, fft_len_g, n)
        print(f"[Adam {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  new global best: {best_score:.6f}")
    # Phase 2: Exp gradient (long)
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(best_a, deadline, n, fft_len_conv, rng, scale=5.0)
        print(f"[ExpGrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  new best after ExpGrad")
    # Phase 3: Fine hill climb
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(best_a, best_score, fft_len_conv, n, deadline, init_step=0.2, min_step=1e-7)
        print(f"[Hill] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  new best after hill")
    # Phase 4: Mixture search among top 5
    # collect top candidates:
    top_vectors = [best_a]
    top_scores = [best_score]
    # add a few random ones
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)
    # keep best 5
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    scores_sel, vecs_sel = zip(*selected)
    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(list(vecs_sel), list(scores_sel), n, fft_len_conv, deadline)
        print(f"[Mix] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  new best after mix")
    # Final evaluation
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] score={final_val:.6f}")
    except Exception as e:
        print("[final] evaluation error:", e)
        final_val = None
    global height_sequence_1
    height_sequence_1 = final_seq
    return final_seq

    # line search: try shrink step sizes
    for factor in [1.0, 0.5, 0.25]:
        a_new = a - η * factor * g
        # project onto simplex
        a_new = simplex_projection(a_new) or set negative to zero and renorm.
        if fast_score(a_new) < current_score:
            a = a_new; current_score = fast_score(a_new); break
    η = η * 0.99   # decay

"""
Propose a low‑score step‑function sequence.

The search uses several complementary optimisation phases:

1.  Initialise a diverse pool of candidates (uniform, exponential,
    power‑law, random Dirichlet, spectral, and the previous best if any).
2.  Run many Adam restarts on a log‑sum‑exp surrogate of the max‑convolution.
    A decreasing “lam_factor’’ schedule refines the smoothing.
3.  Refine the current best with exponentiated‑gradient descent on the true
    max‑convolution (sub‑gradient) for a long run.
4.  Deterministic hill‑climbing that moves a small fraction of mass from the
    index that most contributes to the current maximal convolution entry to an
    index with the smallest partner value, shrinking the step size geometrically.
5.  Convex‑combination (mixture) search among the top few vectors found so far.
    A cheap golden‑section line‑search over the mixing weight is performed.
6.  The final vector is evaluated with the supplied `evaluate_sequence`
    routine, printed and stored as the historic variable `height_sequence_1`.

All helpers are top‑level (no closures) and the implementation respects the
time budget (default 1000 s).  The vector lives on the probability simplex
(`∑a = 1, a ≥ 0`) which is the natural normalisation for the evaluation
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector of various shapes."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Create a start vector from a flat square‑rooted spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed surrogate of the max‑convolution using a log‑sum‑exp.
    Returns (objective value, gradient w.r.t. logits `y`).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n,
                                            fft_len_conv,
                                            fft_len_g,
                                            lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that most contributes to the
    current maximal convolution entry toward a coordinate with the smallest
    partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # Choose destination with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among a small set of candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates, perform a cheap golden‑section line search on
    the mixing weight t∈[0,1] to minimise the bound.
    Returns the best vector and its score discovered in this phase.
    """
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]

            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)

            fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
            fd = fast_score((1.0 - d) * a + d * b, fft_len, n)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd = fast_score((1.0 - d) * a + d * b, fft_len, n)

            tm = 0.5 * (a0 + b0)
            fs = fast_score((1.0 - tm) * a + tm * b, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = ((1.0 - tm) * a + tm * b) / ((1.0 - tm) * a + tm * b).sum()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – keep 4000 (used by historic results)
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise a diverse pool of starting vectors
    # ------------------------------------------------------------------
    candidates = []

    # 1. uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2. exponential (rate ≈ 1/n)
    rate = 1.0 / n
    a_exp = np.exp(-rate * np.arange(n))
    a_exp /= a_exp.sum()
    candidates.append(a_exp)

    # 3. power‑law
    p = 0.8
    a_pow = (np.arange(1, n + 1, dtype=float) ** (-p))
    a_pow /= a_pow.sum()
    candidates.append(a_pow)

    # 4. random Dirichlet (alpha = 0.5)
    a_rand = rng.gamma(0.5, 1.0, size=n)
    a_rand /= a_rand.sum()
    candidates.append(a_rand)

    # 5. spectral initialiser
    candidates.append(spectral_init(n))

    # 6. previous global result if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.maximum(a_prev, 0.0)
            a_prev /= a_prev.sum()
            candidates.append(a_prev)

    # evaluate initial pool
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – multiple Adam restarts (smoothed surrogate)
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001
    ]
    steps_per_lam = 200     # more iterations per λ than before
    restarts = 8

    for r in range(restarts):
        if time.time() >= deadline:
            break
        # perturb the current best before each restart
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after Adam: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement (long run)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(

            deadline,
            n,
            fft_len_conv,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fine deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,

            fft_len_conv,
            n,
            deadline,
            init_step=0.2,
            min_step=1e-7,

        print(f"[Hill‑climb] score={sc_opt:.6f}")

            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – convex‑combination search among the top‑K vectors
    # --------------------------------------------------------------
    # Gather a few top candidates (including current best)
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the best K (K=5)
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len_conv,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
