"""
Propose a step‐function sequence with a low evaluation score.

The algorithm works in three stages:

1. **Extensive parametric sweep** – samples a variety of analytic families
   (uniform, pure exponential, linear ramps, mixtures with the uniform
   distribution, power‐law decays and a random mixture of two exponentials).
   The best candidate from this cheap exploration seeds the next stage.

2. **Exponentiated‑gradient (mirrored SGD)** – optimises the true
   non‑smooth objective `max(conv(a,a))`.  At each iteration a sub‑gradient
   of the active convolution coefficient(s) is computed using FFT, and
   a multiplicative update is performed.  The step size decays like
   `η₀/√t`.  Two passes are performed: a coarse pass with a larger
   initial step‑size followed by a fine pass with a much smaller
   `η₀`.

3. **Deterministic local descent** – repeatedly moves a tiny amount of
   mass from the index that contributes most to the current maximal
   convolution coefficient to the index that contributes least.
   Each move is accepted only if it strictly improves the fast (FFT‑based)
   score.  This fine‑grained hill‑climbing is able to push the solution
   a little further than the stochastic annealer of the reference code.

The whole search respects the 1000 s budget; the time budget is split
approximately 20 % / 60 % / 20 % among the three phases.  The best
sequence found is stored in the global `height_sequence_1` variable so
that subsequent runs can start from it.

All numbers are non‑negative; the final evaluation is performed with the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the official score.
    Assumes `a` is already normalised (sum = 1) – returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the L∞‑norm of `a*a` and the current
    maximal convolution value.
    Returns (grad, max_val) where `grad` has the same shape as `a`.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # reversed a for the sub‑gradient (because convolution is not symmetric)
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)

    # gradient of the active constraints (the mask)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    # keep the part that aligns with the original vector size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – rich parametric sweep
# ----------------------------------------------------------------------


def param_scan_extended(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Scan a collection of low‑dimensional families, including a random
    mixture of two exponentials.  Return the best candidate found.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # Helper to update best
    def upd(a, score):
        nonlocal best_a, best_score
        if score < best_score:
            best_score, best_a = score, a.copy()

    # ----- pure exponential -------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=300):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        upd(a, fast_score(a, fft_len))
        if time.time() >= deadline:
            return best_a, best_score

    # ----- linear decreasing ------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=300):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        upd(a, fast_score(a, fft_len))
        if time.time() >= deadline:
            return best_a, best_score

    # ----- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=80):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=13):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            upd(a, fast_score(a, fft_len))
            if time.time() >= deadline:
                return best_a, best_score

    # ----- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=200):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        upd(a, fast_score(a, fft_len))
        if time.time() >= deadline:
            return best_a, best_score

    # ----- random mixture of two exponentials -------------------------------
    # we draw a few thousand random triples (lam1, lam2, alpha)
    num_samples = 3000
    for _ in range(num_samples):
        lam1 = 10 ** rng.uniform(np.log10(1e-5), np.log10(0.05))
        lam2 = 10 ** rng.uniform(np.log10(1e-5), np.log10(0.05))
        alpha = rng.random()
        a = alpha * np.exp(-lam1 * np.arange(n)) + (1.0 - alpha) * np.exp(
            -lam2 * np.arange(n)
        )
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        upd(a, fast_score(a, fft_len))
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent (coarse + fine)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
):
    """
    Mirrored SGD with multiplicative updates.
    Returns the best vector encountered before `deadline`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (exponentiated gradient)
        exponents = -eta * g
        # prevent overflow/underflow
        exponents = np.clip(exponents, -50, 50)
        a = a * np.exp(exponents)

        # projection onto simplex
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic local descent guided by the sub‑gradient
# ----------------------------------------------------------------------


def local_descent(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    fraction: float = 0.01,
):
    """
    Repeatedly move a small amount of mass from the index that most
    contributes to the current maximal convolution coefficient to the
    index that contributes the least.  Only improving moves are kept.
    """
    a = a_start.copy()
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    while time.time() < deadline:
        # compute sub‑gradient of the active constraints
        g, _ = subgradient(a, fft_len)

        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] <= 0.0:
            # fall back to a random pair if the guided one is useless
            src, dst = rng.integers(0, n, size=2)
            if src == dst:
                continue

        # take a modest fraction of the source mass
        delta = a[src] * rng.random() * fraction
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        # re‑normalise (numerical drift may appear)
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()

        new_score = fast_score(a_candidate, fft_len)

        # accept only if we improve
        if new_score < curr_score:
            a = a_candidate
            curr_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()

    return best_a, best_score

    if time.time() < deadline:
        best_a, best_score = local_descent(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
            fraction=0.01,
        )
        print(f"[local‑descent] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the result for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
