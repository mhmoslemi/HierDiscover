```python
"""
Optimization of non‑negative step function to minimize
2·n·max(conv(a,a))/(sum a)^2.
We use a multi‑phase approach:
    1. Diverse initial vectors.
    2. Adam on a log‑softmax surrogate (smooth max) with many
       annealing‑like λ‑schedule.
    3. Exponentiated‑gradient on the true max‑convolution.
    4. Simulated‑annealing / hill‑climb local search.
    5. Pairwise mixture (golden‑section) among the best few.
The whole search respects the 1000 s budget.
"""
import time
import numpy as np

def next_pow2(x):
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a, fft_len, n):
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def random_initial(n, rng):
    """Generate a random probability vector using several patterns."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sinc'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # mode == 'sinc'
        x = np.arange(n) / (n - 1)
        a = np.abs(np.sinc(4 * (x - 0.5))) + 1e-12
    a /= a.sum()
    return a

def add_noise(a, sigma, rng):
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def spectral_init(n):
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30):
    """Smoothed max of self‑convolution via log‑sum‑exp.
    Returns (value, gradient w.r.t y)."""
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t conv entries
    w = exp_shift / Z   # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t a (size n) is slice (n-1) to (2n-2)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def adam_optimize(a_start, lam_schedule, steps_per_lam, deadline,
                  fft_len_conv, fft_len_g, n):
    """Adam optimiser on the smoothed objective."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv,
                                            fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize(a_start, deadline, n, fft_len_conv,
                         rng, scale=5.0):
    """Exponentiated‑gradient on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def hill_climb(a, score, fft_len, n, deadline,
               init_step=0.15, min_step=1e-6):
    """Deterministic hill‑climbing moving mass from a bad coordinate."""
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside range)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner (avoid i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.7, min_step)
    return a, score

def mixture_search(candidates, scores, n, fft_len, deadline):
    """Golden‑section line search on convex combinations of pairs."""
    def eval_mix(t, a, b):
        mix = (1.0 - t) * a + t * b
        mix /= mix.sum()
        return fast_score(mix, fft_len, n), mix

    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b)
            fd, _ = eval_mix(d, a, b)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b)

            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score

def anneal_optimize(a_start, deadline, n, fft_len, rng):
    """Simple simulated‑annealing on the exact score."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    # initial temperature proportional to current score
    T = best_score * 0.02
    min_T = 1e-8

    while time.time() < deadline:
        # pick source index with mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # pick destination index different from i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # move a fraction (up to 10%) of source mass
        delta = a[i] * rng.uniform(0.0, 0.1)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise to offset numerical drift
        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new /= s
        new_score = fast_score(a_new, fft_len, n)
        dE = new_score - best_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        # geometric cooling
        T = max(T * 0.9995, min_T)
    return best_a, best_score

"""
Optimisation of a non‑negative step function to minimise

    2·n·max(conv(a,a)) / (∑a)²

where `conv` is the full linear convolution.  The method consists of
five phases, each respecting the overall 1000 s budget:

1. Build a diverse pool of initial probability vectors (uniform,
   random, exponential, power‑law, edge‑heavy, sinc‑shaped, spectral,
   plus a warm‑start from the previously found best).
2. Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with a
   decreasing “λ‑schedule”.  Several restarts and many steps per λ
   are performed.
3. Exponentiated‑gradient descent on the true max‑convolution.
4. Local stochastic annealing (random mass transfers with Metropolis
   acceptance) followed by a deterministic hill‑climb that moves a
   small fraction of mass from the worst to the best coordinate.
5. Convex‑combination (golden‑section) search among the top‑K vectors.

The best vector is finally evaluated with the provided
`evaluate_sequence` routine and returned.  All vectors stay on the
simplex (∑a = 1, a ≥ 0) throughout the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) (assumes ∑a ≈ 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers (various shapes)
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector using several patterns."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sinc']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # mode == 'sinc'
        x = np.arange(n) / (n - 1)
        a = np.abs(np.sinc(4 * (x - 0.5))) + 1e-12
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the self‑convolution via log‑sum‑exp.
    Returns (value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    steps_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    rng: np.random.Generator,
    scale: float = 5.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the hard max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb with geometric step reduction
# ----------------------------------------------------------------------
def hill_climb(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_step: float = 0.15,
    min_step: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from the worst to the best coordinate."""
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.7, min_step)
    return a, score


# ----------------------------------------------------------------------
# Convex‑combination search among top candidates
# ----------------------------------------------------------------------
def eval_mix(t: float, a: np.ndarray, b: np.ndarray, n: int, fft_len: int) -> tuple[float, np.ndarray]:
    """Linear mixture of two probability vectors (renormalised)."""
    mix = (1.0 - t) * a + t * b
    mix /= mix.sum()
    return fast_score(mix, fft_len, n), mix


def mixture_search(
    candidates,
    scores,
    n,
    fft_len,
    deadline,
) -> tuple[np.ndarray, float]:
    """Gold‑section search on convex combinations of all pairs."""
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b, n, fft_len)
            fd, _ = eval_mix(d, a, b, n, fft_len)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b, n, fft_len)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b, n, fft_len)

            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b, n, fft_len)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing optimiser on the exact score
# ----------------------------------------------------------------------
def anneal_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass transfers with Metropolis acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    # start temperature proportional to the current score
    T = best_score * 0.02
    min_T = 1e-8

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.uniform(0.0, 0.1)
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        dE = new_score - best_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        T = max(T * 0.9995, min_T)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    `evaluate_sequence` score.  The routine respects the time budget,
    runs several optimisation phases and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the historic length (4000) for compatibility.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build a small set of diverse seed vectors.
    # ------------------------------------------------------------------
    cand0 = a_start.copy()
    cand1 = spectral_init(n)
    cand2 = random_initial(n, rng)
    cand3 = add_noise(cand0, rng.uniform(0.001, 0.01), rng)
    candidates = [cand0, cand1, cand2, cand3]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for c in candidates[1:]:
        sc = fast_score(c, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    start_time = time.time()
    total_deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a smooth maximum (≈45 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002,
    ]
    steps_per_lam = 600          # more iterations per λ
    max_restarts = 12
    for r in range(max_restarts):
        if time.time() >= phase1_deadline:
            break
        a0 = best_a.copy() if r == 0 else add_noise(best_a, rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            phase1_deadline,
            fft_len,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (≈25 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len,
            rng,
            scale=6.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after Exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.85 * budget_s
    if time.time() < phase3_deadline:
        a_opt, sc_opt = anneal_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[Anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after anneal: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic hill‑climb (≈10 % of budget)
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.95 * budget_s
    if time.time() < phase4_deadline:
        a_opt, sc_opt = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase4_deadline,
            init_step=0.12,
            min_step=1e-7,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Convex‑combination (mixture) search among top K vectors
    # ------------------------------------------------------------------
    extra_vecs = []
    extra_scores = []
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len, n)
        extra_vecs.append(a_tmp)
        extra_scores.append(sc_tmp)

    all_vecs = [best_a] + extra_vecs
    all_scores = [best_score] + extra_scores
    ordered = sorted(zip(all_scores, all_vecs), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    phase5_deadline = total_deadline
    if time.time() < phase5_deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len,
            phase5_deadline,
        )
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the security‑checked routine.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the vector for future runs
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
