"""
Propose a step‑function (non‑negative heights) with a low evaluation score.

The optimizer works directly on the probability simplex by representing the
sequence as a soft‑max of real logits `x`.  A smooth approximation of the
max‑convolution (log‑sum‑exp) is used as a differentiable loss; its gradient
is computed analytically via FFT‑based convolutions.  An Adam‑style update
with a gradually increasing temperature `τ` (which makes the smooth max
approach the true max) drives the solution towards a flat autocorrelation.
The best sequence seen (according to the fast FFT‑based score, which equals
the official score when the sum of the entries is 1) is kept and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(convolution(a, a)).
    The function assumes `a` is a probability vector (sum ≈ 1),
    but does not enforce it – it simply returns the required expression.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the given
    evaluation function.  The method uses a soft‑max parameterisation,
    a smoothed max‑convolution loss and Adam optimisation with a
    continuation on the temperature `τ`.  The best sequence found
    before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a few thousand entries gives a good balance.
    # --------------------------------------------------------------
    n = 4000
    # FFT length that can handle a convolution of lengths n and (2n‑1)
    fft_len = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise from a previous run if possible, otherwise uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the required length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # Ensure a proper probability vector
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    # Keep the best vector seen so far (we store the probability version)
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Logits for a soft‑max parametrisation (so sum(a)=1 automatically)
    # --------------------------------------------------------------
    eps = 1e-14
    x = np.log(best_a + eps)          # initial logits
    m = np.zeros_like(x)              # Adam first moment
    v = np.zeros_like(x)              # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin
    it = 1

    # temperature τ for the smooth max – start small, increase slowly
    tau = 1.0
    tau_max = 200.0
    tau_factor = 1.03          # multiply every few hundred iterations

    # learning‑rate schedule (Adam base‑lr)
    lr0 = 0.3

    while time.time() < deadline:
        # ----- convert logits to probability vector -----
        mx = np.max(x)
        p = np.exp(x - mx)
        p /= p.sum()                     # now ∑p = 1

        # ----- convolution p * p (FFT) -----
        fa = np.fft.rfft(p, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # ----- smooth max weights (softmax over convolution) -----
        # stability: shift by the maximum before exponentiating
        cmax = conv.max()
        exp_term = np.exp(tau * (conv - cmax))
        w = exp_term / exp_term.sum()    # length 2n‑1, sums to 1

        # ----- gradient of the smooth loss w.r.t. p -----
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa * fw, fft_len)[: 3 * n - 2]
        g = 2.0 * grad_full[n - 1 : n - 1 + n]   # shape (n,)

        # ----- chain rule: gradient w.r.t. logits x -----
        dot_pg = np.dot(p, g)
        grad_x = p * (g - dot_pg)               # ∂L/∂x

        # ----- Adam update (minimise) -----
        lr = lr0 / np.sqrt(it)                  # simple decay
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- keep track of the best vector (using the cheap FFT score) -----
        if it % 30 == 0:
            cur_score = fast_score(p, fft_len)
            if cur_score < best_score:
                best_score, best_a = cur_score, p.copy()
                print(f"[iter {it:6d}] new best fast_score = {best_score:.6f}")

        # ----- increase temperature every few hundred steps -----
        if it % 300 == 0 and tau < tau_max:
            tau = min(tau * tau_factor, tau_max)

        it += 1

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for a possible warm‑start in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
