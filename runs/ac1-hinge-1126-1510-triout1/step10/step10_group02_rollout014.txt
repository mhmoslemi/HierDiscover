def targeted_local_search(a, cur_score, deadline, n, rng):
    a = a.copy()
    best_a = a.copy()
    best_score = cur_score
    # precompute conv for speed
    while time.time() < deadline:
        # compute conv once
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # find all indices where conv is close to max (within tolerance)
        eps = max_val * 1e-6 + 1e-12
        ks = np.where(conv >= max_val - eps)[0]
        # choose a random k from ks
        k = rng.choice(ks)
        # find index pair with largest product that contributes to this k
        # iterate over i in allowed range:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # compute product list
        # for speed, vectorize:
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        products = a[i_vals] * a[j_vals]
        # find i with largest product
        idx = np.argmax(products)
        i_star = i_vals[idx]
        j_star = j_vals[idx]
        # Move delta from larger of a_i_star, a_j_star to a random other index
        src = i_star if a[i_star] >= a[j_star] else j_star
        dst = rng.integers(0, n)
        if src == dst: continue
        # propose move
        delta = a[src] * rng.random() * 0.1  # up to 10% of src
        if delta <= 0: continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # renormalize to sum = 1
        cand = cand / cand.sum()
        # compute new score
        cand_score = fast_score(cand, fft_len)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = cand.copy()
        # optional acceptance for SA to avoid local minima
        else:
            # accept with some probability decreasing over time
            t = (deadline - time.time()) / (deadline - start_time)
            temp = 1e-3 * t
            if rng.random() < np.exp(-(cand_score - best_score)/temp):
                a = cand
    return best_a, best_score

    v = np.zeros_like(a)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))

        # Actually gradient for exponentiated gradient: a <- a * exp(-eta * g), corresponds to y <- y - eta * g.
        # So we can treat y = log(a) and update y with Adam.

            y = np.log(a + 1e-12)
        # gradient wrt y:
        grad_y = -g  # because y = log a, and we want to move opposite direction to reduce score? Actually we previously used exp(-eta * g)
        # So delta y = -eta * g
        # We'll compute Adam on grad_y
        t += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)

def simulated_annealing_targeted(a_start, start_score, deadline, n, rng):
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.1, 1e-6)  # start relatively high
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1 - frac), 1e-12)

        # propose a move type
        move_type = rng.choice(['pair', 'triplet', 'scale'])
        if move_type == 'pair':
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j: continue
            # move some mass from larger to smaller
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0: continue
            delta = a[src] * rng.random() * 0.5  # up to 50%
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
        elif move_type == 'triplet':
            i, j, k = rng.integers(0, n, size=3)
            # move from i to j and k proportionally
            if a[i] <= 0: continue
            delta = a[i] * rng.random() * 0.3
            cand = a.copy()
            cand[i] -= delta
            # split delta equally
            cand[j] += delta * 0.5
            cand[k] += delta * 0.5
        else:  # scale
            idx = rng.integers(0, n)
            factor = 1.0 + rng.uniform(-0.5, 0.5)  # between 0.5 and 1.5
            cand = a.copy()
            cand[idx] = max(cand[idx] * factor, 0.0)
            # re-normalize
            cand /= cand.sum()

        # reproject to simplex is automatic for pair/triplet proposals due to mass conservation; for scale we renormalized
        cand_score = fast_score(cand, next_pow2(2*n - 1))
        dE = cand_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / temp):

"""
Propose a low‑score step‑function via a multi‑phase search.

Phase 1 – many random starts are refined by a cascade of L‑BFGS
optimisations on a smoothed log‑sum‑exp approximation of the
max‑convolution.  The λ‑schedule is long and reaches very small
values, allowing the optimiser to approach the true L∞‑objective.

Phase 2 – a mirror‑descent (exponentiated‑gradient) optimiser with an
Adam‑style momentum term is run on the exact max‑convolution until
the deadline for this phase is reached.

Phase 3 – a targeted simulated‑annealing hill‑climb (pair, triplet and
scale moves) finishes the search, using a cooling schedule that
adapts to the remaining time.

All iterates remain on the probability simplex, thus the final list
can be passed directly to ``evaluate_sequence``.  The best sequence
found before the 1000 s budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic helpers
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of: 2·n·max(conv(a,a))   (a is assumed to sum to 1).
    The convolution is performed by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def low_overlap_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a vector supported on a random subset of size ≈2·sqrt(n)
    with equal weight.  This often yields small max‑convolution.
    """
    m = max(10, int(2 * np.sqrt(n)))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = rng.random(m) + 1e-6
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project to the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# --------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Work in log‑space y where a = softmax(y).
    """
    # softmax → a on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# --------------------------------------------------------------
# Mirror‑descent with Adam momentum on the exact max‑convolution
# --------------------------------------------------------------

def exp_gradient_optimize_adam(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               eta0: float = 10.0,
                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8,
                               max_iters: int = 10_000_000):
    """
    Adam‑style exponentiated‑gradient on the true L∞‑objective.
    The iterate stays on the simplex via y = log(a) updates.
    """
    a = a_start.copy()
    y = np.log(a + 1e-12)          # log‑domain
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # compute convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # gradient of hard max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # Adam on the log‑variables (y ← y − η·g)
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta_t = eta0 / (np.sqrt(it) + 1.0)

        y = y - eta_t * m_hat / (np.sqrt(v_hat) + eps)
        # back to a, keep on simplex
        a = np.exp(y - np.max(y))
        a_sum = a.sum()
        if a_sum == 0.0:
            a = best_a.copy()
        else:
            a /= a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Targeted simulated‑annealing hill‑climb
# --------------------------------------------------------------

def simulated_annealing_targeted(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Random mass‑transfer moves (pair / triplet / scale) accepted with a
    Metropolis rule.  Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()

    # initial temperature: a modest fraction of the current score
    temp0 = max(best_score * 0.1, 1e-6)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['pair', 'triplet', 'scale'])
        if move_type == 'pair':
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # move mass from larger to smaller entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * 0.5   # up to 50 % of src
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
        elif move_type == 'triplet':
            i, j, k = rng.integers(0, n, size=3)
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.3
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta * 0.5
            cand[k] += delta * 0.5
        else:  # scale a single entry
            idx = rng.integers(0, n)
            factor = 1.0 + rng.uniform(-0.5, 0.5)   # between 0.5 × and 1.5 ×
            cand = a.copy()
            cand[idx] = max(cand[idx] * factor, 0.0)
            total = cand.sum()
            if total == 0.0:
                continue
            cand /= total

        # keep the candidate on the simplex (pair/triplet already conserve mass)
        cand_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = cand_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True
        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # ----------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ----------------------------------------------------------
    # Build a diverse pool of starting vectors
    # ----------------------------------------------------------
    pool = []

    # 1) Use the previous construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families
    for _ in range(6):
        pool.append(random_initial(n, rng))
    # 3) Low‑overlap family
    for _ in range(3):
        pool.append(low_overlap_initial(n, rng))

    # Classic windows (if available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ----------------------------------------------------------
    # Time budget split
    # ----------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS phase
    phase2_end = start_time + 0.80 * budget_s   # Adam‑mirrored SGD phase
    # remaining → targeted simulated annealing

    # ----------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smoothed objective
    # ----------------------------------------------------------
    lam_schedule = [
        0.8, 0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001, 0.000005
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # occasional noise to explore new basins
        if rng.random() < 0.25:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.015), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ----------------------------------------------------------
    # Phase 2 – Adam‑mirrored SGD on the true max‑convolution
    # ----------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize_adam(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=12.0,
            max_iters=8_000_000
        )
        print(f"[exp‑grad‑adam] fast_score={best_score:.9f}")

    # ----------------------------------------------------------
    # Phase 3 – targeted simulated‑annealing hill‑climb
    # ----------------------------------------------------------
    if time.time() < deadline:

            best_a,

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)

    globals()['height_sequence_1'] = final_seq
    return final_seq
