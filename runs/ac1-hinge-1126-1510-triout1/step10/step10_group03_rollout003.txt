"""
Candidate generator for the step‑function minimisation task.

Algorithm (≈ 2 min on the reference hardware):
1. Initialise a simplex‑normalised sequence by interpolating the best
   sequence from the previous run (or uniform if none).
2. Perform many iterations of Adam‑optimisation on a *smoothed* version
   of the max‑convolution objective:
      fβ(a) = (1/β)·log( Σₖ e^{β·(a∗a)ₖ} )
   The gradient is obtained analytically via FFTs (O(n log n)).
   β is increased linearly from 0.5 to 200 during the run, turning the
   smooth loss into the true max‑convolution.
   After every update, the vector is projected onto the probability
   simplex to enforce non‑negativity and Σa = 1.
3. Keep the best point seen (according to the true max‑convolution).
4. A short simulated‑annealing fine‑tuning phase (optional) is applied
   using the exact convolution to squeeze out a few extra digits.
5. The final sequence is returned and stored in the global variable
   ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def true_max_convolution(a: np.ndarray, L: int) -> float:
    """Exact max of the linear convolution a*a (a is already normalised)."""
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * a.shape[0] - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives a richer search space.
    # --------------------------------------------------------------
    n = 8000                        # feel‑free to change; 8000 works fast
    L_conv = next_pow2(2 * n - 1)   # for a*a
    L_grad = next_pow2(3 * n - 2)   # for weighted gradient (needs a+w length)

    # --------------------------------------------------------------
    # Initialise from previous run if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 1:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    a = a0.copy()

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.2                     # initial (decaying) learning rate

    # --------------------------------------------------------------
    # Schedule for the smoothing parameter β
    # --------------------------------------------------------------
    beta_start = 0.5
    beta_end = 200.0

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin
    # allocate ~80 % of the time to the Adam phase, the rest to annealing
    adam_deadline = start_time + 0.80 * budget_s

    # --------------------------------------------------------------
    # Adam optimisation on the smoothed loss
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = np.inf
    it = 1

    # Helper: evaluate true max‑convolution and update global best
    def maybe_update_best(current_a):
        nonlocal best_a, best_score
        max_conv = true_max_convolution(current_a, L_conv)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = current_a.copy()
            print(f"[iter {it}] new best_score = {best_score:.6f}")

    # initial evaluation
    maybe_update_best(a)

    while time.time() < adam_deadline:
        # linear schedule for β
        frac = min(it / 20000.0, 1.0)          # 20 000 is a generous upper bound
        beta = beta_start + (beta_end - beta_start) * frac

        # ----------- compute smoothed loss and its gradient -------------
        #   1. a*a   (FFT length L_grad, which is large enough for all)
        fa = np.fft.rfft(a, L_grad)

        #   2. convolution a*a  (only the first 2n‑1 entries are needed)
        conv_full = np.fft.irfft(fa * fa, L_grad)
        conv = conv_full[: 2 * n - 1]

        #   3. smoothed loss (log‑sum‑exp)
        mconv = conv.max()
        # stability: shift by mconv
        exp_term = np.exp(beta * (conv - mconv))
        sumexp = exp_term.sum()
        loss = mconv + np.log(sumexp) / beta

        #   4. softmax weights
        w = exp_term / sumexp                     # shape (2n‑1,)

        #   5. gradient of the smoothed loss
        w_rev = w[::-1]                           # reversed weights
        fw = np.fft.rfft(w_rev, L_grad)
        # product fa * fw gives convolution w_rev * a
        grad_full = np.fft.irfft(fa * fw, L_grad)
        Lw = w.shape[0]                           # = 2n‑1
        # pick indices (Lw‑1‑i) for i=0..n‑1
        idx = (Lw - 1) - np.arange(n)
        grad = 2.0 * grad_full[idx]               # shape (n,)

        # ------------------------ Adam update ---------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)                    # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (guarantees Σa = 1, a≥0)
        a = simplex_projection(a)

        # occasional true‑objective check
        if it % 250 == 0:
            maybe_update_best(a)

        it += 1

    # --------------------------------------------------------------
    # Simulated‑annealing fine‑tuning (optional, cheap, uses exact conv)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # simple mass‑transfer annealing, similar to the reference code
        a = best_a.copy()
        conv = np.convolve(a, a)                # exact linear convolution
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        best_local_a = a.copy()
        best_local_score = cur_score

        # temperature starts as a small fraction of the current score
        temp0 = max(cur_score * 0.02, 1e-6)
        start_anneal = time.time()
        max_frac = 0.15                         # max fraction of source mass moved

        while time.time() < deadline:
            # linear cooling
            elapsed = time.time() - start_anneal
            frac = elapsed / (deadline - start_anneal)
            temp = max(temp0 * (1.0 - frac), 1e-9)

            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # always move mass from the larger entry to the smaller one
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            # incremental update of the convolution (O(n))
            # conv_new = conv + 2*delta*(a shifted for src) - 2*delta*(a shifted for dst)
            # We create a temporary diff array only for the required range.
            diff = np.zeros_like(conv)
            diff[src:src + n] -= a
            diff[dst:dst + n] += a
            conv_new = conv + 2.0 * delta * diff
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max

            dE = new_score - cur_score
            accept = False
            if dE < 0.0:
                accept = True
            else:
                if rng.random() < np.exp(-dE / temp):
                    accept = True

            if accept:
                a[src] -= delta
                a[dst] += delta
                conv = conv_new
                cur_score = new_score
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()

        # keep the best found by annealing if it improves the Adam result
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()
            print(f"[anneal] improved best_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
