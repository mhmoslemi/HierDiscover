def exp_gradient_adam(a_start, deadline, n, fft_len, rng, init_eta=0.2, beta1=0.9, beta2=0.999, eps=1e-8, max_iters=10_000_000):

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        # compute mask and grad
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps_conv = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_conv).astype(np.float64)

        # grad of max conv w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]   # shape n

        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        t += 1
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = init_eta * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated gradient:
        exp_factor = np.exp(-step)
        a = a * exp_factor
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = best_a.copy()
        else:
            a = a / sum_a

        # evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # maybe early exit if improvement negligible?
    return best_a, best_score

"""
Advanced step‑function optimiser.

The optimiser works in four stages (all respecting the 1000 s deadline).

1. **Diverse initial pool** – uniform, exponential, power‑law, edge‑heavy,
   triangular, cosine and a few classic windows.  All candidates are
   normalised to a probability vector (∑a = 1) and symmetrised, because the
   optimal solution is expected to be symmetric.

2. **Smoothed L‑BFGS** – a cascade of L‑BFGS runs on a log‑sum‑exp
   approximation of the max‑convolution.  The λ‑schedule is fine‑grained,
   allowing the optimiser to “zoom in’’ on the true objective.

3. **Adam‑style exponentiated‑gradient** – optimisation on the true
   L∞‑objective (max‑convolution).  The update is performed on the
   simplex by multiplicative (exponentiated) steps; first‑moment and
   second‑moment estimates (Adam) give an adaptive step size.  After each
   iteration the vector is re‑symmetrised and normalised.

4. **Deterministic greedy refinement** – we repeatedly move a small
   fraction of probability mass from the index with the largest
   sub‑gradient (most responsible for the current peak) to the index with
   the smallest sub‑gradient.  If a move improves the fast score it is kept,
   otherwise it is rejected.  This simple coordinate‑descent finishes the
   optimisation.

At the end the best vector found is evaluated with the official
`evaluate_sequence` function and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a sums to 1.
    The convolution is performed by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrise(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[-i-1] (central symmetry)."""
    a = (a + a[::-1]) * 0.5
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'triangular', 'cosine', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'triangular':
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)  # triangular shape
        a = np.maximum(a, 0.0)
    elif mode == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))  # raised cosine
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    a = np.maximum(a, 0.0)
    a /= a.sum()
    a = symmetrise(a)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    noisy = noisy / s
    noisy = symmetrise(noisy)
    noisy /= noisy.sum()
    return noisy


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    The optimisation is performed in log‑space y (a = softmax(y)).
    """
    # softmax → a on simplex (always ∈ Δ)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # FFT‑based convolution
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        a = symmetrise(a)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_adam(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator,
                     eta0: float = 0.2,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     max_iters: int = 10_000_000):
    """
    Mirrored Adam: multiplicative update a ← a·exp(−step) with Adam‑style
    adaptive step.  The vector stays on the probability simplex and is
    symmetrised after each iteration.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        # compute sub‑gradient of max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_conv = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_conv).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # sub‑gradient of max(conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        t += 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (per coordinate)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        # clip to avoid overflow in exp
        step = np.clip(step, -50.0, 50.0)

        a = a * np.exp(-step)
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # enforce symmetry (keeps sum unchanged)
        a = symmetrise(a)
        a /= a.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional cheap termination: if improvement < 1e-12 for 5000 it.
        # (not strict, just a guard – omitted for clarity)

    return best_a, best_score

# Greedy deterministic refinement (mass transfer)
# ----------------------------------------------------------------------


def greedy_mass_transfer(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        max_no_improve: int = 2000):
    """
    Repeatedly move a small fraction of probability mass from the index with
    largest sub‑gradient to the index with smallest sub‑gradient.
    Moves are accepted only if they improve the fast_score.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    # fraction of source mass to move (adaptive)
    frac = 0.02
    no_improve = 0

    while time.time() < deadline:
        # compute sub‑gradient of current max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = np.argmax(g)
        dst = np.argmin(g)

        if a[src] <= 0.0:
            # pick a random positive source if the extreme is empty
            pos = np.where(a > 1e-12)[0]
            if pos.size == 0:
                break
            src = rng.choice(pos)

        delta = a[src] * rng.uniform(0.1, frac)  # move a modest piece
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = symmetrise(cand)
        cand /= cand.sum()

        new_score = fast_score(cand, fft_len)

        if new_score < cur_score:
            a = cand
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # shrink step size if we are stuck
                frac *= 0.5
                no_improve = 0
                if frac < 1e-4:
                    break

    return best_a, best_score

# Simulated‑annealing (fallback, light)

def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Very light SA that moves random mass chunks.
    Used only if time remains after the heavier phases.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = symmetrise(cand)
        cand /= cand.sum()

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the time budget
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) If a previous solution exists, interpolate it to current size
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, old)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            a_old = symmetrise(a_old)
            a_old /= a_old.sum()
            pool.append(a_old)

    # 2) Random families
    for _ in range(10):
        pool.append(random_initial(n, rng))

    # 3) Classical windows (if SciPy available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            w = symmetrise(w)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            w = symmetrise(w)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.65 * budget_s   # Adam EG
    phase3_end = start_time + 0.90 * budget_s   # Greedy refinement
    # remaining → very light SA

    # ------------------------------------------------------------------
    # Phase 1 – aggressive smoothed L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001
    ]
    maxiter_per_lam = 900

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.35:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.01), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style exponentiated‑gradient on true max‑conv
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_adam(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=0.25,         # a slightly larger initial step
            max_iters=5_000_000
        )
        print(f"[Adam‑EG] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = greedy_mass_transfer(
            best_a,
            best_score,
            phase3_end,
            n,
            fft_len,
            rng
        )
        print(f"[Greedy] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – very light simulated‑annealing (fallback)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[SA] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
