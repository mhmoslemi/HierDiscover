"""
Improved search for a low‑score non‑negative step function.

Key ideas
---------
* Work with a larger number of points (default 8000).  The
  asymptotic optimum of the objective 2·n·max(conv) is known to be
  around 1.5, and a larger n lets us get closer to this limit.
* Keep the multi‑stage strategy of the previous solution, but
  – allocate more time to the gradient‑based phases,
  – use a finer exponential‑rate grid that scales with n,
  – tighten the sub‑gradient step‑size schedule,
  – run a longer simulated‑annealing phase.
* All heavy lifting (convolutions, gradients) is done with FFTs,
  which makes the extra work for n≈8000 still comfortably inside the
  1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (output length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate for the official score:
    2·n·max(conv) when Σa = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# --------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(conv) where a = softmax(y).
    Returns (value, gradient w.r.t. y).
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # ----- smooth max via log‑sum‑exp -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv entries -----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # back‑propagation through convolution (conv = a * a)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient of smooth_max w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f_val), grad_y

    """
    Run L‑BFGS for every λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best vector seen.
    """
    y = np.log(a_start + 1e-12)          # logits → stays on simplex
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as exc:               # pragma: no cover
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the true max‑conv."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9                     # a bit more momentum than before
    it = 0

    while time.time() < deadline:
        # ----- true max‑convolution (full) -----
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the *max* (see original code)
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len_g)
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        conv_g = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # decreasing step size – slower decay for fine tuning
        step = 0.8 / np.sqrt(it + 1.0)

        a_candidate = simplex_projection(a - step * v)
        cand_sc = fast_score(a_candidate, fft_len_conv, n)

        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction of mass between
    two indices; the temperature follows a square‑root cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    # initial temperature – a noticeable fraction of the current score
    T0 = max(0.10 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)   # sqrt cooling

        i = rng.integers(n)                     # source
        j = rng.integers(n - 1)                 # destination (different)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue
        # transfer up to 35 % of source mass
        delta = a[i] * rng.random() * 0.35
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc

# Parametric families
# --------------------------------------------------------------

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Systematic sweep over exponential, power‑law and logistic families.
    Returns the best vector found.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # ----- exponential family -------------------------------------------------
    # Rate is expressed as c / n, with c ∈ [0.8, 4.0]
    for c in np.linspace(0.8, 4.0, 45):
        if time.time() >= deadline:
            return best_a, best_sc
        rate = c / n
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp c={c:.3f} → score={best_sc:.6f}")

    # ----- power‑law family ----------------------------------------------------
    for p in np.linspace(-5.0, -0.5, 22):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # ----- logistic family ------------------------------------------------------
    mus = np.linspace(0.15 * n, 0.85 * n, 9)
    sigmas = np.linspace(0.08 * n, 0.20 * n, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Quick random sampling of a handful of families – serves as a sanity‑check.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "power", "exp", "gauss", "cos", "bimodal",
        "logistic", "binary_sparse", "double_exp", "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_sc:.6f} (family={fam})")

    return best_a, best_sc


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a vector from a simple parametric family and normalise it."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.5, 3.0) / n          # same scaling as grid
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(0.08, 0.22) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(0.1, 0.5) * n
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, 0.45) * n
        mu2 = rng.uniform(0.55, 1.0) * n
        sigma1 = rng.uniform(0.08, 0.15) * n
        sigma2 = rng.uniform(0.08, 0.15) * n
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(0.08, 0.20) * n
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.5, 3.0) / n
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step sequence under the given
    time budget.  The function returns a list of floats (the sequence)
    before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives a tighter asymptotic bound.
    # --------------------------------------------------------------
    n = 8000                                          # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)               # for a*a
    fft_len_g = next_pow2(3 * n - 2)                  # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform, possibly interpolated old result)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate old sequence to new length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_sc:
                best_a, best_sc = a1, sc1
                print(f"[init] interpolated old best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – systematic grid search + cheap random sweep
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_sc:
        best_a, best_sc = a_grid, sc_grid
        print(f"[grid] best fast_score={best_sc:.6f}")

    # a short random sweep (helps escape pathological zones)
    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_sc:
        best_a, best_sc = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Timing allocation for the remaining phases
    # --------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s          # more time for L‑BFGS
    phase2_end = start_time + 0.80 * budget_s          # majority for sub‑gradient
    phase3_end = start_time + 0.97 * budget_s          # final annealing
    # leftovers → final evaluation

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (more λ steps, less maxiter)
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.30, 0.20, 0.14, 0.09, 0.06, 0.04,
        0.025, 0.015, 0.009, 0.005, 0.003, 0.0015,
        8e-04, 4e-04, 2e-04, 1e-04, 5e-05, 2e-05,
        1e-05, 5e-06,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional jitter to avoid stagnation
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            # fresh exponential start
            c = rng.uniform(0.8, 4.0)
            rate = c / n
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (longer)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] sub‑gradient best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (final stochastic polishing)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
