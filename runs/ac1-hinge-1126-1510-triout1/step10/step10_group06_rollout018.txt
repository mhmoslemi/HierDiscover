def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Simple yet powerful simulated‑annealing optimiser for the step‑function
evaluation task.

The code follows three very cheap stages:

1.  Initialise from a uniform distribution or from the previously‑found
    best vector (`height_sequence_1` if it exists).  Also try a few
    random sparse masks (equal mass on a random subset of indices) – this
    often yields a starting point with a lower peak autocorrelation than
    the plain uniform vector.

2.  Run a long simulated‑annealing loop (the bulk of the time budget).
    Each move transfers a random fraction of the mass from one index to
    another while keeping the total sum exactly 1.  Acceptance follows the
    classic Metropolis rule with a square‑root cooling schedule.
    Occasionally the current state is “jittered’’ (small Gaussian noise)
    to escape flat plateaus.

3.  Finally the best vector found is evaluated with the official
    `evaluate_sequence` function and returned.

All helper utilities are top‑level, there are no closures or lambdas,
and the implementation stays well within the 1000 s time limit while
trying to beat the current best score (~1.506).
"""

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast version of the official objective for a normalised vector
    (∑a = 1):   2·n·max(conv(a,a)).
    """
    # linear convolution via FFT (only the first 2·n‑1 entries are correct)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Main optimiser
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function runs for at most `budget_s` seconds and returns the
    best list of floats discovered.
    """
    rng = np.random.default_rng(seed)

    # problem size -------------------------------------------------
    n = 4000                     # length of the step function
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0 – simple initial candidates (uniform + optional previous best)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # uniform start
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # reuse a previous construction if present
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            s = a1.sum()
            if s > 0:
                a1 /= s
                sc1 = fast_score(a1, fft_len, n)
                if sc1 < best_score:
                    best_a, best_score = a1, sc1
                    print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1 – cheap random sparse masks (equal mass on a random subset)
    # --------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.2      # small safety margin

    for trial in range(2000):
        if time.time() >= deadline:
            break
        # random support size between 50 and 200
        k = rng.integers(50, 201)
        pos = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[pos] = 1.0 / k
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[rand_sparse] k={k} → new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2 – long simulated annealing
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score

    total_anneal = deadline - time.time()
    if total_anneal <= 0:
        total_anneal = 1.0

    # initial temperature – a noticeable fraction of the current score
    T0 = max(0.08 * best_score, 1e-6)
    t_start = time.time()
    it = 0

    while time.time() < deadline:
        # square‑root cooling (slower than linear)
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_anneal)

        # pick source (i) and destination (j)
        i = rng.integers(n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30 %) from i to j
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        # Metropolis acceptance
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] it={it} new best fast_score={best_score:.6f}")

        # occasional global jitter to hop out of plateaus
        if rng.random() < 0.001:
            a = best_a * rng.normal(1.0, 0.02, size=n)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a /= s
                cur_score = fast_score(a, fft_len, n)

        it += 1

    # --------------------------------------------------------------
    # 3 – final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
