def simplex_projection(v):
    # v is np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # find last true index
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def proj_subgradient_optimize(a_start, deadline, n, fft_len, init_scale=0.5):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient of max (subgradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]
        # normalize gradient
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break
        step = init_scale / np.sqrt(t + 1.0)
        a_new = a - step * g / norm_g
        # project onto simplex
        a_new = simplex_projection(a_new)
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        a = a_new
        t += 1
    return best, best_score

def mixture_search(candidates, scores, n, fft_len, deadline):
    best_a = candidates[0]
    best_score = scores[0]
    K = len(candidates)
    phi = (1 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1) / 2.0
    for i in range(K):
        for j in range(i+1, K):
            a = candidates[i]
            b = candidates[j]
            a0 = 0.0
            b0 = 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b, fft_len, n)
            fd, _ = eval_mix(d, a, b, fft_len, n)
            while time.time() < deadline and (b0 - a0) > 1e-5:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b, fft_len, n)
            tm = 0.5 * (a0 + b0)
            fmid, vec = eval_mix(tm, a, b, fft_len, n)
            if fmid < best_score:
                best_score = fmid
                best_a = vec.copy()
    return best_a, best_score

def triangular_init(n):
    # symmetrical triangle shape peaked at centre
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(float)
    a = a / a.sum()
    return a

"""
search for a non‑negative step‐function vector that minimizes

    2·n·max(a*a) / (sum a)**2

using a combination of
* diverse initialisers,
* many Adam restarts on a log‑sum‑exp smoothed max,
* exponentiated‑gradient descent on the true max,
* projected sub‑gradient descent on the true max,
* a deterministic hill‑climb,
* pairwise convex‑combination (golden‑section) search.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0).
The routine runs until the supplied time budget expires (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the bound assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑normalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start vector derived from a flat spectrum (square‑rooted)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def triangular_init(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=float)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=float)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    a = a / a.sum()
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0.0
    rho_arr = np.where(cond)[0]
    if rho_arr.size == 0:
        # fallback to uniform distribution
        return np.full(n, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the auto‑convolution via log‑sum‑exp.
    Returns (objective value, gradient wrt y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Multiplicative (exponentiated) sub‑gradient descent on the true max.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Projected sub‑gradient descent on the (hard) max‑convolution
# ----------------------------------------------------------------------
def proj_subgradient_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len_conv: int,
                              init_scale: float = 0.5):
    """
    Euclidean projected sub‑gradient descent on the true max,
    normalising onto the simplex after each step.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    t = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        step = init_scale / np.sqrt(t + 1.0)
        a_new = a - step * g / norm_g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        t += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb with geometric step reduction
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate contributing most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)

    return a, score

# Convex‑combination exploration among top candidates
# ----------------------------------------------------------------------
def eval_mix(t: float, a: np.ndarray, b: np.ndarray,
             fft_len: int, n: int) -> tuple[float, np.ndarray]:
    """
    Linear mixture (convex combination) of two probability vectors.
    Returns (score, normalised mixture).
    """
    mix = (1.0 - t) * a + t * b
    mix = mix / mix.sum()
    return fast_score(mix, fft_len, n), mix


def first_elem(tup):
    return tup[0]


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates perform a cheap golden‑section line search
    over t ∈ [0,1] to minimise the bound.
    Returns the best vector and its score found during this phase.
    """
    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b, fft_len, n)
            fd, _ = eval_mix(d, a, b, fft_len, n)

            while time.time() < deadline and (b0 - a0) > 1e-5:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b, fft_len, n)

            tm = 0.5 * (a0 + b0)
            fmid, vec = eval_mix(tm, a, b, fft_len, n)
            if fmid < best_score:
                best_score = fmid
                best_a = vec.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – keep the historic length (4000) for compatibility.
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # Initialise from previous global result if present.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # Diverse initial pool
    cand0 = a_start.copy()
    cand1 = spectral_init(n)
    cand2 = random_initial(n, rng)
    cand3 = add_noise(cand0, rng.uniform(0.001, 0.01), rng)
    cand4 = triangular_init(n)
    candidates = [cand0, cand1, cand2, cand3, cand4]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many Adam restarts (≈60 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 200  # more refinements per lam
    restart_idx = 0
    while time.time() < deadline - 200:  # keep ~200 s for later phases
        if restart_idx == 0:
            a0 = best_a.copy()
        else:
            sigma = rng.uniform(0.001, 0.015)
            a0 = add_noise(best_a, sigma, rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len,
            n,
        )
        print(f"[Adam restart {restart_idx + 1}] score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   -> new global best after Adam: {best_score:.8f}")
        restart_idx += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   -> new global best after exp‑gradient: {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (fine‑grained)
    # ------------------------------------------------------------------
    if time.time() < deadline - 40:
        a_opt, sc_opt = proj_subgradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            init_scale=0.6,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   -> new global best after proj‑subgrad: {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.20,
            min_step=1e-7,
        )
        print(f"[Hill‑climb] score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   -> new global best after hill‑climb: {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – convex‑combination (mixture) search among top candidates
    # ------------------------------------------------------------------
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    paired = list(zip(top_scores, top_vectors))
    paired.sort(key=first_elem)
    K = 5
    selected = paired[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline - 5:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture] score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   -> new global best after mixture: {best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.8f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
