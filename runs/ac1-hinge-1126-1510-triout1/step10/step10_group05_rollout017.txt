"""
Optimise a non‑negative step function to minimise
    2·n·max(conv(a,a)) / (∑a)^2

The algorithm proceeds in stages:
1.  Initialise from a known solution or from a family of parametric
    shapes (exponential, power‑law, cosine‑bell).  The best among them is
    kept.
2.  A cascade of L‑BFGS optimisations on a log‑sum‑exp smoothed
    approximation of the max‑convolution, using a decreasing smoothing
    schedule.
3.  Several short bursts of exponentiated‑gradient descent on the true
    max‑convolution (sub‑gradient of the hard max).
4.  Projected sub‑gradient descent with a line‑search for the exact
    max‑convolution.
5.  A directed simulated‑annealing / hill‑climbing phase that repeatedly
    moves a small amount of mass from the index that contributes most to
    the current maximal convolution entry to a random destination.
All phases respect the probability simplex, so the final score can be
computed with the fast FFT‑based routine `fast_score`.  The best sequence
found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  Variables are logits y; a = softmax(y) lies on
    the probability simplex.
    Returns (objective, gradient) w.r.t. y.
    """
    # soft‑max
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # linear convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # gradient wrt a before softmax

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_burst(a_start: np.ndarray,
                      iters: int,
                      eta0: float,
                      fft_len: int,
                      n: int,
                      rng: np.random.Generator):
    """
    Perform `iters` steps of exponentiated‑gradient descent on the hard max.
    Returns the best a encountered and its score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    for t in range(1, iters + 1):
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max entry (sub‑gradient)
        fm = np.fft.fft(mask, fft_len)
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(t)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s == 0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with line‑search (hard max)
# ----------------------------------------------------------------------


def subgradient_line_search(a_start: np.ndarray,
                            max_iters: int,
                            deadline: float,
                            fft_len: int,
                            n: int):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses a few geometric step‑sizes each iteration and keeps the best
    sequence seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    alphas = np.geomspace(1.0, 1e-4, num=7)  # decreasing candidate step factors
    while time.time() < deadline and it < max_iters:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        if idx.size:
            g[idx] = a[k - idx]   # one‑sided derivative (the factor 2 is absorbed
                                  # into the step size)

        improved = False
        for alpha in alphas:
            a_candidate = a - alpha * g
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s <= 0.0:
                continue
            a_candidate /= s
            score = fast_score(a_candidate, fft_len)
            if score < best_score:
                best_score = score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
        if not improved:
            # fallback: very small step
            a = a - 0.05 * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Directed simulated annealing / hill‑climb
# ----------------------------------------------------------------------


def directed_hill_climb(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        fft_len: int,
                        n: int,
                        rng: np.random.Generator):
    """
    Greedy hill‑climb that always moves a small fraction of mass from the
    index that contributes most to the current maximal convolution entry
    to a random destination.  Worsening moves are accepted according to a
    simple linear cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    curr_score = start_score

    total_time = deadline - time.time()
    start_time_local = time.time()

    while time.time() < deadline:
        # recompute convolution (FFT is cheap for n≈4000)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # indices that actually appear in this convolution term
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution of each pair (i, k-i)
        contrib = a[idx] * a[k - idx]
        src = idx[np.argmax(contrib)]

        # pick a destination that is not the source (uniformly at random)
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # move a modest fraction of the source mass
        delta = a[src] * rng.random() * 0.08  # up to 8 % of src
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # total mass unchanged, positivity preserved by construction

        # evaluate new score
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max()

        # linear cooling: temperature proportional to remaining time
        remaining = max(0.0, deadline - time.time())
        T = max(1e-7, remaining / total_time) * (best_score - 1.0 + 1e-6)

        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - curr_score) / T):
                accept = True

        if accept:
            a = a_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

Optimise a non‑negative step function to minimise

    2·n·max(conv(a,a)) / (∑a)^2

The algorithm runs several stages within the 1000 s budget:

1. Initialise from a known solution (if any) or from a set of handcrafted
   families (exponential, power‑law, cosine‑bell, linear).  The best among

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_burst(a_start: np.ndarray,
                      iters: int,
                      eta0: float,
                      fft_len: int,
                      n: int,
                      rng: np.random.Generator):
    """
    Perform `iters` steps of exponentiated‑gradient descent on the hard max.
    Returns the best a encountered and its score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    for t in range(1, iters + 1):
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        fm = np.fft.fft(mask, fft_len)
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(t)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s == 0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with line‑search (hard max)
# ----------------------------------------------------------------------


def subgradient_line_search(a_start: np.ndarray,
                            max_iters: int,
                            deadline: float,
                            fft_len: int,
                            n: int):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses a handful of geometric step‑sizes each iteration.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    alphas = np.geomspace(1.0, 1e-4, num=7)
    it = 0
    while time.time() < deadline and it < max_iters:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        if idx.size:
            g[idx] = a[k - idx]

        improved = False
        for alpha in alphas:
            a_cand = a - alpha * g
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s <= 0.0:
                continue
            a_cand /= s
            score = fast_score(a_cand, fft_len)
            if score < best_score:
                best_score = score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            a = a - 0.05 * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Directed simulated annealing / hill‑climb
# ----------------------------------------------------------------------


def directed_hill_climb(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        fft_len: int,
                        n: int,
                        rng: np.random.Generator):
    """
    Greedy hill‑climb moving a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to a random
    destination, with a linear cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    curr_score = start_score

    total_time = deadline - time.time()
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx] * a[k - idx]
        src = idx[np.argmax(contrib)]

        dst = rng.integers(0, n)
        if dst == src:
            continue
        delta = a[src] * rng.random() * 0.08  # up to 8 % of src mass
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max()

        remaining = max(0.0, deadline - time.time())
        T = max(1e-7, remaining / total_time) * (best_score - 1.0 + 1e-6)

        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - curr_score) / T):
                accept = True

        if accept:
            a = a_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if possible, otherwise uniform
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0.0, 1.0, old.size)
                x_new = np.linspace(0.0, 1.0, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
        else:
            a0 = np.full(n, 1.0 / n)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] best_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # Phase 0 – sweep over a few handcrafted families
    # --------------------------------------------------------------
    # exponential decay
    for beta in np.linspace(0.0003, 0.015, 120):
        a = np.exp(-beta * np.arange(n, dtype=float))
        a /= a.sum()
        s = fast_score(a, fft_len)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"[family‑exp] new best {best_score:.6f}")

    # power‑law
    for p in np.linspace(0.3, 2.2, 120):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        s = fast_score(a, fft_len)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"[family‑pow] new best {best_score:.6f}")

    # cosine bell (smooth, symmetric)
    a = 0.5 * (1.0 + np.cos(np.pi * np.arange(n, dtype=float) / n))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    s = fast_score(a, fft_len)
    if s < best_score:
        best_score, best_a = s, a.copy()
        print(f"[family‑cos] new best {best_score:.6f}")

    # linear decreasing (triangle)
    a = np.linspace(1.0, 0.0, n, endpoint=False)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    s = fast_score(a, fft_len)
    if s < best_score:
        best_score, best_a = s, a.copy()
        print(f"[family‑lin] new best {best_score:.6f}")

    # a handful of random starts
    for _ in range(12):
        a = random_initial(n, rng)
        s = fast_score(a, fft_len)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"[rand‑init] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade on a smoothed max
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005, 0.00002]
    maxiter_per_lam = 1500
    phase1_deadline = start_time + 0.30 * budget_s
    best_a, best_score = lbfgs_continuation(
        best_a, lam_schedule, maxiter_per_lam,
        phase1_deadline, fft_len, n,
    )
    print(f"[L‑BFGS] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – bursts of exponentiated‑gradient descent
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    while time.time() < phase2_deadline:
        iters = 1200
        eta0 = 10.0 * (0.5 + rng.random())
        a_cand, s_cand = exp_gradient_burst(
            best_a, iters, eta0, fft_len, n, rng,
        )
        if s_cand < best_score:
            best_score, best_a = s_cand, a_cand.copy()
            print(f"[exp‑grad‑burst] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with line‑search
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.78 * budget_s
    a_cand, s_cand = subgradient_line_search(
        best_a, max_iters=6000,
        deadline=phase3_deadline,
        fft_len=fft_len,
        n=n,
    )
    if s_cand < best_score:
        best_score, best_a = s_cand, a_cand.copy()
        print(f"[sub‑grad‑line] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – directed hill‑climb / simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_cand, s_cand = directed_hill_climb(
            best_a, best_score,
            deadline,
            fft_len, n, rng,
        )
        if s_cand < best_score:
            best_score, best_a = s_cand, a_cand.copy()
            print(f"[anneal‑direct] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
