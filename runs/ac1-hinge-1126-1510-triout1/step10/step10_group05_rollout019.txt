"""
Propose a step‐function that beats the 1.503 upper bound.

The algorithm works in three stages:

1. **Initialisation** – start from the best known construction (if it is
   stored in the global variable `height_sequence_1`) or from a uniform
   distribution.  The sequence is interpolated to a manageable size
   (`n≈2500`) and normalised to a probability vector.

2. **Linear‑programming refinement** – given the current sequence `a`,
   compute `M = max(conv(a,a))`.  Solve the LP  

        maximise   Σ g_i
        subject to conv(a,g) ≤ M   (component‑wise)
                  g_i ≥ 0

   The LP is a linear relaxation of the “alternating‑projection’’ scheme
   described in the statement.  The optimiser (`scipy.optimize.linprog`,
   HiGHS) returns a direction `g` that can carry more mass without increasing
   the current worst convolution value.  After rescaling `g` to the same
   ℓ₁‑norm as `a`, we mix the two vectors (`t≈0.5`).  If the mixed vector
   yields a strictly smaller `max(conv)`, we accept it and repeat the LP
   step a few times.

3. **Exponential‑gradient polish** – a fast sub‑gradient loop on the true
   max‑convolution (the “exp‑grad’’ phase of the reference code) finishes
   the optimisation.

All work respects the 1000 s time budget; the heavy LP stage runs only a
few iterations, after which cheap gradient iterations are used to squeeze
out the last improvement.

The returned list is ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a probability vector a (no normalisation)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random simplex point – used only as a fall‑back."""
    a = rng.random(n) + 1e-6
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 2500 entries give a good trade‑off between quality
    #                and the size of the LP (≈8 Mi elements).
    # ------------------------------------------------------------------
    n = 2500
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the previously best construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:

            x_new = np.linspace(0.0, 1.0, n)

            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    best_score = fast_score(a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    max_lp_iter = 5

    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
