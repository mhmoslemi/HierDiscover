"""
Step‑function minimiser.

The code searches for a non‑negative sequence a = (a_0,…,a_{n‑1}) that
minimises   E(a) = 2·n·max(conv(a,a)) / (sum a)^2.
Because the denominator cancels when we keep the sequence on the
probability simplex, the objective reduces to 2·n·max(conv(a,a)).

Algorithm
---------
1. Cheap parametric sweep (exponential, linear, power‑law, mixtures,
   and Beta‑family shapes) to obtain a good seed.
2. Smooth exponentiated‑gradient descent:
   * a soft‑max weight w_k ∝ exp(β·conv_k) approximates the true max.
   * The sub‑gradient of  Σ_k w_k·conv_k is 2·(w * a) and is obtained
     efficiently with two FFTs.
   * Multiplicative updates keep a non‑negative; Euclidean projection
     onto the simplex restores the normalisation.
   * β is increased gradually so that the optimisation first sees a
     smooth landscape and later focuses on the true maximum.
3. Simulated‑annealing fine‑tuning – random mass transfers between two
   indices with a Metropolis rule.  The temperature schedule is
   exponential, allowing larger moves early on.

All phases respect the total time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
so that a subsequent run can start from it.

The implementation works for any n; we use n≈5000 which gives a good
trade‑off between quality and speed on a typical 2‑core machine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = v.astype(np.float64, copy=True)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force over a few low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # ----- exponential ---------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=200):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ----- linear decreasing ---------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ----- mixture exponential + uniform ---------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=50):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # ----- power‑law decay ------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=150):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ----- Beta‑family (a_i ∝ x^{α‑1} (1‑x)^{β‑1}) --------------------
    # α and β are sampled on a coarse grid; values <0.5 are avoided to
    # keep the shape well behaved at the edges.
    xs = np.linspace(0.0, 1.0, n)
    for alpha in np.linspace(0.5, 4.0, num=12):
        for beta in np.linspace(0.5, 4.0, num=12):
            # small epsilon avoids 0**neg.
            eps = 1e-12
            a = (np.maximum(xs, eps) ** (alpha - 1)) * (np.maximum(1.0 - xs, eps) ** (beta - 1))
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smooth exponentiated‑gradient descent
# ----------------------------------------------------------------------


def softmax_exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    beta_inc: float = 5.0,
):
    """
    Multiplicative (exponentiated) SGD where the sub‑gradient is computed
    from a soft‑max weighting of the convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 1
    beta = beta0

    while time.time() < deadline:
        # ----- convolution of a with itself ---------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max mask (smooth approximation of the true max) -------
        # shift by max_conv to avoid overflow; the mask sums to 1.
        exp_vals = np.exp(beta * (conv - max_conv))
        mask = exp_vals / exp_vals.sum()

        # ----- gradient of Σ_k mask_k·conv_k -------------------------------
        a_rev = a[::-1]
        fft_len2 = next_pow2((2 * n - 1) + n - 1)  # length for mask * a_rev
        fm = np.fft.rfft(mask, fft_len2)
        fa_rev = np.fft.rfft(a_rev, fft_len2)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len2)[: (2 * n - 1) + n - 1]
        # central slice gives the gradient w.r.t. a
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- multiplicative update (exponentiated gradient) ---------------
        eta = eta0 / np.sqrt(it)
        exponents = -eta * grad
        # clip to avoid overflow/underflow
        exponents = np.clip(exponents, -50, 50)
        a = a * np.exp(exponents)

        # ----- projection back onto the probability simplex -----------------
        a = simplex_projection(a)

        # ----- keep the best ------------------------------------------------
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # ----- schedule for β (smoothness) ---------------------------------
        if it % 200 == 0:
            beta = min(beta + beta_inc, beta_max)

        it += 1

    return best_a, best_score

# Phase 2 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers with Metropolis acceptance; linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # initial temperature: a modest fraction of the current score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.30  # allow a larger move than the original code
    recompute_every = 5000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # always move from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Incremental update of the convolution O(n)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

    Returns the best list of floats found before the deadline.

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 5000 gives a fine trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if present, otherwise uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate the old sequence onto the new grid.
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split (5 % / 60 % / 35 %)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.65 * budget_s
    # remaining time will be used for simulated annealing

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = softmax_exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng,
            eta0=6.0, beta0=5.0, beta_max=200.0, beta_inc=5.0
        )
        print(f"[smooth‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for a possible next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
