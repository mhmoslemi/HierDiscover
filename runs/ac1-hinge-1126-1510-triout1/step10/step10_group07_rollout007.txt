def mixture_search(candidates, scores, n, fft_len, deadline):
    # candidates: list of numpy arrays
    # scores: list of scores (float)
    best_idx = np.argmin(scores)
    best_a = candidates[best_idx].copy()
    best_score = scores[best_idx]
    K = len(candidates)
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2
    for i in range(K):
        for j in range(i+1, K):
            a = candidates[i]
            b = candidates[j]
            a0 = 0.0
            b0 = 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
            fd = fast_score((1.0 - d) * a + d * b, fft_len, n)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0 = d
                    fd = fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
                else:
                    a0 = c
                    fc = fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd = fast_score((1.0 - d) * a + d * b, fft_len, n)
            tm = 0.5 * (a0 + b0)
            fs = fast_score((1.0 - tm) * a + tm * b, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = ((1.0 - tm) * a + tm * b).copy()
    return best_a, best_score

Propose a low‑score non‑negative step‑function sequence.

Algorithm outline
-----------------
1. **Build a diverse initial pool** – uniform, spectral, random,
   exponential and power‑law parametrisations, and the previous global best
   (if any).  All vectors are normalised to sum = 1 and clipped to [0, 1000].

2. **Select the most promising few** (by a cheap FFT based bound) to be
   refined later.

3. **Smooth‑max Adam optimisation** – minimise a log‑sum‑exp surrogate of
   the maximal self‑convolution using a log‑softmax parametrisation.
   Several restarts with added noise explore the landscape.

4. **Exponentiated‑gradient descent** on the true (hard) max‑convolution.
   This mirrors the behaviour of a sub‑gradient method while keeping the
   simplex constraint.

5. **Simulated annealing** – random Gaussian perturbations with a
   temperature schedule allow escaping shallow local minima.

6. **Deterministic hill‑climb** – a simple move that transfers a small
   fraction of mass from the “worst” coordinate to the “best” one.

7. **Convex‑combination (mixture) search** among the top‑K vectors found
   so far, using a cheap golden‑section line search.

8. **Final evaluation** with the provided `evaluate_sequence` routine.
   The best vector is stored globally as ``height_sequence_1`` for reuse.

All heavy work is performed with FFT‑based convolutions; the total
running time is kept well below the 1000 s budget.

import time

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the bound 2·n·max(a*a) assuming Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def _score_key(item):
    """Key function for sorting (item = (score, vector))."""
    return item[0]

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def exponential_init(rate: float, n: int) -> np.ndarray:
    """Exponential decay shape a_i ∝ exp(-rate·i)."""
    a = np.exp(-rate * np.arange(n, dtype=float))
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def powerlaw_init(p: float, n: int) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the true objective
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """
    Simple SA: Gaussian perturbations + Metropolis acceptance.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    T0 = 0.02          # initial temperature
    sigma0 = 0.03      # initial perturbation scale
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(1e-9, T0 * (1.0 - frac))
        sigma = max(1e-9, sigma0 * (1.0 - frac))

        proposal = a + rng.normal(0.0, sigma, size=n)
        proposal = np.maximum(proposal, 0.0)
        s = proposal.sum()
        if s <= 0.0:
            continue
        proposal /= s

        score = fast_score(proposal, fft_len, n)

        if score < cur_score:
            a = proposal
            cur_score = score
            if score < best_score:
                best_score = score
                best_a = proposal.copy()
        else:
            delta = score - cur_score
            accept_prob = np.exp(-delta / T)
            if rng.random() < accept_prob:
                a = proposal
                cur_score = score

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb with geometric step reduction
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that most contributes
    to the maximal convolution entry toward a coordinate with the smallest partner.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] where defined
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates perform a cheap golden‑section line search
    over the mixing weight t ∈ [0,1] to minimise the bound.
    Returns (best_a, best_score).
    """
    # initialise with the best known candidate
    best_idx = int(np.argmin(scores))
    best_a = candidates[best_idx].copy()
    best_score = scores[best_idx]

    K = len(candidates)
    phi = (1 + np.sqrt(5)) / 2.0
    invphi = (np.sqrt(5) - 1) / 2.0

    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, fft_len, n)

            tm = 0.5 * (a0 + b0)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            fs = fast_score(mix_tm, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = mix_tm.copy()

    return best_a, best_score

# Main entry point required by the task
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)          # for convolution
    fft_len_g = next_pow2(3 * n - 2)        # for gradient steps

    # --------------------------------------------------------------
    # Initialise from previous global result if available
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Assemble a large pool of diverse candidates
    # --------------------------------------------------------------
    candidates = []
    scores = []

    # uniform
    a_uni = np.full(n, 1.0 / n)
    candidates.append(a_uni)
    scores.append(fast_score(a_uni, fft_len, n))

    # spectral
    a_spec = spectral_init(n)
    candidates.append(a_spec)
    scores.append(fast_score(a_spec, fft_len, n))

    # random
    for _ in range(2):
        a_rand = random_initial(n, rng)
        candidates.append(a_rand)
        scores.append(fast_score(a_rand, fft_len, n))

    # previous best
    candidates.append(a_start)
    scores.append(fast_score(a_start, fft_len, n))

    # exponential family
    for rate in np.logspace(-4, -1, 20):
        a_exp = exponential_init(rate, n)
        candidates.append(a_exp)
        scores.append(fast_score(a_exp, fft_len, n))

    # power‑law family
    for p in np.linspace(0.2, 4.0, 20):
        a_pow = powerlaw_init(p, n)
        candidates.append(a_pow)
        scores.append(fast_score(a_pow, fft_len, n))

    # --------------------------------------------------------------
    # Keep the best K vectors as the elite pool
    # --------------------------------------------------------------
    K = 5
    ranked = sorted(zip(scores, candidates), key=_score_key)
    elite_scores, elite_vectors = zip(*ranked[:K])
    elite_scores = list(elite_scores)
    elite_vectors = list(elite_vectors)

    best_a = elite_vectors[0].copy()
    best_score = elite_scores[0]

    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule_fine = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 150
    restarts = 8

    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule_fine,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len_g,
            n,
        )
        print(f"[Adam‑restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            elite_vectors.append(best_a.copy())
            elite_scores.append(best_score)
            # keep only top K
            ranked = sorted(zip(elite_scores, elite_vectors), key=_score_key)
            elite_scores, elite_vectors = zip(*ranked[:K])
            elite_scores = list(elite_scores)
            elite_vectors = list(elite_vectors)
            print(f"   ==> new global best after Adam: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            elite_vectors.append(best_a.copy())
            elite_scores.append(best_score)
            ranked = sorted(zip(elite_scores, elite_vectors), key=_score_key)
            elite_scores, elite_vectors = zip(*ranked[:K])
            elite_scores = list(elite_scores)
            elite_vectors = list(elite_vectors)
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing for fine stochastic search
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[Sim‑Anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            elite_vectors.append(best_a.copy())
            elite_scores.append(best_score)
            ranked = sorted(zip(elite_scores, elite_vectors), key=_score_key)
            elite_scores, elite_vectors = zip(*ranked[:K])
            elite_scores = list(elite_scores)
            elite_vectors = list(elite_vectors)
            print(f"   ==> new global best after SA: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.15,
            min_step=1e-6,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            elite_vectors.append(best_a.copy())
            elite_scores.append(best_score)
            ranked = sorted(zip(elite_scores, elite_vectors), key=_score_key)
            elite_scores, elite_vectors = zip(*ranked[:K])
            elite_scores = list(elite_scores)
            elite_vectors = list(elite_vectors)
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Convex‑combination search among elites
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            elite_vectors,
            elite_scores,
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
