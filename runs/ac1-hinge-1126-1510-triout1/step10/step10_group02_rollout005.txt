        if s <= 0.0:
            a = best_a.copy()
        else:

        a = symmetrize(a)

            a = symmetrize(a)

        if score < best_score:

            best_a = a.copy()

"""
step‑function optimiser – 1000 s budget (symmetrised, Adam‑EG)

The search works on a probability vector (∑ a = 1) on the simplex.
All candidates are forced to be symmetric (a[i] = a[n‑1‑i]); this
reduces the search space and tends to lower the maximal auto‑convolution.
The pipeline consists of:

1. A diverse pool of symmetric initial vectors (random, edge‑heavy,
   power‑law, and classic windows).
2. A cascade of L‑BFGS optimisations on a smooth log‑sum‑exp
   approximation of the max‑convolution, with a decreasing smoothing
   schedule.
3. A long exponentiated‑gradient (mirrored‑SGD) descent using an
   Adam‑style adaptive step‑size; occasional isotropic noise helps to
   escape plateaus.  After each update the vector is projected back onto
   the simplex and symmetrised.
4. A simple simulated‑annealing hill‑climb that respects the symmetry
   by symmetrising every proposal.

The best vector found is finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic helpers
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast objective: 2·n·max(conv(a,a))  (a is already normalised to sum=1).
    The convolution is evaluated via FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] == a[-1-i] (mirror symmetry)."""
    return (a + a[::-1]) * 0.5


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector (unsymmetrised)."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# --------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# --------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Optimisation is performed in log‑space (y) with a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        a = symmetrize(a)                 # enforce symmetry after each stage
        a = a / a.sum()                    # re‑normalise (symmetrising preserves sum)
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize_adam(a_start: np.ndarray,

                               n: int,

                               rng: np.random.Generator,
                               eta0: float = 12.0,

                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8,
                               noise_every: int = 40_000,
                               noise_sigma: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Mirrored SGD (exponentiated‑gradient) on the exact L∞‑objective,
    enhanced with Adam‑style moment estimates and occasional isotropic noise.
    The vector is kept on the probability simplex and symmetrised after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        # compute convolution and active mask for the current max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size schedule (decays like 1/sqrt(t))
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (clipped exponent to avoid overflow)
        direction = m_hat / (np.sqrt(v_hat) + eps)
        exp_term = np.exp(np.clip(-eta * direction, -50, 50))
        a = a * exp_term

        # normalise
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # enforce symmetry
        a = symmetrize(a)

        # occasional noise injection to escape flat regions
        if (t % noise_every) == 0:
            a = add_noise(a, sigma=noise_sigma, rng=rng)
            a = symmetrize(a)

        # evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but effective)
# --------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass transfers accepted with a Metropolis rule.
    The temperature is linearly cooled.  All proposals are symmetrised
    to keep the search on the symmetric sub‑space.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    # initial temperature: a few percent of the current score
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = symmetrize(cand)            # keep symmetry
        cand = cand / cand.sum()           # re‑normalise (symmetrising preserves sum, but numeric drift possible)

        new_score = fast_score(cand, next_pow2(2 * n - 1))

        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

# Main entry point
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4000                         # keep a size that allows many iterations
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse symmetric starting vectors
    # --------------------------------------------------------------
    pool = []

    # 1) Previous construction (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            a0 = symmetrize(a0)
            pool.append(a0)

    # 2) Random symmetric families
    for _ in range(10):
        a = random_initial(n, rng)
        a = symmetrize(a)
        pool.append(a)

    # 3) Classic window functions (if SciPy is available)
    try:
        from scipy.signal import get_window
        window_specs = [
            ('hann', None),
            ('hamming', None),
            ('blackman', None),
            ('kaiser', [4, 8, 12]),
            ('chebwin', [30, 50, 70]),
            ('gaussian', [0.4]),  # sigma parameter
        ]
        for name, params in window_specs:
            if params is None:
                w = get_window(name, n, sym=True)
                w = np.maximum(w, 0.0)
                w /= w.sum()
                w = symmetrize(w)
                pool.append(w)
            else:
                for p in params:
                    if name == 'kaiser':
                        w = get_window((name, p), n, sym=True)
                    elif name == 'chebwin':
                        w = get_window((name, p), n, sym=True)
                    elif name == 'gaussian':
                        w = get_window((name, p), n, sym=True)
                    else:
                        continue
                    w = np.maximum(w, 0.0)
                    w /= w.sum()
                    w = symmetrize(w)
                    pool.append(w)
    except Exception:
        pass  # optional – continue without windows

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS phase
    phase2_end = start_time + 0.70 * budget_s   # EG‑Adam phase
    # remainder → simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 600

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.30:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.012), rng=rng)
            a_start = symmetrize(a_start)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:                     # fallback – uniform vector
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent with Adam momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize_adam(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=18.0,
            max_iters=8_000_000,
        )
        print(f"[exp‑grad‑adam] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (simple but powerful)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
