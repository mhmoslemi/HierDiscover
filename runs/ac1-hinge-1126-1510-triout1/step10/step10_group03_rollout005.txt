def smoothed_gradient_descent(a_start, deadline, n, fft_len, rng, lr=0.05, alpha_start=1.0, alpha_end=200.0, max_iter=None):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_val = conv.max()
        # compute softmax weights
        # gradually increase alpha
        # compute progress fraction
        time_frac = (time.time() - start_time) / (deadline - start_time) # we need to capture start_time at function start.
        alpha = alpha_start + (alpha_end - alpha_start) * time_frac
        # compute weights using exponent
        # subtract max for numerical stability
        shifted = conv - max_val  # values <= 0
        w_unnorm = np.exp(alpha * shifted)
        w_sum = w_unnorm.sum()
        if w_sum == 0:
            # fallback to mask at max location
            mask = (conv == max_val).astype(np.float64)
            w = mask / mask.sum()
        else:
            w = w_unnorm / w_sum

        # gradient: conv_rev = irfft(fft(a_rev) * fft(w))
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        conv_rev_full = np.fft.irfft(fa_rev * fw, fft_len)[:2 * n - 1]
        grad = 4.0 * n * conv_rev_full[n - 1 : n - 1 + n]  # shape n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project onto simplex
        a = simplex_projection(a)

        # evaluate fast_score
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

1. **Exploratory families** – test many 1‑D parametric families
   (exponential, linear, power‑law, mixtures, Beta, binomial, Gaussian,
   Hann).  Each candidate is normalised to Σa=1 and scored by the fast
   FFT‑based bound `fast_score`.  The best candidate seeds the continuous
   optimisation.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   true objective

        F(a) = 2·n·log‑sum‑exp(α·conv(a,a))/α ,

   where `α` is increased from 1 to a large value (≈200) as the optimisation
   progresses.  The gradient of `F` is obtained via FFT (cost O(n log n)).
   An Adam optimiser performs additive updates, after which the vector is
   projected onto the probability simplex.

3. **Fine‑grained simulated annealing** – random mass transfers between two
   indices are accepted with a Metropolis rule.  Incremental convolution
   updates keep the routine fast while further shaving off the maximum
   convolution value.

All phases respect the total 1000 s budget.  The best sequence found is
returned, stored in the global `height_sequence_1` for possible warm‑starts,
and finally evaluated with the official `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families (including Beta & Binomial)
# ----------------------------------------------------------------------


def param_scan_extended(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Brute‑force over many low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # 1. Pure exponential
    # --------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 2. Linear decreasing
    # --------------------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=250):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 3. Mixture exponential + uniform
    # --------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=80):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=13):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --------------------------------------------------------------
    # 4. Power‑law decay
    # --------------------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=180):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 5. Beta family (symmetric around 0.5)
    # --------------------------------------------------------------
    xs = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    for alpha in np.logspace(np.log10(0.3), np.log10(5.0), num=30):
        for beta in np.logspace(np.log10(0.3), np.log10(5.0), num=30):
            a = (np.maximum(xs, eps) ** (alpha - 1)) * (
                np.maximum(1.0 - xs, eps) ** (beta - 1)
            )
            if a.sum() == 0.0:
                continue
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --------------------------------------------------------------
    # 6. Binomial family (parameter p)
    # --------------------------------------------------------------
    from math import comb

    fact = np.arange(n)  # placeholder to avoid recomputing inside loop
    for p in np.linspace(0.05, 0.95, num=30):
        # use log‑space to avoid overflow
        log_coeffs = np.log([comb(n - 1, k) for k in range(n)])
        a = np.exp(log_coeffs + k * np.log(p) + (n - 1 - k) * np.log(1 - p))
        # numeric issues for extreme p – guard
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 7. Gaussian windows (σ varies)
    # --------------------------------------------------------------
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, num=30):
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 8. Hann window (raised cosine)
    # --------------------------------------------------------------
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    a /= a.sum()
    score = fast_score(a, fft_len)
    if score < best_score:
        best_score, best_a = score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smoothed Adam gradient descent on F(a) = 2·n·log‑sum‑exp
# ----------------------------------------------------------------------


def smoothed_adam_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lr: float = 0.04,
    alpha_start: float = 1.0,
    alpha_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smoothed version of the true objective.
    The smoothing parameter α is increased linearly with time.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    start_time = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # FFT of a (used for both conv and gradient)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # progress fraction for α schedule
        now = time.time()
        prog = (now - start_time) / (deadline - start_time)
        alpha = alpha_start + prog * (alpha_end - alpha_start)

        # softmax weights (log‑sum‑exp trick)
        shifted = conv - max_val  # all ≤ 0
        w_unnorm = np.exp(alpha * shifted)
        w_sum = w_unnorm.sum()
        if w_sum == 0.0 or np.isinf(w_sum):
            # fallback to hard mask
            mask = (conv == max_val).astype(np.float64)
            w = mask / mask.sum()
        else:
            w = w_unnorm / w_sum

        # gradient of the smoothed objective:
        #   grad = 4·n·(a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        conv_rev_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 4.0 * n * conv_rev_full[n - 1 : n - 1 + n]

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # Project onto the simplex (guarantees non‑negativity & Σ=1)
        a = simplex_projection(a)

        # Evaluate current score (fast bound)
        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated annealing (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass transfers with Metropolis acceptance and linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.18   # maximum proportion of source mass moved per proposal
    recompute_every = 5000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Incremental convolution update (O(n))
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality & speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Warm‑start from a previous run if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # resize by linear interpolation
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Split the overall budget (≈5 % / 75 % / 20 %)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.78 * budget_s
    # remaining time → simulated annealing

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (extended families)
    # --------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan_extended(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed Adam gradient descent
    # --------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = smoothed_adam_descent(
            best_a, phase1_end, n, fft_len, rng,
            lr=0.04, alpha_start=1.0, alpha_end=200.0
        )
        print(f"[adam‑gd] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
