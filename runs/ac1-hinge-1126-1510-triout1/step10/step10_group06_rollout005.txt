Initialize a = a0 copy

best_a = a0, best_score = fast_score

  max_val = conv.max()

  mask = (conv >= max_val - eps).astype(float)

  beta1=0.9; beta2=0.999; epsilon=1e-8

  m = beta1 * m + (1-beta1) * grad

  v = beta2 * v + (1-beta2) * (grad**2)

  sc = fast_score(a_new, fft_len_conv, n)

  if sc < best_score:

    best_score = sc; best_a = a_new.copy()

  a = a_new

  t += 1

2. **L‑BFGS on a smooth max proxy** – the true objective
   2·n·max(conv) is approximated by a log‑sum‑exp with a decreasing
   smoothing factor λ.  A continuation over a λ‑schedule quickly
   approaches a (local) optimum.

3. **Adam‑styled projected sub‑gradient descent** – the genuine
   max‑convolution is handled; its sub‑gradient is computed by FFT and
   the update follows Adam with projection onto the probability simplex.
   This phase performs a large number of cheap iterations.

4. **Simulated annealing** – stochastic hill‑climb with a square‑root
   cooling schedule for final fine‑tuning.

All phases respect a global 1000 s deadline and the best sequence seen
at any moment is returned.  The final sequence is evaluated with the
official `evaluate_sequence` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a vector with ∑a = 1.
    The true score is 2·n·max(conv) / (∑a)^2, but ∑a == 1 here.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def generate_exp_power(
    n: int, rate: float, p: float, inc: bool = True
) -> np.ndarray:
    """Family a_i ∝ (i+1)^p·exp(-rate·i) (or reversed if inc=False)."""
    i = np.arange(n, dtype=float)
    power = (i + 1) ** p if inc else (n - i) ** p
    a = power * np.exp(-rate * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_grid_combined(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Grid search over the (rate, p) parameters of the
    (i+1)^p·exp(-rate·i) family.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # search ranges – chosen from exploratory runs
    rates = np.linspace(0.00010, 0.00080, 35)   # fine‐grained exponential decay
    powers = np.linspace(-5.0, -0.4, 30)       # negative powers → decreasing shapes

    for rate in rates:
        for p in powers:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_exp_power(n, rate, p, inc=True)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] rate={rate:.6f} p={p:.2f} → fast_score={best_sc:.6f}")

    # a short random refinement, still inside the deadline
    while time.time() < deadline:
        rate = rng.uniform(0.00005, 0.0010)
        p = rng.uniform(-6.0, -0.2)
        a = generate_exp_power(n, rate, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[rand] rate={rate:.6f} p={p:.2f} → fast_score={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# L‑BFGS on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(conv) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # initial logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑styled projected sub‑gradient descent on true max‑convolution
# ----------------------------------------------------------------------


def adam_subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # true convolution (linear, zero‑padded)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # mask of indices attaining (or almost attaining) the max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (see previous implementation)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam iteration
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        # a learning‑rate schedule decreasing like 1/√t
        lr = 0.15 / np.sqrt(it + 1.0)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
            # occasional progress report
            if it % 200 == 0:
                print(f"[adam] iter={it} best_score={best_score:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (final refinement)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves consist of transferring a random fraction of mass between two
    positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)  # initial temperature
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # cooling (slower than linear)
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # pick source and destination
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        # move a random fraction (up to 30 %) from i to j
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (can be changed, but 4000 works well for the budget)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate – uniform distribution
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a previous construction is present, import it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] imported previous fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s

    # Stage allocations (fractions of total budget)
    phase0_end = start_time + 0.07 * budget_s   # parametric grid + random sweep
    phase1_end = start_time + 0.22 * budget_s   # L‑BFGS refinement
    phase2_end = start_time + 0.55 * budget_s   # Adam sub‑gradient descent
    phase3_end = start_time + 0.85 * budget_s   # simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – rich parametric sweep (exp·power family)
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_combined(n, fft_len_conv, rng, phase0_end)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[phase0] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 500  # modest budget per λ

    # a few restarts – each will stop if the overall deadline is hit
    while time.time() < phase1_end:
        # jitter the current best to possibly escape a flat basin
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            # fresh exponential start – a reliable baseline for L‑BFGS
            rate = rng.uniform(0.00015, 0.00070)
            a_start = generate_exp_power(n, rate, -1.0)  # pure exponential

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam projected sub‑gradient descent (major workhorse)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_subgradient_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
