def flatten_descent(a_start, deadline, n, fft_len_conv, fft_len_grad, rng, max_iter=20000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    target_val = 1.0 / (2 * n - 1)
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute conv
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        # compute residual
        residual = conv - target_val

        # gradient of L2 loss
        # conv gradient formula: g = 4 * (residual * a_rev) conv slice
        a_rev = a[::-1]
        # use length L for FFT of residual and a_rev, we can compute with next_pow2
        L = fft_len_grad  # predetermined as next_pow2(3 * n - 2)
        # Compute FFT of residual padded
        fr = np.fft.rfft(residual, L)
        fr_rev = np.fft.rfft(a_rev, L)
        grad_full = np.fft.irfft(fr * fr_rev, L)
        # Extract slice n-1 .. n-1 + n
        grad = 4.0 * grad_full[(n - 1):(n - 1 + n)]

        # gradient of loss w.r.t a is 2 * something? Wait we derived g = 4 * conv_rev_slice. That gradient is for L2 loss L = sum (r_k)^2. Indeed derivative dL/da_i = 2 * Σ_k r_k * ∂r_k/∂a_i = 2 * Σ_k r_k * 2 * a_{k-i} = 4 * Σ_k r_k * a_{k-i} matches above. Good.

        # Descent direction: negative gradient
        d = -grad

        # Projected onto simplex: we need to move a by step and then project.
        # Use step size schedule:
        eta = 0.5 / np.sqrt(it + 1.0)  # can adjust
        a_new = a + eta * d
        # Ensure non-negative then project:
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = best_a.copy()

        # evaluate fast_score of new a (since sum=1 it's correct)
        sc_new = fast_score(a_new, fft_len_conv, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------
1.  Build a rich pool of initial guesses (uniform, Hann, Kaiser, Gaussian,
    spectral, random Dirichlet‑like, DPSS‑window if SciPy provides it).
2.  Choose the best candidate using the exact fast objective
    `2·n·max(conv(a,a))` (the sum of the coefficients is forced to 1).
3.  **Flattening phase** – minimise the L₂‑distance of the self‑convolution
    to the uniform target `1/(2·n‑1)`.  The gradient of the loss is obtained
    by a double FFT (complexity O(N log N)) and a projected (simplex) gradient
    step with a diminishing step‑size.  This phase quickly makes the
    convolution nearly flat, which is a strong lower bound for the max‑norm.
4.  **Smoothed‑max polishing** – run a short L‑BFGS optimisation on the
    log‑sum‑exp (smooth‑max) surrogate of the max‑convolution, starting from
    the current best vector.  The λ‑schedule drives the surrogate to the
    true max.
5.  **Fine tuning** – a few rounds of exponentiated‑gradient descent on the
    exact max‑convolution (masking the top entries) and a final greedy
    hill‑climb with tiny mass‑transfer moves.
6.  Return the best list of floats found before the deadline.  The official
    `evaluate_sequence` rescales internally, so we keep the coefficients on
    the simplex (sum = 1) throughout.

All phases respect a global time budget (`budget_s`, default 1000 s) and
periodically check the deadline, guaranteeing a result before termination.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast objective (exact for sum=1)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum=1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u = np.zeros(L)
    u[:target_len] = uniform
    S = np.fft.fft(u)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_cplx = np.fft.ifft(A)
    a = np.real(a_cplx[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random draws on the simplex (several families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def dpss_init(n: int, rng: np.random.Generator) -> np.ndarray | None:
    """Discrete prolate spheroidal (Slepian) window – if SciPy supplies it."""
    try:
        from scipy.signal import windows
    except Exception:
        return None
    # Choose a bandwidth parameter in a reasonable range.
    nw = rng.uniform(2.0, max(4.0, n / 4.0))
    a = windows.dpss(n, nw, sym=True)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space, guarantees positivity

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max (log‑sum‑exp) of the self‑convolution and its gradient
    w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    The mask now contains the top `top_fraction` of values (instead of just
    the exact maximum).  This yields a smoother sub‑gradient and improves
    convergence.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask = top‑fraction of entries (including the maximum)
        kth = int(np.ceil(top_fraction * conv.size))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of (smoothed) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule (decreasing)
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.07):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# ----------------------------------------------------------------------


def flatten_descent(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len_conv: int,
                    fft_len_grad: int,
                    rng: np.random.Generator,
                    max_iter: int = 200000):
    """
    Projected gradient descent on the L2 distance between the self‑convolution
    and the uniform target 1/(2·n‑1).  This makes the convolution almost flat,
    which is a very good proxy for minimising its maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    target = 1.0 / (2.0 * n - 1.0)
    it = 0

    while time.time() < deadline and it < max_iter:
        # convolution a*a  (needed for the target residual)
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        # residual = conv - constant
        residual = conv - target

        # gradient of L2 loss = 4 * (residual * a_rev)  (valid slice)
        a_rev = a[::-1]
        fr = np.fft.rfft(residual, fft_len_grad)
        fr_rev = np.fft.rfft(a_rev, fft_len_grad)
        grad_full = np.fft.irfft(fr * fr_rev, fft_len_grad)
        grad = 4.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction
        d = -grad

        # diminishing step size
        eta = 0.8 / np.sqrt(it + 1.0)

        a_new = a + eta * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = best_a.copy()

        sc = fast_score(a_new, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """

    # Problem size – a slightly larger n gives more freedom.
    # --------------------------------------------------------------
    n = 5000
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = next_pow2(3 * n - 2)          # for gradient of L2 loss

    # --------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse a previous construction if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    # randomised families
    for _ in range(3):
        candidates.append(random_initial(n, rng))
    for _ in range(2):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # optional DPSS windows
    dpss = dpss_init(n, rng)
    if dpss is not None:
        candidates.append(dpss)

    # choose the best candidate according to the exact objective
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – flatten the convolution (L2 minimisation)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.18 * budget_s
    best_a, best_score = flatten_descent(best_a,
                                          phase0_end,
                                          n,
                                          fft_len_conv,
                                          fft_len_grad,
                                          rng,
                                          max_iter=150000)
    print(f"[flatten] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing on the smooth‑max surrogate
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        # blend current best with a fresh random vector – keeps diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35          # mix ratio (≤0.35)
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_conv,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the exact max
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len_conv,
                                                   rng,
                                                   top_fraction=0.02)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=80000,
                                        delta_frac=0.06)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Convert to Python list and evaluate with the official function.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
