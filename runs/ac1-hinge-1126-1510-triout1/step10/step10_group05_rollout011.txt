```
def upsample_knots(knot_vals, n):
    # knot_vals length K, nonnegative.
    # we linearly interpolate to length n.
    K = len(knot_vals)
    # positions array of length K equally spaced within [0, n-1]
    knots_pos = np.linspace(0, n-1, K)
    # Use np.interp to get values at integer indices.
    idx = np.arange(n)
    a = np.interp(idx, knots_pos, knot_vals)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a.sum()
    return a
```

```
def lowdim_objective_and_grad(y, ...):
    # compute w (softmax)
    m = np.max(y)
    e = np.exp(y - m)
    w = e / e.sum()
    # compute a from w
    a = w[region_of_index] / repeats[region_of_index]  # vector length n
    # compute conv via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2*n -1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. a
    w_conv = exp_shift / Z   # size 2n-1
    # compute gradient using FFT as before:
    a_rev = a[::-1]
    fw = np.fft.fft(w_conv, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_a = conv_rev[n-1: n-1 + n]   # gradient wrt a

    # convert gradient w.r.t. w (region-level) using mapping
    # For each region j: g_w[j] = sum_{i where region[i]==j} g_a[i] / repeats[j]
    g_w = np.zeros_like(w)
    # sum per region
    # Using np.add.at
    np.add.at(g_w, region_of_index, g_a / repeats[region_of_index])
    # Gradient w.r.t. y (softmax param)
    mean_g = np.dot(w, g_w)
    grad_y = w * (g_w - mean_g)
    return f, grad_y
```

```
def lowdim_lbfgs_continuation(y_start, lam_schedule, maxiter_per_lam, deadline, n, K, repeats, region_of_index, fft_len):
    y = y_start.copy()
    best_y = y.copy()
    # compute initial a
    w = np.exp(y - np.max(y))
    w /= w.sum()
    a = w[region_of_index] / repeats[region_of_index]
    a /= a.sum() # maybe redundant? Actually sum should be 1 already due to mapping, but we can ensure.
    best_score = fast_score(a, fft_len)
    best_a = a.copy()
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=lowdim_objective_and_grad,
                           x0=y,
                           args=(n, K, repeats, region_of_index, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam, 'gtol': 1e-12, 'disp': False})
        except Exception:
            continue
        y = res.x
        w = np.exp(y - np.max(y))
        w /= w.sum()
        a = w[region_of_index] / repeats[region_of_index]
        # ensure sum=1
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_y = y.copy()
    return best_y, best_a, best_score
```

```
def compute_blocks(n, K):
    base = n // K
    remainder = n % K
    repeats = np.full(K, base, dtype=int)
    repeats[:remainder] += 1
    # region array
    region = np.empty(n, dtype=int)
    idx = 0
    for j, r in enumerate(repeats):
        region[idx: idx + r] = j
        idx += r
    return repeats, region
```

"""
Proposes a non‑negative step‑function sequence minimising

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in two stages.

Stage 1 – low‑dimensional optimisation:
    * The distribution is represented by K block‑wise constant values w (sum = 1).
    * A block j of length repeats[j] gets a_i = w_j / repeats[j] → the full
      vector a of length n has sum = 1.
    * Optimisation is performed on the logits y (soft‑max of w) using a
      smoothed version of the max‑convolution (log‑sum‑exp) and L‑BFGS.
      Several random restarts are performed.

Stage 2 – fine‑tuning on the full vector:
    * Starting from the best full‑vector a obtained above, we apply
      exponentiated‑gradient descent, projected sub‑gradient descent and a
      simulated‑annealing hill‑climb (all as in the reference implementation).
    * The whole budget (default 1000 s) is split between the two stages.

The final list of floats is returned; it is guaranteed to be non‑negative.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_blocks(n: int, K: int):
    """Return block lengths (repeats) and region index for each element."""
    base = n // K
    remainder = n % K
    repeats = np.full(K, base, dtype=int)
    repeats[:remainder] += 1
    region = np.empty(n, dtype=int)
    idx = 0
    for j, r in enumerate(repeats):
        region[idx: idx + r] = j
        idx += r
    return repeats, region


def random_initial_full(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fallback uniform initial vector (full dimension)."""
    a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed objective for the full‑dim vector (used by L‑BFGS, Adam, …)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  y are logits; a = softmax(y) is a probability vector.
    Returns (objective, gradient w.r.t. y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                       # weights for each conv entry
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # ∂smooth_max/∂a before soft‑max

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)           # chain rule through soft‑max
    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Cascade of L‑BFGS optimisations for the full‑dim problem.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the smoothed objective (full‑dim).
    Returns best a and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (hard max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """Exponentiated‑gradient descent directly on the hard max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]      # true sub‑gradient

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard max)
# ----------------------------------------------------------------------


def compute_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        alpha0: float = 0.5):
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        conv = compute_conv_fft(a, fft_len)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        if idx.size:
            g[idx] = a[k - idx]

        alpha = alpha0 / np.sqrt(it + 1.0)
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = best_a.copy()
            s = a_new.sum()

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (full‑dim)
# ----------------------------------------------------------------------


def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """Greedy hill‑climb with temperature schedule."""
    a = a_start.copy()
    conv = np.convolve(a, a)          # exact convolution (length 2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    T0 = max((best_score - 1.5) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1200
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (ignore O(delta²) terms)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Low‑dimensional smoothed objective (K‑dim logits)
# ----------------------------------------------------------------------


def lowdim_objective_and_grad(y: np.ndarray, n: int, K: int,
                              repeats: np.ndarray, region: np.ndarray,
                              fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective for the block‑wise representation.
    y are logits; w = softmax(y) are block probabilities.
    a_i = w_{region_i} / repeats[region_i}.
    Returns (objective, gradient w.r.t. y).
    """
    # softmax over K blocks
    m = np.max(y)
    e = np.exp(y - m)
    w = e / e.sum()                     # shape (K,)

    # full vector a (length n) – the sum of a is exactly 1
    a = w[region] / repeats[region]      # shape (n,)

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. a (same as in full‑dim version)
    w_conv = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w_conv, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_a = conv_rev[n - 1: n - 1 + n]

    # aggregate per block (region)
    sum_g = np.bincount(region, weights=g_a, minlength=K)   # Σ_{i∈region_j} g_a[i]
    g_w = sum_g / repeats                                   # ∂f/∂w_j

    # chain rule through the soft‑max
    mean_g = np.dot(w, g_w)
    grad_y = w * (g_w - mean_g)
    return f, grad_y


def lowdim_lbfgs_continuation(y_start: np.ndarray,
                              lam_schedule,
                              maxiter_per_lam: int,
                              deadline: float,
                              n: int,
                              K: int,
                              repeats: np.ndarray,
                              region: np.ndarray,
                              fft_len: int):
    """
    L‑BFGS cascade on the low‑dimensional representation.
    Returns best logits y, the corresponding full vector a and its fast_score.
    """
    y = y_start.copy()
    best_y = y.copy()
    # initial a
    w = np.exp(y - np.max(y))
    w /= w.sum()
    a = w[region] / repeats[region]
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=lowdim_objective_and_grad,
                       x0=y,
                       args=(n, K, repeats, region, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        w = np.exp(y - np.max(y))
        w /= w.sum()
        a = w[region] / repeats[region]

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_y = y.copy()
    return best_y, best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – larger n gives finer resolution at modest cost
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # Low‑dimensional parametrisation
    K = max(50, n // 40)                # ~200 blocks for n=8000
    repeats, region = compute_blocks(n, K)

    # Initialise logits y (log‑softmax of block probabilities)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0 and old.sum() > 0.0:
            old = old / old.sum()
            # map old vector onto K blocks
            m_old = old.size
            repeats_old, region_old = compute_blocks(m_old, K)
            w0 = np.bincount(region_old, weights=old, minlength=K)
            if w0.sum() == 0:
                w0 = np.full(K, 1.0 / K)
            else:
                w0 = w0 / w0.sum()
            y0 = np.log(w0 + 1e-12)
        else:
            y0 = np.log(np.full(K, 1.0 / K))
    else:
        y0 = np.log(np.full(K, 1.0 / K))

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – low‑dimensional L‑BFGS with several restarts
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.35 * budget_s
    lam_schedule_low = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
                        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
                        0.0008, 0.0004, 0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 400

    best_a = None
    best_score = np.inf
    best_y = None
    restart = 0
    while time.time() < phase0_end and restart < 7:
        if restart == 0:
            y_start = y0.copy()
        else:
            # random Dirichlet start
            w_rand = rng.dirichlet(np.ones(K))
            y_start = np.log(w_rand + 1e-12)
        y_opt, a_opt, score_opt = lowdim_lbfgs_continuation(
            y_start, lam_schedule_low, maxiter_per_lam,
            phase0_end, n, K, repeats, region, fft_len
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[low‑dim L‑BFGS restart {restart}] new best fast_score={best_score:.6f}")
        restart += 1

    # Fallback if low‑dim failed to produce a vector
    if best_a is None:
        best_a = random_initial_full(n, rng)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent on the full vector
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.65 * budget_s
    if time.time() < phase1_end:
        best_a, score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng, eta0=12.0)
        if score < best_score:
            best_score = score
            print(f"[exp‑grad post] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (hard max)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    if time.time() < phase2_end:
        best_a, score = subgradient_descent(
            best_a, phase2_end, n, fft_len, rng, alpha0=0.5)
        if score < best_score:
            best_score = score
            print(f"[sub‑grad post] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_hill_climb(
            best_a, best_score, deadline, n, fft_len, rng)
        print(f"[anneal final] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
