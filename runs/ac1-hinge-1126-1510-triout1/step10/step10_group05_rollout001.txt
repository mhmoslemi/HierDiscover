from scipy.optimize import linprog
from scipy.sparse import diags

def lp_optimize(a, M):
    n = len(a)
    # Build sparse matrix A (2n-1) x n, each diagonal d has a[d] on offset d
    diagonals = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)  # offsets from 0 to n-1
    A = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    # constraints: A @ b <= M (vector)
    b_ub = np.full(2*n-1, M)
    # maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # bounds for b
    bounds = [(0, None)] * n
    # method: 'highs'
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': ???}) # can set maybe no limit.

    if res.success:
        b = res.x
        return b
    else:
        return None

def lp_optimize(a, M):
    n = len(a)
    # Build sparse matrix A using diags with constant diagonals
    # duplicates: we create data = np.tile(a[:, None], (1, n))? that's n x n dense.
    # Instead, build using 'scipy.sparse.dia_matrix' with data = np.vstack([np.full(n, a[i]) for i in range(n)])
    # Use format='csr'
    diags_data = np.vstack([np.full(n, a[i]) for i in range(n)])  # shape (n, n)
    offset = np.arange(n)
    A = scipy.sparse.dia_matrix((diags_data, offset), shape=(2*n-1, n)).tocsr()
    # constraints
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    return res

"""
Proposes a non‑negative step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser consists of three stages, all bounded by a 1000 s budget:

1. **Initialisation** – use the best known construction (global
   `height_sequence_1`) if available, otherwise a uniform distribution.

2. **Linear‑programming based improvement** – given the current distribution
   `a` with maximal autocorrelation `M`, solve the linear programme

        maximise   ∑ b_j
        subject to   (a * b)[k] ≤ M   for all k
                     b_j ≥ 0

   The solution `b` has total mass larger than 1 whenever the bound `M`
   can be improved.  Normalising `b` to a probability vector `g` and mixing
   it with `a` (`a←(1‑t)a + t g`) strictly reduces the objective.
   The process is repeated until no improvement is possible or the time
   limit is reached.

3. **Fine‑grained simulated annealing** – a hill‑climb that moves a small
   fraction of mass between two indices, updates the convolution
   incrementally and accepts uphill moves according to a cooling schedule.
   This stage polishes the result obtained by the LP phase.

The final sequence (a plain list of floats) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Score = 2·n·max(conv(a,a)) for a probability vector.
    (Exact value of the official scorer up to the normalisation.)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# -------------------------------------------------------------
# Linear‑programming improvement (LP‑step)
# -------------------------------------------------------------


def build_constraint_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Construct the (2n‑1)×n matrix A with A[k,i] = a[k‑i] for k≥i,
    otherwise 0.  This encodes the constraints (a * b)[k] ≤ M.
    """
    n = a.shape[0]
    # each diagonal d (0 ≤ d < n) consists of the constant a[d]
    # Build a (n, n) dense array where row i is a[i] repeated n times
    # and then treat rows as diagonals.
    data = np.tile(a[:, None], (1, n))          # shape (n, n)
    offsets = np.arange(n)                      # 0 … n‑1
    # dia_matrix stores each diagonal in a separate row of `data`
    A = sparse.dia_matrix((data, offsets), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_step(a: np.ndarray, M: float, time_deadline: float):
    """
    Solve the LP
        maximise   ∑ b_j
        subject to   A b ≤ M   (A encodes convolution with ``a``)
                     b ≥ 0
    Returns the optimal `b` (or None on failure) together with the LP solve time.
    """
    n = a.shape[0]
    A = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # minimise -∑b  → maximise ∑b
    bounds = [(0.0, None)] * n

    # HiGHS is fast for large sparse problems; we give it a soft time limit.
    # The `options` dict is passed to HiGHS; we only set `presolve`.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None, None
    return res.x, res


def lp_improve(a_start: np.ndarray, fft_len: int, deadline: float):
    """
    Repeatedly apply the LP‑step / mixing scheme while time permits.
    Returns the best sequence found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    n = a.shape[0]

    iteration = 0
    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        M = conv.max()

        # Solve LP
        b, _ = lp_step(a, M, deadline)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No further improvement possible (within numerical tolerance)
            break

        # Normalise b to obtain a probability vector g
        g = b / sum_b

        # Try several mixing coefficients and keep the best improvement
        best_local_score = best_score
        best_t = 0.0
        # Candidate mixing coefficients (coarse → fine)
        candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 1.0]
        for t in candidates:
            a_new = (1.0 - t) * a + t * g
            # Re‑normalise (numerical drift)
            a_new /= a_new.sum()
            score = fast_score(a_new, fft_len)
            if score < best_local_score:
                best_local_score = score
                best_t = t
                best_candidate = a_new

        if best_t == 0.0:
            # No mixing coefficient gave improvement – stop the LP phase
            break

        a = best_candidate
        best_a = a.copy()
        best_score = best_local_score
        iteration += 1
        if iteration % 5 == 0:
            print(f"[LP] iter={iteration} score={best_score:.6f}")

    return best_a, best_score

# Simulated‑annealing hill‑climb refinement
# -------------------------------------------------------------


def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb with temperature schedule.
    Uses O(n) incremental updates of the convolution.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact convolution, length 2n‑1
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule parameters (more aggressive than the previous run)
    T0 = max((best_score - 1.2) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.12
    max_fraction_end = 0.015
    recompute_every = 1500
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # linear cooling
        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        # pick source (with larger mass) and destination (with smaller mass)
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Incremental update of convolution (ignoring O(delta²) term)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    return best_a, best_score
