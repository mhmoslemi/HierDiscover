def local_random_search(a, score, fft_len, n, deadline, step_start=0.1, step_decay=0.99, min_step=1e-7):
   step = step_start
   while time.time() < deadline and step > min_step:
       # compute conv
       conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len)*np.fft.fft(a, fft_len)))[:2*n-1]
       max_idx = np.argmax(conv)
       # compute partner values
       i_range_start = max(0, max_idx-(n-1))
       i_range_end = min(n-1, max_idx)
       idx = np.arange(i_range_start, i_range_end+1)
       partner = np.zeros_like(a)
       partner[idx] = a[max_idx - idx]
       contrib = a * partner
       i_max = np.argmax(contrib)
       # choose i_min among indices with smallest partner, not equal i_max
       # random among low values for variety.
       low_idxs = np.where(partner == partner.min())[0]
       # ensure not i_max
       low_idxs = low_idxs[low_idxs != i_max]
       if low_idxs.size==0:
           low_idx = np.argmin(partner) if np.argmin(partner)!=i_max else (np.argmin(partner)+1)%n
       else:
           low_idx = rng.choice(low_idxs)
       delta = min(a[i_max]*step, a[i_max]/2) # ensure not drop too far.
       if delta <= 0:
           break
       a_new = a.copy()
       a_new[i_max] -= delta
       a_new[low_idx] += delta
       a_new[a_new < 0] = 0
       a_new /= a_new.sum()
       new_score = fast_score(a_new, fft_len, n)
       if new_score < score:
           a = a_new
           score = new_score
           # keep step
       else:
           step *= step_decay
   return a, score

"""
Propose a low‑score (non‑negative) step‑function sequence.

The routine follows several complementary optimisation stages inside the
1000 s budget:

1. Initialise a diverse pool (uniform, spectral‑square‑root, random and a
   symmetrised previous result if available).
2. Run many Adam restarts on a log‑softmax parametrisation with a
   decreasing “smooth‑max’’ λ‑schedule (log‑sum‑exp surrogate).  Adam’s
   learning‑rate is fixed and we keep the vector symmetric after every
   update.
3. Refine the best vector with an exponentiated‑gradient descent that
   uses the true (hard) max‑convolution sub‑gradient.
4. Deterministic hill‑climb that moves a fraction of probability mass
   from the coordinate that contributes most to the current maximal
   convolution entry toward a coordinate with the smallest partner value.
5. Random local “mass‑transfer’’ search that repeats the same idea with
   a geometrically shrinking step‑size and stochastic destination choice.
6. Exhaustive convex‑combination (golden‑section) search among the top‑K
   vectors discovered so far.
7. Final symmetrisation and evaluation with the provided security‑checked
   `evaluate_sequence` routine.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0) either by
soft‑max (phases 2) or by explicit normalisation (phases 3‑5).  The code
uses only NumPy and runs well within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming ∑a = 1.
    This is exactly the value returned by `evaluate_sequence`
    after clipping (our vectors stay in the allowed range).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a = a[::-1] (mirror symmetry) and renormalise."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s <= 0.0:
        return a.copy()
    return a_sym / s


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def uniform_init(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_root_init(n: int) -> np.ndarray:
    """
    Construct a start‑vector whose *auto‑convolution* approximates a flat
    spectrum: take the square‑root of the FFT of a uniform vector,
    invert, exponentiate to enforce positivity, and normalise.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    pad = np.zeros(L)
    pad[:target_len] = uniform
    S = np.fft.fft(pad)
    sqrt_S = np.sqrt(S)                 # principal complex square‑root
    a_complex = np.fft.ifft(sqrt_S)
    a_raw = np.real(a_complex[:n])
    # Exponentiate to obtain a strictly positive vector
    a = np.exp(a_raw)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max of the self‑convolution using log‑sum‑exp.
    Returns tuple (objective value, gradient w.r.t. logits y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient wrt a (size n) from conv_rev
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score) where sum(best_a)=1.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            a = symmetrize(a)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 8.0):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = symmetrize(a)
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with smallest partner (avoid i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = symmetrize(a_new)
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)

    return a, score

# Random local mass‑transfer search
# ----------------------------------------------------------------------
def random_local_search(a: np.ndarray,
                        score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        start_step: float = 0.10,
                        decay: float = 0.98,
                        min_step: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Stochastic version of the hill‑climb: pick a random low‑partner index
    as destination and shrink the step size geometrically.
    """
    step = start_step
    while time.time() < deadline and step > min_step:
        # compute convolution once
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner values for the dominant sum
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # choose a random low‑partner index (excluding i_max)
        low_candidates = np.where(partner <= partner.min() + 1e-12)[0]
        low_candidates = low_candidates[low_candidates != i_max]
        if low_candidates.size == 0:
            i_min = np.argmin(partner)
            if i_min == i_max:
                i_min = (i_max + 1) % n
        else:
            i_min = rng.choice(low_candidates)

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = symmetrize(a_new)
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step
        else:
            step *= decay

    return a, score


# ----------------------------------------------------------------------
# Convex‑combination exploration among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates, perform a cheap golden‑section line search
    over the mixing weight t ∈ [0,1] to minimise the bound.
    Returns the best vector and its score found during this phase.
    """
    def eval_mix(t, a, b):
        mix = (1.0 - t) * a + t * b
        mix = mix / mix.sum()
        return fast_score(mix, fft_len, n), mix

    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0].copy()
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b)
            fd, _ = eval_mix(d, a, b)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b)
            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score

        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
