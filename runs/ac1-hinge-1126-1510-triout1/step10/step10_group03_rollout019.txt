def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    n = 4000
    fft_len = next_pow2(2*n-1)

    # Initialize
    a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    # parameters
    lr0 = 5.0
    temp_factor = 0.2
    it = 1

    while time.time() < deadline:
        # FFT of a
        fa = np.fft.rfft(a, fft_len)
        # convolution
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n - 1]

        max_c = conv.max()
        # temperature
        T = max(max_c * temp_factor, 1e-12)

        # softmax weights
        shifted = conv - max_c  # max becomes 0
        w = np.exp(shifted / T)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient via convolution a_rev * w
        fa_rev = np.fft.rfft(a[::-1], fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2*n - 1]
        grad = 2.0 * grad_full[n-1:n-1 + n]

        # step size
        eta = lr0 / np.sqrt(it)
        # gradient descent update (additive)
        a_new = a - eta * grad

        # project onto simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new / s
        else:
            a_new = best_a.copy()

        # compute fast_score
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # optional print
            print(f"[iter {it}] new best fast_score={best_score:.6f}")

        a = a_new
        it += 1

    # final evaluation
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Propose a low‑score step sequence.

The core idea is to minimise the maximum entry of the self‑convolution
of a non‑negative vector a (Σa=1).  The objective is
  f(a)=2·n·max(conv(a,a)).
We treat max(·) via a soft‑max (log‑sum‑exp) surrogate so that the
gradient is well‑defined.  Using FFT we obtain the convolution and the
gradient in O(n log n) time.  An additive projected gradient descent with
a decaying step size and a temperature schedule (the “soft‑max”
parameter) is run for the majority of the time budget; the final
solution is optionally refined by a short simulated‑annealing phase.
The algorithm starts from the best sequence stored in the global
`height_sequence_1` if present, otherwise from the uniform distribution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming Σa = 1.
    The convolution is computed via FFT.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the
    provided `evaluate_sequence` function.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a few thousand entries provide a good trade‑off.
    # --------------------------------------------------------------
    default_n = 4000
    # initialise from a previous run if possible
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # re‑interpolate to the current length
            n = default_n
            x_old = np.linspace(0, 1, old.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = default_n
            a = np.full(n, 1.0 / n)
    else:
        n = default_n
        a = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # phase 1 – soft‑max gradient descent
    lr0 = 5.0                   # initial step size
    temp_factor = 0.2           # temperature as a fraction of current max
    it = 1

    while time.time() < deadline * 0.85:   # keep ~15% for a final tweak
        # FFT of the current vector
        fa = np.fft.rfft(a, fft_len)

        # full convolution a*a
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        # temperature – a small fraction of the current maximum
        T = max(max_c * temp_factor, 1e-12)

        # soft‑max weights (numerically stable: shift by max_c)
        shifted = conv - max_c          # max becomes 0
        w = np.exp(shifted / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smoothed max (2 * a_rev * w)
        fa_rev = np.fft.rfft(a[::-1], fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # step size decays like η₀ / √t
        eta = lr0 / np.sqrt(it)

        # additive gradient step, then project onto the simplex
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new / s
        else:
            a_new = best_a.copy()

        # evaluate (fast) and keep the best seen so far
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score, best_a = score, a_new.copy()
            print(f"[iter {it}] new best fast_score={best_score:.6f}")

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – short simulated annealing fine‑tuning
    # --------------------------------------------------------------
    # simple random mass transfers between two indices
    a = best_a.copy()
    conv = np.convolve(a, a)
    curr_score = best_score
    best_score = curr_score
    best_a = a.copy()
    temp0 = max(best_score * 0.02, 1e-6)
    start_sa = time.time()
    n_sa = n
    max_frac = 0.15   # up to 15 % of the source mass per move

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_sa
        frac = elapsed / (deadline - start_sa)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n_sa, size=2)
        if i == j:
            continue
        # always move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        diff = np.zeros_like(conv)
        diff[src : src + n_sa] -= a
        diff[dst : dst + n_sa] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n_sa * conv_new.max()

        dE = new_score - curr_score
        accept = dE < 0.0 or (rng.random() < np.exp(-dE / temp))

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

    print(f"[final] fast_score after optimisation = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
