def exp_gradient_exact(a_start, deadline, n, fft_len_conv, fft_len_g, rng):
    a = a_start.copy()
    best = a.copy()
    best_sc = fast_score(best, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)
        # gradient of exact max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2 * grad_full[(n-1):(n-1+n)]
        # exponentiated update
        eta = 0.5 / np.sqrt(it + 1)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum == 0:
            a = best.copy()
        else:
            a = a / a_sum
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        it += 1
    return best, best_sc

"""
Search for a low‑score non‑negative step function.

Algorithm Overview
------------------
1. **Problem size** – choose a moderately large n (8000) to give the optimizer
   enough degrees of freedom while keeping FFT work cheap.
2. **Rich initial pool** – deterministic windows (spectral, Gaussian with a
   sweep of σ, triangular, Hann, Blackman) plus several random Dirichlet‑like
   draws; the best of these is used as the seed.
3. **Phase 1 – L‑BFGS on a smoothed max‑convolution**.  The objective is a
   log‑sum‑exp approximation of the true max; a long λ‑schedule and many
   restarts aggressively polish the shape.
4. **Phase 2 – Exponentiated‑gradient descent on the *exact* max‑convolution**.
   The sub‑gradient of the true max is obtained by convolving the reversed
   sequence with a mask that is 1 at the maximal convolution indices.
5. **Phase 3 – Projected sub‑gradient descent (exact max)** for further
   refinement.
6. **Phase 4 – Simulated annealing** and **Phase 5 – greedy hill‑climb** for
   final polishing.
7. Return the best sequence (list of floats).  The official score is obtained
   by a single call to the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize
# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector a (sum = 1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (various families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalised to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window (good sidelobe suppression)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_exact(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    The mask contains the exact maximal entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        # gradient of (exact) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # step size schedule (decreasing)
        eta = 0.5 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact(a_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      rng: np.random.Generator,
                      step0: float = 0.3):
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        mask = (conv >= max_val - 1e-12).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction with zero‑sum constraint
        d = -(g - np.mean(g))

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # move up to 12 % of the source mass
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed.
    # --------------------------------------------------------------
    n = 8000                       # larger than before for extra freedom
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool (diverse families + random draws)
    # --------------------------------------------------------------
    candidates = []

    # If a previous construction exists, interpolate it to our size.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    # Several Gaussian spreads
    for sf in [0.22, 0.25, 0.29, 0.33, 0.37, 0.40]:
        candidates.append(gaussian_init(n, sigma_factor=sf))

    # Pick the best of the pool (fast_score is exact for sum=1)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,

        alpha = rng.random() * 0.3          # mix ratio (≤0.3)
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient on the exact max (≈25 % of budget)

    phase2_end = start_time + 0.65 * budget_s

                                                phase2_end,
                                                n,
                                                fft_len, fft_len,
                                                rng)
        print(f"[exp‑grad‑exact] best fast_score={best_score:.8f}")

        best_a, best_score = subgradient_exact(best_a,
                                                phase3_end,
                                                n,
                                                fft_len,
                                                rng,
                                                step0=0.25)
        print(f"[sub‑grad‑exact] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=6e-4,
                                                 final_temp=1e-7,
                                                 max_moves=200000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=120000,
                                        delta_frac=0.05)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
