def local_optimize(seq, n_iters=5000, step_scale=0.01):
    a = seq.copy()
    n = len(a)

        if time.time() >= deadline: break
        # random pair indices i,j where i != j

        # move delta from i to j

    n = len(vals)
    order = np.empty(n, dtype=int)
    left = 0
    right = n - 1
    idx = 0
    place_left = True
    while left <= right:
        if place_left:
            order[idx] = left
            left += 1
        else:
            order[idx] = right
            right -= 1
        idx += 1
        place_left = not place_left
    permuted = np.empty_like(vals)
    # assign sorted vals to these positions:
    permuted[order] = vals
    return permuted

            # 1. original sorted descending (base already descending)

            for _ in range(2):

                if time.time() >= deadline: break

                if score < best_score:
                    best_score = score; best_seq = seq.copy(); best_n = n; best_p = p

    if best_seq is not None:
        arr = best_seq.copy()
        n = len(arr)
        current_score = best_score
        temperature = 0.01 * current_score # maybe relative
        iter_count = 0
        while time.time() < deadline and iter_count < 300000:
            # propose swap
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose transfer amount: move delta from i to j
            # Instead of swapping values, we can just swap them
            # swapping simplifies to preserve sum exactly;
            # we can also try small delta moves.
            # Try swap values:
            arr[i], arr[j] = arr[j], arr[i]

            if accept:

    """
    Search for a non‑negative height sequence that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2)

    The method explores power‑law shaped sequences of various lengths,
    applies a simple “extreme‑interleave” permutation that spreads the
    largest values across the ends of the array, evaluates a handful of
    random shuffles, and finally performs a hill‑climbing/annealing phase
    that swaps two positions whenever the score improves (or with a
    decreasing probability if it worsens).  The best sequence found before
    the time budget expires is returned.

    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    def generate_power_law(n, p):
        idx = np.arange(1, n+1, dtype=np.float64)
        seq = idx ** (-p)
        seq /= seq.sum()
        return seq

        # seq assumed sorted descending
        n = len(seq)
        out = np.empty_like(seq)
        left = 0
        right = n - 1
        i = 0
        while left <= right:
            out[left] = seq[i]; i+=1; left += 1
            if left <= right:
                out[right] = seq[i]; i+=1; right -= 1
        return out

    best_p = None

"""
Propose a candidate step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search proceeds in three stages:

1. **Coarse exploration** – try power‑law shaped sequences
   a_i ∝ (i+1)^{-p} for several lengths `n` and exponents `p`.
   For every generated shape we evaluate three permutations:
   * original (monotonically decreasing),
   * an “extreme‑interleave” ordering that spreads the largest values
     to the two ends of the array,
   * two random shuffles.

2. **Fine‑grained refinement** – around the best exponent we sweep a
   finer grid and keep the best permutation.

3. **Local optimisation** – starting from the best sequence we perform
   a simulated‑annealing hill‑climb that swaps two positions.
   Swaps that improve the score are always kept; worsening swaps are
   occasionally accepted with a temperature that decays over time.

All evaluations use the provided ``evaluate_sequence`` function, which
includes the required security checks.  The routine stops as soon as the
time budget (default 1000 s) is almost exhausted and returns the best
sequence found.
"""

import numpy as np
import time
import random
import math

def generate_power_law(n: int, p: float) -> np.ndarray:
    """
    Return a length‑`n` non‑negative sequence following the power‑law

        a_i = (i+1)^{-p}   (i = 0 … n‑1)

    and normalised so that sum(a) == 1.  The result is a NumPy array of
    ``float64``.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def interleave_extremes(seq: np.ndarray) -> np.ndarray:
    """
    Given a descending sequence ``seq`` (largest entry first), return a
    new sequence where the entries are placed alternatingly at the left
    and right ends of the array:

        out[0]   = seq[0]      (largest)
        out[-1]  = seq[1]      (second largest)
        out[1]   = seq[2]      (third largest)
        out[-2]  = seq[3]      …

    This spreads the largest values across the extremes of the domain,
    which tends to flatten the self‑convolution.
    """
    n = len(seq)
    out = np.empty_like(seq)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[left] = seq[i]
        i += 1
        left += 1
        if left <= right:
            out[right] = seq[i]
            i += 1
            right -= 1
    return out

    The algorithm consists of a grid search over (n, p) with a few simple

    hill‑climbing phase that swaps two entries at a time.  The best
    sequence discovered before the time limit is returned as a plain
    Python list.
    """
    # ---------- initialisation ----------
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # retrieve length of a previously found construction, if any
    try:
        n_start = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821

    best_score = float("inf")
    best_n = None
    best_p = None

    # ---------- stage 1: coarse grid ----------
    # lengths to test (roughly logarithmic spread)
    n_candidates = [int(x) for x in np.linspace(2000, 8000, 7)]
    if n_start not in n_candidates:
        n_candidates.append(n_start)
    n_candidates = sorted(set(n_candidates))

    # Return plain list as required by the external harness
    return best_seq.tolist()
