"""
Propose a low‑score step‑function.

The algorithm consists of three phases:

1. **Parameterized sweep** –  generate sequences of the form
   a[i] = (i+1+δ)^(-p) for many lengths `n`, exponents `p∈[0.5,0.95]`
   and offsets `δ`.  The evaluation function is homogeneous of degree 0,
   therefore the absolute scale of the sequence does not matter.
   The best shape found in this inexpensive grid search is kept.

2. **Fine‑grained local optimisation** – starting from the best shape,
   we run a projected (sub‑)gradient descent.  The objective
   `2·n·max(a*a)/sum(a)^2` depends only on the maximal entry of the
   autocorrelation, so the sub‑gradient is the sum of the gradients of
   all entries that are within a tiny relative tolerance of that maximum.
   After each step we project onto the simplex (non‑negativity + fixed
   ℓ₁‑norm) and occasionally add a tiny random jitter to escape flat
   regions.

3. **Return the best sequence** – the best candidate seen throughout the
   whole budget (default 1000 s) is returned as a plain Python list.

The method is deliberately simple, avoids heavy linear programming
machinery and exploits the analytic insight that power‑law shapes with
exponent > 0.5 give scores well below the previous 1.57 limit (≈π/2).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas – top level)
# ----------------------------------------------------------------------


def _shape_sequence(length: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    """
    Return a non‑negative vector a[i] = (i+1+offset)^(-exponent).
    The values are clipped to the interval required by ``evaluate_sequence``.
    """
    idx = np.arange(1, length + 1, dtype=np.float64) + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _max_convolution_and_active(seq: np.ndarray, rel_tol: float = 1e-8):
    """
    Compute the autocorrelation of ``seq`` and return the maximal value
    together with the list of indices whose value is within ``rel_tol`` of the
    maximum (sub‑gradient active set).
    """
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # treat floating point noise – active set may contain multiple indices
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    return max_val, active


def _subgradient_active(seq: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(seq,seq))`` w.r.t. ``seq`` obtained by
    summing the sub‑gradients of all active convolution entries.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=np.int64)
            i = k - j
            # each contribution is 2 * a[i] (see analytical derivation)
            grad[j] += 2.0 * seq[i]
    return grad


def _project_onto_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project non‑negative vector ``x`` onto the hyperplane sum(x)=target_sum.
    This is simply a positive scaling because all components are already ≥0.
    """
    cur = np.sum(x)
    if cur == 0.0:
        # cannot scale – return a uniform distribution with the required sum
        return np.full_like(x, target_sum / len(x))
    return x * (target_sum / cur)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds early).

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search over length / exponent / offset
    # ------------------------------------------------------------------
    best_seq = None

    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
