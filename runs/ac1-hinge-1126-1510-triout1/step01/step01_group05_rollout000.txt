"""
Propose a low‑evaluation step sequence.

Algorithm Overview
------------------
1. **Problem size** – we work with n = 2000 coefficients.
2. **Initial guess** – use the previously discovered best sequence
   (global variable ``height_sequence_1``) if available, otherwise the
   uniform distribution.
3. **Smooth‑max L‑BFGS phase** – many random restarts of L‑BFGS on the
   differentiable surrogate
   `2·n·smooth_max(a*a)` where `smooth_max` is a log‑sum‑exp
   approximation.  The smoothing factor is reduced as the search
   progresses, allowing a coarse‑to‑fine optimisation.
4. **Greedy hill‑climbing** – after each L‑BFGS run we apply a fast
   random mass‑transfer hill‑climb (up to a few thousand moves) that
   strictly improves the true objective `2·n·max(conv)`.
5. **Simulated annealing** – a long annealing phase makes stochastic
   moves (transfer a random fraction of mass between two positions) and
   accepts worsening moves with a temperature schedule.  The fast FFT‑
   based score is used for each trial.
6. **Final refinement** – a short deterministic hill‑climb finishes the
   optimisation.
7. **Return** – the sequence (list of floats) is handed to the provided
   `evaluate_sequence` for the final score.

All heavy work is performed with FFTs (length is the next power of two
≥ 2·n‑1, i.e. 4096 for n = 2000), so each score evaluation costs only a
few hundred micro‑seconds, allowing many thousands of iterations within
the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    ``a`` must have length ``n``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence that already sums to 1:
        score = 2·n·max(conv(a, a))
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max surrogate for 2·n·max(a*a) and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables; ``a = softmax(y)`` lies on the simplex.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for convolutions.
    lam_factor : float
        Determines the smoothing width: λ = lam_factor·max(conv).

    Returns
    -------
    f : float
        Smooth objective value.
    g : np.ndarray
        Gradient w.r.t. y (shape (n,)).
    """
    # map to simplex
    a = softmax(y)                       # a_i ≥ 0, sum a_i = 1

    # ----- a * a -------------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12      # smoothing width, strictly positive
    shifted = conv - max_c                # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max ---------------------------------------
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i  (weights)

    # g_vec = ∂smooth_max/∂a (size n)
    # It is the convolution of reversed a with the weight vector w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice: indices (n‑1) … (n‑1 + n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max:
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climb.
    Randomly moves a small fraction of mass from one coordinate to another.
    Keeps the move only if the true score (2·n·max conv) improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the final evaluation
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 2000
    fft_len_conv = next_pow2(2 * n - 1)     # 4096 for n=2000
    fft_len_g = next_pow2(3 * n - 2)        # 8192 for n=2000

    # ------------------------------------------------------------------
    # Initialise a good starting point (previous best if available)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=float)
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        if init_seq.size != n:
            # fall back to uniform if size mismatches
            init_seq = np.full(n, 1.0 / n)
        else:
            # Normalise to sum = 1 (soft‑max domain)
            init_seq = init_seq / init_seq.sum()
    except NameError:
        init_seq = np.full(n, 1.0 / n)

    best_a = init_seq.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts on a smooth surrogate
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 1.0:
        # ------- choose a start point ---------------------------------
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'best'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # mode == 'best'
            # perturb the current best point
            log_best = np.log(best_a + 1e-12)
            y0 = log_best + rng.normal(scale=0.1, size=n)

        # ------- choose smoothing factor (coarse to fine) ------------
        if iteration < 40:
            lam_factor = 0.5
        elif iteration < 150:
            lam_factor = 0.2
        else:
            lam_factor = 0.08

        # ------- run L‑BFGS -----------------------------------------
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})
        a_opt = softmax(res.x)   # already normalised
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration:04d}] mode={mode:6s} lam={lam_factor:.2f} "
                  f"new best = {best_score:.6f}")

        # ------- short greedy refinement ------------------------------
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on the true objective
    # ------------------------------------------------------------------
    # Temperature schedule: start relatively warm, cool to near zero.
    temp_start = 0.12
    temp_end = 0.001
    start_time = time.time()
    total_sa_time = max(0.0, deadline - time.time() - 1.0)
    sa_deadline = time.time() + total_sa_time

    # keep a copy of the current best score for acceptance decisions
    cur_a = best_a.copy()
    cur_score = best_score

    while time.time() < sa_deadline:
        # linear interpolation of temperature
        elapsed = time.time() - start_time
        frac = elapsed / total_sa_time if total_sa_time > 0 else 1.0
        temp = temp_start * (1.0 - frac) + temp_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction (≤10 %) of mass from i to j
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = fast_score(new_a, fft_len_conv, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_a = new_a
            cur_score = new_score
        else:
            accept_prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if rng.random() < accept_prob:
                cur_a = new_a
                cur_score = new_score

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[SA] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic hill‑climb (short)
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    deadline,
                                    max_iter=2500)

    # ------------------------------------------------------------------
    # Final evaluation (the official scorer applies clipping etc.)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")

    return final_seq
