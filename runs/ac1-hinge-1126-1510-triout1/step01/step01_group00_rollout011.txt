"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

under the usual security checks.  The algorithm works on the *probability
simplex* (∑a = 1, a ≥ 0) and uses a smoothed “soft‑max’’ of the convolution
to obtain a dense sub‑gradient.  An Adam‑style optimiser with Euclidean
projection onto the simplex drives the iterates.  A simple annealing schedule
for the soft‑max temperature (τ) and occasional random restarts help escape
local minima.  The final sequence is rescaled to the required sum
√(2·n) before being returned.

The routine respects a time budget (default 1000 s) and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective for a *scaled* sequence.
# For ∑a = √(2·n) the objective equals max(conv(a, a)).
def _score_normalised(conv_max: float, n: int) -> float:
    """Return the evaluation value when the current normalised vector has
    max‑convolution = conv_max (its sum equals 1)."""
    return 2.0 * n * conv_max  # because target sum = sqrt(2 n) cancels out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a step‑function with a low evaluation value.
    The function respects the given time budget and returns a plain list
    of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the vector.
    try:
        # use the previously discovered construction if it exists
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        n_prev = prev.shape[0]
        # we aim for a larger, power‑of‑two length for FFT efficiency
        n = max(20000, 1 << ((n_prev - 1).bit_length()))  # at least 20 k
    except Exception:
        n = 20000                    # default length if no global is present

    # target sum used by the evaluation routine
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialisation of the probability vector (sum = 1)
    if 'prev' in locals():
        # interpolate the previous shape to the new length
        x_old = np.linspace(0.0, 1.0, n_prev)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, prev)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        # a smooth increasing power‑law (p≈1.5 works well)
        p0 = 1.5
        a = np.arange(1, n + 1, dtype=np.float64) ** p0
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Adam‑type hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                      # base learning rate

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    # FFT lengths (power of two for speed)
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a * a
    L_corr = 1 << ((3 * n - 2).bit_length())          # for w * a_rev

    # bookkeeping
    best_seq = a * target_sum
    # compute initial max‑convolution (on the normalised vector)
    fft_a = np.fft.fft(a, n=L_conv)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
    max_conv = conv_full.max()
    best_score = _score_normalised(max_conv, n)

    it = 0
    max_iters = 50000            # hard cap; the time budget will stop earlier

    print(f"--- Start optimisation: n={n}, target_sum={target_sum:.3f} ---")
    print(f"Initial score = {best_score:.6f}")

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # 1) convolution of the current iterate (normalised, sum = 1)
        fft_a = np.fft.fft(a, n=L_conv)
        conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_conv = conv_full.max()

        # --------------------------------------------------------------
        # 2) soft‑max weights on the convolution entries (temperature τ)
        #   τ is decreased linearly from 0.5·max to 0.02·max
        tau = max(max_conv * 0.02,
                  max_conv * 0.5 * max(0.0, 1.0 - it / max_iters))
        # stable soft‑max
        exp_arg = (conv_full - max_conv) / tau
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the smoothed objective:
        #    grad = 2 * correlation(w, a)
        fft_w = np.fft.fft(w, n=L_corr)
        fft_a_rev = np.fft.fft(a[::-1], n=L_corr)
        corr_full = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr_full[n - 1: n - 1 + n]   # length n

        # --------------------------------------------------------------
        # 4) Adam update (projected onto the simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # 5) evaluate the new candidate (fast: re‑use max_conv)
        #    After the projection the convolution changes, recompute it.
        fft_a = np.fft.fft(a_new, n=L_conv)
        conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_score = _score_normalised(cur_max, n)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (a_new * target_sum).copy()
            print(f"Iter {it:5d}: improved score = {best_score:.6f}")

        a = a_new
        it += 1

        # --------------------------------------------------------------
        # 6) occasional random restart around the current best
        if it % 5000 == 0:
            # perturb the best known point and continue from there
            pert = best_seq / target_sum
            pert += np.random.randn(n) * 0.003   # small Gaussian noise
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()                   # back to simplex
            a = pert
            m.fill(0.0)
            v.fill(0.0)
            print(f"Iter {it:5d}: restart with jitter")

    # ------------------------------------------------------------------
    print(f"--- Finished after {it} iterations: best score = {best_score:.6f} ---")
    return best_seq.tolist()
