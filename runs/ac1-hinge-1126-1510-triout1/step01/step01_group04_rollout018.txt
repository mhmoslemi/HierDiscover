```
import time, numpy as np
from scipy.optimize import linprog

def propose_candidate(...):
    np.random.seed(0)
    deadline = time.time() + budget_s - 10.0

    # initial sequence: maybe start from arcsine shape with length n=2000
    n = 2000
    a = _initial_sequence(n, exponent=0.5, offset=0.0)   # define as earlier
    # optionally apply random perturbation.
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print("initial score", best_score)

    total_mass = np.sum(a)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute M
        conv = np.convolve(a, a)
        M = conv.max()
        # build A_ub
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=np.float64)
        # fill zeros
        A.fill(0.0)
        # vector f = a
        for d in range(n):
            A[d:d+n, d] = a   # broadcast a (len n) into column d
        # constraints: A @ g <= M
        c = -np.ones(n)   # maximize sum => minimize -sum
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("LP failed", iteration)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break

        # rescale g to have same total mass as a
        g = g * (total_mass / sum_g)

        # convex combination
        t = 0.5
        a_new = (1 - t) * a + t * g
        # enforce nonnegativity (should hold), maybe clip.
        a_new = np.maximum(a_new, 0.0)

        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"iter {iteration}: improved score {score}")

        # accept update if improves
        if score <= best_score * 1.001:
            a = a_new
        else:
            # maybe skip or keep old a
            pass

        # optionally random jitter occasionally
        if iteration % 10 == 0:
            a += np.random.randn(n)*1e-4
            a = np.maximum(a, 0.0)
            a = a * (total_mass / np.sum(a))

    print("final best score", best_score)
    return best_seq.tolist()
```

def frank_wolfe_step(a):
    conv = np.convolve(a, a)
    M = conv.max()
    g = solve_lp_max_sum(a, M)
    # scale to have same L1
    total = a.sum()
    g = g * (total / g.sum())
    # convex combination
    t = 0.5
    a_new = (1 - t) * a + t * g
    # ensure nonneg
    a_new = np.maximum(a_new, 0)
    return a_new, M

"""
Propose a low‑score step function.

The algorithm works in three stages:

1. **Coarse shape search** – try power‑law sequences
      a[i] = (i+1+δ)^(-p)
   for several lengths, exponents p and offsets δ.
   The best candidate (lowest value of ``evaluate_sequence``) is kept.

2. **Frank‑Wolfe style improvement** – given a candidate ``a`` solve the
   linear program

        max   Σ g_i
        s.t.  (a * g)_k ≤ M   for all k
              g_i ≥ 0

   where `M = max(a*a)`.  The solution `g` moves mass towards indices where
   `a` is small, thereby reducing the maximal self‑convolution.
   A convex combination `a ← (1‑t) a + t·g` (several values of `t` are tried)
   is performed, always keeping the ℓ₁‑norm fixed.

3. **Fine‑grained projected‑gradient polishing** – a few hundred cheap
   projected‑gradient steps are applied to the result of stage 2 to
   escape any remaining plateaus.

All steps respect the 1000 s time budget, use only NumPy/SciPy and never
touch the file system.  The best sequence found is returned as a plain
Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse


def _power_sequence(length: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Create a non‑negative power‑law sequence a[i] = (i+1+offset)^(-exponent)."""
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _build_constraint_matrices(n: int):
    """
    Pre‑compute row / column index arrays for the convolution matrix.
    For any weight vector `a`, the matrix A with entries
        A[k, j] = a_{k‑j}   (if 0 ≤ k‑j < n, else 0)
    satisfies (A @ g)_k = (a * g)[k].
    The row/col pattern is independent of `a`; only the data values change.
    """
    nnz = n * n
    row = np.empty(nnz, dtype=np.int32)
    col = np.empty(nnz, dtype=np.int32)
    pos = 0
    for j in range(n):
        row[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        col[pos:pos + n] = j
        pos += n
    return row, col


def _solve_lp_max_sum(a: np.ndarray, M: float, row_idx, col_idx):
    """
    Solve  max Σ g_i  s.t. (a * g)_k ≤ M   for all k,
    g_i ≥ 0.
    Returns the optimal vector g.
    """
    n = a.shape[0]
    # data for the sparse matrix = a repeated n times (one column per shift)
    data = np.tile(a, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # linprog minimizes; -Σg  ⇔ maximize Σg
    bounds = [(0, None)] * n
    # use the HiGHS interior‑point solver (fast and accepts sparse matrices)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _projected_gradient_step(a: np.ndarray, total_mass: float, step: float):
    """
    One cheap projected‑gradient update that tries to decrease the
    currently maximal convolution entry.
    """
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    n = a.shape[0]

    # sub‑gradient of the active entry (see original code)
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k_max - j
        grad[j] = 2.0 * a[i]

    # keep total mass unchanged (project onto the hyperplane Σa = total_mass)
    grad -= grad.mean()
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    cur = a_new.sum()
    if cur > 0:
        a_new *= (total_mass / cur)
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best bound of 1.503.  The function respects the time limit and prints
    intermediate scores.
    """
    import math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # -----------------------------------------------------------------
    # 1) Coarse search over length / exponent / offset
    # -----------------------------------------------------------------
    print("=== coarse shape search ===")
    best_seq = None
    best_score = float("inf")
    lengths = [800, 1500, 2000, 3000, 4000]   # moderate sizes – fast to evaluate
    exponents = [0.45, 0.48, 0.5, 0.52, 0.55]
    offsets = [0.0, 0.5, 1.0, 2.0, 5.0]

    for n in lengths:
        for p in exponents:
            for off in offsets:
                seq = _power_sequence(n, exponent=p, offset=off)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"new best   len={n:4d}  p={p:.2f}  off={off:4.1f} → {score:.6f}")
                # early exit if we already beat the target
                if best_score < 1.503:
                    break
        if best_score < 1.503:
            break

    if best_seq is None:
        # fallback – a simple uniform vector (guaranteed to be valid)
        best_seq = np.ones(1000, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("fallback to uniform, score:", best_score)

    # -----------------------------------------------------------------
    # 2) Frank‑Wolfe style LP improvement (use a modest length)
    # -----------------------------------------------------------------
    # To keep the LP cheap we cap the length at 2000.
    max_fw_len = 2000
    if best_seq.shape[0] > max_fw_len:
        a = best_seq[:max_fw_len].copy()
        print(f"truncating to {max_fw_len} entries for LP phase")
    else:
        a = best_seq.copy()
    n = a.shape[0]
    total_mass = float(np.sum(a))
    row_idx, col_idx = _build_constraint_matrices(n)

    fw_iterations = 0
    while time.time() < deadline and fw_iterations < 12:
        fw_iterations += 1
        conv = np.convolve(a, a)
        M = float(conv.max())

        # solve LP: max Σ g_i   s.t. (a * g) <= M
        try:
            g = _solve_lp_max_sum(a, M, row_idx, col_idx)
        except Exception as e:
            print("LP error:", e)
            break

        # rescale g to keep the same ℓ₁‑norm as `a`
        sum_g = float(g.sum())
        if sum_g == 0.0:
            break
        g *= (total_mass / sum_g)

        # try a few convex‑combination coefficients
        improved = False
        best_local = a
        best_local_score = best_score
        for t in (0.2, 0.5, 0.8, 1.0):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            # (scaling not needed – the combination already preserves the sum)
            score = evaluate_sequence(a_new.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local = a_new.copy()
                improved = True

        if improved:
            a = best_local
            best_score = best_local_score
            best_seq = a.copy()
            print(f"FW iter {fw_iterations:2d}: improved score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3) Fine‑grained projected‑gradient polishing
    # -----------------------------------------------------------------
    print("=== gradient polishing ===")
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        step = 0.02 / (1.0 + iteration * 0.001)
        a_new = _projected_gradient_step(a, total_mass, step)

        # occasional random jitter to avoid tiny plateaus
        if iteration % 200 == 0:
            jitter = np.random.randn(n) * 1e-3
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= (total_mass / np.sum(a_new))

        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            a = a_new
            print(f"grad iter {iteration:5d}: improved score = {best_score:.6f}")
            if best_score < 1.503:
                break
        else:
            # accept moves that do not deteriorate too much
            if score <= best_score * 1.001:
                a = a_new

    print("=== finished ===")
    print(f"final best score: {best_score:.6f}   length: {len(best_seq)}")
    return best_seq.tolist()
