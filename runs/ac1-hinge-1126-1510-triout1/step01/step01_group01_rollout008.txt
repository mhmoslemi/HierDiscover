def simplex_projection(v):
    """Project v onto the probability simplex {x>=0, sum x=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v))+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient‑guided stochastic local search for the minimal self‑convolution bound.

The program starts from a Catalan (central‑binomial) coefficient sequence
or, if the global variable ``height_sequence_1`` is present, from that
already‑known good sequence.  At each iteration it computes the convolution
a*a, finds the lag where the maximum occurs, and evaluates the “load’’ of
each index – how much that index contributes to the current maximal entry.
Mass is transferred from a high‑load index to a low‑load index; the move is
always kept on the probability simplex (non‑negative entries summing to one).
If no improvement is observed for a while the step size is reduced.
The routine runs for the allotted wall‑clock budget (default 1000 s) and
returns the best sequence found.
"""

import numpy as np
import time

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan (central‑binomial) coefficients."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_max_and_argmax(x: np.ndarray, nfft: int):
    """
    Compute the full linear convolution of x with itself using a real FFT,
    returning both the maximal value and the index where it occurs.
    """
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[: 2 * len(x) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1 - cssv) / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Stochastic local optimisation that tries to shrink the quantity

        2 * n * max( a * a )   (with Σ a_i = 1, a_i ≥ 0)

    The routine respects the 1000 s time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # 1️⃣  Choose a starting point (Catalan or the global best so far)
    # -----------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be supplied by the surrounding framework
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size == 0:
            raise Exception
        # normalise to ℓ¹‑norm 1 (the evaluation is scale‑invariant)
        if init.sum() > 0:
            a = init / init.sum()
        else:
            raise Exception
        n = a.size
        print("[init] using external sequence, length =", n)
    except Exception:
        # fall back to the classic Catalan construction
        n = 6402  # the length used in the reference code
        raw = _catalan_sequence(n)
        a = raw / raw.sum()
        print("[init] Catalan start, length =", n)

    # keep the best one seen so far
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_max_and_argmax(a, _next_pow2(2 * n - 1))[0]
    print(f"[init] initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2️⃣  Pre‑computations
    # -----------------------------------------------------------------
    nfft = _next_pow2(2 * n - 1)          # size for the FFT (fixed)
    step_frac = 0.02                      # initial fraction of source mass to move
    no_improve = 0
    iter_cnt = 0
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------
    # 3️⃣  Main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # -------- compute convolution and locate the worst lag ----------
        max_val, kmax = _conv_max_and_argmax(a, nfft)

        # -------- load vector = derivative of the active lag  ----------
        # load[i] = a[i+k] (if i+k < n) + a[i‑k] (if i‑k ≥ 0)
        load = np.zeros_like(a)
        if kmax < n:
            load[: n - kmax] += a[kmax:]          # a[i+kmax] term
            load[kmax:] += a[: n - kmax]          # a[i‑kmax] term

        # -------- select source (high load) and destination (low load) ----------
        high_thr = np.percentile(load, 80)
        low_thr = np.percentile(load, 20)
        high_idxs = np.where(load >= high_thr)[0]
        low_idxs = np.where(load <= low_thr)[0]

        if high_idxs.size == 0 or low_idxs.size == 0:
            # fall back to a completely random pair
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n)
        else:
            i_src = np.random.choice(high_idxs)
            i_dst = np.random.choice(low_idxs)

        # -------- propose a small mass transfer -------------------------
        # we move at most ``step_frac`` of the source entry
        delta = step_frac * np.random.rand() * a[i_src]
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_src] -= delta
        new_a[i_dst] += delta

        # Numerical drift can make the sum slightly off – re‑project occasionally
        if iter_cnt % 5000 == 0:
            new_a = _project_to_simplex(new_a)

        # -------- evaluate the new candidate ----------------------------
        new_max, _ = _conv_max_and_argmax(new_a, nfft)
        new_score = 2.0 * n * new_max

        # -------- accept if we improve ---------------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = new_a.copy()
            a = new_a
            no_improve = 0
            print(f"[improve] t={int(time.time()-start_time)}s "
                  f"score={best_score:.6f}  step={step_frac:.5f}")
        else:
            # keep the current point (a) unchanged, just count stagnation
            no_improve += 1

        # -------- adapt step size when we are stuck --------------------
        if no_improve >= 2000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:   # too tiny – stop the search
                print("[stop] step size vanished")
                break

        # occasional random “kick’’ to escape flat plateaus
        if iter_cnt % 10000 == 0 and iter_cnt > 0:
            jitter = (np.random.rand(n) - 0.5) * step_frac * 0.1
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a)

        iter_cnt += 1

    # -----------------------------------------------------------------
    # 4️⃣  Return the best sequence found (as a plain list)
    # -----------------------------------------------------------------
    print(f"[final] length={n}, best_score={best_score:.6f}, "
          f"iterations={iter_cnt}")
    # Clip tiny negatives caused by floating point noise (evaluate_sequence does this anyway)
    best_seq = np.maximum(best_seq, 0.0)
    # Ensure ℓ¹‑norm is exactly 1 (helps the external evaluator)
    if best_seq.sum() > 0:
        best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()
