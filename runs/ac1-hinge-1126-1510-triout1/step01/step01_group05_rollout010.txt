def simplex_projection(v):
    # Project v onto probability simplex {x >=0, sum=1}
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimisation of a non‑negative step sequence to minimise

    E(a) = 2·n·max(a * a) / (sum a)^2

The scaling of a does not affect the score, therefore we restrict the
search to the probability simplex (sum = 1).  The search consists of three
phases:

1.  Stochastic L‑BFGS on a smooth approximation of the max (log‑sum‑exp)
    with a continuation schedule for the smoothing parameter.
    Multiple random restarts (different initial shapes) are performed.

2.  A projected sub‑gradient method that minimises the true max
    directly.  The sub‑gradient of max(a*a) is obtained analytically
    (convolution with the active‑set mask) and each step is projected back
    onto the simplex.

3.  A fast greedy hill‑climb that transfers a small fraction of mass
    between two positions while keeping the sum unchanged.

All heavy computations use FFT‑based convolutions (O(n log n)).  The
procedure adapts its run time to the supplied `budget_s` (default 1000 s)
and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen for non‑negative v)
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice the part that corresponds to indices (n‑1) … (n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5          # initial step size multiplier
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (binding) indices – tolerance avoids numerical gaps
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score
