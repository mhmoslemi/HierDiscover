def project_simplex(v):
    # returns projection onto {x >=0, sum x = 1}
    # standard algorithm: sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    # Implementation of the algorithm from Duchi et al. (2008)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_conv_max(a, fft_size):
    """Return full convolution a*a (real) and its maximum."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[:2 * a.shape[0] - 1]
    max_b = b.max()
    return b, max_b

def subgradient_descent(a0, n_iters, step0, deadline):
    """Run projected sub‑gradient descent on the max self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    fft_size = 1 << ((2 * n - 1).bit_length())
    best_local_seq = a.copy()
    best_local_score = float('inf')
    sum_a = a.sum()
    # Ensure sum = 1 for scaling
    if sum_a == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / sum_a
    for it in range(1, n_iters + 1):
        if time.time() >= deadline:
            break
        # convolution and max
        b, max_b = compute_conv_max(a, fft_size)
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)
        if score < best_local_score:
            best_local_score = score
            best_local_seq = a.copy()
        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # vectorised slice could be used, but a small loop is fine
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]
        # diminishing step size
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        # projection onto the simplex
        a = project_simplex(a_new)
    return best_local_seq, best_local_score

    # If we have spare budget we can safely enlarge the sequence a little.

"""
Projected sub‑gradient optimizer for the self‑convolution peak.

The score to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The function is convex (max of convex quadratics) on the simplex
{a_i ≥ 0, Σ a_i = 1}.  Starting from a few random power‑law seeds we run a
projected sub‑gradient descent that directly reduces the active peak of the
full convolution.  The gradient of the active peak is cheap to compute
(only the slice of a that overlaps the peak), the projection onto the
simplex is performed by the O(n) algorithm of Duchi et al., and the
convolution is evaluated with an FFT (O(n log n)).  Several independent
restarts are performed until the time budget runs out; the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}.

    Implementation of the O(n) algorithm from
    Duchi, Shalev‑Shwartz, Singer, Chandra (2008).
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho (the last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """Return the full non‑circular convolution a*a and its maximal entry."""
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that minimises the peak of the
    self‑convolution.

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (will be normalised to sum 1).
    max_iters : int
        Upper bound on iteration count.
    step0 : float
        Initial step size (scaled by 1/√iter).
    deadline : float
        Unix time after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        The best sequence encountered during the descent.
    best_score : float
        Corresponding evaluation‑function value.
    """
    a = a0.copy()
    n = a.shape[0]
    # Normalise to the simplex (the exact scale does not affect the score)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_size = 1 << ((2 * n - 1).bit_length())  # next power of two

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Convolution and peak
        b, max_b = compute_conv_max(a, fft_size)

        # Score for the current a (sum(a) should be ≈1)
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional reporting
            print(f"[it {it}] new best score={score:.6f}")

        # --- sub‑gradient of the active peak ---------------------------------
        k = int(np.argmax(b))                     # index of the active peak
        grad = np.zeros_like(a)

        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)

        # gradient_i = 2 * a_{k-i} for all i contributing to the peak
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]

        # Diminishing step size (standard for sub‑gradient methods)
        step = step0 / np.sqrt(it)

        # Gradient step followed by projection onto the simplex
        a = project_simplex(a - step * grad)

    return best_seq, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.

    The routine performs several independent restarts.  Each restart starts

    and then runs the projected sub‑gradient descent described above.
    The whole search stops when the allocated time budget expires;
    the best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose a working length.  If a previous construction exists we

    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821

            n = 3000
    except Exception:
        n = 3000

    # Slightly enlarge the problem (larger n gives a lower theoretical bound)
    if n < 4000:
        n = 4000

    # --------------------------------------------------------------
    # Global best across all restarts

    global_best_seq = None
    global_best_score = float("inf")

    # Parameters controlling the inner optimisation
    max_iters_per_restart = 5000
    step0 = 0.8  # initial sub‑gradient step size

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ---- initialise a fresh starting point ----------------------
        r = np.random.rand()
        if r < 0.2:
            # Uniform start

        elif r < 0.6:
            # Random power‑law
            p = np.random.uniform(0.45, 0.70)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a0 = idx ** (-p)
            a0 /= a0.sum()
        else:
            # Perturbed copy of the best sequence obtained so far

                eps = 0.02 * np.random.randn(n)
                a0 = np.maximum(a0 + eps, 0.0)

                if s == 0:

            else:

                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)

        # ---- decide how many iterations we can afford ----------------
        time_left = deadline - time.time()
        # Empirical estimate: ~0.003 s per iteration (very conservative)
        it_budget = min(max_iters_per_restart, int(time_left / 0.003))
        if it_budget <= 0:

        # ---- run the sub‑gradient descent ---------------------------
        seq, score = subgradient_descent(a0, it_budget, step0, deadline)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if global_best_seq is None:
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()

    # Convert to plain Python list as required by the external harness.
    return global_best_seq.tolist()
