def project_simplex(v):
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a coefficient sequence that minimizes ``evaluate_sequence``.

Algorithm (high‑level)
----------------------
1. **Initialisation** – start from the best known sequence
   (``height_sequence_1`` if it exists) or from the classic
   √‑difference profile.  The vector is normalised to have L1‑norm = 1.
2. **Smooth max optimisation** – minimise the *log‑sum‑exp* of the
   autocorrelation (which is a smooth approximation of the peak).
   The gradient of this objective can be written as a simple
   correlation, so each iteration costs only a few FFT‑size
   convolutions.  After a gradient step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).
3. **Annealing / exploration** – the smoothing parameter ``lambda`` is
   increased gradually, turning the soft objective into the true peak.
   Small random redistributions of mass are occasionally applied and
   very rare uphill moves are accepted (simulated‑annealing style).
4. **Polishing** – a short random‑pair local search (move a tiny
   fraction of mass from one index to another) finishes the optimisation.
5. **Rescaling** – finally the sequence is scaled to the full
   ``[0, 1000]`` range.  The score is invariant under this scaling.

The method is deterministic except for the random seeds, runs fully
inside the allotted time budget and returns the best sequence found.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum(x_i) = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – enforce exact sum = 1
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    return w / w_sum


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute a stable soft‑max over the convolution values.
    Returns a probability vector of the same length as ``conv``.
    """
    shift = conv - np.max(conv)          # avoid overflow
    exp_vals = np.exp(lam * shift)
    total = exp_vals.sum()
    if total == 0.0:
        # fallback – uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size, dtype=np.float64)
    return exp_vals / total


def _logmax_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L(a) = log( sum_k exp(lam * (a*a)_k) )
    with respect to ``a`` (a is non‑negative, sum(a)=1).

    The gradient can be expressed as:
        grad_i = 2 * sum_j w_{i+j} * a_j
    where w are the soft‑max weights of the convolution.
    """
    conv = np.convolve(a, a)                     # length 2n‑1
    w = _softmax_weights(conv, lam)              # same length
    # correlation w with a (full) gives an array of length 3n‑2
    corr = np.correlate(w, a, mode="full")
    n = a.shape[0]
    # grad_i = 2 * corr[(n‑1)‑i]
    grad = 2.0 * corr[(n - 1) - np.arange(n)]
    return grad

    # ------------------------------------------------------------------
