def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative elements summing to 1)."""
    n = len(v)

"""
Propose a coefficient sequence that minimizes the evaluation function.

Algorithm
---------
1. **Initialization**
   - Use the best known global sequence `height_sequence_1` if it exists,
     otherwise start from the classic “square‑root‑difference’’ profile
     a_i = sqrt(i+1)−sqrt(i).  The vector is projected onto the
     probability simplex (non‑negative, unit L¹ norm).

2. **Smooth‑max gradient descent**
   - Instead of minimizing the non‑differentiable maximum of the
     autocorrelation, minimise a smooth approximation:
         B_λ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k) .
   - The gradient of B_λ w.r.t. a can be written analytically and is
     computed with two convolutions only.
   - After each gradient step the vector is projected back onto the
     simplex (Duchi et al. 2008).  The step size is adapted
     multiplicatively depending on success.
   - The temperature‑like parameter λ is increased slowly (λ≈5→200)
     so the optimisation first follows a smooth landscape and then
     concentrates on the true maximum.

3. **Escaping local minima**
   - Occasionally accept a non‑improving step with a tiny probability.
   - Random “jitter’’ moves (mass transfer between two coordinates)
     are injected regularly.
   - If no improvement is seen for many iterations the current
     iterate is re‑initialised by adding small Gaussian noise to the
     best solution so far.

4. **Final polishing**
   - A short local search (pure random mass transfers) is performed
     for the last few seconds.

5. **Return**
   - The best simplex‑projected vector found is scaled to the
     [0, 1000] range (the score is homogeneous, so scaling does not
     affect it) and returned as a plain Python list.

All heavy operations are NumPy‑based and the code stays well within
the 1000 s budget on a single CPU.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would become 0; return the uniform simplex point.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: fast score (identical to evaluate_sequence for normalised vectors)
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Compute 2·n·max(a∗a) for a with Σa = 1 (same as evaluate_sequence)."""
    n = a.shape[0]
    max_b = np.max(np.convolve(a, a))
    return 2.0 * n * max_b

# ----------------------------------------------------------------------
# Gradient of the smooth maximum of the autocorrelation
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Return the gradient of
        B_λ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k)
    w.r.t. a, assuming Σa = 1.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2n‑1
    # soft‑max weights in a numerically stable way
    scaled = lam * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)     # shape (2n‑1,)

    # Gradient g_i = 2· Σ_k w_k·a_{k‑i}  (indices out of range give 0)
    # This can be obtained by a convolution of a with the reversed weight vector.
    grad = 2.0 * np.convolve(a, w[::-1])[(2 * n - 2) - np.arange(n)]
    return grad

    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # Transfer a small fraction from i to j
        eps = min(a[i], a[i] * 0.015)
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = _project_onto_simplex(a_new)
        new_score = _score_of(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"Polish improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Scale to the [0,1000] range required by the outer judge
    # --------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Returning sequence with final score {best_score:.6f}")
    return final_seq.tolist()
