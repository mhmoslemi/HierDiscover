"""
Propose a low‑value step‑function for the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of:
  • a large “library” of deterministic initial shapes (power‑law, cosine windows,
    the previous best sequence if available and a few random Dirichlet draws);
  • a smoothed sub‑gradient descent on the simplex (sum = 1, a_i ≥ 0) using a
    soft‑max (log‑sum‑exp) approximation of the max‑convolution, momentum,
    and a simple learning‑rate schedule;
  • occasional random mass‑redistribution steps to escape shallow plateaus.

All vectors are kept on the probability simplex, which guarantees feasibility
with respect to the security checks in ``evaluate_sequence``.  The final
sequence is scaled to Σa = sqrt(2·n) (the normalisation used in the reference
code) before being returned.  The routine respects the supplied time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x) = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence but without the
# safety checks).  The function is scale‑invariant; we work with vectors that sum
# to 1.
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq) ** 2)."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.fft(seq, n=L)
    conv = np.real(np.fft.ifft(fft_seq * fft_seq))[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

    function.  The routine respects a time budget (default 1000 s) and returns
    the best sequence found as a plain Python list of floats.

    target_sum = math.sqrt(2.0 * n)   # scaling used in the reference code

    print(f"--- propose_candidate ---------------------------------------------------")
    print(f"Length n = {n}, target L1‑norm = {target_sum:.3f}")

                top[-1] = (sc, a.copy())
                top.sort(key=lambda t: t[0])
    print(f"--- initial pool : best score = {top[0][0]:.6f}")

    # --------------------------------------------------------------
    # 3. Smoothed sub‑gradient descent with momentum.
    best_seq = top[0][1].copy()
    best_score = top[0][0]

    # FFT sizes (kept constant throughout the run)
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    # Parameters for the optimiser
    base_step = 0.5               # initial step magnitude
    mu = 0.9                      # momentum coefficient
    beta_start = 0.1              # start of the smoothing parameter
    beta_max = 5.0                # maximal smoothing
    beta_inc = 1.02               # multiplicative increase per iteration

    max_iters_per_seed = 12000    # hard cap per seed (will be throttled by deadline)

    for seed_idx, (seed_score, seed_vec) in enumerate(top):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed {seed_idx+1}/{len(top)} (init score = {seed_score:.6f})")
        a = seed_vec.copy()
        v = np.zeros_like(a)               # momentum term
        beta = beta_start
        it_local = 0
        while it_local < max_iters_per_seed and time.time() < deadline:
            # ----- linear convolution a * a (real FFT) -----
            fft_a = np.fft.fft(a, n=L_conv)
            conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

            max_val = conv_full.max()

            # ----- smooth approximation of the max (log‑sum‑exp) -----
            # w_i ∝ exp(β (conv_i – max_val))  →  sum_i w_i = 1
            w = np.exp(beta * (conv_full - max_val))
            w_sum = w.sum()
            if w_sum == 0.0:               # numerical safeguard
                w_norm = np.full_like(w, 1.0 / w.size)
            else:
                w_norm = w / w_sum

            # ----- gradient of the smooth max -----
            fft_w = np.fft.fft(w_norm, n=L_corr)
            fft_rev_a = np.fft.fft(a[::-1], n=L_corr)
            corr_full = np.real(np.fft.ifft(fft_w * fft_rev_a))
            grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length‑n gradient

            # ----- step size (diminishing) -----
            step = base_step / math.sqrt(it_local + 1.0)

            # ----- momentum update and projection onto the simplex -----
            v = mu * v - step * grad
            a_new = a + v
            a_new = _project_simplex(a_new)

            # ----- evaluate true (non‑smoothed) objective -----
            cur_score = _score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(
                    f"  iter {it_local:5d}    improved score = {best_score:.6f}"
                )

            # ----- occasional tiny random mass transfer (helps escape plateaus) -----
            if (it_local + 1) % 3000 == 0:
                eps = 0.004
                i, j = np.random.choice(n, 2, replace=False)
                delta = min(eps, a_new[i])
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)

            # ----- prepare for next iteration -----
            a = a_new
            beta = min(beta * beta_inc, beta_max)
            it_local += 1

        print(
            f"  seed {seed_idx+1} finished after {it_local} iters, best score = {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 4. Scale the best sequence to the required L1‑norm and return.
    final_seq = (best_seq * target_sum).tolist()
    print(f"\n--- Finished: global best score = {best_score:.6f} (n={n})")
    return final_seq
