def solve_lp_g(f, M):
    n = len(f)
    # Build convolution constraint matrix A (size 2n-1 x n)
    # Use toeplitz
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(col, row)
    # RHS = M (same bound for all k)
    b = np.full(2*n - 1, M)
    # bounds for variables: (0, 1000)
    bounds = [(0.0, 1000.0) for _ in range(n)]
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zero vector
        return np.zeros(n)
    return res.x

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

def catalan_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2.0 * i)
    return a

def fast_convolution(x, y):
    n = x.size
    L = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[: 2*n - 1]
    return conv

def fast_score(seq):   # same as _fast_score; but we may compute from conv
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolution(seq, seq)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

def build_toeplitz_matrix(a):
    n = a.shape[0]
    m = 2 * n - 1
    diagonals = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)  # 0,1,...,n-1
    A = diags(diagonals, offsets, shape=(m, n), format='csr')
    return A

def solve_lp_direction(f, M):
    """Solve max sum(g) s.t. f * g <= M (entrywise), g >= 0, g <= 1000."""
    n = f.size
    A = build_toeplitz_matrix(f)   # shape (2n-1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)   # RHS vector
    # variable bounds
    bounds = [(0.0, 1000.0) for _ in range(n)]
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zero vector
        print("LP failed:", res.message)
        return np.zeros_like(f)
    return res.x
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start = time.time()
    deadline = start + budget_s - 5.0
    n = 2500
    f = catalan_sequence(n)
    cur_score = fast_score(f)
    best_seq = f.copy()
    best_score = cur_score

        # step 1: compute current convolution and its max

        M = conv_f.max()

        conv_fg = fast_convolution(f, g)
        conv_g = fast_convolution(g, g)
        sum_f = f.sum()
        sum_g = g.sum()

        best_t = None
        for t in t_candidates:
            h = (1.0 - t) * f + t * g
            # compute score directly via fast_score (fast enough)
            s = fast_score(h)
            if s < best_local_score:

                best_local_seq = h
                best_t = t

        # Accept if improvement
        if best_local_score < best_score - 1e-9:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"Iter {iteration:06d}: new best score {best_score:.6f} (t={best_t:.2f})")
        else:
            # Not improving; maybe we need to reduce M target
            # Let's try decreasing M a bit and recompute g
            # Decrease M by small factor and resolve
            M_new = M * 0.99
            # If M_new is too close to current max, might still produce something.
            # Build new rhs with reduced bound, i.e., lower M.
            # Solve LP with new bound
            # We'll call solve_lp_direction with M_new.
            g2 = solve_lp_direction(f, M_new)
            if np.any(g2):
                # compute mixture again with t in grid
                conv_fg2 = fast_convolution(f, g2)
                conv_g2 = fast_convolution(g2, g2)
                t_candidates2 = np.linspace(0.0, 1.0, num=21)[1:]
                best_local_score2 = np.inf
                best_local_seq2 = None
                best_t2 = None
                for t in t_candidates2:
                    h2 = (1.0 - t) * f + t * g2

                        best_local_seq2 = h2

                    print(f"Iter {iteration:06d} (reduced M): new best {best_score:.6f} t={best_t2:.2f}")

            # If still no improvement, fallback to small random perturbation
            # Randomly pick some indices and move a tiny fraction of mass
            perturb = f.copy()
            idx = np.random.choice(n, size=5, replace=False)
            perturb[idx] = np.maximum(0.0, perturb[idx] + np.random.randn(5) * 0.01)
            pert_score = fast_score(perturb)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = perturb.copy()
                f = perturb.copy()

        if best_score < 1.503:

    # Ensure final sequence non-negative and within caps (convert to python list)
    final = np.clip(best_seq, 0, 1000).tolist()
    return final

"""
Optimisation of a non‑negative step function via LP‑guided mass transfer.

The algorithm starts from the Catalan (half‑binomial) coefficients,
which have a constant self‑convolution equal to 1 and give the
well‑known bound π/2 ≃ 1.5708.  In each outer iteration we

1. compute M = max(f*f),
2. solve a linear programme that finds a non‑negative vector g of
   maximal L1‑norm under the linear constraints f*g ≤ M  (entrywise),
3. mix the current solution f with g and pick, among a small grid of
   mixing factors t∈(0,1], the mixture that yields the lowest score.
   The mixture is  h = (1‑t)·f + t·g.
4. replace f by the best mixture and repeat until the time budget is
   exhausted or the target score 1.503 is reached.

When the LP‑guided step no longer improves the score we finish with a
short hill‑climbing phase (mass transfers from early to late indices)
to squeeze a few more points.

All operations respect the constraints of `evaluate_sequence`
(non‑negative, ≤1000).  The whole search runs well within the 1000 s
budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence but without the safety checks
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a 1‑D numpy array."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    """Catalan‑type half‑binomial coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
def _conv(seq1: np.ndarray, seq2: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using FFT (full length)."""
    n = seq1.size
    L = 1 << ((2 * n - 1).bit_length())
    f1 = np.fft.rfft(seq1, n=L)
    f2 = np.fft.rfft(seq2, n=L)
    conv = np.fft.irfft(f1 * f2, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _build_toeplitz(a: np.ndarray):
    """
    Build the (2n‑1)×n matrix A such that (A @ x)[k] = (a * x)[k].
    The matrix is Toeplitz with diagonal d (d≥0) equal to a[d].
    """
    n = a.shape[0]
    # each diagonal contains n copies of the same coefficient a[d]
    diagonals = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)               # 0,1,…,n‑1
    m = 2 * n - 1
    A = diags(diagonals, offsets, shape=(m, n), format="csr")
    return A


# ----------------------------------------------------------------------
def _solve_lp_direction(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                    0 ≤ g_i ≤ 1000
    Returns the optimal vector g (may be all zeros if the LP fails).
    """
    n = f.size
    A = _build_toeplitz(f)                # (2n‑1)×n matrix
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # bounds for the variables
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # maximise Σ g_i  →  minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # on failure fall back to a zero vector – the caller will ignore it
        return np.zeros_like(f)
    return res.x


# ----------------------------------------------------------------------
def _grid_mixture(f: np.ndarray, g: np.ndarray, steps: int = 21) -> tuple:
    """
    Scan a grid of mixing factors t∈(0,1] and return the mixture with the
    lowest score together with the corresponding t.
    """
    best_score = np.inf
    best_seq = None
    best_t = None
    # we never try t=0 (no change)
    for t in np.linspace(0.0, 1.0, steps)[1:]:
        h = (1.0 - t) * f + t * g
        sc = _fast_score(h)
        if sc < best_score:
            best_score = sc
            best_seq = h
            best_t = t
    return best_seq, best_score, best_t


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    The method combines an LP‑driven mass‑transfer step (as described
    in the paper sketch) with a final short hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0           # safety margin

    # ---- 1. initialise with Catalan coefficients ---------------------------------
    n = 2500
    f = _catalan_seq(n)
    cur_score = _fast_score(f)
    best_seq = f.copy()
    best_score = cur_score
    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    iteration = 0
    # ---- 2. LP‑guided outer loop --------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # current self‑convolution and its peak
        conv_f = _conv(f, f)
        M = conv_f.max()

        # solve for a direction g that maximises Σ g_i under f*g ≤ M
        g = _solve_lp_direction(f, M)
        if not np.any(g):
            # nothing useful could be found – break out
            print("LP returned all‑zero vector, stopping LP loop.")
            break

        # try mixtures of f and g and keep the best one
        cand_seq, cand_score, cand_t = _grid_mixture(f, g, steps=21)

        # accept the mixture if it improves the best score so far
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            f = cand_seq.copy()               # continue from the improved point
            print(f"Iter {iteration:07d} – new best score {best_score:.6f} (t={cand_t:.3f})")
        else:
            # No improvement from the current LP direction.
            # Reduce the allowed convolution bound a little and try again.
            M_new = M * 0.99
            if M_new < M * 0.5:               # guard against excessive shrinking
                M_new = M * 0.99
            g2 = _solve_lp_direction(f, M_new)
            if np.any(g2):
                cand_seq2, cand_score2, cand_t2 = _grid_mixture(f, g2, steps=21)
                if cand_score2 + 1e-12 < best_score:
                    best_score = cand_score2
                    best_seq = cand_seq2.copy()
                    f = cand_seq2.copy()
                    print(f"Iter {iteration:07d} (reduced M) – new best {best_score:.6f} (t={cand_t2:.3f})")
                    continue

            # As a last resort perform a tiny random shake to escape possible plateaus
            shake = f.copy()
            idxs = np.random.randint(0, n, size=5)
            shake[idxs] = np.maximum(0.0, shake[idxs] + np.random.randn(5) * 0.01)
            shake_score = _fast_score(shake)
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                f = shake.copy()
                print(f"Iter {iteration:07d} – random shake improved to {best_score:.6f}")

        # early exit if the target is already met
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration} – score {best_score:.6f}")
            break

    # ---- 3. final small hill‑climbing phase ---------------------------------------
    # use whatever time is left for cheap local moves
    remaining = deadline - time.time()
    if remaining > 2.0:
        low_end = max(1, n // 10)          # source region (early)
        high_start = n // 2                # target region (late)
        cur_seq = best_seq.copy()
        cur_sc = best_score

        it = 0
        while time.time() < deadline:
            it += 1
            src = np.random.randint(0, low_end)
            if cur_seq[src] <= 1e-12:
                continue
            tgt = np.random.randint(high_start, n)
            frac = np.random.rand() * 0.5
            delta = cur_seq[src] * frac
            delta = min(delta, 1000.0 - cur_seq[tgt])
            if delta <= 0.0:
                continue
            trial = cur_seq.copy()
            trial[src] -= delta
            trial[tgt] += delta
            sc = _fast_score(trial)
            if sc < cur_sc - 1e-12:
                cur_seq = trial
                cur_sc = sc
                if sc < best_score:
                    best_score = sc
                    best_seq = trial.copy()
                    print(f"Hill‑climb step {it:07d} – improved to {best_score:.6f}")

    print(f"Finished after {iteration} LP iterations. Best score {best_score:.6f}")
    # convert to plain list; the evaluation harness will clip negatives / >1000 automatically
    return best_seq.tolist()
