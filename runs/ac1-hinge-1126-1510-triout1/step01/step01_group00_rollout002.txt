def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        # default: exponential decreasing
        n = 6000
        init_seq = np.exp(-np.arange(n)*0.001)  # sum ensures
    n = init_seq.shape[0]
    # Normalize to sum=1 for internal.
    a = np.maximum(init_seq, 0.0)
    a = a / a.sum()
    target_sum = math.sqrt(2.0 * n)
    # Precompute FFT length
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())
    # Prepare Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1 # maybe adapt
    t = 0
    # Softmax beta schedule
    beta = 1.0
    beta_inc = (100.0 - 1.0) / 50000  # schedule approximate 50k iterations
    best_seq = a.copy()
    best_score = _score(best_seq * target_sum)
    print(f"Initial best score: {best_score:.6f} (n={n})")
    while time.time() < deadline:
        t += 1
        # compute convolution
        fft_a = np.fft.rfft(a, n=L_conv)  # use rfft for real
        # compute conv via irfft
        conv = np.fft.irfft(fft_a * fft_a, n=L_conv)[:2*n-1]
        # compute softmax weights
        # For numerical stability: subtract max before exponent
        maxc = conv.max()
        exp_weights = np.exp(beta * (conv - maxc))
        sum_exp = exp_weights.sum()
        w = exp_weights / sum_exp
        # correlation for gradient
        # compute fft of w for longer length
        fft_w = np.fft.rfft(w, n=L_corr)
        rev_a = a[::-1]
        fft_rev_a = np.fft.rfft(rev_a, n=L_corr)
        corr = np.fft.irfft(fft_w * fft_rev_a, n=L_corr)
        grad = 2.0 * corr[n-1 : n-1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a_new = _project_simplex(a_new)
        # evaluate
        cur_score = _score(a_new * target_sum)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"Iter {t:6d}: improved score = {best_score:.6f}")

the denominator disappears:

\[
F(a)=2\,n\;\max\bigl((a*a)\bigr).
\]

Hence we can work with a **probability simplex** (non‑negative, sum = 1) and

The optimisation problem is non‑convex, so we use a **stochastic
first‑order method** with a sophisticated step‑size rule (Adam) and a
smooth approximation of the “max’’ operator. The algorithm consists of:

1. **Initialisation** – start from the best known sequence (`height_sequence_1`)

2. **Soft‑max smoothing** – replace  
   \(\max_k c_k\) (where \(c = a*a\)) by  

   \[
   \phi_\beta(c)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta c_k}\Bigr) .
   \]

   For a fixed \(\beta\) the gradient of \(\phi_\beta\) is a weighted sum of
   the gradients of the individual convolutions; the weights are a soft‑max
   distribution and are cheap to compute.

   \[
   \nabla_a \bigl(w^\top (a*a)\bigr)

# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (scaled to sum(a)=1)
def objective(a: np.ndarray, n: int, L_conv: int) -> float:
    """Return 2 * n * max(a * a) for a that already satisfies sum(a)=1."""
    # convolution via real FFT, full length = 2*n-1
    fft_a = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(fft_a * fft_a, n=L_conv)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# Gradient of the soft‑max smoothed max‑convolution
def softmax_gradient(a: np.ndarray,
                     beta: float,
                     L_corr: int) -> np.ndarray:
    """
    Compute grad = 2 * corr( w , a ), where
    w_k = exp(beta * c_k) / sum_j exp(beta * c_j)
    and c = a * a (full linear convolution).
    Returns a vector of length len(a).
    """

    # --- full convolution a*a ------------------------------------------------
    L_conv = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(fft_a * fft_a, n=L_conv)[: 2 * n - 1]

    # --- soft‑max weights ------------------------------------------------------
    max_c = conv.max()
    # shift for numerical stability
    shifted = conv - max_c

    w = w_unnorm / w_unnorm.sum()          # shape (2*n-1,)

    fft_w = np.fft.rfft(w, n=L_corr)

    corr = corr_full[n - 1 : n - 1 + n]
    grad = 2.0 * corr
    return grad
