def compute_conv(a, L=None):
    if L is None: L = 1 << ((2*len(a)-1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*len(a)-1]

"""
Propose a non‑negative step‑function (list of heights) that minimizes
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three independent phases:

1. **Broad parametric sweep** – a set of analytical families
   (power‑law, exponential, half‑sine, Gaussian, triangular) is sampled.
   Every candidate is normalised to sum = 1, its convolution is obtained
   by a single FFT and the corresponding score (2·n·max) is recorded.
   The best candidate becomes the starting point for the next phase.

2. **Smooth exponentiated‑gradient descent** – the current distribution
   `p` (sum = 1) is updated by a mirror‑descent step on the simplex:
       p ← p * exp(‑η·g),   p ← p / sum(p)
   where `g` is the (soft) sub‑gradient of the maximum convolution entry.
   The soft‑gradient is obtained by a temperature‑controlled soft‑max over
   the convolution values, i.e. `w_k = exp((c_k‑c_max)/τ)`.  A diminishing
   step size `η = η0 / sqrt(t+1)` and a cooling schedule
   `τ = τ0 / sqrt(t+1)` are used.  The update is inexpensive (three FFTs
   per iteration) and automatically respects non‑negativity and the sum
   constraint.

3. **Fine‑tuning with binary active‑set sub‑gradient** – after a fixed
   number of EG iterations the algorithm switches to the classic
   sub‑gradient scheme (binary active mask) for a few thousand extra
   steps, which often yields a small final improvement.

Throughout the search the currently best sequence and its score are kept.
The routine stops as soon as the allocated time budget (default 1000 s)
is exhausted and returns the best sequence found (any scaling is acceptable;
the routine keeps the sum = 1 normalisation).  The implementation uses only
NumPy, respects the time limit and prints occasional progress reports.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using a real‑FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * a.size - 1]
    return conv


def _corr_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of weight vector w (length 2n‑1) with a reversed.
    Result has length n and equals  (w ⋆ a)[i] = Σ_k w_{i+k} a_{k}.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_rev = np.fft.rfft(a[::-1], n=L)
    corr_full = np.fft.irfft(fft_w * fft_rev, n=L)
    n = a.size
    return corr_full[n - 1 : n - 1 + n]


def _soft_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max version of the active mask, stable under large negative exponents."""
    # conv_max is the current maximum
    cmax = conv.max()
    # shift by cmax to avoid overflow; max weight becomes 1.0
    shifted = (conv - cmax) / max(tau, 1e-12)
    # exponentiate; values far below the max become ~0
    w = np.exp(shifted)
    # ensure at least the maximum entry is kept (numerical safety)
    w[conv == cmax] = 1.0
    return w

    Search for a step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (sum is 1, any scaling works).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin to avoid being cut off
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    try:
        n = len(height_sequence_1)   # type: ignore
    except Exception:
        n = 5600                     # a reasonable default if not provided

    # Pre‑computed FFT lengths (next power of two for speed)
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Helper to compute the objective quickly (sum is 1 → score = 2·n·max)
    def _score_from_conv(conv):
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep over several analytic families
    best_seq = None
    best_score = float("inf")

    def _evaluate_candidate(a):
        """Normalise a to sum = 1, compute score, update globals."""
        nonlocal best_seq, best_score
        if a.sum() == 0:
            return
        a = a / a.sum()
        conv = _conv_fft(a, L_conv)
        sc = _score_from_conv(conv)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"New best (phase‑1) score = {best_score:.6f}")

    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    # 1) Power‑law shapes a[i] ∝ (i+1)^p
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        w = np.power(idx, p)
        _evaluate_candidate(w)
        _evaluate_candidate(w[::-1])

    # 2) Exponential decay a[i] ∝ exp(-α·i)
    for alpha in np.linspace(5e-4, 1.0e-2, 30):
        if time.time() > deadline:

        w = np.exp(-alpha * np.arange(n, dtype=np.float64))
        _evaluate_candidate(w)
        _evaluate_candidate(w[::-1])

    # 3) Half‑sine shape a[i] ∝ sin(π·(i+1)/(n+1))^q
    base = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    for q in np.linspace(0.5, 2.2, 24):
        if time.time() > deadline:

        w = np.power(base, q)
        _evaluate_candidate(w)        # symmetric already

    # 4) Gaussian bell a[i] ∝ exp( - (i-µ)² / (2σ²) )
    centre = (n - 1) / 2.0

        if time.time() > deadline:

        sigma = sigma_frac * n
        w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        _evaluate_candidate(w)

    # 5) Triangular shape (peak in the middle)
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))

    # ------------------------------------------------------------------
    # Phase 2 – smooth exponentiated‑gradient descent on the simplex
    a = best_seq / best_seq.sum()               # start with sum = 1
    print(f"--- Phase 2: exponentiated‑gradient descent (n={n}) ---")
    eta0 = 0.6                                   # base step size
    tau0 = 1.0                                   # initial temperature

        else:
            no_improve += 1

        # 2) soft active mask (temperature‑controlled)
        tau = max(tau0 / math.sqrt(it + 1), 1e-5)

        # 3) gradient of the smoothed max (correlation)
        grad = 2.0 * _corr_fft(a, w, L_corr)

        # 4) exponential‑gradient step on the simplex
        eta = eta0 / math.sqrt(it + 1)

        grad = np.clip(grad, -50, 50)
        a = a * np.exp(-eta * grad)
        # Renormalise (always stays positive)
        s = a.sum()
        if s == 0 or not np.isfinite(s):

        else:

    final_seq = best_seq.tolist()
    print(f"--- Finished: best score = {best_score:.6f} after {it + it2} iterations ---")
    return final_seq
