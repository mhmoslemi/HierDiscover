def refine(a_init, total_budget, start_time, max_iter=20000, length=None):
    a = a_init.copy()
    n = len(a)
    total_mass = 1.0  # assume sum(a) already normalized; ensure
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # hyperparameters
    step0 = 0.1
    tau0 = 0.01
    tau_min = 1e-5
    tau_decay = 0.999  # per iteration
    beta = 0.9

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter+1):
        if time.time() > start_time + total_budget - 0.5:
            break
        # compute softmax weights for conv
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # compute softmax w
        maxv = conv.max()
        exps = np.exp((conv - maxv) / tau0)  # using current tau
        w = exps / exps.sum()
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # momentum
        m = beta * m + (1 - beta) * grad
        # step size
        step = step0 / np.sqrt(it)
        # update
        a_new = a - step * m
        # projection onto simplex
        a_new = project_simplex(a_new)
        # evaluate actual objective
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"iter {it} new best {score}")
        a = a_new
        # update tau gradually maybe
        tau0 = max(tau_min, tau0 * tau_decay)
        # random jitter occasional
        if it % 500 == 0:
            jitter = np.random.randn(n) * 0.01
            a = project_simplex(a + jitter)
    return best_a, best_score
```

def project_simplex(v):
    n = len(v)
    # shift to make sum positive
    # algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Euclidean projection of vector v onto probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # shift to ensure sum positive?
    # Use algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative -> return uniform distribution
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Refinement of step‑function coefficients via a multi‑start active‑set
gradient descent.  A coarse grid of arcsine‑type seeds (varying length,
exponent and offset) is evaluated, supplemented by a few random
initialisations.  The best seed is then improved by iteratively
projecting a momentum‑gradient step onto the probability simplex.
The gradient is taken as the average sub‑gradient of the
largest few convolution entries, which approximates the sub‑gradient
of the max‑convolution objective while keeping each iteration cheap.
The loop runs until the time budget is exhausted and returns the best
sequence found.

def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Create a non‑negative decreasing sequence a[i] = (i+1+offset)^{-exponent}.
    Returns a NumPy array (not normalised).
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    # clip to the range required by evaluate_sequence (never needed here)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to numerical issues the sum may be slightly off; renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    w /= w_sum
    return w

def _active_gradient(a: np.ndarray, conv: np.ndarray,
                     top_k: int = 3) -> np.ndarray:
    """
    Compute the average sub‑gradient of the largest ``top_k`` entries of
    the autocorrelation ``conv = a*a``.  For each index k the sub‑gradient
    w.r.t. a_i is 2 * a_{k-i} whenever the pair is inside the support.
    The function returns an (n,) array.
    """
    n = a.shape[0]
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        grad[i_range] += 2.0 * a[k - i_range]
    grad /= len(ks)
    return grad

    Returns the best sequence encountered and its exact evaluated score.

    # safety margin so that we return before the global deadline

        conv = np.convolve(a, a)
        # compute gradient from the most critical convolution entries
        grad = _active_gradient(a, conv, top_k=top_k)
        # momentum update

        step = step0 / np.sqrt(iteration)
        a_new = a - step * m
        a_new = _project_simplex(a_new)
        # occasional random jitter to escape flat regions

            a_new = _project_simplex(a_new + jitter)

        # accept the step if it does not worsen the objective dramatically

        else:

            if np.random.rand() < 0.05:

"""
Refinement of a step‑function via a multi‑start active‑set projected gradient
descent.

The search proceeds in three stages:
  1) A coarse grid of analytically motivated “arcsine‑type’’ seeds
     (varying length, exponent and offset) is evaluated.
  2) A handful of random Dirichlet‑like seeds of the best length are
     tried.
  3) The best seed (and a few secondary ones) are improved with an
     active‑set gradient descent.  The gradient is the average
     sub‑gradient of the largest convolution entries, which approximates
     the sub‑gradient of the max‑convolution objective while keeping each
     iteration cheap (one convolution + O(k·n) work).  After each step the
     vector is projected onto the probability simplex (non‑negative,
     ℓ₁‑norm = 1).  Momentum (beta≈0.9) and a diminishing step size are
     used, together with occasional random jitter to escape plateaus.
The loop runs until the supplied time budget expires and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Return a decreasing sequence a[i] = (i+1+offset)**(-exponent).
    The values are already clipped to the range required by
    ``evaluate_sequence`` (never needed for the ranges we use).
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm of Duchi et al.
    (2008).  The returned vector is guaranteed to be non‑negative and sum
    to exactly one (up to floating‑point rounding).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]                  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to eliminate tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _active_gradient(a: np.ndarray, conv: np.ndarray,
                     top_k: int = 3) -> np.ndarray:
    """
    Average sub‑gradient of the ``top_k`` largest entries of the
    autocorrelation ``conv = a*a``.
    For a given convolution index k the sub‑gradient w.r.t. a_i equals
    2·a_{k‑i} whenever the pair (i,k‑i) lies inside the support.
    """
    n = a.shape[0]
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        grad[i_range] += 2.0 * a[k - i_range]
    grad /= len(ks)
    return grad


def _refine_sequence_active(a_init: np.ndarray,
                           time_budget: float,
                           top_k: int = 3,
                           beta: float = 0.9,
                           step0: float = 0.5) -> (np.ndarray, float):
    """
    Improve ``a_init`` with an active‑set projected‑gradient scheme.
    The routine stops when ``time_budget`` seconds have elapsed.
    Returns the best sequence encountered and its exact score.
    """
    start = time.time()
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float('inf')
    n = a.shape[0]
    m = np.zeros_like(a)                # momentum vector
    iteration = 0

    # run until we consume the allotted budget (leave a tiny safety margin)
    while time.time() - start < time_budget * 0.95:
        iteration += 1
        conv = np.convolve(a, a)         # full autocorrelation
        grad = _active_gradient(a, conv, top_k=top_k)

        # momentum update
        m = beta * m + (1.0 - beta) * grad
        step = step0 / np.sqrt(iteration)
        a_new = a - step * m
        a_new = _project_simplex(a_new)

        # occasional jitter to escape flat plateaus
        if iteration % 500 == 0:
            jitter = np.random.randn(n) * 1e-3
            a_new = _project_simplex(a_new + jitter)

        # exact evaluation of the objective
        try:
            cur_score = evaluate_sequence(a_new.tolist())
        except Exception:
            cur_score = float('inf')

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"iter {iteration:6d}: new best score {best_score:.6f}")

        # accept the step if it does not deteriorate dramatically
        if cur_score <= best_score * 1.02:
            a = a_new
        else:
            # with a small probability jump back to the best known point
            if np.random.rand() < 0.05:
                a = best_a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(a*a) / (sum(a)**2)

    The algorithm combines a small grid of analytically motivated seeds,
    a handful of random Dirichlet‑like seeds and an active‑set projected
    gradient descent that runs until the time budget expires.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s

    # ------------------------------------------------------------------
    # 1) Coarse grid of arcsine‑type seeds
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]
    exponents = [0.45, 0.5, 0.55, 0.6]
    offsets = [0.0, 0.5, 1.0, 2.0]

    best_seq = None
    best_score = float('inf')
    best_candidates = []                # store a few promising seeds

    print("=== grid search ===")
    for n in lengths:
        for p in exponents:
            for off in offsets:
                seq = _initial_sequence(n, exponent=p, offset=off)
                seq = seq / seq.sum()                 # enforce ℓ₁‑norm = 1
                try:
                    sc = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                # keep the overall best
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                # maintain a small pool of good seeds for possible polishing
                if len(best_candidates) < 5:
                    best_candidates.append((sc, seq.copy()))
                else:
                    worst_idx = np.argmax([c[0] for c in best_candidates])
                    if sc < best_candidates[worst_idx][0]:
                        best_candidates[worst_idx] = (sc, seq.copy())
                print(f"grid n={n:4d} p={p:.2f} off={off:.1f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Random Dirichlet (exponential) seeds of the same length as the
    #    current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.shape[0]
        print("=== random seeds ===")
        for i in range(12):
            rand_seq = np.random.exponential(scale=1.0, size=n_best)
            rand_seq = rand_seq / rand_seq.sum()
            try:
                sc = evaluate_sequence(rand_seq.tolist())
            except Exception:
                continue
            if sc < best_score:
                best_score = sc
                best_seq = rand_seq.copy()
            if len(best_candidates) < 5:
                best_candidates.append((sc, rand_seq.copy()))
            else:
                worst_idx = np.argmax([c[0] for c in best_candidates])
                if sc < best_candidates[worst_idx][0]:
                    best_candidates[worst_idx] = (sc, rand_seq.copy())
            print(f"rand {i:2d} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Use the historic solution if it is provided
    # ------------------------------------------------------------------
    try:
        seq_provided = np.array(height_sequence_1, dtype=np.float64)
        seq_provided = seq_provided / seq_provided.sum()
        sc = evaluate_sequence(seq_provided.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq_provided.copy()
        print(f"height_sequence_1 score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4) Refinement phase – active‑set projected gradient descent
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 10.0 and best_seq is not None:
        # Most of the remaining budget is spent polishing the overall best
        top_budget = time_remaining * 0.70
        print(f"=== refinement of top candidate (budget {top_budget:.1f}s) ===")
        refined_seq, refined_score = _refine_sequence_active(
            best_seq, top_budget, top_k=3)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()

        # Use whatever time is left for a quick polish of the secondary seeds
        leftover = deadline - time.time()
        if leftover > 5.0 and len(best_candidates) > 1:
            sec_budget = leftover / max(1, len(best_candidates) - 1)
            for idx, (_, cand) in enumerate(best_candidates):
                if time.time() >= deadline - 1.0:
                    break
                print(f"=== polishing secondary #{idx+1} (budget {sec_budget:.1f}s) ===")
                seq, sc = _refine_sequence_active(cand, sec_budget, top_k=5)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()

    print(f"=== finished === final best score {best_score:.6f} length {len(best_seq)}")
    # The harness expects a plain Python list
    return best_seq.tolist()
