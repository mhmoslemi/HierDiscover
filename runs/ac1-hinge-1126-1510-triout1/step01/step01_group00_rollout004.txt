"""
propose_candidate
=================

Generate a non‑negative step‑function (a list of heights) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Broad parametric sweep** – try simple families of shapes
   (power‑law, exponential, and their reversals) and a few random
   Dirichlet draws.  The best of those is taken as the initial point.

2. **Smooth‑max Adam optimisation** – minimise the supremum norm of the
   self‑convolution on the probability simplex.  The non‑smooth max
   is approximated by a soft‑max (log‑sum‑exp) with a temperature that
   grows during the run (continuation).  Gradients are obtained via FFT
   and the update uses the Adam optimiser; after each step the vector
   is projected onto the simplex.

3. **Result scaling** – the best normalised vector (sum = 1) is rescaled
   to the normalisation used by the reference code
   (`sum = sqrt(2·n)`) and returned as a plain list.

The routine respects a time budget (default 1000 s) and prints progress
information.  It can start from the previously discovered construction
`height_sequence_1` if that global variable is present, otherwise it
falls back to a length of 3000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x >= 0, sum(x) = 1 }.
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical clean‑up: renormalise to guarantee exact sum 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L_conv: int) -> float:
    """
    Compute max_i (a * a)[i] using FFT.  a must be a 1‑D float array.
    L_conv must be a power of two >= 2*len(a)-1.
    """
    fft_a = np.fft.fft(a, n=L_conv)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return the soft‑max weights w_i = exp(beta*conv_i) / sum_j exp(beta*conv_j).
    The implementation is numerically stable.
    """
    # subtract the maximum to avoid overflow
    m = conv.max()
    # avoid underflow/overflow for very large beta
    exp_vals = np.exp(beta * (conv - m))
    s = exp_vals.sum()
    if s == 0.0:  # should not happen, but guard against it
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s
