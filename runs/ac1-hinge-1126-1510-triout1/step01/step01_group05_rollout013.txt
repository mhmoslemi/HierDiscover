"""
Improved step‑function search.

The algorithm consists of three phases:

1. **Random parametric sampling** – generate families a_i ∝ (i+1)^α·(n‑i)^β,
   evaluate the fast convolution score and keep the best seed.

2. **Progressive smooth‑max optimisation** – start from the best seed (or a
   random one) and run a few L‑BFGS stages with decreasing smoothing
   parameter `lam_factor`.  The optimiser works on the soft‑max
   parametrisation `a = softmax(y)`, which guarantees non‑negativity and
   Σa = 1.

3. **Direct max‑convolution descent** – a small Adam‑style sub‑gradient
   descent that uses the true L_∞ objective (gradient of the maximum
   convolution entry).  Finally a short greedy hill‑climb cleans up the
   result.

The whole search is time‑bounded (default 1000 s).  It keeps the best
sequence seen and returns it as a plain list of floats for the official
evaluator.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic helpers (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns an array of length 2*n‑1 (the same as np.convolve(a, a)).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a).  The vector `a` is assumed to sum to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max (λ·log Σexp) objective and its gradient wrt the
    unconstrained logits `y`.  The sequence a = softmax(y) lives on the
    probability simplex.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z               # ∂smooth_max/∂conv_i
    # (a reversed) * w  convolution
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Hill‑climb (greedy move of a random fraction of mass)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Greedy local search: move a small random fraction of the mass from one
    coordinate to another while keeping the sum exactly 1.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Simple parametric family: a_i ∝ (i+1)^α·(n‑i)^β (α,β ∈ ℝ)
# ----------------------------------------------------------------------
def generate_powerlaw(alpha: float, beta: float, n: int) -> np.ndarray:
    """Return a probability vector defined by the two exponents."""
    i = np.arange(n, dtype=np.float64)
    # use log‑space to avoid overflow
    log_vals = alpha * np.log(i + 1.0) + beta * np.log(n - i)
    log_vals -= np.max(log_vals)               # shift for stability
    a = np.exp(log_vals)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent on the true L_∞ objective
# ----------------------------------------------------------------------
def adam_maximize(a_start: np.ndarray,
                  n: int,
                  fft_len: int,
                  deadline: float,
                  max_iter: int = 3000,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> (np.ndarray, float):
    """
    Adam optimisation of the maximum convolution entry.
    Works directly on the logits `y = log(a)` (soft‑max ensures feasibility).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        a = softmax(y)

        conv = conv_fft(a, fft_len, n)
        k_max = int(conv.argmax())
        # gradient of the max entry (sub‑gradient)
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        valid = (j_idx >= 0) & (j_idx < n)
        g_a = np.zeros(n)
        g_a[valid] = 2.0 * a[j_idx[valid]]

        # back‑prop through soft‑max
        mean_g = np.dot(a, g_a)
        grad_y = a * (g_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y = y - lr * (m_hat / (np.sqrt(v_hat) + eps))

        # keep the best seen vector
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to run several L‑BFGS stages with decreasing λ‑factor
# ----------------------------------------------------------------------
def multi_stage_lbfgs(y0: np.ndarray, n: int,
                      fft_len_conv: int, fft_len_g: int,
                      lam_factors: list) -> np.ndarray:
    """Run L‑BFGS repeatedly, each time with a smaller smoothing factor."""
    y = y0
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The algorithm mixes random parametric seeds, progressive L‑BFGS,
    a dedicated Adam max‑descent and a final greedy hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # problem size – a bit larger than the previous 2000 for extra flexibility
    n = 4000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise with a reasonable baseline
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    # if a previous construction is available, try it as a seed
    try:
        # pylint: disable=all
        global height_sequence_1
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size > 0:
            if prev.size < n:
                tmp = np.zeros(n)
                tmp[:prev.size] = prev
                prev = tmp
            else:
                prev = prev[:n]
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = np.full(n, 1.0 / n)
            else:
                prev = prev / prev.sum()
            best_a = prev.copy()
            best_score = fast_score(best_a, fft_len_conv, n)
            print(f"[init] loaded height_sequence_1 → score {best_score:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed)
    iteration = 0

    # ------------------------------------------------------------------
    # Phase 1 – random parametric sampling (fast, cheap)
    # ------------------------------------------------------------------
    samplings = 25000  # adjustable; fits well inside the time budget
    for _ in range(samplings):
        if time.time() >= deadline:
            break
        alpha = rng.uniform(-3.0, 3.0)
        beta = rng.uniform(-3.0, 3.0)
        a_candidate = generate_powerlaw(alpha, beta, n)
        score = fast_score(a_candidate, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            print(f"[sample] α={alpha:.2f}, β={beta:.2f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – iterative refinement (L‑BFGS + Adam + hill‑climb)
    # ------------------------------------------------------------------
    # smoothing schedule for L‑BFGS (coarse → fine)
    lam_schedule = [0.50, 0.25, 0.12, 0.06, 0.03]

    while time.time() < deadline:
        # ----- choose an initial point for this outer iteration -----
        mode = rng.choice(['uniform', 'random', 'exp', 'edge', 'prev'])
        if mode == 'uniform':
            y0 = np.log(np.full(n, 1.0 / n) + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # 'prev' – start from the current best
            y0 = np.log(best_a + 1e-12)

        # ----- smooth‑max optimisation (multi‑stage L‑BFGS) -----
        y_opt = multi_stage_lbfgs(y0, n, fft_len_conv, fft_len_g, lam_schedule)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  L‑BFGS score = {best_score:.6f}")

        # ----- direct max‑conv descent with Adam (short budget) -----
        adam_deadline = min(deadline, time.time() + 0.7)
        a_adam, score_adam = adam_maximize(best_a, n, fft_len_conv,
                                           adam_deadline, max_iter=4000)
        if score_adam < best_score:
            best_score = score_adam
            best_a = a_adam.copy()
            print(f"[{iteration}] Adam refinement → {best_score:.6f}")

        # ----- final short greedy hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=5000)

        iteration += 1

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (clamping etc.)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
