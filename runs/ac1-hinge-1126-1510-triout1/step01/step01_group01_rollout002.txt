"""
propose_candidate
=================

A two‑stage optimizer for the “step‑function” score.

*Stage 1 – global search*  
Generate a family of power‑law sequences  
  a_i ∝ (i+δ)^(-β)     (β∈[0.3,0.9], δ∈[0.1,2.0])  
and a Catalan reference.  Each candidate is normalized to ℓ¹‑norm = 1 and scored
with a fast FFT‑based convolution (score ≈ 2·n·max(conv) because the sum
is 1).  The best of these ≈ few hundred candidates becomes the starting
point.

*Stage 2 – directed hill‑climbing*  
From the current best sequence we look at the index `t` where the
self‑convolution attains its maximum.  The contribution of entry `i`
to this peak is `a_i·a_{t‑i}`; moving mass away from the largest such
contributor reduces the peak most efficiently.  A small amount `δ`
is transferred from a source index `src` (largest contribution) to a
random destination `dst`.  If the move improves the score it is kept,
otherwise it is rejected.  The step size is halved after many
rejections, and the loop stops when time expires or the step becomes
vanishingly small.

The whole routine runs for the allotted time (≈ 1000 s) and returns the
best sequence found.  All numbers are non‑negative, sum ≈ 1, and comfortably
below the 1000‑clip used by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """First *n* Catalan‑type coefficients:  C_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _self_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using a real FFT (full length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return conv


def _score(seq: np.ndarray) -> float:
    """
    Return the evaluation score for a *normalized* sequence.

    For a sequence whose ℓ¹‑norm is (almost) 1 the exact value is
        2·n·max(conv) / (sum²)
    The function computes the denominator explicitly to be safe.
    """
    s = seq.sum()
    if s <= 0.0:
        return np.inf
    conv = _self_convolution(seq)
    return float(2.0 * seq.shape[0] * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step‑function.
    The function runs for at most *budget_s* seconds and returns the best
    sequence (as a Python list) found during the search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # --------------------------------------------------------------
    # 1. Choose a convenient length.
    #    A power of two makes the FFT cheap; 8192 ≈ 2ⁱ⁺¹ is a good trade‑off.
    # --------------------------------------------------------------
    n = 8192

    # --------------------------------------------------------------
    # 2. Global (grid) search over a simple parametric family.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a) Catalan reference (the previous state‑of‑the‑art)
    cat_raw = _catalan_sequence(n)
    cat_seq = cat_raw / cat_raw.sum()
    sc = _score(cat_seq)
    best_seq = cat_seq.copy()
    best_score = sc
    print(f"[init] Catalan length={n} score={best_score:.6f}")

    # b) Power‑law grid (β controls decay, δ shifts the origin)
    betas = np.linspace(0.30, 0.90, 31)          # 0.30, 0.32, …, 0.90
    deltas = np.linspace(0.10, 2.00, 10)        # 0.10, 0.31, …, 2.0
    idx = np.arange(n, dtype=np.float64)

    for beta in betas:
        # pre‑compute i^{-beta} for the current beta (delta handled later)
        base = np.power(idx + 1.0, -beta)       # will be shifted later
        for delta in deltas:
            a = np.power(idx + delta, -beta)
            seq = a / a.sum()
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[grid] β={beta:.3f} δ={delta:.2f} → score={best_score:.6f}")

    # c) Optional mixture of the best power‑law with Catalan (helps fine‑tune)
    #    We try a few mixing ratios after the grid has finished.
    if best_seq is not None:
        for lam in (0.2, 0.5, 0.8):
            mix = lam * best_seq + (1.0 - lam) * cat_seq
            mix /= mix.sum()
            sc = _score(mix)
            if sc < best_score:
                best_score = sc
                best_seq = mix.copy()
                print(f"[mix] λ={lam:.1f} → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Directed stochastic hill climbing.
    #    Moves are biased to reduce the (currently) maximal convolution entry.
    # --------------------------------------------------------------
    step_frac = 0.04          # initial fraction of a source entry to move
    no_improve = 0
    deadline = start_time + budget_s - 5.0   # small safety margin

    while time.time() < deadline:
        # current convolution and index of its maximum
        conv = _self_convolution(best_seq)
        t = int(conv.argmax())

        # indices that actually contribute to conv[t]
        i_min = max(0, t - (n - 1))
        i_max = min(n - 1, t)
        idxs = np.arange(i_min, i_max + 1, dtype=np.int64)
        if idxs.size == 0:
            break

        # contribution of each index to the peak (a_i * a_{t‑i})
        contrib = best_seq[idxs] * best_seq[t - idxs]

        # source = index with *largest* contribution to the peak
        src = idxs[int(contrib.argmax())]

        # destination – pick a random index that is *not* the source.
        dst = np.random.randint(0, n)
        while dst == src:
            dst = np.random.randint(0, n)

        # amount of mass to move (small fraction of the source entry)
        delta = step_frac * np.random.rand() * best_seq[src]
        if delta <= 0.0 or best_seq[src] < delta:
            continue

        new_seq = best_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # everything stays non‑negative; the total sum is unchanged.
        sc = _score(new_seq)

        if sc < best_score:
            best_score = sc
            best_seq = new_seq
            no_improve = 0
            elapsed = int(time.time() - start_time)
            print(f"[improve] t={elapsed}s src={src}→dst={dst} "
                  f"Δ={delta:.2e} score={best_score:.6f}")
        else:
            no_improve += 1

        # adapt step size if we stagnate
        if no_improve >= 2500:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-7:          # nothing left to move – stop early
                break

    # --------------------------------------------------------------
    # 4. Final verification with the official evaluator.
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] length={n} evaluate_sequence→{final_score:.6f}")

    # Return the sequence as a plain Python list (required by the framework)
    return best_seq.tolist()
