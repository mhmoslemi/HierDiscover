def correlation_fft(w, a):
    L = 1 << ((len(w) + len(a) -1).bit_length())
    # but we can reuse L_corr but computing each time cheap enough.
    fft_w = np.fft.fft(w, n=L)
    fft_a_rev = np.fft.fft(a[::-1], n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # return slice from len(a)-1 to len(a)-1+len(a) => length n
    start = len(a) - 1
    return corr[start:start+len(a)]

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Multiple starts** – a handful of deterministic shapes (power‑law,
   sine‑squared, reversed versions) together with a few random Dirichlet
   draws are generated for several candidate lengths (the length of the
   provided ``height_sequence_1`` and a couple of larger multiples).

2. **Exponentiated‑gradient descent on the simplex** – each start is
   optimised by an exponential‑gradient (mirror‑descent) update that stays
   inside the probability simplex.  The gradient of the *soft‑max* of the
   self‑convolution is used:
        w_k = exp(β·(conv_k – max(conv)))         (β grows during optimisation)
        ∇M(a) = 2·corr(w, a)                     (correlation = a ⊛ w)
   The update is  a ← a·exp(−η·∇M)  followed by renormalisation.
   The step size η decays as 0.5/√t.

3. **Best‑ever tracking & final polishing** – the best sequence (scaled so
   that Σa = √(2 n) as required by the evaluator) is kept across all starts
   and lengths.  If time permits a final polish on the current best is
   performed.

The whole search respects the supplied time budget (default 1000 s) and
does not perform any file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence)
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2).  seq is a 1‑D array."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.fft(seq, n=L)
    conv = np.real(np.fft.ifft(fft_seq * fft_seq))[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    # denominator never zero because we keep sum > 0
    return 2.0 * n * max_b / (sum_a * sum_a)

# Core optimisation routine for a given length and initial vector
def _optimize_length(
    n: int,
    init_a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the simplex.
    Returns (best_a_scaled, best_score) where best_a_scaled already satisfies
    Σa = target_sum.
    """
    # Pre‑compute FFT sizes
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    # Helper: correlation of a (len n) with a weight vector w (len 2n‑1)
    def _corr_weight(a_vec: np.ndarray, w_vec: np.ndarray) -> np.ndarray:
        fft_w = np.fft.fft(w_vec, n=L_corr)
        fft_a_rev = np.fft.fft(a_vec[::-1], n=L_corr)
        corr_full = np.real(np.fft.ifft(fft_w * fft_a_rev))
        return 2.0 * corr_full[n - 1 : n - 1 + n]   # length n

    # Initialise
    a = np.maximum(init_a, 1e-12)
    a = a / a.sum()                     # unit‑sum (simplex)

    best_a = a.copy()
    best_score = _score(best_a * target_sum)

    beta = 10.0          # temperature for the soft‑max of the convolution
    beta_inc = 0.5       # slowly sharpen the soft‑max
    lr_base = 0.5        # step‑size prefactor

    it = 0
    while it < max_iters and time.time() < deadline:
        # ----- self‑convolution -----
        fft_a = np.fft.fft(a, n=L_conv)
        conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        M = conv_full.max()

        # ----- soft‑max weights over the convolution entries -----
        # w_k = exp(beta * (conv_k - M)), max weight = 1
        w = np.exp(beta * (conv_full - M))

        # ----- gradient of the soft‑max of max(conv) -----
        grad = _corr_weight(a, w)          # shape (n,)

        # ----- exponentiated‑gradient step (mirror descent on simplex) -----
        step = lr_base / math.sqrt(it + 1.0)
        a = a * np.exp(-step * grad)       # positivity preserved
        a = np.maximum(a, 1e-15)           # avoid exact zeros
        a = a / a.sum()                    # re‑project onto simplex

        # ----- evaluate -----
        cur_score = _score(a * target_sum)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"n={n:5d} it={it:5d} new best = {best_score:.6f}")

        # ----- schedule updates -----
        beta = min(200.0, beta + beta_inc)   # avoid overflow
        it += 1

    return best_a * target_sum, best_score
