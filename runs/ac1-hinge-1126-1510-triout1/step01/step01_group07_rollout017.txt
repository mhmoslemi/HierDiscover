def max_convolution(seq):
    n = len(seq)
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
    return conv.max()

"""

upper‑bound 1.576609.

The evaluation function is
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

Because the score is invariant under scaling, we only need to shape the

a score of 1, while a uniform sequence gives 2.  To flatten the convolution

small values in the middle (where many terms overlap).  Simple analytic

  * a_i ∝ (i+β)^{-p}                           (power‑law, decreasing)

  * a_i ∝ r^{i}                               (exponential decay)
  * a_i ∝ (i+β)^{-p} * r^{i}                   (mixed power‑law / exponential)

1.  Choose a length `n`.  If a previous construction is available we

def max_convolution(seq: np.ndarray) -> float:
    """Return max value of the full linear convolution seq * seq.

    Uses an FFT of size a power‑of‑two for speed (O(n log n))."""
    n = seq.shape[0]
    # length of the full convolution is 2*n-1; pick next power of two.
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return float(conv.max())


def generate_sequence(n: int, family: int, p: float, beta: float, r: float) -> np.ndarray:

    elif family == 1:                                 # symmetric power‑law (U‑shape)

    elif family == 2:                                 # pure exponential

        # fall back to a tiny uniform vector to avoid division by zero
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:

    return seq
