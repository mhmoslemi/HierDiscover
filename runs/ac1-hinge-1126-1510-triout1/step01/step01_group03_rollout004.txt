    n = a.shape[0]

    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)

def lp_refine(seq, t=0.2):
    n = len(seq)
    M = convolve_self(seq).max()
    # Build matrix A (2n-1) x n: A[k, j] = seq[k-j] if 0 <= k-j < n else 0.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i ranges max(0,k-(n-1)) to min(n-1,k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For each i, j = k - i
        # A[k, j] = seq[i]
        # j = k - i
        # iterate i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(seq[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n) # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    # Solve LP
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return seq
    b = res.x
    if b.sum() == 0:
        return seq
    g = b / b.sum()
    new_seq = (1 - t) * seq + t * g
    # Ensure non-negative also; projection onto simplex
    new_seq = np.maximum(new_seq, 0)
    new_seq_sum = new_seq.sum()
    if new_seq_sum == 0:
        return seq
    new_seq /= new_seq_sum
    return new_seq
```

"""
Hybrid stochastic search for a low‑value step‑function.
====================================================

The objective for a non‑negative sequence `a` of length `n` is

    F(a) = 2·n·max(conv(a,a)) / (sum(a))²

which is homogeneous, so we fix `sum(a)=1`.  The algorithm combines

* a good analytic start (the half‑order series, a power‑law and an
  exponential decay),
* cheap random moves (global multiplicative noise, pairwise mass transfers),
* targeted gradient steps that decrease the currently maximal convolution entry,
* an occasional linear‑programming refinement (maximise the total mass of a
  second vector `b` while keeping `conv(a,b) ≤ max(conv(a,a))`).  The LP step
  produces a new direction that can be mixed with the current vector.

A simulated‑annealing schedule controls the acceptance of worsening moves,
and the step sizes shrink with time.  The search runs until the allocated
budget (`budget_s` seconds) is exhausted and returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _power_law_series(n: int, p: float = 0.75) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _exponential_series(n: int, decay: float = 0.01) -> np.ndarray:
    """a_i ∝ exp(-decay·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * idx)
    a /= a.sum()
    return a


def _convolve_self(a: np.ndarray) -> np.ndarray:
    """Fast self‑convolution using an FFT of length power‑of‑two."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)
    return conv[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Return the evaluation function (without the security checks)."""
    conv = _convolve_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * max_conv / (s * s)


def _global_noise_move(a: np.ndarray, sigma: float) -> np.ndarray:
    """Multiplicative Gaussian noise followed by renormalisation."""
    noisy = a * (1.0 + np.random.randn(a.size) * sigma)
    noisy = np.clip(noisy, 0.0, None)
    total = noisy.sum()
    if total == 0.0:
        return a
    return noisy / total


def _pairwise_transfer_move(a: np.ndarray, eps: float) -> np.ndarray:
    """Move a small fraction eps of the mass from one index to another."""
    n = a.size
    i, j = np.random.choice(n, 2, replace=False)
    delta = eps * a[i]
    if delta > a[i]:
        delta = a[i]
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    # re‑normalise to keep sum = 1 (numerical drift may occur)
    b_sum = b.sum()
    if b_sum == 0.0:
        return a
    return b / b_sum


def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One gradient descent step on the maximal convolution entry."""
    n = a.size
    conv = _convolve_self(a)
    k = np.argmax(conv)                     # index with the largest value
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = k - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[rev]                # ∂c_k/∂a_i = 2·a_{k-i}
    b = a - step * grad
    b = np.maximum(b, 0.0)
    b_sum = b.sum()
    if b_sum == 0.0:
        return a
    return b / b_sum


def _lp_refinement(a: np.ndarray, t: float = 0.2) -> np.ndarray:
    """
    Solve a linear program:
        maximise   sum(b)
        subject to  b ≥ 0
                    conv(a, b) ≤ M   (component‑wise)
    where M = max(conv(a, a)).  Return the convex combination
        (1‑t)·a + t·(b / sum(b)).
    """
    n = a.size
    M = _convolve_self(a).max()

    # Build the Toeplitz‑like constraint matrix:
    #   A[k, j] = a[k‑j]  if 0 ≤ k‑j < n else 0
    rows = []
    cols = []
    data = []
    for j in range(n):
        # column j corresponds to a shifted copy of `a`
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                         # minimise –sum(b)  <=> maximise sum(b)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M)

    # Use the HiGHS interior‑point solver (fast for this size)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return a

    b = res.x
    if b.sum() == 0.0:
        return a
    g = b / b.sum()
    new = (1.0 - t) * a + t * g
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        return a
    return new / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic optimisation of the step‑function bound.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.  The function returns before the deadline
        (a small safety margin is kept).

    Returns
    -------
    list[float]
        The best non‑negative sequence found (the sum is normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length and a few analytic starts
    # ------------------------------------------------------------------
    n = 1000
    starters = [
        _half_order_series(n),
        _power_law_series(n, p=0.65),
        _power_law_series(n, p=0.85),
        _exponential_series(n, decay=0.005),
        np.full(n, 1.0 / n),                     # uniform
    ]

    # pick the best analytic start
    best_seq = None
    best_score = np.inf
    for s in starters:
        sc = _score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    curr_seq = best_seq.copy()
    curr_score = best_score

    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Hyper‑parameters and schedule
    # ------------------------------------------------------------------
    initial_temp = 0.02
    final_temp = 1e-6
    total_runtime = budget_s - 2.0

    # noise amplitudes (they shrink linearly)
    sigma0, sigma_end = 0.30, 0.001
    eps0, eps_end = 0.50, 0.001          # for pairwise transfer
    step0, step_end = 0.12, 0.001        # for gradient steps
    lp_interval = 4000                   # run LP roughly every few seconds

    iteration = 0
    last_lp = 0

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        progress = min(elapsed / total_runtime, 1.0)

        # temperature for Metropolis acceptance
        temp = initial_temp * (final_temp / initial_temp) ** progress

        # ------------------------------------------------------------------
        #   Choose a move type
        # ------------------------------------------------------------------
        move_type = np.random.rand()
        if move_type < 0.4:                     # global multiplicative noise
            sigma = sigma0 * (1.0 - progress) + sigma_end
            cand = _global_noise_move(curr_seq, sigma)
        elif move_type < 0.75:                  # pairwise mass transfer
            eps = eps0 * (1.0 - progress) + eps_end
            cand = _pairwise_transfer_move(curr_seq, eps)
        else:                                   # gradient step on the worst index
            step = step0 * (1.0 - progress) + step_end
            cand = _gradient_step(curr_seq, step)

        cand_score = _score(cand)

        # ------------------------------------------------------------------
        #   Metropolis acceptance (we minimise)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - curr_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            curr_seq = cand
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{iteration:6d}] New best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        #   Occasionally invoke the LP refinement
        # ------------------------------------------------------------------
        if iteration - last_lp >= lp_interval:
            # Run the LP on the *best* sequence seen so far
            refined = _lp_refinement(best_seq, t=0.25)
            refined_score = _score(refined)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()
                curr_seq = refined
                curr_score = refined_score
                print(f"[{iteration:6d}] LP refinement improved score to {best_score:.6f}")
            last_lp = iteration

        # ------------------------------------------------------------------
        #   Periodic restart if stagnation (no improvement for many steps)
        # ------------------------------------------------------------------
        if iteration % 30000 == 0:
            # Heavy random perturbation around the current best
            heavy = _global_noise_move(best_seq, sigma=0.5)
            heavy_score = _score(heavy)
            if heavy_score < best_score:
                best_score = heavy_score
                best_seq = heavy.copy()
                curr_seq = heavy
                curr_score = heavy_score
                print(f"[{iteration:6d}] Heavy restart succeeded, score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished. Best score (official evaluator) = {final_score:.6f}")
    return best_seq.tolist()
