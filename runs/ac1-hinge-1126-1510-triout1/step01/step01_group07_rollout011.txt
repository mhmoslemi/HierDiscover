def simplex_projection(v):
    # v may have negative entries
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    # shift to ensure sum=1: we want sum(w)=1, w>=0.
    # Standard algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):

"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
1. Use the previously found sequence (if available) as a starting point,
   otherwise initialise a power‑law sequence a_i ∝ (i+1)^{-p} with p≈0.5.
   The sequence is normalised to sum = 1 (the objective is scale‑invariant).

2. Perform a projected sub‑gradient descent on the objective.
   * At the current a compute the full self‑convolution via FFT,
     obtain the index k of the maximal entry.
   * The sub‑gradient of max(conv) w.r.t. a is
        g_i = 2·a_{k‑i}   for indices i that overlap with k,
        otherwise 0.
   * Take a step a ← a – α·g and project back onto the probability simplex
     (non‑negative, sum = 1).  Projection uses the O(n) algorithm of
     Duchi et al. (2008).

3. A simple back‑tracking line‑search adapts the step size α.
   Successful steps increase α slightly, failed steps shrink α.
   Periodically a tiny random mass transfer is tried to escape possible
   plateaus.

4. The loop runs until the time budget (≈ 1000 s) is exhausted or the
   target score < 1.503 is reached.  The best sequence seen is returned.

The method directly reduces the peak of the self‑convolution, and in
practice drives the score well below the previous 1.576 value.
"""

import time
import numpy as np


def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p} (sum = 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation follows Duchi et al., ICML 2008.
    """
    v = v.astype(np.float64, copy=True)
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort v in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_score_fft(a: np.ndarray, nfft: int) -> tuple[float, float, np.ndarray]:
    """
    Compute the objective score, the maximal convolution entry, and the full
    convolution (first 2*len(a)-1 elements) using FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a never drops below 0.01 (we always keep it =1 after projection)
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return float(score), float(max_b), conv

    """
