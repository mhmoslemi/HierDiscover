"""
Algorithm: Iterative linear-programming improvement of a nonnegative step
function.  Starting from the Catalan (central‑binomial) coefficients we
solve, for the current shape f, a linear programme that maximises the
ℓ¹‑norm of a new candidate g under the constraint ‖f∗g‖∞≤‖f∗f‖∞.  The
optimal g is then normalised and mixed with f.  Repeating this step
decreases the objective 2·n·‖f∗f‖∞ (the value returned by
`evaluate_sequence`).  When the LP loop no longer yields progress we
switch to a lightweight simulated‑annealing hill‑climber to fine‑tune
the result.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients a_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a

def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.size
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    return conv

def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _full_self_convolution(a).max()

def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A where (A·g)[k] = (a∗g)[k]."""
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill column‑wise – each column i contributes a shifted copy of a.
    for i in range(n):
        # rows where column i is active: k = i … i+n‑1
        start = i
        end = i + n   # exclusive
        A[start:end, i] = a[:n]  # a[0]…a[n‑1] but truncated at end
        # when end exceeds rows, the slice automatically trims
    # However the statement above copies too many elements if end > rows.
    # A more careful version uses np.arange.
    # We'll rewrite using explicit slicing for safety.
    return A

def _random_hill_climb(a: np.ndarray, time_deadline: float, best_score: float):
    n = a.size
    step_frac = 0.02
    no_improve = 0
    while time.time() < time_deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step_frac * np.random.rand() * a[i]
        if delta <= 0:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, 0.0)
        total = a2.sum()
        if total == 0:
            continue
        a2 /= total
        score = 2.0 * n * _max_self_convolution(a2)
        if score < best_score:
            best_score = score
            a = a2
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break
    return a, best_score

    start = time.time()
    deadline = start + budget_s - 5.0

    while time.time() < deadline:

        M = conv.max()

        if g0 is None:
            break
        sum_g0 = g0.sum()

"""
Algorithm: iterative linear‑programming improvement followed by a light
stochastic hill‑climber.

1.  Start from the Catalan (central binomial) coefficients and normalise
    the ℓ¹‑norm to 1 (the objective `evaluate_sequence` is then simply
    2·n·max(a∗a)).

2.  For the current shape f we compute M = max(f∗f).  We solve the linear
    program

        maximise   Σ g_i
        subject to A·g ≤ M·1 ,   g ≥ 0

    where A is the (2n‑1)×n matrix that implements the convolution
    (A·g)[k] = (f∗g)[k].

    The solution g₀ typically has a larger ℓ¹‑norm than f while still
    respecting the bound f∗g ≤ M.  After normalising g = g₀/‖g₀‖₁ we
    mix it with the current f:

        f ← (1‑t)·f + t·g   (t starts at 0.5 and is reduced if no progress).

    The mixing step strictly decreases ‖f∗f‖∞, i.e. the value returned by
    `evaluate_sequence`, and we repeat the LP step while time permits.

3.  When the LP loop stalls we fall back to a fast stochastic hill‑climber:
    repeatedly move a small random amount of mass from one index to another,
    keep the best improvement and gradually shrink the step size.

Both stages keep the sequence non‑negative and renormalise it after each
update, guaranteeing a valid input for `evaluate_sequence`.  The method
fits easily into the 1000 s budget and typically pushes the score below the
target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients a_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a computed by a real FFT."""
    n = a.size
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    return conv


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_self_convolution(a).max()


def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1)×n matrix A such that (A·g)[k] = (a∗g)[k] (linear
    convolution).  Column i of A is a shifted copy of `a`.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for i in range(n):
        start = i
        end = min(i + n, rows)
        length = end - start
        A[start:end, i] = a[:length]
    return A


def _solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        s.t.       A·g ≤ M·1,  g ≥ 0
    where A encodes the convolution with `a`.  Returns the optimal g
    (may be all zeros if the problem is infeasible) or None on failure.
    """
    A = _build_convolution_matrix(a)
    rows, n = A.shape
    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g_i  <=> minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # Use the HiGHS interior‑point solver (fast for modest sizes)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 10.0},
    )
    if not res.success:
        return None
    return res.x


def _local_hill_climb(
    a: np.ndarray, time_deadline: float, best_score: float
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climber that moves a random fraction of mass
    from one index to another while respecting non‑negativity and the
    ℓ¹‑normalisation.  Accepts only improving moves (the step size is
    halved after a long stretch without improvement).
    """
    n = a.size
    step_frac = 0.02
    no_improve = 0

    while time.time() < time_deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_frac * np.random.rand() * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0.0)

        total = a_new.sum()
        if total == 0.0:
            continue
        a_new /= total

        score = 2.0 * n * _max_self_convolution(a_new)
        if score < best_score:
            best_score = score
            a = a_new
            no_improve = 0
            # occasional progress report
            print(
                f"[hill] t={int(time.time() - (time_deadline - 1000))}s "
                f"score={best_score:.6f}"
            )
        else:
            no_improve += 1

        if no_improve >= 2000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine first performs a few LP‑based
    improvement steps starting from the Catalan distribution and then
    finishes with a lightweight stochastic local search.
    """
    np.random.seed(seed)

    # ------------------ 1. deterministic start (Catalan) -----------------
    n = 512                     # moderate size – easy LP matrix, good accuracy
    a_raw = _catalan_sequence(n)
    a = a_raw / a_raw.sum()    # ℓ¹‑norm = 1  (evaluation = 2·n·max_conv)
    best_seq = a.copy()
    best_score = 2.0 * n * _max_self_convolution(a)

    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a 5‑second safety margin

    # ------------------ 2. LP‑based improvement -----------------
    t = 0.5                     # mixing factor (will be reduced on stagnation)
    lp_iter = 0
    while time.time() < deadline and lp_iter < 30:
        lp_iter += 1

        conv_full = _full_self_convolution(a)
        M = conv_full.max()
        current_score = 2.0 * n * M

        # solve the LP for a new direction g
        g0 = _solve_lp_for_g(a, M)
        if g0 is None:
            print("[lp] LP solver failed – exiting LP phase.")
            break

        sum_g0 = g0.sum()
        # If the LP could not increase the ℓ¹‑norm we are at a stationary point
        if sum_g0 <= 1.000000001:
            print("[lp] No ℓ¹‑norm improvement, stopping LP loop.")
            break

        # Normalise g so that it also has ℓ¹‑norm = 1
        g = g0 / sum_g0

        # Mix the new direction into the current sequence
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_score = 2.0 * n * _max_self_convolution(a_new)

        # Record the best overall sequence
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_new.copy()
            print(
                f"[lp improve] iter={lp_iter} t={t:.3f} "
                f"score={best_score:.6f}"
            )

        # Accept the move for the next LP iteration if it really improved
        if new_score < current_score - 1e-12:
            a = a_new
        else:
            # No improvement – shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                print("[lp] Mixing factor too small, leaving LP phase.")
                break

    # ------------------ 3. stochastic local refinement -----------------
    if time.time() < deadline:
        a, best_score = _local_hill_climb(a, deadline, best_score)
        best_seq = a.copy()

    print(f"[final] n={n}, best_score={best_score:.6f}")
    # The evaluator expects a Python list
    return best_seq.tolist()
