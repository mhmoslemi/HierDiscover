def compute_grad_active(act, a):
    # act is binary mask of length 2*n-1
    n = a.shape[0]
    grad = np.zeros_like(a)
    active_idxs = np.where(act > 0)[0]
    for k in active_idxs:
        # shift: for i in [0,n-1], we need a_{k-i} when 0 <= k-i < n
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        # for i from i_low to i_high: associate a_{k-i}
        # vectorized:
        indices_i = np.arange(i_low, i_high+1)
        indices_j = k - indices_i
        grad[indices_i] += 2.0 * a[indices_j]
    # optionally divide by len(active_idxs) i.e., average
    if len(active_idxs) > 0:
        grad /= len(active_idxs)
    return grad
```

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.5 # step size base
    best_a = a.copy()
    best_sc = _score(a)
    # Precompute L for FFT for conv: Lc = 1 << ((2*n-1).bit_length())
    Lc = 1 << ((2*n-1).bit_length())
    Lcorr = 1 << ((3*n-2).bit_length())
    beta1_pow = 1.0
    beta2_pow = 1.0
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # compute conv
        fft_a = np.fft.fft(a, n=Lc)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        max_val = conv.max()
        # tolerance for active set
        eps_active = max_val * 1e-12 + 1e-14
        act = (conv >= max_val - eps_active).astype(np.float64)

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm proceeds in three stages:

1. **Candidate generation** – For each chosen length `n` a few simple
   shapes are created (power‑law increasing / decreasing and a handful of
   random points on the simplex).  The best few of these become seeds.

2. **Local optimisation** – Every seed is refined with an ADAM‑style
   projected sub‑gradient method.  The sub‑gradient of the supremum
   ‑norm of the self‑convolution is obtained from the active set
   (indices where the convolution attains its maximum).  After each
   gradient step the iterate is projected onto the probability simplex
   `{x≥0, Σx=1}`.  The method is inexpensive (FFT based) and runs for as
   many iterations as the time budget permits.

3. **Refinement** – The best sequence found so far is further improved
   by simulated‑annealing‑style random mass transfers.  The temperature
   schedule is linear and the acceptance probability follows the usual
   Metropolis rule.

All computations are performed with the vector normalised to
`Σa = 1`.  The evaluation function is scale‑invariant, so the final
sequence is finally scaled to the conventional sum `√(2 n)` before
returning it.

The implementation respects the 1000 s budget, prints progress messages,
and only uses the allowed libraries (`numpy`, `math`, `time`).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector `v` onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: evaluate the objective 2·n·max(conv(a,a)) / (sum(a)^2)
def _score(seq: np.ndarray) -> float:
    """Return the evaluation score for `seq` (no clipping performed)."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.fft(seq, n=L)
    conv = np.real(np.fft.ifft(fft_seq * fft_seq))[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    # protecting against division by zero – the calling code never reaches here
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# ADAM‑style projected sub‑gradient optimizer
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Minimise the evaluation function starting from `a0` (already on the simplex)
    using a projected ADAM update on the sub‑gradient of the supremum norm.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.8                     # base step size (tuned empirically)

    # FFT lengths – pre‑computed once for speed
    Lc = 1 << ((2 * n - 1).bit_length())
    Lcorr = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    best_sc = _score(a)

    beta1_pow = 1.0
    beta2_pow = 1.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- convolution and active set ---------------------------------
        fft_a = np.fft.fft(a, n=Lc)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_val = conv.max()
        # numerical tolerance for “active’’ indices
        eps_act = max_val * 1e-12 + 1e-14
        act = (conv >= max_val - eps_act).astype(np.float64)

        # ---- gradient (sum of shifted copies of a) ----------------------
        fft_act = np.fft.fft(act, n=Lcorr)
        fft_a_rev = np.fft.fft(a[::-1], n=Lcorr)
        corr_full = np.real(np.fft.ifft(fft_act * fft_a_rev))
        grad = 2.0 * corr_full[n - 1:n - 1 + n]   # length‑n gradient

        # ---- ADAM update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = alpha / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_simplex(a_new)

        # ---- evaluate ----------------------------------------------------
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    ADAM it {it:5d}: new best score = {best_sc:.6f}")

        a = a_new

    return best_a, best_sc

                     deadline: float,
                     max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Perform a simple Metropolis‑style annealing on the simplex.
    Returns the best vector encountered and its score.
    """

    cur_sc = _score(a)
    best_a = a.copy()
    best_sc = cur_sc

    n = a.shape[0]

    # Linear temperature schedule
    t0 = cur_sc * 1e-3               # start temperature (≈0.001 · score)
    anneal_start = time.time()
    total_time = max(deadline - anneal_start, 1e-6)

                accept = False
            else:
                prob = math.exp(-(sc_new - cur_sc) / T)
                accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                print(f"    Anneal it {it:6d}: new best score = {best_sc:.6f}")
        it += 1

    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence found as a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer environment
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore.
    try:
        default_n = len(height_sequence_1)        # type: ignore
    except Exception:
        default_n = 4000

    # Try the default length and a couple of neighbours (half / +20 %).
    lengths = [default_n,
               max(500, default_n // 2),
               min(12000, int(default_n * 1.2))]
    lengths = sorted(set(int(x) for x in lengths if x > 0))

    global_best_seq = None
    global_best_score = float("inf")
    global_best_n = None

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimisation for length n = {n} ===")

        # ------------------ Phase 1: generate seed vectors ------------------
        seeds = []
        # a) power‑law sweep (both forward and reversed)
        exponents = np.linspace(0.0, 2.0, 41)      # 0, 0.05, …, 2.0
        for p in exponents:
            if time.time() > deadline:
                break
            w = np.arange(1, n + 1, dtype=np.float64) ** p
            a_inc = w / w.sum()          # simplex normalisation (sum = 1)
            sc_inc = _score(a_inc)
            seeds.append((sc_inc, a_inc.copy()))
            a_dec = a_inc[::-1]
            sc_dec = _score(a_dec)
            seeds.append((sc_dec, a_dec.copy()))
        # b) random Dirichlet points
        for _ in range(10):
            if time.time() > deadline:
                break
            a_rand = np.random.rand(n)
            a_rand = _project_simplex(a_rand)
            sc_rand = _score(a_rand)
            seeds.append((sc_rand, a_rand.copy()))

        # keep the 5 most promising seeds
        seeds.sort(key=lambda item: item[0])
        top_seeds = [vec for _, vec in seeds[:5]]

        # ------------------ Phase 2: local ADAM optimisation ---------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            print(f"> Starting ADAM from seed {idx + 1} (score {seeds[idx][0]:.6f})")
            opt_vec, opt_score = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=25000
            )
            if opt_score < global_best_score:
                global_best_score = opt_score
                global_best_seq = opt_vec.copy()
                global_best_n = n
                print(f"*** New global best: score = {global_best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3: refinement with simulated annealing on the current best
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑annealing refinement ===")
        refined_seq, refined_score = _simulated_anneal(
            global_best_seq,
            deadline,
            max_iters=200000
        )
        if refined_score < global_best_score:
            global_best_seq = refined_seq
            global_best_score = refined_score
            print(f"*** After annealing: global best score = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Scale to the conventional sum √(2 n) before returning
    if global_best_seq is None:
        # fall back to a simple uniform vector (should never happen)
        n = default_n
        global_best_seq = np.full(n, 1.0 / n)
    else:
        n = global_best_n

    target_sum = math.sqrt(2.0 * n)
    final_seq = (global_best_seq * target_sum).tolist()

    print(f"\n--- Finished: best score = {global_best_score:.6f} for n={n} ---")
    return final_seq
