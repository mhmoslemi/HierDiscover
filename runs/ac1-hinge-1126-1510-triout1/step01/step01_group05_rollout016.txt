def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient
    w = exp_shift / Z  # shape (2n-1,)
    # compute g_i = sum_j a_j * w_{i+j} for i from 0..n-1
    # Equivalent to convolution of a_rev and w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # indices (n-1) to (n-1+n-1)
    g_vec = conv_rev[(n-1):(n-1+n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y
```

def hill_climb_combined(a, current_score, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        move_type = rng.choice([0,1])   # 0: transfer, 1: swap
        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # decide delta up to fraction of source mass
            delta = best[i] * rng.random() * 0.2   # up to 20%
            if delta <= 0:
                continue
            best[i] -= delta
            best[j] += delta
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # swap values
            best[i], best[j] = best[j], best[i]

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep change
        else:
            # revert
            if move_type == 0:
                best[i] += delta
                best[j] -= delta
            else:
                best[i], best[j] = best[j], best[i]
    return best, best_score
```

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step‑function (list of heights) that gives a low
value of the provided ``evaluate_sequence`` function.

The algorithm works on the probability simplex (the elements sum to one) by
parameterising the vector with a soft‑max of an unconstrained vector `y`.
A smooth approximation of the ∞‑norm of the self‑convolution is minimised
using a sequence of L‑BFGS optimisations with decreasing smoothing
parameter.  After each L‑BFGS run a fast greedy hill‑climber (mass transfer
and swaps) refines the candidate.  The whole search runs for the allotted
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using FFTs.
    Returns the first (2*n‑1) entries of the linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2*n*max(conv) for a sequence that
    already sums to 1.  Uses FFT based convolution.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max approximation of the maximum of the self‑convolution and its
    gradient w.r.t. the unconstrained vector `y`.

    Parameters
    ----------
    y : ndarray
        Unconstrained variables; the actual sequence is softmax(y).
    n : int
        Length of the sequence.
    fft_len_conv : int
        Length of the FFT used for the convolution a*a.
    fft_len_g : int
        Length of the FFT used for the gradient convolution.
    lam_factor : float
        Relative smoothing factor (0 < lam_factor < 1).  The smoothing width
        is lam = max(conv) * lam_factor + 1e-12.

    Returns
    -------
    f : float
        Smoothed objective = 2*n*smooth_max(conv).
    g : ndarray
        Gradient of f w.r.t. y (shape (n,)).
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth maximum of the convolution -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    # weights ∂smooth_max / ∂conv_i
    w = exp_shift / Z

    # Convolution a_rev * w (real‑valued) gives the vector g_i = Σ_j a_j w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # Relevant slice corresponds to indices (n‑1) … (n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb_combined(a: np.ndarray, current_score: float,
                       fft_len: int, n: int,
                       deadline: float, max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy improvement of a probability vector.

    Two move types are tried:
        * mass transfer – move a random fraction of mass from i to j,
        * swap – exchange the values of two coordinates.

    The sum of the vector stays exactly 1.  The move is kept only if it
    strictly reduces the `fast_score`.  The routine stops when the time
    budget expires or a maximum number of iterations is reached.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        move_type = rng.integers(0, 2)   # 0 = transfer, 1 = swap

        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move up to 20 % of the mass at i
            delta = best[i] * rng.random() * 0.20
            if delta <= 0.0:
                continue
            best[i] -= delta
            best[j] += delta
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # simple swap
            best[i], best[j] = best[j], best[i]

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert the move
            if move_type == 0:
                best[i] += delta
                best[j] -= delta
            else:
                best[i], best[j] = best[j], best[i]

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).
    **kwargs
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # safety margin for the final call to evaluate_sequence()
    deadline = time.time() + budget_s - 1.5

    # ------------------------------------------------------------------
    # Choose problem size – a few thousand works well with the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 3000          # a bit larger if we have more time
    else:
        n = 2000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidate (uniform) – gives score 2.0
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If a previous good construction is available, use it as a seed
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be a global list of floats
        from __main__ import height_sequence_1  # type: ignore
        a0 = np.array(height_sequence_1, dtype=float)
        if a0.size > 1:
            # Normalise to sum == 1 and adapt size if necessary
            a0 = a0 / a0.sum()
            if a0.size == n:
                best_a = a0.copy()
                best_score = fast_score(best_a, fft_len_conv, n)
                print(f"[init] started from height_sequence_1 score = {best_score:.6f}")
    except Exception:
        # No pre‑existing sequence – keep uniform as starter
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        # ---------- pick a random starting point for the L‑BFGS phase ----------
        mode = rng.choice(['random', 'exp', 'power', 'edge'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.010)
            a_start = np.exp(-rate * np.arange(n))
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 1.5)
            a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)
        else:   # edge‑heavy heuristic
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_start = np.full(n, a_center)
            a_start[:m_edge] = a_edge
            a_start[-m_edge:] = a_edge
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)

        # ---------- multi‑stage L‑BFGS with decreasing smoothing ----------
        y = y0.copy()
        for lam_factor in (0.30, 0.20, 0.10, 0.05):
            if time.time() >= deadline:
                break
            # small number of iterations per stage – fast but still informative
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam_factor),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'gtol': 1e-9,
                                    'disp': False})
            y = res.x

        a_opt = softmax(y)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  improved score = {best_score:.6f}")

        # ---------- local greedy refinement ----------
        # give the hill‑climber a short time slice (≈0.4 s)
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb_combined(best_a, best_score,
                                                 fft_len_conv, n,
                                                 hc_deadline,
                                                 max_iter=3000)

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
