    start = time.time()

        if s == 0:
            return v
        return v * (target_sum / s)

    # initial sequences list
    candidates = []

    # sqrt-difference
    a0 = np.sqrt(np.arange(1,n+1)) - np.sqrt(np.arange(0,n))
    a0 = normalize(a0)
    candidates.append(a0)

    # raised cosine
    b0 = np.sin(np.pi * (np.arange(n)+0.5) / n)
    b0 = np.maximum(b0, 0)
    b0 = normalize(b0)
    candidates.append(b0)

    # Uniform
    c0 = np.ones(n)
    c0 = normalize(c0)
    candidates.append(c0)

    # if height_sequence_1 exists, add its normalized version
    if 'height_sequence_1' in globals():
        d0 = np.array(height_sequence_1, dtype=float)
        d0 = normalize(d0)
        candidates.append(d0)

    # evaluate each initial, keep best
    best_seq = None
    best_score = np.inf
    for seq in candidates:
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"Initial best score {best_score}")

    # Simulated annealing loop
    # Use schedule: temperature = T0 * (1 - elapsed / total) ** alpha
    T0 = 1.0
    alpha = 2.0

    # move parameters gradually reduced
    max_eps = target_sum * 0.2   # max mass transferred
    min_eps = target_sum * 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 5.0)
        # temperature decreasing
        T = T0 * (1 - frac) ** alpha
        # epsilon scale decreasing
        eps_scale = max_eps * (1 - frac) + min_eps * frac

        # propose move
        # pick two indices i != j
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)

        # amount to move: epsilon * a_i (or uniform)
        # if a_i is zero, move a small uniform epsilon
        ai = best_seq[i]
        if ai > 0:
            delta = eps_scale * np.random.rand() * ai
        else:
            delta = eps_scale * np.random.rand()

        # ensure we don't take more than ai
        delta = min(delta, ai)

        # generate candidate:
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalize to target_sum (to correct rounding errors)
        cand = normalize(cand)

        # evaluate

            # active set: indices where conv within 1% of max

            # compute gradient contribution sums positions in active set:

            # step

            cand2 = best_seq - step * grad
            cand2 = np.maximum(cand2, 0)
            cand2 = normalize(cand2)
            cand2_score = evaluate_sequence(cand2.tolist())
            if cand2_score < best_score:
                best_seq = cand2
                best_score = cand2_score
                print(f"{time.time()-start:.1f}s gradient improve {best_score}")

    # final polishing: many small random moves
    polish_deadline = time.time() + min(3.0, deadline - time.time())
    while time.time() < polish_deadline:
        i, j = np.random.randint(0, n, size=2)

            best_score = cand_score
            print(f"Polish new best {best_score}")

    final_seq = best_seq * (1000.0 / best_seq.max()) if best_seq.max() > 0 else best_seq
    final_seq = np.clip(final_seq, 0, 1000)
    print(f"Returning best score {best_score}")
    return final_seq.tolist()

"""
Differential‑Evolution optimiser for the step‑function coefficients.

The evaluation metric is homogeneous, therefore the L¹‑norm of the
coefficients can be fixed without loss of optimality.  We normalise every
candidate to

    Σ a_i = √(2·n)   (n = length of the step function)

and keep all entries non‑negative.

The algorithm proceeds in three phases:

1. **Initialisation** – several sensible start vectors (square‑root‑difference,
   raised‑cosine, uniform, and the optional ``height_sequence_1``) are generated
   and normalised.  A small population (default 12) is built from these
   vectors with added random perturbations.

2. **Differential Evolution (DE)** – the population is evolved using the classic
   DE/rand/1/bin strategy.  After each trial vector is produced it is clipped to
   non‑negative values and re‑normalised to the target L¹‑norm.  If the trial’s
   score (``evaluate_sequence``) is better than its parent it replaces it.
   The global best is tracked and printed whenever it improves.

3. **Polishing** – a short random‑move local search (tiny mass transfers
   between two positions) is performed to fine‑tune the best sequence.

The final sequence is scaled to the full ``[0,1000]`` range (the scaling does not
affect the score because the metric is homogeneous) and returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _normalize_to_target(v: np.ndarray, target: float) -> np.ndarray:
    """Return a copy of *v* scaled so that its L¹‑norm equals *target*."""
    s = v.sum()
    if s == 0.0:
        # Avoid division by zero – return a uniform tiny vector
        return np.full_like(v, target / v.size, dtype=np.float64)
    return v * (target / s)


def _evaluate(v: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    # The external function expects a plain list of Python floats.
    return evaluate_sequence(v.tolist())


def _initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """
    Produce a small collection of promising start vectors, all normalised to
    the given L¹ target.
    """
    vecs = []

    # 1. Square‑root‑difference (known good continuous kernel)
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    vecs.append(_normalize_to_target(a, target_sum))

    # 2. Raised‑cosine shape (smooth, non‑negative)
    b = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # values in (0,1]
    b = np.maximum(b, 0.0)
    vecs.append(_normalize_to_target(b, target_sum))

    # 3. Uniform distribution (very simple, provides diversity)
    c = np.ones(n, dtype=np.float64)
    vecs.append(_normalize_to_target(c, target_sum))

    # 4. The optional previous construction, if present
    if "height_sequence_1" in globals():
        try:
            d = np.asarray(height_sequence_1, dtype=np.float64)
            if d.size == n:
                vecs.append(_normalize_to_target(d, target_sum))
        except Exception:
            pass

    return vecs


def _mutate_de(pop: np.ndarray, idx: int, F: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a DE mutant for individual *idx* using the classic DE/rand/1 scheme.
    """
    pop_size = pop.shape[0]
    # Choose three *distinct* individuals different from idx
    choices = list(range(pop_size))
    choices.remove(idx)
    a_idx, b_idx, c_idx = rng.choice(choices, size=3, replace=False)
    a, b, c = pop[a_idx], pop[b_idx], pop[c_idx]
    mutant = a + F * (b - c)
    # Enforce non‑negativity (clip) – negativity would be repaired by normalisation
    mutant = np.maximum(mutant, 0.0)
    return mutant


def _crossover_bin(target: np.ndarray, mutant: np.ndarray, CR: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Binomial crossover: each coordinate is taken from *mutant* with probability
    ``CR``; at least one coordinate (randomly chosen) is forced to be from the
    mutant to guarantee diversity.
    """
    n = target.size
    cross_mask = rng.random(n) < CR
    # Ensure at least one component from the mutant
    jrand = rng.integers(0, n)
    cross_mask[jrand] = True
    trial = np.where(cross_mask, mutant, target)
    return trial


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Differential‑Evolution based search for a coefficient sequence that
    minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000 s).  The function will stop
        a few seconds earlier to stay safely within limits.
    kwargs
        Ignored – present for compatibility with the runner.

    Returns
    -------
    list[float]
        The best coefficient sequence found (scaled to the [0,1000] range).
    """
    # ------------------------------------------------------------------
    # 0.  Basic set‑up
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we finish early enough
    deadline = start_time + budget_s - 1.0

    # Determine the length of the step function
    try:
        n = len(height_sequence_1)  # use global if it exists
    except Exception:
        n = 1542  # default fallback from the previous run

    # Target L1‑norm (the metric is homogeneous)
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  Initialise a small population
    # ------------------------------------------------------------------
    init_vectors = _initial_vectors(n, target_sum)
    pop_size = max(12, len(init_vectors) * 2)   # at least 12 individuals
    population = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    # Fill the first slots with the handcrafted vectors
    for i, vec in enumerate(init_vectors):
        population[i] = vec
        scores[i] = _evaluate(vec)

    # Fill the remaining slots with perturbed copies of the first vector
    base_vec = init_vectors[0]
    for i in range(len(init_vectors), pop_size):
        # Gaussian perturbation whose scale is a fraction of the mean entry
        noise = rng.normal(loc=0.0, scale=0.1 * base_vec.mean(), size=n)
        candidate = base_vec + noise
        candidate = np.maximum(candidate, 0.0)
        candidate = _normalize_to_target(candidate, target_sum)
        population[i] = candidate
        scores[i] = _evaluate(candidate)

    # Track the global best
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = float(scores[best_idx])
    print(f"Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Differential Evolution loop
    # ------------------------------------------------------------------
    # DE control parameters – we let them decay slowly to encourage fine‑tuning
    F = 0.8          # differential weight
    CR = 0.9         # crossover probability
    min_F = 0.3
    min_CR = 0.5

    # We will adapt F and CR linearly towards their minima as time runs out
    while time.time() < deadline:
        # adapt parameters based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 1.0)
        F = max(min_F, 0.8 * (1.0 - frac) + min_F * frac)
        CR = max(min_CR, 0.9 * (1.0 - frac) + min_CR * frac)

        # Process each individual in the population
        for i in range(pop_size):
            # ---- mutation ----
            mutant = _mutate_de(population, i, F, rng)
            mutant = _normalize_to_target(mutant, target_sum)

            # ---- crossover ----
            trial = _crossover_bin(population[i], mutant, CR, rng)
            trial = np.maximum(trial, 0.0)
            trial = _normalize_to_target(trial, target_sum)

            # ---- selection ----
            trial_score = _evaluate(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
                    print(f"{time.time() - start_time:6.1f}s  new best: {best_score:.6f}")

        # Occasionally inject fresh random individuals to keep diversity
        if rng.random() < 0.02:
            replace_idx = rng.integers(pop_size)
            # start from the current best plus a bigger random kick
            noise = rng.normal(loc=0.0, scale=0.2 * best_seq.mean(), size=n)
            candidate = best_seq + noise
            candidate = np.maximum(candidate, 0.0)
            candidate = _normalize_to_target(candidate, target_sum)
            candidate_score = _evaluate(candidate)
            if candidate_score < scores[replace_idx]:
                population[replace_idx] = candidate
                scores[replace_idx] = candidate_score
                if candidate_score < best_score:
                    best_score = candidate_score
                    best_seq = candidate.copy()
                    print(f"{time.time() - start_time:6.1f}s  diversity improve: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Polishing phase – tiny random mass transfers
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(4.0, deadline - time.time())
    while time.time() < polish_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer a very small fraction of mass from i to j
        delta = best_seq[i] * 0.001
        if delta <= 0.0:
            continue
        candidate = best_seq.copy()
        candidate[i] -= delta
        candidate[j] += delta
        candidate = _normalize_to_target(candidate, target_sum)
        cand_score = _evaluate(candidate)
        if cand_score < best_score:
            best_seq = candidate
            best_score = cand_score
            print(f"{time.time() - start_time:6.1f}s  polishing improved: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final rescaling to the [0,1000] range (does not affect the score)
    # ------------------------------------------------------------------
    max_val = best_seq.max()
    if max_val > 0.0:
        final_seq = best_seq * (1000.0 / max_val)
    else:
        final_seq = best_seq.copy()
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Returning sequence with final score {best_score:.6f}")
    return final_seq.tolist()
