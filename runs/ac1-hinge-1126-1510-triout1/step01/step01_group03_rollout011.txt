def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]

    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimisation for the self‑convolution bound.

* Start from the half‑order (central binomial) sequence which already gives
  ~π/2 ≈ 1.571.
* Perform a cheap parametric sweep (exponent‑tilt, mixing with uniform) to
  obtain a better starting point.
* Main loop (≈1000 s) consists of two complementary moves:
    – *random pair exchange*: move a small amount of mass from one index to
      another, accept if it reduces the maximal convolution entry (or with
      a Metropolis probability that cools down).
    – *sub‑gradient step*: compute the gradient of the current maximal
      convolution entry (or of all entries that are within a tiny tolerance
      of the maximum) and take a projected‑gradient step on the simplex.
* The best sequence seen so far (lowest score) is kept and printed.
"""

import time
import math
import numpy as np


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i,
    normalised so that sum(a)=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence: a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v <= 0):
        # the projection is the uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    half‑order bound.  The routine mixes a quick parametric search with a
    long‑running hill‑climbing / projected sub‑gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin (the caller will also stop if we exceed)
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) parameters
    # ------------------------------------------------------------------
    n = kwargs.get("n", 1000)          # length of the sequence
    total_runtime = deadline - start_time

    # ------------------------------------------------------------------
    # 2) initialise from the half‑order series and a tiny parametric sweep
    # ------------------------------------------------------------------
    base = _half_order_series(n)       # already normalised (sum = 1)
    best_seq = base.copy()
    best_conv = np.convolve(best_seq, best_seq)
    best_max = best_conv.max()
    best_score = 2.0 * n * best_max     # same as evaluate_sequence for sum=1

    # sweep over exponent tilts (a_i *= (i+1)^(-delta))
    for delta in np.linspace(0.0, 0.6, 13):          # 0 … 0.6 in steps of 0.05
        taper = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * taper
        cand /= cand.sum()
        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2.0 * n * m
        if score < best_score:
            best_score = score
            best_max = m
            best_seq = cand.copy()
            print(f"[init] delta={delta:.2f} → score {best_score:.6f}")

    # sweep over a simple convex combination with the uniform sequence
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 11):          # mixing proportion
        cand = (1.0 - alpha) * base + alpha * uniform
        cand /= cand.sum()
        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2.0 * n * m
        if score < best_score:
            best_score = score
            best_max = m
            best_seq = cand.copy()
            print(f"[init] mix={alpha:.2f} → score {best_score:.6f}")

    # start the iterative optimisation from the best found so far
    cur_seq = best_seq.copy()
    cur_max = best_max
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3) optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    # parameters for the stochastic move
    move_prob = 0.6                     # probability of a random pair exchange
    # temperature schedule for Metropolis acceptance (only for random moves)
    init_temp = cur_max * 0.5           # start relatively warm
    final_temp = 1e-12

    # step size for the sub‑gradient move (will be adapted)
    step_g = 0.1
    step_g_min = 1e-8
    step_g_max = 0.5

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        progress = elapsed / total_runtime
        # cool down temperature for random moves
        temp = init_temp * (final_temp / init_temp) ** progress

        # --------------------------------------------------------------
        # 3a) random pair exchange (most frequent move)
        # --------------------------------------------------------------
        if np.random.rand() < move_prob:
            # pick two distinct positions
            i, j = np.random.choice(n, 2, replace=False)
            # propose moving a small fraction of mass from i to j
            max_delta = min(cur_seq[i], 0.02)    # never move more than 2 % mass
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cand_seq = cur_seq.copy()
            cand_seq[i] -= delta
            cand_seq[j] += delta
            # no need to renormalise – sum is unchanged and non‑negativity holds
            cand_conv = np.convolve(cand_seq, cand_seq)
            cand_max = cand_conv.max()

            accept = False
            if cand_max < cur_max:
                accept = True
            else:
                # Metropolis rule (rarely accepts worse moves)
                if temp > 0:
                    prob = math.exp(-(cand_max - cur_max) / temp)
                    if np.random.rand() < prob:
                        accept = True

            if accept:
                cur_seq = cand_seq
                cur_max = cand_max
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_max = cur_max
                    best_seq = cur_seq.copy()
                    print(f"[{iteration:6d}] Random improvement score: {best_score:.6f}")

        # --------------------------------------------------------------
        # 3b) sub‑gradient step on the active set of maximal entries
        # --------------------------------------------------------------
        else:
            conv = np.convolve(cur_seq, cur_seq)
            cur_max = conv.max()
            # active indices (within a tiny tolerance of the maximum)
            tol = 1e-12
            active = np.where(conv >= cur_max - tol)[0]

            # gradient = sum_{k in active} 2 * a_{k-i}
            grad = np.zeros_like(cur_seq)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                rev = k - idx
                grad[idx] += 2.0 * cur_seq[rev]
            # average over the active set
            if len(active) > 1:
                grad /= float(len(active))

            # try a back‑tracking line search on the simplex
            improved = False
            trial_step = step_g
            for _ in range(5):
                cand = cur_seq - trial_step * grad
                cand = _project_simplex(cand)
                cand_conv = np.convolve(cand, cand)
                cand_max = cand_conv.max()
                if cand_max < cur_max:
                    # accept
                    cur_seq = cand
                    cur_max = cand_max
                    cur_score = 2.0 * n * cur_max
                    # slightly increase step size for next iteration
                    step_g = min(step_g * 1.1, step_g_max)
                    improved = True
                    break
                else:
                    trial_step *= 0.5   # shrink step and retry
            if not improved:
                # no improvement – shrink the global step size
                step_g = max(step_g * 0.7, step_g_min)
                # occasional jitter to escape shallow plateaux
                if iteration % 500 == 0:
                    sigma = 0.001
                    noise = np.random.randn(n) * sigma
                    jitter = cur_seq * (1.0 + noise)
                    jitter = np.clip(jitter, 0.0, None)
                    jitter /= jitter.sum()
                    cur_seq = _project_simplex(jitter)

            if cur_score < best_score:
                best_score = cur_score
                best_max = cur_max
                best_seq = cur_seq.copy()
                print(f"[{iteration:6d}] Sub‑gradient improvement score: {best_score:.6f}")

        # optional termination printout every few thousand iterations
        if iteration % 2500 == 0:
            print(f"[{iteration:6d}] Current best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Ensure the output is a plain Python list of floats
    return best_seq.tolist()
