"""
Propose a low‑evaluation step sequence.

The search is performed in three stages:

1. **Exploratory random restarts** – many different initial shapes are
   optimized with a smooth max‑objective using L‑BFGS‑B.  The smoothness
   parameter λ is varied (0.5 → 0.1) to encourage both global exploration
   and fine‑grained optimisation.  After each optimisation a cheap
   greedy hill‑climb (mass transfer between two positions) refines the
   candidate.

2. **Projected sub‑gradient (mirror) descent on the true max‑convolution**.
   Starting from the best candidate, the exact convolution is computed,
   the index (or indices) attaining the maximum is identified and their
   sub‑gradient is used to update the distribution via an exponentiated‑
   gradient step.  The distribution is re‑projected onto the probability
   simplex after every iteration.  The step‑size decays as 1/√t.

3. **Final polishing** – a short hill‑climb with a larger move budget.

The whole routine respects the provided time budget (default 1000 s) and
returns a list of non‑negative floats whose sum is 1 (the evaluator
rescales internally, the objective is scale‑invariant).  The algorithm
makes heavy use of FFT‑based convolutions (O(n log n)) and runs comfortably
within the allowed time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps y to a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the real part of length 2*n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return the exact objective 2·n·max(conv) for a (assumed summed to 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective  f(y)=2·n·smooth_max(a*a) and its gradient,
    where a = softmax(y).  The smoothness λ is taken as
    lam = lam_factor * max(conv) + ε.
    """
    a = softmax(y)

    # ----- convolution a*a (linear, length 2n‑1) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12   # smoothing width (never zero)

    # ----- smooth max over convolution entries -----
    shifted = conv - max_c                      # to avoid overflow
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. convolution entries -----
    w = exp_shift / Z                 # ∂smooth_max / ∂conv_i

    # ----- back‑propagate through the convolution (a * w) -----
    # g_i = Σ_j a_j * w_{i+j}  , i = 0 .. n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice needed part: indices (n‑1) … (n‑1 + n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # ----- chain rule through the soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Greedy mass‑transfer hill‑climb.
    Randomly moves a fraction (up to 5 %) of the mass from one index
    to another and keeps the move only if the objective improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # give a small safety margin for the final evaluation
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # problem size – a trade‑off between resolution and speed
    # ------------------------------------------------------------------
    n = 3000                                 # can be increased if time permits
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g    = next_pow2(3 * n - 2)      # for a * w in the smooth objective

    # ------------------------------------------------------------------
    # initialise best candidate (uniform, or a provided global variable)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined by the harness; use it if available
        init_seq = np.asarray(height_sequence_1, dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            init_seq = np.maximum(0.0, init_seq)       # ensure non‑negative
            init_seq = init_seq / init_seq.sum()       # project onto simplex
            best_a = init_seq
            best_score = fast_score(best_a, fft_len_conv, n)
            print(f"[init] using global height_sequence_1 → fast score = {best_score:.6f}")
        else:
            raise NameError
    except Exception:
        best_a = np.full(n, 1.0 / n)                 # uniform start
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[init] uniform → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1: random restarts + L‑BFGS on a smooth approximation
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 30.0:   # reserve ~30 s for later phases
        # ----------- choose an initial shape ------------
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'tri', 'U'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'tri':
            i = np.arange(n, dtype=float)
            a0 = 1.0 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
            a0 = np.clip(a0, 0, None)
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # mode == 'U'   – a simple U‑shape
            i = np.arange(n, dtype=float)
            a0 = np.sqrt(i + 1) + np.sqrt(n - i)
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # smoothness parameter – smaller values focus more on the true max
        lam_factor = rng.choice([0.5, 0.35, 0.2, 0.1])

        # ----------- run L‑BFGS on the smooth objective -------------
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 350,
                                'gtol': 1e-12,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode:5s} lam={lam_factor:.2f}  improved fast score = {best_score:.6f}")

        # ---------- local greedy refinement (hill climbing) ----------
        hc_deadline = min(deadline, time.time() + 0.6)   # short burst per iteration
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=3000)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase‑2: projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    sub_iter = 0
    while time.time() < deadline - 10.0:   # keep about 10 s for final polishing
        # compute the full convolution once
        conv = conv_fft(a, fft_len_conv, n)
        max_c = conv.max()

        # collect all indices whose value is within a tiny relative tolerance
        tol = max_c * 1e-8
        ks = np.where(conv >= max_c - tol)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in ks:
            # i runs over the overlap of a and shifted a
            i_start = max(0, k - (n - 1))
            i_end   = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k - i_vals

            # contribution a_j to ∂/∂a_i
            g[i_vals] += a[j_vals]

            # contribution a_i to ∂/∂a_j (skip diagonal where i=j)
            mask = i_vals != j_vals
            g[j_vals[mask]] += a[i_vals[mask]]

        # exponentiated‑gradient (mirror) step; step size decays as 1/√t
        step = 0.6 / np.sqrt(sub_iter + 1.0)
        a = a * np.exp(-step * g)

        # project back onto the probability simplex (simple normalisation)
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # evaluate the exact objective
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if sub_iter % 100 == 0:
                print(f"[subgrad {sub_iter:4d}] improved fast score = {best_score:.6f}")
        sub_iter += 1

    # ------------------------------------------------------------------
    # Phase‑3: final polishing hill‑climb (a few seconds)
    # ------------------------------------------------------------------
    final_deadline = time.time() + 5.0
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=5000)

    # ------------------------------------------------------------------
    # Return the result (the official evaluator will clip/normalise again)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
