def proj_simplex(v):
    # Return projection of v onto the simplex {x>=0, sum x = 1}
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate sequence that (approximately) minimises
`evaluate_sequence`.  The algorithm starts from several analytically
motivated initialisations (Catalan, arcsine‑like and semicircle‑like
distributions) and then performs a projected sub‑gradient descent on the
ℓ¹‑simplex.  At each iteration the self‑convolution is obtained with a
real FFT, the K largest convolution entries are used to form a
sub‑gradient of the `max` operator, and a small step is taken away from
the gradient.  After each step we project back onto the non‑negative
simplex by clipping negatives and renormalising.  The step size is
adaptively halved when no improvement is seen for a while.  Random
jitter is occasionally added to help escape plateaus.  The best
sequence encountered over the whole allotted budget is returned.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan (central‑binomial) coefficients
    a_k = binom(2k, k) / 4**k  (exact, not normalised)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _compute_self_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x using a real FFT.
    Length of result is 2*len(x)-1."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return conv


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {v >= 0, sum(v)=1}.
    This simple clip‑and‑renormalise works because we only need a feasible
    point, not the Euclidean projection."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to a uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine performs a simplex‑projected
    sub‑gradient descent starting from several analytically motivated
    distributions.  It runs until the time budget expires and returns the
    best sequence found.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off between
    # quality and speed (the original solution used 6402).
    # ------------------------------------------------------------------
    n = 6402

    # ------------------------------------------------------------------
    # Prepare a few plausible initial sequences.
    # ------------------------------------------------------------------
    init_sequences = []

    # 1) Catalan (central‑binomial) coefficients, ℓ¹‑normalised.
    catalan = _catalan_sequence(n)
    catalan = catalan / catalan.sum()
    init_sequences.append(catalan)

    # 2) Arcsine‑like shape a_i ∝ 1/√(i+1)
    idx = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt(idx + 1.0)
    arcsine = arcsine / arcsine.sum()
    init_sequences.append(arcsine)

    # 3) Semicircle‑like shape a_i ∝ √(i+1)·√(n‑i)
    semicircle = np.sqrt(idx + 1.0) * np.sqrt(n - idx)
    semicircle = semicircle / semicircle.sum()
    init_sequences.append(semicircle)

    # Keep track of the overall best.
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Hyper‑parameters (can be tweaked without breaking the interface).
    # ------------------------------------------------------------------
    K = 5                 # number of top convolution entries to average
    init_step = 0.02      # initial gradient step size
    max_no_improve = 2000  # stagnation threshold before step reduction

    # ------------------------------------------------------------------
    # Loop over the different starts.
    # ------------------------------------------------------------------
    for start_id, a in enumerate(init_sequences):
        # Initial convolution and score.
        conv = _compute_self_convolution_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        print(f"[init {start_id}] initial score = {cur_score:.6f}")

        step = init_step
        no_improve = 0
        iteration = 0

        # Main optimisation loop for this start.
        while time.time() < deadline:
            # --------------------------------------------------------------
            # 1) Determine a sub‑gradient of the max‑convolution.
            # --------------------------------------------------------------
            if K < conv.size:
                # indices of the K largest convolution values
                top_idx = np.argpartition(-conv, K)[:K]
            else:
                top_idx = np.arange(conv.size)

            # average shift term:   shift_j = (1/K) Σ_k a_{k−j}  (where defined)
            shift = np.zeros_like(a)
            for k in top_idx:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                if i_start <= i_end:
                    inds = np.arange(i_start, i_end + 1)
                    shift[inds] += a[k - inds]
            shift /= K

            # sub‑gradient of max(conv) with respect to a (ignoring the constant factor 2·n)
            grad = a + shift

            # --------------------------------------------------------------
            # 2) Gradient step and projection onto the simplex.
            # --------------------------------------------------------------
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new)

            # --------------------------------------------------------------
            # 3) Occasional random jitter to escape shallow plateaus.
            # --------------------------------------------------------------
            if np.random.rand() < 0.01:
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if i != j and a_new[i] > 0.0:
                    delta = step * a_new[i] * np.random.rand()
                    a_new[i] -= delta
                    a_new[j] += delta
                    a_new = _project_to_simplex(a_new)

            # --------------------------------------------------------------
            # 4) Evaluate the new candidate (fast via FFT, full score later).
            # --------------------------------------------------------------
            conv_new = _compute_self_convolution_fft(a_new)
            new_score = 2.0 * n * conv_new.max()

            # --------------------------------------------------------------
            # 5) Keep the best solution seen so far.
            # --------------------------------------------------------------
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[improve] start={start_id} t={int(time.time() - start_time)}s "
                      f"score={new_score:.6f} step={step:.5f}")
                no_improve = 0
            else:
                no_improve += 1

            # --------------------------------------------------------------
            # 6) Adaptive step‑size control.
            # --------------------------------------------------------------
            if no_improve >= max_no_improve:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    # step too tiny – stop improving this start
                    break

            # Prepare for the next iteration.
            a = a_new
            conv = conv_new
            iteration += 1

        print(f"[end start {start_id}] best_score so far = {best_score:.6f}")

        # If the global time budget has been exhausted we stop early.
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    print(f"[final] best_score = {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
