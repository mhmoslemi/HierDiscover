```python
def propose_candidate(seed: int = 123, budget_s: float = 1000):
    '''
    Search for a non‑negative sequence that minimizes `evaluate_sequence`.
    Uses a parametric family a_i ∝ (i + beta)^(-p) with random sampling
    of the exponent p∈[0.4,0.6] and offset beta∈[1e‑4, n] for several lengths.
    The best sequence found is refined with a small stochastic hill‑climbing
    that moves mass between arbitrary positions.
    '''
    import numpy as np, time

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # baseline Catalan sequence as reference
    def _catalan_sequence(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k-1] * (2.0*k)*(2.0*k-1.0) / (4.0*k*k)
        return a

    n0 = 6402
    raw = _catalan_sequence(n0)
    best_seq = raw / raw.sum()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] baseline Catalan length={n0} score={best_score:.6f}")

    # helper convex convolution max
    def max_self_convolution(x):
        n = x.shape[0]
        m = 2*n - 1
        nfft = 1 << (m-1).bit_length()
        fx = np.fft.rfft(x, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[:m]
        return conv.max()

    # Parameter search
    # Pre-define set of lengths we consider
    lengths = [4000, 6000, 8000, 10000, 12000]
    # Use best found during random exploration
    while time.time() < deadline:
        # Choose length randomly with bias toward larger (more promising)
        n = int(np.random.choice(lengths, p=np.array([0.1,0.2,0.3,0.25,0.15])))
        # Sample exponent p around 0.5
        p = np.random.normal(loc=0.5, scale=0.05)
        p = np.clip(p, 0.35, 0.65)
        # Sample offset beta on log scale
        log_beta = np.random.uniform(-4, np.log10(max(1.0, n)))
        # Actually choose beta uniform in [1e-4, n]
        beta = 10**np.random.uniform(-4, np.log10(max(1.0, n)))
        # Ensure beta > 0 to avoid division by zero
        if beta <= 0.0:
            beta = 1e-4

        i = np.arange(n, dtype=np.float64)
        raw = 1.0 / np.power(i + beta, p)
        seq = raw / raw.sum()
        # Compute score using fast conv
        conv_max = max_self_convolution(seq)
        score = 2.0 * n * conv_max  # sum is 1 by construction

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] t={int(time.time()-start)}s n={n} p={p:.4f} beta={beta:.4f} score={score:.6f}")

    # Now perform a lightweight local hill climbing on the best sequence
    # to fine‑tune individual entries.
    n = best_seq.shape[0]
    step_frac = 0.02
    no_improve = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # choose direction
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = step_frac * np.random.rand() * best_seq[src]
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # renormalise to keep sum ≈ 1
        total = new_seq.sum()
        if total <= 0:
            continue
        new_seq /= total
```

The algorithm proceeds in two phases:

   family  

        a_i ∝ (i + beta)^{-p} ,  i = 0,…,n‑1

   where the length `n`, the exponent `p` (around 0.5) and the offset
   `beta` are sampled randomly.  The sequence is normalised to unit ℓ¹‑norm,
   and the objective value is obtained quickly via an FFT based convolution
   (`max_self_convolution`).  The best candidate found so far is kept.

2. **Fine‑grained hill climbing** – starting from the best sequence,
   mass is moved between arbitrary positions (both directions are allowed)
   with a diminishing step size.  Each move is accepted only if it improves
   the objective.  This stage can further lower the score.

The method respects all security checks of `evaluate_sequence` and returns
the best list found before the time budget expires (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients: binom(2k,k)/4**k (exact)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _max_self_convolution(x: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution x*x using a real FFT.
    The result is identical to `np.convolve(x, x).max()`, but much faster.
    """
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()           # next power of two ≥ m
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def _generate_powerlaw_seq(n: int, p: float, beta: float) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n with
        a_i ∝ (i + beta)^{-p},   beta > 0 .
    The result is normalised to unit ℓ¹‑norm.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.power(idx + beta, p)
    # avoid division‑by‑zero for pathological beta (should not happen)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        # fall back to a uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / raw_sum
