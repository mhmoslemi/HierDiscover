    Starts from the classical a[i] = 1/sqrt(i+1) shape (ℓ1‑norm = 1),
    then repeatedly moves a tiny amount of mass from an index that
    contributes heavily to the current maximal autocorrelation to a
    random smaller index, keeping the total ℓ1‑norm fixed.
    The autocorrelation is kept incrementally updated, which makes each
    iteration O(n) instead of O(n²).  Temperature and step‑size decay
    with the remaining budget; the best sequence seen is returned.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin

    a = 1.0 / np.sqrt(np.arange(N, dtype=np.float64) + 1.0)   # a[i] = 1/√(i+1)
    a = a / a.sum() * total_mass  # project onto the simplex (sum = total_mass)

    max_conv = conv.max()

    best_score = curr_score

    step_max = 0.20
    step_min = 0.001
    temp_start = 0.02
    temp_end = 1e-6

        elapsed = time.time() - start_time
        prog = min(1.0, elapsed / budget_s)

        new_conv[j:j + N] += 2.0 * delta * a

        new_max = new_conv.max()
        new_score = 2.0 * N * new_max / (total_mass ** 2)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # probability to accept a worse move
            prob_up = math.exp(-(new_score - curr_score) / (temperature + 1e-12))
            if np.random.rand() < prob_up:
                accept = True

        if accept:
            # apply the move permanently
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            max_conv = new_max
            curr_score = new_score

            # keep the best ever seen
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()
                print(f"iter {iteration:6d} – improved score {best_score:.6f}")

        # ------------------------------------------------------------------
        # occasional global jitter to escape stagnation
        # ------------------------------------------------------------------
        if iteration % 20000 == 0:
            noise = np.random.randn(N) * 1e-4
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum() * total_mass
            conv = np.convolve(a, a)
            max_conv = conv.max()
            curr_score = 2.0 * N * max_conv / (total_mass ** 2)

    # ----------------------------------------------------------------------
    # 3) final validation – call the provided evaluate_sequence once
    # ----------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:
        final_score = float('inf')
    print(f"finished – best score reported by evaluate_sequence: {final_score:.6f}")

    return final_seq
