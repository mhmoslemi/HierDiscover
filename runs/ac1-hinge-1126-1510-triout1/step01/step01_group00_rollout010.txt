"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Broad initialization** – a quick sweep over simple parametric families
   (power‑law and a triangular shape) to obtain a decent starting point.

2. **Accelerated projected Adam optimisation** – the objective
   `J(a)=2·n·max(conv(a,a))` is convex on the probability simplex
   (∑a=1, a≥0).  At each iteration we:
   * compute the self‑convolution via FFT,
   * build an active mask of indices that attain (or nearly attain) the maximum,
   * obtain a sub‑gradient `∇J = 2·corr(mask, a)` (correlation computed by FFT),
   * perform an Adam update and project back onto the simplex.

   The Adam hyper‑parameters are chosen for fast convergence and the step
   size is automatically adapted.

3. **Local stochastic refinement** – after the main Adam run we apply a cheap
   hill‑climbing step that transfers a small amount of mass between two random
   coordinates; any improvement is kept.  This can escape tiny plateaus.

The routine respects the supplied time budget (default 1000 s) and prints
progress information.  The returned sequence is scaled to the required
normalisation `sum = sqrt(2·n)` so that it can be passed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_and_mask(a: np.ndarray, L_conv: int, L_corr: int) -> tuple:
    """
    Compute the linear convolution a * a (via FFT) and return
    - the maximal entry,
    - an active mask (float array length 2n-1) where entries within
      eps of the maximum are set to 1.
    """
    n = a.shape[0]

    # convolution
    fft_a = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(fft_a * fft_a, n=L_conv)[: 2 * n - 1]
    max_val = conv.max()

    # active set mask (floating, used later for gradient)
    eps = max_val * 1e-12 + 1e-14
    mask = (conv >= max_val - eps).astype(np.float64)

    return max_val, mask


# ----------------------------------------------------------------------
def _gradient(mask: np.ndarray, a: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current point a.
    It is 2 * correlation(mask, a) restricted to the first n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L_corr)
    fft_a_rev = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(fft_mask * fft_a_rev, n=L_corr)
    # the relevant slice is indices n-1 … n-1+n-1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return J(a) = 2·n·max(conv(a,a)) for a that already satisfies Σa=1."""
    # use the fast FFT based max computation
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    # ------------------------------------------------------------------
    # Determine the vector length.
    try:
        n = len(height_sequence_1)   # type: ignore
        print(f"Using existing length n={n}")
    except Exception:
        n = 5600                     # a safe default close to the previous run
        print(f"No existing sequence, falling back to n={n}")

    target_sum = math.sqrt(2.0 * n)      # the normalisation used elsewhere

    # Pre‑compute FFT lengths once
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_corr = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – very coarse parametric sweep for a good start
    best_a = None          # will store the best *normalized* (sum=1) vector
    best_score = float("inf")

    print("--- Phase 0: simple parametric sweep ---")
    # power‑law shapes (increasing and decreasing)
    exponents = np.linspace(0.0, 2.0, 41)     # 0, 0.05, …, 2.0
    for p in exponents:
        if time.time() > deadline:
            break

        w = np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a_inc = w / w.sum()                   # Σa = 1
        score_inc = _objective(a_inc, n)
        if score_inc < best_score:
            best_score, best_a = score_inc, a_inc.copy()
            print(f"p={p:.3f} inc  new best J={best_score:.6f}")

        a_dec = a_inc[::-1]
        score_dec = _objective(a_dec, n)
        if score_dec < best_score:
            best_score, best_a = score_dec, a_dec.copy()
            print(f"p={p:.3f} dec  new best J={best_score:.6f}")

    # triangular (linear rise then fall)
    if time.time() < deadline:
        tri = np.concatenate(
            [np.arange(1, n // 2 + 1), np.arange(n - n // 2, 0, -1)]
        ).astype(np.float64)
        tri = tri / tri.sum()
        score_tri = _objective(tri, n)
        if score_tri < best_score:
            best_score, best_a = score_tri, tri.copy()
            print(f"triangular new best J={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser on the simplex
    print("--- Phase 1: Adam optimisation (projected onto simplex) ---")
    # Adam hyper‑parameters
    alpha = 0.05        # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    a = best_a.copy()                     # current iterate (sum=1)
    it = 0
    max_iters = 50000

    while time.time() < deadline and it < max_iters:
        # compute max convolution and active mask
        max_val, mask = _max_conv_and_mask(a, L_conv, L_corr)

        # gradient of the objective J = 2*n*max_val is proportional to the
        # gradient of max_val; the factor 2*n can be absorbed into the step.
        grad = _gradient(mask, a, L_corr)

        it += 1
        t = it
        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam step with projection onto the simplex
        a_temp = a - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = _project_simplex(a_temp)

        # evaluate the new point
        cur_score = _objective(a_new, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            print(f"Iter {it:5d}: improved J = {best_score:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # Phase 2 – cheap stochastic refinement (random mass transfers)
    print("--- Phase 2: stochastic local refinement ---")
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0:
            continue
        # propose moving a fraction of a[i] to a[j]
        delta = np.random.uniform(0.0, a[i] * 0.5)
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # re‑project is not needed – we kept the sum exactly the same
        cand_score = _objective(a_candidate, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"Refine: improved J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Produce the final sequence with the required normalisation.
    final_seq = (best_a * target_sum).tolist()
    print(f"--- Finished: best J = {best_score:.6f} after {it} Adam iters ---")
    return final_seq
