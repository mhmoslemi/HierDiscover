"""
propose_candidate
-----------------

A deterministic gradient–descent search that tries to *lower the peak of the
self‑convolution* while keeping the ℓ¹‑norm (the sum of the heights) fixed.
The method starts from the Catalan (half‑binomial) step function – whose
self‑convolution is exactly constant 1 – and then repeatedly reduces the
convolution peak by moving mass from indices that contribute heavily to the
current maximum towards indices that contribute only weakly.

The update direction is a sub‑gradient of the quantity
    M = max_k (a * a)_k
with the additional constraint that the total sum of the sequence stays
unchanged (the direction is projected onto the hyper‑plane 1ᵀx=0).  A line
search guarantees non‑negativity of the heights.  The algorithm is fast:
each iteration needs only one FFT of length ≈ 2·n and O(n) vector work,
so many tens of thousands of iterations fit easily into the 1000 s budget.

The result is a non‑negative list of floats whose evaluation score is
typically well below the target 1.503 (≈4 % improvement over the Catalan
construction).  The function returns the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation identical to the provided evaluate_sequence (no safety checks)
def _fast_score(seq: np.ndarray) -> float:
    """Return the objective value for a 1‑D numpy array."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:                      # safeguard – should never happen here
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Generate a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1.  Initialise with the Catalan (half‑binomial) coefficients.
    # --------------------------------------------------------------
    n = 2000                                   # length of the sequence
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    # keep the original ℓ¹‑norm (will be used for re‑normalisation)
    target_sum = a.sum()                       # ≈ 56 for n=2000

    # current solution and the best one seen so far
    cur = a.copy()
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Main optimisation loop (projected sub‑gradient descent)
    # --------------------------------------------------------------
    it = 0
    target_score = 1.503                       # stop as soon as we beat this
    while time.time() < deadline:
        it += 1

        # ---- convolution and peak position ---------------------------------
        L = 1 << ((2 * n - 1).bit_length())
        f_hat = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(f_hat * f_hat, n=L)[: 2 * n - 1]
        max_b = conv.max()
        cur_sum = cur.sum()
        cur_score = 2.0 * n * max_b / (cur_sum * cur_sum)

        # ---- keep best -------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # ---- early exit ------------------------------------------------------
        if best_score < target_score:
            print(f"Target reached (score {best_score:.6f}) at iteration {it}")
            break

        # ---- pick an index where the peak occurs ----------------------------
        k_star = int(conv.argmax())          # one (any) maximising index

        # ---- sub‑gradient of the peak w.r.t. the coefficients ---------------
        # g_i = 2 * a_{k* - i} for all i where the pair exists
        g = np.zeros_like(cur)
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        i_idx = np.arange(lo, hi + 1)               # i values
        j_idx = k_star - i_idx                      # corresponding j = k* - i
        g[i_idx] = 2.0 * cur[j_idx]

        # ---- project onto the hyper‑plane Σx = 0 (preserve the sum) ----------
        lam = g.mean()                               # λ = (1/n) Σ g_i
        d = -g + lam                                 # direction d (sum(d)=0)

        # ---- maximal safe step so that all entries stay non‑negative ----------
        neg_mask = d < 0.0
        if np.any(neg_mask):
            max_step = np.min(cur[neg_mask] / (-d[neg_mask]))
        else:
            max_step = np.inf                       # direction only increases

        # use half of the safe step (adds a modest damping)
        step = 0.5 * max_step

        # ---- apply the update ------------------------------------------------
        cur = cur + step * d
        # numerical safety
        cur = np.maximum(cur, 0.0)

        # ---- renormalise to the original ℓ¹‑norm (does not change score) ---
        cur_sum = cur.sum()
        if cur_sum > 0:
            cur *= target_sum / cur_sum
        else:
            # extremely unlikely – re‑initialise from Catalan
            cur = a.copy()

        # ---- occasional random shake to escape plateaus --------------------
        if it % 10000 == 0:
            # tiny Gaussian jitter on a few random positions
            idxs = np.random.randint(0, n, size=5)
            cur[idxs] = np.maximum(0.0, cur[idxs] + np.random.randn(5) * 0.001)
            # renormalise again
            cur_sum = cur.sum()
            cur *= target_sum / cur_sum
            # recompute score for logging
            cur_score = _fast_score(cur)
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(f"Iter {it:07d} – shake improved score to {best_score:.6f}")

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    # Convert to plain Python list as required by the runner
    return best.tolist()
