def project_simplex(v):
    # onto the simplex {x: x>=0, sum(x)=1}
    # from https://arxiv.org/pdf/1101.6081.pdf
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random

    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5

    # Get initial length
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise Exception
        init_len = len(init_seq)
    except Exception:
        init_len = 3000
        init_seq = np.arange(1, init_len+1, dtype=np.float64) ** (-0.5)
        init_seq /= init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)  # compute using fast method

    print(f"Starting best score from init: {best_score:.6f}")

    # Random search over families
    while time.time() < deadline - 5:
        # Choose random length
        n = np.random.choice([init_len, init_len+500, init_len+1000, 4000, 6000, 8000, 10000])
        # Ensure n reasonable not too high
        if n > 12000:
            n = 12000

        mode = np.random.choice(['power', 'exp', 'twosided'])
        if mode == 'power':
            p = np.random.uniform(0.4, 0.7)
            seq = generate_power_law(n, p)
        elif mode == 'exp':
            lam = np.random.uniform(0.0005, 0.02)
            seq = generate_exponential(n, lam)
        else:  # twosided
            p1 = np.random.uniform(0.4, 0.7)
            p2 = np.random.uniform(0.4, 0.7)
            w1 = np.random.uniform(0.2, 2.0)
            w2 = np.random.uniform(0.2, 2.0)
            seq = generate_two_sided_power_law(n, p1, p2, w1, w2)

        # normalise sum to 1
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum

        score = fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] New best ({mode}) n={n} score={score:.6f}")

    # Gradient descent refinement on best_seq
    a = best_seq.copy()
    n = len(a)
    fft_len = 1
    while fft_len < 2*n:
        fft_len <<= 1

    # We'll use simple loop with decreasing step size
    iteration = 0
    while time.time() < deadline - 1:
        iteration += 1
        # compute conv and max index
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:2*n-1]
        conv = np.real(conv)
        # ensure nonnegative
        #conv[conv < 0] = 0
        max_idx = np.argmax(conv)
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end+1)
            j_range = max_idx - i_range
            g[i_range] = 2.0 * a[j_range]

        # step size schedule
        eta = 0.5 / np.sqrt(iteration)
        # exponentiated gradient update
        a = a * np.exp(-eta * g)
        # ensure positivity and normalise
        sum_a = a.sum()
        if sum_a == 0:
            a = best_seq.copy()

        a = a / sum_a

        # periodically evaluate score

                best_score = score

    # final confirm with evaluate_sequence
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)

"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm has two phases:

1. **Diverse random search** – generate hundreds of candidate shapes
   (power‑law, exponential, two‑sided power‑law) with a variety of
   lengths and parameters.  Each candidate is normalised to sum to 1 and
   scored with an inexpensive FFT‑based evaluation.

2. **Projected sub‑gradient refinement** – starting from the best
   candidate, repeatedly reduce the current peak of the self‑convolution
   by a projected gradient step on the simplex (non‑negative entries that
   sum to 1).  The sub‑gradient of the peak ‑ the set of pairs that
   contribute to the maximal convolution entry – is cheap to compute
   (O(n)).  A decreasing step size and periodic re‑evaluation yield a
   steadily improving score.

The whole search respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the evaluation function quickly using an FFT convolution.
    Assumes `seq` is a 1‑D array of non‑negative floats.
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")

    # convolution via FFT (zero‑padded to a power of two)
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (i = 0 … n‑1)  – not normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)  – not normalised."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}
    – not normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

# Main search routine
# ----------------------------------------------------------------------
