    a = np.empty(n, dtype=np.float64)

"""
Optimization of the step‑function bound  evaluate_sequence.

Approach
--------
1. **Generate many deterministic candidates** from several parametric families
   (Catalan, power‑law, beta‑shaped, exponential, uniform and a few Dirichlet draws).
   Each candidate is normalised to ℓ¹‑norm 1 and its score
   `2·n·max(conv)` (identical to `evaluate_sequence`) is evaluated.
   The best of those seeds becomes the starting point.

2. **Directed stochastic hill‑climbing / simulated annealing**.
   * The current maximal convolution index `k*` is identified.
   * A source index `i` is chosen with probability proportional to the
     contribution `a[i]·a[k*‑i]` to that maximal entry (i.e. we move mass from
     coordinates that hurt the worst spot the most).
   * A destination index `j` is drawn uniformly (different from `i`).
   * A fraction of the source mass is transferred (`δ = step_frac·a[i]·U`).
   * The self‑convolution is updated **incrementally** in O(n) time using
     the analytical formula for the effect of moving `δ` from `i` to `j`.
   * The new objective `2·n·max(conv)` is computed.
   * The move is accepted if it improves the score; otherwise it is accepted
     with a small temperature‑dependent probability (simulated annealing).
   * After a long streak without improvement the step size `step_frac` is
     halved; the process stops when it becomes tiny or the time budget expires.

3. The best sequence found within the time limit is returned as a plain list.

The algorithm runs in roughly O(n·iterations) time, where `n` is the length of
the sequence (here a power‑of‑two ≈ 4096 to balance quality and speed).
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Catalan (central‑binomial) coefficients – the classical seed
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan (central‑binomial) terms a_k = C(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine builds several analytic seeds,
    selects the best one, and then performs a directed simulated‑annealing
    local search with an O(n) incremental convolution update.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # length of the step function (power‑of‑two for fast FFT‑style updates)
    n = 4096

    # --------------------------------------------------------------
    # utilities
    # --------------------------------------------------------------
    def normalise(v: np.ndarray) -> np.ndarray:
        """Scale to ℓ¹‑norm 1 (ignore zero‑vector)."""
        s = v.sum()
        if s == 0.0:
            return v
        return v / s

    def score_of(v: np.ndarray) -> float:
        """Return 2·n·max(self‑convolution) for a vector with ℓ¹‑norm 1."""
        conv = np.convolve(v, v)          # length 2·n‑1
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 1.  Scan analytic families for a good seed
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def consider(seq: np.ndarray):
        """Normalise the sequence, evaluate, and keep it if it improves."""
        nonlocal best_seq, best_score
        seq = normalise(seq)
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[seed] score={best_score:.6f}")

    # Catalan
    consider(_catalan_sequence(n))

    # Uniform
    consider(np.ones(n, dtype=np.float64))

    idx = np.arange(n, dtype=np.float64)

    # Power‑law a_k ∝ (k+1)^{-p}
    for p in np.linspace(0.30, 0.90, 31):
        consider(1.0 / (idx + 1.0) ** p)

    # Exponential decay a_k ∝ exp(-λ k)
    for lam in np.logspace(-4, -2, 15):
        consider(np.exp(-lam * idx))

    # Beta‑shaped a_k ∝ (k+1)^{α‑1} (n‑k)^{β‑1}
    alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    betas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    for a in alphas:
        for b in betas:
            consider((idx + 1) ** (a - 1) * (n - idx) ** (b - 1))

    # Small random Dirichlet draws (concentration 0.5 → sparse vectors)
    for _ in range(30):
        consider(np.random.dirichlet(np.full(n, 0.5)))

    # --------------------------------------------------------------
    # 2.  Directed stochastic local optimisation
    # --------------------------------------------------------------
    if best_seq is None:                  # fallback (should never happen)
        best_seq = np.ones(n) / n
        best_score = score_of(best_seq)

    # working copies
    a = best_seq.copy()                 # current candidate (ℓ¹‑norm 1)
    conv = np.convolve(a, a)            # exact self‑convolution
    cur_score = best_score

    # annealing / step‑size parameters
    step_frac = 0.20                     # initial fraction of source mass to move
    no_improve = 0
    iter_cnt = 0
    T0 = 1e-3                            # initial temperature for uphill moves

    while time.time() < deadline:
        iter_cnt += 1

        # ---- locate the current maximal convolution entry -----------------
        k_star = int(np.argmax(conv))

        # ---- choose a source index weighted by its contribution to that peak
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[i_range] * a[k_star - i_range]
        sum_contrib = contrib.sum()
        if sum_contrib > 0:
            probs = contrib / sum_contrib
            r = np.random.rand()
            i = i_range[np.searchsorted(np.cumsum(probs), r)]
        else:
            # fallback: any index proportional to its mass
            probs = a / a.sum()
            i = np.searchsorted(np.cumsum(probs), np.random.rand())

        if a[i] <= 1e-15:
            continue                     # source has no mass – retry

        # ---- choose a destination index (uniform, different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ---- amount to transfer ------------------------------------------------
        delta = step_frac * a[i] * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- backup current state ------------------------------------------------
        a_old = a.copy()
        conv_old = conv.copy()

        # ---- apply the move ------------------------------------------------------
        a[i] -= delta
        a[j] += delta

        # Incremental convolution update (uses a_old, i.e. the vector before the move)
        conv[i:i + n] -= 2.0 * delta * a_old
        conv[j:j + n] += 2.0 * delta * a_old
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # ---- evaluate new objective ------------------------------------------------
        new_max = conv.max()
        new_score = 2.0 * n * new_max
        improve = new_score < cur_score - 1e-12

        # ---- temperature‑based acceptance for occasional uphill moves ------------
        elapsed = time.time() - start_time
        progress = (elapsed) / (deadline - start_time)
        T = T0 * (1.0 - progress)          # linear cooling to 0

        accept = improve or (T > 0.0 and np.random.rand() <
                            np.exp(-(new_score - cur_score) / T))

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[improve] it={iter_cnt} t={int(time.time() - start_time)}s "
                      f"score={best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject – restore previous state
            a = a_old
            conv = conv_old
            no_improve += 1

        # ---- adapt step size ----------------------------------------------------
        if no_improve >= 5000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break

        # ---- periodic re‑synchronisation (avoid drift) -------------------------
        if iter_cnt % 20000 == 0:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()

    print(f"[final] length={n} best_score={best_score:.6f} "
          f"iterations={iter_cnt}")
    return best_seq.tolist()
