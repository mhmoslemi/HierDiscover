Implementation:

```
def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    # If already sum 1 and all >=0, return as is.
    # Implementation: sort descending.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i > cssv_i / i}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All negative?
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Better to ensure numeric stability; but fine.

def subgradient_descent(a, steps, step0, step_decay):
    for t in range(steps):
        conv = conv_fft(a, fft_len_conv, n)
        i_max = np.argmax(conv)
        # gradient of 2*n*conv_i_max w.r.t a is 2*n * (a shifted)
        grad = np.zeros_like(a)
        # fill contributions:
        # for k <= i_max: grad[k] += a[i_max - k]
        # for k >= i_max: grad[k] += a[k - i_max]
        # We can implement via slicing.
        # case k <= i_max:
        grad[:i_max+1] += a[i_max::-1]   # a[i_max - k] (k from 0..i_max)
        # case k >= i_max:
        grad[i_max:] += a[:n-i_max]      # a[k-i_max] (for k from i_max..n-1, index k-i_max)
        # Actually second slice: a[0:n-i_max] corresponds to a[k-i_max] for k from i_max to n-1.
        # So correct.

        # gradient factor:
        grad *= 2.0 * n

        # step size schedule: can use step0 / sqrt(t+1) or step0 * (decay**t)
        eta = step0 / np.sqrt(t+1)  # decays with sqrt
        a_new = a - eta * grad
        # project onto simplex:
        a = project_simplex(a_new)
    return a
```

"""
Hybrid global–local optimizer for the step‑function evaluation problem.

The algorithm searches for a non‑negative sequence `a` (∑a_i = 1) that
minimises   2 n·max(a∗a),  i.e. the value returned by `evaluate_sequence`.
It combines several complementary heuristics:

1. Randomised restarts on the simplex via the smooth‑max objective.
   The smooth objective is differentiable and L‑BFGS‑B is used.
2. A fast greedy hill‑climber that transfers a small fraction of mass
   between two positions while keeping the sum constant.
3. A few sub‑gradient descent steps on the exact max‑convolution
   (non‑smooth) with projection onto the simplex.
4. A brief simulated‑annealing phase that can escape shallow local minima.

All operations are performed with FFT‑based convolutions (O(n log n)).
The search runs for the allotted time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Utility functions (no closures, top‑level)
# -------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that always returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a of length 2 n‑1 using an FFT of length `fft_len`.
    Returns the real part of the convolution (length 2 n‑1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2 n·max(conv(a,a)) for a sequence `a`
    that (by construction) sums to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max surrogate of the objective and its gradient w.r.t. `y`.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables; the actual sequence is `a = softmax(y)`.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        Pre‑computed FFT lengths for the convolution and the gradient.
    lam_factor : float
        Fraction of the current maximal convolution value used as the
        smoothing width (λ = lam_factor·max + ε).

    Returns
    -------
    f : float
        Smooth surrogate of 2 n·max(a∗a).
    g : np.ndarray
        Gradient ∂f/∂y (same shape as `y`).
    """
    # map to simplex
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient of smooth max ----------
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # (a reversed) * w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 5000,
               move_frac: float = 0.10) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climber.

    A random fraction (≤ `move_frac`) of the mass at a source index is moved
    to a destination index.  The move is kept only if the objective improves.
    The sum stays exactly 1, so the sequence always remains on the simplex.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, ∑x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_descent(a: np.ndarray,
                        fft_len_conv: int,
                        n: int,
                        steps: int = 200,
                        step0: float = None) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    Each iteration moves opposite to the sub‑gradient of 2 n·max(conv)
    and projects back onto the simplex.
    """
    if step0 is None:
        step0 = 1.0 / (2.0 * n)   # a safe default

    cur = a.copy()
    for t in range(steps):
        conv = conv_fft(cur, fft_len_conv, n)
        i_max = int(np.argmax(conv))

        # gradient of 2 n·conv[i_max] w.r.t. a
        grad = np.zeros_like(cur)

        # contributions a[i_max - k] for k ≤ i_max
        grad[:i_max + 1] += cur[i_max::-1]
        # contributions a[k - i_max] for k ≥ i_max
        grad[i_max:] += cur[:n - i_max]

        grad *= 2.0 * n

        eta = step0 / np.sqrt(t + 1.0)
        cur = project_simplex(cur - eta * grad)

    return cur


def simulated_annealing(a: np.ndarray,
                        score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_moves: int = 60000) -> (np.ndarray, float):
    """
    Very light simulated‑annealing on the simplex.
    Random mass transfers are occasionally accepted even if they worsen the
    objective, according to an exponential cooling schedule.
    """
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = score
    cur_a = a.copy()
    cur_score = score

    # initial temperature proportional to the current score
    T0 = max(0.05 * best_score, 1e-4)
    T_end = 1e-9

    move = 0
    while time.time() < deadline and move < max_moves:
        move += 1
        # exponential cooling
        frac = move / max_moves
        T = T0 * (T_end / T0) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] == 0.0:
            continue
        # transfer up to 10 % of the source mass
        delta = cur_a[i] * rng.random() * 0.10
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert move
            cur_a[i] += delta
            cur_a[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin so that the final evaluation can run
    deadline = time.time() + budget_s - 2.0

    # problem size – a larger n gives a finer search space
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initialise with the uniform distribution (known baseline)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    iteration = 0
    modes = ['random', 'exp', 'power', 'edge', 'sin', 'gauss', 'prev', 'perturb']

    # reserve ~10 % of the budget for a final refinement phase
    final_phase_deadline = deadline - (budget_s * 0.10)

    while time.time() < final_phase_deadline:
        # ----------- choose a starting point ---------------------------------
        mode = rng.choice(modes)

        # handle the "prev" mode only if the global variable exists
        if mode == 'prev' and 'height_sequence_1' not in globals():
            mode = 'random'

        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'sin':
            a0 = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'gauss':
            sigma = rng.uniform(0.05, 0.30)
            x = np.arange(n) - n / 2.0
            a0 = np.exp(-0.5 * (x / (sigma * n)) ** 2)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'prev':
            # use the previously discovered construction if present
            seq = np.array(height_sequence_1, dtype=float)
            if seq.size != n:
                # pad or truncate to match the current size
                if seq.size < n:
                    seq = np.concatenate([seq, np.zeros(n - seq.size)])
                else:
                    seq = seq[:n]
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0:
                seq = np.full(n, 1.0 / n)
            else:
                seq /= seq.sum()
            y0 = np.log(seq + 1e-12)
        else:  # 'perturb' – slight random perturbation of the current best
            log_best = np.log(best_a + 1e-12)
            y0 = log_best + rng.normal(scale=0.2, size=n)

        # ----------- L‑BFGS on the smooth objective -------------------------
        # pick a smoothing factor; occasionally use a larger one for exploration
        lam_factor = rng.choice([0.5, 0.35, 0.30, 0.20, 0.12, 0.08])

        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-9,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  improved fast score = {best_score:.6f}")

        # ----------- greedy local refinement ---------------------------------
        # allocate a short time slice for hill climbing
        hc_deadline = min(deadline, time.time() + 0.55)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=3000,
                                        move_frac=0.07)

        # ----------- optional sub‑gradient fine tuning -----------------------
        # keep this very cheap – only a few hundred steps
        sub_steps = 120
        best_a = subgradient_descent(best_a, fft_len_conv, n,
                                     steps=sub_steps,
                                     step0=1.0 / (2.0 * n))
        best_score = fast_score(best_a, fft_len_conv, n)

        # ----------- very brief simulated annealing -------------------------
        # (only if we still have a couple of seconds left)
        if time.time() < deadline - 3.0:
            sa_deadline = min(deadline, time.time() + 0.45)
            best_a, best_score = simulated_annealing(best_a, best_score,
                                                     fft_len_conv, n,
                                                     sa_deadline,
                                                     max_moves=25000)

        iteration += 1

    # -----------------------------------------------------------------
    # Final refinement (extra time if any)
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # One final hill‑climb pass with a tighter move budget
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        deadline,
                                        max_iter=4000,
                                        move_frac=0.05)

    # Convert to plain Python list for the official evaluator
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
