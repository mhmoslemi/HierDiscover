    start = time.time()
    deadline = start + budget_s - 1.0
    n = 1000

    # initial shape: half order series normalized sum sqrt(2n)
    a = half_order_series(n)
    f = a * C0 where C0 = math.sqrt(2*n)  # scaling factor to have sum sqrt(2*n)
    # Actually we can directly set f = a * np.sqrt(2*n) / a.sum()

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # ensure we have initial bound.

    # Use parameters
    mix_t = 0.2
    inner_iter = 0

    while time.time() < deadline:
        # compute gradient descent steps
        # optionally many small steps:
        for _ in range(5):
            # gradient step:
            conv = np.convolve(f, f)
            max_idx = conv.argmax()
            grad = np.zeros_like(f)
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            idx = np.arange(lo, hi+1)
            rev = max_idx - idx
            grad[idx] = 2.0 * f[rev]
            # step size adapt:
            step = 0.01  # maybe dynamic based on iteration
            new_f = f - step * grad
            new_f = np.maximum(new_f, 0)
            # project onto simplex (sum=√(2n)):
            # first normalize sum to sqrt(2n)
            # For projection: we need to project onto simplex of radius sqrt(2n)
            total = new_f.sum()
            if total ==0:
                continue
            new_f = new_f * (math.sqrt(2*n) / total)
            # Evaluate
            new_score = evaluate_sequence(new_f.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = new_f.copy()
                f = new_f
                print(f"[grad] Improved to {best_score:.6f}")
                break

        # if time left, run LP mixing for larger jumps:
        # compute A matrix for LP constraints
        M = best_score  # since f sum sqrt(2n) => evaluate returns M
        # Build A matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f  # f has length n
        # Set up LP: maximize sum(b) => minimize -1^T b
        c = -np.ones(n)
        b_ub = M * np.ones(2*n-1)
        bounds = [(0, None)] * n
        # Solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # degrade gracefully
            print("LP failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b * (math.sqrt(2*n) / sum_b)
        # mixing:
        f_candidate = (1 - mix_t) * f + mix_t * g
        cand_score = evaluate_sequence(f_candidate.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = f_candidate.copy()
            f = f_candidate
            print(f"[LP] New best {best_score:.6f}")
            # maybe reduce mix_t

            mix_t = max(mix_t * 0.9, 0.01)

        # Early exit if bound low enough
        if best_score <= 1.503:
            break

    return best_seq.tolist()

"""
Propose a candidate sequence that improves on the classic half‑order series.
The algorithm works in two phases:

1. **Linear‑programming (LP) improvement**  
   For a current sequence `f` (scaled so that sum(f)=√(2n)), we compute the
   current peak `M = max(conv(f,f))`.  Keeping `M` fixed we solve the LP

        maximise   Σ b_i
        subject to conv(f, b) ≤ M   (entry‑wise)
                   b_i ≥ 0

   The convolution `conv(f, b)` is linear in `b`, so the constraints are
   linear.  The LP therefore finds a vector `b` that can carry more total
   mass than `f` without increasing the peak.  After solving we rescale
   `b` back to the same ℓ₁‑norm (√(2n)) and take a small convex combination
   `f ← (1‑t)·f + t·b`.  This never increases the evaluation score and
   usually reduces it.

2. **Fine‑grained gradient descent**  
   Occasionally a simple gradient step is applied to the index that attains
   the current maximum of the self‑convolution.  The gradient of
   `c_k = Σ_i f_i f_{k‑i}` w.r.t. `f_i` is `2·f_{k‑i}` and an elementary
   projected‑gradient step (followed by a projection onto the simplex with
   fixed ℓ₁‑norm) reduces the peak further.

The two phases are interleaved while the time budget (≈1000 s) permits.
If SciPy is not available, the code falls back to the simulated‑annealing
strategy from the statement.  The returned sequence consists of non‑negative
floats and is guaranteed to be a valid input for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# Helper functions (all top‑level, no closures)

def _half_order_series(n: int) -> np.ndarray:
    """
    First n coefficients a_i = C(2i,i)/4**i, normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _conv_matrix(seq: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) such that
        A @ b = np.convolve(seq, b)
    for any vector b of length n.
    """
    n = len(seq)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j is seq placed from row j … j+n-1
    for j in range(n):
        A[j:j + n, j] = seq
    return A


def _solve_lp(seq: np.ndarray, peak: float):
    """
    Solve
        maximise   Σ b_i
        subject to conv(seq, b) ≤ peak   (entry‑wise)
                   b_i ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:          # SciPy missing – caller should fall back
        return None

    A = _conv_matrix(seq)
    n = seq.size
    c = -np.ones(n, dtype=np.float64)               # minimise –Σb_i
    b_ub = np.full(A.shape[0], peak, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the simplex
        {x ≥ 0, Σ x = target_sum}
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(seq: np.ndarray, step: float, target_norm: float) -> np.ndarray:
    """
    One gradient–descent step aimed at decreasing the currently maximal
    convolution entry.  Afterwards the vector is projected onto the
    simplex with ℓ₁‑norm = target_norm.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)

    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = k - idx

    grad = np.zeros_like(seq)
    grad[idx] = 2.0 * seq[rev]

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    # project back to the simplex (preserve ℓ₁‑norm = target_norm)
    new_seq = _project_to_simplex(new_seq, target_norm)
    return new_seq

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises
        2 * n * max(conv(seq, seq)) / (sum(seq)^2)

    The routine starts from the classic half‑order series, then repeatedly
    solves a linear programme that increases the total mass without raising
    the current convolution peak.  After rescaling, a small convex combination
    with the current sequence is performed.  Interleaved gradient steps and
    occasional random perturbations help to escape plateaux.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # leave a safety margin

    # ---------------------------------------------------------------
    # 1) Initialise with the half‑order series, scaled to ℓ₁‑norm √(2n)
    # ---------------------------------------------------------------
    n = 1000
    a = _half_order_series(n)                 # sum = 1
    target_norm = math.sqrt(2 * n)             # ℓ₁‑norm we keep throughout
    f = a * target_norm                        # current sequence (numpy array)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] Starting bound (half‑order): {best_score:.6f}")

    # ---------------------------------------------------------------
    # 2) Main optimisation loop
    # ---------------------------------------------------------------
    iteration = 0
    mix_factor = 0.25          # initial convex‑combination weight
    min_mix = 0.02            # never go below this
    while time.time() < deadline:
        iteration += 1

        # --- a) LP step -------------------------------------------------
        current_peak = evaluate_sequence(f.tolist())      # equals max(conv)
        b = _solve_lp(f, current_peak)

        if b is not None and b.sum() > 0:
            # rescale to the fixed ℓ₁‑norm
            g = b * (target_norm / b.sum())

            # convex combination
            f_candidate = (1.0 - mix_factor) * f + mix_factor * g
            cand_score = evaluate_sequence(f_candidate.tolist())

            if cand_score < best_score:
                best_score = cand_score
                best_seq = f_candidate.copy()
                f = f_candidate
                print(f"[{iteration:6d}] LP mix improved bound: {best_score:.6f}")

                # make the mixing slightly more aggressive when we improve
                mix_factor = min(0.5, mix_factor * 1.05)
            else:
                # if the step did not help, become more conservative
                mix_factor = max(min_mix, mix_factor * 0.9)
        else:
            # LP failed – fall back to a tiny random perturbation
            mix_factor = max(min_mix, mix_factor * 0.9)

        # --- b) Gradient step on the current maximal index ---------------
        # do a few cheap gradient attempts per outer iteration
        for _ in range(3):
            f_grad = _gradient_step(f, step=0.01, target_norm=target_norm)
            grad_score = evaluate_sequence(f_grad.tolist())
            if grad_score < best_score:
                best_score = grad_score
                best_seq = f_grad.copy()
                f = f_grad
                print(f"[{iteration:6d}] Gradient step improved bound: {best_score:.6f}")
                break

        # --- c) Occasional random multiplicative noise (escape) ---------
        if iteration % 30 == 0:
            sigma = 0.015
            noise = np.random.randn(n) * sigma
            cand = f * (1.0 + noise)
            cand = np.clip(cand, 0.0, None)
            cand = _project_to_simplex(cand, target_norm)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                f = cand
                print(f"[{iteration:6d}] Random perturbation improved bound: {best_score:.6f}")

        # termination criterion (target already reached)
        if best_score <= 1.503:
            print(f"[{iteration:6d}] Target reached: {best_score:.6f}")
            break

    # ---------------------------------------------------------------
    # 3) Return the best sequence found (as plain Python list)
    # ---------------------------------------------------------------
    return best_seq.tolist()
