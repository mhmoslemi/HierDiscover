        return _max_self_convolution(a)

    # LP iteration
    t = 0.2
    max_iter = 30
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        M = max_conv(best_seq)
        # Build convolution matrix of best_seq
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):

        # solve LP: maximize sum(g) i.e., minimize -sum(g)

        g_opt = res.x

        if sum_g_opt <= best_seq.sum() + 1e-12:

        g_scaled = g_opt * (best_seq.sum() / sum_g_opt)

        new_seq = (1 - t) * best_seq + t * g_scaled

        # renormalize sum to 1 (optional)
        total = new_seq.sum()
        if total == 0:

        new_seq /= total

        # evaluate

"""
Propose a candidate sequence for `evaluate_sequence`.

The algorithm has two phases:

1. **LP‐phase** – Starting from the Catalan (central‑binomial) sequence,
   we repeatedly solve a linear program:
       maximise   Σ g_j
       subject to g_j ≥ 0  and  (f * g)_k ≤ M   for all k,
   where f is the current sequence and M = max(f * f).
   The LP finds a “more spread” distribution g that does not increase the
   worst self‑convolution value.  We then mix a fraction `t` of the
   rescaled g into f.  Repeating this step monotonically reduces the
   evaluation score.

2. **Hill‑climbing phase** – After the LP iterations we perform a fast
   stochastic local search (mass transfers) to polish the result.

Both phases respect the non‑negativity and cut‑off rules of the evaluator.
The implementation works for modest lengths (n≈2000–2500) which keep the
linear program tractable while still allowing a significant improvement
over the plain Catalan bound (≈1.571).  The total running time is limited
to the supplied budget (default 1000 s).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# SciPy is optional – if unavailable we fall back to hill‑climbing only.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _catalan_sequence(n: int) -> np.ndarray:
    """First n terms of binom(2k,k)/4**k (central‑binomial coefficients)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _max_self_convolution(x: np.ndarray) -> float:
    """Maximum entry of the linear convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()
    fx = rfft(x, n=nfft)
    conv = irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The method first performs a few LP‑based updates
    starting from the Catalan distribution and then finishes with a light
    stochastic hill‑climbing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # Basic parameters – they can be tuned without breaking the API.
    # -----------------------------------------------------------------
    n = 2400                     # a size that keeps the LP tractable
    raw = _catalan_sequence(n)
    f = raw / raw.sum()         # start with ℓ¹‑norm = 1
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – linear‑programming driven “spreading”.  We limit the
    # number of LP steps to keep the total runtime well below the budget.
    # -----------------------------------------------------------------
    if linprog is not None:
        t = 0.2                 # mixing factor
        max_lp_iters = 12
        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # current worst self‑convolution value
            M = _max_self_convolution(best_seq)

            # Build the convolution matrix A where A @ g = f * g.
            # A has shape (2n‑1, n) and A[k,i] = f[i] if 0≤k‑i<n else 0.
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = best_seq[i]

            # Linear program: maximise Σ g   ⇔ minimise –Σ g
            c = -np.ones(n, dtype=np.float64)               # objective
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # RHS of constraints
            bounds = [(0.0, None)] * n                        # g_j ≥ 0

            # Solve – HiGHS works well for dense problems of this size.
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                print(f"[LP] iteration {it+1} failed: {res.message}")
                break

            g_opt = res.x
            sum_g_opt = g_opt.sum()
            # If the LP could not increase the ℓ¹‑mass we stop.
            if sum_g_opt <= best_seq.sum() + 1e-12:
                print(f"[LP] iteration {it+1}: no ℓ¹‑gain")
                break

            # Rescale g to have the same total mass as f (still =1)
            g_scaled = g_opt * (best_seq.sum() / sum_g_opt)

            # Convex combination  f ← (1‑t)·f + t·g_scaled
            new_seq = (1.0 - t) * best_seq + t * g_scaled
            # Normalise (tiny rounding errors may accumulate)
            new_seq /= new_seq.sum()

            # Quick score using the known normalisation (sum=1)
            new_score = 2.0 * n * _max_self_convolution(new_seq)

            print(f"[LP] iter {it+1:2d} t={t:.3f} score={new_score:.6f}")

            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq
                # slowly shrink the mixing factor once we see progress
                t = max(0.02, t * 0.9)
            else:
                # if no improvement, reduce the step size more aggressively
                t = max(0.02, t * 0.5)

    # -----------------------------------------------------------------
    # Phase 2 – lightweight stochastic hill climbing (the same idea as the
    # reference implementation but with a cooling schedule).
    # -----------------------------------------------------------------
    step_frac = 0.02                     # start with 2 % of the donor mass
    no_improve = 0
    while time.time() < deadline:
        # Pick a source (large index) and a destination (small index)
        i = np.random.randint(n // 2)               # source in the “large” half
        j = np.random.randint(n // 2, n)            # destination in “small” half

        # Transfer a random fraction of the source entry
        delta = step_frac * np.random.rand() * best_seq[i]
        if delta <= 0.0:
            continue

        candidate = best_seq.copy()
        candidate[i] -= delta
        candidate[j] += delta
        candidate /= candidate.sum()                 # keep ℓ¹‑norm = 1

        # Evaluate quickly (exact evaluation is cheap for these sizes)
        conv_max = _max_self_convolution(candidate)
        score = 2.0 * n * conv_max

        if score < best_score:
            best_score = score
            best_seq = candidate
            no_improve = 0
            print(f"[improve] t={int(time.time() - start_time)}s  score={best_score:.6f}")
        else:
            no_improve += 1

        # If we stall for a while, shrink the step size.
        if no_improve >= 2500:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break

    print(f"[final] n={n}, best_score={best_score:.6f}")
    return best_seq.tolist()
```
