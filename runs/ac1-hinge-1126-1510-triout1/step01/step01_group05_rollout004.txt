"""
Propose a low‑evaluation step sequence.

The algorithm works in three phases:

1. **Low‑dimensional parametric sweep** –  
   We generate many simple families of candidate distributions
   (edge‑heavy piecewise‑constant, exponential decay, power‑law)
   that are described by only one or two parameters.
   Each candidate is evaluated with a fast FFT based score
   `2 * n * max(conv)` (the exact evaluation function up to the
   harmless sum‑normalisation).  The best candidate from this sweep
   is used as a warm start for the next phase.

2. **Smooth‑max quasi‑Newton optimisation** –  
   Starting from the best candidate we optimise the *smooth* maximum
   of the convolution using an L‑BFGS run.  The variables are
   unconstrained logits `y`; the sequence is `a = softmax(y)`,
   guaranteeing non‑negativity and unit sum.  The smooth‑max
   approximation is `λ·log Σ exp(conv/λ)` with a λ proportional to the
   current maximum, giving a differentiable objective and an analytic
   gradient that is computed via FFTs (O(n log n)).

3. **Greedy hill‑climbing refinement** –  
   Finally a simple stochastic mass‑transfer hill‑climb moves a small
   fraction of mass between two random positions as long as the
   true score (the exact `evaluate_sequence` objective) improves.

The whole search respects the 1000‑second budget and prints
intermediate progress.  The returned list is a plain Python list of
floats ready for `evaluate_sequence`.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (length 2*n‑1).  a must already sum to 1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2*n*max(conv) for a sequence that already sums to 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective and its gradient w.r.t. the logits y.

    Returns
    -------
    f : float
        2 * n * smooth_max(conv)
    g : np.ndarray
        Gradient ∂f/∂y (shape (n,))
    """
    a = softmax(y)                     # a lives on the simplex

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12   # smoothing width (adaptive)

    # ----- smooth maximum (log‑sum‑exp) -----
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                  # ∂smooth_max/∂conv_i

    # g_i = Σ_j a_j * w_{i+j}
    # This is a convolution of reversed a with w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple stochastic hill‑climb that transfers a small fraction of mass
    from one coordinate to another while keeping the sum exactly 1.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Candidate generators (low‑dimensional families)
# ----------------------------------------------------------------------
def generate_edge_pattern(n: int, m: int, r: float) -> np.ndarray:
    """
    Piecewise‑constant pattern:
        * first m and last m entries have weight r times the interior weight
        * all other entries have the interior weight
    The vector is normalised to unit sum.
    """
    if m == 0:
        return np.full(n, 1.0 / n)

    denom = (n - 2 * m) + 2 * m * r
    w_center = 1.0 / denom
    w_edge = r * w_center
    a = np.full(n, w_center)
    a[:m] = w_edge
    a[-m:] = w_edge
    return a


def generate_power_pattern(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ (i+1)^p + (n-i)^p .
    For negative p the distribution is edge‑heavy,
    for positive p it concentrates in the centre.
    """
    idx = np.arange(n, dtype=float)
    weights = (idx + 1) ** p + (n - idx) ** p
    # Guard against overflow/underflow for extreme p
    if np.any(~np.isfinite(weights)):
        weights = np.nan_to_num(weights, nan=0.0, posinf=0.0, neginf=0.0)
    total = weights.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return weights / total


def generate_exponential_pattern(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay from both ends:
        a_i ∝ exp(-lam*i) + exp(-lam*(n-1-i))
    """
    idx = np.arange(n, dtype=float)
    weights = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    total = weights.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return weights / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The function respects the given time budget and returns a list of
    non‑negative floats (the sequence).  Intermediate progress is printed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Problem size – a compromise between flexibility and speed.
    n = 2000

    # Pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1️⃣  Low‑dimensional parametric sweep
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # uniform baseline
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    # ---- edge‑heavy piecewise constant ----
    max_m = int(0.12 * n)               # up to ~12 % of the indices
    step_m = max(1, max_m // 30)        # ≈ 30 different m‑values
    r_vals = np.logspace(0, 2, num=30)  # r from 1 to 100 (log‑spaced)

    for m in range(0, max_m + 1, step_m):
        for r in r_vals:
            if time.time() >= deadline:
                break
            a_cand = generate_edge_pattern(n, m, r)
            sc = fast_score(a_cand, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a_cand
                print(f"[edge] m={m:4d} r={r:5.2f}  fast score={sc:.6f}")

    # ---- power‑law family ----
    p_vals = np.linspace(-3.0, 1.0, num=35)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a_cand = generate_power_pattern(n, p)
        sc = fast_score(a_cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_cand
            print(f"[power] p={p:5.2f}  fast score={sc:.6f}")

    # ---- exponential decay family ----
    lam_vals = np.logspace(np.log10(0.001), np.log10(0.2), num=30)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        a_cand = generate_exponential_pattern(n, lam)
        sc = fast_score(a_cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_cand
            print(f"[exp]   λ={lam:6.4f}  fast score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max quasi‑Newton refinement (L‑BFGS)
    # ------------------------------------------------------------------
    # Warm‑start from the best candidate discovered above.
    # We run a single, relatively long L‑BFGS optimisation.
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)

        # L‑BFGS options – more iterations than the earlier baseline.
        options = {'maxiter': 500, 'gtol': 1e-10, 'disp': False}
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, 0.30),
                       method='L-BFGS-B',
                       jac=True,
                       options=options)

        a_opt = softmax(res.x)
        sc_opt = fast_score(a_opt, fft_len_conv, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            print(f"[lbfgs] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Greedy hill‑climb refinement (exact score)
    # ------------------------------------------------------------------
    # Use the official evaluator as the true objective for the hill‑climb.
    if time.time() < deadline:
        # First compute the exact score for the current best.
        exact_score = evaluate_sequence(best_a.tolist())
        # The hill climb works with the fast surrogate, but we keep the exact
        # value for the final printout and for early‑stop decisions.
        hc_deadline = min(deadline, time.time() + 0.8 * (deadline - time.time()))
        best_a, fast_sc = hill_climb(best_a, best_score,
                                     fft_len_conv, n,
                                     hc_deadline,
                                     max_iter=5000)
        # Re‑evaluate with the official function after hill‑climbing.
        exact_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluated score = {exact_score:.6f}")

    # Return the sequence as a plain Python list.
    return best_a.tolist()
