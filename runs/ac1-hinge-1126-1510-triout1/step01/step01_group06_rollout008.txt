"""
Propose a coefficient sequence that minimizes the evaluation function.

The algorithm works in three phases:

1. **Coarse parametric scan** – generate candidate step functions for a
   collection of lengths (≈300‑1800) and several simple parametric families
   (power‑law differences, √‑differences with offset, exponentials and a
   linear ramp).  Each candidate is normalised to L1‑norm 1 and its exact
   objective value 2·n·max(a∗a) is obtained with an FFT‑based convolution.
   The best candidate from this scan is kept.

2. **Local refinement** – starting from the best candidate a projected‑gradient
   descent on the active convolution index is combined with random
   “mass‑transfer’’ moves.  The step size and a simulated‑annealing temperature
   are gradually decreased.  Every move is evaluated analytically (no call to
   `evaluate_sequence`) and accepted if it improves the objective, or with a
   Metropolis probability to escape shallow minima.

3. **Polishing** – a short random‑swap phase fine‑tunes the final solution.
   The resulting sequence is finally scaled to the full [0, 1000] range
   (scaling does not change the score) and returned as a plain Python list.

All numbers are kept non‑negative and the L1‑norm is fixed to 1, so the
search stays inside the feasible region enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max_and_index(seq: np.ndarray):
    """
    Return (max value, index) of the self‑convolution of `seq`.
    Uses an FFT of length a power of two for speed.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    conv = np.real(conv)                            # discard tiny imag part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _conv_max(seq: np.ndarray) -> float:
    """Return only the maximal convolution value."""
    max_val, _ = _conv_max_and_index(seq)
    return max_val


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` (scale invariant):
        2 * n * max(a∗a)   with sum(a)=1
    """
    n = seq.shape[0]
    return 2.0 * n * _conv_max(seq)


def _make_power_seq(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^p - i^p, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    raw = (i + 1) ** p - i ** p
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones_like(raw)
        raw_sum = raw.sum()
    return raw / raw_sum


def _make_sqrt_seq(n: int, offset: float = 0.0) -> np.ndarray:
    """a_i = sqrt(i+1+offset) - sqrt(i+offset), normalised."""
    i = np.arange(n, dtype=np.float64)
    raw = np.sqrt(i + 1.0 + offset) - np.sqrt(i + offset)
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones_like(raw)
        raw_sum = raw.sum()
    return raw / raw_sum


def _make_exp_seq(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha*i), normalised."""
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(-alpha * i)
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones_like(raw)
        raw_sum = raw.sum()
    return raw / raw_sum


def _make_linear_seq(n: int) -> np.ndarray:
    """Simple decreasing linear ramp, normalised."""
    raw = n - np.arange(n, dtype=np.float64)
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones_like(raw)
        raw_sum = raw.sum()
    return raw / raw_sum


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function.
    Returns a list of floats (each in [0,1000]).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # 0.  Keep a copy of an externally supplied sequence if it exists
    # ------------------------------------------------------------------
    try:
        # pylint: disable=invalid-name
        height_sequence_1  # noqa: F821
        external = np.asarray(height_sequence_1, dtype=np.float64)
        if external.size > 0:
            external = external / external.sum()
        else:
            external = None
    except Exception:  # noqa: BLE001
        external = None

    # ------------------------------------------------------------------
    # 1.  Coarse parametric scan over many lengths / families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_desc = ""

    length_candidates = [
        300, 400, 500, 600, 800, 900, 1000, 1100, 1200,
        1300, 1400, 1500, 1600, 1800,
    ]

    power_exponents = [0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60]
    sqrt_offsets = [0.0, 0.1, 0.5, 1.0]
    exp_alphas = [0.001, 0.002, 0.003, 0.005, 0.008, 0.010]

    for n in length_candidates:
        # ----- power‑law differences -----
        for p in power_exponents:
            seq = _make_power_seq(n, p)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                best_desc = f"pow n={n} p={p}"
                print(f"New best (scan): {best_score:.6f} – {best_desc}")

        # ----- sqrt‑difference with offset -----
        for off in sqrt_offsets:
            seq = _make_sqrt_seq(n, off)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                best_desc = f"sqrt n={n} off={off}"
                print(f"New best (scan): {best_score:.6f} – {best_desc}")

        # ----- exponential decay -----
        for a in exp_alphas:
            seq = _make_exp_seq(n, a)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                best_desc = f"exp n={n} α={a}"
                print(f"New best (scan): {best_score:.6f} – {best_desc}")

        # ----- linear ramp -----
        seq = _make_linear_seq(n)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            best_desc = f"linear n={n}"
            print(f"New best (scan): {best_score:.6f} – {best_desc}")

    # ----- try the provided global sequence (if any) -----
    if external is not None:
        sc = _objective(external)
        if sc < best_score:
            best_score, best_seq = sc, external.copy()
            best_desc = "external height_sequence_1"
            print(f"New best (external): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Local refinement (gradient + random mass transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.size
    it = 0
    while time.time() < deadline:
        it += 1

        # decide move type
        if np.random.rand() < 0.55:          # gradient‑type move
            max_b, kmax = _conv_max_and_index(a)
            # gradient of the active convolution term
            grad = np.zeros_like(a)
            i_low = max(0, kmax - (n - 1))
            i_high = min(kmax, n - 1)
            if i_low <= i_high:
                i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
                j_vals = kmax - i_vals
                grad[i_vals] = a[i_vals] + a[j_vals]      # includes the 2*a_i term on the diagonal
            # zero‑sum direction (preserve L1)
            mean_g = grad.mean()
            direction = -grad + mean_g

            # temperature‑controlled step length
            temp = max(1.0 * (0.9995 ** it), 1e-4)
            step = 0.04 * temp * (np.linalg.norm(direction) / n + 1e-12)
            step = max(step, 1e-8)

            cand = a + step * direction
        else:                                 # random transfer move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            eps = a[i] * np.random.rand() * 0.04  # move up to 4 % of source mass
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps

        # projection onto feasible set (non‑negative, L1=1)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        # cheap objective evaluation
        cand_score = _objective(cand)

        # accept if improvement
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            a = cand
            print(f"Iter {it}: improved best to {best_score:.6f} (len={n})")
            continue

        # Metropolis acceptance for uphill moves
        temp = max(1.0 * (0.9995 ** it), 1e-4)
        if np.random.rand() < np.exp(-(cand_score - best_score) / temp):
            a = cand

    # ------------------------------------------------------------------
    # 3.  Very short polishing phase (random tiny swaps)
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    a = best_seq.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = a[i] * np.random.rand() * 0.02 + 1e-12
        a2 = a.copy()
        a2[i] -= eps
        a2[j] += eps
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        sc2 = _objective(a2)
        if sc2 < best_score:
            best_score, best_seq = sc2, a2.copy()
            a = a2
            print(f"Polish: new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Rescale to the [0,1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Returning sequence (len={len(final_seq)}) with final score {best_score:.6f}")
    return final_seq.tolist()
