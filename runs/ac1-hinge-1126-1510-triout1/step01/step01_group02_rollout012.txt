    diagonal_vals = [np.full(n, f[d]) for d in range(n)]

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Frank‑Wolfe style optimization for the convolution‑maximisation bound.
    Starts from the half‑binomial (Catalan) coefficients, then iteratively
    solves a linear programme that finds a direction which can increase the
    L¹‑norm without raising the worst self‑convolution.  A convex combination
    of the current sequence and the LP solution is taken, decreasing the
    evaluation metric.  The loop stops after the time budget or when the
    target score is reached.
    Returns a list of non‑negative floats.
    """
    import time, numpy as np, cvxpy as cp, scipy.sparse as sp

    # set random seed
    np.random.seed(seed)

    # length of sequence (tunable)
    n = 2500

    # generate Catalan half‑binomial coefficients
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2.0*i)

    # ensure sum is not too small; current sum = ~56
    f = a.copy()
    f_sum = f.sum()

    # compute initial score using original evaluate_sequence for safety
    cur_score = evaluate_sequence(f.tolist())
    best_f = f.copy()
    best_score = cur_score
    print(f"Start score: {best_score:.6f}")

    # helper functions
    def fast_score(seq):
        n_local = seq.size
        s = seq.sum()
        if s < 0.01:
            return np.inf
        L = 1 << ((2 * n_local - 1).bit_length())
        ft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(ft * ft, n=L)[: 2 * n_local - 1]
        return 2.0 * n_local * conv.max() / (s * s)

    def max_conv(seq):
        n_local = seq.size
        L = 1 << ((2 * n_local - 1).bit_length())
        ft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(ft * ft, n=L)[: 2 * n_local - 1]
        return conv.max()

    def solve_lp(f_vec, M_val):
        # Build convolution matrix A where (f*b)[k] = sum_i f_i * b_{k-i}
        n_local = f_vec.shape[0]
        # each diagonal -d has constant value f[d]
        diag_data = []
        offsets = []
        for d in range(n_local):
            diag_data.append(np.full(n_local, f_vec[d], dtype=np.float64))
            offsets.append(-d)
        A = sp.diags(diag_data, offsets, shape=(2 * n_local - 1, n_local), format='csc')
        b_var = cp.Variable(n_local, nonneg=True)
        constraints = [A @ b_var <= M_val * np.ones(2 * n_local - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver='ECOS', warm_start=True, abstol=1e-7, reltol=1e-6, feastol=1e-7, max_iters=2000)
        if b_var.value is None:
            raise RuntimeError("LP did not return a solution")
        return b_var.value

    start = time.time()
    deadline = start + budget_s - 5.0  # keep small margin
    iteration = 0
    # schedule: for first few iterations use aggressive steps, later smaller.
    while time.time() < deadline:
        iteration += 1
        # compute current maximum convolution
        M = max_conv(f)

        # solve linear program to find a candidate with larger L1 norm
        try:
            b = solve_lp(f, M)
        except Exception as e:
            print(f"LP failed at iteration {iteration}: {e}")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            print("Trivial LP solution, stopping.")
            break

        # rescale to same L1 norm as f
        g = (f_sum / sum_b) * b

        # try full replacement first
        new_f = g
        new_score = fast_score(new_f)
        if new_score < cur_score:
            f = new_f
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                print(f"Iter {iteration}: improved score to {best_score:.6f} (full replace)")

        else:
            # try convex combo with decreasing step
            # initial step size
            t = 0.5
            while t > 1e-4:
                new_f = (1 - t) * f + t * g
                new_score = fast_score(new_f)
                if new_score < cur_score:
                    f = new_f
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_f = f.copy()
                        print(f"Iter {iteration}: improved score to {best_score:.6f} (t={t:.4f})")
                    break
                t *= 0.5

        # early exit if target reached
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration} with score {best_score:.6f}")
            break

        # occasional re‑compute sum, which stays constant
        # no need to recompute f_sum (should stay same)
        # optionally print progress
        if iteration % 10 == 0:
            elapsed = time.time() - start
            print(f"Iter {iteration:05d} – score {cur_score:.6f} – elapsed {elapsed:.1f}s")

    # Return best sequence as Python list (non‑negative floats)
    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    return best_f.tolist()

import time, numpy as np, cvxpy as cp, scipy.sparse as sp

def _fast_score(seq):
    # compute score quickly
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2*n - 1]
    M = conv.max()
    return 2.0 * n * M / (s * s)

def _max_conv(seq):
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2*n - 1]
    return conv.max()

def _solve_lp(f_vec, M_val):
    n = f_vec.shape[0]
    # Build convolution matrix A such that A @ b = f*b
    diag_data = []
    offsets = []
    for d in range(n):
        diag_data.append(np.full(n, f_vec[d], dtype=np.float64))
        offsets.append(-d)
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M_val * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=2000, abstol=1e-7, reltol=1e-6, feastol=1e-7)
    # status check
    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"LP not solved optimally: {prob.status}")
    if b_var.value is None:
        raise RuntimeError("LP solution is None")
    return np.array(b_var.value, dtype=np.float64)

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Frank‑Wolfe optimisation for the convolution upper bound.
    Starts from the half‑binomial (Catalan) coefficients
    and repeatedly solves a linear programme that finds a direction
    with larger L¹‑norm while keeping the maximum self‑convolution
    below the current value.  A convex combination of the current
    vector and the LP solution strictly decreases the evaluation
    metric.  The loop runs until the time budget expires or the target
    score 1.503 is reached.
    Returns the best sequence found as a plain Python list.
    """
    import numpy as np, time, random

    np.random.seed(seed)
    # length of the step function; can be tuned
    n = 2500

    # Initialise Catalan (half‑binomial) coefficients
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2.0 * i)

    f = a.copy()
    f_sum = f.sum()
    cur_score = _fast_score(f)
    best_f = f.copy()
    best_score = cur_score
    print(f"Initial score: {best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # current maximal convolution value
        M = _max_conv(f)

        # Solve linear programme
        try:
            b = _solve_lp(f, M)
        except Exception as e:
            print(f"Iteration {iteration}: LP failed ({e}); terminating.")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            print(f"Iteration {iteration}: LP returned zero mass, stopping.")
            break

        # Rescale to keep the L1 norm constant
        g = (f_sum / sum_b) * b

        # Evaluate the candidate obtained by full replacement
        new_score = _fast_score(g)
        if new_score < cur_score:
            f = g
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                print(f"Iter {iteration:05d} – full replace improved score to {best_score:.6f}")
        else:
            # Try a convex combination with diminishing step size
            t = 0.5
            improved = False
            while t > 1e-4:
                f_candidate = (1 - t) * f + t * g
                cand_score = _fast_score(f_candidate)
                if cand_score < cur_score:
                    f = f_candidate
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_f = f.copy()
                        print(f"Iter {iteration:05d} – t={t:.4f} improved score to {best_score:.6f}")
                    improved = True
                    break
                t *= 0.5
            if not improved:
                # No improvement; keep current f unchanged
                pass

        # periodic status
        if iteration % 5 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {iteration:05d} – score {cur_score:.6f} – elapsed {elapsed:.1f}s")

        # early exit if target met
        if best_score < 1.503:
            print(f"Target achieved after {iteration} iterations, score {best_score:.6f}")
            break

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Return as plain Python list for the runner
    return best_f.tolist()

"""
Frank‑Wolfe style optimisation for the convolution upper‑bound.

The algorithm starts from the half‑binomial (Catalan) coefficients,
which give the classic score ≈ π/2.  At each iteration it solves a
linear programme

    maximise   Σ b_i
    subject to b_i ≥ 0
               (f * b)_k ≤ M   for all k,

where * is the linear convolution, *f* is the current sequence and
M = max_k (f*f)_k.  The constraints are linear in *b*; the matrix that
realises the convolution (f*b) is built as a sparse Toeplitz matrix.
The LP returns a vector b with a larger ℓ₁‑norm than f while still
respecting the same worst‑case convolution bound M.
After scaling b to have the same total mass as f we obtain a candidate g.
If g alone already reduces the score we replace f by g; otherwise we
take a convex combination (1–t) f + t g with a diminishing step size t.
The process strictly reduces the evaluation metric and stops when the
time budget expires or the target score 1.503 is reached.

Only standard scientific packages (numpy, scipy, cvxpy) are used; the
function returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the metric (identical to ``evaluate_sequence`` but
    without the safety checks).  Returns ``np.inf`` for degenerate inputs.
    """
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    M = conv.max()
    return 2.0 * n * M / (s * s)


def _max_conv(seq: np.ndarray) -> float:
    """Return the maximum value of the self‑convolution of *seq*."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return float(conv.max())


def _solve_lp(f_vec: np.ndarray, M_val: float) -> np.ndarray:
    """
    Solve
        maximise   Σ b_i
        subject to b_i ≥ 0
                    (f * b)_k ≤ M_val   for all k.
    The convolution matrix (f*b) is built as a sparse Toeplitz matrix.
    Returns the optimal vector ``b`` as a dense ``np.ndarray``.
    """
    n = f_vec.shape[0]

    # Each diagonal of the convolution matrix is constant and equal to f_vec[d].
    # Offsets are 0, -1, -2, …, -(n‑1).
    data = [float(v) for v in f_vec]           # list of scalar entries
    offsets = -np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M_val * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS works well for this modest‑size LP.
    prob.solve(
        solver='ECOS',
        warm_start=True,
        max_iters=2000,
        abstol=1e-6,
        reltol=1e-6,
        feastol=1e-6,
    )

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"LP not solved to optimality (status={prob.status})")
    if b.value is None:
        raise RuntimeError("LP returned no solution")
    return np.array(b.value, dtype=np.float64)


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step function for the given evaluation metric.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility (not used directly by the core
        optimisation but kept for API compatibility).
    budget_s : int
        Time budget in seconds.
    **kwargs
        Ignored; present for compatibility with the runner.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)

    # Length of the sequence – 2500 offers a good trade‑off between quality
    # and the size of the linear programme.
    n = 2500

    # ------------------------------------------------------------------
    # 1.  Initialise with the half‑binomial (Catalan) coefficients.
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    f = a.copy()
    f_sum = f.sum()                     # stays constant throughout the run
    cur_score = _fast_score(f)

    best_f = f.copy()
    best_score = cur_score
    print(f"Initial score: {best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # tiny safety margin
    iteration = 0

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (Frank‑Wolfe style).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # 2a.  Current worst self‑convolution value.
        M = _max_conv(f)

        # 2b.  Linear programme to find a direction with larger ℓ₁‑norm.
        try:
            b = _solve_lp(f, M)
        except Exception as exc:
            print(f"Iteration {iteration}: LP failed ({exc}); stopping.")
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            print(f"Iteration {iteration}: LP returned zero mass, aborting.")
            break

        # 2c.  Rescale to keep the ℓ₁‑norm unchanged.
        g = (f_sum / sum_b) * b

        # 2d.  Try to replace f by g completely; otherwise fall back to a
        #      convex combination with a decreasing step size.
        new_score = _fast_score(g)
        if new_score < cur_score:
            f = g
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                print(f"Iter {iteration:05d} – full replace improved score to {best_score:.6f}")
        else:
            # convex combination with halving step size
            t = 0.5
            improved = False
            while t > 1e-4:
                f_candidate = (1.0 - t) * f + t * g
                cand_score = _fast_score(f_candidate)
                if cand_score < cur_score:
                    f = f_candidate
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_f = f.copy()
                        print(
                            f"Iter {iteration:05d} – t={t:.4f} improved score to {best_score:.6f}"
                        )
                    improved = True
                    break
                t *= 0.5
            if not improved:
                # No improvement in this iteration – keep the current f.
                pass

        # occasional progress report
        if iteration % 5 == 0:
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:05d} – score {cur_score:.6f} – elapsed {elapsed:.1f}s"
            )

        # early exit if the target is already reached
        if best_score < 1.503:
            print(
                f"Target reached after {iteration} iterations with score {best_score:.6f}"
            )
            break

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Convert to a plain Python list as required by the evaluation harness.
    return best_f.tolist()
