def proj_simplex(v):
    # Project onto simplex {x >=0, sum x =1}
    # reference from Duchi et al. 2008
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    w = np.maximum(v - theta, 0)
    return w

def conv_fft(a):
    n = len(a)

def grad_of_active(a, conv, active_idxs):
    n = len(a)
    g = np.zeros_like(a)
    for k in active_idxs:
        i_low = max(0, k - (n - 1))

        j_vals = k - i_vals
        g[i_vals] += a[j_vals]

    if len(active_idxs)>0:
        g /= len(active_idxs)
    return g

assistantfinal**Solution Overview**

The objective  
\[
\Phi(a)=\frac{2n\;\max (a*a)}{(\sum a)^2}
\]
is *scale invariant*: multiplying the whole vector by a constant does not change the
score.  By fixing \(\sum a = 1\) the problem reduces to minimizing the peak of the
autocorrelation
\[
M(a)=\max_k (a*a)_k .
\]

For a fixed vector the function \(M(a)\) is the maximum of a finite family of
quadratic forms
\[
b_k(a)=\sum_{i}a_i a_{k-i}\qquad(k=0,\dots ,2n-2).
\]
Its sub‑gradient at a point is any convex combination of the gradients of the
active indices (those attaining the maximum).  
If *K* is the set of active indices,
\[
\partial M(a)=\operatorname{conv}\{\nabla b_k(a)\;|\;k\in K\},
\quad   \nabla b_k(a)_i = 
\begin{cases}
a_{k-i}& 0\le k-i < n\\[2pt]
0&\text{otherwise.}
\end{cases}
\]

We perform a **projected sub‑gradient descent** on the simplex  
\(\{a\ge 0,\;\sum a = 1\}\).  
Each iteration consists of

1. Compute the full convolution `c = a*a` (FFT based, \(O(n\log n)\)).
2. Determine the current peak `M = max(c)` and the active set
   `K = {k | c[k] >= M - ε}` (ε a small tolerance).
3. Build the averaged sub‑gradient  
   `g = (1/|K|) Σ_{k∈K} ∇b_k(a)`.
4. Take a step `a ← a – α_t g` with a diminishing step size  
   `α_t = α₀ / √t`.
5. Project the result back onto the simplex (Euclidean projection).

A few random restarts (different initial shapes: the classical
√‑difference, a power‑law, a random Dirichlet) are executed until the time budget
expires.  After each restart a short “polishing” phase performs
pairwise mass transfers to exploit possible local improvements.

The algorithm is completely self‑contained, uses only `numpy` and the given
`evaluate_sequence`, and respects the 1000 s budget.

"""
Projected sub‑gradient optimisation for the autocorrelation peak.
The algorithm repeatedly:
  * evaluates the full convolution via FFT,
  * builds a sub‑gradient from all active peak indices,
  * takes a diminishing step,
  * projects onto the probability simplex (non‑negative, sum = 1).

Multiple random restarts (sqrt‑difference, power‑law, random) are performed
within the allotted time.  A final local polishing stage swaps small amounts
of mass between two positions.  The resulting sequence is scaled to the
[0, 1000] range (scaling does not change the score).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    # Duchi et al. (2008) – O(n) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                        # all entries are zero or negative
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution a*a using FFT (real input → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)      # real inverse FFT
    return conv[: 2 * n - 1]                 # keep only needed part


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the maximal autocorrelation entry
# ----------------------------------------------------------------------
def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of b_k over the active indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_idx.size == 0:
        return g
    for k in active_idx:
        # all i such that 0 <= i < n and 0 <= k-i < n
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g /= float(active_idx.size)
    return g

# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function using projected sub‑gradient descent
    with several random restarts and a final polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Determine length of the sequence (use previously found one if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1542                       # default length used in earlier runs

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Parameters for the optimisation loop
    # ------------------------------------------------------------------
    max_iter_per_restart = 5000
    step0 = 0.5                # initial step size (will be divided by sqrt(t))
    epsilon_rel = 1e-6        # tolerance for defining the active set

    restart = 0
    # ------------------------------------------------------------------
    # Main loop over random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # ----- Initialise a candidate vector -----------------------------------------
        init_mode = np.random.choice(["sqrtdiff", "power", "random"])
        if init_mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif init_mode == "power":
            # power‑law a_i = (i+1)^{-p}  with random exponent p in [0.1, 1.8]
            p = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:  # "random"
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n)
        a = a / a.sum()                     # keep the sum exactly 1

        # ----- Sub‑gradient descent for the current restart -------------------------
        t = 0
        while time.time() < deadline and t < max_iter_per_restart:
            # 1) convolution and peak
            conv = _conv_fft(a)
            M = conv.max()

            # 2) active indices (within a relative tolerance)
            tol = max(epsilon_rel * M, 1e-12)
            active = np.where(conv >= M - tol)[0]

            # 3) sub‑gradient
            g = _subgrad_active(a, active)

            # 4) step (diminishing)
            step = step0 / math.sqrt(t + 1)
            a_candidate = a - step * g

            # 5) project back onto the simplex (non‑negative, sum = 1)
            a_candidate = _project_to_simplex(a_candidate)

            # 6) evaluate
            score = evaluate_sequence(a_candidate.tolist())
            if score < best_score:
                best_score = score
                best_seq = a_candidate.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart}] iter {t:04d} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            # Accept the new point unconditionally (helps to wander out of local minima)
            a = a_candidate
            t += 1

        # ----- Polishing phase for the best sequence found so far ------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(6.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # transfer a small fraction from i to j
                eps = min(a_local[i], a_local[i] * np.random.uniform(0.0, 0.05))
                if eps <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                a_new = _project_to_simplex(a_new)
                s = evaluate_sequence(a_new.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # Decide whether another restart is possible
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        best_seq = a_fallback / a_fallback.sum()

    # ------------------------------------------------------------------
    # Scale to the [0, 1000] range required by the evaluator (no effect on score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()

**Explanation of key choices**

* **Projection onto the simplex** – guarantees the denominator of the
  evaluation function stays constant (`sum a = 1`), so the problem reduces to
  minimizing the autocorrelation peak `max(a*a)`.
* **Active‑set sub‑gradient** – using *all* indices whose convolution value is
  within a tiny tolerance of the maximum yields a more informative descent
  direction than considering only the single peak index.
* **FFT‑based convolution** – reduces the cost of each iteration from
  \(O(n^2)\) to \(O(n\log n)\), allowing thousands of iterations within the
  time budget.
* **Multiple random restarts** – different initial shapes (square‑root‑difference,
  power‑law, uniform random) help escape the basins of attraction of poor local minima.
* **Polishing stage** – cheap pairwise mass transfers can still improve the peak after the main descent has converged.
