def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

* Compute the gradient of the smoothed objective analytically.
  For a weight vector w (soft‑max of the convolution) the gradient is

      ∇_j = 2 * Σ_i a_i · w_{i+j} ,

  which can be obtained by a single convolution w * a_rev.
* Perform a projected Adam update on the probability simplex (non‑negative,
  sum = 1).  The Euclidean projection onto the simplex is done by the
  standard O(n log n) algorithm.
* Occasionally inject a small random multiplicative perturbation to escape
  shallow local minima and keep a short simulated‑annealing style refinement
  at the end.

The whole loop runs until the time budget (≈1000 s) is exhausted and the
best sequence found (with the smallest true evaluation score) is returned.
"""

import time
import math
import numpy as np


def _half_order_series(n: int) -> np.ndarray:
    """Generate the first n coefficients a_i = C(2i,i)/4**i (arcsine series)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Returns the projected vector.
    """
    # https://arxiv.org/pdf/1101.6081.pdf  (Algorithm 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (fast for repeated calls).
    Returns the full convolution (len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights whose evaluation score is as low as
    possible (within the given time budget).  The algorithm is a projected
    Adam optimiser on a smoothed version of the max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a reasonable dimension.
    #    Larger n does not change the asymptotic bound of the arcsine shape,
    #    but gives the optimiser more freedom to flatten the tails.
    # ------------------------------------------------------------------
    n = 1500                # length of the step function
    a = _half_order_series(n)   # start from the arcsine shape, already sum=1

    # best solution seen so far (objective = 2·n·max(conv))
    conv = _conv_fft(a, a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam state (first and second moments of the gradient)
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters and schedules
    # ------------------------------------------------------------------
    # soft‑max temperature α : start small (smooth) → end large (sharp)
    alpha_start = 5.0
    alpha_end = 150.0
    # learning‑rate schedule (exponential decay)
    lr_start = 0.25
    lr_end = 0.002

    iteration = 0
    total_runtime = budget_s - 1.0   # a bit less than the full budget

    while time.time() < deadline:
        iteration += 1
        # progress ∈ [0,1] over the whole run
        elapsed = time.time() - start_time
        progress = min(elapsed / total_runtime, 1.0)

        # gradually raise α and lower the learning rate
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # --------------------------------------------------------------
        # 4. Current convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)                # length 2n‑1
        max_val = conv.max()
        # true objective (no need to call evaluate_sequence every iteration)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:6d}] new best score = {best_score:.6f}")

        # soft‑max over the convolution values (stabilised)
        shift = max_val
        exp_weights = np.exp(alpha * (conv - shift))
        w = exp_weights / exp_weights.sum()   # length 2n‑1, sum = 1

        # --------------------------------------------------------------
        # 5. Gradient of the smoothed objective
        #    ∇_j = 2 * Σ_i a_i * w_{i+j}
        #    can be obtained by convolving w with a reversed.
        # --------------------------------------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # keep the central n entries

        # --------------------------------------------------------------
        # 6. Adam update with projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)

        # --------------------------------------------------------------
        # 7. Occasional random jitter (simulated‑annealing style)
        # --------------------------------------------------------------
        if iteration % 2500 == 0:
            # multiplicative Gaussian noise, then re‑project
            noise = np.random.randn(n) * 0.015
            a_candidate = a_candidate * (1.0 + noise)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _simplex_projection(a_candidate)

        a = a_candidate

    # ------------------------------------------------------------------
    # 8. Final evaluation (using the provided evaluate_sequence for safety)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] best score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
