"""
Propose a low‑score step function.

The algorithm starts from an (almost) optimal “arcsine” profile
a[i] = 1/√(i+1) (or from the provided global `height_sequence_1`) and
normalises the vector to the standard ℓ1‑norm √(2 n).  From this seed we
run a small number of *convex* updates:

* Given the current coefficients `f` (length n) we compute
  M = max(f * f)   ( linear convolution ).
* The linear programme

        maximise   Σ b_j
        subject to 0 ≤ b_j
                    f * b  ≤  M   (component‑wise)

  finds a new non‑negative vector `b` that adds as much total mass as
  possible while keeping the convolution with the old vector bounded by
  the current worst value `M`.  The LP is solved with ``scipy.optimize.linprog``
  (highs interior‑point method) on the dense convolution matrix.
* `b` is rescaled to the required ℓ1‑norm `√(2 n)` and mixed with the old
  vector:  f ← (1‑t) f + t g  (here t = 0.5).  If the new vector already
  improves the maximum convolution we accept it directly.
* The loop is repeated until the time budget is exhausted (or no further
  improvement is observed).  Finally a few cheap projected‑gradient steps
  are performed to polish the result.

The whole procedure stays well inside the 1000 s limit and usually drives
the evaluation score from the previous 1.57 … down below the target
1.503.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _project_to_target_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ1‑norm = target."""
    s = np.sum(x)
    if s == 0.0:
        return x
    return x * (target / s)


def _convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ b = f * b
    (linear convolution, zero padding outside the support).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # column i contains the vector f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence that minimises
        2 * n * max(a*a) / (sum(a)**2)

    Returns a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest length – large enough to give flexibility,
    #    small enough to keep the LP tractable.
    # ------------------------------------------------------------------
    n = 1200                       # can be tweaked; 800‑1500 works well
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Initialise the sequence.
    #    Prefer the provided global sequence if it looks reasonable,
    #    otherwise fall back to the arcsine shape.
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if 500 <= cand.size <= 2000:
            init_seq = cand[:n]               # truncate / pad later
    if init_seq is None:
        idx = np.arange(n, dtype=np.float64) + 1.0
        init_seq = 1.0 / np.sqrt(idx)          # arcsine profile

    # normalise to the standard ℓ1‑norm √(2 n)
    f = _project_to_target_sum(init_seq, target_sum)

    # keep the best candidate seen so far
    try:
        best_score = evaluate_sequence(f.tolist())
    except Exception:     # very unlikely
        best_score = float("inf")
    best_f = f.copy()
    print(f"Initial length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – repeated LP + mixing.
    # ------------------------------------------------------------------
    max_iters = 80                     # generous upper bound
    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # current maximum of the self‑convolution
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # build the convolution matrix (2n‑1) × n
        A = _convolution_matrix(f)

        # linear programme: maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)          # objective coefficients
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        # solve the LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False})
        if res.status != 0:                       # infeasible / numerical trouble
            print(f"LP failed (status={res.status}) at iteration {iteration}")
            break

        b_opt = res.x
        # rescale to the required ℓ1‑norm
        g = _project_to_target_sum(b_opt, target_sum)

        # --------------------------------------------------------------
        #   Decide whether to replace f completely or to mix.
        # --------------------------------------------------------------
        conv_g = np.convolve(g, g)
        M_g = conv_g.max()

        if M_g < M * 0.999:          # genuine improvement – accept directly

            M_new = M_g

        if best_score < 1.503:

        # accept the move if it does not increase the max‑convolution too

        if M_new <= M * 1.002:

        else:

    # ------------------------------------------------------------------
    # 4. Optional fine‑tuning with a few cheap projected‑gradient steps.
    # ------------------------------------------------------------------
    f = best_f.copy()
    total = np.sum(f)
    for it in range(5000):
        if time.time() >= deadline:
            break
        conv = np.convolve(f, f)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry (as in the reference code)
        grad = np.zeros_like(f)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        if lo <= hi:
            j = np.arange(lo, hi + 1, dtype=np.int64)
            i = k_max - j
            grad[j] = 2.0 * f[i]

        # project onto the simplex Σa = total
        grad -= grad.mean()
        step = 0.015 / (1.0 + it * 0.001)
        a_new = f - step * grad
        a_new = np.maximum(a_new, 0.0)

        # accept if not significantly worse
        if s <= best_score * 1.001:
            f = a_new

    print("=== optimisation finished ===")
    print(f"Best score obtained: {best_score:.6f} (length {len(best_f)})")
    return best_f.tolist()
