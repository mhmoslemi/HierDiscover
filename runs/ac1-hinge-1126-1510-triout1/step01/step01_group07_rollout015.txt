def project_simplex(v):
    # projects onto {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project vector v onto probability simplex (nonnegative, sum=1)."""
    # v is np.ndarray
    # reference: Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all zeros? return uniform?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = len(init)
        if n == 0:
            raise Exception()
    except Exception:
        n = 3000
        init = generate_power_law(n, 0.505)

    # ensure nonneg, normalize to sum 1
    init = np.maximum(init, 0.0)
    init = init / init.sum()
    a = init.copy()

    best_seq = a.copy()
    # compute best directly
    c = np.convolve(a, a)
    best_score = 2.0 * n * c.max() / (a.sum()**2)  # should be same as evaluate_sequence
    print(f"Starting score: {best_score:.6f}")

    iteration = 0
    step_factor = 0.1  # initial step size coefficient

    while time.time() < deadline:
        # compute convolution and max index
        c = np.convolve(a, a)
        max_c = c.max()
        cur_score = 2.0 * n * max_c / (a.sum()**2)  # a.sum() should be 1
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration}] improved to {best_score:.6f}")

        # locate index with max conv
        k_max = np.argmax(c)

        # compute gradient for max index
        idx = np.arange(n)
        src_idx = k_max - idx
        mask = (src_idx >= 0) & (src_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[src_idx[mask]]

        eta = step_factor / np.sqrt(iteration + 1)

            a_new = 0.99 * a_new + 0.01 * random_dir
            a_new = simplex_projection(a_new)

"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm starts from a good known construction (the global
``height_sequence_1`` if it exists, otherwise a power‑law sequence with
exponent ≈0.505).  It then runs a projected sub‑gradient descent that
directly reduces the maximal entry of the self‑convolution.

For a current sequence ``a`` the convolution entry that attains the
maximum, say at index ``k_max``, is a quadratic form

    c_k = Σ_i a_i * a_{k-i} .

Its (sub‑)gradient with respect to ``a`` is
``g_i = 2 * a_{k_max-i}`` (zero outside the support).  Moving a small
step opposite to this gradient reduces the peak value.  After each
step the vector is projected onto the probability simplex
(non‑negative, sum = 1) using the O(n) algorithm of Duchi et al.
If a step does not improve the score the step size is shrunk; occasional
random jitter helps to escape plateaus.

The loop runs until the time budget (≈1000 s) expires and the best
sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all at top level, no closures)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a length‑``n`` power‑law sequence a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation follows Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the largest rho satisfying the condition
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive; return the uniform distribution
        theta = np.max(v) - 1.0 / n
        w = np.maximum(v - theta, 0.0)
        w /= w.sum()
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform if projection collapsed to zero
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step aimed at lowering the
    maximum entry of the self‑convolution.
    """
    n = a.shape[0]
    # full convolution (float64)
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)

    # build sub‑gradient g_i = 2 * a_{k_max-i} (zero out‑of‑range)
    idx = np.arange(n)
    src = k_max - idx
    mask = (src >= 0) & (src < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[src[mask]]

    # gradient descent step (moving opposite to the gradient)
    a_candidate = a - step * g
    # project back onto the simplex (non‑negative, unit‑sum)
    a_candidate = simplex_projection(a_candidate)
    return a_candidate
