"""
Propose a step‑function that beats the Catalan baseline.

The algorithm works in two phases:

1. **Linear‑programming expansion** –  
   For the current sequence `f` we compute its peak `M = max(f*f)`.  
   We then solve a linear program

        maximise   Σ g_i
        subject to conv_f_g(k) ≤ M  (k = 0 … 2n‑2)
                  0 ≤ g_i ≤ 1000

   where `conv_f_g(k) = Σ_i f_i·g_{k‑i}` is linear in the unknown vector `g`.  
   The LP pushes as much mass as possible into indices that have only a
   weak influence on the convolution peak, therefore increasing the L¹‑norm
   without enlarging the peak.

2. **Optimal mixing** –  
   With the old vector `f` and the LP‑solution `g` we minimise the
   evaluation metric over the one‑dimensional family

        h(t) = (1‑t)·f + t·g ,   t∈[0,1].

   The self‑convolution of `h(t)` can be written analytically as a quadratic
   combination of the three known convolutions `f*f`, `g*g` and `f*g`.
   A short golden‑section search finds the `t` that minimises the
   evaluation score `2·n·max(h*h) / (Σh)²`.  If the score improves, the new
   sequence becomes the current one.

After a few LP‑mixing steps (usually only 1–3 are enough) we fall back to
a lightweight hill‑climber that moves a small amount of mass from indices
with a large gradient (i.e. those that contribute most to the current peak)
to indices with a small gradient.  The whole procedure respects the
non‑negativity and 0 … 1000 bounds and finishes well within the 1000 s
budget.

The returned list can be fed directly to `evaluate_sequence`.
"""
import time
import numpy as np

# optional imports – the LP phase is used only when SciPy is available
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    sp = None
    linprog = None


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Fast convolution via FFT (real FFT, full length)."""
    n = a.size + b.size - 1
    L = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[:n]
    return conv


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A for the linear constraints
        (A @ g) <= M,
    where (A @ g)[k] = Σ_i f_i * g_{k-i} = (f * g)[k].
    """
    n = f.size
    rows = []
    cols = []
    data = []
    # each row k corresponds to convolution index k
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in [i_min, i_max] we have coefficient f[k-i]
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        rows.extend([k] * idx.size)
        cols.extend(idx.tolist())
        data.extend(f[k - idx].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)
    return A


def _line_search_best_t(conv_f, conv_g, conv_fg, sum_f, sum_g, n,
                        max_iters: int = 25):
    """
    Golden‑section search on t∈[0,1] to minimise the evaluation metric.
    Returns the best t and the corresponding score.
    """
    lo, hi = 0.0, 1.0
    phi = (1.0 + 5**0.5) / 2.0
    for _ in range(max_iters):
        t1 = hi - (hi - lo) / phi
        t2 = lo + (hi - lo) / phi

        conv1 = (1 - t1) ** 2 * conv_f + t1 ** 2 * conv_g + 2 * t1 * (1 - t1) * conv_fg
        M1 = conv1.max()
        s1 = (1 - t1) * sum_f + t1 * sum_g
        score1 = 2.0 * n * M1 / (s1 * s1)

        conv2 = (1 - t2) ** 2 * conv_f + t2 ** 2 * conv_g + 2 * t2 * (1 - t2) * conv_fg
        M2 = conv2.max()
        s2 = (1 - t2) * sum_f + t2 * sum_g
        score2 = 2.0 * n * M2 / (s2 * s2)

        if score1 < score2:
            hi = t2
        else:
            lo = t1
    t_best = (lo + hi) / 2.0
    conv_best = (1 - t_best) ** 2 * conv_f + t_best ** 2 * conv_g + 2 * t_best * (1 - t_best) * conv_fg
    M_best = conv_best.max()
    s_best = (1 - t_best) * sum_f + t_best * sum_g
    score_best = 2.0 * n * M_best / (s_best * s_best)
    return t_best, score_best

        init_seq[0] = 1.0
        for i in range(1, n):
            init_seq[i] = init_seq[i - 1] * (2 * i - 1) / (2.0 * i)
        print(f"Generated Catalan start (len={n})")
    cur = init_seq.copy()
    n = cur.size
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. LP‑based expansion + optimal mixing
    # ------------------------------------------------------------------
    if sp is not None and linprog is not None:
        lp_iter = 0
        while time.time() < deadline and best_score > 1.503 and lp_iter < 10:
            lp_iter += 1
            conv_f = _conv_fft(cur, cur)
            M = conv_f.max()
            sum_f = cur.sum()

            # Build constraints A @ g ≤ M
            print(f"[LP {lp_iter}] Building constraint matrix (n={n})")
            A = _build_constraint_matrix(cur)
            b = np.full(2 * n - 1, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)          # maximise Σg_i
            bounds = [(0.0, 1000.0)] * n

            # Solve the LP – give it the remaining time budget
            time_left = max(0.0, deadline - time.time())
            options = {'presolve': True, 'time_limit': time_left}
            print(f"[LP {lp_iter}] Solving linear programme (time limit ≈{time_left:.1f}s)…")
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options=options)

            if not res.success:
                print(f"[LP {lp_iter}] LP failed (status {res.status}): {res.message}")
                break

            g = np.maximum(0.0, res.x)          # clip tiny negatives
            sum_g = g.sum()
            if sum_g <= sum_f + 1e-9:
                print(f"[LP {lp_iter}] No increase in total mass – stop LP phase.")
                break

            # Compute the three required convolutions
            conv_g = _conv_fft(g, g)
            conv_fg = _conv_fft(cur, g)

            # Find the optimal mixing proportion
            t_opt, score_opt = _line_search_best_t(conv_f, conv_g, conv_fg,
                                                   sum_f, sum_g, n,
                                                   max_iters=25)
            new_seq = (1.0 - t_opt) * cur + t_opt * g
            new_seq = np.clip(new_seq, 0.0, 1000.0)
            new_score = evaluate_sequence(new_seq.tolist())
            print(f"[LP {lp_iter}] best t={t_opt:.4f}, score={new_score:.6f}")

            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                cur = new_seq
                print(f"*** New best after LP #{lp_iter}: {best_score:.6f}")
            else:
                # even if not better we keep the mixture as the new base – it may
                # allow a further LP to increase the mass further.
                cur = new_seq

            if best_score < 1.503:
                print(f"Target reached after LP iteration {lp_iter}.")
                break
    else:
        print("SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 2. Local hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    it = 0
    # Pre‑compute a convolution once and reuse its peak index for a while
    conv_cur = _conv_fft(cur, cur)
    while time.time() < deadline and best_score > 1.503 and it < 2_000_000:
        it += 1

        # Refresh convolution and peak every few thousand iterations
        if it % 1000 == 0:
            conv_cur = _conv_fft(cur, cur)
            max_val = conv_cur.max()
            peak_idxs = np.where(conv_cur >= max_val * 0.999)[0]
            k_peak = np.random.choice(peak_idxs)

        # gradient of the current peak k_peak w.r.t. each a_i ( = 2·a_{k−i} )
        low = max(0, k_peak - (n - 1))
        high = min(n - 1, k_peak)
        idxs = np.arange(low, high + 1, dtype=np.int64)
        grad = np.zeros(n, dtype=np.float64)
        grad[idxs] = 2.0 * cur[k_peak - idxs]

        # pick a source with large gradient and positive mass
        src_candidates = np.where((grad > 0.0) & (cur > 1e-12))[0]
        if src_candidates.size == 0:
            continue
        src = src_candidates[np.argmax(grad[src_candidates])]

        # pick a target where the gradient is tiny (prefer far indices)
        tgt = np.argmin(grad)
        if tgt == src:
            continue

        # move a random fraction (up to 50 %) of the source to the target
        frac = np.random.rand() * 0.5
        delta = cur[src] * frac
        # respect the global upper bound 1000
        if cur[tgt] + delta > 1000.0:
            delta = 1000.0 - cur[tgt]
        if delta <= 0.0:
            continue

        trial = cur.copy()
        trial[src] -= delta
        trial[tgt] += delta
        trial_score = evaluate_sequence(trial.tolist())

        if trial_score < best_score:
            best_score = trial_score
            best_seq = trial.copy()
            cur = trial
            print(f"Iter {it:07d} – improved score: {best_score:.6f}")

        # occasional random shake to escape tiny plateaus
        if it % 50000 == 0:
            shake = cur.copy()
            idxs = np.random.randint(0, n, size=10)
            shake[idxs] = np.clip(shake[idxs] + np.random.randn(10) * 0.02,
                                  0.0, None)
            shake_score = evaluate_sequence(shake.tolist())
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                cur = shake
                print(f"Iter {it:07d} – shake improved score: {best_score:.6f}")

    print(f"Finished optimisation. Best score = {best_score:.6f}")
    return best_seq.tolist()
