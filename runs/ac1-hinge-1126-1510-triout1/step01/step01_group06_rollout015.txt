import numpy as np
from scipy.optimize import linprog

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense convolution matrix A such that (A @ g) = np.convolve(f, g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j holds f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Frank‑Wolfe optimiser for the step‑function problem.

    The method works on a probability vector `f` (non‑negative entries,
    sum = 1).  In each iteration it solves a linear programme that finds
    a vector `g` with maximal L1 mass while respecting the current
    convolution peak `M = max(f * f)`.  A convex combination
    `f ← (1‑η) f + η g/∑g` is taken; the step size η is decreased
    slowly.  When progress stalls a tiny random jitter is added.
    The loop stops before the supplied time budget expires and the best
    vector seen (scaled to the [0, 1000] range required by
    ``evaluate_sequence``) is returned.
    """
    import numpy as np
    import time
    from scipy.optimize import linprog

    # ---- configuration -------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # keep a safety margin

    # length of the step function
    try:
        n = len(height_sequence_1)   # noqa: F821 – may be defined globally
    except Exception:
        n = 1542  # default length that was used previously

    # ---- initial profile ------------------------------------------------
    # sqrt‑difference (discrete derivative of √x) – a well‑known low‑peak kernel
    f = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    f = f / f.sum()  # normalise to a probability distribution

    # best solution tracking
    best_f = f.copy()
    best_M = np.convolve(f, f).max()
    best_score = 2.0 * n * best_M   # same as evaluate_sequence after scaling
    print(f"Initial peak {best_M:.8f} → score {best_score:.6f}")

    # ---- optimisation loop ---------------------------------------------
    iteration = 0
    eta = 0.5                      # initial convex‑combination weight
    eta_decay = 0.99               # multiplicative decay
    eta_min = 0.01
    no_improve = 0
    max_no_improve = 200

    while time.time() < deadline:
        # 1. current peak bound
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # 2. build convolution matrix A (dense, but modest size)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        # 3. linear programme: maximise sum(g)  s.t. A @ g ≤ M, g ≥ 0
        c = -np.ones(n)  # linprog minimises → maximise sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # 'highs' is fast and handles medium‑sized problems well
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0 or np.isnan(sum_g):
            print(f"Degenerate LP solution at iteration {iteration}")
            break

        # 4. normalise g to keep the total mass equal to 1
        g_norm = g / sum_g

        # 5. convex update with decreasing step size
        eta = max(eta * eta_decay, eta_min)
        f_new = (1.0 - eta) * f + eta * g_norm
        f_new = np.clip(f_new, 0.0, None)
        f_new /= f_new.sum()          # protect against accumulated rounding error

        # 6. evaluate new peak and score
        new_M = np.convolve(f_new, f_new).max()
        new_score = 2.0 * n * new_M

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            best_M = new_M
            print(f"Iter {iteration:4d}: improved score {best_score:.6f} (η={eta:.4f})")
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        iteration += 1

        # occasional random perturbation if we seem stuck
        if no_improve >= max_no_improve:
            jitter = np.random.randn(n) * 1e-3
            f = np.clip(f + jitter, 0.0, None)
            f /= f.sum()
            print(f"Iter {iteration}: jitter injected")
            no_improve = 0

        # safety stop if we are already extremely close to the theoretical lower bound
        # theoretical bound = 2 * n / (2 * n - 1)
        if best_score <= 1.05:   # well below the target, we can finish early
            break

    # ---- final polishing (tiny local swaps) ---------------------------
    polishing_deadline = time.time() + min(5.0, deadline - time.time())
    f = best_f.copy()
    while time.time() < polishing_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = min(f[i], f[i] * 0.01)

        f_new = f.copy()
        f_new[i] -= eps

        if new_score < best_score:

            best_f = f_new.copy()
            best_M = new_M
            f = f_new

    final_seq = best_f * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return final_seq.tolist()

"""
Frank‑Wolfe optimiser for the step‑function problem.

The algorithm works on a probability vector `f` (non‑negative entries,
sum = 1).  In each iteration it solves a *linear programme* that finds a
vector `g` with maximal L¹‑mass while respecting the current convolution
peak

    M = max(f * f) ,

i.e. it enforces  (f * g)[k] ≤ M  for every k.  Afterwards a convex
combination

    f ← (1‑η)·f + η·g/∑g

is taken (η is slowly reduced).  When progress stalls a tiny random
jitter is added.  The loop runs until the supplied time budget (≈1000 s)
expires; the best sequence found is finally scaled to the [0, 1000] range
required by ``evaluate_sequence`` and returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that (A @ g) == np.convolve(f, g).
    For length n the matrix has shape (2·n‑1, n) and each column j
    contains the vector f shifted by j positions.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The method follows the Frank‑Wolfe scheme
    described in the problem statement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer environment
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Determine the length and an initial feasible vector f
    # ------------------------------------------------------------------
    try:
        # a previously discovered construction may be available globally
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # noqa: F821
        n = len(init_seq)
        f = np.maximum(init_seq, 0.0)
        if f.sum() == 0.0:
            raise ValueError
        f = f / f.sum()
    except Exception:
        # fall back to the classic sqrt‑difference (discrete derivative of √x)
        n = 1542
        f = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
        f = f / f.sum()

    # ------------------------------------------------------------------
    # 2. Track the best solution seen so far
    # ------------------------------------------------------------------
    best_f = f.copy()
    best_M = np.convolve(f, f).max()
    best_score = 2.0 * n * best_M          # identical to evaluate_sequence after scaling
    print(f"Initial peak M={best_M:.8e} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main Frank‑Wolfe loop
    # ------------------------------------------------------------------
    iteration = 0
    eta = 0.5                # initial convex‑combination weight
    eta_decay = 0.99
    eta_min = 0.01
    no_improve = 0
    max_no_improve = 200    # after this many stagnant steps we inject jitter

    while time.time() < deadline:
        # current peak bound
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # build the convolution matrix A for the constraint A @ g ≤ M
        A = _conv_matrix(f)

        # linear programme : maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[Iter {iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0 or not np.isfinite(sum_g):
            print(f"[Iter {iteration}] Degenerate LP solution")
            break

        # normalise g to have the same total mass as f (i.e. 1)
        g_norm = g / sum_g

        # convex update – step size η slowly shrinks
        eta = max(eta * eta_decay, eta_min)
        f_new = (1.0 - eta) * f + eta * g_norm
        f_new = np.clip(f_new, 0.0, None)
        f_new /= f_new.sum()          # protect against rounding drift

        # evaluate the new peak
        new_M = np.convolve(f_new, f_new).max()
        new_score = 2.0 * n * new_M

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            best_M = new_M
            print(f"[Iter {iteration:4d}] improved → score {best_score:.6f} (η={eta:.4f})")
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        iteration += 1

        # if we have stalled for a while, perturb a little to escape flat regions
        if no_improve >= max_no_improve:
            jitter = np.random.randn(n) * 1e-3
            f = np.clip(f + jitter, 0.0, None)
            f /= f.sum()
            print(f"[Iter {iteration}] jitter injected")
            no_improve = 0

        # early stop when we are already very close to the theoretical limit
        # theoretical lower bound = 2·n / (2·n−1)
        if best_score <= 1.05:
            break

    # ------------------------------------------------------------------
    # 4. Optional short polishing phase (tiny local moves)
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    f = best_f.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = min(f[i], f[i] * 0.01)
        if eps == 0:
            continue
        f_new = f.copy()
        f_new[i] -= eps
        f_new[j] += eps
        f_new /= f_new.sum()
        new_M = np.convolve(f_new, f_new).max()
        new_score = 2.0 * n * new_M
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            best_M = new_M
            f = f_new
            print(f"[Polish] improved → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Scale to the [0, 1000] range required by `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_f * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return final_seq.tolist()
