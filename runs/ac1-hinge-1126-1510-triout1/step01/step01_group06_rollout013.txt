"""
Propose a coefficient sequence that minimizes the evaluation function.

Algorithm Overview
------------------
1. **Warm‑start** – try a family a_i ∝ (i+δ)^(-p) (p≈0.5 gives the classic
   √‑difference shape) and keep the best of a quick parameter sweep.

2. **Projected sub‑gradient descent** – repeatedly move a in the direction that
   reduces the current convolution peak (the active index of a*a).  The step
   direction is projected to keep the L1‑norm fixed.  Temperature and step size
   are annealed.

3. **Linear‑programming refinement** – given the current sequence ``a`` and its
   peak M = max(conv(a,a)), solve the LP

       maximise   ∑ g_j
       subject to conv(a, g) ≤ M   (2 n‑1 linear inequalities)
                  g_j ≥ 0

   The LP pushes mass to positions where ``a`` is small while respecting the
   same peak bound.  After normalising ``g`` we form convex combinations
   ``a_new = (1‑t)·a + t·g`` for several candidate mixing factors ``t`` and keep
   the best improvement.

4. **Polishing** – a short random‑move local search (transfer a small amount of
   mass between two indices) to catch any remaining tiny improvements.

All operations respect the non‑negativity and L1‑norm constraints; the final
sequence is scaled to the full [0, 1000] range (the score is scale‑invariant).

The routine stops automatically after the allocated time budget (default
≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _conv(seq: np.ndarray) -> np.ndarray:
    """Full convolution of a non‑negative sequence with itself."""
    return np.convolve(seq, seq)


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


def _initial_candidates(n: int, time_limit: float) -> tuple[np.ndarray, float]:
    """Run a cheap parametric sweep and return the best start sequence."""
    best_seq = None
    best_sc = float("inf")
    start = time.time()
    # Parameter sweep: a_i ∝ (i+δ)^(-p)  (δ avoids division by zero)
    deltas = [0.5, 1.0]
    ps = np.linspace(0.30, 0.80, 11)          # around 0.5 (sqrt‑difference)
    for δ in deltas:
        for p in ps:
            i = np.arange(n, dtype=np.float64)
            a = (i + δ) ** (-p)
            a /= a.sum()                       # L1‑normalise
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_seq = a.copy()
            # early abort if time is out
            if time.time() - start > time_limit:
                return best_seq, best_sc
    # Fallback: classic sqrt‑difference shape
    if best_seq is None:
        a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(
            np.arange(0, n, dtype=np.float64)
        )
        a = a / a.sum()
        best_seq = a
        best_sc = _score(best_seq)
    return best_seq, best_sc


def _subgradient_step(a: np.ndarray, step_scale: float, temperature: float) -> np.ndarray:
    """One projected sub‑gradient descent step targeting the current max‑convolution."""
    n = a.size
    conv = _conv(a)
    k_max = int(np.argmax(conv))

    # Gradient of the active quadratic term b_k = Σ a_i a_{k-i}
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    if i_low <= i_high:
        i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_vals = k_max - i_vals
        grad[i_vals] = a[i_vals] + a[j_vals]

    # Project to the tangent plane (zero‑sum) to keep L1‑norm unchanged
    direction = -grad + grad.mean()

    # Adaptive step length
    step = step_scale * temperature * (np.linalg.norm(direction) / n + 1e-12)
    step = max(step, 1e-8)

    a_new = a + step * direction
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() == 0.0:
        return a
    a_new /= a_new.sum()
    return a_new


def _build_conv_matrix(a: np.ndarray) -> sparse.csc_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(a, g).
    Row k (0‑based) contains a_{k‑j} for each column j.
    """
    n = a.size
    # For offset i (i ≥ 0) the diagonal consists of a_i repeated n times.
    diags_vals = [np.full(n, a_i, dtype=np.float64) for a_i in a]
    offsets = np.arange(n)                     # 0 … n‑1
    A = sparse.diags(diags_vals, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_refinement(a: np.ndarray, M: float, remaining_time: float) -> np.ndarray:
    """
    Perform one LP refinement step:
      maximise Σ g_j  subject to conv(a, g) ≤ M, g ≥ 0.
    Return a new sequence obtained by mixing a with the normalised LP solution.
    """
    n = a.size
    # Build the constraint matrix A (full) once per call.
    A = _build_conv_matrix(a)

    # rhs vector – the current peak bound for every convolution entry.
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: maximise Σ g_j  <=> minimise −Σ g_j.
    c = -np.ones(n, dtype=np.float64)

    # Bounds: g_j ≥ 0, no explicit upper bound.
    bounds = [(0, None) for _ in range(n)]

    # Solve with HiGHS (fast for moderate‑size LPs).
    # Use a modest time limit per LP to avoid overshooting the overall budget.
    lp_time_limit = min(30.0, remaining_time * 0.2)  # at most 30 s or 20% of remaining
    options = {"presolve": True, "time_limit": lp_time_limit}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=options)

    if res.status != 0 or res.x is None:
        # LP failed or timed out – fall back to the current a.
        return a

    g_raw = res.x
    if g_raw.sum() == 0.0:
        return a

    # Normalise g to have L1 = 1 (same scale as a).
    g = g_raw / g_raw.sum()

    # Try several mixing factors and keep the best improvement.
    best_candidate = a
    best_sc = _score(a)
    for t in np.linspace(0.05, 0.45, 9):   # 5% … 45%
        cand = (1.0 - t) * a + t * g
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_candidate = cand.copy()
    return best_candidate

                best_seq = a_lp.copy()
                a = a_lp
                print(f"[lp]  Improved to {best_score:.6f} (t={time.time()-start_time:.1f}s)")
                # after an LP step we tighten the step parameters a bit
                step_scale = max(step_scale * 0.9, min_step_scale)
                temperature = max(temperature * 0.9, 0.1)
            lp_cooldown = time.time()  # enforce spacing between LP attempts

    # ------------------------------------------------------------------
    # 4. Final polishing – short random‑move local search
    # ------------------------------------------------------------------
    polishing_deadline = min(deadline, time.time() + 6.0)
    a = best_seq.copy()
    while time.time() < polishing_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = min(a[i], a[i] * 0.03)
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new /= a_new.sum()
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            a = a_new
            print(f"[polish] Improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Rescale to the [0,1000] range (no effect on the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Returning final sequence with score {best_score:.6f}")
    return final_seq.tolist()
