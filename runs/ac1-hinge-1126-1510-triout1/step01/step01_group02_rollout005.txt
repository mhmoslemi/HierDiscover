def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # choose n based on time budget, maybe 2000
    n = 2000
    # generate Catalan
    a = np.empty(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2.0*i)
    # optionally scale to sum sqrt(2n) to get sum approx 63? sum about sqrt(2n)?? Actually sum maybe approx
    f = a  # start
    sum_f = f.sum()
    M = _conv_max(f)  # compute max conv
    best_seq = f.copy()
    best_score = _fast_score(f)
    print(f"Start: n={n}, initial score={best_score:.6f}, M={M:.6f}")

    # warmstart for LP
    warm_b = None

    iteration = 0
    # parameters
    alpha = 0.4
    min_alpha = 0.05
    alpha_decay = 0.995  # per iteration

    while time.time() < deadline:
        iteration += 1
        # solve LP
        b_opt, status = solve_lp(f, M, warmstart=warm_b)
        if b_opt is None or status not in ("optimal", "optimal_inaccurate"):
            # if fail, break
            print("LP solve failed or not optimal, breaking")
            break
        warm_b = b_opt  # for next warm start

        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            print("LP returned zero vector, break")
            break

        # scale to same sum as f
        g = b_opt * (sum_f / sum_b)

        # compute self-convolution max of g
        M_g = _conv_max(g)
        # compute score of g
        score_g = _fast_score(g)

        improvement = False

        if M_g < M - 1e-8:
            # adopt g directly
            f = g
            M = M_g
            cur_score = score_g
            improvement = True
            print(f"Iter {iteration:05d} – direct replace improved: score={cur_score:.6f}, M={M:.6f}")
        else:
            # try mixture
            new_f = (1 - alpha) * f + alpha * g
            M_new = _conv_max(new_f)
            score_new = _fast_score(new_f)
            if score_new < best_score - 1e-6:
                f = new_f
                M = M_new
                cur_score = score_new
                improvement = True
                print(f"Iter {iteration:05d} – mixed improve: score={cur_score:.6f}, alpha={alpha:.3f}, M={M:.6f}")
            else:
                # maybe accept with small probability? but for now just skip
                # optionally continue with same f
                cur_score = best_score  # unchanged

        if score_g < best_score:
            best_score = score_g
            best_seq = g.copy()

            # recompute M
            M = _conv_max(f)

"""...

"""
import time, random, numpy as np
import cvxpy as cp

def _fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence: returns objective value for a 1‑D numpy array."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum value of the self‑convolution of seq (no safety checks)."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return float(conv.max())

def _solve_lp(f: np.ndarray, M: float, warm_start: np.ndarray = None):
    """Linear program:
    maximize   sum(b)
    subject to  b >= 0
                max( conv(f, b) ) <= M
    Returns the optimal b (numpy array) and the solver status.
    """
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)
    # linear convolution of constant f and variable b
    conv_expr = cp.conv(f, b)  # length 2*n-1
    constraints = [cp.max(conv_expr) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start is not None:
        b.value = warm_start
    # ECOS works well for this size; fallback to SCS if needed
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)
    if b.value is None:
        # second attempt with a different solver
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)
    return b.value, prob.status

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.

    The algorithm starts from the Catalan (half‑binomial) coefficients,
    then iteratively solves a linear program that finds a direction `b`
    with maximal total mass while keeping the cross‑convolution ``f*b``
    below the current peak ``M``.  A convex combination of the current
    sequence `f` and the scaled direction `b` is taken; this reduces the
    self‑convolution peak and therefore the final score.  The loop runs
    until the allotted time expires or the target score (≈1.503) is reached.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with Catalan (half‑binomial) coefficients.
    # ------------------------------------------------------------------
    n = 2000                       # length – can be tuned
    f = np.empty(n, dtype=np.float64)
    f[0] = 1.0
    for i in range(1, n):
        f[i] = f[i - 1] * (2 * i - 1) / (2.0 * i)

    # Normalise sum to the theoretical Catalan value sqrt(4n/π) –
    # this does not affect the objective but yields a more stable scale.
    target_sum = np.sqrt(4.0 * n / np.pi)
    f = f * (target_sum / f.sum())

    sum_f = f.sum()
    M = _conv_max(f)
    best_f = f.copy()
    best_score = _fast_score(f)
    print(f"Initial Catalan‑type sequence: n={n}, score={best_score:.6f}, M={M:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (LP + mixing)
    # ------------------------------------------------------------------
    warm_b = None           # for warm‑starting the LP
    alpha = 0.35            # mixing weight for the new direction
    min_alpha = 0.02
    decay_factor = 0.99
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # --- solve the LP: find b with maximal total mass under current bound ---
        b_opt, status = _solve_lp(f, M, warm_start=warm_b)
        if b_opt is None or status not in ("optimal", "optimal_inaccurate"):
            print(f"Iteration {iteration}: LP not optimal (status={status}). Stopping.")
            break
        warm_b = b_opt  # keep for the next iteration

        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            print(f"Iteration {iteration}: LP returned zero vector. Stopping.")
            break

        # Scale b to have the same total mass as f.
        g = b_opt * (sum_f / sum_b)

        # Evaluate the candidate.
        M_g = _conv_max(g)
        score_g = _fast_score(g)

        # Try a direct replacement if it already improves the peak.
        if M_g < M - 1e-12:
            f = g
            M = M_g
            cur_score = score_g
            print(f"Iter {iteration:05d} – direct replace: score={cur_score:.6f}, M={M:.6f}")
        else:
            # Mix the current sequence with the new direction.
            new_f = (1.0 - alpha) * f + alpha * g
            M_new = _conv_max(new_f)
            cur_score = _fast_score(new_f)

            # Accept the mixture only if we make progress.
            if cur_score < best_score - 1e-8:
                f = new_f
                M = M_new
                print(f"Iter {iteration:05d} – mixed move: score={cur_score:.6f}, alpha={alpha:.3f}, M={M:.6f}")
                # increase mixing weight slightly after a successful step
                alpha = min(0.6, alpha / decay_factor)
                last_improve_iter = iteration
            else:
                # No improvement – shrink the mixing weight.
                alpha = max(min_alpha, alpha * decay_factor)

        # Keep the overall best sequence seen so far.
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"Iter {iteration:05d} – new global best: score={best_score:.6f}")

        # Early exit if we have already reached the target bound.
        if best_score < 1.503:
            print(f"Target bound reached at iteration {iteration} with score {best_score:.6f}")
            break

        # Occasionally perturb the current solution to escape a flat region.
        if iteration % 30 == 0 and iteration - last_improve_iter > 100:
            # Add a tiny random bump to the tail of the sequence.
            bump = np.zeros_like(f)
            tail_start = n // 2
            bump[tail_start:] = np.random.rand(n - tail_start) * 0.005
            f = np.maximum(0.0, f + bump)
            f = f * (sum_f / f.sum())          # re‑normalise total mass
            M = _conv_max(f)
            print(f"Iter {iteration:05d} – random bump applied, new M={M:.6f}")

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Convert to plain Python list as required by the runner.
    return best_f.tolist()

"""
Iterative LP‑based optimisation of a non‑negative step function.

The construction starts from the Catalan (half‑binomial) coefficients,
which have a constant self‑convolution (M≈1) and give a score ≈π/2≈1.57.
In each iteration we solve a linear programme

    maximise   Σ b_i
    subject to  b ≥ 0
                (f * b)_k ≤ M   for all k

where * denotes discrete convolution and M is the current peak of the
self‑convolution of f.  The solution b places as much mass as possible
in positions that do not increase the peak.  After scaling b to keep the
total mass unchanged we mix it into the current sequence f; this reduces
the peak and consequently the final evaluation score.
The loop runs until the time budget expires (≈1000 s) or the target
score 1.503 is reached.
"""

import time
import random
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` (no safety checks)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the self‑convolution of `seq` (no safety checks)."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return float(conv.max())


def _solve_lp(f: np.ndarray, M: float, warm_start: np.ndarray = None):
    """
    Linear programme:
        maximise   sum(b)
        subject to  b ≥ 0
                    conv(f, b) ≤ M (component‑wise)

    Returns the optimal b (numpy array) and the solver status string.
    """
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)

    # Linear convolution of the constant vector f with the variable b.
    conv_expr = cp.conv(cp.Constant(f), b)          # length 2*n‑1

    # Upper‑bound vector (same bound for every entry).
    M_vec = np.full(2 * n - 1, M, dtype=np.float64)

    constraints = [conv_expr <= cp.Constant(M_vec)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    if warm_start is not None:
        # provide a warm start for the solver
        b.value = warm_start

    # Try a fast interior‑point method first; fall back to SCS if needed.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=3000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)

    if b.value is None:
        # last resort: another call to SCS
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)

    return b.value, prob.status


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the given
    evaluation function.  The method follows the LP‑based scheme described
    above.  The returned object is a plain Python list of floats.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise with Catalan (half‑binomial) coefficients.
    # ------------------------------------------------------------------
    n = 2000                                             # length (tunable)
    f = np.empty(n, dtype=np.float64)
    f[0] = 1.0
    for i in range(1, n):
        f[i] = f[i - 1] * (2 * i - 1) / (2.0 * i)

    # Normalise the total mass to the theoretical Catalan sum √(4n/π).
    # This does not affect the objective (scale‑invariant) but stabilises
    # the numerical range.
    target_sum = np.sqrt(4.0 * n / np.pi)
    f = f * (target_sum / f.sum())
    sum_f = f.sum()

    M = _conv_max(f)
    best_f = f.copy()
    best_score = _fast_score(f)
    print(f"Initial Catalan‑type sequence: n={n}, score={best_score:.6f}, M={M:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    warm_b = None               # warm‑start vector for the LP
    alpha = 0.35                # mixing weight for the new direction
    min_alpha = 0.02
    decay = 0.99
    iteration = 0
    last_improve = 0

    while time.time() < deadline:
        iteration += 1

        # ---- LP: find a direction with maximal total mass under current bound.
        b_opt, status = _solve_lp(f, M, warm_start=warm_b)
        if b_opt is None or status not in ("optimal", "optimal_inaccurate"):
            print(f"Iter {iteration}: LP failed (status={status}). Stopping.")
            break
        warm_b = b_opt.copy()

        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            print(f"Iter {iteration}: LP returned zero vector. Stopping.")
            break

        # Scale b to keep the total mass unchanged.
        g = b_opt * (sum_f / sum_b)

        # Evaluate candidate g.
        M_g = _conv_max(g)
        score_g = _fast_score(g)

        # Direct replacement if the peak already improves.
        if M_g < M - 1e-12:
            f = g
            M = M_g
            cur_score = score_g
            print(f"Iter {iteration:05d} – direct replace: score={cur_score:.6f}, M={M:.6f}")
        else:
            # Convex combination of the current f and the new direction g.
            new_f = (1.0 - alpha) * f + alpha * g
            M_new = _conv_max(new_f)
            cur_score = _fast_score(new_f)

            # Accept only if the mixture yields a lower score.
            if cur_score < best_score - 1e-8:
                f = new_f
                M = M_new
                print(f"Iter {iteration:05d} – mixed move: score={cur_score:.6f}, "
                      f"alpha={alpha:.3f}, M={M:.6f}")
                # after a successful step we can be a bit more aggressive
                alpha = min(0.6, alpha / decay)
                last_improve = iteration
            else:
                # no progress – shrink the step size
                alpha = max(min_alpha, alpha * decay)

        # Record the best sequence ever seen.
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"Iter {iteration:05d} – new global best: score={best_score:.6f}")

        # Target reached?
        if best_score < 1.503:
            print(f"Target bound reached at iteration {iteration} with score {best_score:.6f}")
            break

        # Occasionally add a tiny random perturbation to the tail if we are stuck.
        if iteration % 30 == 0 and iteration - last_improve > 100:
            tail_start = n // 2
            bump = np.zeros_like(f)
            bump[tail_start:] = np.random.rand(n - tail_start) * 0.005
            f = np.maximum(0.0, f + bump)
            # re‑normalise the total mass
            f = f * (sum_f / f.sum())
            M = _conv_max(f)
            print(f"Iter {iteration:05d} – random bump applied, new M={M:.6f}")

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Convert to a plain Python list as required by the runner.
    return best_f.tolist()
