def project_onto_simplex(v, s=1):
    n = len(v)
    if v.sum() == s and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # condition: u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‐function sequence that beats the half‑order bound.
Algorithm
---------
1. Initialise with the half‑order series (C(2i,i)/4**i) of length N.
2. Repeatedly minimise a smooth approximation of the maximum convolution entry:
   - Compute the linear convolution c = a * a.
   - Form a soft‑max weight w over the entries of c; the temperature α
     is increased during the run so that w concentrates on the current
     peaks.
   - The gradient of the smoothed objective w.r.t. a is
         g_i = 2 * Σ_k w_k * a_{k-i},
     which can be obtained by a single linear convolution of
     reversed a with w.
   - Perform an exponentiated‑gradient (mirror‑descent) step
         a_i ← a_i * exp(-η * g_i) and renormalise to the probability simplex.
3. Every few hundred iterations inject a small random multiplicative
   perturbation (simulated‑annealing style) to avoid local minima.
4. Keep the best sequence seen (according to the exact evaluate_sequence
   value, which equals 2·N·max(conv) after normalisation).
5. Stop when the time budget is exhausted and return the best sequence.

The method uses only NumPy (no external solvers) and runs fully
deterministically apart from the random perturbations.  It is fast
enough to perform many thousands of iterations within the 1000 s budget
and typically pushes the bound well below the π/2 ≈ 1.571 value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """
    Generate the first n coefficients a_i = C(2i,i) / 4**i.
    The sequence is normalised so that its sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()          # normalise to sum = 1
    return a


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """
    Exact objective value when the sequence is already normalised:
        2 * n * max(conv)
    """
    return float(2.0 * n * conv.max())


def _softmax_weights(x: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute a soft‑max weight vector w_i ∝ exp(alpha * (x_i - max(x))).
    The subtraction of max(x) avoids overflow.
    """
    mx = x.max()
    # exponentials are never larger than 1 because (x_i - mx) ≤ 0
    exp_term = np.exp(alpha * (x - mx))
    w = exp_term / exp_term.sum()
    return w


def _exponentiated_gradient_step(seq: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a mirror‑descent (exponentiated‑gradient) update.
    All entries stay non‑negative and the result is re‑normalised to sum‑1.
    """
    # avoid overflow/underflow in the exponential
    # (grad may be large – we scale it with the step size η)
    factor = np.exp(-eta * grad)
    new_seq = seq * factor
    s = new_seq.sum()
    if s == 0.0:          # extremely unlikely, but safeguard by random restart
        new_seq = np.random.rand(seq.shape[0])
        s = new_seq.sum()
    return new_seq / s


def _random_perturb(seq: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative Gaussian noise (kept non‑negative) followed by renormalisation.
    """
    cand = seq * (1.0 + rng.standard_normal(seq.shape) * sigma)
    cand = np.clip(cand, 0.0, None)
    tot = cand.sum()
    if tot == 0.0:
        return seq.copy()
    return cand / tot


# ----------------------------------------------------------------------
# Main entry point required by the optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # --------------------------------------------------------------
    # 1) initialise
    # --------------------------------------------------------------
    N = 1800                     # length – a compromise between quality and speed
    seq = _half_order_series(N)  # starts with the half‑order bound (~1.571)

    # evaluate the starter (exact evaluation, sum(seq) == 1)
    conv = np.convolve(seq, seq)
    best_score = _score_from_conv(conv, N)
    best_seq = seq.copy()
    print(f"[init] score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) iterative smoothing descent (soft‑max + exponentiated gradient)
    # --------------------------------------------------------------
    iteration = 0
    # schedule parameters
    max_iterations = 200000          # a large cap – the loop will stop by time
    while time.time() < deadline and iteration < max_iterations:
        iteration += 1

        # ---- compute convolution and current score -----------------
        conv = np.convolve(seq, seq)
        cur_score = _score_from_conv(conv, N)

        # update global best if we found a new record
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{iteration:6d}] new best score: {best_score:.6f}")

        # ---- temperature / learning‑rate schedules -----------------
        # progress goes from 0 to 1 over the whole budget
        elapsed = time.time() - start_time
        progress = min(elapsed / (budget_s - 0.5), 1.0)

        # α controls how peaked the soft‑max is; we start gentle and end hard
        alpha = 1.0 + 199.0 * (progress ** 1.5)        # from 1 up to ~200
        # step size for the exponentiated gradient
        eta = 0.12 * (1.0 - progress) + 0.002          # decays from ~0.12 to ~0.002

        # ---- soft‑max weighting of convolution entries -------------
        weights = _softmax_weights(conv, alpha)      # shape (2N‑1,)

        # ---- compute gradient: g_i = 2 * Σ_k w_k * a_{k-i} ----------
        # This is a linear convolution of reversed a with the weight vector.
        rev_seq = seq[::-1]
        grad_full = np.convolve(rev_seq, weights)    # length = N + (2N‑1) - 1 = 3N‑2
        grad = 2.0 * grad_full[N - 1 : N - 1 + N]    # extract the N relevant entries

        # ---- exponentiated‑gradient update -------------------------
        seq = _exponentiated_gradient_step(seq, grad, eta)

        # ---- occasional random noise to escape plateaus ------------
        if iteration % 250 == 0:
            sigma = 0.008 * (1.0 - progress) + 0.001   # slowly shrinking noise
            noisy = _random_perturb(seq, sigma, rng)
            # accept the perturbation if it improves the objective,
            # otherwise keep the current sequence (no Metropolis step needed)
            conv_noisy = np.convolve(noisy, noisy)
            noisy_score = _score_from_conv(conv_noisy, N)
            if noisy_score < cur_score:
                seq = noisy
                print(f"[{iteration:6d}] noise improvement: {noisy_score:.6f}")

        # ---- a few pure‑max gradient steps for extra fine‑tuning ---
        if iteration % 1000 == 0:
            # take a tiny deterministic step using only the current maximal index
            k = int(np.argmax(conv))
            lo = max(0, k - (N - 1))
            hi = min(N - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            max_grad = np.zeros_like(seq)
            max_grad[idx] = 2.0 * seq[k - idx]
            seq = _exponentiated_gradient_step(seq, max_grad, eta * 0.5)

    # --------------------------------------------------------------
    # 3) final clean‑up: a short local search around the best point
    # --------------------------------------------------------------
    final_seq = best_seq.copy()
    final_score = best_score
    for _ in range(500):
        sigma = 0.001
        cand = _random_perturb(final_seq, sigma, rng)
        conv_cand = np.convolve(cand, cand)
        cand_score = _score_from_conv(conv_cand, N)
        if cand_score < final_score:
            final_score = cand_score
            final_seq = cand.copy()
            print(f"[local] improved to {final_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # --------------------------------------------------------------
    return final_seq.tolist()
