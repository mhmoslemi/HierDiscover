import numpy as np
import time

def project_simplex(v):
    # Projects onto the probability simplex (sum=1, nonnegative)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u > css / np.arange(1, n+1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def max_convolution(seq):
    n = len(seq)
    N = 2 * n - 1
    L = 1 << (N-1).bit_length()
    # using real FFT
    fft_seq = np.fft.rfft(seq, n=L)
    prod = fft_seq * fft_seq
    conv_full = np.fft.irfft(prod, n=L)[:N]
    return conv_full.max(), conv_full

def evaluate_obj(seq):
    n = len(seq)
    max_conv, _ = max_convolution(seq)
    return 2.0 * n * max_conv

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation starts from the best known construction (if available)
or from a simple power‑law sequence.  It then performs a projected‑gradient
descent on the *maximum* self‑convolution (the bottleneck of the score).

* The current sequence `a` is kept on the probability simplex
  (non‑negative, sum = 1).  This removes the scaling invariance of the
  original formulation.
* At each iteration the convolution `c = a * a` is obtained with a real FFT,
  the index `k_max` where `c` attains its maximum is identified and the
  sub‑gradient of `max(c)` with respect to `a` is
      g_i = 2·a_{k_max‑i}  (if the term exists, otherwise 0).
* A line‑search with a diminishing step size is performed:
      a' = Proj_Simplex(a – step·g)
  The projection onto the simplex is the standard O(n log n) algorithm.
* If the new point improves the objective it is accepted (and the step
  size is slightly increased); otherwise the step size is halved.
* Every few hundred iterations a small random perturbation is tried to
  escape possible plateaus.
* The loop runs until the allotted time budget (≈ 1000 s) expires.
  The best sequence seen so far is returned as a plain Python list.

The whole routine uses only NumPy, the standard library and the supplied
`evaluate_sequence` for a final sanity check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i >= 0 , sum(w) = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_candidates = u - css / np.arange(1, n + 1)
    rho = np.where(rho_candidates > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution_fft(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute the full self‑convolution a*a using a real FFT and return
    both the maximum value and the whole convolution array.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # length for FFT – next power of two (fastest for np.fft)
    L = 1 << (conv_len - 1).bit_length()

    # real FFT of the zero‑padded sequence
    fft_a = np.fft.rfft(a, n=L)
    prod = fft_a * fft_a                     # element‑wise square
    conv_full = np.fft.irfft(prod, n=L)[:conv_len]

    max_val = conv_full.max()
    return max_val, conv_full


def objective_on_simplex(a: np.ndarray) -> float:
    """
    For a sequence that already lies on the simplex (sum = 1) the
    evaluation function reduces to 2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    max_val, _ = max_convolution_fft(a)
    return 2.0 * n * max_val

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score sequence within the given time budget.
    The algorithm performs a projected gradient descent on the
    supremum norm of the self‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣ Determine length and initialise the sequence
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n == 0:
            n = 3000
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821

        if init_sum <= 0.0:
            raise ValueError
        a = init / init_sum                     # put on the simplex
    except Exception:
        # fallback: a simple power‑law (exponent 0.55 works well)
        n = 3000
        idx = np.arange(1, n + 1, dtype=np.float64)
        p = 0.55
        a = idx ** (-p)
        a = a / a.sum()                         # simplex

    # best solution seen so far
    best_seq = a.copy()
    best_score = objective_on_simplex(best_seq)
    print(f"[init] n = {n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Parameters for the descent
    # --------------------------------------------------------------
    step = 0.12           # initial step size (learning rate)
    min_step = 1e-7

    # --------------------------------------------------------------

    print(f"[final] evaluate_sequence score = {final_score:.6f}")
