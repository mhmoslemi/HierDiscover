Implement subgradient descent with momentum and "Adam" style:

```
m = np.zeros(n)
v = np.zeros(n)
beta1 = 0.9
beta2 = 0.999
epsilon = 1e-8
lr = initial_lr

t = 0
while ...:
    t += 1
    # compute convolution peak
    conv = ... (FFT method)
    max_b = conv.max()
    k_max = conv.argmax()
    # gradient of max_b:
    grad = np.zeros(n)
    i_low = max(0, k_max - (n-1))
    i_high = min(k_max, n-1)
    i_vals = np.arange(i_low, i_high+1)
    j_vals = k_max - i_vals
    grad[i_vals] = a[i_vals] + a[j_vals]  # Because derivative w.r.t each a_i in region is a_i + a_{k-i}
    # gradient of denominator (sum_a^2) part: we want to minimize 2n*max_b/(sum_a**2). Equivalent to minimize log objective? Hard.
    # Simpler: Keep normalization after each step: we keep sum_a = 1 (L1). So denominator is constant => ignore.
    # for simplicity we ensure sum=1 after each step.
    # We can consider using gradient of max_b only.
    # adjust direction: subtract mean to keep sum zero:
    grad = grad - grad.mean()
    # Adam update:
    m = beta1 * m + (1-beta1) * grad
    v = beta2 * v + (1-beta2) * (grad**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
    a = a - step   # moving against gradient for minimization
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n) / n
    else:
        a = a / a_sum
```

   single convolution; we project the step onto the simplex (non‑negative,
   unit‑L1) and adapt the step‑size and the smoothing parameter β.
   Adam‑style momentum is employed to accelerate convergence.

3. **Polishing** – a short random‑move local search that swaps a tiny

The whole procedure runs until the time budget (≈1000 s) is exhausted and
returns the best sequence found, scaled to the [0, 1000] range required by
`evaluate_sequence`.  The implementation uses only NumPy and the provided
`evaluate_sequence` function.

import numpy as np

# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using an FFT of size a power of two."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)
    return conv[: 2 * n - 1]                 # length 2n‑1


def smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Softmax weights centred at the current maximum.
    Returns a vector w that sums to 1.
    """
    max_val = conv.max()
    # shift by max to avoid overflow; all entries ≤ 0 after shift
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:                # safety, should never happen
        w = np.ones_like(conv) / conv.size
    else:
        w = w / w_sum
    return w


def gradient_of_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a∗a)_k .
    The formula reduces to  grad_i = a_i·S_i + T_i
    where S_i = Σ_{k=i}^{i+n-1} w_k   (moving window of length n)
    and   T_i = Σ_{k} w_k·a_{k-i}    (cross‑correlation).
    Both quantities are obtained in O(n) time.
    """
    n = a.shape[0]

    # ----- moving sum S_i (window length n) -----
    pref = np.concatenate(([0.0], np.cumsum(w)))          # length 2n
    S = pref[n:] - pref[:n]                               # length n

    # ----- cross‑correlation term T_i -----
    rev_a = a[::-1]
    corr = np.convolve(w, rev_a, mode='full')             # length 3n‑2
    T = corr[n - 1 : n - 1 + n]                           # slice of length n

    grad = a * S + T
    return grad
