"""
Propose a low‑score step‑function using a smoothed‑max gradient descent.
The method
  * builds a few power‑law seeds (a[i] ~ (i+δ)^(-α)) of various lengths,
    exponents and offsets and picks the best one according to
    ``evaluate_sequence``;
  * then runs a continuous optimisation on the simplex Σa = 1 using a
    differentiable approximation of the max‑convolution (soft‑max with a
    temperature λ).  The gradient of the smoothed objective is obtained
    efficiently with FFT‑based convolutions.
  * a simple projected‑gradient + momentum scheme keeps all a_i ≥ 0,
    Σa = 1 and respects the clipping limits of ``evaluate_sequence``.
The loop stops when the time budget (≈1000 s) is exhausted and the best
sequence found is returned.
"""

import time
import numpy as np

# Helper utilities (all top‑level, no closures, no lambda)
# ----------------------------------------------------------------------


def _power_law_sequence(length: int, exponent: float = 0.5,
                        offset: float = 0.0) -> np.ndarray:
    """Return a non‑negative power‑law sequence a[i] = (i+1+offset)^(-exponent)."""
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)          # respect the hard caps of evaluate_sequence
    total = seq.sum()
    if total == 0.0:
        return seq
    return seq / total                       # normalise to ℓ₁‑norm 1 (scale‑invariant)


def _conv_fft(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution of two real vectors via FFT, result length = len(a)+len(b)-1."""
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    return conv[: a.size + b.size - 1]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute a probability vector proportional to exp(lam * (conv - max(conv))).
    The shift by max(conv) prevents overflow.
    """
    shift = conv.max()
    # all entries are ≤ 0 after the shift, exp is safe
    w = np.exp(lam * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to a uniform distribution (should never happen)
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _gradient_softmax(a: np.ndarray, weights: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smoothed objective ϕ(a)=∑_k w_k (a*a)_k.
    It can be written as 2 * (a * w_rev) evaluated on a specific slice.
    """
    w_rev = weights[::-1]
    # convolution a * w_rev
    conv = _conv_fft(a, w_rev, N)                # length = len(a)+len(w_rev)-1 = 3n‑2
    n = a.size
    # needed slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    segment = conv[n - 1 : 2 * n - 1]            # length n
    grad = 2.0 * segment[::-1]                  # reverse to obtain correct ordering
    # project onto the hyperplane Σa = 0 (remove the component along the all‑ones vector)
    grad -= grad.mean()
    return grad


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the ℓ₁‑simplex Σx = 1.
    Since we never get negative entries after the projection step,
    this is simply a renormalisation.
    """
    s = x.sum()
    if s == 0.0:
        return x
    return x / s

        conv = _conv_fft(a, a, N)                # length = 2n‑1
        max_conv = conv.max()

        # ----- soft‑max weights -----
        weights = _softmax_weights(conv, lam)

        # ----- gradient of the smoothed objective -----
        grad = _gradient_softmax(a, weights, N)   # already zero‑sum projected

        # ----- normalise the step -----
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            # nothing to do – break out early
            break
        step = step_factor / (grad_norm + 1e-12)

        # ----- momentum update -----
        momentum = beta * momentum - step * grad
        a_new = a + momentum

        # ----- enforce feasibility -----
        a_new = np.maximum(a_new, 0.0)            # non‑negativity
        a_new = _project_onto_simplex(a_new)      # Σa = 1

        # ----- occasional jitter to escape plateaus -----
        if iteration % 400 == 0:
            jitter = np.random.randn(n) * 1e-4
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_onto_simplex(a_new)

        # ----- evaluate -----
        try:
            new_score = evaluate_sequence(a_new.tolist())
        except Exception:
            # numerical safeguard – discard this move
            a_new = a.copy()
            continue

        # ----- keep best ever -----
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"iter {iteration:5d}: improved best score = {best_score:.6f}")

        # ----- accept or reject the move -----
        # we accept strictly improving moves; a small tolerance helps stability
        if new_score <= cur_score * (1.0005):
            a = a_new
            cur_score = new_score
        else:
            # with a tiny probability we still move (helps escaping tiny local traps)
            if np.random.rand() < 0.005:
                a = a_new
                cur_score = new_score
            else:
                # rollback momentum to avoid building up on a bad direction
                momentum = np.zeros_like(a)

        # ----- adapt λ (sharpen the soft‑max) -----
        if iteration % 500 == 0:
            lam = min(lam_max, lam * lam_inc)

    # --------------------------------------------------------------
    # 3) Return the best sequence found (clipped to the allowed range)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    print("=== finished ===")
    print(f"final best score: {best_score:.6f}  length: {len(best_seq)}")
    return best_seq
