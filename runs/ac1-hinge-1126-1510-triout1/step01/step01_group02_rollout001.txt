def _fast_score(seq: np.ndarray) -> float:
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

def _grad_max_convolution(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Compute subgradient of 2 n max(conv) / sum^2 at current sequence.

    Returns gradient w.r.t seq (to be used for descent).
    """
    n = seq.size
    s = seq.sum()
    # find the index where conv is maximal (any argmax)
    k = int(np.argmax(conv))
    # contributions a_rev: a_rev[i] = seq[k - i] if 0 <= k-i < n else 0
    a_rev = np.zeros_like(seq)

    hi = min(k, n - 1)
    i_range = np.arange(lo, hi + 1)

    C2 = 4.0 * n * conv[k] / (s * s * s)

"""
Gradient‑descent refinement of a Catalan‑type step function.

The routine starts from a half‑binomial (Catalan) sequence – or from the
previously discovered `height_sequence_1` if it exists – normalises the
ℓ¹‑norm and then repeatedly reduces the maximum entry of the self‑convolution
by moving mass from indices that contribute to the peak towards the tail.
At each iteration we

* compute the convolution via FFT,
* locate the index k where the convolution attains its maximum,
* build the sub‑gradient of the evaluation functional
  (2·n·max(conv)/(∑a)²) with respect to the coefficients,
* take a small step in the descent direction, clip to non‑negative values
  and renormalise the ℓ¹‑norm,
* accept improvements unconditionally and occasional worsening moves
  with a Metropolis probability (simulated‑annealing flavour),
* occasionally add a tiny random shake to escape plateaus.

The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np


def _fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the metric (identical to evaluate_sequence)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:  # protect against division by zero
        return np.inf
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _grad_max_convolution(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the evaluation functional with respect to `seq`.

    The gradient is derived from the active (maximum) convolution index.
    """
    n = seq.size
    s = seq.sum()
    k = int(np.argmax(conv))                     # index of the current peak
    a_rev = np.zeros_like(seq)                  # a_rev[i] = a_{k-i} when defined
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range                    # j_range = k - i
        a_rev[lo:hi + 1] = seq[j_range]

    C1 = 4.0 * n / (s * s)
    C2 = 4.0 * n * conv[k] / (s * s * s)
    grad = C1 * a_rev - C2
    return grad
