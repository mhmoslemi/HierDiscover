def solve_lp_max_sum(a, M):
    n = len(a)
    # constraints: conv(a,g) <= M
    # Build A matrix of shape (2n-1, n)
    # Let's create A as a dense np.ndarray (2n-1, n), may be large but okay.
    # For each k (0..2n-2), row A[k, j] = a_{k - j} if 0 <= k-j < n else 0.
    # That equals a shifted version of a.
    # We can build via Toeplitz: each row is a shifted and padded.
    # Use np.zeros((2n-1, n)).
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # for j in [lo..hi], coefficient A[k, j] = a[k - j]
        # we can fill A[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]
        idx = np.arange(lo, hi+1)
        A[k, idx] = a[k - idx]
    b = np.full(2*n-1, M)
    # obj: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds for g_j >= 0
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.fun
    else:
        # fallback: return a copy
        return a.copy(), -a.sum()
```

def lp_max_sum(a, M):
    n = len(a)
    rows = 2*n-1
    # Build matrix A (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        A[k, idx] = a[k - idx]
    # Solve LP
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        g = a.copy()
    # Normalize g to sum =1
    if g.sum() > 0:
        g = g / g.sum()
    return g
```

"""
Propose a candidate sequence with a value better than the simple half‑order
series (≈ 1.571).  

The algorithm works in three phases:

1. **Initialization & cheap parameter sweep** – start from the classical
   half‑order series a_i = C(2i,i)/4**i and apply a tiny exponential tilt
   a_i ← a_i·exp(θ·i).  A small grid of θ values (both positive and negative)
   is exhaustively checked and the best tilted sequence is kept.

2. **Smooth‑max gradient descent** – the objective
   ``2 n·max(conv(a,a))`` is approximated by the smooth function
   ``L(a)=log Σ_k exp(α·(a*a)_k)``.  Its gradient can be written in closed
   form using a single convolution:
   ∂L/∂a_i = 2·α·(p ⋆ a_rev)[i+n‑1] where p_k are the softmax weights of the
   current convolution and a_rev is a reversed copy of a.  We perform a
   projected gradient step on the probability simplex (non‑negative, sum = 1)
   with a decreasing step size and an annealing schedule for α (from 10 to
   200).  Occasionally a small random multiplicative perturbation is added
   to escape flat regions.

3. **LP refinement (optional)** – every few hundred iterations we solve a
   tiny linear program that maximises Σ_j g_j under the constraints
   ``(a⋆g)_k ≤ M`` where M is the current maximal self‑convolution value.
   The LP solution is blended into the current a (mixing factor t≈0.5) and
   renormalised.  This is the same idea as the “alternating LP” method from
   the reference description and provides monotone improvements when the
   LP solver is available.

The loop runs until the allotted time (default 1000 s) is almost exhausted.
The best sequence seen is returned as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# SciPy is used only for the optional LP refinement.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _half_order_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i)/4**i,
    normalised so that the sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _score(a: np.ndarray) -> float:
    """
    Compute the exact evaluation function (without the safety clamps)
    for a sequence that already satisfies the non‑negativity and sum>0.01.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def _tilted_sequence(base: np.ndarray, theta: float) -> np.ndarray:
    """
    Apply an exponential tilt a_i ← a_i·exp(theta·i) and re‑normalise.
    """
    i = np.arange(base.shape[0], dtype=np.float64)
    tilted = base * np.exp(theta * i)
    s = tilted.sum()
    if s == 0.0:
        return base.copy()
    tilted /= s
    return tilted


def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of log‑sum‑exp(α·conv(a,a)) w.r.t. a.
    Returns the gradient vector (same size as a).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                       # length 2n‑1
    # soft‑max weights
    scaled = alpha * conv
    # stable log‑sum‑exp
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()              # length 2n‑1, sum(p)=1
    # convolution of reversed a with p gives the required sums
    a_rev = a[::-1]
    tmp = np.convolve(a_rev, p)                    # length 3n‑2
    grad = 2.0 * alpha * tmp[n - 1:n - 1 + n]       # slice of length n
    return grad


def _lp_refine(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear programme:
        maximise   Σ_j g_j
        subject to (a⋆g)_k ≤ M  for all k
                   g_j ≥ 0
    Return a feasible g (renormalised to sum=1).  If the LP fails,
    simply return the original a.
    """
    if not _SCIPY_AVAILABLE:
        return a.copy()

    n = a.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where A[k, j] = a_{k‑j} if 0≤k‑j<n else 0
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        A[k, idx] = a[k - idx]

    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g_j  ⇔ minimise –Σ g_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # Use HiGHS (fast, default in recent SciPy)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            return a.copy()
        g = res.x
    except Exception:  # pragma: no cover
        return a.copy()

    s = g.sum()
    if s == 0.0:
        return a.copy()
    g /= s
    return g


def propose_candidate(
    seed: int = 42,
    budget_s: float = 1000.0,
    **kwargs,
) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a small
    value of 2·n·max(conv(a,a))/sum(a)^2.  The routine respects the 1000 s
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a spare half‑second

    # ----------------------------------------------------------------------
    # 1) Basic parameters & initial sequence
    # ----------------------------------------------------------------------
    n = 1000
    base = _half_order_series(n)

    # Try to use a previously computed global sequence if it exists.
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, list) and len(global_seq) == n:
        try:
            candidate = np.array(global_seq, dtype=np.float64)
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s > 0.0:
                candidate /= s
                best_seq = candidate.copy()
                best_val = _score(best_seq)
                print(f"Initial global sequence used, score={best_val:.6f}")
        except Exception:
            best_seq = base.copy()
            best_val = _score(best_seq)
    else:
        best_seq = base.copy()
        best_val = _score(best_seq)

    # ----------------------------------------------------------------------
    # 2) Tiny tilt parameter sweep (exponential factor)
    # ----------------------------------------------------------------------
    thetas = np.linspace(-0.006, 0.006, 25)  # symmetric grid, step≈0.0005
    for theta in thetas:
        tilted = _tilted_sequence(base, theta)
        val = _score(tilted)
        if val < best_val:
            best_val = val
            best_seq = tilted.copy()
            print(f"Tilt sweep: theta={theta:.5f} improved score={best_val:.6f}")

    # Current working sequence
    a = best_seq.copy()
    cur_val = best_val

    # ----------------------------------------------------------------------
    # 3) Main optimisation loop (smooth‑max gradient + random noise)
    # ----------------------------------------------------------------------
    iteration = 0
    last_improve_iter = 0
    total_iters_est = 1_000_000  # just a large number; we stop by time

    while time.time() < deadline:
        iteration += 1
        progress = (time.time() - start_time) / max(deadline - start_time, 1e-9)

        # --- annealing schedule for the smoothing parameter α
        alpha = 10.0 * (1.0 - progress) + 200.0 * progress

        # --- gradient of the smooth‑max
        grad = _smooth_max_gradient(a, alpha)

        # --- step size schedule (larger steps early, tiny later)
        step = 0.07 * (1.0 - progress) + 0.0008 * progress

        # projected gradient step on the simplex
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = a.copy()
            s = a_new.sum()
        a_new /= s

        new_val = _score(a_new)

        # accept if better, otherwise occasional Metropolis move
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature for Metropolis (very gentle)
            temp = 0.01 * (1.0 - progress) + 0.001
            prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val - 1e-8:
                best_val = new_val
                best_seq = a_new.copy()
                last_improve_iter = iteration
                print(f"[{iteration:6d}] Gradient step improved score={best_val:.6f}")

        # --- occasional random multiplicative noise (helps escape flat areas)
        if iteration % 150 == 0:
            sigma = 0.02 * (1.0 - progress) + 0.001
            noise = np.random.randn(n) * sigma
            cand = a * (1.0 + noise)
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0.0:
                cand /= s
                cand_val = _score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    cur_val = cand_val
                    last_improve_iter = iteration
                    print(f"[{iteration:6d}] Random perturbation improved score={best_val:.6f}")

        # --- LP refinement every few hundred iterations
        if iteration % 500 == 0 and _SCIPY_AVAILABLE:
            M_current = np.convolve(a, a).max()
            g = _lp_refine(a, M_current)
            # blend the two sequences (t=0.5 gives good progress)
            blend_t = 0.5
            a_lp = (1.0 - blend_t) * a + blend_t * g
            a_lp /= a_lp.sum()
            lp_val = _score(a_lp)
            if lp_val < best_val:
                best_val = lp_val
                best_seq = a_lp.copy()
                a = a_lp
                cur_val = lp_val
                last_improve_iter = iteration
                print(f"[{iteration:6d}] LP refinement improved score={best_val:.6f}")

        # safety break‑out if no improvement for a long while
        if iteration - last_improve_iter > 20000:
            # reduce step size dramatically to fine‑tune
            step *= 0.1
            # also reset the counter so we do not exit too early
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ----------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score={best_val:.6f}")
    # Clip just in case of tiny negative round‑off before returning
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum()
    return best_seq.tolist()
