```
def compute_softmax_max(seq, alpha):
    n = len(seq)
    L = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n-1]
    # conv as b array
    max_val = conv.max() # for actual score, but we need soft approximation
    # compute softmax
    # shift conv to avoid overflow: subtract max
    shift = conv.max()
    exps = np.exp(alpha * (conv - shift)) # α*(b - max)
    sum_exps = exps.sum()
    soft_max = shift + np.log(sum_exps) / alpha
    w = exps / sum_exps   # weights sum to 1
    return soft_max, w, conv
```

import numpy as np
from scipy.optimize import linprog

def _max_self_conv(seq):
    """Return max self‑convolution value (M) for a 1‑D array."""
    conv = np.convolve(seq, seq)
    return conv.max(), conv

def _build_toeplitz_conv_matrix(f):
    """Return the (2n‑1)×n matrix A such that A @ g = f * g (linear convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f[0..n‑1] on rows j..j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step function using an alternating
    linear‑programming / convex‑combination scheme.
    The algorithm starts from the Catalan (half‑binomial) sequence,
    normalises it so that ∑a_i = √(2·n) (hence the score equals the
    peak of the self‑convolution), then repeatedly:
        1. solves a linear programme that, for the current f,
           finds a non‑negative g with maximal ℓ₁‑norm under the
           constraint f * g ≤ M (where M = max(f * f));
        2. rescales g to the same ℓ₁‑norm as f;
        3. replaces f ← (1‑t)·f + t·g for a decreasing step size t.
    The process monotonically drives down M, i.e. the evaluation
    metric, and stops when the time budget is exhausted or no
    further improvement is observed.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # small safety margin

    # 1. Initialise with Catalan (half‑binomial) coefficients.
    #    a_i = C(2i,i) / 4**i   (recurrence a_{i+1}=a_i*(2i+1)/(2(i+1)) )
    #    Then scale so that sum = sqrt(2*n).
    # ------------------------------------------------------------------
    n = 1000                      # length of the sequence (tunable)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    iteration = 0
    t = 0.5                         # initial mixing weight, will decay
    min_t = 0.02

        A = _build_toeplitz_conv_matrix(f)

2.  Solve a linear programme that for this f finds a non‑negative g
    with maximal total mass under the linear constraints   f∗g ≤ M.
    (The convolution f∗g is linear in g, therefore the constraints are
    linear.)  The LP also caps each entry at 1000 to respect the
    evaluation limits.
3.  Rescale g so that Σg = Σf (i.e. the target sum √(2·n)).
4.  Update the sequence by a convex combination
        f ← (1‑t)·f + t·g
    with a mixing weight t that slowly decays.
5.  Keep the best sequence seen (according to evaluate_sequence).

The process monotonically drives the peak M down – consequently the
evaluation score (which equals M for the chosen normalisation) improves.
A small random jitter is added every 50 iterations to escape tiny
plateaus.  The loop stops when the time budget (≈1000 s) is exhausted or
when a score below the target 1.503 is found.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _max_self_conv(seq):
    """
    Return the maximal value of the self‑convolution of `seq`
    together with the full convolution array (both as float).
    """
    conv = np.convolve(seq, seq)
    return conv.max(), conv

def _build_toeplitz_conv_matrix(f):
    """
    Build the (2·n‑1) × n matrix A such that A @ g = f ∗ g
    (linear convolution of `f` with a variable vector `g`).

    For each column j the vector f is placed on rows j … j+n‑1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) beats the target
    evaluation score.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # tiny safety margin

    # --------------------------------------------------------------
    # 1. Initialise with Catalan (half‑binomial) coefficients.
    # --------------------------------------------------------------
    n = 1000                                          # length of the sequence
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    # Normalise so that Σa = √(2·n).  With this normalisation the score
    # returned by `evaluate_sequence` is exactly the peak of the
    # self‑convolution.
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    f = a.copy()                                      # current candidate
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score (n={n}): {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Main optimisation loop – alternating LP + mixing.
    # --------------------------------------------------------------
    iteration = 0
    t = 0.5                                           # initial mixing weight
    min_t = 0.02                                      # lower bound for t

    while time.time() < deadline:
        iteration += 1

        # ----- a) current peak M of f∗f --------------------------------
        M, _ = _max_self_conv(f)

        # ----- b) build the linear constraints f∗g ≤ M -----------------
        A = _build_toeplitz_conv_matrix(f)           # shape (2n‑1, n)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # ----- c) linear program: maximise Σg_j subject to f∗g ≤ M -----
        c = -np.ones(n, dtype=np.float64)            # minimise –Σg
        bounds = [(0.0, 1000.0)] * n                 # keep values in allowed range

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g == 0.0:
            print(f"LP returned zero total mass at iteration {iteration}")
            break

        # ----- d) rescale g to the target ℓ₁‑norm --------------------
        g = g_raw * (target_sum / sum_g)

        # ----- e) convex combination (mix) ----------------------------
        f = (1.0 - t) * f + t * g

        # decay the mixing coefficient slowly
        t = max(min_t, t * 0.99)

        # ----- f) evaluate the new candidate ---------------------------
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"Iter {iteration:04d} – new best score: {best_score:.6f}")

        # ----- occasional jitter to avoid tiny plateaus ----------------
        if iteration % 50 == 0:
            jitter = np.random.randn(n) * 1e-3
            f_j = np.maximum(0.0, f + jitter)
            f_j = f_j * (target_sum / f_j.sum())
            jitter_score = evaluate_sequence(f_j.tolist())
            if jitter_score < best_score:
                best_score = jitter_score
                best_seq = f_j.copy()
                f = f_j
                print(f"Iter {iteration:04d} – jitter improved score: {best_score:.6f}")

        # ----- early exit if target already reached --------------------
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}: score {best_score:.6f}")
            break

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
