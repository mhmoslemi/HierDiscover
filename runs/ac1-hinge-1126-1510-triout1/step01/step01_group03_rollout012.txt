```
def project_simplex(v):
    # ensure non-negative
    v = np.maximum(v, 0)
    # project onto sum=1 simplex
    if v.sum() == 0:
        return np.full_like(v, 1.0/v.shape[0])
    # standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a sequence of non‑negative heights minimizing
    2 * n * max(conv(seq,seq)) / (sum(seq)**2)

The algorithm combines three complementary ideas:
1.  Initialise with the classical half‑order series (central binomial
    coefficients) which already gives the π/2 ≈ 1.571 bound.
2.  Main loop (≈ 1000 s budget):
    * Simulated‑annealing style multiplicative noise to explore the space.
    * A sub‑gradient step that reduces the current maximal convolution entry
      (or a small set of active entries) while projecting back onto the simplex.
    * Occasionally solve a small linear programme:
        maximise 1ᵀg  subject to  (f * g) ≤ M   and  g ≥ 0,
      where f is the current sequence and M = max(f * f).
      The solution g tends to move mass into positions that are “inactive”
      for the current max‑convolution and, after mixing with f, reduces M.
3.  After the time limit the best sequence found is returned.
All operations stay on the simplex (non‑negative entries, sum = 1) so
evaluate_sequence() never rejects the candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def half_order_series(n: int) -> np.ndarray:
    """First n coefficients of C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, Σ x_i = 1}.
    Implementation of the algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # ensure non‑negativity (helps numerical stability)
    v = np.maximum(v, 0.0)

    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise just in case of rounding errors
    w /= w.sum()
    return w


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix A such that A @ g == np.convolve(f,g)
    (linear convolution, “full” mode).  Row k corresponds to (f*g)_k.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A


def lp_optimize_g(f: np.ndarray, M: float):
    """
    Solve   maximise Σ g_i
    s.t.    f * g ≤ M   (component‑wise)
            g_i ≥ 0
    Returns the optimal g (may be None if the LP fails).
    """
    try:
        from scipy.optimize import linprog
    except Exception:  # SciPy not available – skip the LP step
        return None

    n = f.shape[0]
    A = build_conv_matrix(f)                # shape (2n‑1) × n
    b = np.full(A.shape[0], M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)       # maximise Σg  →  minimise –Σg
    bounds = [(0, None) for _ in range(n)]

    # “highs” is the recommended modern interior‑point method
    res = linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}
    )
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def subgradient_step(seq: np.ndarray, step: float, epsilon: float = 1e-12) -> np.ndarray:
    """
    One sub‑gradient step that tries to reduce the maximal convolution entry.
    The gradient is accumulated over all (approximately) active indices.
    After the step the vector is projected back onto the simplex.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # consider all indices whose value is within 0.5 % of the maximum
    active = np.where(conv >= max_val * (1.0 - 0.5e-2))[0]

    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev = k - idx
        # derivative of b_k = Σ a_i a_{k‑i}
        grad[idx] += 2.0 * seq[rev]

    # move opposite to the gradient, keep a small step size
    cand = seq - step * grad
    cand = project_to_simplex(cand)
    return cand


def uniform_target_conv(n: int) -> np.ndarray:
    """Uniform target vector of length 2n‑1, each entry = 1/(2n‑1)."""
    L = 2 * n - 1
    return np.full(L, 1.0 / L, dtype=np.float64)


def l2_error_gradient(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Gradient of the squared‑error ‖conv(seq,seq) – target‖²
    where the target is the uniform vector.
    The gradient is   2 · corr(err, seq)   (see derivation in analysis).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    target = uniform_target_conv(n)
    err = conv - target                     # length 2n‑1
    # np.correlate(err, seq, mode='valid') gives Σ_j err_{j+i}·seq_j  for i=0..n‑1
    grad = 2.0 * np.correlate(err, seq, mode='valid')
    cand = seq - step * grad
    cand = project_to_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (floating point numbers) that
    minimises the evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence seen.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ---- 1) initialise -------------------------------------------------
    n = 500                     # a size that balances quality & LP cost
    seq = half_order_series(n)  # sum = 1, already gives ≈1.571 bound

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}  score={best_score:.6f}")

    # annealing schedule parameters
    temp_initial = 0.6
    temp_final = 1e-5
    total_runtime = budget_s - 2.0
    iteration = 0

    # ---- 2) main optimisation loop ------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        progress = elapsed / total_runtime
        progress = min(max(progress, 0.0), 1.0)

        # temperature (for Metropolis acceptance)
        temp = temp_initial * (temp_final / temp_initial) ** progress

        # --------------------------------------------------------------
        # 2.1) random multiplicative perturbation (simulated annealing)
        # --------------------------------------------------------------
        sigma = 0.015 * (1.0 - progress) + 0.001  # decreasing noise
        noise = np.random.randn(n) * sigma
        cand = seq * (1.0 + noise)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
            cand_score = evaluate_sequence(cand.tolist())
            accept = False
            if cand_score < best_score:
                accept = True
            else:
                # Metropolis acceptance
                prob = math.exp(-(cand_score - best_score) / max(temp, 1e-12))
                if np.random.rand() < prob:
                    accept = True
            if accept:
                seq = cand
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"[{iteration:06d}] Random improvement: {best_score:.6f}")

        # --------------------------------------------------------------
        # 2.2) sub‑gradient step on the current maximal convolution entry
        # --------------------------------------------------------------
        if iteration % 7 == 0:
            step = 0.02 * (1.0 - progress) + 0.001
            cand = subgradient_step(seq, step)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                seq = cand
                print(f"[{iteration:06d}] Sub‑gradient step: {best_score:.6f}")

        # --------------------------------------------------------------
        # 2.3) L2‑error based gradient towards a uniform convolution
        # --------------------------------------------------------------
        if iteration % 13 == 0:
            step = 0.05 * (1.0 - progress) + 0.001
            cand = l2_error_gradient(seq, step)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                seq = cand
                print(f"[{iteration:06d}] L2‑error gradient: {best_score:.6f}")

        # --------------------------------------------------------------
        # 2.4) occasional linear‑programming based mass redistribution
        # --------------------------------------------------------------
        if iteration % 31 == 0:
            conv = np.convolve(seq, seq)
            M = conv.max()
            g = lp_optimize_g(seq, M)
            if g is not None and g.sum() > 0.0:
                g = g / g.sum()            # normalise to sum = 1
                # simple mixing – a convex combination with weight t
                t = 0.30 * (1.0 - progress) + 0.05   # start aggressive, fade
                cand = (1.0 - t) * seq + t * g
                cand = project_to_simplex(cand)
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    seq = cand
                    print(f"[{iteration:06d}] LP‑mix improvement: {best_score:.6f}")

        # --------------------------------------------------------------
        # 2.5) occasional pure replacement with the LP solution (if better)
        # --------------------------------------------------------------
        if iteration % 61 == 0 and g is not None and g.sum() > 0.0:
            g = g / g.sum()
            cand_score = evaluate_sequence(g.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = g.copy()
                seq = g
                print(f"[{iteration:06d}] LP pure replacement: {best_score:.6f}")

        # break early if we manage to cross the target bound
        if best_score < 1.503:
            print(f"[{iteration:06d}] Target reached (score={best_score:.6f}).")
            break

    # ---- 3) final result ------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
