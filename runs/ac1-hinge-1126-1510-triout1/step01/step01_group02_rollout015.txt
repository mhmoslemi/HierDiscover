```
def _project_simplex(v):
    # v: ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def _conv_self(p):

    f = np.fft.rfft(p, n=L)
```

```
def _gradient(p, w):
    # w: length 2n-1
    n = len(p)
    w_rev = w[::-1]
```

"""
Propose a candidate step‑function sequence that minimizes the provided
evaluation metric.

The algorithm has two phases:

1. **Soft‑max gradient descent (mirrored on the probability simplex).**
   The distribution of the coefficients is normalised to sum = 1 (the
   evaluation metric is homogeneous, therefore scaling does not matter).
   At each iteration we
   * compute the self‑convolution `c = a * a` via FFT,
   * turn `c` into a soft‑max weight vector `w` (parameter β gradually
     increased from a low to a high value, so early iterations explore,
     later ones focus on the current maximum),
   * compute the gradient of the soft‑max surrogate
       `g_i = 2 Σ_k w_k a_{k‑i}`  (implemented with a second FFT), and
   * perform a multiplicative‑weights update
       `a_i ← a_i·exp(-η·g_i)`  followed by normalisation to keep the sum
       equal to 1.  This is an entropic mirror‑descent step that stays on
       the simplex without an expensive projection.

2. **Local hill‑climbing (optional, for the last ≈20 % of the time).**
   Once the temperature β is large the distribution is already close to
   a flat convolution; we then make small stochastic transfers of mass
   from early indices (which strongly influence the peak) to far‑away
   indices and keep any improvement.

Both phases keep track of the best sequence seen so far and stop early
once a score below the target `1.503` is reached.  The routine runs for
the allocated wall‑clock time (default ≈ 1000 s) and finally returns the
best list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (full length 2·len(a)‑1)."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    f = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max on the convolution values.
    Returns a probability vector of the same length as conv."""
    max_c = conv.max()
    # shift to avoid overflow/underflow
    shifted = conv - max_c
    # exponentiate safely
    exp_beta = np.exp(beta * shifted)
    w = exp_beta / exp_beta.sum()
    return w


def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        g_i = 2 * Σ_k w_k * a_{k-i}
    (out‑of‑range indices contribute 0).  Computed by a convolution with
    the reversed weight vector.
    """
    n = a.size
    w_rev = w[::-1]
    L = 1 << ((n + w_rev.size - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)[: n + w_rev.size - 1]
    offset = w_rev.size - 1                # = len(w) - 1
    # extract the n needed components (see analysis for the index shift)
    grad_raw = conv_rev[offset - np.arange(n)]
    return 2.0 * grad_raw


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the
    allocated wall‑clock time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – we want to finish well before the timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Initial distribution – Catalan (half‑binomial) coefficients,
    #     normalised to sum = 1 (the objective is scale‑invariant).
    # ------------------------------------------------------------------
    n = 2500
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()                     # now a is a probability vector

    cur = a.copy()
    conv = _conv_self(cur)
    cur_score = 2.0 * n * conv.max()
    best = cur.copy()
    best_score = cur_score
    print(f"Initial score (Catalan normalised): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Hyper‑parameters and schedule
    # ------------------------------------------------------------------
    beta_start = 5.0          # soft‑max temperature at the beginning
    beta_end   = 200.0        # final temperature (focuses on true max)
    eta_start  = 0.5          # learning rate for the multiplicative update
    low_end    = max(1, n // 10)   # early part – sources for hill‑climb
    high_start = n // 2            # far part – targets for hill‑climb

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 5.0)          # 0 … 1

        # --------------------------------------------------------------
        # Phase selection
        # --------------------------------------------------------------
        if prog < 0.80:            # -------- gradient descent phase ----------
            # ----- schedules -------------------------------------------------
            beta = beta_start + (beta_end - beta_start) * prog
            eta  = eta_start * (1.0 - prog)    # linear decay to ~0

            # ----- compute convolution and soft‑max weights ------------------
            conv = _conv_self(cur)
            w = _softmax_weights(conv, beta)

            # ----- gradient (2 · Σ_k w_k a_{k-i}) ---------------------------
            grad = _gradient(cur, w)

            # ----- multiplicative (mirror‑descent) update -------------------
            # protect against under‑flow / overflow
            exp_factor = np.exp(-eta * grad)
            cur = cur * exp_factor
            cur_sum = cur.sum()
            if cur_sum == 0.0:            # extremely unlikely, re‑initialise
                cur = np.full_like(cur, 1.0 / n)
            else:
                cur /= cur_sum

            # ----- occasional jitter to escape plateaus --------------------
            if it % 3000 == 0:
                jitter = np.random.randn(n) * 1e-4
                cur = np.maximum(cur + jitter, 0.0)
                cur /= cur.sum()

        else:                     # -------- hill‑climbing phase -------------
            # Pick a source from the first 10 % (must contain some mass)
            src = np.random.randint(0, low_end)
            if cur[src] <= 1e-12:
                continue
            # Pick a far target
            tgt = np.random.randint(high_start, n)
            # Transfer a random fraction (up to 30 %) of the source mass
            frac = np.random.rand() * 0.3
            delta = cur[src] * frac
            trial = cur.copy()
            trial[src] -= delta
            trial[tgt] += delta

            # Evaluate the trial sequence (fast self‑convolution)
            conv_trial = _conv_self(trial)
            trial_score = 2.0 * n * conv_trial.max()

            # Accept only improving moves (no temperature here)
            if trial_score < cur_score:
                cur = trial
                cur_score = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
                    print(f"Iter {it:07d} – new best (hill‑climb) {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3.  Track global best (once per outer loop)
        # ------------------------------------------------------------------
        conv = _conv_self(cur)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            print(f"Iter {it:07d} – new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4.  Early exit if target already reached
        # ------------------------------------------------------------------
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, score {best_score:.6f}")
            break

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    # Return a plain Python list as required by the grader.
    return best.tolist()
