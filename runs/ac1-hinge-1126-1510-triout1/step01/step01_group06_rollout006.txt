    rng = np.random.default_rng(seed)
    start = time.time()

    conv = np.convolve(a, a)
    cur_max = conv.max()
    cur_score = 2 * n * cur_max

    step_decay = 0.99995
    min_step = 1e-6
    iter_count = 0
    last_print = start

    while time.time() < deadline:
        iter_count += 1
        # Choose i based on probability proportional to a (mass moves from large entries)
        i = int(rng.choice(n, p=a))
        # Choose j uniformly not equal i
        j = int(rng.integers(0, n-1))
        if j == i:

        # epsilon: random fraction up to step_factor of a[i]
        eps = a[i] * rng.random() * step_factor

        conv_new = conv.copy()
        # contributions from delta_i = -eps, delta_j = +eps

        start_i = i
        end_i = i + n

        # update slice for j

        conv_new[idx_2j] += eps * eps

        if new_max < cur_max:

        else:

            delta = new_max - cur_max

                best_score = cur_score
                best_seq = a.copy()
                best_max = cur_max

"""
Simulated‑annealing search for a low‑autocorrelation step function.

The algorithm works with a probability vector a (∑a = 1, a_i ≥ 0) and
keeps the full self‑convolution `conv = a*a` up‑to‑date.  A move transfers a
small amount of mass from an index i (chosen proportionally to its mass)
to a different index j.  The convolution can be updated in O(n) using the
identity

    (a+Δ)*(a+Δ) = a*a + 2·(a*Δ) + Δ*Δ ,

where Δ has only two non‑zero entries (−ε at i and +ε at j).  This makes
hundreds of thousands of candidate evaluations cheap enough to run for the
full time budget.

Occasionally (with a small probability) a gradient‑descent step on the
active convolution peak is performed – this is the same direction used in
the reference implementation but with an adaptive step size.  The
Metropolis acceptance rule with a slowly decreasing temperature allows
up‑hill moves early on and focuses on descent later.

The best sequence found is scaled to the full [0,1000] range before being
returned.  All numbers stay non‑negative, and the code respects the 1000 s
time limit.
"""

    conv = np.convolve(a, a)

    min_step = 1e-8
    step_decay = 0.99997

    gd_step = 0.02
    gd_decay = 0.99999

    # ------------------------------------------------------------------

    while time.time() < deadline:

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        if rng.random() < gd_prob:
            # index of the current convolution peak
            k_max = int(np.argmax(conv))

            # build gradient of that peak (entries outside the active range stay 0)
            grad = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
