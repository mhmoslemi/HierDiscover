Let's design the projection algorithm:

```
def project_onto_simplex(v):
    # Returns projection of v onto {x >= 0, sum = 1}
    # Using algorithm by Wang & Carreira-Perpinan.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v):
    # Based on algorithm by Duchi et al (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all <=0 -> return uniform?
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    t_initial = 0.5
    for outer in range(max_outer):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()

        # T shape (2n-1, n)

        if b_var.value is None:

            break

    conv = np.convolve(a, a)

    corr = np.correlate(a, mask, mode='full')

    refinement (maximise the total mass while keeping the mixed
    convolution ≤ the current worst value). Starts from the
    half‑order series and iteratively improves the bound.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

def project_onto_simplex(v):
    # Duchi et al. algorithm
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero or negative, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1  # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numeric drift
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    w /= w_sum
    return w
```

"""
Propose a candidate sequence of non‑negative heights that improves the
convolution bound.

The algorithm combines two complementary ideas:

1. **Linear‑programming refinement** –  Starting from a good seed (the
   half‑order series) we maximise the total mass of a new non‑negative
   vector `b` under the linear constraints
       (a * b)[k] ≤ M   for all k,
   where `a` is the current sequence and `M = max(a*a)`.  This is a small
   LP (≈ 2 n variables and 2 n constraints).  The optimiser returns a
   vector with as much mass as possible while keeping the mixed convolution
   below the current worst value.  Normalising `b` to unit sum and mixing
   it with the old sequence (convex combination) reduces the
   `max(conv(a,a))` value.

2. **Projected sub‑gradient polishing** –  After each LP step we perform
   a few iterations of projected sub‑gradient descent on the objective
   `max(conv(a,a))`.  The sub‑gradient of the maximal convolution entry is
   obtained efficiently via a correlation, then a gradient step is taken
   and the iterate is projected back onto the probability simplex
   (non‑negative entries summing to 1).

The process is repeated with a decreasing mixing weight.  The whole
search runs for the allotted time (≈ 1000 s) and returns the best sequence
found.  The implementation uses only `numpy` and `scipy.optimize.linprog`
and never calls `evaluate_sequence` inside the inner loop (the bound is
computed analytically as `2*n*max(conv)` because the sum is fixed to 1).
"""

import time
import math

import numpy as np
from scipy.optimize import linprog


def _half_order_series(n: int) -> np.ndarray:
    """First n coefficients of a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i-1} * ((2i−1)(2i)) / (4 i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _powerlaw_series(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum 1 (p>0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a[0] = 1.0           # keep the first entry reasonably large
    a /= a.sum()
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        w = np.full_like(v, 1.0 / n)
        return w
    rho = ind[cond].max() - 1               # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical drift may cause the sum to deviate slightly
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def _subgradient_max_conv(a: np.ndarray):
    """
    Returns (g, m) where m = max_k (a*a)[k] and g is a sub‑gradient of
    the function max_k (a*a)[k] at the point a.
    """
    conv = np.convolve(a, a)
    m = conv.max()
    # treat all entries that are numerically equal to the maximum as active
    tol = 1e-12
    mask = (conv >= m - tol).astype(np.float64)
    # correlation a * mask gives Σ_i a_i * mask_{k−i}
    corr = np.correlate(a, mask, mode="full")
    start = len(mask) - 1
    g = 2.0 * corr[start:start + len(a)]
    return g, m


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `2 * n * max(conv(seq, seq)) / (sum(seq)**2)`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) problem size and initial seeds
    # ------------------------------------------------------------------
    n = 1500                     # length of the sequence (trade‑off speed / quality)

    candidates = [_half_order_series(n)]
    # a few power‑law seeds around the half‑order exponent (0.5)
    for p in np.linspace(0.35, 0.65, 7):
        candidates.append(_powerlaw_series(n, p))
    # a few random Dirichlet vectors
    for _ in range(3):
        v = np.random.rand(n)
        v /= v.sum()
        candidates.append(v)

    best_seq = None
    best_score = np.inf
    for seq in candidates:
        cur = 2.0 * n * np.convolve(seq, seq).max()
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
    print(f"Initial best bound  {best_score:.6f}")

    a = best_seq.copy()                 # current iterate (always sum 1)

    # ------------------------------------------------------------------
    # 2) main optimisation loop: LP refinement + sub‑gradient polishing
    # ------------------------------------------------------------------
    outer = 0
    max_outer = 30                      # enough iterations for the time budget
    t0 = 0.5                            # initial mixing weight

    while outer < max_outer and time.time() < deadline:
        # current worst convolution value
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # ---- build the Toeplitz convolution matrix T ( (2n‑1) × n ) ----------
        # T[k, j] = a_{k‑j}   if 0 ≤ k‑j < n, else 0
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            T[j:j + n, j] = a

        # ---- linear programme ------------------------------------------------
        # maximise Σ b_j   ⇔   minimise -Σ b_j
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=T,
            b_ub=ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"LP failed at outer iteration {outer} – terminating.")
            break

        b_raw = np.maximum(res.x, 0.0)
        b_sum = b_raw.sum()
        if b_sum == 0.0:
            # nothing useful obtained – stop the outer loop
            break
        b = b_raw / b_sum                # normalise to unit sum

        # ---- convex combination ------------------------------------------------
        t = t0 / (outer + 1)             # decreasing mixing weight
        a_new = (1.0 - t) * a + t * b
        a_new = _project_onto_simplex(a_new)

        # evaluate the new candidate (exact bound, sum is 1)
        new_score = 2.0 * n * np.convolve(a_new, a_new).max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[outer {outer:2d}] LP‑mix improvement: {best_score:.6f}")

        a = a_new

        # ---- a few projected sub‑gradient steps to polish --------------------
        for inner in range(5):
            if time.time() >= deadline:
                break
            grad, _ = _subgradient_max_conv(a)
            # step size decreasing with iteration counter
            step = 0.07 / math.sqrt(inner + 1 + outer * 5)
            a = a - step * grad
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)
            cur_score = 2.0 * n * np.convolve(a, a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[outer {outer:2d} inner {inner}] sub‑grad improvement: {best_score:.6f}")

        outer += 1

    # ------------------------------------------------------------------
    # 3) final validation via the provided evaluator (optional)
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Finished – best bound returned {final_val:.6f}")
    return best_seq.tolist()
