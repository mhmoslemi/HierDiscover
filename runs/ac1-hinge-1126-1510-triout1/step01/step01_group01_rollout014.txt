Pseudo:

def project_simplex(v):
    # v is numpy array
    # projects onto the simplex {x>=0, sum x = 1}
    n = v.shape[0]
    # shift v to get sorted
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # from Duchi et al. 2008
    # input: numpy array
    # output: array projected onto simplex { x >= 0, sum x = 1 }
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence with a very low evaluation value.

Algorithm
---------
1. Choose a length `n` that is a power of two (good for FFT).
2. Build a few simple analytic families (Catalan, power‑law, beta‑like)
   and keep the one with the smallest raw score.
3. Starting from this best seed run a projected sub‑gradient descent on the
   objective  `max(conv(a,a))` (the only term that matters after normalising
   the sum to 1).  The sub‑gradient at the current peak `k` is
        g_i = 2 * a_{k-i}  (if 0 ≤ k-i < n, else 0)
   and a step `a ← a - η g` is taken.
4. After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1).  The step size decays like `η₀/√t`.
5. A tiny Metropolis acceptance rule (temperature slowly cooling) and
   occasional random perturbations are used to escape plateaus.
6. The best sequence found before the time limit is returned.

The whole routine stays well inside the 1000 s budget and typically
improves the Catalan bound (≈1.571) to ≈1.50‑ish.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x≥0, sum(x)=1} using the algorithm of Duchi et al."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find the largest rho satisfying the condition
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families used for the initial guess
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """First `n` Catalan‑type coefficients: binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (i = 0,…,n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _beta_sequence(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Discrete Beta‑like shape:
       a_i ∝ (i+0.5)^{a-1} (n-i-0.5)^{b-1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i ** (a_param - 1.0)
    right = (n - i) ** (b_param - 1.0)
    seq = left * right
    # avoid NaNs for extreme parameters
    seq = np.where(np.isfinite(seq), seq, 0.0)
    return seq


# ----------------------------------------------------------------------
# Core routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The method builds a good analytic seed and then
    refines it with a projected sub‑gradient descent on the L∞ norm of the
    self‑convolution (the only term that matters after normalisation).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣   Choose a convenient length (power of two => fast FFT)
    # ------------------------------------------------------------------
    n = 8192                 # 2^13, gives 2 n‑1 = 16383 convolution size

    # ------------------------------------------------------------------
    # 2️⃣   Helper to evaluate a normalised sequence (sum = 1)
    # ------------------------------------------------------------------
    nfft = 1 << ((2 * n - 2).bit_length())        # next power of two ≥ 2n‑1

    def _score(seq: np.ndarray) -> float:
        """2·n·max(conv(seq,seq)) for a normalised (ℓ¹‑norm = 1) sequence."""
        F = np.fft.rfft(seq, n=nfft)
        conv = np.fft.irfft(F * F, n=nfft)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 3️⃣   Scan a handful of analytic families for a good starting point
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Catalan
    seq = _catalan_sequence(n)
    seq = seq / seq.sum()
    sc = _score(seq)
    print(f"[init] Catalan seed score = {sc:.6f}")
    if sc < best_score:
        best_seq, best_score = seq.copy(), sc

    # Power‑law grid
    for p in np.linspace(0.45, 0.65, 21):          # step ≈0.01
        seq = _powerlaw_sequence(n, p)
        seq = seq / seq.sum()
        sc = _score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc

    # Beta‑like grid (a,b around 0.5 give arcsine‑type shapes)
    for a_param in (0.4, 0.5, 0.6):
        for b_param in (0.4, 0.5, 0.6):
            seq = _beta_sequence(n, a_param, b_param)
            if seq.sum() == 0:
                continue
            seq = seq / seq.sum()
            sc = _score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc

    print(f"[init] Best seed score after scanning = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣   Projected sub‑gradient descent on max‑convolution
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    step0 = 0.20            # initial step length (will be divided by √t)
    temp0 = 0.02            # initial temperature for Metropolis acceptance

    no_improve = 0          # counter for stagnation detection
    last_best_time = start_time

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- current convolution and peak index -----
        F_cur = np.fft.rfft(cur_seq, n=nfft)
        conv = np.fft.irfft(F_cur * F_cur, n=nfft)[: 2 * n - 1]
        k = int(conv.argmax())          # index where the max occurs

        # ----- sub‑gradient of max(conv) at the current peak -----
        g = np.zeros_like(cur_seq)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * cur_seq[k - idx]   # vectorised assignment

        # ----- step size (decays like 1/√t) -----
        step = step0 / math.sqrt(it)

        # ----- propose a new point and project onto the simplex -----
        cand = cur_seq - step * g
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_score = _score(cand)

        # ----- acceptance rule (strict improvement + occasional uphill) -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # temperature cools linearly to zero
            temperature = temp0 * max(0.0, 1.0 - (time.time() - start_time) / (budget_s - 5.0))
            if temperature > 0 and np.random.rand() < math.exp(-(cand_score - cur_score) / temperature):
                accept = True
        if accept:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- keep global best -----
        if cur_score < best_score:
            best_seq, best_score = cur_seq.copy(), cur_score
            last_best_time = time.time()
            no_improve = 0
            print(f"[improve] it={it:,}  t={int(time.time() - start_time)}s  best={best_score:.6f}")

        # ----- occasional random jitter to escape plateaus -----
        if it % 2000 == 0:
            noise = np.random.randn(n) * 0.005   # very small Gaussian noise
            pert = cur_seq + noise
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            pert_score = _score(pert)
            if pert_score < cur_score:
                cur_seq, cur_score = pert, pert_score
                if cur_score < best_score:
                    best_seq, best_score = cur_seq.copy(), cur_score
                    print(f"[jitter] it={it:,}  t={int(time.time() - start_time)}s  best={best_score:.6f}")

        # ----- restart from the best known point if stuck for long -----
        if no_improve > 30000:
            # re‑inject a tiny random perturbation around the best solution
            eps = 0.01
            cur_seq = best_seq + eps * np.random.randn(n)
            cur_seq = np.maximum(cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _score(cur_seq)
            no_improve = 0
            print(f"[restart] it={it:,}  t={int(time.time() - start_time)}s  score={cur_score:.6f}")

    print(f"[final] best_score={best_score:.6f} after {it:,} iterations")
    # Convert to plain Python list as required by the surrounding harness
    return best_seq.tolist()
