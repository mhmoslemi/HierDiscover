"""
propose_candidate
-----------------
Generate a non‑negative step‑function whose evaluation score
(2·n·max(conv)/sum²) is as small as possible.

The algorithm works in two phases:

1. **Structured scaling** – start from the Catalan (half‑binomial)
   coefficients `a_i = C(2i,i)/4**i`.  The first 10 % of the entries
   (the “early” part) is multiplied by a factor α∈[0,1] and the mass that
   disappears is redistributed uniformly over the second half of the
   vector (the “tail”).  For many values of α the score drops well below
   the Catalan baseline (π/2 ≈ 1.5708); we scan a fine grid of α and keep
   the best candidate.

2. **Local stochastic refinement** – starting from the best scaled
   sequence we perform a simulated‑annealing / hill‑climbing loop.
   Moves transfer a random fraction of a selected early entry to a set
   of *k* distant tail indices (k≈5‑30).  This spreads the transferred
   mass, keeping the convolution peak low while the L¹‑norm stays
   essentially unchanged.  An exponentially decaying temperature
   accepts occasional worsening moves, and occasional “shakes” (tiny
   Gaussian perturbations) help escape plateaus.

Both phases use an FFT‑based convolution (`_fast_score`) that mirrors the
provided `evaluate_sequence` but without its safety checks, allowing
hundreds of thousands of evaluations per second.  The whole search runs
for at most ~995 s (a 5 s safety margin) and returns the best sequence
found.
"""

# Fast evaluation – identical to the supplied evaluate_sequence (no checks)
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a 1‑D numpy array (no safety checks)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # FFT length – next power of two for efficiency
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _catalan_half_binomial(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function returns before the budget
        expires; a 5‑second safety margin is kept internally).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the runner (≈5 s)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1️⃣  Base Catalan sequence
    # ------------------------------------------------------------------
    n = 5000                     # length – a good trade‑off between speed & quality
    base_seq = _catalan_half_binomial(n)

    # ------------------------------------------------------------------
    # 2️⃣  Define early / tail zones (the zones used for structured moves)
    # ------------------------------------------------------------------
    early_len = max(1, n // 10)          # first 10 %
    tail_start = n // 2                  # start of the “tail”
    tail_len = n - tail_start
    early_sum = base_seq[:early_len].sum()

    # ------------------------------------------------------------------
    # 3️⃣  Structured scaling: scan many α ∈ [1, 0.05]
    # ------------------------------------------------------------------
    best_seq = base_seq.copy()
    best_score = _fast_score(best_seq)
    print(f"Base Catalan score (n={n}): {best_score:.6f}")

    # Scan α from 1.0 down to a small value – 200 steps give fine resolution
    alphas = np.linspace(1.0, 0.05, 200)
    for alpha in alphas:
        # Scale early part and redistribute the removed mass uniformly over the tail
        cand = base_seq.copy()
        cand[:early_len] *= alpha
        removed = (1.0 - alpha) * early_sum          # total mass taken from early zone
        delta = removed / tail_len                    # uniform addition per tail entry
        cand[tail_start:] += delta

        # Clip to stay inside the allowed range (0 … 1000)
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = _fast_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"α={alpha:6.3f} → new best score {best_score:.6f}")

        # Stop early if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Local stochastic refinement (Simulated Annealing)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    cur_best_seq = best_seq.copy()
    cur_best_score = best_score

    # Temperature settings – start proportional to current score
    init_temp = cur_best_score * 0.5
    it = 0

    # Helper to compute temperature (linear decay)
    def temperature(elapsed: float) -> float:
        frac = max(0.0, 1.0 - elapsed / (budget_s - 5.0))
        return init_temp * frac

    # Main SA loop
    while time.time() < deadline:
        it += 1

        # ---- move type: spread mass from an early index to several tail indices ----
        src = np.random.randint(0, early_len)
        src_val = cur_seq[src]
        if src_val <= 1e-12:
            continue

        # fraction of the source to move (0 – 50 %)
        frac = np.random.rand() * 0.5
        mass = src_val * frac
        if mass <= 0.0:
            continue

        # number of distinct tail targets (5‑30 is a good range)
        k = np.random.randint(5, 31)
        targets = np.random.choice(np.arange(tail_start, n), size=k, replace=False)

        # Ensure we do not exceed the per‑element upper bound (1000)
        max_allowed = np.min(1000.0 - cur_seq[targets])
        if max_allowed < mass / k:
            # Scale down the whole transfer so that every target stays ≤1000
            mass = max_allowed * k
        if mass <= 0.0:
            continue
        delta = mass / k

        # Build trial sequence
        trial = cur_seq.copy()
        trial[src] -= mass
        trial[targets] += delta

        # Fast evaluation
        trial_score = _fast_score(trial)

        # Acceptance test (Simulated Annealing)
        elapsed = time.time() - start_time
        temp = temperature(elapsed)
        accept = False
        if trial_score < cur_score:
            accept = True
        elif temp > 1e-12:
            # Metropolis criterion
            prob = math.exp((cur_score - trial_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = trial
            cur_score = trial_score
            if trial_score < cur_best_score:
                cur_best_score = trial_score
                cur_best_seq = trial.copy()
                print(f"Iter {it:07d} – new best score {cur_best_score:.6f}")

        # ---- occasional “shake” to break plateaus ----
        if it % 5000 == 0:
            shake = cur_seq.copy()
            # pick five random tail positions and add tiny Gaussian noise
            idxs = np.random.randint(tail_start, n, size=5)
            shake[idxs] = np.maximum(0.0, shake[idxs] + np.random.randn(5) * 1e-4)
            shake_score = _fast_score(shake)
            if shake_score < cur_best_score:
                cur_best_score = shake_score
                cur_best_seq = shake.copy()
                cur_seq = shake
                cur_score = shake_score
                print(f"Iter {it:07d} – shake improved score to {cur_best_score:.6f}")

        # early exit if target achieved
        if cur_best_score < 1.503:
            print(f"Target reached at iteration {it}, score {cur_best_score:.6f}")
            break

    total_iters = it
    print(f"Finished after {total_iters} iterations. Best score: {cur_best_score:.6f}")
    # Return as plain Python list (required by the outer harness)
    return cur_best_seq.tolist()
