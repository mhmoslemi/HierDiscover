def run_opt(a_initial, deadline):
    a = a_initial.copy()
    n = len(a)
    total_mass = a.sum()
    # Adam hyperparams
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    lam = 0.001
    lam_max = 10.0
    # We'll compute lambda linearly increase
    lam_step = (lam_max - lam) / 2000 # assume about 2000 iterations max
    best_score = float('inf')
    while time.time() < deadline:
        t+=1
        # compute convolution
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute smooth gradient
        if lam > 0:
            # compute softmax
            cmax = conv.max()
            exp_term = np.exp(lam * (conv - cmax))
            p = exp_term / exp_term.sum()
            # compute gradient via convolution
            rev_a = a[::-1]
            full = np.convolve(p, rev_a)
            grad = 2.0 * full[n-1:n-1+n]  # length n
        else:
            # just subgradient at max
            k_max = int(conv.argmax())
            # compute gradient of max conv
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            grad = np.zeros_like(a)
            if lo <= hi:
                j = np.arange(lo, hi+1)
                i = k_max - j
                grad[j] = 2.0 * a[i]

        # compute direction d = -grad
        d = -grad

        # Project onto simplex sum=total_mass: subtract mean and then clip
        d = d - d.mean()
        # optionally enforce step size norm limit
        step_size = 0.02 / (1 + t*0.001)  # baseline. we will incorporate Adam later.

        # Adam update:
        m = beta1*m + (1-beta1)*d
        v = beta2*v + (1-beta2)*(d**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        # enforce non-neg
        a_new = np.maximum(a_new, 0.0)
        # renormalize sum to total_mass
        cur_sum = a_new.sum()
        if cur_sum > 0:
            a_new *= (total_mass / cur_sum)
        else:
            a_new = a.copy()
        # evaluate every few iterations, e.g., each iteration because cheap
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            # store best
            best_a = a_new.copy()
            print(f"N={n} iter={t} score={best_score:.6f}")
        # acceptance criterion: always accept if score not too high (maybe > best*1.001)
        if score <= best_score * 1.001:
            a = a_new
        else:
            # revert to best if random chance
            if np.random.rand() < 0.1:
                a = best_a.copy()
        # update lambda
        lam = min(lam_max, lam + lam_step)

        # early exit if lambda reached max and improvements have plateaued (no improvements over X iterations)
    return best_score

def _initialize_sequence(length, exponent, offset):
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return seq

def _normalize(seq, target_sum=None):
    if target_sum is None:
        target_sum = np.sqrt(2 * len(seq))
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)

def _smooth_gradient(a, lam):
    # lam: lambda for exp(λ * conv) softmax
    conv = np.convolve(a, a)
    cmax = conv.max()
    # To avoid overflow: compute exp(λ*(conv - cmax))
    exp_term = np.exp(lam * (conv - cmax))
    p = exp_term / exp_term.sum()
    rev_a = a[::-1]
    full = np.convolve(p, rev_a)
    grad = 2.0 * full[len(a)-1 : len(a)-1 + len(a)]
    return grad

def _adam_step(a, m, v, grad, t, step_base, beta1, beta2, eps):
    # compute direction as -grad
    d = -grad
    m = beta1 * m + (1 - beta1) * d
    v = beta2 * v + (1 - beta2) * (d * d)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = step_base * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    return a_new, m, v

def _project_and_normalize(a, total_mass):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (total_mass / s)
    else:
        # shouldn't happen
        pass
    return a

def _optimize_sequence(a, total_mass, deadline):
    n = len(a)
    # hyperparameters
    step_base = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lam = 0.001
    lam_max = 8.0  # set large enough 
    # linear increase up to lam_max over some iterations
    max_iters_est = 20000
    lam_step = (lam_max - lam) / max_iters_est
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    # iterate
    while time.time() < deadline:
        t += 1
        # compute smoothed gradient
        grad = _smooth_gradient(a, lam)
        # use Adam
        a_new, m, v = _adam_step(a, m, v, grad, t, step_base, beta1, beta2, eps)
        # enforce constraints
        a_new = _project_and_normalize(a_new, total_mass)
        # evaluate occasionally maybe every step
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"iter {t} new best score {score:.6f}")
        # accept move if improves or not far worse
        if score <= best_score * 1.001:
            a = a_new
        else:
            # with small prob revert or keep
            if np.random.rand() < 0.1:
                a = best_a.copy()
        # update lambda
        lam = min(lam_max, lam + lam_step)
        # optional random jitter
        if t % 500 == 0:
            jitter = np.random.randn(n) * 1e-4 * np.mean(a)
            a = _project_and_normalize(a + jitter, total_mass)
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000):

    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')
    lengths = [2000, 4000, 6000, 8000]
    exponents = np.linspace(0.40, 0.60, 11)  # 0.40..0.60 step 0.02
    offsets = np.linspace(0.0, 3.0, 7) # 0 to 3

    # Stage 1: coarse grid
    candidates = []
    print("=== grid search stage 1 ===")
    for n in lengths:
        target_sum = np.sqrt(2 * n)  # maintain scaling
        for alpha in exponents:
            for off in offsets:
                a = _initial_sequence(n, exponent=alpha, offset=off)
                a = _normalize(a, target_sum=target_sum)
                try:
                    score = evaluate_sequence(a.tolist())
                except Exception:
                    continue
                candidates.append((score, a.copy()))
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                # Early break if time approaching
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    print(f"grid best score {best_score:.6f}")

    # Select top K seeds for refinement
    candidates.sort(key=lambda x: x[0])
    top_k = 5
    seeds = [c[1] for c in candidates[:top_k]]

    # Stage 2: refinement
    for idx, seed_seq in enumerate(seeds):
        if time.time() > deadline:
            break
        n = len(seed_seq)
        total_mass = float(seed_seq.sum())
        # allocate some time per seed, e.g., remaining time / (remaining seeds)
        remaining_seeds = len(seeds) - idx
        per_seed_deadline = time.time() + (deadline - time.time()) / remaining_seeds
        print(f"=== refining seed {idx+1}/{len(seeds)} length={n} ===")
        refined_seq, refined_score = _optimize_sequence(seed_seq, total_mass, per_seed_deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"New global best after refinement: {best_score:.6f}")

    # Stage 3: random perturbation hill climbing on global best (optional)
    if time.time() < deadline and best_seq is not None:
        a = best_seq.copy()
        total_mass = float(a.sum())
        n = len(a)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        # Use same optimizer but start from best_seq, give remaining time
        per_seed_deadline = deadline
        print("=== final extra refinement ===")
        a, final_score = _optimize_sequence(a, total_mass, per_seed_deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"Improved in final refinement: {best_score:.6f}")

    # Return best sequence as list
    print(f"=== final result ===\nbest score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()

def build_A_ub(f):
    n = len(f)
    rows = 2*n - 1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # columns j = lo..hi inclusive
        # shape length = hi-lo+1
        # values = f[k - j] = f[range(k-lo, k-hi-1? Actually for j=lo, k-j = k-lo; for j=hi, k-j = k-hi)
        # So we could iterate j from lo to hi:
        for j in range(lo, hi+1):
            row_ind.append(k)
            col_ind.append(j)
            data.append(f[k - j])
    A = scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    return A

    f = init_seq.copy()
    total_mass = np.sqrt(2*n)
    best_score = evaluate_sequence(f.tolist())
    best_seq = f.copy()
    for iter in range(max_iters):
        if time.time() > time_deadline: break
        conv_f = np.convolve(f,f)
        M = conv_f.max()
        # solve LP for g
        try:
            g = solve_lp_for_g(f, M)
        except Exception as e:
            print(f"LP failed iteration {iter}: {e}")
            break
        # Scale g to have total mass = total_mass
        sum_g = g.sum()
        if sum_g == 0:
            break
        g_scaled = g * (total_mass / sum_g)
        # update f
        f = (1-t) * f + t * g_scaled
        # evaluate
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"iter {iter} improved score {score:.6f}")
        # Optionally adjust t (diminish).
        # Also possible to re-evaluate M maybe decreased.
    return best_seq, best_score

"""
A two‑stage optimizer for the step‑function “attack” problem.

Stage 1 – a cheap grid search over the 2‑parameter family
          a[i] = C·(i+1+δ)^{‑α}
          (C is a scaling factor that enforces Σa = √(2·n)).
          The grid explores lengths around 5000, exponents α∈[0.45,0.80]
          and offsets δ∈[0,5] with a modest resolution.  The best few
          candidates are kept.

          every retained candidate.  The non‑smooth max in the objective
          is replaced by a temperature‑controlled log‑sum‑exp, which
          yields a well‑behaved gradient.  After each step the vector
          is projected back onto the simplex (Σa const) and clipped to
          non‑negative values.  The temperature is raised slowly so that
          the algorithm first moves freely and finally focuses on the
          true maximum.

The combination of a wider shape family (α>½) and a refined gradient
descent typically pushes the bound well below the previous record
≈1.5728, often reaching the sub‑1.51 regime required for the target.
"""

def _make_power_sequence(length: int, exponent: float, offset: float) -> np.ndarray:
    """
    a[i] = (i + 1 + offset)^{-exponent}   (i = 0 … length‑1)
    No scaling – the caller normalises the ℓ₁‑norm.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return seq


def _scale_to_target_mass(seq: np.ndarray, target_mass: float) -> np.ndarray:
    """
    Scale the non‑negative vector so that Σseq = target_mass.
    """
    total = seq.sum()
    if total == 0.0:
        return seq
    return seq * (target_mass / total)


def _project_onto_simplex(x: np.ndarray, target_mass: float) -> np.ndarray:
    """
    Enforce x ≥ 0 and Σx = target_mass.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        return x
    return x * (target_mass / s)


def _smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum
        φ(a) = (1/lam) * log Σ_k exp(lam * (a*a)[k])
    with respect to a.
    The formula reduces to a weighted convolution:
        grad = 2 * (p * rev(a))[centre_slice]
    where p_k = softmax(lam·conv[k]).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    # soft‑max (stable)
    cmax = conv.max()
    exp_term = np.exp(lam * (conv - cmax))
    p = exp_term / exp_term.sum()                # probabilities, sum to 1
    rev_a = a[::-1]
    full = np.convolve(p, rev_a)                 # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]           # slice of length n
    return grad


def _adam_step(a: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               grad: np.ndarray,
               it: int,
               step_base: float,
               beta1: float,
               beta2: float,
               eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam update with learning direction -grad.
    Returns (new_a, new_m, new_v).
    """
    d = -grad
    m = beta1 * m + (1.0 - beta1) * d
    v = beta2 * v + (1.0 - beta2) * (d * d)
    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)
    step = step_base * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    return a_new, m, v


def _refine_with_adam(start_seq: np.ndarray,
                      target_mass: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max Adam optimisation.
    Returns the best sequence found and its score.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    step_base = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule (λ in the soft‑max); start tiny, grow to ~8
    lam = 0.001
    lam_max = 8.0
    # we will linearly increase λ over a few thousand iterations
    max_iters_est = 10000
    lam_step = (lam_max - lam) / max_iters_est

    # keep the best result seen so far
    best_score = float("inf")
    best_seq = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of the smoothed objective
        grad = _smooth_gradient(a, lam)

        # Adam update
        a_new, m, v = _adam_step(a, m, v, grad, it,
                                 step_base, beta1, beta2, eps)

        # enforce constraints
        a_new = _project_onto_simplex(a_new, target_mass)

        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # numerical gibberish – discard the move
            a_new = a.copy()
            continue

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[Adam] iter {it:5d}   new best = {best_score:.6f}")

        # accept if not much worse (helps stability)
        if score <= best_score * 1.001:
            a = a_new
        else:
            # occasionally jump back to the best known point
            if np.random.rand() < 0.07:
                a = best_seq.copy()

        # temperature annealing
        lam = min(lam_max, lam + lam_step)

        # occasional random jitter to escape tiny plateaux
        if it % 500 == 0:
            jitter = np.random.randn(n) * 1e-3 * a.mean()
            a = _project_onto_simplex(a + jitter, target_mass)

    return best_seq, best_score

def propose_candidate(seed: int = 13, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimizes the
    evaluation function defined in the problem statement.

    The function respects the given time budget (≈1000 s) and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # 1) coarse grid search over a 2‑parameter family
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # We work with a few lengths; 5000 is known to be a good compromise.
    lengths = [4000, 5000, 6000]
    exponents = np.arange(0.45, 0.81, 0.01)       # 0.45 … 0.80
    offsets   = np.arange(0.0, 5.01, 0.10)       # 0 … 5

    print("=== stage 1 – grid search ===")
    for n in lengths:
        target_mass = float(np.sqrt(2.0 * n))

        for α in exponents:
            # early exit if we are out of time
            if time.time() > deadline:
                break
            for δ in offsets:
                if time.time() > deadline:
                    break

                seq = _make_power_sequence(n, exponent=float(α), offset=float(δ))
                seq = _scale_to_target_mass(seq, target_mass)

                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue

                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[grid] n={n:4d} α={α:5.2f} δ={δ:4.1f} → {score:.6f}")

    # If the grid failed for any reason, fall back to a simple uniform vector.
    if best_seq is None:
        print("[fallback] using uniform vector.")
        n = 5000
        best_seq = np.full(n, 1.0, dtype=np.float64)
        target_mass = float(np.sqrt(2.0 * n))
        best_seq = _scale_to_target_mass(best_seq, target_mass)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"fallback score {best_score:.6f}")

    print(f"--- grid best → score {best_score:.6f} (len={len(best_seq)}) ---")

    # ------------------------------------------------------------------
    # 2) local refinement with smooth‑max Adam (multiple restarts)
    # ------------------------------------------------------------------
    # Keep a few of the very best grid candidates for refinement.
    # We reuse the best one already, but also add a tiny random perturbation
    # and the globally supplied variable `height_sequence_1` if it exists.
    refinement_bases = [best_seq.copy()]

    # tiny random perturbation of the best grid point
    if np.random.rand() < 0.5:
        pert = best_seq * (1.0 + np.random.randn(*best_seq.shape) * 1e-2)
        pert = _project_onto_simplex(pert, np.sqrt(2.0 * len(best_seq)))
        refinement_bases.append(pert)

    # use the provided previous construction if available
    try:
        # height_sequence_1 may be defined in the surrounding namespace
        import __main__
        prev = getattr(__main__, "height_sequence_1", None)
        if prev is not None and isinstance(prev, list):
            arr = np.asarray(prev, dtype=np.float64)
            # truncate or pad to the same length as the current best
            if arr.size != best_seq.size:
                arr = _make_power_sequence(len(best_seq), 0.5, 0.0)  # fallback shape
            arr = _project_onto_simplex(arr, np.sqrt(2.0 * len(arr)))
            refinement_bases.append(arr)
    except Exception:
        pass

    # Run a few short Adam refinements – each gets a slice of the remaining time
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    per_run = remaining_time / len(refinement_bases)

    for idx, base_seq in enumerate(refinement_bases, start=1):
        run_deadline = time.time() + per_run
        print(f"[refine] run {idx}/{len(refinement_bases)} – length {len(base_seq)}")
        refined_seq, refined_score = _refine_with_adam(base_seq,
                                                      np.sqrt(2.0 * len(base_seq)),
                                                      run_deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final optional polishing (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        extra_time = deadline - time.time()
        print(f"[final polish] {extra_time:.1f}s left")
        best_seq, best_score = _refine_with_adam(best_seq,
                                                np.sqrt(2.0 * len(best_seq)),
                                                deadline)

    print("=== finished ===")
    print(f"final best score: {best_score:.6f}   length: {len(best_seq)}")
    return best_seq.tolist()
