    n = len(a)
    M = np.max(np.convolve(a, a))
    # variables b of length n
    # build A_ub as list of rows
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # for j in lo..hi, coefficient a[j] for b[k-j]
        # For each j: row k, col (k-j) gets a[j]
        # Might be many entries; easier to fill using loops
        for j in range(lo, hi+1):
            rows.append(k)
            cols.append(k - j)
            data.append(a[j])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Additional constraint for sum? We'll scale later.
    # Bounds for b: (0, None)
    # Objective: minimize -sum(b) => c = -np.ones(n)
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')

    sum_b0 = b0.sum()
    # scaling factor

    # mixing
    a_new = (1 - t) * a + t * b_scaled

def lp_step(a, t=0.5):
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()

    # Build A_ub
    i = np.arange(n)
    j = np.arange(n)
    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.repeat(i, n)
    data = np.repeat(a, n)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))

    # Build bounds
    bounds = [(0, None)] * n

    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)

    # RHS
    b_ub = np.full(2*n - 1, M)

    a_new = np.maximum(a_new, 0.0)

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. **Initial seed** – try a few modest lengths (1500…2500) and use the
   classic arcsine shape a[i] = (i+1+offset)^‑0.5, scaled so that
   Σa = sqrt(2·n). Keep the best seed (including the optional
   ``height_sequence_1`` if it exists).

2. **Linear‑programming refinement** – for the current sequence a
   compute M = max(a∗a).  Build the linear constraints
        (a * b)_k ≤ M   for every convolution index k,
   where b is a new non‑negative vector of the same length.
   Maximise Σb (i.e. minimise –Σb) using `scipy.optimize.linprog`
   (high‑s) with a sparse constraint matrix.  The optimal b pushes
   as much ℓ₁‑mass as possible into positions that do **not** increase
   the maximal convolution value.

   Scale b to the required ℓ₁‑norm (sqrt(2·n)) and replace a by the
   scaled vector (full replacement).  This operation never increases
   M and typically reduces it substantially.

3. **Iterate** – repeat the LP step while time permits (≈30 – 50
   iterations for a length ≈2000).  Occasionally add a tiny random
   jitter to escape plateaus and keep the sequence feasible.

4. **Return** – the best sequence seen during the run.

The whole routine respects the 1000 s budget, prints progress and
never writes to disk.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------
# ----------------------------------------------------------------------


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Non‑negative power‑law seed:  a[i] = (i+1+offset)^{-exponent}.
    The values are already clipped to the [0,1000] range required by
    ``evaluate_sequence``.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _lp_update(a: np.ndarray, total_sum: float) -> np.ndarray:
    """
    Perform a single LP update for a given sequence ``a``.
    Returns a new sequence (scaled LP solution) or the original ``a`` if
    the LP fails.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()

    # Build the sparse matrix representing (a * b)_k ≤ M.
    # Row k corresponds to convolution index k (0 … 2n‑2).
    i = np.arange(n, dtype=np.int64)
    rows = (i[:, None] + i[None, :]).ravel()          # shape (n*n,)
    cols = np.repeat(i, n)                           # each column i repeated n times
    data = np.repeat(a, n)                           # a[i] repeated n times

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear programme: maximise Σb  <=> minimise –Σb
    c = -np.ones(n, dtype=np.float64)                # objective
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # RHS of the inequality constraints
    bounds = [(0.0, None) for _ in range(n)]         # b ≥ 0

    # Solve (highs is fast and accepts sparse matrices)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'disp': False})

    if not res.success:
        # LP failed – return the original vector unchanged
        return a

    b0 = res.x
    sum_b0 = b0.sum()
    if sum_b0 <= 0.0:
        return a

    # Scale the LP solution to the required ℓ₁‑norm (total_sum)
    scale = total_sum / sum_b0
    g = b0 * scale

    # Full replacement (the theory guarantees M does not increase)
    a_new = np.clip(g, 0.0, 1000.0)
    return a_new


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 990, **kwargs):
    """
    Search for a coefficient sequence that minimises the provided
    evaluation function.  The search runs for approximately ``budget_s``
    seconds and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    print("=== Phase 0 – initial seeds ===")
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0a – optional user‑provided construction (if present)
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be a global list if it exists.
        hs1 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if hs1.size > 0:
            # normalise to the required ℓ₁‑norm (the evaluation is scale‑invariant)
            n_hs1 = hs1.shape[0]
            hs1 = hs1 * (np.sqrt(2 * n_hs1) / hs1.sum())
            sc = _score(hs1)
            best_seq, best_score = hs1, sc
            print(f"seed from height_sequence_1 (len={n_hs1}) → score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 0b – grid of arcsine‑type seeds (moderate lengths)
    # --------------------------------------------------------------
    lengths = [1500, 1800, 2100, 2400, 2700]
    offsets = [0.0, 0.3, 0.7, 1.2]     # a few offsets to break symmetry
    exponent = 0.5                     # the classic arcsine exponent

    for n in lengths:
        for off in offsets:
            seq = _initial_sequence(n, exponent=exponent, offset=off)
            # normalise ℓ₁‑norm to sqrt(2·n) (the objective is homogeneous)
            seq = seq * (np.sqrt(2 * n) / seq.sum())
            sc = _score(seq)
            print(f"seed len={n:4d} offset={off:4.1f} → score={sc:.6f}")
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

    if best_seq is None:
        # fallback: uniform vector (guaranteed to be feasible)
        n = 2000
        best_seq = np.ones(n, dtype=np.float64) * (np.sqrt(2 * n) / n)
        best_score = _score(best_seq)
        print("fallback uniform seed → score={:.6f}".format(best_score))

    print(f"Initial best: len={len(best_seq)} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – LP based refinement (the heart of the approach)
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)   # Σa is kept constant
    iteration = 0
    last_improve = 0

    while time.time() < deadline:
        iteration += 1

        # one LP step (full replacement)
        a_new = _lp_update(a, target_sum)

        # tiny random jitter every few iterations to escape flat regions
        if iteration % 7 == 0:
            jitter = np.random.randn(n) * 1e-4
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * (target_sum / a_new.sum())   # re‑normalise

        # evaluate the candidate
        try:
            sc = _score(a_new)
        except Exception:
            # defensive – keep the old vector
            sc = best_score + 1.0
            a_new = a

        # keep the globally best sequence
        if sc < best_score:
            best_score, best_seq = sc, a_new.copy()
            last_improve = iteration
            print(f"iter {iteration:5d}: improved score={best_score:.6f}")

        # accept the new vector if it does not worsen the objective
        if sc <= best_score * 1.001:
            a = a_new
        else:
            # occasionally fall back to the best known vector
            if np.random.rand() < 0.1:
                a = best_seq.copy()

        # early stop if we have already beaten the target comfortably
        if best_score <= 1.51:
            print("Target region reached – stopping early.")
            break

        # safety break – if no improvement for many iterations, quit
        if iteration - last_improve > 150:
            print("No improvement for a while – terminating.")
            break

    print("=== finished ===")
    print(f"final best score: {best_score:.6f}   length: {len(best_seq)}")
    # The evaluator expects a plain Python list
    return best_seq.tolist()
