from scipy.optimize import linprog

def improve_via_lp(a, target_sum, max_iter=5, time_limit=None):
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    start = time.time()
    for it in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        # compute max convolution
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        # build matrix
        A = build_convolution_matrix(best_seq)
        b = np.full(A.shape[0], M)
        # objective: minimize -sum(b) (i.e., maximize sum(b))
        c = -np.ones(best_seq.shape[0])
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break
        b_opt = res.x
        if b_opt.sum() == 0:
            break
        # scale to target sum
        b_opt = b_opt * (target_sum / b_opt.sum())
        # mixing
        t = 0.3  # could reduce gradually
        a_candidate = (1 - t) * best_seq + t * b_opt
        a_candidate = np.maximum(a_candidate, 0.0)
        # Normalize to target sum (should already be)
        a_candidate = a_candidate * (target_sum / a_candidate.sum())
        candidate_score = evaluate_sequence(a_candidate.tolist())
        if candidate_score < best_score:
            best_score = candidate_score
            best_seq = a_candidate.copy()
            print(f"LP improvement {it}: new best {best_score:.6f}")
            # maybe increase t for next iteration
        else:
            # no improvement, break
            print(f"LP iteration {it} no improvement (score {candidate_score:.6f})")
            break
    return best_seq, best_score

"""
Propose a coefficient sequence that minimizes the evaluation function.

The algorithm combines several ideas:
1.  A quick parametric sweep over families that are known to give low
    autocorrelation (square‑root differences, exponentials, half‑sine,
    Gaussian, triangular, random).
2.  A simulated‑annealing / hill‑climbing phase that repeatedly moves a
    small amount of mass between two positions (or mixes two positions)
    while keeping the L1‑norm fixed to √(2 n).  Moves that improve the
    score are always kept, worsening moves are occasionally accepted
    according to a linear temperature schedule.
3.  An optional linear‑programming refinement (uses SciPy’s `linprog` if
    available).  For the current best shape `a` the LP maximises the total
    mass of a new vector `b` under the constraints
        (a * b)[k] ≤ max(a * a)  for all lags k,
    which yields a vector that spreads the mass more evenly.  A convex
    combination of the old and the new vector is taken and the process is
    repeated a few times.
4.  A short polishing phase with very small random transfers.

All sequences are kept non‑negative, normalized to the target L1 norm
√(2 n) (which makes the score equal to the maximal convolution value) and
finally rescaled to the full [0, 1000] range – the rescaling does not
affect the score because the metric is homogeneous.

The function respects the 1000 s budget and prints progress information.
"""

import time
import math
import numpy as np

# Optional linear‑programming support
try:
    from scipy.optimize import linprog  # type: ignore
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False


def _normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and scale to the required L1 sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return v
    return v * (target_sum / s)


def _generate_diff_pow(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^p - i^p   (discrete derivative of x^p)."""
    x = np.arange(n + 1, dtype=np.float64)
    a = np.power(x[1:], p) - np.power(x[:-1], p)
    return _normalize(a, target_sum)


def _generate_exponential(n: int, r: float, target_sum: float) -> np.ndarray:
    """a_i = r^i   (geometric decay)."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    return _normalize(a, target_sum)


def _generate_half_sine(n: int, power: float, target_sum: float) -> np.ndarray:
    """a_i = sin(π(i+0.5)/(2n))^power."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / (2.0 * n)
    a = np.sin(theta) ** power
    return _normalize(a, target_sum)


def _generate_gaussian(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    centre = (n - 1) / 2.0
    xs = np.arange(n, dtype=np.float64)
    a = np.exp(-((xs - centre) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a, target_sum)


def _generate_triangular(n: int, target_sum: float) -> np.ndarray:
    """a_i = min(i+1, n‑i)   (symmetric triangular shape)."""
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    return _normalize(a, target_sum)


def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ b)[k] = (a * b)[k].
    Row k corresponds to lag k, column j to b_j.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    j = np.arange(n, dtype=np.int64)
    for i in range(n):
        rows_idx = i + j                     # k = i + j
        A[rows_idx, j] += a[i]              # contribution a_i * b_j
    return A


def _gradient_descent_step(a: np.ndarray, target_sum: float,
                          step_scale: float = 0.001) -> np.ndarray:
    """
    One projected sub‑gradient step that reduces the current convolution peak.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    if i_low <= i_high:
        i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_vals = k_max - i_vals
        grad[i_vals] = a[i_vals] + a[j_vals]

    # zero‑mean to keep the total mass unchanged after the step
    grad = grad - grad.mean()
    direction = -grad
    step = step_scale * (np.linalg.norm(direction) / n + 1e-12)
    a_new = a + step * direction
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, target_sum)
    return a_new


def _lp_refinement(a: np.ndarray, target_sum: float,
                   max_iter: int = 6, time_limit: float = None):
    """
    Linear‑programming refinement.
    For the current shape a, maximise Σ b_j under the constraints
        (a * b)[k] ≤ max(a * a)   ∀k,
    then mix a and the optimal b.
    """
    if not SCIPY_AVAILABLE:
        return a, evaluate_sequence(a.tolist())

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    start = time.time()

    for it in range(max_iter):
        if time_limit is not None and (time.time() - start) > time_limit:
            break

        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        A = _build_convolution_matrix(best_seq)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(best_seq.shape[0], dtype=np.float64)   # maximise Σb

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        except Exception as exc:      # pragma: no cover
            print(f"LP solve exception: {exc}")
            break

        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break

        b_opt = res.x
        if b_opt.sum() == 0.0:
            break

        # scale to the required L1 sum
        b_opt = _normalize(b_opt, target_sum)

        # mix – start with a relatively large step, then become more conservative
        t = 0.5 if it == 0 else 0.2
        candidate = (1.0 - t) * best_seq + t * b_opt
        candidate = _normalize(candidate, target_sum)

        cand_score = evaluate_sequence(candidate.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"LP iteration {it} improved score to {best_score:.6f}")
        else:
            print(f"LP iteration {it} no improvement (score {cand_score:.6f})")
            break

    return best_seq, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given metric.
    The function respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                # use the provided global if present
    except Exception:                            # noqa: BLE001
        n = 1542

    target_sum = math.sqrt(2.0 * n)               # makes the score equal to max convolution

    # ------------------------------------------------------------------
    # 2. Helper to evaluate a candidate (avoids re‑creating the list each time)
    # ------------------------------------------------------------------
    def _score(vec):
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 3. Initial parametric sweep
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # baseline (square‑root‑difference)
    a0 = _generate_diff_pow(n, 0.5, target_sum)
    best_seq = a0.copy()
    best_score = _score(best_seq)
    print(f"Baseline sqrt‑diff score {best_score:.6f}")

    # power‑law differences
    for p in np.linspace(0.15, 0.9, 15):
        a = _generate_diff_pow(n, p, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Better diff p={p:.3f} → {sc:.6f}")

    # exponential decay
    for r in np.linspace(0.85, 0.9999, 30):
        a = _generate_exponential(n, r, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Better exp r={r:.5f} → {sc:.6f}")

    # half‑sine family
    for pw in [0.5, 1.0, 2.0, 3.0]:
        a = _generate_half_sine(n, pw, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Better half‑sine power={pw} → {sc:.6f}")

    # Gaussian shapes
    for sigma in np.linspace(n / 8.0, n / 2.0, 12):
        a = _generate_gaussian(n, sigma, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Better Gaussian σ={sigma:.1f} → {sc:.6f}")

    # triangular shape
    a = _generate_triangular(n, target_sum)
    sc = _score(a)
    if sc < best_score:
        best_score, best_seq = sc, a.copy()
        print(f"Better triangular → {sc:.6f}")

    # a few random starts
    for _ in range(20):
        a = np.random.rand(n)
        a = _normalize(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"Better random → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing / hill‑climbing phase
    # ------------------------------------------------------------------
    a = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        T = max(0.01, 1.0 - elapsed / (deadline - start_time))

        a_new = a.copy()
        move_type = np.random.rand()

        if move_type < 0.6:                     # move a small mass from i to j
            i, j = np.random.choice(n, 2, replace=False)
            if a_new[i] > 0.0:
                eps = a_new[i] * np.random.rand() * 0.07   # ≤7 % of source
                a_new[i] -= eps
                a_new[j] += eps
        elif move_type < 0.85:                  # mix two positions
            i, j = np.random.choice(n, 2, replace=False)
            alpha = np.random.rand() * 0.25
            ai, aj = a_new[i], a_new[j]
            a_new[i] = (1 - alpha) * ai + alpha * aj
            a_new[j] = alpha * ai + (1 - alpha) * aj
        else:                                   # recreate a small block
            start_idx = np.random.randint(0, n - 4)
            block_len = np.random.randint(3, min(30, n - start_idx))
            a_new[start_idx:start_idx + block_len] = np.random.rand(block_len)

        a_new = _normalize(np.maximum(a_new, 0.0), target_sum)
        sc_new = _score(a_new)

        if sc_new < best_score:
            best_score, best_seq = sc_new, a_new.copy()
            a = a_new
            print(f"Iter {it:5d} improved best → {best_score:.6f}")
        else:
            # Metropolis acceptance of worsening moves
            if np.random.rand() < math.exp(-(sc_new - best_score) / (T + 1e-12)):
                a = a_new

        # every few hundred iterations perform a tiny sub‑gradient refinement
        if it % 750 == 0 and time.time() < deadline:
            a = _gradient_descent_step(a, target_sum)

    # ------------------------------------------------------------------
    # 5. Linear‑programming refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE and time.time() < deadline:
        remaining = deadline - time.time()
        # allow up to half of the remaining time for the LP stage
        lp_time = min(remaining * 0.5, 30.0)
        refined_seq, refined_score = _lp_refinement(best_seq, target_sum,
                                                    max_iter=6,
                                                    time_limit=lp_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"LP refinement yielded new best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short polishing using tiny random transfers
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    a = best_seq.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = a[i] * np.random.rand() * 0.02
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = _normalize(np.maximum(a_new, 0.0), target_sum)
        sc = _score(a_new)
        if sc < best_score:
            best_score, best_seq = sc, a_new.copy()
            a = a_new
            print(f"Polish improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Rescale to the full [0, 1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * (1000.0 / best_seq.max())
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning final sequence with score {best_score:.6f}")
    return final_seq.tolist()
