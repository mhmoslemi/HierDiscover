   vectors.  All candidates are normalised to have sum = 1.

2. **Fast Screening** – Evaluate the exact objective
   `2·n·max(conv)` (the denominator cancels for normalised vectors) by
   an FFT‑based routine and keep the best few seeds.

3. **Intensive L‑BFGS** – For each seed run a continuation of L‑BFGS on a
   log‑sum‑exp smoothed version of the max‑convolution.  The smoothing
   parameter λ is driven geometrically from 0.4·max to 1e‑8·max,
   allowing the optimiser to approach the true non‑smooth objective.
   The loop is repeated with random blends of the current best and fresh
   seeds until the allocated time slice expires.

4. **Adam on the Exact Objective** – Switch to an Adam optimiser that
   works on the exact max (sub‑gradient obtained via a single FFT).  This
   refines the solution on the true landscape.

5. **Large‑Scale Simulated Annealing** – Perform a cheap pairwise
   mass‑transfer annealing with a long schedule (≈ 120 k moves) to escape
   possible plateaus.

6. **Deterministic Hill‑Climb Polish** – Finish with a greedy
   pairwise‑transfer hill‑climb that only accepts strictly improving
   moves.

All phases respect the global deadline (≈ 1000 s).  The final sequence
is returned as a plain list of floats; `evaluate_sequence` will be
called by the harness to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a * a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f   = 2·n·smooth_max(conv)
       grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)    # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z                    # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (various deterministic / random families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def power_law_init(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^{-p} + (n-i)^{-p}, symmetrised version."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n - i + 1) ** (-p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in ``lam_schedule``, starting each stage
    from the previous result. Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits (a = softmax(y)) so that the simplex constraint
    is satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and its max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient wrt a of the max entry (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # full objective gradient (2·n·max)
        grad_a = 2.0 * n * g_a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)      # decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaux
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Pairwise mass‑transfer annealing.  Moves are accepted with the
    usual Metropolis criterion.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # move a small random fraction of mass from i to j
        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power‑of‑two FFT size works best)
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # Deterministic families
    candidates.append(np.full(n, 1.0 / n))                # uniform
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # Explicit arcsine (Beta(0.5,0.5))
    candidates.append(beta_init(n, 0.5, 0.5))

    # Parametric beta shapes (grid search)
    for _ in range(20):
        alpha = rng.uniform(0.1, 2.5)
        beta = rng.uniform(0.1, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # Power‑law shapes
    for _ in range(10):
        p = rng.uniform(0.2, 2.5)
        candidates.append(power_law_init(n, p))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Keep the best few (fast_score is exact for sum = 1)
    scored = [(fast_score(c, fft_len, n), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 6
    best_sc, best_a = scored[0]
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS intensive refinement
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # allocate ~45 %
    lam_schedule = np.geomspace(0.4, 1e-8, num= fifty)  # will be defined below
    # Use a reasonable number of λ‑steps
    lam_schedule = np.geomspace(0.4, 1e-8, num=45)
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # blend current best with a fresh random vector for diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Long simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(30.0, deadline - time.time())
        print("[SA] start long annealing")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-8,
                                              max_moves=200000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + min(20.0, deadline - time.time())
        print("[Hill‑climb] start polishing")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=120000,
                                     delta_frac=0.025)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
