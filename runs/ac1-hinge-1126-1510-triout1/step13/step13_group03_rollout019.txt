"""
Propose a very low‑score non‑negative step function.

Algorithm
---------
1. Choose a problem size.  We use a power‑of‑two length because FFTs are
   fastest there.  If the time budget permits we go up to 8192
   (twice the default 4096) – a larger search space gives a better
   chance to beat the 1.503 target.

2. Build a diverse pool of initial candidates (deterministic windows,
   random Dirichlet draws, mixtures of Gaussians, a few sparse vectors).
   The pool is normalised to a true probability vector (∑a=1).

3. Phase 1 – many independent L‑BFGS runs on the *smooth* max
   (log‑sum‑exp) objective.  Each run starts from a random convex
   combination of the current best solution and a fresh random vector.
   A schedule of smoothing parameters (λ) drives the optimisation from
   a very smooth surrogate to the true max.

4. Phase 2 – exact sub‑gradient descent on the real max‑convolution.
   The sub‑gradient is built from the active constraints (those
   lags where the current autocorrelation is within a tiny ε of the
   maximum).  An exponentiated‑gradient (multiplicative) update keeps the
   iterate on the simplex.

5. Phase 3 – Adam on the exact objective (still on the simplex via a
   softmax parametrisation).  The learning‑rate decays as 1/√t and a
   tiny Gaussian jitter is added periodically to escape plateaus.

6. Phase 4 – a cheap simulated‑annealing mass‑transfer move set.
   Randomly pick two positions i≠j and move a small fraction of a_i
   to a_j.  Acceptance follows a standard Boltzmann rule with an
   exponential cooling schedule.

7. Phase 5 – deterministic greedy hill‑climb (tiny pairwise transfers)
   for the last polishing.  Moves that improve the objective are kept,
   otherwise they are undone.

All phases respect a global 1000 s deadline; each phase stops early
if the deadline is reached.  The best vector seen during the whole run
is returned.  The official ``evaluate_sequence`` rescales internally,
so the optimiser works on the probability simplex (∑a=1).
"""

# ----------------------------------------------------------------------
# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax: ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean project onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective for vectors on the simplex (∑a = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return the exact evaluation for a on the simplex:
        2·n·max(a∗a)
    (the denominator Σa² == 1, therefore the full formula reduces to this.)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective   f = 2·n·smooth_max(conv)
    and its gradient w.r.t. the logits y (a = softmax(y)).
    """
    a = softmax(y)

    # ---- convolution a*a (linear, zero‑padded) ----
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # ---- log‑sum‑exp smooth approximation of the max ----
    shifted = conv_full - max_c                # ≤ 0
    exp_shift = np.exp(shifted / lam)          # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. the convolution entries ----
    w = exp_shift / Z                           # shape (2n‑1,)

    # ---- back‑prop through the convolution ----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # ---- back‑prop through softmax (standard formula) ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """A smooth shape obtained by taking the sqrt of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using several heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'lognorm'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def mixture_gaussian_init(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Gaussians with random parameters."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def dirichlet_sparse_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet draw."""
    alpha = rng.uniform(0.01, 0.2, size=n)
    a = rng.gamma(alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run a cascade of L‑BFGS optimisations, each with a tighter
    smoothing parameter λ.  Each stage starts from the result of the
    previous stage.  Returns the best vector seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits (ensures positivity)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:               # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         eps: float = 1e-12):
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Multiplicative (exponentiated‑gradient) updates keep the iterate on
    the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # ---- convolution a*a (full) ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        # mask of “active” lags (within eps of the max)
        mask = (conv >= max_c - eps).astype(np.float64)

        # sub‑gradient of the max wrt a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # step size 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # EG update & renormalise
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a /= s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[Subgrad] it={it}, best fast_score={best_sc:.9f}")

        it += 1

    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Parameters are logits y (a = softmax(y)) to keep the simplex constraint.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # ---- exact convolution & max ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # ---- sub‑gradient w.r.t. a ----
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # ∂ (2·n·max) / ∂ a
        grad_a = 2.0 * n * g_a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out tiny plateaus
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.9f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Simple pairwise mass‑transfer annealing.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # propose moving a small random fraction of a_i to a_j
        delta = a[i] * rng.random() * 0.16
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – try to use the largest power‑of‑two that comfortably
    # fits into the time budget.  8192 usually works within 1 000 s.
    # --------------------------------------------------------------
    n = 8192                               # raise to 8192 if time permits
    fft_len = next_pow2(2 * n - 1)         # length for zero‑padded FFTs

    # --------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse a previous solution if the global variable is present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(mixture_gaussian_init(n, rng, K=rng.integers(2, 5)))
    candidates.append(dirichlet_sparse_init(n, rng))

    # pick the best of the pool using the *exact* fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS continuation (global restarts)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # about 30 % of total time
    lam_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1,
                    0.07, 0.05, 0.03, 0.02, 0.01,
                    0.006, 0.004, 0.002, 0.001, 5e-4,
                    2e-4, 1e-4, 5e-5]          # increasingly tight
    maxiter_per_lam = 1400

    while time.time() < phase1_end:
        # blend the current best with a fresh random start
        rand_vec = random_initial(n, rng)
        if rng.random() < 0.30:
            rand_vec = mixture_gaussian_init(n, rng, K=rng.integers(2, 4))
        alpha = rng.random() * 0.28   # keep most of the old solution
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on the true max
    # --------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.20 * budget_s
        print("[Subgrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                                subgrad_deadline,
                                                n,
                                                fft_len,
                                                eps=1e-14)
        print(f"[Subgrad] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start (exact objective)")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(18.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=150000)
        print(f"[SA] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – Final greedy hill‑climb (polish)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 22.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=80000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Return the best vector found.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.9f}")
    return final_seq
