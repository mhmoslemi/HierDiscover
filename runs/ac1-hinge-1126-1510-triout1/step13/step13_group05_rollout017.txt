"""
Advanced search for a low‑score step function.

The algorithm builds a rich pool of candidates (deterministic windows,
randomised families, beta‑density families, spectral “sqrt‑uniform”
guess).  It then performs several optimisation phases:

1.  L‑BFGS polishing of a smoothed max‑convolution objective with a
    decreasing smoothing schedule.
2.  Adam optimisation of the same smoothed objective.
3.  Adam minimisation of the L₂ distance between the self‑convolution
    and the perfectly uniform distribution (the theoretical lower bound).
4.  Exponentiated‑gradient descent on the exact max‑convolution.
5.  Projected sub‑gradient descent on the exact max‑convolution.
6.  Simulated‑annealing + fine‑grained hill‑climbing until the time
    budget expires.

All updates keep the vector on the probability simplex, so the final
score returned by `evaluate_sequence` is invariant to the normalisation
performed inside that function.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact when the vector is normalised to sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density discretisation on the simplex."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    """(1‑cos)ᵖ window (generalised Hann)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a). Adaptive step size helps escape plateaus.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Focuses on the top `top_fraction` of convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # project gradient onto tangent space of simplex
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.003):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Flattening phase – minimise L2 distance to uniform convolution
# ----------------------------------------------------------------------


def flatten_adam(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 target_conv: np.ndarray,
                 rng: np.random.Generator,
                 max_iters: int = 6000,
                 lr: float = 0.06):
    """
    Adam optimiser for the L2 loss ‖a*a – target‖².
    Keeps the iterate on the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 1
    while time.time() < deadline and it <= max_iters:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        h = conv - target_conv

        # gradient = 2 * rev_a convolve h
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fh = np.fft.rfft(h, fft_len)
        grad_full = np.fft.irfft(fa_rev * fh, fft_len)
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # Adam update on a (before projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score

"""
Advanced search for a low‑score step function.

The algorithm builds a rich pool of candidates (deterministic windows,
random families, Beta‑density families and a spectral “sqrt‑uniform”
guess).  It then runs several optimisation phases:

1. L‑BFGS polishing of a smoothed max‑convolution objective.
2. Adam optimisation of the same smoothed objective.
3. Adam minimisation of the L₂ distance between the self‑convolution
   and the perfectly uniform distribution (the theoretical lower bound).
4. Exponentiated‑gradient descent on the exact max‑convolution.
5. Projected sub‑gradient descent on the exact max‑convolution.
6. Simulated‑annealing + fine‑grained hill‑climbing until the time
   budget expires.

All updates keep the vector on the probability simplex, so the final
score returned by `evaluate_sequence` is invariant to the normalisation
performed inside that function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact when the vector is normalised to sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density discretisation."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    """(1‑cos)ᵖ window (generalised Hann)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a). Adaptive step size helps escape plateaus.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Focuses on the top `top_fraction` of convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.003):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Flattening phase – minimise L₂ distance to the uniform convolution
# ----------------------------------------------------------------------


def flatten_adam(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 target_conv: np.ndarray,
                 rng: np.random.Generator,
                 max_iters: int = 6000,
                 lr: float = 0.06):
    """
    Adam optimiser for the L₂ loss ‖a*a – target‖².
    Keeps the iterate on the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 1
    while time.time() < deadline and it <= max_iters:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        h = conv - target_conv

        # gradient = 2 * rev_a convolve h
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fh = np.fft.rfft(h, fft_len)
        grad_full = np.fft.irfft(fa_rev * fh, fft_len)
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Interpolate previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Spectral “sqrt‑uniform” guess (solve a*a ≈ uniform)
    target_conv = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    t_fft = np.fft.rfft(target_conv, fft_len)
    sqrt_fft = np.sqrt(t_fft)                     # element‑wise complex sqrt
    a_spec = np.real(np.fft.ifft(sqrt_fft))[:n]
    a_spec[a_spec < 0] = 0.0
    a_spec = a_spec / a_spec.sum() if a_spec.sum() > 0 else np.full(n, 1.0 / n)
    candidates.append(a_spec)

    # Random families
    for _ in range(12):
        candidates.append(random_initial(n, rng))
    for _ in range(8):
        candidates.append(pow_decay_init(n, rng))

    # Random Beta densities
    for _ in range(300):
        alpha = rng.exponential(scale=0.8) + 0.05
        beta_param = rng.exponential(scale=0.8) + 0.05
        candidates.append(beta_init(n, alpha, beta_param))

    # Raised‑cosine family
    for _ in range(120):
        p = rng.uniform(0.4, 3.5)
        candidates.append(raised_cosine_init(n, p))

    # ------------------------------------------------------------------
    # Choose best initial candidate (fast_score = 2·n·max(conv))
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing on smoothed max‑convolution
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha_mix = rng.random() * 0.3
        a_start = (1.0 - alpha_mix) * best_a + alpha_mix * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation of the smoothed max‑convolution
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=8000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Flattening: minimise L₂ distance to uniform convolution
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        best_a, best_score = flatten_adam(best_a,
                                          phase3_end,
                                          n,
                                          fft_len,
                                          uniform_target,
                                          rng,
                                          max_iters=8000,
                                          lr=0.06)
        print(f"[flatten‑Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exponentiated‑gradient on the exact max‑convolution
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.80 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase4_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.015)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Exact‑max sub‑gradient descent
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.90 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase5_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.20)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Simulated annealing + fine hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Simulated annealing block (if enough time left)
        if deadline - time.time() > 30.0:
            a_sa, sc_sa = simulated_annealing(best_a,
                                              deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=250000)
            if sc_sa < best_score:
                best_score = sc_sa
                best_a = a_sa.copy()
                print(f"[SA] new best fast_score={best_score:.8f}")

        # Fine‑grained hill climb
        a_hc, sc_hc = hill_climb_fine(best_a,
                                      best_score,
                                      fft_len,
                                      n,
                                      deadline,
                                      rng,
                                      max_iter=250000,
                                      delta_start=0.04,
                                      delta_end=0.003)
        if sc_hc < best_score:
            best_score = sc_hc
            best_a = a_hc.copy()
            print(f"[hill‑climb fine] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose for possible future runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
