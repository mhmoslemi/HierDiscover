    best_a = a.copy()

    iteration = 0
    no_improve_iters = 0
    while time.time() < deadline:
        # compute conv
        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
        k_max = conv.argmax()

        i_start = max(0, k_max-(n-1))
        i_end = min(k_max, n-1)

        contributions = a[i_range] * a[j_range]

        sum_contrib = contributions.sum()

            i = np.random.choice(i_range, p=probs)
        else:
            i = np.random.randint(i_start, i_end+1)

"""
Propose a low‑score step function.

The algorithm:
1. Build a diverse set of candidate windows (spectral, classic windows,
   parametric families, random draws) and keep the best one w.r.t. the
   exact fast_score (2·n·max(self‑convolution) with Σa=1).
2. Perform a short L‑BFGS polishing on a soft‑max‐parametrised version.
3. Run a long, aggressive hill‑climb that always moves mass away from the
   current convolution peak.  The destination is chosen among indices whose
   contribution to the peak is minimal.  When no improvement is seen for a
   while, a small random perturbation is applied to escape local minima.
4. Return the best vector (as a plain list) and evaluate it with the
   official ``evaluate_sequence`` routine.

The hill‑climb occupies most of the 1000 s budget, which is crucial for
pushing the score below the previous best of 1.505999.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisers (probability vectors that sum to 1)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ i^{‑exponent} (i = 1…n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_polish(a_start: np.ndarray,
                 fft_len: int,
                 n: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    One L‑BFGS run with a decreasing smoothing schedule.
    Returns the improved vector and its fast_score.
    """
    # log‑softmax parametrisation
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    # geometrically decreasing λ (the λ‑factor inside the objective)
    lam_schedule = np.geomspace(0.5, 1e-9, num=25)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 2000,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def refined_hill_climb(a_start: np.ndarray,
                       start_score: float,
                       n: int,
                       fft_len: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_no_improve: int = 80000) -> tuple[np.ndarray, float]:
    """
    Intensive greedy hill‑climb.
    - Pick a source index weighted by its contribution to the current peak.
    - Pick a destination among indices that contribute the *least* to the same peak.
    - Transfer a random fraction (up to 50 %) of the source mass.
    - Accept only improving moves; after ``max_no_improve`` unsuccessful moves,
      perform a small random perturbation around the current best.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    no_improve = 0
    it = 0

    # Pre‑allocate arrays that are reused each iteration
    idxs = np.arange(n)

    while time.time() < deadline:
        # --- compute self‑convolution and locate the current peak ----------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # --- source selection (weighted by contribution to the peak) -------
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(i_range, p=probs)
        else:
            # fallback – pick any index inside the overlapping region
            i = rng.integers(i_start, i_end + 1)

        # --- destination selection (indices with minimal partner weight) ----
        # partner[d] = a[k_max - d] if defined else 0
        partner = np.zeros(n, dtype=np.float64)
        mask = (k_max - idxs >= 0) & (k_max - idxs < n)
        partner[mask] = a[k_max - idxs[mask]]
        # take the lower 10 % quantile as the candidate set
        thresh = np.quantile(partner, 0.10)
        low_cand = np.where(partner <= thresh)[0]
        if low_cand.size == 0:
            j = rng.integers(0, n)
        else:
            j = rng.choice(low_cand)

        # ensure a move (j ≠ i)
        if j == i:
            # pick a random other index
            j = (i + rng.integers(1, n)) % n

        # --- transfer a random fraction of source mass --------------------
        max_delta = a[i] * 0.5  # up to 50 % of the source
        if max_delta <= 0:
            # nothing to move – skip
            continue
        delta = max_delta * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # --- evaluate candidate -------------------------------------------
        sc = fast_score(cand, fft_len, n)

        it += 1
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            cur_score = sc
            no_improve = 0
            if it % 5000 == 0:
                print(f"[Hill‑climb] iter {it:,} best fast_score={best_score:.8f}")
        else:
            no_improve += 1
            # optionally keep the move if it does not worsen the current score
            # (greedy hill‑climb – we discard non‑improving moves)
            if sc < cur_score:
                cur_score = sc
                a = cand
            # otherwise reject the move (keep a unchanged)

        # --- random perturbation after a long stagnation ------------------
        if no_improve >= max_no_improve:
            # a small blend with a fresh random vector helps escaping plateaus
            mix = 0.02 * rng.random()
            rand_vec = random_initial(n, rng)
            a = (1.0 - mix) * best_a + mix * rand_vec
            a /= a.sum()
            cur_score = fast_score(a, fft_len, n)
            no_improve = 0
            print(f"[Hill‑climb] stagnation reset at iter {it:,}, cur_score={cur_score:.8f}")

    return best_a, best_score

    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:

            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows with several sigmas
    for sigma in np.linspace(0.12, 0.35, 7):

    best_a = candidates[0].copy()

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------

    phase1_end = start_time + 0.10 * budget_s
    if time.time() < phase1_end:
        best_a, best_score = lbfgs_polish(best_a, fft_len, n, phase1_end)
        print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Heavy hill‑climb for the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = refined_hill_climb(best_a,
                                                best_score,
                                                n,
                                                fft_len,
                                                deadline,
                                                rng,
                                                max_no_improve=80000)
        print(f"[Hill‑climb‑final] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation (the score is invariant under scaling)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
