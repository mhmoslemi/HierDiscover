    y = np.log(w_start + 1e-12)
    best_w = w_start.copy()

        try:

                           x0=y,

                           method='L-BFGS-B',
                           jac=True,

"""
Optimisation of a non‑negative step function (length 5000) to minimise
            2·n·max(a*a)  =  evaluate_sequence(a)

The algorithm builds a modest library of analytic windows (Hann, Blackman,
Kaiser, Beta‑density, etc.) and searches for the best convex combination of
them.  The combination weights live on the probability simplex and are
optimised with a smooth approximation of the L∞‑norm (log‑sum‑exp) using
L‑BFGS and Adam.  The resulting distribution is then polished by a suite of
exact‑max sub‑gradient, simulated‑annealing and fine‑grained hill‑climb
procedures, all respecting the simplex constraint.

The whole search respects a 1000 s time budget; each phase stops early if
the deadline is reached.  The best sequence found is returned and stored in
the global variable ``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact when the vector is normalised to sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families (each returns a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target = 2 * n - 1
    uniform = np.full(target, 1.0 / target)
    padded = np.zeros(L)
    padded[:target] = uniform
    S = np.fft.fft(padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density discretisation on the simplex."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    """(1‑cos)ᵖ window (generalised Hann)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max objective for a single vector (used by hill‑climb phases)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Smoothed max objective for a convex combination of base windows
# ----------------------------------------------------------------------


def objective_and_grad_w(y: np.ndarray, B: np.ndarray, n: int,
                         fft_len_conv: int, fft_len_g: int,
                         lam_factor: float = 0.30):
    """
    Smoothed max objective when a = B @ w,  w = softmax(y).
    Returns (objective, gradient) w.r.t. logits y of the weight vector.
    """
    w = softmax(y)                     # weights on simplex
    a = B @ w                          # candidate distribution (length n)

    # convolution of a with itself
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    dw = exp_shift / Z                   # d smooth_max / d conv_i

    # back‑propagation to a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(dw, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    grad_a = conv_rev[(n - 1):(n - 1 + n)]

    # gradient wrt w (since a = B @ w)
    grad_w = B.T @ grad_a

    # chain rule through softmax
    mean_g = np.dot(w, grad_w)
    grad_y = w * (grad_w - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives for weight vectors (mixture of bases)
# ----------------------------------------------------------------------


def lbfgs_continuation_w(w_start: np.ndarray,
                         B: np.ndarray,
                         lam_schedule,
                         maxiter_per_lam,
                         deadline,
                         n,
                         fft_len):
    """
    L‑BFGS optimisation of the smoothed max for a weight vector.
    Returns best weight vector and its fast_score.
    """
    y = np.log(w_start + 1e-12)          # logits for soft‑max

    best_w = w_start.copy()
    best_score = fast_score(B @ best_w, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad_w,
                           x0=y,
                           args=(B, n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs_w] exception:", e)
            continue
        y = res.x
        w = softmax(y)
        sc = fast_score(B @ w, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_w = w.copy()
    return best_w, best_score


def adam_optimize_w(w_start: np.ndarray,
                    B: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator,
                    max_iters: int = 4000,
                    lam_init: float = 0.30,
                    lam_decay: float = 0.9995,
                    lr: float = 0.07):
    """
    Adam optimisation of the smoothed max for weight vectors.
    Operates on logits y = log(w). Adaptive step size helps escape plateaus.
    """
    y = np.log(w_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_w = softmax(y)
    best_score = fast_score(B @ best_w, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad_w(y, B, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        w = softmax(y)
        sc = fast_score(B @ w, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_w = w.copy()

        it += 1
    return best_w, best_score


# ----------------------------------------------------------------------
# Optimisation primitives for a full vector (the final polishing steps)
# ----------------------------------------------------------------------


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        accept = False
        if sc < best_score:
            accept = True
        else:
            prob = np.exp(-(sc - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        move += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.04,
                    delta_end: float = 0.003):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich library of base windows (all are normalised to sum=1)
    # ------------------------------------------------------------------
    base_windows = []

    # previous solution (if any) – interpolated to current length
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            if s > 0:
                a_prev /= s
            else:
                a_prev = np.full(n, 1.0 / n)
            base_windows.append(a_prev)

    # deterministic families
    base_windows.append(spectral_init(n))
    base_windows.append(triangular_init(n))
    base_windows.append(hann_init(n))
    base_windows.append(blackman_init(n))
    base_windows.append(gaussian_init(n))
    base_windows.append(inv_sqrt_init(n))
    base_windows.append(arcsine_init(n))

    # a few random initial windows (normalised)
    for _ in range(4):
        base_windows.append(random_initial(n, rng))
    for _ in range(5):
        base_windows.append(pow_decay_init(n, rng))

    # Kaiser windows with a range of betas
    for beta in [4, 6, 8, 10, 12, 14]:
        base_windows.append(kaiser_init(n, beta))

    # Raised‑cosine windows with various exponents
    for p in [0.5, 0.75, 1, 1.5, 2, 3]:
        base_windows.append(raised_cosine_init(n, p))

    # Random Beta‑densities
    for _ in range(20):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        base_windows.append(beta_init(n, alpha, beta_param))

    # Ensure every base window is a proper probability vector
    base_windows = [w / w.sum() if w.sum() > 0 else np.full(n, 1.0 / n) for w in base_windows]

    # Stack into a matrix B (n × M)
    B = np.column_stack(base_windows)          # shape (n, M)
    M = B.shape[1]

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – mixture optimisation (L‑BFGS + occasional Adam)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 2000

    best_w = None
    best_score = np.inf

    # a few random restarts
    while time.time() < phase0_end:
        # start from uniform mixture, perturbed a little
        w0 = np.full(M, 1.0 / M)
        perturb = rng.normal(scale=0.08, size=M)
        w0 = np.maximum(w0 + perturb, 0.0)
        w0_sum = w0.sum()
        if w0_sum > 0:
            w0 /= w0_sum
        else:
            w0 = np.full(M, 1.0 / M)

        w_opt, sc_opt = lbfgs_continuation_w(w0, B, lam_schedule,
                                             maxiter_per_lam,
                                             phase0_end, n, fft_len)
        if sc_opt < best_score:
            best_score = sc_opt
            best_w = w_opt.copy()
            print(f"[Mixture‑L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam polishing of the mixture (if time permits)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    if best_w is not None and time.time() < phase1_end:
        best_w, best_score = adam_optimize_w(best_w, B,
                                             phase1_end,
                                             n,
                                             fft_len,
                                             rng,
                                             max_iters=6000,
                                             lam_init=0.30,
                                             lam_decay=0.9993,
                                             lr=0.07)
        print(f"[Mixture‑Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Convert mixture to a concrete candidate vector
    # ------------------------------------------------------------------
    if best_w is None:
        # fallback – use a simple deterministic window
        a = spectral_init(n)
        best_score = fast_score(a, fft_len, n)
    else:
        a = B @ best_w
        best_score = fast_score(a, fft_len, n)
    print(f"[init‑mix] mixture score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max sub‑gradient descent
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        a, sc = subgradient_exact_optimize(a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           step0=0.20)
        if sc < best_score:
            best_score = sc
            print(f"[sub‑grad] improved to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.65 * budget_s
    if time.time() < phase3_end:
        a, sc = simulated_annealing(a,
                                    phase3_end,
                                    n,
                                    fft_len,
                                    rng,
                                    init_temp=5e-4,
                                    final_temp=1e-7,
                                    max_moves=300000)
        if sc < best_score:
            best_score = sc
            print(f"[SA] improved to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine‑grained hill‑climb (the final polishing stage)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        a_new, sc_new = hill_climb_fine(a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=250000,
                                        delta_start=0.04,
                                        delta_end=0.003)
        if sc_new < best_score:
            best_score = sc_new
            a = a_new.copy()
            print(f"[hill‑climb fine] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation (official scoring)
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
