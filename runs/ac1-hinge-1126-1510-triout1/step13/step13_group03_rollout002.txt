"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Create a diverse pool of initial vectors (deterministic windows,
   several random Dirichlet‑like draws, mixtures of Gaussians).  Keep the
   best according to the exact objective 2·n·max(a∗a).

2. Phase A – aggressive L‑BFGS continuation on a log‑sum‑exp smoothed
   max.  The smoothing factor λ is decreased from 0.5 down to 5e‑6.
   Each λ‑stage is run for up to 2000 L‑BFGS iterations; the start
   point for a stage is the result of the previous stage.  Several
   random restarts (mixing the current best with a fresh random vector)
   are performed while time permits.

3. Phase B – exact sub‑gradient descent on the true max‑convolution.
   The step size follows 1/√t, the ε‑mask that selects “almost‑max”
   entries is tightened from 1e‑2 down to 1e‑14 during the run,
   and an exponentiated‑gradient update keeps the iterate on the
   simplex.

4. Phase C – a short L‑BFGS refinement with the smallest λ (5e‑6)
   to polish the solution found by Phase B.

5. Phase D – deterministic greedy hill‑climb that performs tiny
   pairwise mass transfers while strictly improving the objective.

The whole search respects a 1000 s wall‑clock limit; the best vector
found is returned as a plain Python list.  The official
``evaluate_sequence`` rescales the vector internally, so we work
on the probability simplex (sum = 1) throughout.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a∗a) for a on the simplex."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective f = 2·n·smooth_max(conv) and its gradient wrt logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤0
    exp_shift = np.exp(shifted / lam)    # in (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution entries
    w = exp_shift / Z                     # (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) (*) w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root of flat spectrum) – smooth shape."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:

    centre = (n - 1) / 2.0
    sigma = sigma_factor * n

    a /= a.sum()
    return a

    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using several heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'lognorm'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def mixture_gaussian_init(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Gaussians with random parameters."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector and its fast_score found.
    """
    # Initialise logits from the start vector
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f} (λ={lam})")
    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         eps_start: float = 1e-2,
                         eps_end: float = 1e-14):
    """
    Exact sub‑gradient descent on 2·n·max(conv) using exponentiated‑gradient.
    The tolerance ε used to select “active” constraints is annealed from
    eps_start to eps_end.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # progressive epsilon
        t_frac = it / max(1.0, (deadline - time.time()))
        eps = eps_start * (eps_end / eps_start) ** (min(1.0, it / 20000.0))

        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        mask = (conv >= max_c - eps).astype(np.float64)

        # sub‑gradient (2 * rev(a) * mask)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)   # classic 1/√t schedule

        a = a * np.exp(-eta * g)       # exponentiated‑gradient step
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a /= s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[SubGrad] it={it}, best fast_score={best_sc:.9f}")

        it += 1
    return best_a, best_sc


def final_hill_climb(a: np.ndarray,
                     cur_sc: float,
                     fft_len: int,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iter: int = 120000,
                     delta_frac: float = 0.03):
    """
    Strictly improving greedy hill‑climb with tiny pairwise mass moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(mixture_gaussian_init(n, rng, K=rng.integers(2, 5)))
    candidates.append(dirichlet_sparse_init(n, rng))

    # pick the best candidate according to the exact fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Global time accounting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS continuation with aggressive restarts.
    # ------------------------------------------------------------------
    phase_a_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.4, 0.3, 0.2, 0.15, 0.1,
        0.07, 0.05, 0.03, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4,
        1e-4, 5e-5, 2e-5, 1e-5, 5e-6
    ]
    maxiter_per_lam = 2000

    while time.time() < phase_a_end:
        # blend current best with a fresh random draw to diversify restarts
        rand_vec = random_initial(n, rng)
        if rng.random() < 0.2:
            rand_vec = mixture_gaussian_init(n, rng, K=rng.integers(2, 4))
        alpha = rng.random() * 0.33
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase_a_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

        # early exit if we already beat the target
        if best_sc < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (longer run).
    # ------------------------------------------------------------------
    if time.time() < deadline and best_sc >= 1.503:
        subgrad_deadline = time.time() + 0.25 * budget_s
        print("[Subgrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                                subgrad_deadline,
                                                n,
                                                fft_len,
                                                eps_start=1e-2,
                                                eps_end=1e-14)
        print(f"[Subgrad] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase C – Final L‑BFGS refinement with the smallest λ (5e‑6).
    # ------------------------------------------------------------------
    if time.time() < deadline and best_sc >= 1.503:
        print("[L‑BFGS‑final] start")
        final_lam_schedule = [5e-6, 2e-6, 1e-6, 5e-7]
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                           final_lam_schedule,
                                           maxiter_per_lam,
                                           deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑final] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase D – Greedy hill‑climb polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline and best_sc >= 1.503:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 25.0)
        print("[Hill‑climb] start")
        best_a, best_sc = final_hill_climb(best_a,
                                           best_sc,
                                           fft_len,
                                           n,
                                           hill_deadline,
                                           rng,
                                           max_iter=200000,
                                           delta_frac=0.025)
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result – official evaluator will rescale internally.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # sanity check – report the official score for logging purposes
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
