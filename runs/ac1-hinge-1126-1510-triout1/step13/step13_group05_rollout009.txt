"""
A more aggressive multi‑phase optimizer for the non‑negative step‑function problem.

Key ideas
---------
* Try a few different problem sizes (n = 4000, 5000, 6000).  The optimum
  sometimes appears at a size slightly different from the default.
* Build an enlarged palette of deterministic initialisers (spectral,
  triangular, Hann, Blackman, Gaussian, inverse‑sqrt, arcsine,
  exponential‑growth, power‑growth, raised‑cosine, and a dense Beta‑grid).
* Use a focused grid search over the Beta‐distribution parameters (α,β)
  – the best shape from the grid seeds the subsequent continuous optimisation.
* Run a longer L‑BFGS continuation with a finer λ‑schedule, then Adam,
  exponentiated‑gradient, exact sub‑gradient, and a very long fine‑grained
  hill‑climb.  The time budget is split dynamically so that later phases
  receive the remaining slack.
* All updates keep the vector on the probability simplex, therefore the
  official `evaluate_sequence` is invariant to the final scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (same as evaluate_sequence after normalisation)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a∗a) for a probability vector a (Σa=1).
    Fast because it uses an FFT of length `fft_len`.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialiser families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto the simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                                 # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density discretisation on the simplex."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    """(1‑cos)ᵖ window (generalised Hann)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient for L‑BFGS/Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # self‑convolution (full length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a). Adaptive step size helps escape plateaus.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Focuses on the top `top_fraction` of convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # contribution of a[k] where k runs over overlapping indices
        for k in range(j_low, j_high + 1):
            g[k] = 2.0 * a[idx - k]

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.005):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a few problem sizes – flexibility sometimes helps.
    # ------------------------------------------------------------------
    size_options = [4000, 5000, 6000]
    best_overall_score = np.inf
    best_overall_a = None
    best_n = None

    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # small safety margin

    for n in size_options:
        # quick check for remaining time
        if time.time() >= overall_deadline:
            break

        fft_len = next_pow2(2 * n - 1)

        # --------------------------------------------------------------
        # Build a rich pool of deterministic and parametric candidates.
        # --------------------------------------------------------------
        candidates = []

        # deterministic families
        candidates.append(spectral_init(n))
        candidates.append(triangular_init(n))
        candidates.append(hann_init(n))
        candidates.append(blackman_init(n))
        candidates.append(gaussian_init(n))
        candidates.append(inv_sqrt_init(n))
        candidates.append(arcsine_init(n))

        # parameterised families – a modest grid
        for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
            candidates.append(raised_cosine_init(n, p))

        beta_vals = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
        for a in beta_vals:
            for b in beta_vals:
                candidates.append(beta_init(n, a, b))

        # exponential‑growth (right‑heavy) shapes
        for gamma in [0.001, 0.003, 0.005, 0.008, 0.012]:
            i = np.arange(n)
            a = np.exp(gamma * i)
            a /= a.sum()
            candidates.append(a)

        # power‑growth shapes (i^p, p>0)
        for pwr in [0.2, 0.5, 1.0, 2.0, 3.0]:
            i = np.arange(1, n + 1, dtype=float)
            a = i ** pwr
            a /= a.sum()
            candidates.append(a)

        # a few random initialisers
        for _ in range(6):
            candidates.append(random_initial(n, rng))

        # --------------------------------------------------------------
        # Pick the best direct candidate (fast_score) for this n.
        # --------------------------------------------------------------
        best_a_n = None
        best_score_n = np.inf
        for cand in candidates:
            sc = fast_score(cand, fft_len, n)
            if sc < best_score_n:
                best_score_n = sc
                best_a_n = cand.copy()

        print(f"[init n={n}] best fast_score={best_score_n:.8f}")

        # --------------------------------------------------------------
        # Allocate a fraction of the remaining budget to this size.
        # --------------------------------------------------------------
        # Roughly split remaining time equally among still‑to‑process sizes.
        remaining_sizes = len(size_options) - size_options.index(n)
        time_for_this_n = (overall_deadline - time.time()) / remaining_sizes

        # Phase 1 – L‑BFGS polishing (≈30 % of the time for this n)
        phase1_end = time.time() + 0.30 * time_for_this_n
        lam_schedule = [
            0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
            0.05, 0.03, 0.018, 0.01, 0.006,
            0.003, 0.0015, 0.0008, 0.0004, 0.0002,
            0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
            2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
            5e-08, 2e-08, 1e-08
        ]
        maxiter_per_lam = 1500

        while time.time() < phase1_end and time.time() < overall_deadline:
            # start from a random convex blend of the current best and a fresh random vector
            rand_vec = random_initial(n, rng)
            alpha = rng.random() * 0.3          # ≤ 0.3
            a_start = (1.0 - alpha) * best_a_n + alpha * rand_vec
            a_start /= a_start.sum()
            a_opt, sc_opt = lbfgs_continuation(a_start,
                                               lam_schedule,
                                               maxiter_per_lam,
                                               phase1_end,
                                               fft_len, fft_len, n)
            if sc_opt < best_score_n:
                best_score_n = sc_opt
                best_a_n = a_opt.copy()
                print(f"[L‑BFGS n={n}] new best fast_score={best_score_n:.8f}")

        # Phase 2 – Adam optimisation (≈25 % of time for this n)
        phase2_end = time.time() + 0.25 * time_for_this_n
        if time.time() < phase2_end:
            best_a_n, best_score_n = adam_optimize(best_a_n,
                                                   phase2_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   max_iters=12000,
                                                   lam_init=0.30,
                                                   lam_decay=0.9993,
                                                   lr=0.07)
            print(f"[Adam n={n}] best fast_score={best_score_n:.8f}")

        # Phase 3 – exponentiated‑gradient (≈15 % of time for this n)
        phase3_end = time.time() + 0.15 * time_for_this_n
        if time.time() < phase3_end:
            best_a_n, best_score_n = exp_gradient_optimize(best_a_n,
                                                           phase3_end,
                                                           n,
                                                           fft_len,
                                                           rng,
                                                           top_fraction=0.015)
            print(f"[exp‑grad n={n}] best fast_score={best_score_n:.8f}")

        # Phase 4 – exact sub‑gradient (≈10 % of time for this n)
        phase4_end = time.time() + 0.10 * time_for_this_n
        if time.time() < phase4_end:
            best_a_n, best_score_n = subgradient_exact_optimize(best_a_n,
                                                                phase4_end,
                                                                n,
                                                                fft_len,
                                                                step0=0.20)
            print(f"[sub‑grad n={n}] best fast_score={best_score_n:.8f}")

        # Phase 5 – long fine‑grained hill‑climb (use whatever time is left)
        while time.time() < overall_deadline:
            a_new, sc_new = hill_climb_fine(best_a_n,
                                            best_score_n,
                                            fft_len,
                                            n,
                                            overall_deadline,
                                            rng,
                                            max_iter=250000,
                                            delta_start=0.04,
                                            delta_end=0.003)
            if sc_new < best_score_n:
                best_score_n = sc_new
                best_a_n = a_new.copy()
                print(f"[hill‑climb fine n={n}] new best fast_score={best_score_n:.8f}")
            # if no improvement after a climb we break to avoid endless loops
            # (the loop will anyway stop when time expires)
        # --------------------------------------------------------------
        # Record the best across all n‑values.
        # --------------------------------------------------------------
        if best_score_n < best_overall_score:
            best_overall_score = best_score_n
            best_overall_a = best_a_n.copy()
            best_n = n
            print(f"[global] new overall best fast_score={best_overall_score:.8f} (n={best_n})")

    # ------------------------------------------------------------------
    # Final official evaluation and expose for future runs.
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] n={best_n}, evaluated score = {final_eval:.8f}")

    # make the sequence globally available for a possible warm‑start
    globals()['height_sequence_1'] = final_seq
    return final_seq
```

This implementation expands the search space (multiple `n`, exhaustive Beta‑grid and additional parametric families), allocates a longer L‑BFGS budget, and finishes with an extensive fine‑grained hill‑climb.  All updates stay on the probability simplex, so the final `evaluate_sequence` call is guaranteed to be meaningful.
