"""
improved_step_function_search.py

Search for a non‑negative step‑function (a list of heights) that minimises
the objective used by ``evaluate_sequence``.
The algorithm combines several complementary heuristics:

1. **Diverse pool of initialisations** – deterministic windows, spectral
   constructions, and random sparse supports.
2. **L‑BFGS continuation** on a log‑sum‑exp smooth approximation of the
   max‑convolution.  A generous λ‑schedule (down to 1e‑6) and many L‑BFGS
   iterations per λ are used.
3. **Exact‑max gradient descent** – repeatedly identifies the index where
   the convolution attains its maximum and moves a small amount of mass
   from positions that contribute most to this peak to positions that
   contribute least.
4. **Adam optimiser (top‑k)** and **Adam (smooth‑max)** – further
   refinement with adaptive step‑size.
5. **Simulated annealing** and a final **greedy hill‑climb**.

The search respects the 1000 s time budget (configurable) and returns the
best sequence found.  All helper functions are top‑level – no closures
or lambdas are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Exact scoring (fast, no safety checks)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective 2·n·max(conv)/sum(a)² for a (the sum is assumed ≈1).
    Used only for bookkeeping – the official score is obtained via
    ``evaluate_sequence`` after the search finishes.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (smooth objective, gradient) for the soft‑max of the
    self‑convolution.  The variable y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess: inverse FFT of a flat spectrum, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random draw from a variety of patterns, projected onto the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse – pick a random support of size ~sqrt(n)
        m = max(2, int(np.sqrt(n)))
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS continuation on the smooth‑max objective.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    """
    Adam optimizer that gradually focuses on the true maximum (top‑k mask).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # linear decay of the top‑fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # gradient w.r.t. a (ignoring 2·n factor)
        grad_a = 2.0 * n * g_a                         # full objective gradient

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # decreasing step size
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 0.05,
                         lam_end: float = 1e-6):
    """
    Adam optimiser on the log‑sum‑exp smoothed max.
    λ decays exponentially.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor=lam / (a @ a) if (a @ a) > 0 else lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:

            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 250000):
    """
    Simple simulated‑annealing on the simplex with small mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22   # up to ~22% of source mass
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


def exact_max_grad_optimize(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            max_steps: int = 20000,
                            step_frac: float = 0.001):
    """
    Dedicated local optimiser that repeatedly moves a small fraction of
    mass from positions that contribute most to the current peak
    (max entry of the convolution) to positions that contribute least.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    step = 0
    while time.time() < deadline and step < max_steps:
        # compute full self‑convolution (FFT is cheap)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        t_max = int(np.argmax(conv))

        # gradient of the peak w.r.t. a (derived analytically)
        # g[i] = 2 * a[t_max - i]  when the index is inside the support
        idx = t_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # choose source (largest contribution) and destination (smallest)
        # we also require that source still has some mass to give away.
        src_candidates = np.where((g > 0) & (a > 1e-12))[0]
        if src_candidates.size == 0:
            break
        i_src = src_candidates[np.argmax(g[src_candidates])]
        i_dst = np.argmin(g)  # may be the same; handled below
        if i_src == i_dst:
            # fallback: random move
            i_src, i_dst = rng.integers(0, n, size=2)
            if i_src == i_dst:
                step += 1
                continue

        eps = a[i_src] * step_frac
        if eps <= 1e-12:
            step += 1
            continue

        cand = a.copy()
        cand[i_src] -= eps
        cand[i_dst] += eps

        # maintain non‑negativity (numerical safety)
        if cand[i_src] < 0:
            cand[i_dst] += cand[i_src]
            cand[i_src] = 0.0

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand         # accept improvement
        else:
            # occasional random perturbation to escape shallow plateaus
            if step % 1000 == 0:
                a = (0.97 * a) + (0.03 * random_initial(n, rng))
                a /= a.sum()
        step += 1

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a trade‑off between granularity and speed
    # --------------------------------------------------------------
    n = 4096                     # power‑of‑two for fast FFTs
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool (may reuse a previous solution)
    # --------------------------------------------------------------
    candidates = []

    # reuse previous result if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=8.0))

    # random families (including sparse supports)
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # pick the best by cheap fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -----------------------------
    # Phase 1 – L‑BFGS continuation
    # -----------------------------
    phase1_end = start_time + 0.25 * budget_s   # 25 % of time budget
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # perturb current best slightly before each L‑BFGS run
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # -------------------------
    # Phase 2 – Adam (top‑k)
    # -------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.15,
            end_top=0.001
        )
        print(f"[Adam‑topk] best fast_score={best_sc:.9f}")

    # -------------------------
    # Phase 3 – Adam (smooth‑max)
    # -------------------------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.20 * budget_s
        print("[Adam‑smooth] start")
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n,
            fft_len,
            rng,
            lam_start=5e-2,
            lam_end=1e-6
        )
        print(f"[Adam‑smooth] best fast_score={best_sc:.9f}")

    # -------------------------
    # Phase 4 – Exact‑max gradient descent
    # -------------------------
    if time.time() < deadline:
        gm_deadline = time.time() + 0.15 * budget_s
        print("[Exact‑max‑grad] start")
        best_a, best_sc = exact_max_grad_optimize(
            best_a,
            gm_deadline,
            n,
            fft_len,
            rng,
            max_steps=25000,
            step_frac=0.001
        )
        print(f"[Exact‑max‑grad] best fast_score={best_sc:.9f}")

    # ------------------------------------------------
    # Phase 5 – Simulated annealing
    # ------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=300000
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------
    # Phase 6 – final greedy hill‑climb
    # ------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 12.0)
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=150000,
            delta_frac=0.04
        )
        print(f"[hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------
    # Return the result (evaluate_sequence will rescale internally)
    # ------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
