def blend_with_patterns(a, patterns, t_values, fft_len, n):
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    for p in patterns:
        # ensure p normalized

        for t in t_values:

                best = cand.copy()

def random_perturb(a, deadline, fft_len, n, rng, sigma=0.02):
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    while time.time() < deadline:
        i = rng.integers(0, n)
        delta = max(0, best[i] * rng.normal(scale=sigma))
        if delta <= 0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        cand = best.copy()
        cand[i] = max(0, cand[i] - delta)
        cand[j] = cand[j] + delta
        cand = cand / cand.sum()
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
    return best, best_sc

"""
Propose a low‑score step‑function sequence.

The optimiser works in several stages, all respecting the simplex
constraint (non‑negative entries that sum to 1).  The pipeline is:

1.  Build a rich pool of initial windows (deterministic, random, beta,
    raised‑cosine, etc.) and pick the best according to the fast
    exact score 2·n·max(a*a).

2.  L‑BFGS on a smoothed log‑sum‑exp approximation of the max‑conv.
    A long λ‑schedule is used and many random restarts are performed.

3.  Adam optimisation on the same smoothed objective.

4.  Exponentiated‑gradient descent that concentrates on the top
    convolution entries.

5.  Projected sub‑gradient descent on the exact max‑convolution.

6.  Simulated‑annealing with mass‑transfer moves.

7.  Greedy hill‑climb followed by a fine‑grained hill‑climb (the
    latter runs until the time budget expires).

8.  A short “pattern‑blending” phase: the current best vector is mixed
    with a few classic window shapes (uniform, triangular, Hann, …)
    over a grid of mixing coefficients.

9.  A final high‑precision L‑BFGS polish using only the smallest
    smoothing parameters.

The routine stops just before the allotted budget (default 1000 s) and
returns the best list of floats found.  The global name
``height_sequence_1`` is updated so that a subsequent run can start
from the last solution.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (exact for vectors that sum to 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).
    This matches `evaluate_sequence` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS/Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = rfft(a, fft_len_conv)
    conv_full = irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, fft_len_g)
    fw = rfft(w, fft_len_g)
    conv_rev = irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Rough “spectral” guess – inverse square‑root of the FFT magnitude."""
    L = next_pow2(2 * n - 1)
    u = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    u_padded = np.zeros(L)
    u_padded[: 2 * n - 1] = u
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def welch_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - ((i - (n - 1) / 2.0) / ((n - 1) / 2.0)) ** 2
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def sinc_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / (n / 2.0)           # roughly [-1, 1]
    a = np.sinc(5.0 * x)                         # 5 × bandwidth, tweakable
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS optimisation on the smoothed max‑convolution.
    Returns the best vector found during the whole schedule.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimiser on the smoothed objective.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent focusing on the top fraction of convolution entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        fa = rfft(a, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = rfft(a_rev, fft_len)
        fm = rfft(mask, fft_len)
        grad_conv = irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(time.time() - deadline + 1.0)  # decreasing step
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = rfft(a, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # contribution from a[idx - np.arange(j_low, j_high + 1)]
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 150000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Greedy hill‑climb: random mass transfer that is kept only if it strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.005):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Pattern‑blending refinement (stage 8)
# ----------------------------------------------------------------------


def blend_refinement(a: np.ndarray,
                     fft_len: int,
                     n: int,
                     patterns: list[np.ndarray],
                     t_grid: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Try convex combinations of `a` with a set of fixed patterns.
    Returns the best vector found.
    """
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)
    for pat in patterns:
        pat = pat / pat.sum()
        for t in t_grid:
            cand = (1.0 - t) * best + t * pat
            cand = cand / cand.sum()
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best_sc = sc
                best = cand.copy()
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # Use the previous solution if it exists (helps with continuity)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))
    candidates.append(welch_init(n))
    candidates.append(sinc_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(3):
        beta_param = rng.uniform(4.0, 12.0)
        candidates.append(beta_init(n, rng.uniform(0.5, 5.0), beta_param))

    # Systematic exploration of beta / raised‑cosine families
    for _ in range(400):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))

    for _ in range(150):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # --------------------------------------------------------------
    # Pick the best initial candidate
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4          # blend factor ≤ 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=8000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        print(f"[Adam] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent (≈12 % of budget)
    # --------------------------------------------------------------
    phase3_end = start_time + 0.67 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.015)
        print(f"[exp‑grad] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient descent (≈10 % of budget)
    # --------------------------------------------------------------
    phase4_end = start_time + 0.77 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.20)
        print(f"[sub‑grad] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈8 % of budget)
    # --------------------------------------------------------------
    phase5_end = start_time + 0.85 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-7,
                                                 max_moves=200000)
        print(f"[SA] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 6 – Initial greedy hill‑climb (≈5 % of budget)
    # --------------------------------------------------------------
    phase6_end = start_time + 0.90 * budget_s
    if time.time() < phase6_end:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        phase6_end,
                                        rng,
                                        max_iter=120000,
                                        delta_frac=0.06)
        print(f"[hill‑climb 1] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 7 – Fine‑grained hill‑climb for the remaining time
    # --------------------------------------------------------------
    while time.time() < deadline:
        # allocate a short chunk for fine hill‑climb; the function stops early
        # when the internal deadline is hit.
        a_new, sc_new = hill_climb_fine(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=150000,
                                        delta_start=0.04,
                                        delta_end=0.003)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"[hill‑climb fine] new best fast_score = {best_score:.8f}")
        # after each fine‑climb we also try a cheap pattern‑blend pass
        # (the overhead is tiny compared with the FFT‑based hill‑climb).
        blend_patterns = [
            np.full(n, 1.0 / n),                 # uniform
            triangular_init(n),
            hann_init(n),
            blackman_init(n),
            gaussian_init(n),
            welch_init(n),
            sinc_init(n)
        ]
        t_grid = np.linspace(0.0, 0.4, 9)        # 0, 0.05,…, 0.4
        best_a, best_score = blend_refinement(best_a,
                                               fft_len,
                                               n,
                                               blend_patterns,
                                               t_grid)
        # loop again until time expires
        # (the while‑condition will break automatically)

    # --------------------------------------------------------------
    # Phase 8 – Final high‑precision L‑BFGS polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # a short schedule with only the smallest smoothing parameters
        final_lam_schedule = [1e-08, 5e-09, 2e-09, 1e-09, 5e-10]
        a_final, sc_final = lbfgs_continuation(best_a,
                                               final_lam_schedule,
                                               maxiter_per_lam=2000,
                                               deadline=deadline,
                                               fft_len_conv=fft_len,
                                               fft_len_g=fft_len,
                                               n=n)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[final‑L‑BFGS] new best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
