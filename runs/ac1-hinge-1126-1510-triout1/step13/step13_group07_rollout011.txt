def max_subgradient_descent(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    iteration = 0
    while time.time() < deadline:
        # Compute convolution and peak
        fa = np.fft.rfft(a, fft_len)

def simulated_annealing(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    # Determine initial temperature as maybe 0.01 * cur_score? but we can calibrate.
    T = 0.001 * cur_score  # relative.
    iter = 0
    while time.time() < deadline:
        # Choose source i weighted by contributions at peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        p = int(conv.argmax())
        # compute contributions c_i = a[i] * a[p-i] if indices in range
        i_idx = np.arange(n)
        j_idx = p - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        # Normalize contributions to probabilities
        total = contrib.sum()
        if total == 0:
            # fallback random source
            src = rng.integers(0, n)
        else:
            probs = contrib / total
            src = rng.choice(n, p=probs)
        # Choose target randomly (may be any other index)
        tgt = rng.integers(0, n)
        while tgt == src:
            tgt = rng.integers(0, n)
        # Determine max delta
        max_delta = a[src] * 0.5
        delta = max_delta * rng.random()
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        # Ensure nonnegativities and maybe small epsilon.
        # Normalize sum (should remain same).
        # Compute score
        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < cur_score:
            accept = True
        else:
            # compute acceptance probability
            prob = np.exp(-(sc_new - cur_score) / T)
            accept = rng.random() < prob
        if accept:
            a = a_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best = a.copy()
        # Cool down temperature
        T *= 0.9999995  # adjust factor to gradually cool
        iter += 1
    return best, best_score

def beta_init(n, alpha, beta):
    x = (np.arange(n) + 0.5) / n  # midpoints
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

"""
Propose a low‑score step function.

Algorithm overview
-------------------
1.  Build a rich pool of initial vectors:
    * classical windows (hann, blackman, kaiser, gaussian, …)
    * spectral & triangular guesses
    * a grid of Beta‑shaped densities (including the arcsine case)
    * several random draws on the simplex.
2.  Keep the best start according to the exact fast score.
3.  L‑BFGS polishing on the smooth log‑sum‑exp surrogate of the max‑convolution.
    The smoothing factor λ is decreased geometrically; each λ receives many
    L‑BFGS iterations.
4.  Adam optimisation on the same smooth surrogate (larger iteration budget).
5.  Exponentiated‑gradient descent that focuses on the top‑fraction of the
    convolution entries.
6.  Targeted hill‑climb that moves mass away from the indices contributing
    most to the current peak of the autocorrelation.
7.  Simulated‑annealing refinement – moves are biased towards the peak
    contributions (source) and the smallest coefficients (target); uphill
    moves are occasionally accepted to escape local minima.
8.  Final evaluation with the official ``evaluate_sequence`` routine.

The whole routine respects a 1000 s wall‑clock budget; each phase runs for a
pre‑determined fraction of that budget.  All vectors are kept on the probability
simplex (non‑negative, sum = 1) – the evaluation score is invariant under scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast evaluation (Σa = 1 → score = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Returns (smooth objective, gradient w.r.t. logits y).

    The smooth objective approximates 2·n·max(a*a) with a log‑sum‑exp
    with smoothing factor λ = lam_factor·max(conv).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on [0,1] (mid‑point discretisation)."""
    x = (np.arange(n) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len,
    n,
):
    """Sequence of L‑BFGS runs on the smooth surrogate."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    lam_factor: float = 0.001,
    lr: float = 0.08,
    max_iter: int = 20000,
):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    top_fraction: float = 0.04,
):
    """Exponentiated‑gradient descent focusing on the top‑fraction of conv."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_moves: int = 500000,
):
    """
    Greedy hill‑climb that moves mass away from the positions contributing
    most to the current peak of the autocorrelation.
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        if i_start > i_end:
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            if products.sum() == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / products.sum()
                i = rng.choice(i_range, p=probs)

        # target biased toward smallest coefficients (edges / low mass)
        inv_a = 1.0 / (best_a + 1e-12)
        inv_a = inv_a / inv_a.sum()
        j = rng.choice(n, p=inv_a)
        while j == i:
            j = rng.choice(n, p=inv_a)

        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # safeguard against tiny negatives due to rounding
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp_factor: float = 0.001,
    cooling_rate: float = 0.99999,
    max_move_frac: float = 0.5,
):
    """
    Simulated‑annealing refinement.
    Moves mass from a source (chosen proportionally to its contribution to the
    current peak) to a destination (biased towards the smallest coefficients).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    T = init_temp_factor * cur_score
    while time.time() < deadline:
        # current convolution and peak index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        p = int(conv.argmax())

        # contribution of each index to conv[p]
        i_idx = np.arange(n)
        j_idx = p - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        if contrib.sum() > 0.0:
            probs_src = contrib / contrib.sum()
            src = rng.choice(n, p=probs_src)
        else:
            src = rng.integers(0, n)

        # target biased to low‑mass positions
        inv_a = 1.0 / (a + 1e-12)
        inv_a = inv_a / inv_a.sum()
        tgt = rng.choice(n, p=inv_a)
        if tgt == src:
            tgt = (src + rng.integers(1, n)) % n

        max_delta = a[src] * max_move_frac
        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        sc = fast_score(cand, fft_len, n)
        if sc < cur_score:
            accept = True
        else:
            prob = np.exp(-(sc - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        T *= cooling_rate
        if T < 1e-12:
            T = 1e-12
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------

    # Build an enriched pool of starting vectors
    # ------------------------------------------------------------------
    candidates = []

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)

    candidates.append(blackman_init(n))

    # Kaiser windows
    for beta in np.linspace(4.0, 16.0, 13):

    for a in alphas:
        for b in betas:

    best_score = fast_score(best_a, fft_len, n)

        sc = fast_score(cand, fft_len, n)

            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of budget)

    lam_schedule = np.geomspace(0.5, 1e-9, num=45)  # decreasing λ‑factors
    maxiter_per_lam = 3500

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.62 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            top_fraction=0.02,
        )
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted hill‑climb (≈15 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.77 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase4_end,
            rng,
            max_moves=800000,
        )
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, deadline, n, fft_len, rng
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
