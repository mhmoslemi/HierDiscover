"""
search for a step‑function (non‑negative sequence) that minimises
    2·n·max(convolution(a,a)) / (sum(a)**2)

The algorithm works in two resolutions (4096 → 8192 points) and
combines:
  • diverse deterministic & random initialisations,
  • a multi‑stage L‑BFGS continuation on a log‑sum‑exp smoothed max,
  • long Adam runs with a decaying top‑k mask and with a decaying
    smoothing parameter,
  • an aggressive simulated‑annealing phase,
  • a final greedy hill‑climb.

All work respects the global time‑budget (default 1000 s).  The best
sequence found (as a simple Python list) is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (used only for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation 2·n·max(conv(a,a)).  `a` is assumed to sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full linear self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t each convolution entry
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation families

def spectral_init(n: int) -> np.ndarray:
    """Spectral (random‑phase) guess, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    # uniform target spectrum -> sqrt gives a flat autocorrelation
    uniform = np.full(target_len, 1.0 / target_len)
    pad = np.zeros(L)
    pad[:target_len] = uniform
    S = np.fft.fft(pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'dirichlet', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        a = rng.dirichlet(alpha=np.full(n, 0.5))
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """L‑BFGS for every λ in `lam_schedule`. Returns best (a, score)."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    start_top: float = 0.15,
    end_top: float = 0.001,
):
    """Adam with a decaying top‑k mask."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # linear convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # determine current top‑k mask
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient through masked max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1) : (n - 1 + n)]  # ∂objective/∂a (ignoring 2·n)
        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lam_start: float = 5e-2,
    lam_end: float = 1e-8,
):
    """Adam on the log‑sum‑exp smoothed max with exponential λ decay."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        f, grad_y = objective_and_grad(

        )

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:

            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    final_temp: float = 1e-9,
    max_moves: int = 500_000,
):
    """Simple SA on the simplex with small mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(
    a: np.ndarray,
    cur_sc: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 120_000,
    delta_frac: float = 0.04,
):
    """Strictly improving greedy hill‑climb with tiny mass‑transfer moves."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes (small → large)
    # ------------------------------------------------------------------
    n_small = 4096
    n_large = 8192
    fft_len_small = next_pow2(2 * n_small - 1)
    fft_len_large = next_pow2(2 * n_large - 1)

    # ------------------------------------------------------------------
    # Gather diverse initialisers for the small resolution
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous result if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n_small)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n_small, 1.0 / n_small)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n_small))
    candidates.append(triangular_init(n_small))
    candidates.append(hann_init(n_small))
    candidates.append(blackman_init(n_small))
    candidates.append(gaussian_init(n_small))

    # random families
    for _ in range(10):
        candidates.append(random_initial(n_small, rng))
    for beta in np.linspace(4.0, 12.0, 5):
        candidates.append(kaiser_init(n_small, beta))

    # pick the best by cheap exact fast_score
    best_a_small = candidates[0].copy()
    best_sc_small = fast_score(best_a_small, fft_len_small, n_small)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_small, n_small)
        if sc < best_sc_small:
            best_sc_small = sc
            best_a_small = cand.copy()
    print(f"[init‑small] n={n_small}, best fast_score={best_sc_small:.9f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on the small problem
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.1,
        0.05,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # create a slightly perturbed start point
        rand_vec = random_initial(n_small, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a_small + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_small,
            fft_len_small,
            n_small,
        )
        if sc_opt < best_sc_small:
            best_sc_small = sc_opt
            best_a_small = a_opt.copy()
            print(f"[L‑BFGS‑small] new best fast_score={best_sc_small:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam (top‑k) on the small problem
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk‑small] start")
        best_a_small, best_sc_small = adam_topk_optimize(
            best_a_small,
            deadline,
            n_small,
            fft_len_small,
            rng,
            start_top=0.20,
            end_top=0.001,
        )
        print(f"[Adam‑topk‑small] best fast_score={best_sc_small:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam (smooth) on the small problem
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑smooth‑small] start")
        best_a_small, best_sc_small = adam_smooth_optimize(
            best_a_small,
            deadline,
            n_small,
            fft_len_small,
            rng,
            lam_start=5e-2,
            lam_end=1e-8,
        )
        print(f"[Adam‑smooth‑small] best fast_score={best_sc_small:.9f}")

    # ------------------------------------------------------------------
    # Upscale to the larger resolution (if time permits)
    # ------------------------------------------------------------------
    large_allowed = deadline - time.time()
    if large_allowed > 20.0:
        # interpolate the current best onto the larger grid
        x_small = np.linspace(0, 1, n_small)
        x_large = np.linspace(0, 1, n_large)
        a_large = np.interp(x_large, x_small, best_a_small)
        a_large = np.clip(a_large, 0.0, None)
        a_large /= a_large.sum()
        best_a_large = a_large
        best_sc_large = fast_score(best_a_large, fft_len_large, n_large)
        print(f"[upscale] n={n_large}, fast_score={best_sc_large:.9f}")

        # --------------------------------------------------------------
        # Phase 4 – L‑BFGS continuation on the large problem
        # --------------------------------------------------------------
        phase4_end = time.time() + 0.20 * budget_s
        while time.time() < phase4_end:
            rand_vec = random_initial(n_large, rng)
            alpha = rng.random() * 0.30
            a_start = (1.0 - alpha) * best_a_large + alpha * rand_vec
            a_start /= a_start.sum()
            a_opt, sc_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase4_end,
                fft_len_large,
                fft_len_large,
                n_large,
            )
            if sc_opt < best_sc_large:
                best_sc_large = sc_opt
                best_a_large = a_opt.copy()
                print(f"[L‑BFGS‑large] new best fast_score={best_sc_large:.9f}")

        # --------------------------------------------------------------
        # Phase 5 – Adam (top‑k) on the large problem
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("[Adam‑topk‑large] start")
            best_a_large, best_sc_large = adam_topk_optimize(
                best_a_large,
                deadline,
                n_large,
                fft_len_large,
                rng,
                start_top=0.15,
                end_top=0.001,
            )
            print(f"[Adam‑topk‑large] best fast_score={best_sc_large:.9f}")

        # --------------------------------------------------------------
        # Phase 6 – Adam (smooth) on the large problem
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("[Adam‑smooth‑large] start")
            best_a_large, best_sc_large = adam_smooth_optimize(
                best_a_large,
                deadline,
                n_large,
                fft_len_large,
                rng,
                lam_start=5e-2,
                lam_end=1e-8,
            )
            print(f"[Adam‑smooth‑large] best fast_score={best_sc_large:.9f}")

        # --------------------------------------------------------------
        # Phase 7 – Simulated annealing (large)
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
            best_a_large, best_sc_large = simulated_annealing(
                best_a_large,
                sa_deadline,
                n_large,
                fft_len_large,
                rng,
                init_temp=5e-4,
                final_temp=1e-9,
                max_moves=600_000,
            )
            print(f"[SA‑large] best fast_score={best_sc_large:.9f}")

        # --------------------------------------------------------------
        # Phase 8 – Final hill‑climb (large)
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            hill_deadline = time.time() + min(remaining, 12.0)
            best_a_large, best_sc_large = hill_climb(
                best_a_large,
                best_sc_large,
                fft_len_large,
                n_large,
                hill_deadline,
                rng,
                max_iter=150_000,
                delta_frac=0.04,
            )
            print(f"[hill‑climb‑large] best fast_score={best_sc_large:.9f}")

        # choose whatever is best overall
        if best_sc_large < best_sc_small:
            best_a = best_a_large
            best_sc = best_sc_large
            final_n = n_large
        else:
            best_a = best_a_small
            best_sc = best_sc_small
            final_n = n_small
    else:
        # not enough time for the large problem – keep small result
        best_a = best_a_small
        best_sc = best_sc_small
        final_n = n_small

    # --------------------------------------------------------------
    # Final evaluation using the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] n={final_n}, evaluated score = {final_eval:.9f}")
    return final_seq
