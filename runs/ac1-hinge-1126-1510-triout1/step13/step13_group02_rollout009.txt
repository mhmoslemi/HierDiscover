def projected_subgradient_descent(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute conv and max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_c = conv.max()
        # if max_c is within tolerance maybe break?
        # compute mask of area near max
        mask = (conv >= max_c - 1e-12).astype(np.float64)
        # gradient wrt a:
        a_rev = a[::-1]
        grad_rev = np.fft.irfft(np.fft.rfft(a_rev, fft_len) * np.fft.rfft(mask, fft_len), fft_len)
        g = 2.0 * grad_rev[n-1:n-1+n]
        # step size schedule
        eta = 0.5 / np.sqrt(it + 1.0)
        # compute candidate a2 = proj_simplex(a - eta * g)
        a2 = simplex_projection(a - eta * g)
        sc = fast_score(a2, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            # optionally reduce step size slower
        a = a2
        it += 1
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4096
    fft_len = next_pow2(2*n - 1)

    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim ==1 and prev.size >0:
            # Interpolate to current size:
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0,0.0,None)
            a0 /= a0.sum()
            candidates.append(a0)

    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0,12.0)))
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        alpha = rng.uniform(0.2,2.5)
        beta = rng.uniform(0.2,2.5)
        candidates.append(beta_init(n, alpha, beta))

    # Pick best
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_sc:.9f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Phase 1 L-BFGS
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = np.geomspace(0.4, 1e-8, num=40)
    maxiter_per_lam = 2000
    while time.time() < phase1_end:
        # random start near current best
        rand_vec = random_initial(n, rng)
        alpha_mix = rng.random() * 0.5
        a_start = (1 - alpha_mix) * best_a + alpha_mix * rand_vec

        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule,
                                           maxiter_per_lam,

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # Phase 2 projected subgradient descent (exact max)

        sub_end = time.time() + 0.20 * budget_s

        best_a, best_sc = projected_subgradient_descent(best_a, sub_end,

"""
Propose a low‑score step function.

Algorithm overview
-------------------
1. Build a diverse pool of initial candidates (deterministic windows,
   random Dirichlet‑like draws, interpolated previous best, ...).  Keep the
   best by the exact objective 2·n·max(conv).

2. Phase 1 – L‑BFGS on a smoothed (log‑sum‑exp) approximation of the
   max‑convolution.  A geometric λ‑schedule (λ→0) with a relatively large
   iteration budget per λ aggressively reduces the peak.

3. Phase 2 – Projected sub‑gradient descent on the *exact* max‑convolution.
   The sub‑gradient is obtained by convolving the reversed current vector
   with a mask of the (almost) maximal autocorrelation entries; after each
   step we project back onto the probability simplex.

4. Phase 3 – Random convex‑mix refinement.  Randomly generated vectors are
   mixed with the current best and the mixing weight is tuned by a few
   sampled α∈[0,1] to obtain a lower max‑convolution.

5. Phase 4 – Simulated annealing with many pairwise mass‑transfer moves.
   The temperature schedule is long and the move count is large, giving
   fine‑grained exploration of the neighbourhood.

6. Phase 5 – Final greedy hill‑climb (pairwise transfers) with a tiny
   transfer fraction for polishing.

All phases respect a global deadline (≈ 1000 s).  The best sequence found
is finally scored with the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a * a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f   = 2·n·smooth_max(conv)
       grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation helpers (various deterministic / random families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage
    from the previous result. Returns the best vector found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc


def projected_subgradient_descent(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len: int,
                                  rng: np.random.Generator):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses Euclidean projection onto the simplex after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # mask of (almost) maximal entries
        tol = 1e-12
        mask = (conv >= max_c - tol).astype(np.float64)

        # sub‑gradient of the max entry
        a_rev = a[::-1]
        grad_rev = np.fft.irfft(np.fft.rfft(a_rev, fft_len) *
                               np.fft.rfft(mask, fft_len), fft_len)
        g = 2.0 * grad_rev[n - 1:n - 1 + n]

        # step size
        eta = 0.5 / np.sqrt(it + 1.0)

        # projected update
        a_new = simplex_projection(a - eta * g)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_sc


def random_mix_refinement(a: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          mix_trials: int = 120):
    """
    Random convex‑mix refinement.
    For each trial a random vector v is generated and a few random mixing
    coefficients α∈[0,1] are examined; the best mixture replaces the current
    solution if it improves the score.
    """
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for _ in range(mix_trials):
        if time.time() >= deadline:
            break
        v = random_initial(n, rng)
        # test a handful of random alphas
        alphas = rng.random(5)
        for alpha in alphas:
            cand = (1.0 - alpha) * best_a + alpha * v
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = fast_score(cand, fft_len, n)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 250000):
    """
    Pairwise mass‑transfer simulated annealing.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer a random fraction of a[i] (up to 15 % of its current mass)
        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            a = cand
            best_a = cand.copy()
            best_sc = sc_cand
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                a = cand
        move += 1
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 200000,
               delta_frac: float = 0.01):
    """
    Greedy hill‑climb with tiny pairwise mass transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

        if prev.ndim == 1 and prev.size > 0:

            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates.append(a0)

    # Deterministic families

    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Random families

    # Beta‑shaped families
    for _ in range(4):
        alpha = rng.uniform(0.2, 2.5)

        candidates.append(beta_init(n, alpha, beta))

    # ------------------------------------------------------------------
    # Choose the best of the initial pool
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------

                                     delta_frac=0.01)
        print(f"[Hill‑climb] final fast_score={best_sc:.9f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
