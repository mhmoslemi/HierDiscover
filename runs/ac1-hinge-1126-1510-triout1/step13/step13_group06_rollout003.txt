"""
Propose a step‑function (non‑negative vector) that minimizes
    2·n·max(a*a)/(sum(a)**2)

The algorithm builds a rich set of initial candidates (including a
random‑Fourier construction), picks the best, and then refines it with
a multi‑phase pipeline:

1. L‑BFGS on a smoothed “log‑sum‑exp” approximation of the max‑convolution
   (many λ‑values, many inner iterations) – this quickly reduces the
   global objective.
2. Adam optimisation on the same smooth objective (large number of steps,
   exponential λ‑decay) to escape shallow plateaus.
3. Projected sub‑gradient descent with momentum on the exact max‑convolution.
4. A heavy simulated‑annealing stage that moves mass between arbitrary
   positions (both transfers and swaps) with a geometric temperature schedule.
5. A final short L‑BFGS run with very small λ to polish the solution.
6. A greedy hill‑climb that only accepts strictly improving mass‑transfer
   moves.

All vectors are kept on the probability simplex (non‑negative, sum = 1) by
Euclidean projection.  The routine respects the 1000 s time budget,
printing intermediate scores and finally returning the best sequence
found (also stored in the global ``height_sequence_1`` for reuse).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float):
    """
    Returns (objective, gradient) for the smoothed max_{k}(a*a)_k
    where a = softmax(y).  The smoothing parameter is lam = max_c * lam_factor.
    """
    a = softmax(y)

    # full self‑convolution via FFT
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS through a whole λ‑schedule."""
    y = np.log(a_start + 1e-12)  # logits for softmax
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iters: int = 25000,
                  lr: float = 0.07,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9993):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def momentum_subgradient_optimize(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len: int,
                                  step0: float = 0.30,
                                  beta: float = 0.9):
    """Projected sub‑gradient descent with momentum on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)
    vel = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # compute exact convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # remove the mean so that the update respects the simplex (∑a=1)
        d = -(g - np.mean(g))

        # momentum update
        vel = beta * vel + (1.0 - beta) * d
        step = step0 / np.sqrt(it + 1.0)
        a_new = a + step * vel
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 500000):
    """SA with two move types (transfer & swap) and a geometric temperature schedule."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    for move in range(max_moves):
        if time.time() >= deadline:
            break
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        if rng.random() < 0.7:          # transfer move
            delta = a[i] * rng.random() * 0.20
            if delta <= 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
        else:                           # swap move (exchange values)
            a_candidate = a.copy()
            a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]

        sc_candidate = fast_score(a_candidate, fft_len, n)

        # Metropolis acceptance
        if sc_candidate < best_sc or rng.random() < np.exp(-(sc_candidate - best_sc) / max(T, 1e-12)):
            a = a_candidate
            if sc_candidate < best_sc:
                best_sc = sc_candidate
                best_a = a_candidate.copy()
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 400000,
               delta_frac: float = 0.03):
    """Strictly improving hill‑climb (mass‑transfer moves)."""
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:

    return a

def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)

def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_fourier_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Fourier construction: random phases, constant magnitude,
    clipping to non‑negative and re‑normalisation."""
    L = next_pow2(2 * n - 1)
    F = np.zeros(L, dtype=np.complex128)

    # DC component = 1 (so that the sum before clipping is 1)
    F[0] = 1.0

    # Random phases for non‑DC frequencies, Hermitian symmetry to obtain a real signal
    for k in range(1, L // 2):
        amp = rng.exponential(scale=0.5)          # random magnitude
        phase = rng.random() * 2.0 * np.pi
        val = amp * np.exp(1j * phase)
        F[k] = val
        F[-k] = np.conj(val)

    if L % 2 == 0:                               # Nyquist term for even length
        amp = rng.exponential(scale=0.5)
        phase = rng.random() * 2.0 * np.pi
        F[L // 2] = amp * np.exp(1j * phase)

    a_complex = np.fft.ifft(F)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list[float] (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of candidates (including previously discovered one)
    # ------------------------------------------------------------------
    candidates = []

    # try to reuse a previous solution if it exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # random families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(kaiser_init(n, beta=rng.uniform(4.0, 12.0)))

    # systematic Beta / raised‑cosine search (moderate size to stay fast)
    for _ in range(120):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))
    for _ in range(80):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # random Fourier constructions
    for _ in range(20):
        candidates.append(random_fourier_init(n, rng))

    # ------------------------------------------------------------------
    # Pick the best starting candidate (by fast_score)
    # ------------------------------------------------------------------
    best_a = None
    best_sc = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11,
        2e-11, 1e-11, 5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # Blend best with a fresh random vector to diversify starts
        rand_vec = random_initial(n, rng)
        a_start = 0.85 * best_a + 0.15 * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam (≈25 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_sc = adam_optimize(
            best_a, phase2_end, n, fft_len,
            max_iters=25000, lr=0.07, lam_init=0.30, lam_decay=0.9992
        )
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – momentum sub‑gradient on exact max (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        best_a, best_sc = momentum_subgradient_optimize(
            best_a, phase3_end, n, fft_len, step0=0.30, beta=0.9
        )
        print(f"[Momentum‑subgrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = simulated_annealing(
            best_a, deadline, n, fft_len, rng,
            init_temp=5e-4, final_temp=1e-9, max_moves=500000
        )
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – polishing L‑BFGS with very small λ (≈5 % of budget)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.85 * budget_s
    if time.time() < phase5_end:
        fine_lam = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
        best_a, best_sc = lbfgs_continuation(
            best_a, fine_lam, maxiter_per_lam=4000,
            deadline=phase5_end, fft_len=fft_len, n=n
        )
        print(f"[Fine L‑BFGS] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – final greedy hill‑climb (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = hill_climb(
            best_a, best_sc, fft_len, n, deadline, rng,
            max_iter=400000, delta_frac=0.03
        )
        print(f"[final hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq

This implementation follows the same overall structure as the reference code but adds several enhancements:

* a random‑Fourier family of initial vectors,
* a richer λ‑schedule and more L‑BFGS inner iterations,
* a longer Adam run,
* momentum‑based exact sub‑gradient descent,
* an annealing phase that also performs swaps,
* a polishing L‑BFGS pass with extremely small smoothing,
* a slightly more aggressive hill‑climb.

These changes give the optimizer more expressive moves and finer control, increasing the chance of lowering the objective below the previous best of 1.50599 and getting closer to the target 1.503.
