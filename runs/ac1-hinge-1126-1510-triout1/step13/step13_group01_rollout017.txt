def lp_max_sum(f, M):
    n = len(f)
    # Build sparse constraint matrix for convolution f * g <= M
    # For each shift k from 0 to 2n-2.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # g_j appears if 0 <= j <= n-1 and 0 <= k-j <= n-1 => k-n+1 <= j <= k
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        # For j in [j_min, j_max], coefficient = f[j] (since a_i * g_j with i = k - j)
        # Actually conv_f_g[k] = sum_i f_i * g_{k-i} = sum_{i=j}??? Equivalent.
        # For each j, coefficient = f_{k-j} if that index in [0,n-1].
        # Let's compute: for each i from 0..n-1, if 0 <= k-i < n, then add f_i * g_{k-i}.
        # Equivalent: for each j = k-i, index in range. So coefficient for variable g_j is f_{k-j} if k-j within [0,n-1].
        # So we can loop over j.

        # Loop over j from j_min to j_max:
        for j in range(j_min, j_max+1):
            coeff = f[k-j]
            rows.append(k)
            cols.append(j)
            data.append(coeff)

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=float)
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback: return current f
        g = f.copy()
    return g

"""
Optimiser for the “max‑autocorrelation step‑function” challenge.

The algorithm works on a probability vector (sum = 1) of length *n*.
Its objective is   2·n·max(conv(a,a))   which is exactly the value
returned by ``evaluate_sequence`` after the internal √(2n) scaling.

Main ideas
----------
1. Use a relatively large problem size (n = 8192) – the optimal
   constant approaches ≈1.5 from above as n grows.
2. Initialise a rich pool of deterministic and random shapes and keep
   the best one according to a cheap FFT based “fast_score”.
3. Phase 1 – L‑BFGS continuation on a log‑sum‑exp smoothed
   approximation of the max‑convolution.  The λ‑schedule reaches
   1e‑8 and each λ receives many iterations.
4. Phase 2 – Adam optimisation with a decaying top‑k mask that
   gradually focuses on the true maximum.
5. Phase 3 – Adam on the smooth‑max with an exponential λ‑decay to
   1e‑8 for fine‑tuning.
6. Phase 4 – aggressive simulated‑annealing on the simplex with
   tiny mass‑transfer moves.
7. Phase 5 – deterministic hill‑climb that greedily accepts any
   improvement.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain list of floats; ``evaluate_sequence``
will normalise it internally.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (fast) – used for comparisons only
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS/Adam)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv.max()
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. the convolution entries
    w = exp_shift / Z            # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    grad_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_a = grad_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_a)
    grad_y = a * (g_a - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """FFT‑based “white‑noise” guess projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    u = np.random.RandomState(0).normal(size=L)
    A = np.fft.rfft(u)
    # force a non‑negative spectral envelope
    amp = np.sqrt(np.abs(A))
    phase = np.angle(A) / 2.0
    S = amp * np.exp(1j * phase)
    a = np.fft.irfft(S)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    """Deterministic triangular window."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def exponential_init(n: int, rate: float = 0.002) -> np.ndarray:
    """Geometric‑like decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomised draws on the simplex."""
    # pick a distribution family at random
    mode = rng.choice(
        ["uniform", "exp", "power", "edge", "bimodal", "random"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=np.float64))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        weight = np.sqrt(n / 2.0) / n
        a[:m_edge] = weight
        a[-m_edge:] = weight
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        left = max(0, centre - width)
        right = min(n, centre + width)
        a = np.full(n, 1.0 / n)
        a[left:right] = rng.uniform(0.5, 1.5) / n
    else:  # "random"
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the previous result.
    """
    y = np.log(a_start + 1e-12)          # logits → guarantee positivity

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    start_top: float = 0.2,
    end_top: float = 0.001,
):
    """
    Adam optimiser with a top‑k mask that decays from ``start_top``
    to ``end_top``.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution (full)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient w.r.t. a (via convolution with reversed a)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = rev_grad[(n - 1) : (n - 1 + n)]
        grad_a = 2.0 * n * g_a          # full objective gradient

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        cur_a = softmax(y)
        sc = fast_score(cur_a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = cur_a.copy()

    return best_a, best_sc


def adam_smooth_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lam_start: float = 5e-2,
    lam_end: float = 1e-8,
):
    """
    Adam optimiser on the log‑sum‑exp smoothed max,
    λ decays exponentially from ``lam_start`` to ``lam_end``.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        # exponential λ schedule
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        f, grad_y = smooth_obj_and_grad(y, n, fft_len, lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        cur_a = softmax(y)
        sc = fast_score(cur_a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = cur_a.copy()

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    final_temp: float = 1e-9,
    max_moves: int = 400000,
):
    """
    Classical SA on the simplex; each move transfers a small amount
    of mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer up to ~20 % of the source mass
        delta = a[i] * rng.random() * 0.22
        if delta <= 0.0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(
    a_start: np.ndarray,
    cur_sc: float,
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 200000,
    delta_frac: float = 0.04,
):
    """
    Greedy improvement – strictly decreasing moves only.
    """
    best = a_start.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` score.  Returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size ---------------------------------------------------
    n = 8192                     # larger n improves the theoretical bound
    fft_len = next_pow2(2 * n - 1)

    # ----- build a diverse initial pool ---------------------------------
    candidates = []

    # reuse previous solution if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(exponential_init(n, rate=0.0015))

    # random families
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # pick the best according to the cheap fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ----- time budget handling ------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (≈25 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.1,
        0.05,
        0.02,
        0.01,
        5e-3,
        2e-3,
        1e-3,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
        5e-7,
        2e-7,
        1e-7,
        5e-8,
        2e-8,
        1e-8,
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # random perturbation of the current best
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * random_initial(n, rng)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start")
        best_a, best_sc = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.2,
            end_top=0.001,
        )
        print(f"[Adam‑topk] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on smooth‑max (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mid_deadline = time.time() + 0.20 * budget_s
        print("[Adam‑smooth] start")
        best_a, best_sc = adam_smooth_optimize(
            best_a,
            mid_deadline,
            n,
            fft_len,
            rng,
            lam_start=5e-2,
            lam_end=1e-8,
        )
        print(f"[Adam‑smooth] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=500000,
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            n,
            fft_len,
            hill_deadline,
            rng,
            max_iter=300000,
            delta_frac=0.04,
        )
        print(f"[hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation (official score) and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")

    # expose the best sequence for possible reuse in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
