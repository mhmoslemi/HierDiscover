"""
Multi‑phase heuristic for the step‑function attack.

The search consists of:
1.  Building a diverse pool of candidates (deterministic windows,
    random families and the previously best sequence if available).
2.  A Differential‑Evolution (DE) optimisation on the probability simplex
    that directly minimises the fast surrogate 2·n·max(a*a) (score when
    the sum of the coefficients is 1).  This yields a good basin of
    attraction without gradients.
3.  Several L‑BFGS refinements of the best DE individual using a
    smooth approximation of the max‑convolution (log‑sum‑exp) and a
    decreasing smoothing schedule.
4.  A final greedy hill‑climb that strictly improves the fast score.
5.  Evaluation with the official ``evaluate_sequence`` routine.

The whole routine respects the 1000 s time budget, works for any
sequence length (default n = 5000) and returns a list of non‑negative
heights with a lower bound than the previous best (≈1.50599).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast surrogate score (sum of a is 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth approximation of max_i (a*a)_i where a = softmax(y).

    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # self‑convolution by FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (rev‑a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initial families (used for the pool of candidates)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS through the whole λ‑schedule."""
    y = np.log(a_start + 1e-12)      # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(kaiser_init(n, beta=rng.uniform(4.0, 12.0)))

    # systematic beta / raised‑cosine search (moderate size)
    for _ in range(100):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))
    for _ in range(80):
        p = rng.uniform(0.5, 5.0)
        candidates.append(raised_cosine_init(n, p))

    # --------------------------------------------------------------
    # Normalise and evaluate the pool
    # --------------------------------------------------------------
    population = []
    pop_score = []
    for a in candidates:
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        sc = fast_score(a, fft_len, n)
        population.append(a)
        pop_score.append(sc)

    # keep only the best 30 individuals for the DE phase
    sorted_idx = np.argsort(pop_score)[:30]
    population = [population[i] for i in sorted_idx]
    pop_score = [pop_score[i] for i in sorted_idx]
    pop_size = len(population)

    best_idx = int(np.argmin(pop_score))
    best_a = population[best_idx].copy()
    best_score = pop_score[best_idx]
    print(f"[init] pool={len(candidates)} → DE pop={pop_size}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Differential Evolution (≈55 % of budget)
    # ------------------------------------------------------------------
    de_end = start_time + 0.55 * budget_s
    Cr = 0.9
    F = 0.8
    while time.time() < de_end:
        for i in range(pop_size):
            # three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = population[a_idx]
            b_vec = population[b_idx]
            c_vec = population[c_idx]

            mutant = a_vec + F * (b_vec - c_vec)
            mutant = np.clip(mutant, 0, None)
            sm = mutant.sum()
            if sm <= 0:
                mutant = np.full(n, 1.0 / n)
            else:
                mutant /= sm

            target = population[i]
            mask = rng.random(n) < Cr
            trial = np.where(mask, mutant, target)
            trial_sum = trial.sum()
            if trial_sum <= 0:
                trial = np.full(n, 1.0 / n)
            else:
                trial /= trial_sum

            trial_score = fast_score(trial, fft_len, n)
            if trial_score < pop_score[i]:
                population[i] = trial
                pop_score[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
        # occasional tiny random injection to keep diversity
        if rng.random() < 0.03:
            idx = rng.integers(0, pop_size)
            pop = population[idx]
            noise = rng.normal(scale=0.05, size=n)
            pop_noise = np.exp(np.log(pop + 1e-12) + noise)
            pop_noise /= pop_noise.sum()
            population[idx] = pop_noise
            pop_score[idx] = fast_score(pop_noise, fft_len, n)

    print(f"[DE] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS refinement (≈25 % of budget)
    # ------------------------------------------------------------------
    lbfgs_end = start_time + 0.80 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11,
        2e-11, 1e-11, 5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 2000

    while time.time() < lbfgs_end:
        # perturb the current best before each restart
        perturb = np.exp(rng.normal(scale=0.15, size=n))
        y0 = (best_a * perturb)
        y0 /= y0.sum()
        a_opt, sc_opt = lbfgs_continuation(
            y0, lam_schedule, maxiter_per_lam,
            lbfgs_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑restart] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_score = fast_score(best_a, fft_len, n)
        best_a, best_score = hill_climb(
            best_a, cur_score, fft_len, n,
            deadline, rng,
            max_iter=500000, delta_frac=0.08
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation (official scoring)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
