def build_convolution_ub_matrix(a):
    n = len(a)
    rows = 2*n-1
    # create list of diagonals
    data = []
    offsets = []
    for i in range(n):
        diag_len = rows - i
        # diag values: a[i] repeated
        data.append(np.full(diag_len, a[i]))
        offsets.append(i)
    A = diags(data, offsets, shape=(rows, n), format='csr')
    return A

"""
Propose a low‑score step function.

Algorithm Overview
-------------------
1.  Build a rich pool of symmetric initial vectors (spectral, windows,
    parametric families, random draws).  Keep the best according to the
    exact fast_score (2·n·max self‑convolution).

2.  Iteratively improve the best vector by a *linear‑programming blend*:
    - Compute the current peak M of the self‑convolution.
    - Select the indices where the convolution is within a small
      relative tolerance of the peak.
    - Solve a linear program that maximises the total mass of a new
      candidate g under the constraints (a * g)[k] ≤ M for the selected
      indices.
    - Normalise g to a probability vector and replace a ← (1‑t)·a + t·g
      (symmetrised and projected onto the simplex).

    The LP step moves mass away from the positions that currently
    contribute most to the peak.

3.  Refine the result with a few L‑BFGS optimisations on a
    smooth log‑sum‑exp approximation of the max‑convolution.

4.  A short targeted hill‑climb (as in the previous baseline) removes
    any remaining local spikes.

5.  Return the best sequence found; final evaluation is performed with
    the official ``evaluate_sequence`` routine.

The whole search respects the 1000 s budget and is deterministic for a
given seed.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def self_convolution(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    # enforce symmetry
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified symmetric random draw on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal (two symmetric Gaussians)
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        sigma = width / 2.0
        xs = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
        a += np.exp(-0.5 * ((xs - (n - centre - 1)) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    a = (a + a[::-1]) / 2.0
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = opt.minimize(fun=objective_and_grad,
                               x0=y,
                               args=(n, fft_len, fft_len, lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter_per_lam,
                                        'gtol': 1e-12,
                                        'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 600000):
    """
    Greedy hill‑climb that moves mass away from the positions that
    contribute most to the current peak of the self‑convolution.
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        if i_start > i_end:
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            if products.sum() == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / products.sum()
                i = rng.choice(i_range, p=probs)

        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming blend step
# ----------------------------------------------------------------------


def lp_blend_step(a: np.ndarray,
                  M: float,
                  conv: np.ndarray,
                  n: int,
                  rel_tol: float = 0.01,
                  max_constraints: int = 250,
                  rng: np.random.Generator = None):
    """
    Solve a LP that maximises total mass of a new vector g under
    (a * g)[k] ≤ M for the indices k where the current convolution is
    within rel_tol of the peak.
    Returns a normalised candidate g (or None if LP fails).
    """
    # Select constraint indices
    idx_candidates = np.where(conv >= M * (1.0 - rel_tol))[0]
    if idx_candidates.size == 0:
        return None
    # Keep only the most extreme ones (limit matrix size)
    if idx_candidates.size > max_constraints:
        # pick the top values by conv magnitude
        top_idx = np.argpartition(conv[idx_candidates],
                                  -max_constraints)[-max_constraints:]
        idx = idx_candidates[top_idx]
    else:
        idx = idx_candidates

    m = len(idx)
    rows = []
    cols = []
    data = []

    for row, k in enumerate(idx):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        j_vals = np.arange(j_start, j_end + 1, dtype=int)
        i_vals = k - j_vals
        rows.extend([row] * len(j_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise sum(g)

    # bounds: g_j ≥ 0
    bounds = [(0.0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    g_norm = g / s
    # enforce symmetry (helps stability)
    g_norm = (g_norm + g_norm[::-1]) / 2.0
    g_norm /= g_norm.sum()
    return g_norm

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – choose a moderate n (large enough for precision,
    #   small enough for the LP step to stay fast)
    # ------------------------------------------------------------------
    n = 3000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Use a previously exported solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            a0 = (a0 + a0[::-1]) / 2.0
            a0 /= a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows with several sigmas
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows with a grid of betas
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Power‑law decays with a grid of exponents
    for expo in np.linspace(0.15, 0.70, 12):
        candidates.append(power_decay_init(n, exponent=expo))

    # Randomised families (multiple draws)
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming blend cycles (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    blend_t = 0.25          # base blending factor
    rel_tol = 0.01          # indices within 1 % of the peak are constrained
    while time.time() < phase1_end:
        conv = self_convolution(best_a, fft_len, n)
        M = conv.max()
        g = lp_blend_step(best_a, M, conv, n,
                          rel_tol=rel_tol,
                          max_constraints=250,
                          rng=rng)
        if g is None:
            # fallback: a random symmetric perturbation
            rand_vec = random_initial(n, rng)
            g = (best_a + 0.5 * rand_vec)
            g /= g.sum()
        # convex blend
        new_a = (1.0 - blend_t) * best_a + blend_t * g
        new_a = (new_a + new_a[::-1]) / 2.0
        new_a /= new_a.sum()
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[LP‑blend] new best fast_score={best_score:.8f}")
        # gradually reduce the blending factor to fine‑tune
        blend_t = max(0.05, blend_t * 0.95)

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS polishing (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        lam_schedule = np.geomspace(0.4, 1e-9, num=30)
        maxiter_per_lam = 2500
        best_a, best_score = lbfgs_continuation(best_a,
                                                lam_schedule,
                                                maxiter_per_lam,
                                                phase2_end,
                                                fft_len,
                                                n)
        print(f"[L‑BFGS] after polishing fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on the smoothed max (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase3_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.06,
                                           max_iter=30000)
        print(f"[Adam] after refinement fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted hill‑climb with remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(best_a,
                                                best_score,
                                                fft_len,
                                                n,
                                                deadline,
                                                rng,
                                                max_moves=600000)
        print(f"[Hill‑climb] final fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
