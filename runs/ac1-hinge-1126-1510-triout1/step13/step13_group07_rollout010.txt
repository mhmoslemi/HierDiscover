def gradient_hill_climb_fast(a, score, n, fft_len, deadline, rng):
    best_a = a.copy()
    best_score = score
    iter = 0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        idx = int(conv.argmax())
        # gradient
        i_start = max(0, idx-(n-1))
        i_end = min(idx, n-1)
        # compute g via indexing
        g = np.zeros(n)
        i_range = np.arange(i_start, i_end+1)
        g[i_range] = 2 * best_a[idx - i_range]
        # source prob proportional to g
        if g.sum() == 0:
            src_probs = np.full(n, 1.0/n)
        else:
            src_probs = g / g.sum()
        # destination prob inverse
        maxg = g.max()
        dest_weights = maxg - g
        if dest_weights.sum() == 0:
            dest_weights = np.full(n, 1.0/n)
        dest_probs = dest_weights / dest_weights.sum()
        # sample i and j
        i = rng.choice(n, p=src_probs)
        j = rng.choice(n, p=dest_probs)
        # ensure i != j
        if i == j:
            continue
        # step size decays
        step_factor = 0.2 * np.exp(-iter / 50000) + 0.001
        delta = step_factor * best_a[i] * rng.random()
        # limit delta to not make a[i] negative
        delta = min(delta, best_a[i])
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute new score
        sc = fast_score(cand, fft_len, n)
        # accept if better or with small probability
        if sc < best_score:
            best_score = sc
            best_a = cand
        else:
            # occasional uphill acceptance for escape
            # compute temperature
            T0 = 1e-5
            T = T0 * np.exp(-iter / 200000)
            if rng.random() < np.exp( -(sc - best_score) / max(T, 1e-12) ):
                best_a = cand
                best_score = sc
        iter += 1
        # optional symmetrize every 10000 steps
        if iter % 10000 == 0:
            sym = (best_a + best_a[::-1]) / 2
            sym /= sym.sum()
            best_a = sym
            best_score = fast_score(best_a, fft_len, n)
    return best_a, best_score

"""
Improved step‑function search for the evaluation metric.

Algorithm
---------
1. Build a diverse pool of feasible start vectors (spectral, classical windows,
   random draws etc.) and keep the one with the lowest *exact* fast_score.
2. Perform several short L‑BFGS refinements on the smoothed max‑convolution
   objective (log‑sum‑exp).  Each run starts from a random convex blend between
   the current best and a fresh random vector.
3. Exhaust the remaining time with a custom gradient‑guided hill‑climb.  At
   every iteration the index of the current peak of the self‑convolution is
   identified, the exact sub‑gradient of that peak is computed (which is simply
   a shifted copy of the sequence), and a small amount of mass is moved from
   high‑gradient positions to low‑gradient positions.  The step size decays
   geometrically, and occasional random moves help escaping plateaus.
4. Finally enforce symmetry (a ≈ a[::-1]) and evaluate with the official
   ``evaluate_sequence`` routine.

The routine respects the 1000 s wall‑clock limit and returns the best
sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast objective (self‑convolution, Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    """Power‑law decay a_i ∝ i^{‑exponent} (i = 1…n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run a sequence of L‑BFGS optimisations over a decreasing λ schedule.
    Returns the best vector found and its fast_score.
    """
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def gradient_hill_climb(a_start: np.ndarray,
                        cur_score: float,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = None):
    """
    Gradient‑guided hill‑climb.
    Moves mass from positions with large sub‑gradient of the current peak
    (i.e. indices that contribute most to the maximal self‑convolution entry)
    to positions with small sub‑gradient.
    """
    best_a = a_start.copy()
    best_score = cur_score
    a = best_a.copy()
    it = 0

    # parameters for step‑size schedule
    def step_factor(iter_idx):
        # decays from ~0.2 to ~1e‑4
        return max(1e-4, 0.2 * math.exp(-iter_idx / 50000.0))

    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        # full self‑convolution (FFT based, cheap)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        idx = int(conv.argmax())

        # exact sub‑gradient of the peak (derivative of max convex term)
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        g = np.zeros(n, dtype=np.float64)                # ∂max/∂a
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            g[i_range] = 2.0 * a[idx - i_range]          # factor 2 from convolution symmetry

        # source probabilities – proportional to gradient magnitude
        g_sum = g.sum()
        if g_sum == 0.0:
            src_probs = np.full(n, 1.0 / n)
        else:
            src_probs = g / g_sum

        # destination probabilities – favor low‑gradient entries
        maxg = g.max()
        dest_weights = maxg - g
        w_sum = dest_weights.sum()
        if w_sum == 0.0:
            dest_probs = np.full(n, 1.0 / n)
        else:
            dest_probs = dest_weights / w_sum

        # --- main move -------------------------------------------------
        # Occasionally (2 %) perform a completely random redistribution to
        # help escaping shallow plateaux.
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            i = rng.choice(n, p=src_probs)
            j = rng.choice(n, p=dest_probs)
            if i == j:
                continue

        sf = step_factor(it)
        delta = sf * a[i] * rng.random()
        delta = min(delta, a[i])          # keep non‑negative

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand
        else:
            # very occasional uphill acceptance (simulated‑annealing style)
            T = 1e-5 * math.exp(-it / 200000.0)
            if rng.random() < math.exp(-(sc - best_score) / max(T, 1e-12)):
                a = cand

        # occasional symmetrisation (every 20 k iterations)
        if (it + 1) % 20000 == 0:
            sym = (a + a[::-1]) * 0.5
            sym /= sym.sum()
            a = sym
            sc_sym = fast_score(a, fft_len, n)
            if sc_sym < best_score:
                best_score = sc_sym
                best_a = a.copy()

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – moderate length gives a good trade‑off
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the current size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows with several sigmas
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows with a grid of betas
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Power‑law decays with a grid of exponents
    for expo in np.linspace(0.15, 0.70, 12):
        candidates.append(power_decay_init(n, exponent=expo))

    # Randomised families (multiple draws)
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Pick the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-12, num=50)   # decreasing λ‑fractions
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30                # ≤ 0.30 to keep a close to current best
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Gradient‑guided hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = gradient_hill_climb(best_a,
                                                best_score,
                                                n,
                                                fft_len,
                                                deadline,
                                                rng,
                                                max_iters=None)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Optional final symmetrisation (helps flatten the convolution)
    # ------------------------------------------------------------------
    sym = (best_a + best_a[::-1]) * 0.5
    sym /= sym.sum()
    sym_score = fast_score(sym, fft_len, n)
    if sym_score < best_score:
        best_a = sym
        best_score = sym_score
        print(f"[Symmetrise] improved fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
