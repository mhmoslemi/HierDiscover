"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(a*a) / (sum(a)**2)

The score is homogeneous, so we work with probability vectors
(sum(a)=1) and minimise 2·n·max(a*a).  The algorithm uses several
phases:

1.   Build a large pool of sensible initialisations (uniform, cosine,
     Gaussian, beta, etc.) and keep the best.
2.   A short L‑BFGS refinement on a smooth log‑sum‑exp approximation of
     the max‑convolution, with a fine λ‑schedule.
3.   Several random restarts of the L‑BFGS phase, each starting from a
     perturbed version of the current best.
4.   A long simulated‑annealing phase that moves small amounts of mass
     between random coordinates.
5.   A final adaptive hill‑climb with a decreasing step‑size, allowing
     very fine adjustments.

All steps respect the constraints of ``evaluate_sequence`` (non‑negative,
finite numbers) and keep the vector on the probability simplex.
The best vector found is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) for the smooth approximation of
    max_i (a * a)_i where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:

    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:

    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS through the whole λ‑schedule."""
    y = np.log(a_start + 1e-12)      # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iters: int = 20000,
                  lr: float = 0.07,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9993):
    """Adam on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 800000):
    """Simple SA with many mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for move in range(max_moves):
        if time.time() >= deadline:
            break
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score or rng.random() < np.exp(-(sc_candidate - best_score) / max(T, 1e-12)):
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
    return best_a, best_score

                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000000,
                        delta_start: float = 0.20,
                        delta_end: float = 0.001):
    """
    Greedy hill‑climb with a linearly decaying move‑fraction.
    Only strictly improving moves are kept.
    """
    best = a_start.copy()
    best_score = cur_score
    start = time.time()
    total = deadline - start

    it = 0
    while time.time() < deadline and it < max_iter:
        # linearly decay delta fraction
        elapsed = time.time() - start
        t = min(1.0, elapsed / total)
        delta_frac = delta_start * (1.0 - t) + delta_end * t

        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list[float] (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse previous best if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # random families
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(kaiser_init(n, beta=rng.uniform(4.0, 12.0)))

    # systematic search over beta / raised‑cosine families
    for _ in range(200):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))
    for _ in range(120):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # --------------------------------------------------------------
    # Pick the best starting candidate
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – focused L‑BFGS (≈20 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11,
        5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 1800
    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        a_start = 0.85 * best_a + 0.15 * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.35 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, n, fft_len,
            max_iters=20000, lr=0.07, lam_init=0.30, lam_decay=0.9993
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – random L‑BFGS restarts (≈15 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.50 * budget_s
    y0 = np.log(best_a + 1e-12)
    while time.time() < phase3_end:
        # perturb logits slightly
        sigma = 0.5 * np.exp(- (time.time() - start_time) / budget_s)
        y_pert = y0 + sigma * rng.normal(size=n)
        a_start = softmax(y_pert)
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule[-12:], 1200,
            phase3_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            y0 = np.log(best_a + 1e-12)   # reset central point
            print(f"[restart L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – heavy Simulated Annealing (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, deadline, n, fft_len, rng,
            init_temp=1e-3, final_temp=1e-10, max_moves=800000
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – final adaptive Hill‑Climb (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_adaptive(
            best_a, best_score, fft_len, n, deadline, rng,
            max_iter=2000000, delta_start=0.20, delta_end=0.001
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation (official scoring)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
```

This implementation expands on the previous multi‑phase approach: it adds a larger pool of initial families, performs a dedicated L‑BFGS refinement, carries out several random restarts of the L‑BFGS stage, runs a longer simulated‑annealing pass (up to 800 k moves), and finishes with an adaptive hill‑climb whose step‑size shrinks from 0.20 to 0.001. All phases respect the time budget (≈1000 s) and the constraints of `evaluate_sequence`.
