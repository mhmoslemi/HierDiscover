"""
Optimisation of a non‑negative step function for the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of:

1.  A rich pool of initial shapes (window functions, random Dirichlet draws,
    beta‑densities, raised‑cosine families, etc.).  The best of these is
    selected using the fast FFT based score.

2.  Many L‑BFGS refinements on a smoothed version of the max‑convolution.
    The smoothing factor (λ) is driven to the limit with a fine schedule.
    Random perturbations of the current best are used as fresh starts.

3.  A final Polyak‑step sub‑gradient descent on the exact max‑convolution,
    which is the natural non‑smooth objective.  The step size is chosen
    adaptively from the observed improvement, providing fine‑grained
    descent that often beats the previous best score.

All vectors are kept on the probability simplex, so the final call to
`evaluate_sequence` (which rescales internally) is invariant to the
normalisation.  The routine returns the best list found before the
deadline (default 1000 s).  The global name ``height_sequence_1`` is
updated so that later runs can continue from the best solution.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact for probability vectors)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum = 1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Parameters
    ----------
    y : ndarray
        Log‑its; the actual probability vector is ``softmax(y)``.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        Length of the FFT used for the convolution and for the back‑propagation
        through the linear map.
    lam_factor : float
        Multiplicative factor that determines the smoothing parameter λ:
        λ = max(conv) * lam_factor.
    Returns
    -------
    f : float
        Smoothed objective value (2·n·smooth_max).
    grad_y : ndarray
        Gradient w.r.t. the logits ``y``.
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers – a variety of deterministic / random shapes
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def logistic_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Logistic (sigmoid) shaping – a smooth step."""
    centre = n // 2
    scale = rng.uniform(0.05 * n, 0.5 * n)
    x = np.arange(n) - centre
    a = 1.0 / (1.0 + np.exp(x / scale))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule`` starting from ``a_start``.
    Returns the best probability vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_polyak(a_start: np.ndarray,
                                deadline: float,
                                n: int,
                                fft_len: int,
                                best_known_score: float = None):
    """
    Projected sub‑gradient descent with a Polyak step size.
    Works directly on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    # current best max‑convolution value (not the scaled score)
    conv_max = fast_score(a, fft_len, n) / (2.0 * n)
    best_max = conv_max if best_known_score is None else best_known_score / (2.0 * n)

    while time.time() < deadline:
        # exact convolution via FFT (fast and sufficiently accurate)
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv_full.argmax())
        cur_max = conv_full[idx]

        # gradient of the max entry: g_i = 2 * a_{idx-i}  when i contributes
        lower = max(0, idx - (n - 1))
        upper = min(idx, n - 1)
        g = np.zeros_like(a)
        js = np.arange(lower, upper + 1)
        g[js] = 2.0 * a[idx - js]

        g_norm_sq = np.dot(g, g)
        if g_norm_sq == 0:
            break

        # Polyak step: α = (cur_max - best_max) / ||g||²  (scaled by 0.9 for safety)
        step = (cur_max - best_max) / g_norm_sq
        if step <= 0:
            # fall back to a tiny step to keep moving
            step = 1e-8
        else:
            step *= 0.9

        a_new = a - step * g
        a_new = simplex_projection(a_new)

        # evaluate new max
        fa_new = np.fft.rfft(a_new, fft_len)
        conv_new = np.fft.irfft(fa_new * fa_new, fft_len)[: 2 * n - 1]
        new_max = conv_new.max()

        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()

        a = a_new

    return best_a, 2.0 * n * best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ----------------  problem size  -----------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ----------------  build candidate pool  -----------------
    candidates = []

    # reuse previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # re‑interpolate onto current size
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # randomised families
    for _ in range(12):
        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(beta_init(n,
                                    alpha=rng.exponential(scale=0.8) + 0.1,
                                    beta=rng.exponential(scale=0.8) + 0.1))
    for _ in range(4):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))
    for _ in range(6):
        candidates.append(logistic_init(n, rng))

    # ----------------  pick the best initial candidate  -----------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # ----------------  time handling  -----------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ----------------  Phase 1 – intensive L‑BFGS refinements  -----------------
    phase1_end = start_time + 0.55 * budget_s
    # extended smoothing schedule – pushes λ to a very tight smoothing
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08,
        5e-09, 2e-09, 1e-09, 5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # a small random perturbation of the current best
        perturb = rng.normal(scale=0.02, size=n)
        a_start = best_a + perturb
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ----------------  Phase 2 – final Polyak sub‑gradient descent  -----------------
    phase2_end = deadline
    if time.time() < phase2_end:
        a_opt, sc_opt = subgradient_optimize_polyak(best_a,
                                                    phase2_end,
                                                    n,
                                                    fft_len,
                                                    best_known_score=best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Polyak‑SGD] new best fast_score={best_score:.8f}")

    # ----------------  final evaluation (official) -----------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # store for possible next runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
