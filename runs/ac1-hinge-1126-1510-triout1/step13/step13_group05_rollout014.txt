"""
Low‑score step function search.

The optimizer starts from the uniform distribution (the provable lower bound
for the unconstrained problem is a fast‑score of 2).  It then repeatedly
solves a small linear programme that maximises the total mass of a new
sequence g while keeping the self‑convolution f∗g below the current worst
convolution value M = max(f∗f).  After each LP step the new sequence is
mixed into the current one (convex combination) and the process is repeated.
A final fine‑grained hill‑climb cleans up any remaining slack.

All optimisation respects the probability simplex (non‑negative, sum fixed)
so the final objective value returned by ``evaluate_sequence`` is invariant
to any subsequent scaling performed by the judge.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point uniformly from the probability simplex."""
    x = rng.random(n)
    x /= x.sum()
    return x


# ----------------------------------------------------------------------
# Fast evaluation (exact for a probability vector)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (Σa = 1).
    This equals the value returned by ``evaluate_sequence`` after
    normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return conv


# ----------------------------------------------------------------------
# Linear‑programme step (maximise Σg  subject to  f∗g ≤ M)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float, active_tol: float, eps: float = 1e-12):
    """
    Solve the LP
        maximise   Σ g_j
        subject to  (f∗g)[i] ≤ M  for all i with (f∗f)[i] ≈ M
                    g_j ≥ 0.
    Only the “active’’ convolution indices (within ``active_tol`` of M)
    are kept – this makes the problem tiny (a few dozen constraints).

    Returns the optimal g (numpy array) or ``None`` if the solver fails.
    """
    n = f.shape[0]
    # Find the active constraints
    conv_f = full_conv(f, next_pow2(2 * n - 1))
    active = np.where(conv_f >= M - active_tol)[0]
    if active.size == 0:
        return None

    # Build the (dense) constraint matrix A g ≤ b  (only active rows)
    A = np.zeros((active.size, n), dtype=np.float64)
    for row, i in enumerate(active):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # (f∗g)[i] = Σ_k f[i‑k]·g_k  → coefficients are f[i‑k]
        A[row, idx] = f[i - idx]

    b = np.full(active.size, M + eps)

    # maximise Σ g  ⇔ minimise −Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use HiGHS – fast for large‑variable, few‑constraint problems
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Fine‑grained hill‑climbing (same as in the reference code)
# ----------------------------------------------------------------------


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.003):
    """Linearly decaying greedy mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 42):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000                                   # length of the step function
    fft_len = next_pow2(2 * n - 1)             # FFT length for convolutions

    # ------------------------------------------------------------------
    # Start from the uniform distribution (the provable optimum for the
    # unconstrained problem) – it already yields a score of exactly 2.
    # ------------------------------------------------------------------
    f = np.full(n, 1.0 / n, dtype=np.float64)  # Σf = 1
    best_f = f.copy()
    best_score = fast_score(best_f, fft_len, n)
    print(f"[init] uniform fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Alternating LP + convex‑mixing phase
    # ------------------------------------------------------------------
    # Parameters (tuned empirically)
    mix_factor = 0.25            # how much of the new g to blend in
    active_tol = 1e-6            # tolerance for selecting active constraints
    lp_iter = 0
    while time.time() < deadline:
        # 1) current worst convolution value
        conv_f = full_conv(f, fft_len)
        M = conv_f.max()

        # 2) linear‑programming step
        g0 = lp_step(f, M, active_tol)
        if g0 is None:
            # LP failed – fall back to a tiny random perturbation
            perturb = rng.normal(scale=1e-4, size=n)
            f = f + perturb
            f = np.clip(f, 0.0, None)
            f = simplex_projection(f)
            continue

        # 3) rescale g0 to the same total mass as f (the evaluation does not
        #    depend on the absolute scale, only on the shape)
        total_f = f.sum()
        if g0.sum() == 0:
            # degenerate LP – skip this iteration
            continue
        g = g0 * (total_f / g0.sum())

        # 4) convex combination  f ← (1‑t)·f + t·g
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = simplex_projection(f)          # keep Σf = 1 exactly

        # 5) keep the best solution seen so far
        cur_score = fast_score(f, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP‑iter {lp_iter:03d}] new best fast_score = {best_score:.8f}")
        lp_iter += 1

    # ------------------------------------------------------------------
    # Final fine‑grained hill‑climbing
    # ------------------------------------------------------------------
    cur_score = fast_score(best_f, fft_len, n)
    best_f, best_score = hill_climb_fine(best_f,
                                         cur_score,
                                         fft_len,
                                         n,
                                         deadline,
                                         rng,
                                         max_iter=250000,
                                         delta_start=0.04,
                                         delta_end=0.003)

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence for possible future runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
