"""
Propose a low‑score step function.

Algorithm (≈ 1000 s budget)
---------------------------------
1. Build a rich pool of initial vectors (classical windows, spectral,
   beta‑power families and random draws).  Keep the best according to the
   exact fast_score.
2. Repeated L‑BFGS polishing: blend the current best with a fresh random
   vector, then run a sequence of L‑BFGS optimisations on the *smoothed*
   max‑convolution.  The λ‑schedule is dense (50 values) and the optimiser
   is allowed many iterations – this is the heavy lifting.
3. Fine‑tune with several Adam runs at decreasing smoothing factors
   (lam_factor = 0.005 → 0.001 → 0.0005 → 0.0002 → 0.0001).  Each run respects
   the remaining time budget.
4. Targeted hill‑climb: repeatedly move a modest fraction of mass from the
   positions that contribute most to the current peak of the convolution
   to positions where the convolution is currently low.  The move‑size is
   chosen adaptively and the best improvement is kept.
5. Final official evaluation with ``evaluate_sequence`` and return the
   best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    """Power‑law decay a_i ∝ i^{‑exponent} (i = 1…n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_power_init(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (x_i(1‑x_i))^{‑beta}  with x_i = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x * (1.0 - x)) ** (-beta)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_sym_init(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta distribution a_i ∝ x^{α‑1}(1‑x)^{α‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    x = i / (n + 1.0)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 500000):
    """
    Greedy hill‑climb that moves mass away from the positions contributing
    most to the current peak of the self‑convolution, directing the mass
    towards indices where the convolution is low.
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        # current convolution and its peak
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # indices that actually contribute to conv[idx]
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        if i_start > i_end:
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            if products.sum() == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / products.sum()
                i = rng.choice(i_range, p=probs)

        # pick destination with low convolution values
        # we use the bottom 15 % of the conv array as a pool
        low_thresh = np.percentile(conv, 15)
        low_candidates = np.where(conv <= low_thresh)[0]
        # map convolution indices back to a‑indices (choose a centre of the zone)
        if low_candidates.size == 0:
            j = rng.integers(0, n)
        else:
            # pick a random convolution position and back‑project to an a‑index
            conv_idx = rng.choice(low_candidates)
            # pick a source index that participates in that conv position
            # (i.e. find some k such that 0 ≤ k < n and 0 ≤ conv_idx‑k < n)
            # we simply take the centre of the feasible interval.
            k_min = max(0, conv_idx - (n - 1))
            k_max = min(conv_idx, n - 1)
            j = rng.integers(k_min, k_max + 1)

        if j == i:
            continue

        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep n moderately large (5000 is a good trade‑off)
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # Re‑use a previously exported solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows (various sigmas)
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows (grid of betas)
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Power‑law decays (grid of exponents)
    for expo in np.linspace(0.15, 0.70, 12):
        candidates.append(power_decay_init(n, exponent=expo))

    # Beta‑power families – a useful parametric shape
    for beta in np.linspace(0.20, 0.60, 9):
        candidates.append(beta_power_init(n, beta=beta))

    # Symmetric beta distributions (α‑parameter)
    for alpha in np.linspace(0.3, 2.0, 10):
        candidates.append(beta_sym_init(n, alpha=alpha))

    # Randomised families (multiple draws)
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # Choose the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of the budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-10, num=50)   # dense decreasing λ‑schedule
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30               # ≤ 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam with decreasing smoothing (≈30 % of the budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    lam_factors = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for lam in lam_factors:
        if time.time() >= phase2_end:
            break
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=lam,
                                           lr=0.06,
                                           max_iter=40000)
        print(f"[Adam λ={lam:.5f}] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Targeted hill‑climb (use the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining_moves = int(1e6)  # generous move budget
        best_a, best_score = targeted_hill_climb(best_a,
                                                best_score,
                                                fft_len,
                                                n,
                                                deadline,
                                                rng,
                                                max_moves=remaining_moves)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
