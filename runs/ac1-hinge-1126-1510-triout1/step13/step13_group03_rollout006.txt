"""
Multi‑phase optimizer for the “max‑convolution” objective.

The procedure works on the probability simplex (sum = 1) and uses
the following stages (all respecting a global 1000 s budget):

1. Initialise a diverse pool (deterministic windows, random Dirichlet‑like,
   Gaussian mixtures, etc.) and keep the best by the exact objective
   2·n·max(conv).

2. Global L‑BFGS with a continuation on a smoothed max (log‑sum‑exp).
   Several restarts are performed, each start being a convex combination
   of the current best and a fresh random candidate.  The smoothing
   factor (lam_factor) is decreased on a schedule reaching 1e‑5.

3. Exact sub‑gradient descent on the true max‑convolution (projected
   onto the simplex with an exponentiated‑gradient update).

4. Adam optimisation on the exact max (gradient w.r.t. logits, i.e.
   softmax‑parametrised).

5. Simulated‑annealing with pairwise mass‑transfer moves for fine‑tuning.

6. Deterministic greedy hill‑climb (tiny pairwise transfers) for the
   final polishing.

All phases stop automatically when the overall deadline is reached.
The returned list is fed to the official ``evaluate_sequence`` which
rescales it internally, so the optimisation works on the simplex.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact (fast) objective for vectors on the simplex (sum == 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a on the simplex (Σa=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective f = 2·n·smooth_max(conv) and gradient w.r.t. logits y.
    y → a = softmax(y) lives on the simplex.
    lam_factor is the smoothing schedule entry (relative to max(conv)).
    """
    a = softmax(y)

    # convolution a*a (zero‑padded)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter λ = max_c * lam_factor
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp (stable, shifted by max_c)
    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)    # in (0, 1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z                     # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n) – take the central window
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root of flat spectrum)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Randomised families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using several heuristics."""
    mode = rng.choice(
        ["uniform", "exp", "power", "edge", "bimodal", "dirichlet", "lognorm"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "dirichlet":
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def mixture_gaussian_init(
    n: int, rng: np.random.Generator, K: int = 3
) -> np.ndarray:
    """Mixture of K Gaussians with random parameters."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def dirichlet_sparse_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw (small α)."""
    alpha = rng.uniform(0.01, 0.2, size=n)
    a = rng.gamma(alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule (starting from a_start).
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc


def subgradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    eps: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on 2·n·max(conv) using an
    exponentiated‑gradient update on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        mask = (conv >= max_c - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * rev_grad[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a /= s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[Subgrad] it={it}, best fast_score={best_sc:.9f}")

        it += 1

    return best_a, best_sc


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y (a = softmax(y)) so that Σa=1 holds automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1) : (n - 1 + n)]
        grad_a = 2.0 * n * g_a

        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.9f}")

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    final_temp: float = 1e-7,
    max_moves: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Simple pairwise mass‑transfer annealing.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.16
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(
    a: np.ndarray,
    cur_sc: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 100000,
    delta_frac: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – use a larger support if we have enough time
    # ------------------------------------------------------------------
    # 4096 works well; switch to 8192 when we have a very generous budget.
    n = 8192 if budget_s > 800 else 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # reuse previous solution if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(mixture_gaussian_init(n, rng, K=rng.integers(2, 5)))
    candidates.append(dirichlet_sparse_init(n, rng))

    # pick the best by the exact fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Global time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.4  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with continuation (aggressive restarts)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.38 * budget_s
    lam_schedule = [
        0.5,
        0.3,
        0.15,
        0.07,
        0.03,
        0.015,
        0.007,
        0.003,
        0.0015,
        5e-4,
        2e-4,
        1e-4,
        5e-5,
        2e-5,
        1e-5,
        5e-6,
        2e-6,
        1e-6,
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_deadline:
        # blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        if rng.random() < 0.2:
            rand_vec = mixture_gaussian_init(
                n, rng, K=rng.integers(2, 4)
            )
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.20 * budget_s
        print("[Subgrad] start")
        best_a, best_sc = subgradient_optimize(
            best_a, sub_deadline, n, fft_len, eps=1e-14
        )
        print(f"[Subgrad] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + 0.20 * budget_s
        print("[Adam] start")
        best_a, best_sc = adam_optimize(
            best_a, adam_deadline, n, fft_len, rng
        )
        print(f"[Adam] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(30.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=1e-3,
            final_temp=1e-7,
            max_moves=250000,
        )
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – Greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + min(30.0, deadline - time.time())
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=200000,
            delta_frac=0.02,
        )
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result – the official evaluator rescales internally.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
