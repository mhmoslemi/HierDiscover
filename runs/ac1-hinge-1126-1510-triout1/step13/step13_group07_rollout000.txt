def beta_init(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, alpha-1) * np.power(1 - x, beta-1)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    return a

"""
Propose a low‑score step function.

The algorithm consists of three stages

1.  **Broad parametric sweep** – generate many candidates from several
    analytic families (Beta, exponential, power‑law, classic windows,
    random draws).  Each candidate is evaluated with an exact fast score
    (2·n·max(conv)) and the best few are kept.

2.  **Continuous refinement** – each of the top candidates is polished by
    a sequence of L‑BFGS optimisations on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  The schedule of smoothing parameters
    is geometrically decreasing, allowing us to converge to the true
    max‑objective.

3.  **Combinatorial fine‑tuning** – a hill‑climbing / simulated‑annealing
    phase moves small packets of mass away from the positions that
    contribute most to the current peak of the self‑convolution.  This
    further reduces the peak while keeping the vector on the probability
    simplex.

All phases respect a total wall‑clock budget of ~1000 s.  The final
sequence is returned as a plain Python list; `evaluate_sequence`
will be called externally to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast exact objective (probability vector → 2·n·max(conv))
# ------------------------------------------------------------

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ------------------------------------------------------------
# Initialisation families
# ------------------------------------------------------------

def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on a uniform grid, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # use logs to avoid overflow for extreme parameters
    log_a = (alpha - 1) * np.log(x + 1e-16) + (beta - 1) * np.log(1 - x + 1e-16)
    a = np.exp(log_a)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Arcsine distribution (Beta(0.5,0.5))."""
    return beta_init(n, 0.5, 0.5)


def exponential_init(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a = a / a.sum()
    return a


def power_decay_init(n: int, exponent: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^{-exponent}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    a = a / a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False, dtype=np.float64)
    dec = np.linspace(1, 0, n - half, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

# Optimisation primitives
# ------------------------------------------------------------

def lbfgs_refine(a_start: np.ndarray,
                 deadline: float,
                 fft_len: int,
                 n: int) -> tuple[np.ndarray, float]:
    """
    Run a series of L‑BFGS optimisations on the smoothed max objective.
    Returns (best_a, best_score).
    """
    lam_schedule = np.geomspace(0.5, 1e-9, num=60)   # decreasing smoothing factor
    maxiter_per_lam = 200
    y = np.log(np.maximum(a_start, 1e-12))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a_start: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_moves: int = 800000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass away from the positions that
    contribute most to the current peak of the self‑convolution.
    """
    best_a = a_start.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # overlapping index range that contributes to conv[idx]
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)

        if i_start > i_end:
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            total = products.sum()
            if total == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / total
                i = rng.choice(i_range, p=probs)

        # choose a different destination index
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidates = []

    # 1.  Existing solution (if any) – resample to the current length
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:

            x_new = np.linspace(0, 1, n)

            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2.  Analytic families ------------------------------------------------
    candidates.append(arcsine_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.18))
    candidates.append(gaussian_init(n, sigma_factor=0.28))

    # Beta family (symmetric)
    for alpha in np.linspace(0.2, 2.5, 22):

    # Beta family (asymmetric – a small grid)
    for a in [0.3, 0.6, 1.0, 1.5]:

    # Exponential decays

        else:

                top.sort(key=lambda x: x[0])

    best_score, best_a = top[0]
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing of the top few candidates
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.42 * budget_s
    for sc, a in top:
        if time.time() >= phase1_end:

        a_opt, sc_opt = lbfgs_refine(a, phase1_end, fft_len, n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed max
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,

        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
