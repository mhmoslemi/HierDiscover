def opt_beta(start_alpha, start_beta, n, fft_len):
    def obj(log_params):
        a = np.exp(log_params[0])
        b = np.exp(log_params[1])
        seq = beta_init(n, a, b)
        return fast_score(seq, fft_len, n)
    res = minimize(obj, x0=np.log([start_alpha, start_beta]), method='L-BFGS-B', bounds=[(np.log(0.01), np.log(100)), (np.log(0.01), np.log(100))], options={'maxiter': 200})
    a_opt = np.exp(res.x[0]); b_opt = np.exp(res.x[1])
    best_seq = beta_init(n, a_opt, b_opt)
    return best_seq, fast_score(best_seq, fft_len, n)

"""
search for a low‑score non‑negative step sequence.

The algorithm proceeds in three phases:

1️⃣  Build a diverse pool of candidate windows (Beta‑density,
    raised‑cosine, Kaiser, Gaussian, classic windows, random draws,
    and the previous best if present).  A systematic grid search on
    the Beta and raised‑cosine parameters is performed, followed by
    a short smoothing sweep (moving‑average kernels).  The best few
    candidates are kept.

2️⃣  Refine the current best by a cascade of continuous optimisers:
    * L‑BFGS on a log‑sum‑exp smoothed max‑convolution (decreasing
      smoothing temperature schedule).
    * Projected sub‑gradient descent on the exact max‑convolution.
    * A greedy hill‑climb that relocates a small fraction of mass
      between random indices.

    The cascade runs until the allocated time budget is exhausted,
    repeatedly restarting L‑BFGS from a slight random perturbation
    of the current best.

3️⃣  The final normalised vector is evaluated with the official
    `evaluate_sequence` function and stored in the global
    `height_sequence_1` for possible future runs.

All optimisation works on the probability simplex (∑a_i = 1), which
matches the scale‑invariant nature of the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(v: np.ndarray) -> np.ndarray:
    """Clip negatives, then rescale to sum‑1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    return np.full_like(v, 1.0 / v.size)


# ----------------------------------------------------------------------
# Fast evaluation (exact for sum‑1 vectors)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    return normalize(a)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = np.arange(1, n + 1, dtype=float) ** (-p)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    return normalize(a)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    return normalize(a)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return normalize(a)


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    return normalize(a)


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    return normalize(a)


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(a)


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    return normalize(a)


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return normalize(a)


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return normalize(a)


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    return normalize(a)

# Smoothing helper

def smooth_moving_average(a: np.ndarray, window: int) -> np.ndarray:
    """Convolve with a uniform kernel of length `window` (odd preferred)."""
    if window <= 1:
        return a.copy()
    kernel = np.ones(window) / float(window)
    b = np.convolve(a, kernel, mode='same')
    return normalize(b)


# ----------------------------------------------------------------------
# Objective & gradient for smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the previous result.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Greedy hill‑climb: move a random fraction of mass from i to j if it strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def generate_candidate_pool(n: int, rng: np.random.Generator):
    """Create a rich pool of initial windows."""
    candidates = []

    # reuse previous solution if it exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            candidates.append(normalize(a0))

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # systematic Beta grid
    alpha_grid = np.concatenate([
        np.linspace(0.2, 6.0, 30),
        np.logspace(np.log10(6.1), np.log10(20.0), 20)
    ])
    beta_grid = alpha_grid
    for a in alpha_grid:
        for b in beta_grid:
            candidates.append(beta_init(n, float(a), float(b)))

    # raised‑cosine grid
    for p in np.linspace(0.3, 3.0, 28):
        candidates.append(raised_cosine_init(n, float(p)))

    # Kaiser grid
    for k in np.linspace(3.0, 12.0, 20):
        candidates.append(kaiser_init(n, float(k)))

    # Gaussian sigma grid
    for sigma in np.linspace(0.15, 0.35, 16):
        candidates.append(gaussian_init(n, sigma_factor=float(sigma)))

    # random draws for diversity
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    return candidates


def smooth_candidate_set(pool, fft_len, n):
    """Apply a short moving‑average sweep to every candidate."""
    windows = [3, 5, 7, 9, 11, 15, 21, 31]
    extended = list(pool)      # keep originals
    for a in pool:
        for w in windows:
            sm = smooth_moving_average(a, w)
            extended.append(sm)
    # keep the best few after smoothing
    scored = [(fast_score(a, fft_len, n), a) for a in extended]
    scored.sort(key=lambda x: x[0])
    return [a for _, a in scored[:30]]  # keep 30 strongest


def convex_combine_top(pool, fft_len, n):
    """Form convex mixtures of the top‑5 candidates."""
    # evaluate once
    scored = [(fast_score(a, fft_len, n), a) for a in pool]
    scored.sort(key=lambda x: x[0])
    top5 = [a for _, a in scored[:5]]
    mixed = []
    for i in range(len(top5)):
        for j in range(i + 1, len(top5)):
            for w in [0.2, 0.4, 0.6, 0.8]:
                a = w * top5[i] + (1.0 - w) * top5[j]
                mixed.append(normalize(a))
    # also keep the original pool entries
    mixed.extend(top5)
    scored2 = [(fast_score(a, fft_len, n), a) for a in mixed]
    scored2.sort(key=lambda x: x[0])
    return [a for _, a in scored2[:20]]  # keep 20 best mixtures

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial candidate pool
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    pool = generate_candidate_pool(n, rng)
    print(f"[pool] generated {len(pool)} raw candidates")

    # keep the best 30 after a quick smoothing sweep
    pool = smooth_candidate_set(pool, fft_len, n)
    print(f"[pool] after smoothing keep {len(pool)} candidates")

    # add convex mixtures of the top few
    pool = convex_combine_top(pool, fft_len, n)
    print(f"[pool] after convex combination keep {len(pool)} candidates")

    # ------------------------------------------------------------------
    # Identify the current best candidate
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for a in pool:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score={best_score:.10f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 700

    while time.time() < phase1_end:
        # perturb the current best a little and re‑optimise
        perturb = random_initial(n, rng)
        alpha = rng.random() * 0.35  # ≤ 0.35
        a_start = (1.0 - alpha) * best_a + alpha * perturb
        a_start = normalize(a_start)

        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, fft_len, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.10f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈25 % of time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        a_opt, sc_opt = subgradient_exact_optimize(
            best_a, phase2_end, n, fft_len, step0=0.25)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[sub‑grad] new best fast_score={best_score:.10f}")

    # ------------------------------------------------------------------
    # Phase 3 – Greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_sc = fast_score(best_a, fft_len, n)
        a_opt, sc_opt = hill_climb(
            best_a, cur_sc, fft_len, n, deadline, rng,
            max_iter=120000, delta_frac=0.045)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[hill‑climb] improved fast_score={best_score:.10f}")

        # occasional small random perturbation to escape shallow plateaus
        if rng.random() < 0.12:
            pert = random_initial(n, rng)
            best_a = normalize(0.92 * best_a + 0.08 * pert)

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.10f}")

    # expose the best sequence for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
