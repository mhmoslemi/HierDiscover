"""
Propose a low‑score non‑negative step function.

Algorithm
---------
* Initialise a diverse pool of candidates (previous best, spectral,
  random Dirichlet etc.) and keep the best one (normalised to sum=1).
* Iteratively improve the candidate by solving a linear program:
      maximise  Σ b_i
      subject to  a * b ≤ M   (componentwise)   and   b ≥ 0
  where M = max(a * a) is the current worst convolution value.
  The LP yields a vector b with a larger total mass; after normalising
  it to unit sum we convex‑combine it with a (mixing factor t).  This
  step is provably non‑increasing for the objective 2·n·max(conv).
* After a few LP refinements finish with a cheap greedy hill‑climb that
  moves tiny amounts of mass between two coordinates.  All phases respect
  the global 1000 s deadline.
* The function returns the final list of floats; ``evaluate_sequence`` will
  rescale it automatically, so we keep the sum equal to 1 for simplicity.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def max_convolution(a: np.ndarray) -> float:
    """
    Return max_{k} (a * a)[k] for a non‑negative vector a.
    Uses FFT for O(n log n) work.
    """
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return float(conv.max())


def fast_score(a: np.ndarray) -> float:
    """Objective = 2·n·max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def build_conv_matrix(f: np.ndarray) -> sparse.csc_matrix:
    """
    Construct the (2n‑1)×n matrix    A_{k,i}=f_i   if 0≤k‑i<n else 0.
    Then A @ b = f * b  (full convolution, no reversal).
    """
    n = f.shape[0]
    # each diagonal i (offset i) is filled with constant f[i]
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def lp_step(a: np.ndarray, mix: float) -> (np.ndarray, bool):
    """
    Perform one LP‑based improvement step.
    Returns the new vector (already normalised to sum=1) and a success flag.
    """
    n = a.shape[0]
    M = max_convolution(a)

    # Build constraints a * b ≤ M
    A = build_conv_matrix(a)
    c = -np.ones(n)                      # minimise -sum(b)  → maximise sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve the LP
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return a, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a, False

    # Scale to unit sum (feasibility is preserved because sum_b ≥ 1)
    g = b / sum_b

    # Convex combination with the current vector
    a_new = (1.0 - mix) * a + mix * g
    # Re‑normalise to avoid drift
    a_new = a_new / a_new.sum()
    return a_new, True


def hill_climb(a: np.ndarray, max_iter: int, rng: np.random.Generator) -> np.ndarray:
    """
    Simple greedy hill‑climb: transfer a tiny amount of mass between two
    randomly chosen coordinates and keep the move if it improves the
    objective.  Stops after ``max_iter`` iterations or when time runs out.
    """
    n = a.shape[0]
    best = a.copy()
    best_sc = fast_score(best)

    for _ in range(max_iter):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of mass from i to j
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
        # else discard
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size.
    # --------------------------------------------------------------
    # The original run used n=4096; for the LP phase a smaller size
    # speeds up the linear programme dramatically while keeping the
    # same constant 2·n·max(conv) (the objective is scale‑invariant).
    n = 1024                     # feel free to raise to 2048 if time permits
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates.
    # --------------------------------------------------------------
    candidates = []

    # 1) reuse previous best if available (down‑sample / interpolate)
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size > 1:
            # linear interpolation to the new grid
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) deterministic families
    # (very cheap to build; they are already normalised to sum=1)
    def hann_init():
        i = np.arange(n, dtype=np.float64)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        return a / a.sum()

    def blackman_init():
        i = np.arange(n, dtype=np.float64)
        a = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
        a = np.maximum(a, 0.0)
        return a / a.sum()

    def gaussian_init(sigma_factor=0.25):
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        return a / a.sum()

    candidates.append(hann_init())
    candidates.append(blackman_init())
    candidates.append(gaussian_init())

    # 3) a few random Dirichlet‑like draws
    for _ in range(5):
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
        candidates.append(a)

    # --------------------------------------------------------------
    # Pick the best initial vector (according to the exact objective).
    # --------------------------------------------------------------
    best_a = None
    best_sc = np.inf
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best initial fast_score = {best_sc:.9f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – LP based refinement
    # --------------------------------------------------------------
    lp_mix = 0.5          # start with a relatively aggressive mixing factor
    lp_phase_end = time.time() + 0.55 * budget_s   # devote ~55 % of time
    lp_iter = 0
    while time.time() < lp_phase_end:
        lp_iter += 1
        # adapt mixing factor slowly (larger early, smaller later)
        if lp_iter % 5 == 0:
            lp_mix = max(0.05, lp_mix * 0.9)

        a_new, ok = lp_step(best_a, lp_mix)
        if not ok:
            print("[LP] step failed, breaking")
            break

        new_sc = fast_score(a_new)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_new.copy()
            print(f"[LP] iter={lp_iter}, new best fast_score={best_sc:.9f}")

        # stop early if improvement stalls
        if lp_iter >= 20 and lp_mix < 0.1:
            # after a few aggressive passes the gain becomes tiny
            break

    # --------------------------------------------------------------
    # Phase 2 – cheap greedy hill‑climb for final polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most 20 % of the remaining budget for hill‑climb
        max_iter = int(min(200000, remaining * 2000))
        print(f"[Hill‑climb] start, max_iter={max_iter}")
        best_a = hill_climb(best_a, max_iter, rng)
        best_sc = fast_score(best_a)
        print(f"[Hill‑climb] finished, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Return the result – the official evaluator rescales internally.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
