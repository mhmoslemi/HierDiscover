"""Search for low‑score step function via random parametric families,
extensive random exploration, L‑BFGS smoothing of the max‑convolution,
and a final greedy hill‑climb.  The sequence is kept on the
probability simplex so that the official `evaluate_sequence` is
invariant to its internal scaling."""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact score (2·n·max(a*a)) for a probability vector a."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of the self‑convolution (log‑sum‑exp) and its gradient."""
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # gradient w.r.t. logits y (softmax Jacobian)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage. Returns best a and its score.
    """
    y = np.log(a_start + 1e-12)          # logits for softmax (always positive)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover

            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 150000,
               delta_frac: float = 0.04):
    """Greedy hill‑climb with random mass transfers."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a

def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diverse random vector on the simplex."""
    mode = rng.choice(['uniform', 'dirichlet', 'exp', 'power', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        concentration = rng.uniform(0.05, 1.0)
        a = rng.gamma(concentration, 1.0, size=n)
        a = a / a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = a / a.sum()
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a

def symmetric_power_init(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law decay a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def symmetric_exponential_init(n: int, sigma: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / sigma) + np.exp(-(n - 1.0 - i) / sigma)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

"""
Search for a low‑score step function.

The algorithm consists of three phases:

1.   A very fast random exploration of many parametric families
     (beta, symmetric power / exponential, mixtures of classic windows,
     Dirichlet‑like draws, etc.).  The best vector found is kept.

2.   From the current best, a series of L‑BFGS refinements is applied
     to the smoothed max of the self‑convolution (log‑sum‑exp
     smoothing).  The smoothing parameter is reduced on a schedule,
     each stage starting from the previous result.

3.   A short greedy hill‑climb (random mass transfers) is performed
     until the time budget is exhausted.

All vectors are kept on the probability simplex, which guarantees that
`evaluate_sequence` returns exactly `2·n·max(conv)` – the same value as
our internal `fast_score`.  The final sequence is stored in the global
`height_sequence_1` variable for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (identical to `evaluate_sequence` after normalisation)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum = 1).
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for softmax (strictly positive)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # very rare, just skip the stage
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 150000,
               delta_frac: float = 0.04):
    """
    Greedy hill‑climb: transfer a random fraction of mass from i to j
    if the move strictly improves the fast_score.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)

            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diverse random vector on the simplex."""
    mode = rng.choice(['uniform', 'dirichlet', 'exp', 'power', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        concentration = rng.uniform(0.05, 1.0)
        a = rng.gamma(concentration, 1.0, size=n)
        a = a / a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = a / a.sum()
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


def symmetric_power_init(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def symmetric_exponential_init(n: int, sigma: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / sigma) + np.exp(-(n - 1.0 - i) / sigma)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal score.
    The best sequence (as a list of floats) is returned and stored in
    the global variable `height_sequence_1` for reuse.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # problem size
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # -----------------------------------------------------------------
    # Minimal initialisation (deterministic + previous run)
    # -----------------------------------------------------------------
    best_a = None
    best_score = float('inf')

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
                sc = fast_score(a0, fft_len, n)
                if sc < best_score:
                    best_score = sc
                    best_a = a0.copy()

    for fn in (triangular_init, hann_init, blackman_init,
               gaussian_init, arcsine_init, inv_sqrt_init):
        a = fn(n)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # -----------------------------------------------------------------
    # Phase 1 – intensive random exploration (≈50 % of budget)
    # -----------------------------------------------------------------
    start = time.time()
    phase1_end = start + 0.5 * budget_s
    while time.time() < phase1_end:
        sel = rng.integers(0, 8)
        if sel == 0:
            # symmetric power‑law
            p = rng.uniform(0.1, 2.5)
            a = symmetric_power_init(n, p)
        elif sel == 1:
            # symmetric exponential
            sigma = rng.uniform(0.001 * n, 0.2 * n)
            a = symmetric_exponential_init(n, sigma)
        elif sel == 2:
            # beta density with log‑uniform parameters
            log_alpha = rng.uniform(-2.0, 0.7)
            log_beta = rng.uniform(-2.0, 0.7)
            alpha = 10.0 ** log_alpha
            beta = 10.0 ** log_beta
            a = beta_init(n, alpha, beta)
        elif sel == 3:
            # Dirichlet‑like random draw
            concentration = rng.uniform(0.05, 1.2)
            a = rng.gamma(concentration, 1.0, size=n)
            a /= a.sum()
        elif sel == 4:
            # mixture of a few classic windows
            w = rng.random(5)
            w /= w.sum()
            shapes = [triangular_init(n), hann_init(n), blackman_init(n),
                      gaussian_init(n), arcsine_init(n)]
            a = sum(wi * si for wi, si in zip(w, shapes))
            a = a / a.sum()
        elif sel == 5:
            # double‑edge narrow Gaussian
            sigma = rng.uniform(0.001 * n, 0.02 * n)
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-0.5 * ((i) / sigma) ** 2) + np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
            a = a / a.sum()
        elif sel == 6:
            # one‑sided power‑law (non‑symmetric)
            p = rng.uniform(0.1, 2.0)
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a = a / a.sum()
        else:
            a = random_initial(n, rng)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[rand] new best fast_score={best_score:.8f}")

    print(f"[phase1] random search finished – best fast_score={best_score:.8f}")

    # -----------------------------------------------------------------
    # Phase 2 – L‑BFGS refinement (≈40 % of budget)
    # -----------------------------------------------------------------
    phase2_end = start + 0.90 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.2, 0.12, 0.07,
        0.04, 0.02, 0.01, 0.005, 0.002,
        0.001, 5e-04, 2e-04, 1e-04, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
        5e-07, 2e-07, 1e-07, 5e-08, 1e-08
    ]
    maxiter_per_lam = 1200

    while time.time() < phase2_end:
        # slightly randomise the current best before refinement
        perturb = random_initial(n, rng) * 0.15
        a_start = (1.0 - 0.15) * best_a + perturb
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase2_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    print(f"[phase2] L‑BFGS phase completed – best fast_score={best_score:.8f}")

    # -----------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (remaining time)
    # -----------------------------------------------------------------
    cur_score = fast_score(best_a, fft_len, n)
    while time.time() < start + budget_s - 0.3:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best_a[i] * rng.random() * 0.03
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < cur_score:
            best_a = cand
            cur_score = sc
            best_score = sc
            print(f"[hill] improved fast_score={best_score:.8f}")

    # -----------------------------------------------------------------
    # Final evaluation with the official scorer
    # -----------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    # store for possible reuse
    globals()['height_sequence_1'] = final_seq
    return final_seq
