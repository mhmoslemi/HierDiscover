"""
Propose a low‑score non‑negative step function.

The algorithm consists of several complementary phases that work within a
user‑specified time budget (default 1000 s).

* Phase 0 – Spectral flattening (alternating projections).  Starting from a
  random feasible vector we repeatedly enforce a *constant* magnitude of the
  zero‑padded DFT (which would give a perfectly flat autocorrelation) and
  project back onto the probability simplex.  This quickly produces a shape
  with a much flatter self‑convolution than a plain random or window
  initialization.

* Phase 1 – Exact sub‑gradient descent on the true max‑convolution.
  Uses an additive projected‑gradient scheme on the simplex.  The step size
  decays like 1/√t.

* Phase 2 – Adam optimisation on the exact objective (the true max of the
  self‑convolution).  Operates on soft‑max logits so that the simplex
  constraint is satisfied automatically.

* Phase 3 – Cheap simulated‑annealing moves for fine‑tuning.

* Phase 4 – Deterministic greedy hill‑climb (tiny pairwise mass transfers)
  for last polishing.

All phases respect a global deadline; the best sequence found is returned
as a plain Python list.  The official ``evaluate_sequence`` rescales it
internally, so the optimisation works on the probability simplex
(∑ aᵢ = 1).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact objective for vectors on the simplex (sum == 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a on the simplex (sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of flat spectrum) – produces an initial smooth shape."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Randomised families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using a variety of heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'lognorm'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def mixture_gaussian_init(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Gaussians with random means, sigmas and weights."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def dirichlet_sparse_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw (small alphas)."""
    alpha = rng.uniform(0.01, 0.2, size=n)
    a = rng.gamma(alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Phase 0 – alternating spectral‑flattening projection
# ----------------------------------------------------------------------


def spectral_flattening(a_start: np.ndarray, n: int, fft_len: int,
                        max_iter: int, deadline: float,
                        rng: np.random.Generator):
    """Iteratively enforce constant FFT magnitude and non‑negativity."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded vector
        A = np.fft.rfft(a, fft_len)

        # RMS magnitude (constant that preserves total energy)
        rms = np.linalg.norm(A) / np.sqrt(A.shape[0])

        # Replace magnitudes by the constant rms, keep current phases
        A_new = rms * np.exp(1j * np.angle(A))

        # Inverse FFT, keep only the first n samples
        a_new = np.fft.irfft(A_new, fft_len)[:n]

        # Project onto the probability simplex (non‑negative, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = best_a.copy()

        # Optional symmetry enforcement (helps flattening)
        a_new = (a_new + a_new[::-1]) * 0.5
        s = a_new.sum()
        if s > 0:
            a_new /= s

        a = a_new

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # occasional log
            if it % 50 == 0:
                print(f"[Spectral] it={it}, best fast_score={best_sc:.9f}")

        # Small random jitter every few iterations to avoid stagnation
        if (it + 1) % 200 == 0:
            a += rng.normal(scale=1e-4, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 1 – exact sub‑gradient descent (additive, simplex‑projected)
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray, deadline: float,
                        n: int, fft_len: int, eps: float = 1e-12):
    """Projected additive sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # Linear convolution a*a using FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        # All positions that are (almost) maximal become active
        mask = (conv >= max_c - eps).astype(np.float64)

        # Sub‑gradient of max‑convolution w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # Step size schedule (classical 1/√t)
        eta = 0.5 / np.sqrt(it + 1.0)

        # Additive update + projection onto simplex
        a = a - eta * g
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 100 == 0:
                print(f"[Subgrad] it={it}, best fast_score={best_sc:.9f}")

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – Adam on the exact max‑convolution (softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray, deadline: float,
                  n: int, fft_len: int, rng: np.random.Generator):
    """Adam optimiser on the exact max‑convolution objective."""
    # Work in log‑space to keep the simplex automatically
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_y = y.copy()
    best_sc = fast_score(np.exp(y) / np.exp(y).sum(), fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = np.exp(y)
        a /= a.sum()

        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient of max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # full gradient of 2·n·max
        grad_a = 2.0 * n * g_a

        # back‑prop through soft‑max
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.2 / np.sqrt(t)          # decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to exit tiny plateaux
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = np.exp(y)
        a_curr /= a_curr.sum()
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_y = y.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.9f}")

    # Return the best encountered vector
    best_a = np.exp(best_y)
    best_a /= best_a.sum()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – Simulated annealing (pairwise mass transfers)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        n: int, fft_len: int, rng: np.random.Generator,
                        init_temp: float = 5e-4, final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """Simple pairwise mass‑transfer annealing."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # propose moving a small fraction of mass from i to j
        delta = a[i] * rng.random() * 0.16
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_sc: float, fft_len: int, n: int,
               deadline: float, rng: np.random.Generator,
               max_iter: int = 100000, delta_frac: float = 0.04):
    """Strictly improving greedy hill‑climb with tiny pairwise transfers."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function with a low evaluation score."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a power‑of‑two length (larger = more freedom)
    # ------------------------------------------------------------------
    n = 4096                     # feel free to raise (e.g. 8192) if time permits
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # randomised families
    for _ in range(3):
        candidates.append(random_initial(n, rng))
    for _ in range(2):
        candidates.append(mixture_gaussian_init(n, rng, K=rng.integers(2, 5)))
    candidates.append(dirichlet_sparse_init(n, rng))

    # pick the best of the pool (exact fast_score)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – Spectral flattening (multiple random restarts)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.15 * budget_s
    num_restarts = 0
    while time.time() < phase0_end:
        # start from a random point mixed with current best
        a0 = random_initial(n, rng)
        mix = rng.random() * 0.4
        a_start = (1.0 - mix) * best_a + mix * a0
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, sc_opt = spectral_flattening(a_start, n, fft_len,
                                            max_iter=200,
                                            deadline=phase0_end,
                                            rng=rng)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[Spectral‑flatten] new best fast_score={best_sc:.9f}")
        num_restarts += 1
    print(f"[Spectral‑flatten] performed {num_restarts} restarts")

    # ------------------------------------------------------------------
    # Phase 1 – Exact sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.30 * budget_s
        print("[Subgrad] start")
        best_a, best_sc = subgradient_descent(best_a, subgrad_deadline,
                                              n, fft_len, eps=1e-14)
        print(f"[Subgrad] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a, deadline, n, fft_len, rng)
        print(f"[Adam] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a, sa_deadline,
                                              n, fft_len, rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=120000)
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 15.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a, best_sc,
                                     fft_len, n,
                                     hill_deadline, rng,
                                     max_iter=60000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result – the official evaluator rescales internally.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
