"""
propose_candidate
-----------------

A multi‑phase search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

(the normalised version of ``evaluate_sequence``).

The algorithm proceeds as follows

1. **Parameter sweep** – generate many families of shapes (Beta,
   raised‑cosine, Hann, Blackman, arcsine, triangular).  For each
   shape the fast “probability‑vector” score
   ``2·n·max(conv)`` is computed (the sum is forced to 1, so the scaling
   factor of ``evaluate_sequence`` is omitted).  The best few
   candidates are kept.

2. **Local refinement** – each kept candidate is polished by a
   continuation of L‑BFGS on a smoothed log‑sum‑exp approximation of
   the max‑convolution, followed by Adam optimisation and a
   sub‑gradient descent on the exact max‑convolution.

3. **Fine‑grained greedy search** – a projected exponential‑gradient
   hill‑climb that focuses on the current top‑fraction of convolution
   entries is applied until the time budget expires.

The routine respects the 1000 s wall‑time budget, checks the time
regularly and stops early if needed.  The best sequence found is
returned (and stored in ``height_sequence_1`` for possible reuse).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (vector normalised to sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum=1).  This equals the
    value returned by ``evaluate_sequence`` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers – families of candidate windows
# ----------------------------------------------------------------------


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density discretisation on the simplex."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    """(1‑cos)ᵖ window (generalised Hann)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a).
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    The direction is taken from the unique maximal index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        # the contributing slice of a
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # Projected direction (tangent to simplex)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # Adaptive step with simple back‑tracking
        step = step0 / np.sqrt(it + 1.0)
        for _ in range(12):
            a_new = a + step * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                break
            step *= 0.5
        it += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.001):
    """
    Fine‑grained exponentiated‑gradient hill‑climb.
    The step size decays linearly from ``delta_start`` to ``delta_end``.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem sizes to try – different lengths sometimes give a slight edge
    # ------------------------------------------------------------------
    n_candidates = [4000, 5000, 6000]

    best_seq = None
    best_eval = float('inf')
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # Helper to evaluate a shape and keep the global best
    # ------------------------------------------------------------------
    def consider(a_vec: np.ndarray, n_val: int):
        nonlocal best_seq, best_eval, best_score, best_n
        fft_len_local = next_pow2(2 * n_val - 1)
        sc = fast_score(a_vec, fft_len_local, n_val)
        if sc < best_score:
            best_score = sc
            best_seq = a_vec.copy()
            best_n = n_val

    # ------------------------------------------------------------------
    # Phase A – systematic family sweep (cheap, runs for all n)
    # ------------------------------------------------------------------
    print("[search] Start family sweep")
    for n in n_candidates:
        if time.time() >= deadline:
            break
        fft_len = next_pow2(2 * n - 1)

        # deterministic bases
        family = [
            arcsine_init(n),
            triangular_init(n),
            hann_init(n),
            blackman_init(n)
        ]

        # Beta grid (symmetric)
        for alpha in np.linspace(0.30, 0.80, 26):
            family.append(beta_init(n, alpha, alpha))

        # Beta asymmetric (coarse)
        alphas = np.linspace(0.30, 0.80, 9)
        betas = np.linspace(0.30, 0.80, 9)
        for a in alphas:
            for b in betas:
                if abs(a - b) < 0.001:  # skip symmetric duplicates
                    continue
                family.append(beta_init(n, a, b))

        # Raised‑cosine family
        for p in np.linspace(0.5, 3.0, 21):
            family.append(raised_cosine_init(n, p))

        # evaluate all
        scores = []
        for a in family:
            sc = fast_score(a, fft_len, n)
            scores.append((sc, a))
        scores.sort(key=lambda tup: tup[0])

        # keep the best three for deeper optimisation
        top_three = scores[:3]
        for sc, a in top_three:
            consider(a, n)

    print(f"[search] After sweep: best fast_score={best_score:.9f} (n={best_n})")

    # ------------------------------------------------------------------
    # Phase B – local polishing of the current global best
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_n = max(n_candidates)
        best_seq = np.full(best_n, 1.0 / best_n)
        best_score = fast_score(best_seq, next_pow2(2 * best_n - 1), best_n)

    # Prepare frequently used variables
    n = best_n
    fft_len = next_pow2(2 * n - 1)

    # L‑BFGS continuation with a refined λ schedule
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]

    phase_end = min(deadline, start_time + 0.30 * budget_s)
    while time.time() < phase_end:
        # blend with a fresh random start to escape possible local traps
        rand_vec = np.abs(rng.standard_normal(n))
        rand_vec /= rand_vec.sum()
        alpha = rng.random() * 0.3          # blend factor ≤ 0.3
        a_start = (1.0 - alpha) * best_seq + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule,
                                           maxiter_per_lam=800,
                                           deadline=phase_end,
                                           fft_len_conv=fft_len,
                                           fft_len_g=fft_len,
                                           n=n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[L‑BFGS] updated best fast_score={best_score:.9f}")

    # Adam refinement (≈20 % of remaining budget)
    phase2_end = min(deadline, start_time + 0.55 * budget_s)
    if time.time() < phase2_end:
        best_seq, best_score = adam_optimize(best_seq,
                                             phase2_end,
                                             n,
                                             fft_len,
                                             rng,
                                             max_iters=6000,
                                             lam_init=0.30,
                                             lam_decay=0.9993,
                                             lr=0.07)
        print(f"[Adam] best fast_score={best_score:.9f}")

    # Exact‑max sub‑gradient descent with back‑tracking line search
    phase3_end = min(deadline, start_time + 0.70 * budget_s)
    if time.time() < phase3_end:
        best_seq, best_score = subgradient_exact_optimize(best_seq,
                                                          phase3_end,
                                                          n,
                                                          fft_len,
                                                          step0=0.20)
        print(f"[Sub‑grad] best fast_score={best_score:.9f}")

    # Fine‑grained exponential‑gradient hill‑climb until the very end
    while time.time() < deadline:
        cur_sc = fast_score(best_seq, fft_len, n)
        best_seq, best_score = hill_climb_fine(best_seq,
                                               cur_sc,
                                               fft_len,
                                               n,
                                               deadline,
                                               rng,
                                               max_iter=250000,
                                               delta_start=0.04,
                                               delta_end=0.0008)
        # occasional progress report
        if int(time.time() - start_time) % 30 == 0:
            print(f"[Fine hill] interim fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f} (n={best_n})")

    # expose the best sequence for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
