def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    best_seq = None
    best_score = float('inf')
    
    # Quick global search over n and scaling factor c = n*(1-r)
    # We'll sample n values geometrically or linearly.
    for n in range(50, 5001, step):
        # compute baseline product = 2.0 (approx)
        # Use a set of product candidates around optimum.
        for prod in [1.2, 1.5, 2.0, 2.5, 3.0]:
            r = 1 - prod / n
            if r <= 0 or r >= 1:
                continue
            seq = r ** np.arange(n)  # a0=1
            # Optionally scale to keep max <=1000, but values <=1 so fine.
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best: score={best_score:.6f} n={n} r={r:.6f} prod={prod}")
        if time.time() - start_time > budget_s - 5:
            break
    # Additional local refinement: for best n, perform small random search for r around best r.
    if best_seq is not None:
        n_best = len(best_seq)
        # Use hill climbing on r
        r_best = best_seq[1]/best_seq[0] if n_best>1 else 0.9
        # We have a product approx n*(1-r) = c: compute c_best else.
        c_best = n_best * (1 - r_best)
        for i in range(2000):
            # propose new c by small Gaussian perturbation.
            c_new = c_best + np.random.randn()*0.2  # step
            if c_new <= 0:
                continue
            r_new = 1 - c_new / n_best
            if r_new <= 0 or r_new >= 1:
                continue
            seq = r_new ** np.arange(n_best)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                r_best = r_new
                c_best = c_new
                print(f"Refine: new best {best_score:.6f} n={n_best} r={r_best:.6f}")
            if time.time() - start_time > budget_s - 5:
                break
    if best_seq is None:
        # fallback: return some simple sequence to avoid nan
        best_seq = [1.0, 0.5, 0.25]
    print(f"Final best score: {best_score}")
    return best_seq
