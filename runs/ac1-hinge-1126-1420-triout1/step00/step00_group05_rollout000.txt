"""
Propose a sequence of non‑negative heights that minimises
    2 * n * max(conv(a, a)) / (sum a)^2

The objective is homogeneous, therefore we keep the L¹‑norm of the
sequence equal to 1 (sum = 1).  With this normalisation the score is simply
`2 * n * max(conv)`.  The algorithm is a simple projected sub‑gradient
method on the function `max(conv)`, interleaved with cheap random
perturbations and occasional LP moves (the LP helper from the previous
baseline is reused when available).

Key ideas
---------
* The convolution used by ``evaluate_sequence`` is the *full* (non‑circular)
  convolution: `c[k] = Σ_i a[i]·a[k‑i]`.  The gradient of the
  maximal entry `c[k*]` w.r.t. the vector `a` is
  `∂c[k*]/∂a[j] = 2·a[k*‑j]` (if the index is inside the array, otherwise 0).
  This yields a cheap sub‑gradient that can be used for a descent step.
* After each descent step we project the vector back onto the simplex
  (`a ≥ 0` and `Σ a = 1`).  The projection is simply a clipping to zero
  and renormalisation.
* A line‑search over a few step‑sizes is performed; the best candidate of
  this small search is kept.
* Random perturbations (Gaussian noise on a few coordinates) and occasional
  restarts avoid getting stuck in the uniform fixed‑point.
* When the LP helper from the original code is present we occasionally
  use it as an additional move.

The routine runs for the allotted time (default 1000 s) and returns the
best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, ks):
    """Return a sub‑gradient for the max‑convolution at indices ks."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        if k < n:
            # indices 0 … k   ←→   a[k], a[k‑1], … , a[0]
            g[: k + 1] += 2.0 * a[k :: -1]
        else:
            # k ≥ n: overlapping region starts at `start`
            start = k - (n - 1)
            # a[start … n‑1] reversed
            g[start:] += 2.0 * a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Helper: one projected sub‑gradient descent step with tiny line‑search
# ----------------------------------------------------------------------
def _gradient_step(seq, rng, base_factor=0.2):
    """Perform a sub‑gradient descent step on `max(conv)`.

    Returns a tuple (new_sequence, new_score).
    """
    a = np.asarray(seq, dtype=float)
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.where(conv >= max_val - 1e-12)[0]   # indices attaining the max

    # sub‑gradient of the max entry (or sum of them, if several tie)
    g = _subgradient_max_conv(a, ks)

    # line‑search – try a few decreasing step lengths
    best_seq = seq
    best_score = 2.0 * n * max_val
    max_g = g.max()
    if max_g == 0.0:
        return best_seq, best_score

    # step sizes are expressed as “fraction of the maximal component of g”
    for factor in (1.0, 0.5, 0.25, 0.125):
        step = (base_factor * factor) / max_g
        new_a = a - step * g
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s == 0.0:
            continue
        new_a /= s
        new_conv = np.convolve(new_a, new_a)
        new_score = 2.0 * n * new_conv.max()
        if new_score < best_score:
            best_score = new_score
            best_seq = new_a.tolist()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: small random perturbation of a sequence
# ----------------------------------------------------------------------
def _random_perturb(seq, rng, sigma=0.02):
    a = np.asarray(seq, dtype=float)
    n = a.size
    # perturb a few entries
    for _ in range(rng.randint(1, 4)):
        idx = rng.randint(n)
        a[idx] = max(0.0, a[idx] + rng.randn() * sigma)
    s = a.sum()
    if s == 0.0:
        a = rng.rand(n)
        s = a.sum()
    a /= s
    return a.tolist()

# Helper: optional LP move (uses the helper defined in the starter code)
# ----------------------------------------------------------------------

        if cand is not None:

    return None

    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python list of floats)
    # ------------------------------------------------------------------
    return best_seq
