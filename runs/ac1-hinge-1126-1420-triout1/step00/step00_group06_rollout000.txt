"""
Propose a sequence of non‑negative heights that (hopefully) improves the bound
for ``evaluate_sequence``.

Algorithm
---------
1.  Choose a moderate length (default 2000 – large enough to give a good
    bound but small enough for fast convolutions).
2.  Initialise a random vector and scale it so that the L1‑norm equals
    ``sqrt(2·n)`` – this is the normalisation used by the evaluator and makes
    the objective equal to the maximal convolution value.
3.  Simulated‑annealing / hill‑climbing loop
    *   Compute the current convolution and its maximum.
    *   With probability *p_target* (≈0.1) perform a **targeted move**:
        – locate the index ``k_max`` where the convolution is maximal,
        – pick a random position ``i`` that participates in that term,
          i.e. ``j = k_max‑i`` is inside the array,
        – move a small amount of mass from both ``i`` and ``j`` to a random
          third index.  This directly reduces the dominant convolution term.
    *   Otherwise perform a simple **mass‑exchange move**:
        – pick two distinct indices ``i`` and ``j``,
        – transfer a random fraction of ``a[i]`` to ``a[j]`` while keeping the
          total sum unchanged.
    *   Accept the new state if it improves the score, otherwise accept it
        with a Metropolis probability depending on a temperature that cools
        geometrically.
    *   After each accepted move, re‑normalise the vector to the required
        sum (the evaluator clamps negative values, so we keep the vector
        non‑negative at every step).
4.  Keep the best sequence seen during the whole run and return it.

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
periodically prints the current best score.  The method needs only NumPy
and works without any file or network I/O.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) yields a low value
    of ``evaluate_sequence``.  The function runs for at most ``budget_s``
    seconds and uses a simple simulated‑annealing / targeted‑move search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # keep a safety margin

    # ------------------------------------------------------------------
    # 1.  Problem size – we keep it modest to allow many iterations.
    # ------------------------------------------------------------------
    n = kwargs.get("length", 2000)                     # length of the step function
    target_sum = math.sqrt(2.0 * n)                    # required L1‑norm

    # ------------------------------------------------------------------
    # 2.  Initial vector – random Dirichlet (uniform in the simplex).
    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a = a / a.sum() * target_sum                       # enforce the normalisation

    # Current evaluation
    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing parameters.
    # ------------------------------------------------------------------
    # Start temperature roughly 10 % of the initial score.
    temperature = max(1e-8, 0.1 * best_score)
    decay = 0.9995                                      # geometric cooling
    # Typical amount of mass transferred in a plain exchange move.
    base_step = 0.05 * target_sum / n

    # Store the current score to avoid recomputing it.
    cur_score = best_score

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 3a – decide which type of move to attempt.
        # --------------------------------------------------------------
        if np.random.rand() < 0.1:                     # target‑focused move
            # Compute convolution once – cheap for the chosen size.
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))

            # Choose a random i that participates in the maximal term.
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            if i_low >= i_high:
                # fallback to a plain exchange move if the index range collapses.
                i, j = np.random.choice(n, 2, replace=False)
                delta = base_step * np.random.rand() * a[i]
                new_a = a.copy()
                new_a[i] = max(0.0, a[i] - delta)
                new_a[j] += delta
            else:
                i = np.random.randint(i_low, i_high + 1)
                j = k_max - i
                if i == j:
                    # we need two distinct positions – pick a neighbour.
                    j = i + 1 if i + 1 < n else i - 1
                # Amount to move away from the two “offending’’ positions.
                delta_i = base_step * np.random.rand() * a[i]
                delta_j = base_step * np.random.rand() * a[j]
                delta = min(delta_i, delta_j, a[i], a[j])

                # Random third index that receives the mass.
                k = np.random.choice(np.delete(np.arange(n), [i, j]))
                new_a = a.copy()
                new_a[i] = max(0.0, a[i] - delta)
                new_a[j] = max(0.0, a[j] - delta)
                new_a[k] += 2.0 * delta
        else:                                          # simple mass exchange
            i, j = np.random.choice(n, 2, replace=False)
            # Transfer a random fraction of the smaller entry.
            max_transfer = min(a[i], base_step * np.random.rand())
            delta = max_transfer
            new_a = a.copy()
            new_a[i] = a[i] - delta
            new_a[j] = a[j] + delta

        # 3b – re‑normalise to keep the L1‑norm fixed (the evaluator is
        #      invariant under a global scaling, but keeping the sum exact
        #      avoids the “sum too close to zero’’ safeguard).
        # --------------------------------------------------------------
        new_a = np.maximum(new_a, 0.0)                  # safeguard
        s = new_a.sum()
        if s == 0.0:                                   # extremely unlikely

        # 3c – evaluate and decide on acceptance.
        # --------------------------------------------------------------

        if new_score < best_score:
            best_score = new_score
            best_seq = new_a.tolist()
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability = exp( -(Δ) / T )
            if temperature > 0:
                prob = math.exp(-(new_score - cur_score) / temperature)
                accept = np.random.rand() < prob

        if accept:
            a = new_a
            cur_score = new_score

        # --------------------------------------------------------------
        # 3d – cooling schedule and occasional step‑size adaptation.
        # --------------------------------------------------------------
        temperature *= decay
        # Slowly adapt the step size to the typical magnitude of entries.
        if iteration % 1000 == 0:
            mean_val = a.mean()
            base_step = 0.05 * mean_val if mean_val > 0 else base_step

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq
