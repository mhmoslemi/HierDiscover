def subgradient_step(a, step):
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]  # might be multiple
    grad = np.zeros_like(a)
    N = len(a)
    for k in ks:
        i_min = max(0, k-(N-1))
        i_max = min(N-1, k)
        idx = np.arange(i_min, i_max+1)
        j = k - idx
        grad[idx] += 2*a[j]  # note a[j] corresponds to a at positions j
    # Update using exponential decay
    # For numerical stability, we might compute exp(-step * grad) but grad could be large.
    # Use a = a * np.exp(-step * grad)
    a_new = a * np.exp(-step * grad)
    s = a_new.sum()
    if s == 0:
        return a  # no change
    a_new = a_new / s
    return a_new

import numpy as np
import time
import random
from typing import List

def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One subgradient descent step on the max-convolution objective."""
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # all tying indices
    N = a.shape[0]
    grad = np.zeros_like(a)
    rng = np.arange(N)
    for k in ks:
        j = k - rng
        mask = (j >= 0) & (j < N)
        grad[mask] += 2.0 * a[j[mask]]
    # exponential update guarantees positivity
    a_new = a * np.exp(-step * grad)
    sum_new = a_new.sum()
    if sum_new <= 0:
        return a
    return a_new / sum_new

def random_sequence(N: int) -> np.ndarray:
    """Generate a random positive vector of length N, sum normalized to 1."""
    # Dirichlet with all parameters = 1 => exponential distribution
    x = np.random.rand(N)  # uniform in [0,1)
    # ensure no zero (though zero is fine after normalization)
    if x.sum() == 0:
        x = np.ones(N)
    return x / x.sum()

"""
Simulated‑annealing / sub‑gradient search for a non‑negative step sequence.

The routine repeatedly draws random vectors (Dirichlet‑distributed) of a
moderate length, then applies a few sub‑gradient updates that lower the
maximum value of the autocorrelation (a*a).  An exponential update guarantees
positivity, after which the vector is re‑normalised (the evaluation function
is scale‑invariant).  Small random jitter is added from time to time to
escape plateaus.  The best sequence found before the time budget expires is
returned.
"""

import time
import random
from typing import List

import numpy as np


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """Perform one sub‑gradient descent step on max(conv(a,a))."""
    # convolution of a with itself
    conv = np.convolve(a, a)
    max_val = conv.max()
    # all indices attaining the maximum (might be more than one)
    ks = np.flatnonzero(conv == max_val)

    N = a.shape[0]
    grad = np.zeros_like(a)
    rng = np.arange(N)

    # gradient of max(conv) w.r.t. a_i is 2*a_{k-i} for each active k
    for k in ks:
        j = k - rng
        mask = (j >= 0) & (j < N)
        grad[mask] += 2.0 * a[j[mask]]

    # multiplicative exponential update keeps a_i ≥ 0
    a_new = a * np.exp(-step * grad)

    s = a_new.sum()
    if s <= 0.0:                     # should never happen, but guard against it
        return a
    return a_new / s


def _random_sequence(N: int) -> np.ndarray:
    """Generate a random positive vector of length N and normalise its sum to 1."""
    # Dirichlet(1,…,1) – equivalent to i.i.d. exponential draws
    x = np.random.rand(N) + 1e-12   # avoid exact zeros
    return x / x.sum()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).  The function returns as soon as the
        budget is exhausted (a small safety margin is kept).
    **kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.

    Returns
    -------
    List[float]
        The best sequence found so far (as plain Python list).
    """
    np.random.seed(seed)
    random.seed(seed)

    # tiny safety margin so we don't get killed right at the deadline
    deadline = time.time() + budget_s - 0.2

    best_seq: List[float] = []
    best_score = float("inf")

    # lengths that give a good trade‑off between speed and expressiveness
    length_choices = [30, 40, 50, 60, 80, 100, 120, 150, 200]

    # ------------------------------------------------------------------ #
    # Main outer loop – keep restarting with fresh random vectors.
    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        N = random.choice(length_choices)

        # initialise a random positive vector (sum = 1)
        a = _random_sequence(N)

        # evaluate the starting point
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{time.time():.2f}] New best (init)  score={best_score:.6f}  N={N}")

        # ----------------------------------------------------------------
        # Inner optimisation – a few sub‑gradient steps + occasional jitter
        # ----------------------------------------------------------------
        step = 0.4                # initial step size for the exponential update
        max_inner = 2000          # cap to keep each restart bounded
        it = 0

        while time.time() < deadline and it < max_inner:
            # sub‑gradient descent
            a = _subgradient_step(a, step)

            # occasional random jitter to escape flat regions
            if random.random() < 0.02:
                i, j = random.sample(range(N), 2)
                delta = (random.random() - 0.5) * 0.02   # tiny mass transfer
                if a[j] > delta:
                    a[i] += delta
                    a[j] -= delta
                    a = a / a.sum()                    # re‑normalise

            # evaluate the new candidate
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{time.time():.2f}] New best          score={best_score:.6f}  N={N}")

            # adapt the step size – increase a little on improvement,
            # otherwise decay slowly.
            if cur_score < best_score * 0.999:
                step = min(step * 1.03, 0.6)
            else:

            # stop early if we have already crossed the target bound
