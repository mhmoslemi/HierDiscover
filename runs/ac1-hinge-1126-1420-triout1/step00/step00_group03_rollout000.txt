"""
Propose a good step‑function sequence.

The algorithm is based on the analytic observation that a monotone
power‑law sequence a_i = (i+1)^(-p) (i = 0 … n‑1) yields a nearly flat
auto‑convolution.  For a given length n the exponent p that balances the
large edge value a_0^2 against the interior convolution peak gives a
considerable reduction of the evaluation metric
    2 * n * max(conv(a,a)) / (sum(a)^2).

Steps performed:
1. Determine the problem size n (use the length of the provided
   ``height_sequence_1`` if available, otherwise fall back to 5000).
2. Perform a coarse grid search over exponents p ∈ [0.30, 0.90] and keep
   the best sequence.
3. Refine the exponent with a finer grid around the best p.
4. Starting from the best sequence run a short stochastic local‑search
   loop (random multiplicative perturbations + a simple projected
   gradient step) until the time budget expires.
5. Return the best sequence found.

All operations respect the non‑negativity and scaling invariance of the
objective; the sequence is never normalised because the evaluation
function is homogeneous of degree 0.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------


def generate_powerlaw_sequence(n: int, p: float) -> list[float]:
    """Return a monotone decreasing power‑law sequence a_i = (i+1)^(-p)."""
    # using a list comprehension keeps the result a plain Python list,
    # which is what ``evaluate_sequence`` expects.
    return [(i + 1) ** (-p) for i in range(n)]


def random_perturb(seq: list[float], scale: float = 0.1, n_changes: int = 5) -> list[float]:
    """Multiply a few random coordinates by a log‑normal factor."""
    new_seq = seq.copy()
    for _ in range(n_changes):
        idx = np.random.randint(len(seq))
        factor = math.exp(np.random.normal(scale=scale))
        new_seq[idx] = max(0.0, new_seq[idx] * factor)
    return new_seq


def gradient_step(seq: list[float], step: float = 0.01) -> list[float]:
    """
    One projected gradient step that tries to reduce the currently largest
    convolution entry.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # gradient of conv[k_max] w.r.t. seq_j is 2 * seq_{k_max-j} (if index in range)
    grad = np.zeros(n, dtype=float)
    for j in range(n):
        k = k_max - j
        if 0 <= k < n:
            grad[j] = 2.0 * seq[k]

    # take a step opposite to the gradient, then project onto the non‑negative orthant
    new_seq = [max(0.0, x - step * g) for x, g in zip(seq, grad)]

    # avoid the pathological case sum ≈ 0 (the evaluator returns inf there)
    s = sum(new_seq)
    if s < 1e-6:
        return seq  # fall back to the previous sequence
    # scaling does not change the objective, but we keep the same total mass
    scale_factor = sum(seq) / s
    new_seq = [x * scale_factor for x in new_seq]
    return new_seq


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value coefficient sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Determine the length of the sequence to work with.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 5000

    print(f"[INFO] Working with n = {n}")

    # ------------------------------------------------------------------
    # 1) Coarse grid search over exponent p.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_p = None

    coarse_ps = np.linspace(0.30, 0.90, 61)  # 61 candidates
    for p in coarse_ps:
        if time.time() > deadline:
            break
        seq = generate_powerlaw_sequence(n, p)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p
            print(f"[COARSE] New best p={p:.4f}  score={score:.6f}")

    if best_seq is None:  # fallback – should never happen
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 2) Refine exponent locally around the best coarse value.
    # ------------------------------------------------------------------
    if best_p is not None:
        fine_low = max(0.10, best_p - 0.04)
        fine_high = min(1.20, best_p + 0.04)
        fine_ps = np.linspace(fine_low, fine_high, 41)  # finer grid
        for p in fine_ps:
            if time.time() > deadline:
                break
            seq = generate_powerlaw_sequence(n, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p
                print(f"[FINE]   New best p={p:.4f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stochastic local search (random perturbations + gradient steps)
    # ------------------------------------------------------------------
    current_seq = best_seq
    while time.time() < deadline:
        # ----- random multiplicative perturbation -----
        cand = random_perturb(current_seq, scale=0.07, n_changes=3)
        cand_score = evaluate_sequence(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            current_seq = cand
            print(f"[PERT]   Improved to {cand_score:.6f}")

        # ----- simple gradient descent step -----
        grad_seq = gradient_step(current_seq, step=0.02)
        grad_score = evaluate_sequence(grad_seq)
        if grad_score < best_score:
            best_score = grad_score
            best_seq = grad_seq
            current_seq = grad_seq
            print(f"[GRAD]   Improved to {grad_score:.6f}")

        # occasional random restart to escape possible traps
        if np.random.rand() < 0.005:
            current_seq = [np.random.random() for _ in range(n)]

    print(f"[DONE] Final best score = {best_score:.6f}")
    return best_seq
