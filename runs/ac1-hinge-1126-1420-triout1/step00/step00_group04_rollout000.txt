def fast_eval(seq):
    # Clip negatives to zero and >1000 to 1000
    a = np.clip(np.array(seq, dtype=float), 0, 1000)
    # Ensure sum >0.01 else return large.
    sum_a = a.sum()
    if sum_a < 0.01: return np.inf
    # compute convolution using FFT to get linear conv
    n = len(a)
    size = 1 << (2*n-1).bit_length()
    # Use np.fft.rfft maybe faster
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a**2))
