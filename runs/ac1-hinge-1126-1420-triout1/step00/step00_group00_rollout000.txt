def gradient_descent_optimize(n, max_iters, deadline):
    # initialize p uniform
    p = np.full(n, 1.0/n)
    best_seq = p.copy()
    best_val = evaluate_sequence(p.tolist())
    # parameters
    alpha = 0.5  # step size
    decay = 0.995
    # loop
    for it in range(max_iters):
        # check time
        if time.time() > deadline:
            break
        conv = np.convolve(p, p)
        avg = 1.0/(2*n-1)
        # compute loss as sum squared deviation (optional) maybe not needed but used for gradient.
        # compute gradient
        u = conv - avg
        grad = 4.0 * np.correlate(u, p, mode='valid')
        # step
        p_new = p - alpha * grad
        # keep nonnegative
        p_new = np.maximum(p_new, 0.0)
        # renormalize to sum=1
        s = p_new.sum()
        if s < 1e-12:
            # reinitialize
            p_new = np.full(n, 1.0/n)
            s = 1.0
        p_new /= s
        p = p_new
        # optionally enforce momentum
        # compute evaluation
        if it % 10 == 0:
            score = evaluate_sequence(p.tolist())
            if score < best_val:
                best_val = score
                best_seq = p.copy()
        # adjust alpha
        alpha *= decay
    return best_seq.tolist(), best_val

"""
Gradient‑descent optimisation of a non‑negative step function.

We view the sequence a as a probability distribution (sum = 1).  
The self‑convolution c = a * a is the distribution of the sum of two i.i.d.
draws.  The objective that evaluate_sequence returns is

    2·n·max(c)          (n = len(a))

Thus minimising max(c) – i.e. making the convolution as flat as possible –
directly minimises the score.  The code performs an exponentiated‑gradient
descent on the loss

    L(a) = Σ_k (c_k – μ)² ,   μ = 1/(2n‑1)

which penalises deviation of the convolution from a perfectly uniform
distribution.  The update

    a_i ← a_i·exp(−η·∂L/∂a_i)

keeps the vector non‑negative; a simple renormalisation restores the sum‑1
constraint (scale invariance makes the constraint harmless for the original
objective).  Several random restarts with different lengths are tried within
the given time budget and the best sequence found is returned.
"""

import time
import numpy as np

def _exponential_gradient_step(p: np.ndarray, avg: float, eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step aiming to flatten the self‑convolution.
    """
    # self‑convolution
    conv = np.convolve(p, p)                 # length 2·n‑1
    # deviation from the uniform target μ
    dev = conv - avg
    # gradient of Σ (conv‑μ)² w.r.t. p (derived analytically)
    grad = 4.0 * np.correlate(dev, p, mode='valid')
    # step size is scaled so that the largest update is ≈ η
    max_abs = np.max(np.abs(grad))
    if max_abs == 0:
        return p
    step = eta / (max_abs + 1e-12)
    # exponentiated update (keeps positivity)
    p_new = p * np.exp(-step * grad)
    # avoid exact zeros – they would stagnate the optimisation
    p_new = np.maximum(p_new, 1e-12)
    # renormalise to a probability vector
    p_new /= p_new.sum()
    return p_new


def _optimise_length(n: int, deadline: float, max_iters: int = 2000) -> tuple[list[float], float]:
    """
    Optimise a sequence of length ``n`` using exponentiated‑gradient descent.
    Returns the best sequence (as a plain list) and its score.
    """
    avg = 1.0 / (2 * n - 1)          # target value for each convolution entry
    # start from a uniform distribution; later restarts will provide alternatives
    p = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = p.copy()
    best_score = float('inf')

    # occasional evaluation (every 10 steps) keeps the overhead low
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponentiated‑gradient step
        p = _exponential_gradient_step(p, avg, eta=0.5)

        # evaluate every few iterations
        if it % 10 == 0:
            try:
                score = evaluate_sequence(p.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = p.copy()
            except Exception:
                # the evaluation routine is robust, but we guard against unexpected errors
                pass

    # final check (in case the loop exited before the last evaluation)
    try:
        final_score = evaluate_sequence(p.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = p.copy()
    except Exception:
        pass

    return best_seq.tolist(), best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Entry point for the competition.  It searches over several problem sizes,
    runs a fast exponentiated‑gradient optimiser for each size and keeps the
    best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # lengths to explore – a mix of small, medium and large vectors
    lengths = [200, 400, 600, 800, 1000, 1200, 1400]
    np.random.shuffle(lengths)

    best_overall_seq: list[float] | None = None
    best_overall_score = float('inf')

    for n in lengths:
        # allocate a modest amount of time per length; stop early if we run out
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # we try a few random restarts for each size
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # occasionally start from a random Dirichlet vector to increase diversity
            if np.random.rand() < 0.3:
                raw = np.random.gamma(shape=0.5, size=n)   # yields a sparse start
                p0 = raw / raw.sum()
            else:
                p0 = np.full(n, 1.0 / n)

            # run the optimiser on this initial point
            seq, score = _optimise_length(n, deadline)

            if score < best_overall_score:
                best_overall_score = score
                best_overall_seq = seq
                print(f"[{time.strftime('%H:%M:%S')}] New best: {score:.6f} (len={n})")

    # Fallback – should never be reached because we always initialise at least one run
    if best_overall_seq is None:
        best_overall_seq = [1.0] * 200

    # The evaluator caps values at 1000 and removes negatives, so the raw numbers
    # can be returned directly.
    return best_overall_seq
