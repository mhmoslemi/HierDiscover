def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (v_i ≥ 0, sum v = 1)."""
    n = v.size
    if n == 0:
        return v
    # Duchi et al. 2008 projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
