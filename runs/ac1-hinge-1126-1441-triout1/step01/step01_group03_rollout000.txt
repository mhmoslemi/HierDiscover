            if score < best_score:

            if best_score < 1.45:

            idx = np.random.randint(len(a))
            original = a[idx]

            factor = 1.0 + 0.1 * (np.random.rand() - 0.5)  # ±5%
            a[idx] = max(0.0, original * factor)
            # evaluate
            try:
                score = evaluate_sequence(a.tolist())
            except Exception:
                break
            if score < best_score:
                best_score = score
                best_seq = a.tolist()

            else:
                # revert the change if it does not help
                a[i] = old

    # --------------------------------------------------------------
    # 5.  Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0, 1.0]   # trivial non‑empty sequence

    print(f"Final best score: {best_score:.6f} (n={len(best_seq)}, alpha={best_alpha})")
    return best_seq
