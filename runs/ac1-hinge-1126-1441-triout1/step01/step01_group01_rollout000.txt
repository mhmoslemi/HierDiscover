def project_simplex(v):
    # v >= any real
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Thus ensures sum=1.

"""
Simple stochastic projected‑gradient search for the step‑function heights.

The objective to minimise is

    E(a) = 2·n·max_k (a*a)_k / (sum_i a_i)^2

where “*” is the discrete convolution.  The function is homogeneous,
so we may optimise a probability vector w (w_i ≥ 0, Σw_i = 1) and finally
return the raw w – any scaling leaves the score unchanged.

A theoretical lower bound for E is 2, attained by a uniform vector.
The code therefore starts from the uniform distribution (which is the
global optimum) and performs a short stochastic projected‑gradient run.
If a better historic solution (`height_sequence_1`) exists we try it as
well.  The algorithm keeps the best sequence found before the time
budget expires and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers (no closures, no lambdas – all top‑level as required)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _metric_from_weights(w: np.ndarray) -> float:
    """
    Compute the evaluation metric for a probability vector w (Σw = 1).

    The convolution w*w is obtained via FFT for speed.
    """
    n = w.shape[0]
    size = 2 * n - 1
    fw = np.fft.fft(w, n=size)
    conv = np.real(np.fft.ifft(fw * fw))          # (w*w)_k for k = 0 … 2n‑2
    max_conv = conv.max()
    return 2.0 * n * max_conv


def _subgradient_of_max(w: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of the map w ↦ max_k (w*w)_k.
    The sub‑gradient is the average of the gradients of the constraints
    that are active (i.e. attain the maximal value).
    """
    n = w.shape[0]
    size = 2 * n - 1
    fw = np.fft.fft(w, n=size)
    conv = np.real(np.fft.ifft(fw * fw))
    t = conv.max()
    active = np.where(conv >= t - 1e-12)[0]        # indices where the max is attained
    grad = np.zeros_like(w)

    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        # derivative of Σ_i w_i w_{k-i} w.r.t. w_i is 2·w_{k-i}
        grad[idx] += 2.0 * w[k - idx]

    grad /= max(1, len(active))                  # average over the active set
    return grad

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Stochastic projected‑gradient search for a low‑score step‑function.

    The function runs for at most ``budget_s`` seconds (leaving a small safety
    margin) and returns the best sequence encountered.  The returned list
    contains non‑negative floats; its sum is forced to be 1, but any scaling
    would give the same score because the evaluation function is scale‑invariant.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a working length and initialise the current point.
    # ------------------------------------------------------------------
    n = 600                                          # moderate size – fast FFTs
    w = np.full(n, 1.0 / n)                         # uniform vector (theoretical optimum)

    best_w = w.copy()
    best_val = _metric_from_weights(best_w)

    # 2.  If a historic construction exists, try it as a starting point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

                print("Started from historic sequence – score:", best_val)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop (projected gradient + occasional jitter).
    # ------------------------------------------------------------------
    step = 0.05
    it = 0
    stagnant = 0

    while time.time() < deadline:
        # a) compute a sub‑gradient of the max‑convolution term

    # ------------------------------------------------------------------
    result = [float(v) for v in best_w.tolist()]
