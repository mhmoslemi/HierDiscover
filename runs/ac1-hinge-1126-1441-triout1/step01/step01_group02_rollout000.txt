def subgradient_minimize(n, steps, lr_schedule):
    a = np.full(n, 1.0/n) # sum=1
    best = a.copy()
    best_eval = evaluate_sequence(a)
    for t in range(steps):
        conv = np.convolve(a, a)
        max_val = conv.max()
        idxs = np.flatnonzero(conv == max_val)
        # compute subgradient g
        g = np.zeros_like(a)
        for k in idxs:
            # compute overlapping contributions
            # i from max(0, k - (n-1)) to min(n-1, k)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            for i in range(i_start, i_end+1):

                if i != j:
                    g[i] += 2 * a[j]
                else:
                    g[i] += a[i]  # from derivative of i=j term

def project_simplex(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple simulated‑annealing optimizer for the “max‑convolution” score.

The evaluation function is scale invariant:
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

Hence we can fix the ℓ¹‑norm of the step heights to 1 and only
move mass around the simplex.  Starting from a random point we repeatedly
pick two indices (i -> j) and transfer a small amount of mass.
The move is accepted if it improves the score, otherwise with a
Metropolis probability that decays with time (simulated annealing).

The algorithm runs many short independent runs (different lengths
and random seeds) until the time budget is exhausted and keeps the
best sequence found.  It works for any odd length (the uniform vector
is a stationary point for even lengths, so odd lengths are preferred).

The code uses only `numpy` and the provided `evaluate_sequence`
function; it prints occasional progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index (the last where u_i > cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _anneal_one_run(length, max_iter, rng):
    """
    Perform a single simulated‑annealing run on a vector of given length.
    The vector always lives on the simplex (sum = 1, non‑negative).
    Returns the best vector and its corresponding score.
    """
    # start from a random point on the simplex (Dirichlet(1) = uniform)
    a = rng.random(length)
    a /= a.sum()

    cur_score = evaluate_sequence(a.tolist())
    best_vec = a.copy()
    best_score = cur_score

    # exponential cooling schedule
    start_temp = 1.0
    end_temp = 1e-4

    for it in range(max_iter):
        # temperature
        t = start_temp * ((end_temp / start_temp) ** (it / max_iter))

        # pick two distinct indices
        i, j = rng.integers(0, length, size=2)
        if i == j:
            continue

        # amount of mass to move: a fraction of the donor entry
        eps = rng.random() * a[i] * 0.5  # at most half of a[i]
        if eps <= 0.0:
            continue

        # propose move
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # numerical safety – keep non‑negative
        if a_new[i] < 0.0:
            continue

        # evaluate
        new_score = evaluate_sequence(a_new.tolist())
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            a = a_new
            cur_score = new_score
            # record improvement
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()

    return best_vec, best_score


def _run_multiple_search(total_budget, rng):
    """
    Drive many short annealing runs until the time budget is exhausted.
    Returns the overall best sequence found.
    """
    best_overall_seq = None
    best_overall_score = np.inf

    # some odd lengths – odd because uniform vectors are not
    # stationary points for the max‑convolution when n is odd.
    odd_lengths = [5, 7, 9, 11, 13, 15, 21, 31, 41, 51, 61, 71, 81, 101, 151]
    # maximum number of inner iterations for a single run; larger lengths
    # need slightly fewer iterations to keep the total time reasonable.
    base_iters = 12000

    deadline = time.time() + total_budget - 0.5  # keep a tiny safety margin

    while time.time() < deadline:
        # pick a random length
        n = int(rng.choice(odd_lengths))
        max_iter = max(2000, int(base_iters * 200 / n))  # scale with n
        seq, score = _anneal_one_run(n, max_iter, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.tolist()
            print(
                f"[{time.time():.2f}] New global best: {best_overall_score:.6f} "
                f" (length={n})"
            )

        # occasionally try to improve the current best by a tiny random jitter
        if best_overall_seq is not None and rng.random() < 0.05:
            a = np.array(best_overall_seq, dtype=float)
            # jitter a few entries while keeping the simplex constraint
            for _ in range(3):
                i, j = rng.integers(0, len(a), size=2)
                if i == j:
                    continue
                delta = rng.random() * a[i] * 0.1
                a[i] -= delta
                a[j] += delta
            a = _project_onto_simplex(a)
            jitter_score = evaluate_sequence(a.tolist())
            if jitter_score < best_overall_score:
                best_overall_score = jitter_score
                best_overall_seq = a.tolist()
                print(
                    f"[{time.time():.2f}] Jitter improved best to "
                    f"{best_overall_score:.6f}"
                )

    return best_overall_seq

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(convolve(a, a)) / (sum(a) ** 2)

    The routine runs a simple simulated‑annealing optimiser on several
    odd‑length vectors (to avoid the uniform stationary point) and
    returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # First try to improve the provided SOTA construction (if present)
    try:
        # height_sequence_1 is expected to be a global variable
        init_seq = np.array(height_sequence_1, dtype=float)
        if init_seq.size > 0:
            init_seq = np.clip(init_seq, 0, 1000)
            init_seq = init_seq / init_seq.sum()  # project onto simplex
            init_score = evaluate_sequence(init_seq.tolist())
            best_seq = init_seq.tolist()
            best_score = init_score
            print(f"[init] SOTA score = {best_score:.6f}")
        else:
            best_seq = None
            best_score = np.inf
    except Exception:
        best_seq = None
        best_score = np.inf

    # Run the stochastic search for the remaining time
    remaining_budget = max(0.0, budget_s - 2.0)  # keep a small buffer
    candidate = _run_multiple_search(remaining_budget, rng)

    # Compare with the SOTA start if it existed
    if candidate is not None:
        cand_score = evaluate_sequence(candidate)
        if cand_score < best_score:
            best_seq = candidate
            best_score = cand_score
            print(f"[final] New best from search = {best_score:.6f}")

    # As a safety net, if nothing was found, fall back to a tiny uniform vector
    if best_seq is None:
        best_seq = [1.0 / 3] * 3  # any valid vector with sum > 0.01
        print("[fallback] using tiny uniform vector")

    return best_seq
