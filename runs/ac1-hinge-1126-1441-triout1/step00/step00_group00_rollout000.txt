"""
Propose a (hopefully) good step‐function height sequence.

The algorithm is a multi‑restart stochastic local search that combines
   * simple shape initialisers (uniform, triangular, sine, exponential, or the
     known SOTA sequence),
   * a fast “gradient‑like” move that shifts a little mass from indices that
     contribute most to the current maximum convolution value to indices that
     contribute the least,
   * occasional linear‑programming refinements (the same LP used by the
     reference implementation),
   * random perturbations for escaping plateaus.

All candidates are normalised so that
      sum(a_i) = sqrt(2·n)    (the normalisation used inside ``evaluate_sequence``).

The routine runs until the allotted time budget (default 1000 s, a few seconds
in practice) expires and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _normalise(seq):
    """Scale a sequence so that its L1‑norm equals sqrt(2·len(seq))."""
    seq = np.asarray(seq, dtype=float)
    s = seq.sum()
    if s == 0:
        seq = np.ones_like(seq)
        s = seq.sum()
    target = np.sqrt(2.0 * len(seq))
    seq = seq * (target / s)
    # protect against the bounds used in evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    return seq.tolist()


def _random_sequence(L):
    """Uniform random values on [0,1] then normalised."""
    return _normalise(np.random.rand(L))


def _triangular_sequence(L):
    """Symmetric triangular shape (0 → 1 → 0)."""
    i = np.arange(L, dtype=float)
    seq = 1.0 - np.abs(2.0 * i / (L - 1) - 1.0)   # peak = 1 in the middle
    return _normalise(seq)


def _sine_sequence(L, power=1.0):
    """Half‑sine shape, optionally raised to a power."""
    i = np.arange(L, dtype=float) + 0.5
    seq = np.sin(np.pi * i / (L + 1.0)) ** power
    return _normalise(seq)


def _exponential_sequence(L, r=0.95):
    """Geometric decay r**i (0<r<1)."""
    seq = np.power(r, np.arange(L, dtype=float))
    return _normalise(seq)


# ----------------------------------------------------------------------
# Linear‑programming step (identical to the reference code)
# ----------------------------------------------------------------------
def _solve_convolution_lp(f_seq, rhs):
    """
    Linear program:
        maximise   sum(b)
        s.t.       conv(f_seq, b) <= rhs   (elementwise)
                   b >= 0
    Returns the optimal b as a NumPy array, or None if LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints:
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  (‑I·b <= 0)
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # A modest time limit keeps the outer search responsive.
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs',
                  options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None


def _get_good_direction_to_move_into(seq):
    """
    LP based improvement as described in the statement.
    Returns a new sequence (list) or None if the LP fails.
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    # Normalise to the convention required by the LP.
    norm_seq = np.array(seq, dtype=float) * np.sqrt(2.0 * n) / sum_seq
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = _solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    # Rescale g to the same L1‑norm as the original sequence.
    g = np.array(g, dtype=float) * np.sqrt(2.0 * n) / sum_g
    # Small convex combination step (t=0.01) as in the reference.
    t = 0.01
    new_seq = (1.0 - t) * np.array(seq, dtype=float) + t * g
    return _normalise(new_seq)


# ----------------------------------------------------------------------
# Local moves
# ----------------------------------------------------------------------
def _gradient_move(seq):
    """
    Move a little mass from the index that contributes most to the current
    maximum convolution value to the index that contributes least.
    Returns a new normalised list or None if no move is possible.
    """
    a = np.asarray(seq, dtype=float)
    n = len(a)
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    # Weight of each position i in the current max entry = a_{kmax-i}
    w = np.zeros(n, dtype=float)
    for i in range(n):
        j = kmax - i
        if 0 <= j < n:
            w[i] = a[j]

    # Choose donor (largest weight) and receiver (smallest weight)
    i_donor = int(np.argmax(w))
    i_receiver = int(np.argmin(w))
    if i_donor == i_receiver:
        # pick a random other receiver
        i_receiver = np.random.randint(n)
        while i_receiver == i_donor:
            i_receiver = np.random.randint(n)

    # Move a modest fraction of the donor's value.
    delta = min(a[i_donor], max(1e-8, 0.01 * a[i_donor]))
    if delta <= 0:
        return None

    b = a.copy()
    b[i_donor] -= delta
    b[i_receiver] += delta
    # Numerical safety – clip tiny negatives.
    b = np.clip(b, 0.0, None)
    return _normalise(b)


def _random_perturb(seq):
    """Add a tiny Gaussian perturbation to a random coordinate and renormalise."""
    a = np.asarray(seq, dtype=float)
    n = len(a)
    i = np.random.randint(n)
    # Scale the perturbation to the current magnitude.
    scale = max(1e-6, 0.02 * a[i])
    delta = np.random.randn() * scale
    a[i] = max(0.0, a[i] + delta)
    # Renormalise to keep the L1‑norm fixed.
    return _normalise(a)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine performs many random restarts,
    each followed by a short local optimisation that mixes:

        – a cheap gradient‑like move,
        – the LP refinement from the reference solution,
        – tiny random perturbations.

    The best sequence encountered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + max(0.0, budget_s - 5.0)   # reserve a few seconds for cleanup

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # Prepare a pool of initial shapes and length choices
    # --------------------------------------------------------------
    length_choices = [256, 512, 1024]          # small enough for fast conv. ops
    init_modes = ['random', 'triangular', 'sine', 'exp', 'sota']

    # Helper to build an initial sequence of given length and mode
    def _initial_sequence(L, mode):
        if mode == 'random':
            return _random_sequence(L)
        if mode == 'triangular':
            return _triangular_sequence(L)
        if mode == 'sine':
            power = np.random.uniform(0.5, 2.0)
            return _sine_sequence(L, power=power)
        if mode == 'exp':
            r = np.random.uniform(0.85, 0.98)
            return _exponential_sequence(L, r=r)
        if mode == 'sota' and 'height_sequence_1' in globals():
            # Take a random contiguous window (or zero‑pad) from the known SOTA.
            full = np.asarray(height_sequence_1, dtype=float)
            if len(full) >= L:
                start = np.random.randint(len(full) - L + 1)
                sub = full[start:start + L]
                return _normalise(sub)
            else:
                # Pad with zeros if the SOTA is shorter.
                padded = np.concatenate([full, np.zeros(L - len(full))])
                return _normalise(padded)
        # fallback
        return _random_sequence(L)

    # --------------------------------------------------------------
    # Main stochastic search loop
    # --------------------------------------------------------------
    while time.time() < end_time:
        # ---- pick a random problem size and start mode -----------------
        L = int(np.random.choice(length_choices))
        mode = np.random.choice(init_modes)
        seq = _initial_sequence(L, mode)

        # ---- local optimisation (bounded by a modest iteration count) -
        cur_seq = seq
        cur_score = evaluate_sequence(cur_seq)

        # The inner loop stops either when we are out of time or after a
        # fixed budget of cheap iterations.  The budget is deliberately low
        # for the large lengths we explore; most progress comes from the
        # many cheap restarts.
        inner_iters = 0
        max_inner = 1500   # adjustable; kept low to stay within the global budget
        while time.time() < end_time and inner_iters < max_inner:
            inner_iters += 1
            choice = np.random.rand()
            if choice < 0.55:
                # gradient‑like move
                new_seq = _gradient_move(cur_seq)
            elif choice < 0.85:
                # LP refinement (expensive but powerful)
                new_seq = _get_good_direction_to_move_into(cur_seq)
            else:
                # random perturbation
                new_seq = _random_perturb(cur_seq)

            if new_seq is None:
                continue

            new_score = evaluate_sequence(new_seq)
            # accept only improving moves (strictly smaller score)
            if new_score < cur_score:
                cur_seq = new_seq
                cur_score = new_score

                # update the global best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time():.1f}s] New best: {best_score:.6f} (len={L}, mode={mode})")

        # End of this restart – go for another random seed when time permits
        # (the outer loop will automatically start a new restart).

    # ------------------------------------------------------------------
    # Fallback: if nothing better was found, return a uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        L = 256
        best_seq = _normalise([1.0] * L)
        best_score = evaluate_sequence(best_seq)

    print(f"Finished search – best score {best_score:.6f} with length {len(best_seq)}")
    return best_seq
