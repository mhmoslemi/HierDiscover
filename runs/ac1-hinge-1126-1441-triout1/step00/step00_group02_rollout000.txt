def propose_candidate(seed=42, budget_s=1000, n=None):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave margin
    best_seq = None
    best_score = np.inf

    while time.time() < deadline:
        # choose n randomly each restart
        if n is None:
            n_cur = np.random.randint(800, 2000)  # moderate size
        else:
            n_cur = n
        a = init_sequence(n_cur)
        # maybe also random perturb
        a += np.random.rand(n_cur) * 0.01
        a = np.maximum(a, 0)
        a = a / a.sum()
        score = compute_score(a)

        # local search
        iters = 0
        # schedule step size reduces over time
        step_factor = 0.1
        while time.time() < deadline and iters < 2000:
            # compute convolution and max idx
            b = np.convolve(a, a)
            max_idx = int(np.argmax(b))
            # find i indices for which contributions to max are large
            # get i in range
            i_min = max(0, max_idx - (n_cur - 1))
            i_max = min(n_cur - 1, max_idx)
            # find i that has max product a[i] * a[max_idx - i]
            # compute products
            idx_range = np.arange(i_min, i_max+1)
            j_range = max_idx - idx_range
            # ensure j_range in [0,n-1]
            # compute product contributions
            contrib = a[idx_range] * a[j_range]
            # pick i with max contribution
            i_sel = idx_range[np.argmax(contrib)]
            # choose target index far = 0 if i_sel not 0 else n_cur-1
            target = 0 if i_sel > n_cur//2 else n_cur-1
            if i_sel == target:
                target = n_cur-1 if target==0 else 0
            # amount to move
            delta = step_factor * a[i_sel]
            if delta < 1e-9:
                # break maybe
                break
            # ensure a[i_sel] - delta >= 0
            delta = min(delta, a[i_sel])
            # propose new a
            a_new = a.copy()
            a_new[i_sel] -= delta
            a_new[target] += delta
            # compute new score
            score_new = compute_score(a_new)
            # accept if better
            if score_new < score:
                a = a_new
                score = score_new
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"New best: {best_score:.6f} n={n_cur}")
            else:
                # maybe accept with small probability or occasionally random move
                if np.random.rand() < 0.001:
                    a = a_new
                    score = score_new
            iters += 1

        # after local search maybe random restart
        # continue outer loop

    return best_seq.tolist() if best_seq is not None else [0.0]

def project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    # using algorithm by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max {j | u_j - cssv_j / j > 0}
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate

A fast simplex‐projected sub‑gradient optimiser for the
self‑convolution bound.

The objective to be minimised is

    f(a) = 2·n·max_k (a*a)[k] / (∑a_i)² .

Since the ratio is homogeneous we can fix ∑a_i = 1 and simply
minimise 2·n·max_k (a*a)[k].

The problem is convex: each convolution entry is a convex quadratic
form in a and we minimise the maximum of these forms over the
probability simplex.  A classic sub‑gradient method therefore works:

* compute the convolution,
* locate the index k* where the maximum occurs,
* the sub‑gradient of this maximal quadratic form is
      g_i = 2·a_{k*‑i}  (if the index is inside the support, else 0),
* take a step a ← a − η·g,
* project the result onto the simplex {a ≥ 0, ∑a = 1}.

Uniform weights are a stationary point (the gradient is constant), so
the optimiser adds small random “jitter” moves every few hundred
iterations to break symmetry and to explore the space.  Several
restarts with different lengths and initial shapes are performed until
the time budget expires.  The best sequence found is returned.

The implementation uses only NumPy and pure‑Python code, runs well
inside the 1000 s budget and respects the non‑negative / ≤1000 bounds
required by `evaluate_sequence`.
"""

import time
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (v_i ≥ 0, sum v = 1)."""
    n = v.size
    if n == 0:
        return v
    # Duchi et al. 2008 projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse starting point on the simplex."""
    # mix a few analytic shapes with a tiny random perturbation
    shape = rng.choice(["uniform", "ends", "inverse"])
    i = np.arange(n, dtype=float)

    if shape == "uniform":
        a = np.ones(n)
    elif shape == "ends":
        # larger weights near the two ends, small in the middle
        a = 1.0 / (i + 1) + 1.0 / (n - i)
    else:  # "inverse"
        # a smooth U‑shaped distribution
        a = 1.0 / (np.sqrt(i + 0.5) * np.sqrt(n - i - 0.5) + 1e-12)

    # add a tiny random jitter (helps to escape symmetry)
    a += rng.random(n) * 1e-3
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function (non‑negative heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # a tiny safety margin

    best_seq = None
    best_score = np.inf

    # a few different lengths – larger n gives a tighter theoretical lower bound
    lengths = [1200, 2000, 2800, 3600]

    # sub‑gradient parameters
    step0 = 1.0          # initial step size
    jitter_interval = 250  # iterations between random perturbations

    for n in lengths:
        if time.time() >= deadline:
            break

        # several restarts per length
        for restart in range(3):
            if time.time() >= deadline:
                break

            a = _initial_distribution(n, rng)

            # main optimisation loop
            it = 0
            while time.time() < deadline:
                it += 1

                # convolution and maximal entry
                conv = np.convolve(a, a)
                kmax = int(conv.argmax())
                max_val = conv[kmax]
                # because ∑a = 1 the objective is simply 2·n·max_val
                cur_score = 2.0 * n * max_val

                # keep the best seen so far
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.strftime('%H:%M:%S')}] "
                        f"new best: {best_score:.6f}  n={n}  it={it}"
                    )

                # ---- sub‑gradient step ----
                # gradient of the active quadratic form a^T M_kmax a
                grad = np.zeros_like(a)

                i_min = max(0, kmax - (n - 1))
                i_max = min(n - 1, kmax)
                idx = np.arange(i_min, i_max + 1)
                # g_i = 2 * a_{kmax-i}
                grad[idx] = 2.0 * a[kmax - idx]

                # diminishing step size
                eta = step0 / np.sqrt(it)

                a_candidate = a - eta * grad
                # projection back onto the simplex
                a = _project_to_simplex(a_candidate)

                # ---- occasional random jitter to break symmetry ----
                if it % jitter_interval == 0:
                    i_src = rng.integers(n)
                    i_dst = rng.integers(n)
                    if i_src != i_dst and a[i_src] > 1e-12:
                        # move a small fraction of mass from i_src to i_dst
                        delta = 0.05 * a[i_src]   # 5 % of the source weight
                        a[i_src] -= delta
                        a[i_dst] += delta
                        a = _project_to_simplex(a)

                # safety: prevent the algorithm from stagnating forever
                if it >= 5000:
                    break

    # If nothing was found (unlikely), fall back to a tiny uniform vector
    if best_seq is None:
        best_seq = np.ones(10) / 10.0

    return best_seq.tolist()
