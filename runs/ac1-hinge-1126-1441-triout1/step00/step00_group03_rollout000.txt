def grad_max(a, beta):
    c = np.convolve(a, a)
    max_c = c.max()
    # subtract max for stability
    exp_terms = np.exp(beta * (c - max_c))
    Z = exp_terms.sum()
    # Gradient (dL/da) = (2/(beta * Z)) * correlation of a with exp_terms
    corr = np.correlate(a, exp_terms, mode='full')
    start = len(exp_terms) - 1
    grad = (2.0 / (beta * Z)) * corr[start:start + len(a)]
    return grad

Pseudo:

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):

"""
Optimise a non‑negative step function for the given objective.

The algorithm works directly on the probability simplex (sum = 1,
non‑negative).  For a fixed length *n* the evaluation score
`evaluate_sequence` simplifies to

        score = 2 * n * max(conv(a, a))

because the sum of the coefficients is 1 and the clipping performed in
`evaluate_sequence` never activates for the values we generate.
Thus we minimise the L∞‑norm of the self‑convolution.

A smooth approximation of the max operator (log‑sum‑exp) is used to
obtain a gradient that can be followed by projected (onto the simplex)
gradient descent.  The smoothing parameter β is increased gradually,
making the objective tighter.  Several random restarts with different
support sizes are performed until the time budget is exhausted.
The best sequence found is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex
# ------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum x = 1}."""
    # based on the algorithm by Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.max(np.where(cond)[0])
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    minimises `evaluate_sequence`.  The function runs for up to
    `budget_s` seconds and uses projected gradient descent on a smooth
    surrogate of the L∞‑norm of the self‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # --------------------------------------------------------
    # Determine the working length
    # --------------------------------------------------------
    try:
        # `height_sequence_1` is provided by the host environment
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2620                       # fallback length if the variable is missing

    # --------------------------------------------------------
    # Utility: cheap score (identical to evaluate_sequence when sum=1)
    # --------------------------------------------------------
    def cheap_score(a: np.ndarray) -> float:
        return 2.0 * n * np.max(np.convolve(a, a))

    # --------------------------------------------------------
    # Keep the best sequence found so far
    # --------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------
    # Optionally start from the known construction
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array([float(x) for x in height_sequence_1], dtype=float)  # type: ignore[name-defined]
        seq0 = np.maximum(seq0, 0.0)
        s = seq0.sum()
        if s < 0.01:                     # degenerate – fall back to uniform
            seq0 = np.ones_like(seq0) / seq0.size
        else:
            seq0 = seq0 / s               # normalise to sum = 1
        sc0 = cheap_score(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"Initial sequence (provided) score = {best_score:.6f}")

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ----------------------------------------------------
        # Random sparse initialisation
        # ----------------------------------------------------
        # pick a support size around sqrt(n) (which is a good trade‑off)
        k_min = max(40, int(0.6 * np.sqrt(n)))
        k_max = max(k_min + 1, int(1.4 * np.sqrt(n)))
        k = np.random.randint(k_min, k_max + 1)

        # generate k positive weights, the rest are zero
        a = np.random.exponential(scale=1.0, size=n)
        # keep only the k largest entries -> sparse support
        idx = np.argpartition(a, -k)[-k:]
        sparse = np.zeros_like(a)
        sparse[idx] = a[idx]
        a = sparse / sparse.sum()          # project onto the simplex

        # ----------------------------------------------------
        # Local optimisation: projected gradient descent on smoothed max
        # ----------------------------------------------------
        beta = 5.0                         # smoothing parameter (log‑sum‑exp)
        step = 0.2                         # initial step size
        no_improve_cnt = 0
        inner_iter = 0

        while time.time() < deadline and inner_iter < 3000:
            inner_iter += 1

            # current convolution and score
            conv = np.convolve(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max    # cheap_score(a)

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[restart {restart} it {inner_iter}] New best score = {best_score:.6f}")

                # stop early if we already beat the target
                if best_score <= 1.503:
                    print("Target reached – terminating early.")
                    return best_seq.tolist()

            # ------------------------------------------------
            # Gradient of the smoothed max (log‑sum‑exp)
            # ------------------------------------------------
            # shift by the maximum for numerical stability
            shifted = beta * (conv - cur_max)
            exp_terms = np.exp(shifted)      # values in (0, 1]
            Z = exp_terms.sum()
            # correlation a with exp_terms gives sum_j a_j * exp_{i+j}
            corr = np.correlate(a, exp_terms, mode="full")
            start = exp_terms.size - 1
            grad = (2.0 / (beta * Z)) * corr[start:start + n]

            # ------------------------------------------------
            # Gradient step + projection onto the simplex
            # ------------------------------------------------
            a_new = a - step * grad
            a_new = project_onto_simplex(a_new)

            # quick evaluation of the trial step
            new_score = cheap_score(a_new)

            # accept / reject logic (simple back‑tracking)
            if new_score < cur_score:
                a = a_new
                step = min(step * 1.05, 1.0)   # modestly increase step
                no_improve_cnt = 0
            else:
                step *= 0.5
                no_improve_cnt += 1

            # if we stagnate for many iterations, restart
            if no_improve_cnt >= 50:
                break

            # slowly tighten the smoothing (increase β)
            if inner_iter % 200 == 0:
                beta = min(beta * 1.3, 200.0)

        # end of one restart
        # occasional random perturbation to escape poor basins
        if time.time() < deadline and np.random.rand() < 0.2:
            eps = 0.02 * np.random.randn(n)
            a = a + eps
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            # small improvement attempt after perturbation
            step = 0.2
            beta = 5.0

    # ----------------------------------------------------
    # Return the best sequence found (as a plain list)
    # ----------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (guaranteed to be valid)
        best_seq = np.ones(n, dtype=float) / n

    return best_seq.tolist()
