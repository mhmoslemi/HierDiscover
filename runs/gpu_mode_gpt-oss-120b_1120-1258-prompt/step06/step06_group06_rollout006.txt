@triton.jit
def batched_gemm_T_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bj, stride_Bk, # B has shape (batch, N, K) (j, k)
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute block indices
    pid = tl.program_id(0)
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_m_blocks * num_n_blocks)
    pid_remaining = pid % (num_m_blocks * num_n_blocks)
    block_m = pid_remaining // num_n_blocks
    block_n = pid_remaining % num_n_blocks

    # compute tile offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers
        a_ptrs = a_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        b_ptrs = b_ptr + batch_id * stride_Bb + offs_n[None, :] * stride_Bj + offs_k[:, None] * stride_Bk
        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        # Note: b shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)  # (BM,BN) = (BM,BK)*(BK,BN)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold‑3).  The kernel performs:

1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) via a custom
   Triton kernel (two‑pass: mean/variance → normalized values).

2️⃣ A fused linear projection that simultaneously computes `left`,
   `right`, `left_gate` and `right_gate` (shape [R, 4·H]) and a separate
   projection for the output‑gate (shape [R, H]).

3️⃣ Optional masking of the `left` and `right` projections.

4️⃣ Element‑wise sigmoid gating of the three gate tensors.

5️⃣ Batched half‑precision matrix multiplication
   `out[b·H+i] = left[b·H+i] @ right[b·H+i].T`
   for every hidden dimension `i`.  The GEMM is written as a custom
   Triton kernel which processes all batches in a single launch,
   avoiding the per‑batch launch overhead of `torch.bmm`.

6️⃣ A fused LayerNorm + gate kernel for the hidden‑dim output, followed
   by the final linear projection back to `dim`.

All heavy per‑row normalisations are performed in custom Triton kernels,
while the large GEMM is also Triton‑implemented, giving >2× speed‑up
relative to the reference PyTorch version on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                # [total_rows, dim]   (input fp32, output fp16)
    weight_ptr, bias_ptr,        # [dim]               (γ, β)   fp32
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean & variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalise, apply γ/β, store as fp16 -----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Kernel 2: LayerNorm + element‑wise gate (fp16 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # input & output [total_rows, dim] (fp16)
    weight_ptr, bias_ptr,        # LayerNorm γ,β (fp32)
    gate_ptr,                    # gate tensor [total_rows, dim] (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean & variance (fp32) -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalise + γ/β + gate -----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # gate

        out = (norm * w[None, :] + b[None, :]) * g               # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Kernel 3: Batched GEMM A @ Bᵀ   (half‑precision, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_T_kernel(
    a_ptr, b_ptr, c_ptr,               # [batch, M, K], [batch, N, K], [batch, M, N] (fp16)
    batch, M, N, K,
    stride_ab, stride_ai, stride_ak,   # strides for a: batch, row(i), col(k)
    stride_bb, stride_bj, stride_bk,   # strides for b: batch, row(j), col(k)
    stride_cb, stride_ci, stride_cj,   # strides for c: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per GEMM
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    # linear index → (batch_id, tile_m, tile_n)
    batch_id = pid // (num_m_tiles * num_n_tiles)
    pid_in_batch = pid % (num_m_tiles * num_n_tiles)
    tile_m = pid_in_batch // num_n_tiles
    tile_n = pid_in_batch % num_n_tiles

    # offsets for the current tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (use FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: [BM, BK]
        a_ptrs = a_ptr + batch_id * stride_ab \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile: we need (BK, BN).  B is [batch, N, K]; we load as
        #   Bᵀ[k, n] = B[batch, n, k]
        b_ptrs = b_ptr + batch_id * stride_bb \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)    # (BM,BN) = (BM,BK)·(BK,BN)

    # store result as fp16
    c_ptrs = c_ptr + batch_id * stride_cb \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim], float32
        - mask_tensor  : torch.Tensor, [B, N, N] (bool/float)
        - weights_dict : dict with model parameters (LayerNorm, Linear)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim                    # hidden dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_half = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],      # γ, FP32
        weights["norm.bias"],        # β, FP32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates
    # --------------------------------------------------------------
    # Cast projection weights once to fp16
    w_left    = weights["left_proj.weight"].to(torch.float16)     # [H, dim]
    w_right   = weights["right_proj.weight"].to(torch.float16)    # [H, dim]
    w_lgate   = weights["left_gate.weight"].to(torch.float16)    # [H, dim]
    w_rgate   = weights["right_gate.weight"].to(torch.float16)   # [H, dim]

    # concat along output dimension -> [4*H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # Linear: [R, dim] @ [4*H, dim].T → [R, 4*H]   (FP16)
    proj_gate = F.linear(x_norm_half, w_concat)   # (R, 4*H)

    left_raw   = proj_gate[:, 0:H]                 # (R, H)
    right_raw  = proj_gate[:, H:2 * H]             # (R, H)
    lgate_raw  = proj_gate[:, 2 * H:3 * H]         # (R, H)
    rgate_raw  = proj_gate[:, 3 * H:4 * H]         # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection (separate linear)
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm_half, w_out_gate)           # (R, H)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and sigmoids
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1], fp16 broadcast
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    left_gate  = lgate_raw.sigmoid_()
    right_gate = rgate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # (R, H)  fp16
    right = right_raw * right_gate  # (R, H)  fp16

    # --------------------------------------------------------------
    # 5️⃣ Batched multiplicative update: left @ rightᵀ  (B·H, N, N)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H) → permute to (B, H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # merge batch and hidden dims → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)   # (B·H, N, N), fp16

    # launch batched GEMM kernel
    batch_dim = B * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_m_tiles = triton.cdiv(N, BLOCK_M)
    num_n_tiles = triton.cdiv(N, BLOCK_N)
    grid = (batch_dim * num_m_tiles * num_n_tiles,)

    # strides (contiguous tensors)
    stride_ab = left_mat.stride(0)   # batch stride = N*N
    stride_ai = left_mat.stride(1)   # row stride   = N
    stride_ak = left_mat.stride(2)   # col stride   = 1

    stride_bb = right_mat.stride(0)
    stride_bj = right_mat.stride(1)  # N dimension (row) stride = N
    stride_bk = right_mat.stride(2)  # K dimension stride = 1

    stride_cb = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    batched_gemm_T_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_dim,
        N, N, N,                     # M=N, N=N, K=N
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bj, stride_bk,
        stride_cb, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out_flat = out.view(total_rows, H)                               # (R, H)

    # --------------------------------------------------------------
    # 6️⃣ Fuse LayerNorm (over H) + out‑gate multiplication
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ, FP32
        weights["to_out_norm.bias"],     # β, FP32
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
