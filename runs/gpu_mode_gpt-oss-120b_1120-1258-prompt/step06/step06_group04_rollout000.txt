"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32→FP16) with a custom Triton
    kernel.  The kernel works on a 2‑D view (rows = B·N·N, cols = dim) and
    performs a two‑pass mean/variance reduction.

2️⃣ Fuse each (proj + gate) pair for the left/right branches:
        (dim → hidden) + (dim → hidden) → (dim → 2·hidden) via a single
    `torch.nn.functional.linear`.  Split the result, apply the sigmoid
    gate and (optionally) the mask.

3️⃣ Compute the tri‑multiplicative update
        out_h(i,j) = Σ_k left_h(i,k)·right_h(j,k)
   by reshaping the hidden dimension into the batch axis and using a
   single batched `torch.bmm`.  This is the most compute‑intensive step
   (B·H independent N×N GEMMs) and runs in FP16 on Tensor‑Cores.

4️⃣ Second LayerNorm (over hidden_dim) – same Triton kernel as step 1.

5️⃣ Multiply by the output gate, apply the final linear projection
   (hidden → dim) and cast the result back to FP32.

All heavy arithmetic is performed in FP16; the final result matches the
reference PyTorch module’s dtype (float32).  The implementation keeps
the memory layout dense and avoids any Python‑level loops, enabling
high occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor[float32]  shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be omitted)
        * weights_dict : mapping str → torch.Tensor containing all model
                         parameters (see docstring of the problem statement)
        * config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (default True)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Block sizes chosen to cover dim ≤ 384 efficiently
    BLOCK_M = 128                # rows per program
    BLOCK_D = 64                 # columns per program
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # [B, N, N, dim]  (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left/right branches
    # ------------------------------------------------------------------
    # Concatenate (proj, gate) weights for a single linear call
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    rows = total_rows
    # Linear: (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat = F.linear(x_norm.view(rows, dim), left_w)   # FP16
    right_concat = F.linear(x_norm.view(rows, dim), right_w)

    # Split into projection & pre‑gate
    left_proj = left_concat[:, :hidden_dim]          # (rows, H)
    left_gate_pre = left_concat[:, hidden_dim:]      # (rows, H)
    right_proj = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Output‑gate (shared across the three‑tensor dimension)
    out_gate = torch.sigmoid(
        F.linear(
            x_norm.view(rows, dim),
            weights["out_gate.weight"]
            .to(dtype=torch.float16, device=device, non_blocking=True)
        )
    )   # (rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16,
                                         device=device,
                                         non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)
    right_h = right_proj * right_gate       # (rows, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)      # FP16
    right = right_h.view(B, N, N, hidden_dim)    # FP16

    # Merge H into batch → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: (B·H, N, N) × (B·H, N, N)ᵀ   → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate + final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)       # FP16
    gated = out_norm * out_gate                         # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"]
        .to(dtype=torch.float16, device=device, non_blocking=True)
    )   # (rows_h, dim)   FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
```

**Explanation of key optimizations**

- **Two‑pass LayerNorm kernel** – computes mean and variance with vectorised loads, then writes the normalised result in FP16, avoiding any Python loops.
- **Fused projection+gate** – a single `F.linear` per side produces both the projected features and the pre‑gate values, halving the number of matrix multiplies.
- **Mask fusion** – applied once after projection, broadcasting over the hidden dimension.
- **Batched GEMM** – by folding the hidden dimension into the batch axis we call `torch.bmm` only `B·hidden_dim` times; each call is a dense N×N matrix multiply that fully exploits Tensor‑Core FP16 performance.
- **Second LayerNorm** – re‑uses the same Triton kernel.
- **All heavy arithmetic in FP16** – data stays in half‑precision until the final cast, maximising throughput on H100 while preserving the required FP32 output.

The implementation satisfies the requirement of using at least one Triton kernel, works for all configurations listed in the problem statement, and is written as a single, self‑contained Python block ready for integration.
