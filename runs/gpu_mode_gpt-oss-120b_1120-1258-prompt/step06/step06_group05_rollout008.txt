import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     total_rows, dim, eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # accumulate sums
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D) # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to element
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M, D]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        y = norm * w[None, :] + b[None, :]
        y = y.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, y,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of outgoing TriMul (AlphaFold-3) using Triton kernels for both LayerNorms.
    ...
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # 1st LayerNorm
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128  # can tune
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # fused projections + gates: concatenated weights
    # convert to half for speed
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)  # [4*H, dim]

    # Linear: x_norm (FP16) @ W^T (FP16) => (R, 4*H) FP16
    proj_gate = F.linear(x_norm, w_concat)  # bias None

    H = hidden_dim
    left_proj_raw = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # output gate projection
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)
    )  # shape [R, H]

    # Mask handling
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gating
    left = left_proj_raw * left_gate  # [R, H]
    right = right_proj_raw * right_gate  # [R, H]

    # Reshape for batch matmul
    left = left.view(B, N, N, H)   # [B,N,N,H]
    right = right.view(B, N, N, H)  # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, H)

    # Bring head dimension before N,N for batched GEMM
    # We want: for each batch and head, compute left[b, i, :, h] @ right[b, j, :, h]^T?
    # In reference einsum, left[..., i, k, d] and right[..., j, k, d] -> sum over k
    # left shape after view: [B, N, N, H] corresponds to left[b,i,k,h]
    # right shape: right[b,j,k,h]
    # For GEMM per head, we need left_mat size [B*H, N, N] representing left rows = i dimension? Actually left is (i,k) matrix; right is (j,k). We need out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    # That's equivalent to left_mat[b*H + h] = left[b,:, :, h] (N,N)
    # right_mat = right[b,:, :, h] (N,N)
    # out = left_mat @ right_mat.transpose(-1, -2) gives (N,N) result per head.
    # So we can transpose right before bmm, as done earlier:
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [BH, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [BH, N, N]

    # Matmul using bmm; torch.bmm expects (b, m, k) * (b, k, n)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH, N, N] half

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 2nd LayerNorm over hidden dim H
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)
    )  # shape [R, dim], half

    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim) – Triton kernel.
2️⃣  Fuse the four linear maps (left/right projections + gates) into a single
    half‑precision GEMM:  X_norm @ W_concatᵀ → [R, 4·H].
3️⃣  Separate the result into left_proj, right_proj, left_gate, right_gate.
4️⃣  Compute the output‑gate with a second half‑precision GEMM.
5️⃣  (Optional) apply the pair‑wise mask, then sigmoid the three gates.
6️⃣  Form gated tensors  left = left_proj * left_gate,  right = right_proj * right_gate.
7️⃣  Tri‑multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision GEMM (torch.bmm) over the head dimension.
8️⃣  Row‑wise LayerNorm over the hidden dimension H – Triton kernel.
9️⃣  Multiply by the output‑gate, project back to `dim` with a final linear layer,
    and cast the result to float32 (the reference dtype).

Only the two LayerNorms are written in Triton; all large matrix multiplications
use cuBLAS/Tensor‑Cores via `torch.nn.functional.linear` and `torch.bmm`,
which is already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)
    bias_ptr,       # const   *[dim]                (float32)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β

    Input  : float32, Output : float16 (fp16 for downstream ops).
    Accumulation is performed in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- 1️⃣ mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)                 # [BLOCK_D]
        mask_d = offs_d < dim
        # address of x[row, col]
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]        # [M,D]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                         # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- 2️⃣ write normalized values γ·(x‑μ)·rstd + β (fp16) ---------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ, fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β, fp32
        out = norm * w[None, :] + b[None, :]          # fp32

        out_fp16 = out.to(tl.float16)
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3) using Triton
    for the two LayerNorms and PyTorch/Tensor‑Cores for all GEMMs.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping with all Linear/LayerNorm parameters.
                         All tensors are float32.
        - config_dict  : must contain "dim", "hidden_dim", "nomask".

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack --------------------------------------------------------
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise operations

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) -----------------------------
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 32           # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates -----------------------
    # --------------------------------------------------------------
    H = hidden_dim

    # concatenate weights for the four linear maps (shape: [4·H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # fused GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_concat)   # no bias

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16, shape [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, apply gating -------------------------
    # --------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16) and broadcast
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate      # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM) -------------------
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # reshape for bmm: bring head dim into batch dimension
    # left_mat[b*H + h]  = left[b, :, :, h]  -> shape (N, N)
    # right_mat[b*H + h] = right[b, :, :, h] -> shape (N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [BH, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [BH, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim (fp16) --------------------
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate + final linear projection ----------------
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    # final projection back to `dim`
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
