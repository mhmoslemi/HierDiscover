"""
Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
0️⃣  Unpack inputs.
1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] input (float32 → float16) using a
    custom Triton kernel (no extra synchronisation, fused mean/var).
2️⃣  Fuse the four linear projections/gates (left/right + left/right‑gate) into a
    single half‑precision GEMM:
        X_norm_half @ W_proj_gate.T  →  [R, 4·H]  (R = B·N·N).
   Split the result into the four tensors and apply a sigmoid on the two gate
   tensors (in‑place).  Apply the optional mask and the gates to obtain the
   gated tensors *left* and *right* (still fp16).
3️⃣  Compute the output‑gate with a second half‑precision GEMM.
4️⃣  Reshape *left*, *right* and *out_gate* to (B, H, N, N) (no extra copies) and
    perform the N³ multiplicative update as a batched GEMM:
        out[b, h] = left[b, h] @ right[b, h]ᵀ
   (torch.matmul → cuBLAS Tensor‑Core kernel).
5️⃣  Second LayerNorm over the hidden dimension (fp32 → fp16) with the same
    Triton kernel.
6️⃣  Multiply by the output‑gate, apply the final linear projection back to
    `dim` (half‑precision GEMM) and cast the result to float32.

All heavy per‑row normalisations are done in Triton, while the large dense
matrix multiplications are delegated to cuBLAS for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 input → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]        (float32)   gamma
    bias_ptr,       # const   * [dim]        (float32)   beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # load a [BLOCK_M, BLOCK_D] tile
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # cast to fp32 for the reduction

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x‑μ) * rstd * gamma + beta  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]                     # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D fp32 tensor ``x`` → fp16 tensor.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N        # total rows for the flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp16 output
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()   # [R, C]  float32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )                                            # [R, C]  float16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (four * hidden_dim outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection / gate weight matrices (each [H, C])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [4·H, dim]

    # Single GEMM: X_norm @ W.T -> [R, 4·H]
    proj_gate = F.linear(x_norm, w_proj_gate)        # fp16

    # Split the concatenated result
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]           # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                                 # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1]  (fp16) – broadcast over hidden dim
        mask_flat = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid for the gates (no extra buffers)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate    # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # Back to 4‑D tensors
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Move hidden dim to batch position -> (B, H, N, N) without copying
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched matrix multiplication (B·H, N, N) @ (B·H, N, N)^T
    # This maps to a single cuBLAS batched GEMM (Tensor‑Cores)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # [B, H, N, N]

    # Reshape back to the original 4‑D layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()   # [R, H] fp16
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                     # fp16, shape [R, H]
    out_norm = out_norm.view(B, N, N, H)  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Output gate, final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise fp16
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),    # [dim, H]
    )                                                  # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
