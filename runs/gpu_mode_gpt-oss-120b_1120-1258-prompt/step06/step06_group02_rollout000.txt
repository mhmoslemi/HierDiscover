"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

- Row‑wise LayerNorm is implemented in pure Triton (fp32 accumulation → fp16 output)
  and is applied twice (input norm + hidden‑dim norm).
- The cubic “multiplicative update” is reduced to a batched FP16 GEMM:
      out_h[b,i,j] = Σ_k left_h[b,i,k] * right_h[b,j,k]
  which is equivalent to a batched matrix multiply `left @ right.transpose(-2,-1)`.
  Using `torch.bmm` (with the hidden dimension folded into the batch) gives a
  highly‑optimised Tensor‑Core path on H100.
- All heavy linear work stays in fp16; the final result is cast back to float32
  to match the reference implementation.
- A mask (if present) is applied to the projected tensors before the gating
  non‑linearity, exactly as in the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim]   (float32)
    y_ptr,        # mutable * [total_rows, dim]   (float16)
    w_ptr,        # const   * [dim]               (float32)  γ
    b_ptr,        # const   * [dim]               (float32)  β
    total_rows,   # int (runtime)
    dim,          # int (runtime)
    eps,          # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute per‑row mean & variance in fp32,
    write the normalised result (γ·x̂+β) in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row sum and sum‑of‑squares
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : dict of model parameters (all float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]           # H
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()  # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128                                 # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (one GEMM)
    # ------------------------------------------------------------------
    #   Concatenate left/right projection + left/right gate weights = [4*H, dim]
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4H, dim]

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4H] fp16
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(x_norm,
                            weights["out_gate.weight"].to(torch.float16))   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, σ‑gates & gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask : [B,N,N] → [R,1] (float16)
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors (B, N, N, H)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Fold the hidden dimension into the batch dimension:
    #   left  → (B*H, N, N)
    #   right → (B*H, N, N) and we need its transpose on the second axis.
    left_b = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # [B*H, N, N]
    right_b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)        # [B*H, N, N]

    # Batched matrix multiplication: (i,k) · (j,k)^T  → (i,j)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))              # [B*H, N, N]

    # Restore shape (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                 # element‑wise gate, fp16

    final = F.linear(out_norm.reshape(total_rows, H),
                     weights["to_out.weight"].to(torch.float16))   # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
