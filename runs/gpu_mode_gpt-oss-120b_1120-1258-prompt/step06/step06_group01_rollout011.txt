"""
Outgoing TriMul (AlphaFold‑3) forward pass
-----------------------------------------

The implementation follows the exact PyTorch reference but moves the
expensive N³ “pairwise‑multiplicative” update into a custom Triton
kernel.  The steps are

1️⃣ Row‑wise LayerNorm over the last channel (dim) → fp16.
2️⃣ Fuse the four linear layers (left/right projections + their gates) into
   one GEMM:  [B·N·N, dim] @ [4·H, dim]ᵀ → [B·N·N, 4·H] (H = hidden_dim).
3️⃣ Apply the optional pairwise mask, sigmoid‑gates and element‑wise gating.
4️⃣ Compute the N³ update   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   with a Triton batched GEMM that works directly on the
   [B,N,N,H] layout – no extra permutes are required.
5️⃣ Second row‑wise LayerNorm over the hidden dimension H → fp16.
6️⃣ Multiply by the output‑gate, project back to *dim* with a final linear,
   and cast the result to float32.

All heavy work (the two LayerNorms and the N³ update) runs in Triton,
while the cheap dense linear layers use cuBLAS via ``torch.nn.functional.linear``.
The final tensor has shape ``[B, N, N, dim]`` and dtype ``float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (fp32 reduction → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim] (float32)
    y_ptr,        # mutable * [total_rows, dim] (float16)
    w_ptr,        # const   * [dim]           (float32)  γ
    b_ptr,        # const   * [dim]           (float32)  β
    total_rows,   # int (runtime)
    dim,          # int (runtime)
    eps,          # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- reduction ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ---------- write ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel that computes the outgoing TriMul update:
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   left/right/out have shape [B, N, N, H] (contiguous, fp16)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # const * [B, N, N, H]  (fp16)
    right_ptr,  # const * [B, N, N, H]  (fp16)
    out_ptr,    # mutable * [B, N, N, H] (fp16)

    B, N, H,                # runtime ints
    stride_b, stride_i, stride_k, stride_h,   # strides of the tensors
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,   # tile rows (i dimension)
    BLOCK_N: tl.constexpr,   # tile cols (j dimension)
    BLOCK_K: tl.constexpr,   # reduction tile (k dimension)
):
    pid_i = tl.program_id(0)          # i‑tile id
    pid_j = tl.program_id(1)          # j‑tile id
    pid_bh = tl.program_id(2)         # batch×head id (0 … B·H‑1)

    # start indices for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # derive batch and head indices
    b = pid_bh // H
    h = pid_bh - b * H

    # base offset for the (b, h) slice
    base = b * stride_b + h * stride_h

    # index vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduction over the second sequence dimension (k)
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  [BLOCK_M, BLOCK_K]   = left[b, i, k, h]
        left_ptrs = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile: we need shape [BLOCK_K, BLOCK_N] = right[b, j, k, h]ᵀ
        right_ptrs = right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)      # (M,K)·(K,N) → (M,N)

    # --------------------------------------------------------------
    # Write the result tile
    # --------------------------------------------------------------
    out_ptrs = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor          shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor          shape [B, N, N]   (bool/float)
        weights_dict : dict of torch.Tensors  (nn.Linear / LayerNorm params)
        config_dict  : dict with keys:
            - "dim"        : int
            - "hidden_dim": int
            - "nomask"    : bool (optional, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                # total “row” count for the first LN

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the last *dim* channel
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],        # γ (float32)
        weights["norm.bias"],          # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear layers (proj + gate) into one GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices → [4·H, dim] (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused projection/gate: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_proj_gate)    # fp16

    # split the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # output‑gate (stand‑alone linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid‑gates + element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()    # inplace
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gating
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (custom Triton kernel)
    # ------------------------------------------------------------------
    # reshape to the layout expected by the kernel: [B, N, N, H]
    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out   = torch.empty_like(left)                       # fp16 output

    # strides of the contiguous tensors (all identical)
    s_b, s_i, s_k, s_h = left.stride()   # (B,N,N,H) → (N*N*H, N*H, H, 1)

    # launch the batched GEMM kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles

        right,
        out,

        s_b, s_i, s_k, s_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                  # H100 likes 8‑warp kernels
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm across the hidden dimension (H)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply the output‑gate and final linear projection
    # ------------------------------------------------------------------
    # element‑wise multiplication with the (already sigmoid‑ed) out‑gate
    out_norm = out_norm * out_gate       # still [R, H] fp16

    # final linear back to *dim*
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
