"""
TriMul (outgoing) – fast forward pass on H100
================================================

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim)
and computes

    out[b,i,j,:] = Σ_k  ( (x[b,i,k,:]·W_l)·σ(x[b,i,k,:]·W_lg) ) *
                    ( (x[b,j,k,:]·W_r)·σ(x[b,j,k,:]·W_rg) )

followed by a second LayerNorm, an output‑gate and a final linear
projection back to `dim`.

Key implementation ideas
-------------------------

1️⃣ **Row‑wise LayerNorm (fp32 → fp16) in Triton** – two calls (pre‑ and
   post‑multiplicative update).  Each program processes `BLOCK_M` rows
   and `BLOCK_D` columns, computing mean/variance on‑the‑fly with
   `tl.rsqrt` (no extra synchronisation).

2️⃣ **Fused projection+gate GEMM** – concatenate the four weight
   matrices (left/right proj + their gates) into a single
   (4·H × dim) matrix and perform one half‑precision `F.linear`.
   The result is sliced into the four tensors, saving two extra GEMMs.

3️⃣ **Mask handling** – if a mask is supplied it is broadcasted to
   `[B·N·N, 1]` and multiplied only into the *proj* tensors (the gates
   stay un‑masked).  The mask and all sigmoids are applied in‑place to
   keep memory traffic low.

4️⃣ **Tri‑multiplicative update** – after reshaping to
   `[B·H, N, N]` the update becomes a batched matrix‑multiply
   `left @ rightᵀ`.  This maps directly to cuBLAS Tensor‑Core `bmm`,
   which is the fastest path on H100.

5️⃣ **Second LayerNorm + output gate + final linear** – same Triton
   kernel as step 1, then an in‑place element‑wise gate, then a final
   half‑precision linear projection.  The final result is cast back to
   `float32`.

All heavy normalisations are performed by a custom Triton kernel,
while the large GEMMs leverage cuBLAS for peak TFLOPs.  The code is
fully mixed‑precision (fp16 ⇢ fp32) and works for any supported
`dim`/`hidden_dim`/`seq_len` configuration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # const   * [total_rows, dim] (float32)
    y_ptr,                # mutable * [total_rows, dim] (float16)
    weight_ptr,           # const   * [dim]             (float32)
    bias_ptr,             # const   * [dim]             (float32)
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    eps,                  # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (x, mask, weights, config)
            x        : torch.Tensor, [B, N, N, dim], dtype float32
            mask     : torch.Tensor, [B, N, N] (bool/float) or None
            weights  : dict of all Linear/LayerNorm parameters
            config   : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape                # x.shape == [B, N, N, dim]
    R = B * N * N                        # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim]  fp32
    x_norm = torch.empty(R, dim,
                         dtype=torch.float16,
                         device=device)       # [R, dim]  fp16

    # Choose block sizes: larger blocks reduce kernel launch overhead.
    # For dim up to 384 a BLOCK_D=64 works well; BLOCK_M=1024 covers many rows.
    BLOCK_M = 1024
    BLOCK_D = 64 if dim > 128 else 32

    grid = (triton.cdiv(R, BLOCK_M),)
    _layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ   (float32)
        weights["norm.bias"],      # β   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # ------------------------------------------------------------------
    # Concatenate the four projection/gate weight matrices
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                         # [4·H, dim]

    # R×dim @ dim×4H → R×4H  (half‑precision)
    fused_out = F.linear(x_norm, w_fused)       # [R, 4·H]

    # Slice the fused result
    left_raw      = fused_out[:, 0:H]
    right_raw     = fused_out[:, H:2*H]
    left_gate_raw = fused_out[:, 2*H:3*H]
    right_gate_raw= fused_out[:, 3*H:4*H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                    # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and combine
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # In‑place sigmoid to save temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H]  fp16
    right = right_raw * right_gate  # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    # [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dim to batch axis → [B, H, N, N] → flatten to [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ  (half‑precision)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    _layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    out_norm = out_norm.view(B, N, N, H)        # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(R, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                            # [R, dim] fp16

    # Cast back to float32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
