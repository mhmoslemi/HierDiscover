"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values (γ·x̂+β) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : {
            "norm.weight":          Tensor[dim],
            "norm.bias":            Tensor[dim],
            "left_proj.weight":     Tensor[hidden_dim, dim],
            "right_proj.weight":    Tensor[hidden_dim, dim],
            "left_gate.weight":     Tensor[hidden_dim, dim],
            "right_gate.weight":    Tensor[hidden_dim, dim],
            "out_gate.weight":      Tensor[hidden_dim, dim],
            "to_out_norm.weight":   Tensor[hidden_dim],
            "to_out_norm.bias":     Tensor[hidden_dim],
            "to_out.weight":        Tensor[dim, hidden_dim],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N                     # B·N·N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Block sizes tuned for dim ≤ 384
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # x_norm_fp16 now has shape (total_rows, dim) in FP16
    # --------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections & gates into a single GEMM
    # --------------------------------------------------------------
    # Build concatenated weight matrix of shape (5·hidden_dim, dim)
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    weight_concat = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)  # (5H, D)

    # Single GEMM: (rows, dim) @ (5H, dim).T → (rows, 5H)
    all_out = F.linear(
        x_norm_fp16,    # (rows, dim) in FP16
        weight_concat,  # (5H, dim) in FP16
    )  # -> (rows, 5H) FP16

    # --------------------------------------------------------------
    # Split fused output and apply sigmoids
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj   = all_out[:, 0:H]                 # (rows, H)
    left_gate_pre = all_out[:, H:2 * H]           # (rows, H)
    right_proj  = all_out[:, 2 * H:3 * H]         # (rows, H)
    right_gate_pre = all_out[:, 3 * H:4 * H]      # (rows, H)
    out_gate_pre = all_out[:, 4 * H:5 * H]        # (rows, H)

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → (rows, 1)
        mask_flat = mask.view(total_rows, 1).to(
            dtype=torch.float16, device=device, non_blocking=True
        )
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply gates
    # --------------------------------------------------------------
    left_h = left_proj * left_gate      # (rows, H)
    right_h = right_proj * right_gate   # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Merge H into batch dimension → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, shape [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16, shape [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # --------------------------------------------------------------
    rows_h = total_rows
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)        # FP16
    gated = out_norm * out_gate                 # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device, non_blocking=True),
    )   # (rows_h, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)  # cast back to FP32
    return final
```
