"""
Highly‑optimized forward pass of the **outgoing TriMul** operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32→FP16) on the flattened input (custom Triton kernel).
2️⃣ Fuse the four linear layers (left/right projections + their two gates) *and*
   the output‑gate linear layer into a single FP16 GEMM.
   *Result*: 5·H intermediate tensors (left, right, left_gate, right_gate,
   out_gate) in FP16.
3️⃣ Optional mask on `left/right`, in‑place sigmoid on the three gates and
   element‑wise gating.
4️⃣ Reshape to `[B, H, N, N]` and launch a **batched GEMM kernel** (Triton)
   that computes for every head `h`:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   This replaces `torch.bmm` and removes per‑matrix launch overhead.
5️⃣ Second LayerNorm (row‑wise, FP16) on the hidden dimension (custom kernel).
6️⃣ Multiply by the output‑gate, final FP16 linear projection, and cast to FP32.

All heavy reductions (means/variances) are handled by custom Triton kernels,
the large matrix‑multiplications use Tensor‑Core friendly `tl.dot`.  The
implementation works for any `B∈{1,2}`, `N∈{128,256,512,1024}`,
`dim∈{128,384}` and `hidden_dim∈{128,384,768}`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm kernel (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32) γ
    bias_ptr,         # const   * [dim]               (float32) β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1️⃣ mean / var -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var+eps)

    # ------------------- 2️⃣ normalize & apply γ,β -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Batched GEMM kernel for the N³ multiplicative update
#   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # all equal to N in our use‑case
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index
    pid_z = tl.program_id(2)          # batch*head index

    # --- tile coordinates -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --- pointer base for this batch/head ---------------------------------
    a_base = a_ptr + pid_z * stride_a_batch
    b_base = b_ptr + pid_z * stride_b_batch
    c_base = c_ptr + pid_z * stride_c_batch

    # --- create pointers to the K‑dimension blocks -------------------------
    a_ptrs = a_base + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    b_ptrs = b_base + offs_n[:, None] * stride_b_row + tl.arange(0, BLOCK_K)[None, :] * stride_b_col

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = tl.arange(0, BLOCK_K)
        mask_k = cur_k < (K - k * BLOCK_K)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)          # Tensor‑Core dot → fp32 accumulator

        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_col

    # --- write back --------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry‑point: forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask applied
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],       # γ   (float32)
        weights["norm.bias"],         # β   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections (left, right, left‑gate,
    #    right‑gate, out‑gate) into one GEMM.
    # ------------------------------------------------------------------
    # [5·H, dim] weight matrix, FP16
    w_5h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    fused = F.linear(x_norm_fp16, w_5h)      # [R, 5·H]  FP16

    left_raw      = fused[:, 0:H]
    right_raw     = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw= fused[:, 3*H:4*H]
    out_gate_raw  = fused[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] FP16
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid (saves a temporary)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension next to batch → [B, H, N, N] (contiguous)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head → [B·H, N, N] (contiguous)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM via Triton (Tensor‑Core dot)
    # ------------------------------------------------------------------
    out_mat = torch.empty_like(left_mat)                # FP16, same shape

    # strides for the three tensors (row‑major)
    stride_a_batch = left_mat.stride(0)   # N * N
    stride_a_row   = left_mat.stride(1)   # N
    stride_a_col   = left_mat.stride(2)   # 1

    stride_b_batch = right_mat.stride(0)
    stride_b_row   = right_mat.stride(1)
    stride_b_col   = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_gemm = (
        triton.cdiv(N, BLOCK_M),   # tiles along rows
        triton.cdiv(N, BLOCK_N),   # tiles along cols
        B * H,                     # one program per (batch,head)
    )
    batched_gemm_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                 # M, N, K (all = seq_len)
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (row‑wise over hidden dim) → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)      # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                   # element‑wise FP16

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
