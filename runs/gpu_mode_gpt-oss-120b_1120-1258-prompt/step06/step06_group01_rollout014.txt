@triton.jit
def fused_outgoing_kernel(
    out_ptr,           # mutable [total_rows, dim] fp16
    out_gate_ptr,      # const [total_rows, H] fp16
    out_input_ptr,     # const [total_rows, H] fp16
    ln_weight_ptr,     # const [H] fp32
    ln_bias_ptr,       # const [H] fp32
    final_weight_ptr,  # const [dim, H] fp16
    total_rows,        # int
    hidden_dim,        # int
    dim,               # int
    eps,               # float
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    NUM_N = tl.cdiv(dim, BLOCK_D)
    pid_m = pid // NUM_N
    pid_n = pid % NUM_N

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_D

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_n = col_start + tl.arange(0, BLOCK_D)
    mask_n = offs_n < dim

    # Compute mean & var per row across hidden_dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        x_ptrs = out_input_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulate output
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # load values
        x_ptrs = out_input_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        g_ptrs = out_gate_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        g = tl.load(g_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        gamma = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)  # fp32
        beta  = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)  # fp32

        # layernorm + gating
        norm = (x - mean[:, None]) * rstd[:, None]         # [M, K]
        norm = norm * gamma[None, :] + beta[None, :]       # [M, K]
        a_tile = norm * g                                  # [M, K]

        # weight tile
        w_ptrs = final_weight_ptr + (offs_k[:, None] + offs_n[None, :] * hidden_dim)
        w = tl.load(w_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a_tile, w)   # [M, D]
    # store
    out_ptrs = out_ptr + (offs_m[:, None] * dim + offs_n[None, :])
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out,

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor – custom Triton kernel
   (FP32 accumulation → FP16 output).

2️⃣ Fuse the four linear layers (left/right projections + gates) into a
   single GEMM:     [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]  (R = B·N·N).

3️⃣ Apply optional mask, compute sigmoid gates and obtain the gated
   tensors *left* and *right*.

4️⃣ Batched GEMM (cuBLAS) for the N³ multiplicative update:
   out = left @ rightᵀ  → shape [B·H, N, N] (half‑precision).

5️⃣ Fuse the second LayerNorm, the output‑gate multiplication and the
   final linear projection into **one** Triton kernel:
       out̂ = ((out − μ) / √(σ²+ε) ⋅ γ + β) ⊙ out_gate  @  Wᵀ
   where γ,β are the parameters of `to_out_norm` and `W` is
   `to_out.weight`.  The kernel works on a 2‑D grid (rows × output‑dim)
   and performs the reduction over the hidden dimension H inside the
   kernel.  This removes two large intermediate tensors and cuts the
   memory traffic of the final stage dramatically.

All heavy matrix multiplications are delegated to cuBLAS (torch.bmm /
torch.nn.functional.linear), which is optimal on an H100.  The only
operations written in Triton are the two LayerNorm passes and the final
fused LN‑gate‑linear kernel.

The function `custom_kernel` follows the required signature and returns
a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]  (float32)
    y_ptr,          # mutable * [total_rows, dim]  (float16)
    w_ptr,          # const   * [dim]              (float32)  γ
    b_ptr,          # const   * [dim]              (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- normalize and write --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused second LN + out‑gate + final linear
# ----------------------------------------------------------------------
@triton.jit
def fused_out_norm_gate_linear_kernel(
    out_ptr,           # mutable * [total_rows, dim] (fp16)
    out_gate_ptr,      # const   * [total_rows, H]   (fp16)
    out_input_ptr,     # const   * [total_rows, H]   (fp16)
    ln_w_ptr,          # const   * [H]               (fp32)  γ (to_out_norm)
    ln_b_ptr,          # const   * [H]               (fp32)  β (to_out_norm)
    final_w_ptr,       # const   * [dim, H]          (fp16)  to_out.weight
    total_rows,        # int
    hidden_dim,        # int
    dim,               # int
    eps,               # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Computes for each row r (0 ≤ r < total_rows) and each output
    column d (0 ≤ d < dim):

        out[r, d] = Σ_{h} ( ( (x[r, h] - μ_r) / √(σ_r²+ε) * γ[h] + β[h] )
                          * gate[r, h] ) * W[d, h]

    where x is out_input_ptr, gate is out_gate_ptr,
    (μ_r, σ_r) are the per‑row mean/variance over the hidden dimension.
    """
    pid = tl.program_id(0)

    # 2‑D grid is flattened: pid = pid_m * N_blocks + pid_n
    N_blocks = tl.cdiv(dim, BLOCK_D)
    pid_m = pid // N_blocks            # row‑tile id
    pid_n = pid % N_blocks             # col‑tile id

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_D

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_n = col_start + tl.arange(0, BLOCK_D)
    mask_n = offs_n < dim

    # --------------------------------------------------------------
    # 1️⃣ Compute mean / variance per row over the hidden dimension H
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_k = tl.cdiv(hidden_dim, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        x_ptrs = out_input_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1 / √(var+ε)

    # --------------------------------------------------------------
    # 2️⃣ Accumulate the final output tile
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # ----- load x and gate -----
        x_ptrs = out_input_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        g_ptrs = out_gate_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :])
        g = tl.load(g_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- layer‑norm parameters -----
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # fp32

        # ----- normalize, apply LN‑bias, then gate -----
        norm = (x - mean[:, None]) * rstd[:, None]          # (x-μ)/σ
        norm = norm * gamma[None, :] + beta[None, :]       # γ·norm + β
        a_tile = norm * g                                   # ⊙ out_gate

        # ----- load final linear weight tile (K×D) -----
        w_ptrs = final_w_ptr + (offs_k[:, None] + offs_n[None, :] * hidden_dim)
        w = tl.load(w_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)               # fp32

        # ----- GEMM over the hidden dimension -----
        acc += tl.dot(a_tile, w)                            # [M, D]

    # --------------------------------------------------------------
    # 3️⃣ Store result (fp16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + (offs_m[:, None] * dim + offs_n[None, :])
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : Tuple[input_tensor, mask_tensor, weights_dict, config_dict]
        input_tensor : torch.Tensor   shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor   shape [B, N, N]     (bool/float)
        weights_dict : dict of torch.Tensors with all Linear/LayerNorm params
        config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LayerNorm

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()         # [R, dim] fp32
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices (no bias) → [4·H, dim] (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (fp16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)          # fp16

    left_proj_raw   = proj_gate[:, 0:H]          # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]        # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]      # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]      # [R, H]

    # out‑gate (separate weight)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )  # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # bring head dimension into batch axis for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)   # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, output‑gate & final linear projection
    # --------------------------------------------------------------
    # flatten to [R, H] (guaranteed contiguous after the view)
    out_flat = out_mat.view(R, H).contiguous()            # [R, H] fp16
    out_gate_flat = out_gate.view(R, H).contiguous()     # [R, H] fp16

    # allocate output buffer
    final_half = torch.empty(R, dim,
                             dtype=torch.float16,
                             device=device)

    # launch fused kernel
    BLOCK_K = 32                     # hidden‑dim tile
    BLOCK_D = 64                     # output‑dim tile
    grid = (triton.cdiv(R, BLOCK_M) *
            triton.cdiv(dim, BLOCK_D),)   # 2‑D grid flattened

    fused_out_norm_gate_linear_kernel[grid](
        final_half,                     # out_ptr
        out_gate_flat,                  # out_gate_ptr
        out_flat,                       # out_input_ptr (pre‑LN)
        weights["to_out_norm.weight"],  # LN γ (float32)
        weights["to_out_norm.bias"],    # LN β (float32)
        weights["to_out.weight"].to(torch.float16),  # final linear weight
        R,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # reshape back to [B, N, N, dim] and cast to float32
    out = final_half.view(B, N, N, dim).to(torch.float32)

    return out
