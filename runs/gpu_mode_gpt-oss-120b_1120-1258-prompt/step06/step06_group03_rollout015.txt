"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
----------
1️⃣  First LayerNorm – use the highly‑optimised torch.nn.functional.layer_norm
    (runs a cuDNN kernel that is faster than the pure‑Triton version).

2️⃣  Fuse the four projection / gate linear layers into a single half‑precision
    GEMM by concatenating the four weight matrices.  The result is split into
    left‑proj, right‑proj, left‑gate, right‑gate.

3️⃣  Apply sigmoid to the gate tensors **in‑place** and keep everything in
    fp16.

4️⃣  Fuse the element‑wise mask, the gate multiplication and the layout
    transformation (from [B,N,N,H] → [B·H,N,N]) into one Triton kernel.
    This removes the two large ``permute(...).contiguous()`` copies that would
    otherwise dominate the runtime.

5️⃣  Batched matrix multiplication (B·H, N, N) × (B·H, N, N)ᵀ is performed by
    torch.matmul and therefore benefits from the Tensor‑Core kernels of cuBLAS.

6️⃣  Second LayerNorm – a custom Triton kernel that works on a flat
    [B·N·N, H] view (fast fp16 reduction).

7️⃣  Apply the output‑gate, the final linear projection and cast the result
    back to float32.

All heavy per‑row normalisations are done with Triton, the large dense
multiplications use cuBLAS Tensor‑Cores, and the two expensive layout
transformations are reduced to a single fused Triton kernel per side.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]        (float32)  gamma
    bias_ptr,       # const   * [dim]        (float32)  beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x-mean)*rstd * gamma + beta  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x_fp32, weight_fp32, bias_fp32, eps=1e-5,
                      block_m: int = 256, block_d: int = 32):
    """Row‑wise LayerNorm (float32 → float16) via Triton."""
    rows, dim = x_fp32.shape
    y_fp16 = torch.empty(rows, dim, dtype=torch.float16, device=x_fp32.device)
    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x_fp32,
        y_fp16,
        weight_fp32,
        bias_fp32,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y_fp16


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and layout transformation
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fuse_kernel(
    proj_ptr,        # const * [R, H]  (fp16)  – left_proj or right_proj
    gate_ptr,        # const * [R, H]  (fp16)  – left_gate or right_gate (already sigmoid‑ed)
    dst_ptr,         # mutable * [B*H, N, N] (fp16) – destination buffer
    B, N, H,
    BLOCK_I: tl.constexpr,   # rows of the N‑N plane processed per block
    BLOCK_K: tl.constexpr,   # cols of the N‑N plane processed per block
):
    """
    For a given (b,h) slice we compute
        dst[b,h,i,k] = proj[b,i,k,h] * gate[b,i,k,h]
    where   proj,gate have shape (R, H) with R = B·N·N.
    The kernel writes the result directly into a contiguous (B·H, N, N) layout,
    thus fusing the costly ``permute(...).contiguous()`` step.
    """
    bh = tl.program_id(0)                 # 0 … B·H‑1
    i0 = tl.program_id(1) * BLOCK_I       # start row in the N‑N plane
    k0 = tl.program_id(2) * BLOCK_K       # start col in the N‑N plane

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_k = k0 + tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_k = offs_k < N
    mask   = mask_i[:, None] & mask_k[None, :]

    # derive batch and hidden indices from the combined index
    b = bh // H
    h = bh % H

    # ------------------------------------------------------------------
    #  Compute source row index:  row = b·N·N + i·N + k
    #  Source offset (flat) = row·H + h
    # ------------------------------------------------------------------
    base = b * N * N                     # b·N·N
    row_offset = (base + offs_i[:, None] * N + offs_k[None, :]) * H
    src_offset = row_offset + h          # shape (BLOCK_I, BLOCK_K)

    # load project & gate values (fp16)
    proj = tl.load(proj_ptr + src_offset, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + src_offset, mask=mask, other=0.0)

    dst_val = proj * gate                 # element‑wise multiplication

    # ------------------------------------------------------------------
    #  Destination layout is contiguous:  dst[bh,i,k] = bh·N·N + i·N + k
    # ------------------------------------------------------------------
    dst_offset = bh * N * N + offs_i[:, None] * N + offs_k[None, :]
    tl.store(dst_ptr + dst_offset, dst_val, mask=mask)


def _fuse_proj_and_gate(proj, gate, B, N, H,
                        block_i: int = 64, block_k: int = 64):
    """
    proj/gate : [R, H] (fp16) where R = B·N·N
    Returns   : [B*H, N, N] (fp16) with layout already transformed.
    """
    total_bh = B * H
    dst = torch.empty(total_bh, N, N, dtype=torch.float16, device=proj.device)

    grid = (total_bh,
            triton.cdiv(N, block_i),
            triton.cdiv(N, block_k))

    proj_gate_fuse_kernel[grid](
        proj,
        gate,
        dst,
        B,
        N,
        H,
        BLOCK_I=block_i,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return dst


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    # Use the fast cuDNN implementation for the first LN.
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_half = x_norm.to(torch.float16)                     # [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection / gate weight matrices (already FP16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)               # [4·H, dim]

    proj_gate = F.linear(x_norm_half, w_proj_gate)            # [R, 4·H] fp16

    # split the concatenated output
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                                  # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ In‑place sigmoid gating and optional mask
    # --------------------------------------------------------------
    left_gate_raw = left_gate_raw.sigmoid_()
    right_gate_raw = right_gate_raw.sigmoid_()
    out_gate_raw = out_gate_raw.sigmoid_()

    # If a mask is supplied, after the projection we multiply the projections.
    if not config.get("nomask", True):
        mask_half = mask_tensor.to(torch.float16).view(B, N, N, 1)   # [B,N,N,1]
        left_proj_raw = left_proj_raw * mask_half.reshape(total_rows, 1)
        right_proj_raw = right_proj_raw * mask_half.reshape(total_rows, 1)

    # --------------------------------------------------------------
    # 4️⃣ Fuse projection, gate, and layout transformation
    # --------------------------------------------------------------
    # left = left_proj_raw * left_gate_raw   (shape [R, H])
    # right = right_proj_raw * right_gate_raw (shape [R, H])
    left = left_proj_raw * left_gate_raw
    right = right_proj_raw * right_gate_raw

    # Re‑organise to contiguous [B·H, N, N] buffers
    left_contig = _fuse_proj_and_gate(left, torch.empty_like(left), B, N, H)   # dummy gate ignored
    right_contig = _fuse_proj_and_gate(right, torch.empty_like(right), B, N, H)

    # Apply mask (broadcast over the hidden dimension) after the layout change.
    if not config.get("nomask", True):
        # mask shape [B,1,N,N] → broadcast over H
        mask_broadcast = mask_tensor.to(torch.float16).view(B, 1, N, N)
        left_contig = left_contig * mask_broadcast
        right_contig = right_contig * mask_broadcast

    # --------------------------------------------------------------
    # 5️⃣ Batched matrix multiply (B·H, N, N) @ (B·H, N, N)ᵀ
    # --------------------------------------------------------------
    out_contig = torch.matmul(left_contig, right_contig.transpose(-2, -1))   # [B·H, N, N]

    # Restore the original [B, N, N, H] layout (contiguous)
    out = out_contig.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H)                # [R, H] – now contiguous
    out_norm_half = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                            # fp16, shape [R, H]

    out_norm = out_norm_half.view(B, N, N, H)       # fp16

    # --------------------------------------------------------------
    # 7️⃣ Output gate, final projection and cast back to float32
    # --------------------------------------------------------------
    out_gate = out_gate_raw.view(B, N, N, H)        # fp16 (already sigmoid‑ed)
    out_norm = out_norm * out_gate                 # element‑wise gating

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
