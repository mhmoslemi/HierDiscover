"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated.

The kernel pipeline
-------------------
1️⃣ Row‑wise LayerNorm on the input tensor  (FP32 → FP16).
2️⃣ Fuse projection + gate for the left / right branches:
   (dim → hidden) + (dim → hidden) → (dim → 2·hidden) via one `F.linear`.
   Apply sigmoid gates and an optional mask.
3️⃣ Batched GEMM on the hidden dimension:
   out_h(b,i,j) = Σ_k left_h(b,i,k)·right_h(b,j,k)
   – realised by a custom Triton kernel that avoids the costly
   `permute/contiguous` shuffles required by `torch.bmm`.
4️⃣ Second LayerNorm (over the hidden dimension) – Triton kernel.
5️⃣ Fuse output‑gate multiplication and the final linear projection
   into a single Triton kernel:
   y = (norm_out ⊙ out_gate) @ W_out.T
6️⃣ Cast the result back to FP32.

All heavy arithmetic is performed in FP16, only the tiny reductions
(mean/var) use FP32.  The implementation keeps the memory layout
`[B, N, N, H]` (H – hidden) where **H is the innermost dimension**,
so the tensors are already suitable for the custom GEMM kernel
without any extra `contiguous` calls.

The code below works for the test matrix‑sizes listed in the task
statement and targets ~1 ms wall‑time on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [rows, dim]   (input, FP32)
    y_ptr,          # half*    [rows, dim]   (output, FP16)
    gamma_ptr,      # float*   [dim]         (scale)
    beta_ptr,       # float*   [dim]         (bias)
    rows,           # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- Pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x_tile, axis=1)
        sum_sq  += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_tile - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


@triton.jit
def trimul_gemm_kernel(
    # pointers
    a_ptr,          # left tensor  (B,N,N,H)   fp16
    b_ptr,          # right tensor (B,N,N,H)   fp16
    c_ptr,          # out tensor   (B,N,N,H)   fp16
    # geometry
    B, N, H,
    # strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel is a batched GEMM where the batch dimension is
    B * H, M = N (rows), N = N (cols), K = N (reduction).
    """
    pid_i = tl.program_id(0)          # tile over M (i dimension)
    pid_j = tl.program_id(1)          # tile over N (j dimension)
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # decode batch & hidden from the flattened index
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # offsets for the current tile
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the specific (b,h) slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: shape (BLOCK_M, BLOCK_K)   (i, k)
        a_ptrs = a_base \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a_tile = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B^T: shape (BLOCK_K, BLOCK_N)   (k, j)
        b_ptrs = b_base \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_j
        b_tile = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a_tile, b_tile)   # fp16->fp32 accumulation

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = c_base \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


@triton.jit
def final_proj_kernel(
    # inputs
    norm_ptr,       # (rows, H)   fp16   (post‑norm)
    gate_ptr,       # (rows, H)   fp16   (output gate)
    weight_ptr,     # (D, H)      fp16   (to_out weight)
    # output
    out_ptr,        # (rows, D)   fp16
    # geometry
    rows, hidden, out_dim,
    # strides (all in element units)
    stride_norm_rows, stride_norm_h,
    stride_gate_rows, stride_gate_h,
    stride_w_out_dim, stride_w_h,
    stride_out_rows, stride_out_dim,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute   out = (norm ⊙ gate) @ weight.T
    where   norm,gate  : (rows, hidden)
            weight     : (out_dim, hidden)
    The kernel fuses the element‑wise gate multiplication into the
    inner product to save one memory pass.
    """
    pid_m = tl.program_id(0)   # tile over rows
    pid_n = tl.program_id(1)   # tile over out_dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < rows
    mask_n = offs_n < out_dim

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # (rows, hidden) → (BLOCK_M, BLOCK_K)
        n_ptr = norm_ptr + offs_m[:, None] * stride_norm_rows + offs_k[None, :] * stride_norm_h
        g_ptr = gate_ptr + offs_m[:, None] * stride_gate_rows + offs_k[None, :] * stride_gate_h

        n_tile = tl.load(n_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        g_tile = tl.load(g_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        prod = n_tile * g_tile             # shape (BLOCK_M, BLOCK_K)

        # weight_T : (hidden, out_dim)
        w_ptr = weight_ptr + offs_k[:, None] * stride_w_h + offs_n[None, :] * stride_w_out_dim
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(prod, w_tile)        # FP16 → FP32 accumulation

    out_tile_ptr = out_ptr + offs_m[:, None] * stride_out_rows + offs_n[None, :] * stride_out_dim
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : {
            "norm.weight":          Tensor[dim],
            "norm.bias":            Tensor[dim],
            "left_proj.weight":     Tensor[hidden, dim],
            "right_proj.weight":    Tensor[hidden, dim],
            "left_gate.weight":     Tensor[hidden, dim],
            "right_gate.weight":    Tensor[hidden, dim],
            "out_gate.weight":      Tensor[hidden, dim],
            "to_out_norm.weight":   Tensor[hidden],
            "to_out_norm.bias":     Tensor[hidden],
            "to_out.weight":        Tensor[dim, hidden],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # block sizes: 128 rows and 64 columns give good occupancy for dim ≤ 384
    BLOCK_M = 128
    BLOCK_D = 64
    grid_norm = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # back to 4‑D view (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣  Projection + gate (fused linear)
    # --------------------------------------------------------------
    # concatenate proj + gate weights for a single matmul per side
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    rows = total_rows
    # (rows, 2*H) – one matmul per side
    left_concat = F.linear(x_norm.view(rows, dim), left_w)   # FP16
    right_concat = F.linear(x_norm.view(rows, dim), right_w)

    # split proj / pre‑gate
    left_proj = left_concat[:, :H]          # (rows, H)
    left_gate_pre = left_concat[:, H:]      # (rows, H)
    right_proj = right_concat[:, :H]
    right_gate_pre = right_concat[:, H:]

    # output gate (shared across positions)
    out_gate = torch.sigmoid(
        F.linear(
            x_norm.view(rows, dim),
            weights["out_gate.weight"].to(dtype=torch.float16, device=device)
        )
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)
    right_h = right_proj * right_gate       # (rows, H)

    # reshape to [B, N, N, H] (H is innermost → stride=1)
    left_h = left_h.view(B, N, N, H)
    right_h = right_h.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣  Tri‑multiplicative update – custom batched GEMM
    # --------------------------------------------------------------
    out = torch.empty_like(left_h)   # (B, N, N, H)

    # strides (in element counts) for the four‑dim tensors
    s_a_batch, s_a_i, s_a_k, s_a_h = left_h.stride()
    s_b_batch, s_b_j, s_b_k, s_b_h = right_h.stride()
    s_c_batch, s_c_i, s_c_j, s_c_h = out.stride()

    # tile sizes – each GEMM works on a (BLOCK_M × BLOCK_N) tile of the
    # (N × N) output matrix; BLOCK_K is the inner‑reduction tile.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_gemm = (
        triton.cdiv(N, BLOCK_M),    # i‑tiles
        triton.cdiv(N, BLOCK_N),    # j‑tiles
        B * H,                      # combined batch‑hidden dimension
    )
    trimul_gemm_kernel[grid_gemm](
        left_h,
        right_h,
        out,
        B, N, H,
        s_a_batch, s_a_i, s_a_k, s_a_h,
        s_b_batch, s_b_j, s_b_k, s_b_h,
        s_c_batch, s_c_i, s_c_j, s_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    rows_h = total_rows           # B·N·N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # reuse the same block configuration (rows are many, dim=H≤128)
    grid_norm2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,          # reuse BLOCK_D from the first LN
    )
    out_norm = out_norm_fp16.view(B, N, N, H)  # FP16, same layout as `out`

    # --------------------------------------------------------------
    # 7️⃣  Fuse output‑gate multiplication & final projection
    # --------------------------------------------------------------
    # flatten tensors for the kernel: (rows_h, H) and (rows_h, D)
    out_norm_flat = out_norm.view(rows_h, H)
    out_gate_flat = out_gate.view(rows_h, H)
    w_out = weights["to_out.weight"].to(dtype=torch.float16, device=device)

    # output tensor (rows_h, dim) in FP16
    final_fp16 = torch.empty((rows_h, dim), dtype=torch.float16, device=device)

    # Strides for the flattened tensors
    s_norm_rows, s_norm_h = out_norm_flat.stride()   # (dim, 1)  → (dim, 1)
    s_gate_rows, s_gate_h = out_gate_flat.stride()
    s_w_out_dim, s_w_h = w_out.stride()             # (H, 1) when weight shape (dim, H)

    # Output stride
    s_out_rows, s_out_dim = final_fp16.stride()     # (dim, 1)

    # Tile sizes for the final matmul (hidden → dim)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_final = (
        triton.cdiv(rows_h, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )
    final_proj_kernel[grid_final](
        out_norm_flat,
        out_gate_flat,
        w_out,
        final_fp16,
        rows_h,
        H,
        dim,
        s_norm_rows, s_norm_h,
        s_gate_rows, s_gate_h,
        s_w_out_dim, s_w_h,
        s_out_rows, s_out_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to 4‑D and cast to FP32 (the reference module’s dtype)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
