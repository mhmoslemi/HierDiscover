"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated.

The implementation follows the reference PyTorch module but fuses the two
LayerNorms (the first on the input and the second on the hidden‑dim output)
into custom Triton kernels.  All heavy matrix multiplications are performed
with half‑precision Tensor‑Core GEMMs via `torch.nn.functional.linear` and
`torch.bmm`, which on an H100 are already optimal.  The kernels work for
any batch size (1‑2), sequence length up to 1024, and `dim` ∈ {128,384}
with `hidden_dim` = 128.  Masking and gating are applied exactly as in the
reference code.

Key steps
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – `layernorm_kernel`.
2️⃣ One fused linear projection that simultaneously computes
   left/right projections and their gates.
3️⃣ Separate linear projection for the output gate.
4️⃣ Optional mask multiplication, sigmoid on gates and element‑wise gating.
5️⃣ Batched GEMM (N³ update): (B·H, N, N) @ (B·H, N, N)ᵀ.
6️⃣ Fused hidden‑dim LayerNorm + output‑gate multiplication –
   `layernorm_gate_kernel`.
7️⃣ Final linear projection back to `dim` (FP16 → FP32).

Both kernels are written with a two‑pass algorithm (mean/variance → normalize)
and support arbitrary `dim` / `hidden_dim`.  They are launched with a
reasonable block size (256 rows × 64 columns) and 4 warps per program,
which gives high occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)   γ
    bias_ptr,            # const   * [dim]               (float32)   β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
      y = (x - μ) / sqrt(σ²+eps) * γ + β
    Input x is fp32, output y is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- 1️⃣ compute mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer arithmetic: row * dim + col
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---- 2️⃣ write normalized values (fp16) -------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]             # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + gate multiplication (fp16 inputs)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,            # input / output, [total_rows, dim] (fp16)
    weight_ptr, bias_ptr,    # LN γ,β (float32)
    gate_ptr,                # gate tensor, same shape as x (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Computes  y = ( (x-μ)/σ * γ + β ) * gate
    All arithmetic in fp32, final result stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean / variance --------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply LN params, then gate -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr_x  = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptr_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        # gate values are fp16; promote to fp32 for multiplication
        ptr_gate = gate_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        gate = tl.load(ptr_gate,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]   # LN (fp32)
        out = out * gate                       # gating (fp32)

        out_fp16 = out.to(tl.float16)
        ptr_y = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(ptr_y, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the *outgoing* TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : dict containing all linear / layer‑norm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of (b,i,j) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(R, dim).contiguous()          # fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ   (float32)
        weights["norm.bias"],            # β   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single matmul)
    # ------------------------------------------------------------------
    # Cast linear weights once to fp16 for the fused projection
    w_left = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right = weights["right_proj.weight"].to(torch.float16) # [H, dim]
    w_lgate = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate = weights["right_gate.weight"].to(torch.float16) # [H, dim]

    # Concatenate the four weight matrices (rows stacked)
    # shape => [4*H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # [R, dim] @ [4*H, dim].T -> [R, 4*H]   (all fp16)
    proj_gate = F.linear(x_norm, w_concat)   # fp16

    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]               # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]             # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate projection (separate linear)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] -> [R,1] as fp16; broadcast on hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid for speed
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Element‑wise gating
    left = left_proj_raw * left_gate    # [R, H] fp16
    right = right_proj_raw * right_gate # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dim into the batch axis so we can use torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H)                                          # [R, H]

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + multiplication by the output‑gate
    # ------------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.view(R, H),             # gate (fp16)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16

    # Cast back to fp32 for the model output
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
