"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) with a custom
   Triton kernel (no extra synchronisation, uses two passes to compute
   mean/variance and then write the normalised values).

2️⃣ Fuse the two linear projections of each branch (proj + gate) by
   concatenating the weights → a single `torch.nn.functional.linear`
   call (FP16).  Apply the sigmoid to the gate part and an optional mask
   (broadcast over the hidden dimension).  The result of each branch
   is the hidden‑dim tensor `h = proj * sigmoid(gate)`.

3️⃣ Compute the tri‑multiplicative update directly with `torch.einsum`:
   `out_h[b,i,j,d] = Σ_k left_h[b,i,k,d] * right_h[b,j,k,d]`.
   This avoids the costly `permute → contiguous → view` that the
   reference implementation performs.

4️⃣ Second LayerNorm (over `hidden_dim`) using the same Triton kernel,
   again in FP16.

5️⃣ Multiply by the output‑gate (already in FP16) and apply the final
   linear projection `(hidden → dim)` with `torch.nn.functional.linear`.
   The result is cast back to float32.

All heavy arithmetic is performed in FP16; only the final tensor is
returned in float32.  The code works for any of the shapes listed in the
test specification and supports both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d_iter in range(0, num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values ----------
    for d_iter in range(0, num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]               # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
    mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
    weights_dict : {
        "norm.weight":          Tensor[dim] (FP32),
        "norm.bias":            Tensor[dim],
        "left_proj.weight":     Tensor[hidden_dim, dim],
        "right_proj.weight":    Tensor[hidden_dim, dim],
        "left_gate.weight":     Tensor[hidden_dim, dim],
        "right_gate.weight":    Tensor[hidden_dim, dim],
        "out_gate.weight":      Tensor[hidden_dim, dim],
        "to_out_norm.weight":   Tensor[hidden_dim],
        "to_out_norm.bias":     Tensor[hidden_dim],
        "to_out.weight":        Tensor[dim, hidden_dim],
    }
    config_dict  : {"dim": int,
                    "hidden_dim": int,
                    "nomask": bool (optional, default True)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    # flatten first three dimensions
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tune block sizes – dim ≤ 384 → at most 3 iterations over D
    BLOCK_M = 128          # rows per program
    BLOCK_D = 128          # cols per program
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm = x_norm_fp16.view(B, N, N, dim)            # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse (proj + gate) for left/right branches
    # ------------------------------------------------------------------
    # concatenate proj & gate weights → (2·H, dim)
    left_w  = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    # flatten for Linear
    x_norm_flat = x_norm.view(total_rows, dim)                 # [rows, dim]

    # Linear → (rows, 2·H) for each branch
    left_concat  = F.linear(x_norm_flat, left_w)               # FP16
    right_concat = F.linear(x_norm_flat, right_w)              # FP16

    # split into projection and pre‑gate tensors
    H = hidden_dim
    left_proj   = left_concat[:, :H]               # (rows, H)
    left_gate_p = left_concat[:, H:]               # (rows, H)
    right_proj  = right_concat[:, :H]
    right_gate_p = right_concat[:, H:]

    # output‑gate (shared)
    out_gate_pre = F.linear(
        x_norm_flat,
        weights["out_gate.weight"]
        .to(dtype=torch.float16, device=device, non_blocking=True)
    )
    out_gate = torch.sigmoid(out_gate_pre)        # (rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(
            dtype=torch.float16, device=device, non_blocking=True
        )
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)

    left_h  = left_proj  * left_gate           # (rows, H)
    right_h = right_proj * right_gate          # (rows, H)

    # reshape back to 4‑D tensors
    left_h  = left_h.view(B, N, N, H)          # [B,N,N,H]
    right_h = right_h.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via einsum (no materialising perms)
    # ------------------------------------------------------------------
    # out_h[b,i,j,d] = Σ_k left_h[b,i,k,d] * right_h[b,j,k,d]
    out_h = torch.einsum('b i k d, b j k d -> b i j d', left_h, right_h)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_h_flat = out_h.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_h_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_h_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)          # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Gate & final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)               # FP16
    gated = out_norm * out_gate                        # FP16

    gated_flat = gated.view(rows_h, H)
    final_fp16 = F.linear(
        gated_flat,
        weights["to_out.weight"]
        .to(dtype=torch.float16, device=device, non_blocking=True)
    )                                                  # (rows_h, dim) FP16

    # Cast back to FP32 as required by the reference module
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
```
